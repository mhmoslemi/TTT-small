def project_to_feasible(h, n):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    # compute sum, diff
    sum_h = h.sum()
    diff = sum_h - target
    if abs(diff) < 1e-12:
        return h
    # identify indices that can move:
    # If diff > 0, need to decrease sum: decreasing h values >0
    # else need to increase sum: increasing h values <1
    # We can distribute diff uniformly across those indices.
    # iterative: while diff !=0 and there exist free indices.
    # We'll do simple loop.
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # Update
        if diff > 0:
            # subtract
            h[free] -= delta
        else:
            h[free] -= delta  # diff negative; subtract negative = add
        # Clip again to maintain bounds
        h = np.clip(h, 0.0, 1.0)
        sum_h = h.sum()
        diff = sum_h - target
    # final small adjustment sequentially
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # iterate over indices
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy import optimize

# ---------- Utility functions (top level) ----------

def project_to_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # iterative water‑filling
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        if diff > 0:
            h[free] -= delta
        else:
            h[free] -= delta   # diff negative => adds
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Return C5 for a discretised step function h using FFT convolution."""
    n = h.size
    # length needed for full correlation is 2*n-1
    M = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fh = np.fft.fft(h, M)
    fb = np.fft.fft((1.0 - h)[::-1], M)
    conv = np.fft.ifft(fh * fb).real
    max_corr = np.max(conv[: 2 * n - 1])
    return max_corr * (2.0 / n)


def upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


class InequalityConstraint:
    """t - correlation >= 0 for every shift."""
    def __init__(self, n: int):
        self.n = n

    def __call__(self, x):
        h = x[: self.n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")


class EqualityConstraint:
    """Sum(h) = n/2."""
    def __init__(self, n: int):
        self.n = n

    def __call__(self, x):
        return np.sum(x[: self.n]) - (self.n / 2.0)


def objective(x):
    """Minimise the auxiliary variable t."""
    return x[-1]


def solve_slsqp(h0: np.ndarray, max_iter: int = 2000) -> tuple:
    """Refine a candidate with SLSQP."""
    n = h0.size
    h0 = project_to_feasible(h0, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": InequalityConstraint(n)},
        {"type": "eq", "fun": EqualityConstraint(n)},
    ]
    res = optimize.minimize(
        objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    h_opt = project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    return h_opt, compute_c5(h_opt)


def local_refine(h: np.ndarray, attempts: int = 20, delta: float = 0.01) -> np.ndarray:
    """Small stochastic hill‑climb around h."""
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = project_to_feasible(trial, n)
        c5 = compute_c5(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


def anneal_continuous(start_h: np.ndarray, time_budget: float) -> tuple:
    """Simulated annealing on continuous values."""
    n = start_h.size
    best_h = start_h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    t_start = time.time()
    # temperature schedule
    temp_max = max(0.5 * best_c5, 1e-3)
    temp_min = 0.001 * best_c5
    while time.time() - t_start < time_budget:
        elapsed = time.time() - t_start
        prog = elapsed / time_budget
        temp = temp_max * (1.0 - prog) + temp_min * prog
        k = max(1, int(0.02 * n))
        idx = np.random.choice(n, size=k, replace=False)
        trial = cur_h.copy()
        trial[idx] += temp * np.random.randn(k)
        trial = project_to_feasible(trial, n)
        c5 = compute_c5(trial)
        if c5 < cur_c5 or np.random.rand() < np.exp((cur_c5 - c5) / max(temp, 1e-12)):
            cur_h, cur_c5 = trial, c5
            if c5 < best_c5:
                best_h, best_c5 = trial, c5
    return best_h, best_c5


def anneal_binary(n: int, time_budget: float) -> tuple:
    """Simulated annealing on binary vectors (exactly n/2 ones)."""
    # start from a random binary vector
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    t_start = time.time()
    temp_max = max(best_c5, 1e-3)
    temp_min = 0.001 * best_c5
    while time.time() - t_start < time_budget:
        elapsed = time.time() - t_start
        prog = elapsed / time_budget
        temp = temp_max * (1.0 - prog) + temp_min * prog
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        trial = cur_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        c5 = compute_c5(trial)
        if c5 < cur_c5 or np.random.rand() < np.exp((cur_c5 - c5) / max(temp, 1e-12)):
            cur_h, cur_c5 = trial, c5
            if c5 < best_c5:
                best_h, best_c5 = trial, c5
    return best_h, best_c5
