def improve_h(h0, time_limit):
    h = project_to_feasible(h0)
    best_c = c5_from_h(h)
    best_h = h.copy()
    # compute correlation vector just once per iteration
    temperature = 0.5 # starting high? we can use schedule: T = 0.5 * (time_left / total_time)
    # but tricky without total time
    # We have time_limit as "remaining time", we don't know actual total but we can derive.
    # We'll implement simple annealing: gradually reduce T as we use time.
    total_time = time_limit
    start_iter = time.time()
    iter_cnt = 0
    # max iterations perhaps 20000 or time limit
    while time.time() < start_iter + total_time:
        iter_cnt+=1
        # compute current correlation vector
        n = h.size
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # Determine max shift(s)
        max_idxs = np.flatnonzero(corr == max_corr)
        # Choose one shift randomly among them
        k = np.random.choice(max_idxs) - (n-1)  # shift k such that index in full corr corresponds to shift = idx - (n-1)
        # we can treat positive k (right shift) and negative.
        # We'll consider the sign.
        # We'll attempt targeted move to increase product for shift k.
        # We need i such that i + k in [0,n)
        # Choose i uniformly among possible i
        if k >= 0:
            possible_i = np.arange(n - k)
        else:
            possible_i = np.arange(-k, n)
        # Among them pick i that yields minimal product h[i] * h[i+k] (lowest) but also with capacity to increase.
        # compute product array
        prod = h[possible_i] * h[possible_i + k]
        # filter those where both can increase
        inc_mask = (h[possible_i] < 1) & (h[possible_i + k] < 1)
        if np.any(inc_mask):
            # choose from those inc_mask with smallest product (most to improve)
            cand_idxs = possible_i[inc_mask]
            prod_vals = prod[inc_mask]
            # select index with minimal product
            idx = cand_idxs[np.argmin(prod_vals)]
            # Determine max epsilon we can add to both
            max_eps = min(1 - h[idx], 1 - h[idx + k])
            # choose epsilon as fraction of max_eps (like 0.5*max_eps) random or fixed small
            eps = max_eps * 0.5 * np.random.rand()  # random upto half
            if eps > 1e-8:
                # need to take epsilon from donors
                # donors: pick two random indices with h > eps
                donor_idxs = np.where(h > 0)[0]
                # ensure donor size > 0. Choose 2 indices (may be same)
                # we'll reduce both donors equally epsilon each
                # need total reduction equal to 2*eps
                # choose donor1, donor2 maybe same
                if len(donor_idxs) >= 2:
                    d1, d2 = np.random.choice(donor_idxs, size=2, replace=False)
                else:
                    d1 = d2 = donor_idxs[0]
                # compute reduction capacity
                max_red1 = h[d1]
                max_red2 = h[d2]
                # total reduction capacity
                total_max_red = max_red1 + max_red2
                needed_red = 2*eps
                if total_max_red >= needed_red:
                    # scale reductions proportionally
                    red1 = needed_red * (max_red1 / total_max_red)
                    red2 = needed_red - red1
                    # apply
                    h_new = h.copy()
                    h_new[idx] += eps
                    h_new[idx + k] += eps
                    h_new[d1] -= red1
                    h_new[d2] -= red2
                    # clip ensure limits
                    h_new = np.clip(h_new, 0, 1)
                    # reproject sum to target (due to possible clipping)
                    h_new = project_to_feasible(h_new)
                    c_new = c5_from_h(h_new)
                    if c_new < best_c or np.random.rand() < np.exp((best_c - c_new) / max(temperature, 1e-8)):
                        h = h_new
                        best_c = c_new
                        best_h = h.copy()
        # else fallback random swap
        else:
            # random two indices exchange epsilon
            i,j = np.random.randint(n, size=2)
            if i==j: continue
            max_down = h[i]
            max_up = 1 - h[j]
            max_eps = min(max_down, max_up)
            if max_eps <= 0: continue
            eps = max_eps * np.random.rand() * 0.3
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            h_new = project_to_feasible(h_new)
            c_new = c5_from_h(h_new)
            # temperature as before
            if c_new < best_c or np.random.rand() < np.exp((best_c - c_new) / max(temperature, 1e-8)):
                h = h_new
                best_c = c_new
                best_h = h.copy()
        # update temperature schedule, e.g., decay
        # compute fraction of elapsed time
        elapsed = time.time() - start_iter
        frac = elapsed / total_time
        temperature = max(0.001, 0.5 * (1 - frac))
    return best_h, best_c

def upscale(arr, n):
    m = arr.size
    x_c = (np.arange(m) + 0.5) * (2.0 / m)
    x_n = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x_n, x_c, arr)

def improve_h(h0, time_limit):
    h = _project_to_feasible(h0.copy(), h0.size)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    n = h.size
    start = time.time()
    # parameters
    T0 = 0.1
    # keep track of iteration count
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # update temperature
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + 1e-6

        # pick move type
        if np.random.rand() < 0.7:
            # targeted move using worst shift
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = corr.max()
            max_idxs = np.flatnonzero(corr == max_corr)
            idx = np.random.choice(max_idxs)
            k = idx - (n - 1)
            # compute feasible indices for i
            if k >= 0:
                i_candidates = np.arange(n - k)
            else:
                i_candidates = np.arange(-k, n)
            # compute product and capacity
            product = h[i_candidates] * h[i_candidates + k]
            inc_mask = (h[i_candidates] < 1) & (h[i_candidates + k] < 1)
            good_i = i_candidates[inc_mask]
            if good_i.size > 0:
                # select i with minimum product
                i = good_i[np.argmin(product[inc_mask])]
                # epsilon
                max_eps = min(1 - h[i], 1 - h[i + k])
                # ensure small enough to also find donors
                eps = max_eps * np.random.rand() * 0.5
                if eps > 1e-8:
                    # pick donor indices
                    donor_idxs = np.where(h > 0)[0]
                    if donor_idxs.size >= 2:
                        d1, d2 = np.random.choice(donor_idxs, size=2, replace=False)
                        # reduce equally may not keep total? we need sum(h) unchanged
                        # Let's reduce each by eps
                        # Actually need total reduction 2*eps. If each donor has at least eps.
                        if h[d1] >= eps and h[d2] >= eps:
                            h_new = h.copy()
                            h_new[i] += eps
                            h_new[i + k] += eps
                            h_new[d1] -= eps
                            h_new[d2] -= eps
                            h_new = _project_to_feasible(h_new, n)
                            c_new = _c5_from_h(h_new)
                            if c_new < best_c or np.random.rand() < np.exp((best_c - c_new) / max(T, 1e-12)):
                                h = h_new
                                if c_new < best_c:
                                    best_c = c_new
                                    best_h = h.copy()
                                continue
        # random pairwise move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = h[i]
        max_up = 1 - h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.5
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        h_new = _project_to_feasible(h_new, n)
        c_new = _c5_from_h(h_new)
        if c_new < best_c or np.random.rand() < np.exp((best_c - c_new) / max(T, 1e-12)):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    return best_h, best_c

def _project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # using simple iterative approach
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # if sum too high, decrease those >0 equally
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        count = np.count_nonzero(mask)
        if count == 0:
            break
        delta = diff / count
        h[mask] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction:
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first non-saturated entries
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time

def _project_to_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h)=n/2 by a simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        h[mask] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a given discretised h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    m = coarse_h.size
    x_coarse = (np.arange(m) + 0.5) * (2.0 / m)
    x_fine = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x_fine, x_coarse, coarse_h)


def _shape_h(n: int, r: float) -> np.ndarray:
    """Symmetric shape: zero edges, linear ramps of total width r,
    central plateau of height 1.  r in [0,2]."""
    if r < 1e-12:
        # pure block of ones of length 1 centered.
        Lz = 0.5
        a = 1.0
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.zeros_like(x)
        mask = (x >= Lz) & (x < Lz + a)
        h[mask] = 1.0
        return h
    a = 1.0 - 0.5 * r                     # plateau length
    Lz = 0.5 - 0.25 * r                    # length of each zero side
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.zeros_like(x)

    left_ramp_end = Lz + r / 2.0
    mask = (x >= Lz) & (x < left_ramp_end)
    h[mask] = (x[mask] - Lz) * (2.0 / r)   # slope up

    plateau_end = left_ramp_end + a
    mask = (x >= left_ramp_end) & (x < plateau_end)
    h[mask] = 1.0

    right_ramp_end = plateau_end + r / 2.0
    mask = (x >= plateau_end) & (x < right_ramp_end)
    h[mask] = 1.0 - (x[mask] - plateau_end) * (2.0 / r)  # slope down
    return h


def _find_best_shape(n: int, time_limit: float):
    """Search r∈[0,2] for the shape with smallest C5 (very cheap)."""
    start = time.time()
    best_c = np.inf
    best_h = None
    best_r = 0.0

    # coarse grid
    for r in np.linspace(0.0, 2.0, 101):
        h = _shape_h(n, r)
        h = _project_to_feasible(h, n)
        c = _c5_from_h(h)
        if c < best_c:
            best_c, best_h, best_r = c, h, r

    # three refinement stages
    step = 0.02
    for _ in range(3):
        low = max(0.0, best_r - step)
        high = min(2.0, best_r + step)
        for r in np.linspace(low, high, 31):
            h = _shape_h(n, r)
            h = _project_to_feasible(h, n)
            c = _c5_from_h(h)
            if c < best_c:
                best_c, best_h, best_r = c, h, r
        step *= 0.4
        if time.time() - start > time_limit:
            break
    return best_h, best_c, best_r


def _random_feasible(n: int) -> np.ndarray:
    """Draw a random point in the feasible polytope."""
    return _project_to_feasible(np.random.rand(n), n)


def _improve_h(h0: np.ndarray, time_limit: float):
    """Simulated‑annealing / hill‑climbing refinement."""
    n = h0.size
    h = _project_to_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    start = time.time()
    T0 = 0.001  # initial temperature

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac) + 1e-8

        # ---- targeted move based on worst shift ----
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        idxs = np.flatnonzero(corr == max_corr)
        k = np.random.choice(idxs) - (n - 1)          # shift producing the max
        if k >= 0:
            i_cand = np.arange(n - k)
        else:
            i_cand = np.arange(-k, n)
        prod = h[i_cand] * h[i_cand + k]
        inc_mask = (h[i_cand] < 1.0) & (h[i_cand + k] < 1.0)
        if np.any(inc_mask):
            i_good = i_cand[inc_mask]
            prod_good = prod[inc_mask]
            i = i_good[np.argmin(prod_good)]          # pair with smallest product
            max_eps = min(1.0 - h[i], 1.0 - h[i + k])
            eps = max_eps * np.random.rand()          # any positive amount
            if eps > 1e-12:
                donor_idxs = np.where(h > 0.0)[0]
                if donor_idxs.size:
                    if donor_idxs.size == 1:
                        d1 = d2 = donor_idxs[0]
                    else:
                        d1, d2 = np.random.choice(donor_idxs, size=2, replace=False)
                    if h[d1] >= eps and h[d2] >= eps:
                        h_new = h.copy()
                        h_new[i] += eps
                        h_new[i + k] += eps
                        h_new[d1] -= eps
                        h_new[d2] -= eps
                        h_new = _project_to_feasible(h_new, n)
                        c_new = _c5_from_h(h_new)
                        if c_new < best_c or np.random.rand() < np.exp((best_c - c_new) / T):
                            h = h_new
                            if c_new < best_c:
                                best_c, best_h = c_new, h.copy()
                            continue

        # ---- random pairwise epsilon exchange ----
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = h[i]
        max_up = 1.0 - h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0.0:
            continue
        eps = max_eps * np.random.rand() * 0.5
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        h_new = _project_to_feasible(h_new, n)
        c_new = _c5_from_h(h_new)
        if c_new < best_c or np.random.rand() < np.exp((best_c - c_new) / T):
            h = h_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
    return best_h, best_c
