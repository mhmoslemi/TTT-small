def project_to_feasible(h, target, tol=1e-9, max_iter=1000):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual greedily
        for i in range(len(h)):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-9,
                         max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the required sum (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# SLSQP local optimisation
# ----------------------------------------------------------------------
def _slsqp_obj(x):
    return x[-1]


def _slsqp_ineq(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr  # element‑wise, SciPy treats each as a constraint


def _slsqp_eq(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_slsqp(initial_h: np.ndarray, max_iter: int = 300):
    """Run SLSQP from a feasible start."""
    n = initial_h.size
    target = n / 2.0
    h0 = _project_to_feasible(initial_h, target)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _slsqp_ineq},
        {"type": "eq", "fun": _slsqp_eq},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = minimize(_slsqp_obj, x0, method="SLSQP",
                   bounds=bounds, constraints=constraints, options=opts)
    if res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, target)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, deadline: float):
    """Exchange epsilon mass between two coordinates, accepting improvements."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount we can move from i to j
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Differential Evolution (global search)
# ----------------------------------------------------------------------
def _de_optimize(base_h: np.ndarray, n: int, target_sum: float,
                 deadline: float, popsize: int = 15, max_gen: int = 80,
                 F: float = 0.8, CR: float = 0.7):
    """Simple DE with projection onto the feasible simplex."""
    # initialise population
    pop = np.empty((popsize, n))
    obj = np.empty(popsize)
    pop[0] = _project_to_feasible(base_h.copy(), target_sum)
    obj[0] = _c5_from_h(pop[0])
    for i in range(1, popsize):
        rand_vec = np.random.rand(n)
        pop[i] = _project_to_feasible(rand_vec, target_sum)
        obj[i] = _c5_from_h(pop[i])

    best_idx = int(np.argmin(obj))
    best_h = pop[best_idx].copy()
    best_val = obj[best_idx]

    gen = 0
    while time.time() < deadline and gen < max_gen:
        gen += 1
        for i in range(popsize):
            # choose three distinct others
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = _project_to_feasible(mutant, target_sum)

            cross = np.random.rand(n) < CR
            if not np.any(cross):
                cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _project_to_feasible(trial, target_sum)

            trial_val = _c5_from_h(trial)
            if trial_val < obj[i]:
                pop[i] = trial
                obj[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
    return best_h, best_val
