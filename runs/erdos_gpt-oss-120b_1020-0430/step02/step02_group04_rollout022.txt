import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Box–sum projection: clip to [lower,upper] and adjust to achieve exact sum=target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random feasible vector (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def perturb_h(h, rng, scale):
    """Transfer a random amount from i to j, respecting bounds."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * scale * max_eps
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


def hill_climb(h_start, rng, max_iters, init_scale=0.5, decay=0.9, min_scale=1e-4):
    """Simple hill‑climb with decreasing move size."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = init_scale
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            scale = max(scale * decay, min_scale)
    return best_h, best_c


# ---------- smooth‑max gradient descent ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    C = corr * (2.0 / x.size)
    bc = CURRENT_BETA * C
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP refinement using the log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, n / 2.0)


def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            g = np.zeros_like(h)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g[: n - k] += wk * (1 - h[k:])
                    g[k:] += wk * (-h[: n - k])
                else:
                    kk = -k
                    g[kk:] += wk * (1 - h[: n - kk])
                    g[: n - kk] += wk * (-h[kk:])
            grad = (2.0 / n) * g
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    T = budget_s

    # target resolution
    n = 200
    target_sum = n / 2.0

    # start from parent if available, otherwise random
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            h0 = parent.copy()
        else:
            x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_target = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_target, x_parent, parent)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target_sum)
    else:
        h0 = random_feasible(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------- many hill‑climb restarts ----------
    while time.time() - start < T * 0.70:
        # allocate iterations based on remaining time (roughly 0.5 ms per iteration)
        remaining = T * 0.70 - (time.time() - start)
        max_iters = int(min(20000, max(1000, remaining * 500)))
        h_start = random_feasible(n, rng)
        if rng.random() < 0.15:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, target_sum)
        h_opt, c_opt = hill_climb(h_start, rng, max_iters,
                                 init_scale=0.5, decay=0.95, min_scale=1e-4)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---------- gradient‑descent refinements ----------
    beta_sched = [5, 20, 80, 200, 800, 2000]
    steps_per_beta = 120
    for _ in range(3):
        if time.time() - start > T * 0.90:
            break
        h_opt, c_opt = grad_descent_optimize(best_h, beta_sched, steps_per_beta, rng)
        try:
            h_opt = slsqp_optimize(h_opt, beta=800, max_iter=300)
        except Exception:
            pass
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---------- binary simulated‑annealing search ----------
    if time.time() - start < T * 0.95:
        binary_time = T * 0.20
        bin_start = time.time()
        bin_best_c = float("inf")
        bin_best_h = None
        while time.time() - bin_start < binary_time:
            h_bin = np.zeros(n, dtype=float)
            ones = rng.choice(n, size=n // 2, replace=False)
            h_bin[ones] = 1.0
            c_bin = compute_c5(h_bin)
            best_local_c = c_bin
            best_local_h = h_bin.copy()
            T_ann = 0.02
            T_min = 1e-7
            decay = 0.9995
            ones_set = set(ones.tolist())
            zeros_set = set(range(n)) - ones_set
            while T_ann > T_min and time.time() - bin_start < binary_time:
                i_one = rng.choice(list(ones_set))
                i_zero = rng.choice(list(zeros_set))
                h_new = h_bin.copy()
                h_new[i_one] = 0.0
                h_new[i_zero] = 1.0
                c_new = compute_c5(h_new)
                delta = c_new - c_bin
                if delta <= 0.0 or rng.random() < np.exp(-delta / T_ann):
                    h_bin = h_new
                    c_bin = c_new
                    ones_set.remove(i_one)
                    zeros_set.remove(i_zero)
                    ones_set.add(i_zero)
                    zeros_set.add(i_one)
                if c_bin < best_local_c:
                    best_local_c = c_bin
                    best_local_h = h_bin.copy()
                T_ann *= decay
            if best_local_c < bin_best_c:
                bin_best_c, bin_best_h = best_local_c, best_local_h.copy()
            if bin_best_c < best_c:
                best_h, best_c = bin_best_h, bin_best_c

    # ---------- optional up‑sampling and fine‑tune ----------
    if time.time() - start < T * 0.99 and best_h.size * 2 <= 400:
        h_up = np.repeat(best_h, 2)
        up_c = compute_c5(h_up)
        for _ in range(2):
            h_up, up_c = hill_climb(h_up, rng, max_iters=3000,
                                    init_scale=0.4, decay=0.90, min_scale=1e-4)
            h_up, up_c = grad_descent_optimize(h_up, beta_sched, 80, rng)
            if up_c < best_c:
                best_h, best_c = h_up, up_c

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
