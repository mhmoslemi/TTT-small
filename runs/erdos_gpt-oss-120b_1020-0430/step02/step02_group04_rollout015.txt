def gradient_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.2
        momentum = np.zeros_like(h)
        mu = 0.9
        for it in range(steps_per_beta):
            # compute corr, weights
            corr = np.correlate(h, 1-h, mode='full')
            C = corr * (2.0/n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.ones_like(w)/w.size
            # compute gradient via loop
            grad = np.zeros_like(h)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    grad[:n-k] -= wk * h[k:]
                    grad[k:]   -= wk * h[:n-k]
                else:
                    kk = -k
                    grad[kk:]   -= wk * h[:n-kk]
                    grad[:n-kk] -= wk * h[kk:]
            grad = (2.0/n) * grad
            momentum = mu * momentum + grad
            new_h = h - step * momentum
            new_h = project_onto_box_sum(np.clip(new_h,0,1), n/2.0)
            h = new_h
            # occasionally evaluate
            if (it+1) % 50 == 0 or it == steps_per_beta-1:
                cur_c = compute_c5(h)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                # adapt step maybe
                if cur_c > best_c + 1e-5:
                    step *= 0.9
                else:
                    step *= 1.01
        # after each beta, maybe target best
        h = best_h.copy()
    return best_h, best_c
```

def hill_climb(h_start, rng, max_iters=2000, step_scale=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    for it in range(max_iters):
        # Choose random pair
        i, j = rng.choice(len(h), size=2, replace=False)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # ensure bounds
        new_h = np.clip(new_h, 0.0, 1.0)
        # project sum
        new_h = project_onto_box_sum(new_h, h.size / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
            # optionally increase step
        # step update schedule
        if (it+1) % max(1, max_iters // 10) == 0:
            step *= 0.9
    return best_h, best_c

    step = 0.2

    mu = 0.9
    for t in range(total_steps):
        # compute correlation, weights as before
        corr = np.correlate(h, 1-h, mode='full')
        C = corr * (2.0/n)
        bc = beta * C
        max_bc = np.max(bc)
        w = np.exp(bc - max_bc)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        # compute gradient
        grad = np.zeros_like(h)
        for idx,wk in enumerate(w):
            if wk == 0.0:
                continue
            k = idx - (n - 1)
            if k >= 0:
                grad[:n-k] -= wk * h[k:]
                grad[k:]   -= wk * h[:n-k]
            else:
                kk = -k
                grad[kk:]   -= wk * h[:n-kk]
                grad[:n-kk] -= wk * h[kk:]
        grad = (2.0 / n) * grad
        momentum = mu * momentum + grad
        # update

        new_h = np.clip(new_h,0,1)

                best_h = h.copy()
                step = min(step * 1.05, 0.5)   # increase step a bit after improvement

import time
import numpy as np

# ---------- helpers ----------
def compute_c5(h):
    """Discrete C5 for step function h with dx = 2/len(h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """
    Project v onto the box [lower,upper] while enforcing sum(v) == target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Random vector in [0,1]^n with exact sum = n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def upsample_h(h, new_n):
    """Linear up‑sampling of h from its length to new_n points."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)


def hill_climb(h_start, rng, max_iters=2000, step_scale=0.5):
    """Simple pairwise epsilon‑swap hill climbing."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2, replace=False)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, target)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 10) == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_descent_optimize(
    h0, beta_start, beta_end, total_steps, rng, step0=0.2, mu=0.9
):
    """
    Projected gradient descent on the smooth max surrogate.
    Uses convolution (np.convolve) to compute the gradient efficiently.
    """
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    beta = beta_start
    beta_factor = (beta_end / beta_start) ** (1.0 / max(total_steps - 1, 1))
    step = step0
    momentum = np.zeros_like(h)

    for t in range(total_steps):
        # correlation and soft‑max weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        C = corr * (2.0 / n)
        bc = beta * C
        m = np.max(bc)
        w = np.exp(bc - m)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        # gradient via convolution
        conv = np.convolve(w, h, mode="full")
        conv_rev = np.convolve(w[::-1], h, mode="full")
        off = n - 1
        grad = -(2.0 / n) * (conv_rev[off : off + n] + conv[off : off + n])

        # momentum update
        momentum = mu * momentum + grad
        new_h = h - step * momentum
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, target)

        h = new_h
        beta *= beta_factor

        # occasional evaluation
        if (t + 1) % 200 == 0:
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                step = min(step * 1.05, 0.5)
            else:
                step *= 0.97

    # final check
    final_c = compute_c5(h)
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()

    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=200):
    """Single SLSQP refinement on the log‑sum‑exp surrogate."""
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        C = corr * (2.0 / n)
        bc = beta * C
        m = np.max(bc)
        # smooth max = (log sum exp)/beta
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, target)
    return x


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----
    # target resolution (larger than the parent 180 points)
    target_n = 360
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != target_n:
            h = upsample_h(h0, target_n)
        else:
            h = h0.copy()
    else:
        h = random_feasible(target_n, rng)

    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target_n / 2.0)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- quick hill‑climb on the start ----
    if time.time() - start < budget_s * 0.1:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2500, step_scale=0.6)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---- main gradient‑descent optimisation (single long run) ----
    beta_start, beta_end = 5.0, 2000.0
    total_steps = 5000
    if time.time() - start < budget_s * 0.55:
        h_opt, c_opt = gradient_descent_optimize(
            best_h, beta_start, beta_end, total_steps, rng, step0=0.2, mu=0.9
        )
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- short SLSQP refinement (optional) ----
    if time.time() - start < budget_s * 0.6:
        try:
            best_h = slsqp_optimize(best_h, beta=500.0, max_iter=250)
        except Exception:
            pass
        best_c = compute_c5(best_h)

    # ---- occasional random‑restart improvements ----
    while time.time() - start < budget_s * 0.8:
        h_start = random_feasible(target_n, rng)
        # blend occasionally with current best
        if rng.random() < 0.15:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, target_n / 2.0)

        h_opt, c_opt = gradient_descent_optimize(
            h_start, beta_start, beta_end, 3000, rng, step0=0.25, mu=0.9
        )
        # brief hill‑climb polish
        h_opt, c_opt = hill_climb(h_opt, rng, max_iters=800, step_scale=0.3)

        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- optional up‑sampling to double resolution ----
    if time.time() - start < budget_s * 0.95:
        up_n = target_n * 2
        h_up = upsample_h(best_h, up_n)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = project_onto_box_sum(h_up, up_n / 2.0)

        # refine at higher resolution
        h_up, c_up = gradient_descent_optimize(
            h_up, beta_start, beta_end, 3000, rng, step0=0.15, mu=0.9
        )
        h_up, c_up = hill_climb(h_up, rng, max_iters=1200, step_scale=0.2)

        if c_up < best_c:
            best_h, best_c = h_up, c_up
            target_n = up_n

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
