    return contrib

    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    temp = init_temp
    min_eps = 1e-12
    for it in range(max_iter):
        # choose move type
        if rng.random() < 0.7:
            # target move
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            idx_max = np.argmax(corr)
            k = idx_max - (n - 1)
            # contribution array:
            contrib = np.zeros(n)
            if k >= 0:
                contrib[:n - k] = 1.0 - cur_h[k:]
            else:
                contrib[-k:] = 1.0 - cur_h[:n + k]
            mask_src = cur_h > min_eps
            mask_dst = cur_h < 1 - min_eps
            if np.any(mask_src) and np.any(mask_dst):
                src_candidates = np.where(mask_src)[0]
                src = src_candidates[np.argmax(contrib[src_candidates])]
                dst_candidates = np.where(mask_dst)[0]

                    new_h[dst] += eps

                    # fallback random

                new_h = perturb_transfer(cur_h, rng, max_step * (temp / init_temp))

import time
import numpy as np
from scipy.optimize import minimize

# globals for the surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_transfer(h, rng, step_scale):
    """Transfer a fraction of mass from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h
    # random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_transfer = min(h[src], 1.0 - h[dst])

    new_h[src] -= eps

def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):

def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)

def targeted_anneal_optimize(h_start, rng, max_iter=25000,
                             init_temp=0.05, decay=0.9995, max_step=0.5):
    """Annealing that preferentially transfers mass from indices that
    contribute most to the current worst overlap to those that contribute least."""
    n = h_start.size
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    temp = init_temp
    min_eps = 1e-12

    for it in range(max_iter):
        # 70 % of the time try a targeted move
        if rng.random() < 0.7:
            corr = np.correlate(cur_h, 1 - cur_h, mode="full")
            idx = np.argmax(corr)
            k = idx - (n - 1)                     # shift that attains the max
            contrib = np.zeros(n)
            if k >= 0:
                contrib[:n - k] = 1.0 - cur_h[k:]
            else:
                contrib[-k:] = 1.0 - cur_h[:n + k]

            src_mask = cur_h > min_eps
            dst_mask = cur_h < 1.0 - min_eps
            if src_mask.any() and dst_mask.any():
                src_candidates = np.where(src_mask)[0]
                src = src_candidates[np.argmax(contrib[src_candidates])]
                dst_candidates = np.where(dst_mask)[0]
                dst = dst_candidates[np.argmin(contrib[dst_candidates])]
                max_transfer = min(cur_h[src], 1.0 - cur_h[dst])
                if max_transfer > 0:
                    scale = max_step * (temp / init_temp) * rng.random()
                    eps = max_transfer * scale
                    if eps < 1e-12:
                        eps = max_transfer * 0.01
                    new_h = cur_h.copy()
                    new_h[src] -= eps
                    new_h[dst] += eps
                else:
                    new_h = perturb_transfer(cur_h, rng, max_step * (temp / init_temp))
            else:
                new_h = perturb_transfer(cur_h, rng, max_step * (temp / init_temp))
        else:
            new_h = perturb_transfer(cur_h, rng, max_step * (temp / init_temp))

        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()

        # Metropolis acceptance
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            cur_h = new_h
            cur_c = new_c

        temp *= decay
        if temp < 1e-12:
            temp = 1e-12
    return best_h, best_c

    if elapsed < budget_s * 0.80 and n * 2 <= 800:
        # one up‑sample step
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        best_up_h, best_up_c = quick_refine(h_up, rng)

        # targeted annealing at the higher resolution
        remaining = budget_s - (time.time() - start_time)
        if remaining > budget_s * 0.10:
            iters_up = 30000
            best_up_h, best_up_c = targeted_anneal_optimize(
                best_up_h, rng, max_iter=iters_up)

        if best_up_c < best_c:
            best_h, best_c = best_up_h, best_up_c
            n = n_up

    # ----------------------------------------------------------------------
    # Final feasibility clean‑up
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
