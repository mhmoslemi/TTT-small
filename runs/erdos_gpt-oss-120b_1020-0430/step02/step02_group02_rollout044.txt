```python
def _shift_with_zeros(arr: np.ndarray, shift: int) -> np.ndarray:
    n = arr.size
    if shift >= 0:
        if shift >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[:shift] = 0.0
        out[shift:] = arr[:n-shift]
        return out
    else:
        s = -shift
        if s >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[-s:] = 0.0
        out[:-s] = arr[s:]
        return out
```

```python
def _gradient_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)
    pos = _shift_with_zeros(h, shift)

    grad = 1.0 - pos - neg
    return grad, shift, corr[idx]
```

```python
def _hill_climb(h, max_steps, time_limit):

    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):

            break

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # possible epsilon limited by bounds
        max_down = best_h[i]  # can reduce
        max_up = 1.0 - best_h[j]  # can increase

        eps = np.random.rand() * max_eps * 0.5  # small move

        cand[i] -= eps
```

def _project_to_feasible(x: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 using water-filling."""
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(x > 0)[0]
        else:
            free = np.where(x < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and x[i] < 1 - 1e-12:
                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 1e-12:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a step function h on [0,2] (dx = 2/n)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse vector onto a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


def _shift_with_zeros(arr: np.ndarray, shift: int) -> np.ndarray:
    n = arr.size
    if shift >= 0:
        if shift >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[:shift] = 0.0
        out[shift:] = arr[: n - shift]
        return out
    else:
        s = -shift
        if s >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[-s:] = 0.0
        out[:-s] = arr[s:]
        return out


def _gradient_max_corr(h: np.ndarray):
    """Gradient of the most offending correlation shift."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)
    pos = _shift_with_zeros(h, shift)
    neg = _shift_with_zeros(h, -shift)
    grad = 1.0 - pos - neg
    return grad, shift, corr[idx]


def _grad_descent(h0: np.ndarray, max_iter: int, step0: float, deadline: float):
    """Projected gradient descent targeting the current worst shift."""
    n = h0.size
    h = _project_to_feasible(h0.copy(), n)
    best_h, best_c = h, _c5_from_h(h)
    step = step0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        grad, _, _ = _gradient_max_corr(best_h)
        cand = _project_to_feasible(best_h - step * grad, n)
        cand_c = _c5_from_h(cand)
        if cand_c < best_c - 1e-12:
            best_h, best_c = cand, cand_c
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.5
            if step < 1e-7:
                break
    return best_h, best_c


def _hill_climb(h: np.ndarray, steps: int, deadline: float):
    """Simple pairwise epsilon exchanges improving C5."""
    n = h.size
    best_h, best_c = h.copy(), _c5_from_h(h)
    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = _c5_from_h(cand)
        if cand_c < best_c - 1e-12:
            best_h, best_c = cand, cand_c
    return best_h, best_c


def _random_feasible(n: int):
    """Uniform random vector projected onto the feasible set."""
    rnd = np.random.rand(n)
    return _project_to_feasible(rnd, n)


def _binary_feasible(n: int):
    """Random 0‑1 vector with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n)
    arr[:half] = 1.0
    np.random.shuffle(arr)
    return arr.astype(float)
