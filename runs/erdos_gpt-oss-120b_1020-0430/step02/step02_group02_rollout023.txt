def hill_climb(h, steps, time_limit):

   n = h.size

   best_h = h.copy()

   best_c = compute_c5(best_h)

   # Pre-alloc loop with ability to break early.

   for step in range(steps):

        if time.time() > time_limit: break

        # choose i,j distinct

        i, j = np.random.randint(n, size=2)

        if i == j: continue

        # Randomly decide to swap or transfer

        if np.random.rand() < 0.1: # swap

            if best_h[i] == best_h[j]: continue

            cand = best_h.copy()

            tmp = cand[i]; cand[i] = cand[j]; cand[j] = tmp

        else:

            maxdown = best_h[i]

            maxup = 1.0 - best_h[j]

            maxeps = min(maxdown, maxup)

            if maxeps <= 0: continue

            eps = np.random.rand() * maxeps

            cand = best_h.copy()

            cand[i] -= eps

            cand[j] += eps

        # compute c

        c = compute_c5(cand)

        if c < best_c - 1e-12:

            best_h, best_c = cand, c

   return best_h, best_c

import time
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
# Feasibility projection
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:               # need to reduce sum
            free = np.where(h > 0)[0]
        else:                      # need to increase sum
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to eliminate rounding error
    residual = target - h.sum()
    if abs(residual) > 1e-15:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-15:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-15:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# SLSQP optimisation of the max‑overlap formulation
# ----------------------------------------------------------------------
def _solve_h_slsqp(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)

    # starting bound (unscaled)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    constraints = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    res = opt.minimize(lambda x: x[-1], x0, method="SLSQP",
                       bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Linear up‑sampling of a coarse step function
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Simple hill‑climbing (transfer + occasional swap)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    # pre‑compute indices array for speed
    indices = np.arange(n)
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # with a small probability perform a pure swap
        if np.random.rand() < 0.08:
            if best_h[i] == best_h[j]:
                continue
            cand = best_h.copy()
            cand[i], cand[j] = cand[j], cand[i]          # swap values
        else:
            max_down = best_h[i]
            max_up = 1.0 - best_h[j]
            max_eps = min(max_down, max_up)
            if max_eps <= 0.0:
                continue
            # bias epsilon toward larger values
            eps = (np.random.rand() ** 0.5) * max_eps
            cand = best_h.copy()
            cand[i] -= eps
            cand[j] += eps

        c = _c5_from_h(cand)
        if c < best_c - 1e-14:
            best_h, best_c = cand, c
    return best_h, best_c
