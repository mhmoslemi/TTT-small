import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in-place so that sum(h) == target while keeping 0<=h<=1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Uniform random feasible vector (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        return h.copy()
    eps = (rng.random() * 2.0 - 1.0) * max_transfer * scale
    eps = max(-max_transfer * scale, min(eps, max_transfer * scale))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def smooth_max_obj(x, beta):
    """Log‑sum‑exp smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_refine(h0, beta, max_iter=500):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    res = minimize(lambda x: smooth_max_obj(x, beta), h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny drift)
    rng = np.random.default_rng()
    return enforce_sum(x, n / 2.0, rng)

def anneal(h_start, rng, time_limit):
    """
    Simple simulated‑annealing optimiser.
    Returns best vector found and its C5 value.
    """
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    # temperature schedule parameters
    T0 = max(0.01, 0.05 * best_c)  # a modest initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac)                     # linear cooling
        step_scale = 0.4 * (1.0 - frac) + 0.02    # shrink step size
        # pick i (with mass) and j (with capacity)
        i = rng.integers(0, n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if h[j] >= 1.0 - 1e-12:
            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0.0:
            continue
        eps = rng.random() * max_transfer * step_scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        delta = cur_c - new_c
        if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-12)):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
    return best_h, best_c

def upsample_interpolate(h, rng):
    """Linear interpolation to double the resolution, then enforce sum."""
    n = h.size
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    n2 = 2 * n
    dx2 = 2.0 / n2
    xs2 = (np.arange(n2) + 0.5) * dx2
    h2 = np.interp(xs2, xs, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return enforce_sum(h2, n2 / 2.0, rng)

def search_phase(h_initial, rng, start_time, budget_s, time_fraction, max_seg):
    """
    Generic multi‑restart simulated‑annealing.
    Uses a fraction of the total budget.
    """
    n = h_initial.size
    best_h = h_initial.copy()
    best_c = compute_c5(best_h)
    while True:
        now = time.time()
        remaining = budget_s - (now - start_time)
        if remaining < 5.0:
            break
        seg = min(remaining * time_fraction, max_seg)
        if seg < 1.0:
            break
        # start either from a perturbed elite or a fresh random point
        if rng.random() < 0.7:
            start_h = perturb_h(best_h, rng, scale=0.35)
        else:
            start_h = random_feasible_vector(n, rng)
        h_loc, c_loc = anneal(start_h, rng, seg)
        if c_loc < best_c:
            best_c = c_loc
            best_h = h_loc.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Base problem size and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)

        h0 = random_feasible_vector(n0, rng)

    # ------------------------------------------------------------------
    # Coarse search on the base grid
    # ------------------------------------------------------------------
    h_best, c_best = search_phase(
        h0, rng, start_time, budget_s, time_fraction=0.4, max_seg=30.0
    )

    # If time permits, move to a finer grid via interpolation
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

        h_fine = upsample_interpolate(h_best, rng)
        h_fine, c_fine = search_phase(
            h_fine, rng, start_time, budget_s, time_fraction=0.3, max_seg=25.0

    # Local refinement with increasing β (smooth surrogate)
    # ------------------------------------------------------------------
    for beta in (10, 30, 100, 300, 800):

        except Exception:
            pass
        c_best = compute_c5(h_best)

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    n_final = h_best.size
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = enforce_sum(h_best, n_final / 2.0, rng)
    final_c = compute_c5(h_best)

    return h_best.tolist(), float(final_c), int(n_final)
