    # w shape (2*n-1)
    n = h.size
    corr_fwd = np.correlate(w, h, mode='full')

def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                       # scaled overlaps
            bc = beta * C
            m = bc.max()
            w = np.exp(bc - m)
            w /= w.sum()
            # compute gradient
            # w size is 2*n-1, aligned with shifts = -(n-1)...(n-1)
            # correlation w with h (two directions)
            # compute forward:
            corr_fwd = np.correlate(w, h, mode='full')
            corr_rev = np.correlate(w[::-1], h, mode='full')
            start = n - 1
            g_raw = 1.0 - corr_fwd[start:start+n] - corr_rev[start:start+n]
            grad = (2.0 / n) * g_raw
            # step and project
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # accept with small probability to maintain diversity
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Clamp to [lower,upper] while fixing sum(v) = target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible vector with values in [0,1] and sum = n/2."""
    m = n // 2
    half = rng.random(m)
    half = project_onto_box_sum(half, n / 4.0)
    full = np.concatenate([half, half[::-1]])
    return full


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- gradient descent on smooth max ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected GD on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            # current smooth max weights
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = bc.max()
            w = np.exp(bc - m)
            w /= w.sum()

            # gradient = (2/n)*(1 - sum_k w_k (h_{i+k}+h_{i-k}))
            fw = np.correlate(w, h, mode="full")
            rw = np.correlate(w[::-1], h, mode="full")
            start = n - 1
            g_raw = 1.0 - fw[start:start + n] - rw[start:start + n]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            # enforce symmetry
            new_h = (new_h + new_h[::-1]) * 0.5
            new_h = project_onto_box_sum(new_h, n / 2.0)

            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

def triangular_h(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.where(x <= 1.0, x, 2.0 - x)
    return h
