def perturb_h_feasible(h, rng, scale=0.2):
    n = h.size
    # choose i,j and transfer epsilon.
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i==j:
        return h.copy()
    # epsilon w.r.t transfer: max amount to move from i to j while staying in [0,1].
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2
```

def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta
```

def slsqp_optimize(h0, beta, target, max_iter=400):
    n = h0.size
    bounds = [(0.0,1.0)]*n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x,beta), h0, method="SLSQP", bounds=bounds, constraints=cons,
                         options={"maxiter": max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x,0,1)
    # repair sum (inverse)
    s = np.sum(x)
    diff = target - s
    if abs(diff) > 1e-12:
        x = project_to_sum_box(x, target)
    return x
```

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def compute_c5(h):
    """Exact discrete C5 for step function h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_sum_box(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    """Projection of v onto {lower<=x_i<=upper, sum x_i = target}."""
    lo = np.min(v - upper) - 1.0
    hi = np.max(v - lower) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)


def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_to_sum_box(v, n / 2.0)


def binary_feasible_vector(n, rng):
    # n is even in our usage
    ones = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(start, rng, max_iters=3000, start_step=0.5, time_limit=None, start_time=None, budget=None):
    h = start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stabilized log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0, beta, target, max_iter=600):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0,
                   method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # deterministic repair for possible drift
    return project_to_sum_box(x, target)


def upsample_h(h, factor):
    n = h.size
    n_up = n * factor
    dx = 2.0 / n
    dx_up = 2.0 / n_up
    xs = (np.arange(n) + 0.5) * dx
    xs_up = (np.arange(n_up) + 0.5) * dx_up
    h_up = np.interp(xs_up, xs, h)
    return project_to_sum_box(h_up, n_up / 2.0)
