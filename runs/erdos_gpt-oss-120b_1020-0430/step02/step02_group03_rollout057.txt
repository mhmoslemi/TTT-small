def gradient_descent_max_corr(h, max_iters=2000, step_frac=0.1):
    n = len(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(corr)
        # Compute derivative for shift k0
        # Need arrays of shape n for d
        d = np.zeros(n)
        # for i where i + k0 < n
        idx1 = np.arange(0, n - k0)
        d[idx1] += (1 - h[idx1 + k0])
        # for i where i - k0 >= 0
        idx2 = np.arange(k0, n)
        d[idx2] -= h[idx2 - k0]
        # find max positive and min negative
        pos_idx = np.argmax(d)  # largest
        neg_idx = np.argmin(d)  # smallest
        if d[pos_idx] <= 1e-12 or d[neg_idx] >= -1e-12:
            break
        # transfer epsilon
        max_transfer = min(h[pos_idx], 1 - h[neg_idx])
        epsilon = step_frac * max_transfer
        if epsilon <= 1e-12:
            break
        h[pos_idx] -= epsilon
        h[neg_idx] += epsilon
    return h
```

    corr = np.correlate(h, 1 - h, mode='full')

    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max."""
    corr = np.correlate(x, 1 - x, mode="full")

    """Exact discrete C₅ for step function h (dx = 2/n)."""
