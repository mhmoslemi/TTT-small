def compute_c5_fft(h):
    n = h.shape[0]
    L = 1 << (2*n - 1).bit_length()   # next power of 2 >= 2*n-1
    # Use real FFT for speed
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)
    # conv length L; the full linear convolution of length 2*n-1 is first (2*n-1) entries
    corr = conv[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)
```

def simulated_annealing_binary(h0, rng, max_iters=500000, init_temp=0.2, decay=0.9999):
    h = h0.copy()
    n = h.shape[0]
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_temp
    for it in range(max_iters):
        # Choose random indices i (1) and j (0)
        # We maintain list of indices where h=1 and h=0 for quick choice.
        # Use numpy arrays of indices.
        i = rng.choice(one_idx) # one_idx list
        j = rng.choice(zero_idx)
        # Swap
        h[i] = 0
        h[j] = 1
        # Evaluate
        new_c = compute_c5_fft(h)
        # Accept if better or with probability
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / T):
            cur_c = new_c
            # Update best
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            # Update indices
            one_idx = one_idx[one_idx != i] # remove i
            zero_idx = zero_idx[zero_idx != j] # remove j
            # add swapped
            one_idx = np.append(one_idx, j)
            zero_idx = np.append(zero_idx, i)
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # decay temperature
        T *= decay
        # maybe early exit if T small and no improvement for many steps
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global parameters for the soft‑max surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (full correlation)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_fft(h):
    """Fast C5 using FFT (for binary vectors during SA)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.4):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the sum exactly (numeric drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def simulated_annealing_binary(h0, rng, time_limit, init_temp=0.2, decay=0.9999):
    """Binary SA using swaps; returns best vector and its C5."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = init_temp

    # maintain lists of indices for quick swaps
    ones = [int(i) for i in np.where(cur_h == 1)[0]]
    zeros = [int(i) for i in np.where(cur_h == 0)[0]]

    it = 0
    while time.time() < time_limit:
        # pick a random 1 and a random 0
        i_idx = rng.integers(0, len(ones))
        j_idx = rng.integers(0, len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0

        new_c = compute_c5_fft(cur_h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp((cur_c - new_c) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_c = new_c
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

        T *= decay
        it += 1
        # occasional temperature reset to avoid freezing too early
        if it % 50000 == 0:
            T = max(T, init_temp * 0.5)

    return best_h, best_c

            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=350)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial

    # Final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
