def random_feasible_vector(n, rng):

    # use simple acceptance: start with zeros, fill while maintaining bounds

def slsqp_optimize(h0, beta, max_iter=300):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust first element
        x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for step function h on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(2000):
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 0:
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    target = n / 2.0
    h = np.zeros(n)
    remaining = target
    order = rng.permutation(n)
    for i in order:
        if remaining <= 0:
            break
        max_val = min(1.0, remaining)
        val = rng.random() * max_val
        h[i] = val
        remaining -= val
    # distribute any tiny leftover
    while remaining > 1e-12:
        i = rng.integers(0, n)
        add = min(remaining, 1.0 - h[i])
        h[i] += add
        remaining -= add
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def targeted_step(h: np.ndarray, shift: int, rng: np.random.Generator,
                  max_transfer: float = 0.2) -> np.ndarray:
    """One deterministic move that tries to reduce the correlation at given shift."""
    n = h.size
    i_start = max(0, -shift)
    i_end = min(n, n - shift)
    if i_end <= i_start:
        return h
    h_i = h[i_start:i_end]
    h_j = h[i_start + shift:i_end + shift]

    # source: must have h_i>0 and neighbor (h_j) small (gradient large)
    src_mask = h_i > 1e-12
    if not np.any(src_mask):
        return h
    src_scores = np.where(src_mask, -h_j, -np.inf)  # prefer small neighbor
    src_idx_rel = int(np.argmax(src_scores))
    src_idx = i_start + src_idx_rel

    # target: must have h_i<1 and neighbor large (gradient small)
    tgt_mask = h_i < 1.0 - 1e-12
    if not np.any(tgt_mask):
        return h
    tgt_scores = np.where(tgt_mask, h_j, -np.inf)  # prefer large neighbor
    tgt_idx_rel = int(np.argmax(tgt_scores))
    tgt_idx = i_start + tgt_idx_rel

    if src_idx == tgt_idx:
        # fallback to random admissible pair
        src_candidates = np.where(src_mask)[0] + i_start
        tgt_candidates = np.where(tgt_mask)[0] + i_start
        if src_candidates.size == 0 or tgt_candidates.size == 0:
            return h
        src_idx = int(rng.choice(src_candidates))
        tgt_idx = int(rng.choice(tgt_candidates))

    max_delta = min(h[src_idx], 1.0 - h[tgt_idx], max_transfer)
    if max_delta <= 0.0:
        return h
    delta = max_delta * rng.random()
    nh = h.copy()
    nh[src_idx] -= delta
    nh[tgt_idx] += delta
    return nh


def softmax_obj(x: np.ndarray, beta: float) -> float:
    """Smooth log‑sum‑exp approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """Run SLSQP on the smooth surrogate for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        lambda x: softmax_obj(x, beta),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction to satisfy the sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust first entry safely
        x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x
