def smoothmax_and_gradient(h, beta):
   n = len(h)
   nfft = 1<< ((2*n-1).bit_length())
   H = np.fft.rfft(h, nfft)
   Hc = np.fft.rfft(1 - h, nfft)
   corr = np.fft.irfft(H * np.conj(Hc), nfft)[:2*n-1]
   C = corr * dx  # since dx = 2/n
   bc = beta * C
   bc_max = np.max(bc)
   w = np.exp(bc - bc_max)
   w_sum = np.sum(w)
   w /= w_sum
   # compute term1, term2 via convolution
   W = np.fft.rfft(w, nfft)
   conv1 = np.fft.irfft(W * H, nfft)[: (2*n-1) + n - 1]  # length 3n-2
   # conv2 with reversed w
   w_rev = w[::-1]
   W_rev = np.fft.rfft(w_rev, nfft)

   smooth_val = (bc_max + np.log(w_sum)) / beta  # approximation of max, but not needed

import time
import numpy as np

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while forcing sum(v)=target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric vector in [0,1] with Σh = n/2."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    h = np.concatenate([x, x[::-1]])
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a random amount of mass between two positions."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ------------------------------------------------------------
# smooth‑max objective and its gradient (FFT based)
# ------------------------------------------------------------
def smoothmax_and_gradient(h, beta):
    """Return smooth‑max value of C(k) and its gradient w.r.t. h."""
    n = h.shape[0]
    m_len = 2 * n - 1
    nfft = 1 << ((m_len).bit_length())          # next power of two
    H = np.fft.rfft(h, nfft)
    Hc = np.fft.rfft(1.0 - h, nfft)

    # correlation h * (1‑h)
    corr = np.fft.irfft(H * np.conj(Hc), nfft)[:m_len]
    dx = 2.0 / n
    C = corr * dx                               # C(k) values

    bc = beta * C
    bc_max = np.max(bc)
    w = np.exp(bc - bc_max)
    w_sum = np.sum(w)
    w /= w_sum                                   # soft‑max weights

    # convolutions needed for gradient
    W = np.fft.rfft(w, nfft)
    conv1 = np.fft.irfft(W * H, nfft)[: (m_len + n - 1)]
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, nfft)
    conv2 = np.fft.irfft(W_rev * H, nfft)[: (m_len + n - 1)]

    start = n - 1
    term1 = conv1[start:start + n]
    term2 = conv2[start:start + n]

    g_raw = np.ones(n) - (term1 + term2)        # ∇C_i before scaling
    grad = (2.0 / n) * g_raw

    smooth_val = (bc_max + np.log(w_sum)) / beta
    return smooth_val, grad


# ------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ------------------------------------------------------------
def adam_optimize(h0, betas, iters_per_beta, rng, deadline, best_sofar):
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.02

    for beta in betas:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return h
            t += 1
            _, grad = smoothmax_and_gradient(h, beta)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            h = h - step
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_sum(h, target)

            # enforce exact symmetry (helps the problem)
            h = (h + h[::-1]) * 0.5
            h = project_onto_box_sum(h, target)

            # occasional check – if we already beat the current best we can stop early
            if t % 600 == 0:
                cur = compute_c5(h)
                if cur < best_sofar:
                    best_sofar = cur
    return h

    # --------------------------------------------------------

    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
