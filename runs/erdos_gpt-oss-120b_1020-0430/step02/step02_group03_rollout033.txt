def compute_c5_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.fft(h, L)
    rev = (1 - h)[::-1]  # reversed
    fg = np.fft.fft(rev, L)
    conv = np.fft.ifft(fh * fg).real
    conv = conv[:2*n-1]  # linear conv
    return conv.max() * (2.0 / n)
```

# Helper functions (all top‑level, no closures)

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    # start from uniform 0.5 and perform a random walk that preserves the sum
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum (numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(start_h, rng, deadline, T0=0.02, Tf=1e-5, max_iters=1500):
    """Simple exponential‑cooling SA, returns best discovered point."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)    # exponential schedule
        new_h = perturb_h(cur_h, rng, scale=0.35)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c
