def c5(h):
    n = h.size
    L = 1 << (2*n-1).bit_length()   # next power of two
    # compute convolution of h and reversed(1-h)
    rev = (1 - h)[::-1]
    fft_len = L
    H = np.fft.rfft(h, fft_len)
    R = np.fft.rfft(rev, fft_len)
    conv = np.fft.irfft(H * R, fft_len)[:2*n-1]
    max_corr = conv.max()
    return max_corr * (2.0 / n)

def project_to_feasible(h, n, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # Determine indices that can move: if diff>0 need to reduce sum, so choose indices with h>0
        # else need to increase sum, choose indices with h<1
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # add/subtract to first free entry
        idx = np.where((residual > 0) & (h < 1))[0]
        if idx.size:
            i = idx[0]
            add = min(residual, 1 - h[i])
            h[i] += add
            residual -= add
        else:
            idx = np.where((residual < 0) & (h > 0))[0]
            if idx.size:
                i = idx[0]
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
    return np.clip(h, 0.0, 1.0)

def correlation_array_fft(h):

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-10, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        i = np.argmin(np.abs(h - (1 if residual > 0 else 0)))
        add = min(residual, 1 - h[i]) if residual > 0 else max(residual, -h[i])
        h[i] += add
    return np.clip(h, 0.0, 1.0)


def _upscale(h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, h)


def _corr_array_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.size
    L = 2 * n                     # >= 2n-1, convenient power of two not required
    H = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[: 2 * n - 1]
    return conv


def _c5_val(h: np.ndarray) -> float:
    """C₅ value for a feasible step function."""
    n = h.size
    max_corr = _corr_array_fft(h).max()
    return max_corr * (2.0 / n)


def _gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation at a given shift."""
    n = h.size
    term1 = np.ones_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        # (1‑h[i+shift]) if i+shift < n else 1
        term1[: n - shift] = 1.0 - h[shift:]
        # h[i‑shift] if i‑shift >= 0 else 0
        term2[shift:] = h[: n - shift]
    else:
        k = -shift
        term1[k:] = 1.0 - h[: n - k]
        term2[: n - k] = h[k:]
    return term1 - term2


def _hill_climb(start_h: np.ndarray, time_budget: float, target: float = 0.3808) -> tuple:
    """Simulated‑annealing style local search."""
    n = start_h.size
    h = start_h.copy()
    cur_val = _c5_val(h)
    best_h, best_val = h.copy(), cur_val
    start = time.time()
    max_iter = 800_000
    init_T = 0.018

    for it in range(1, max_iter + 1):
        if time.time() - start > time_budget:
            break

        # ---- propose a move -------------------------------------------------
        if np.random.rand() < 0.22:                      # occasional gradient step
            conv = _corr_array_fft(h)
            idx = int(conv.argmax())
            shift = -(idx - (n - 1))                     # sign correction
            grad = _gradient(h, shift)
            scale = np.max(np.abs(grad)) + 1e-12
            eta = 0.045 / scale
            new_h = h - eta * grad
        else:                                            # random mass transfer
            i, j = np.random.choice(n, 2, replace=False)
            max_pos = min(h[i], 1.0 - h[j])
            max_neg = -min(h[j], 1.0 - h[i])
            if max_pos == max_neg == 0:
                continue
            d = max_neg + (max_pos - max_neg) * np.random.rand()
            new_h = h.copy()
            new_h[i] -= d
            new_h[j] += d

        # ---- enforce feasibility -------------------------------------------
        new_h = _project_to_feasible(new_h, n)
        new_val = _c5_val(new_h)

        # ---- Metropolis acceptance -----------------------------------------
        T = init_T * (1.0 - it / max_iter)
        if new_val < cur_val or np.random.rand() < math.exp((cur_val - new_val) / max(T, 1e-12)):
            h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                if best_val < target:
                    break

    return best_h, best_val
