def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # get n from initial or random
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0,0.,1.)
        target = n/2.0
        enforce_sum(h0,target,rng)
    else:
        n = 180  # choose moderate small size
        h0 = random_feasible_vector(n, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # Phase: random restarts with quick_refine until time 0.3*budget
    while time.time() - start_time < budget_s * 0.3:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # Phase: upsample if time left, to higher resolution
    up_factors = [2, 2]  # upsample twice to get factor 4 (n*4)
    for factor in up_factors:
        if time.time() - start_time >= budget_s * 0.9:
            break
        # upsample
        h_up = upsample_h(best_h, factor=factor, rng=rng)
        n = h_up.size
        # deep refine
        h_up_ref, c_up_ref = deep_refine(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref

    # final cleanup
    n = best_h.size
    best_h = np.clip(best_h, 0., 1.)
    enforce_sum(best_h, n/2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Return a random vector with 0≤h_i≤1 and sum(h)=n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.3):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny repair for drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- local searches ----------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # schedule step reduction
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000,
                        init_temp=0.02, decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def binary_hill_climb(h, rng, max_iters=2000):

    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    return enforce_sum(h_up, target_up, rng)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 180
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = quick_refine(h0, rng)

    # ----- Phase 1: many random restarts (coarse) -----
    while time.time() - start < budget_s * 0.30:
        cand = random_feasible_vector(n0, rng)
        cand_h, cand_c = quick_refine(cand, rng)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ----- Phase 2: upscale and deep refinement -----
    n = best_h.size
    for factor in (2, 2):  # up to 4× total
        if time.time() - start >= budget_s * 0.85:
            break
        up_h = upsample_h(best_h, factor, rng)
        n = up_h.size
        refined_h, refined_c = deep_refine(up_h, rng)
        if refined_c < best_c:
            best_h, best_c = refined_h, refined_c

        # occasional random starts at the higher resolution
        while time.time() - start < budget_s * 0.90:
            rand_h = random_feasible_vector(n, rng)
            rand_ref, rand_c = quick_refine(rand_h, rng)
            if rand_c < best_c:
                best_h, best_c = rand_ref, rand_c

    # ----- final cleanup -----
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    # try a binary rounding of the continuous solution
    n = best_h.size
    half = n // 2
    bin_h = np.zeros_like(best_h)
    idx = np.argpartition(best_h, -half)[-half:]
    bin_h[idx] = 1.0
    bin_c = compute_c5(bin_h)
    if bin_c < final_c:
        best_h, final_c = bin_h, bin_c

    # optional binary local improvement
    if time.time() - start < budget_s * 0.95:
        bin_h2, bin_c2 = binary_hill_climb(best_h, rng, max_iters=2000)
        if bin_c2 < final_c:
            best_h, final_c = bin_h2, bin_c2

    # final exact feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
