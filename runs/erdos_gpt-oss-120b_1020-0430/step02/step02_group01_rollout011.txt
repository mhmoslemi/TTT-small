import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_feasible(h, target):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = 1.0 - h
        else:
            free = -h
        total = free.sum()
        if total == 0:
            break
        delta = diff * (free / total)
        h += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h / h.sum() * target
    return project_feasible(h, target)

def gradient_smooth_max(h, beta, rng, step, target, max_iter, eval_interval, best_h, best_c, start_time, budget):
    """Projected gradient descent on the smooth‑max surrogate."""
    n = h.size
    for it in range(max_iter):
        if time.time() - start_time > budget * 0.98:
            break

        # correlation and overlaps
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr[n - 1 :]               # size n, shift 0..n‑1

        # soft‑max weights (log‑sum‑exp trick)
        max_ov = np.max(overlaps)
        w = np.exp(beta * (overlaps - max_ov))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / n)
        else:
            w_norm = w / w_sum

        # gradient of the smooth max
        grad = np.zeros_like(h)

        # k = 0 term
        w0 = w_norm[0]
        if w0 != 0:
            grad += w0 * (1.0 - 2.0 * h)

        # k >= 1 terms
        for k in range(1, n):
            wk = w_norm[k]
            if wk == 0:
                continue
            nk = n - k
            grad[:nk] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:nk]

        # a little randomness in the step size
        step_eff = step * (1.0 + 0.1 * (rng.random() - 0.5))
        h = h - step_eff * grad
        h = project_feasible(h, target)

        # occasional exact evaluation
        if (it % eval_interval) == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        # tiny random perturbation to escape plateaus
        if (it % 500) == 0 and rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = min(h[i], 1.0 - h[j]) * (rng.random() - 0.5) * 0.2
                h[i] -= eps
                h[j] += eps
                h = project_feasible(h, target)

    return h, best_h, best_c

def hill_climb(h0, rng, target, max_iters=2000, step=0.2):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h0.copy()
    best = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = project_feasible(new_h, target)
        c = compute_c5(new_h)
        if c < best:
            best = c
            h = new_h
    return h, best

        h0 = random_feasible(n, rng)
        # a short gradient run
        h0, best_h, best_c = gradient_smooth_max(
            h0, 30, rng, 0.12, target,
            max_iter=1500,
            eval_interval=75,
            best_h=best_h,
            best_c=best_c,
            start_time=start,
            budget=budget_s,
        )
        # brief hill‑climb
        h0, c0 = hill_climb(best_h, rng, target, max_iters=2000, step=0.1)
        if c0 < best_c:
            best_h, best_c = h0, c0

    # -------------------------------------------------------------
    # final feasibility enforcement and exact evaluation
    best_h = project_feasible(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
