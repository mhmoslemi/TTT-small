def binary_hill_climb(h, rng, max_iters=2000, temp_start=1e-3, temp_end=1e-6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    h_cur = h.copy()
    temp_decay = (temp_end / temp_start) ** (1.0 / max_iters) if max_iters>0 else 1.0
    temp = temp_start
    for it in range(max_iters):
        # random swap
        ones = np.flatnonzero(h_cur)
        zeros = np.flatnonzero(1 - h_cur)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_cur[i] = 0; h_cur[j] = 1
        c = compute_c5(h_cur)
        if c < best_c or rng.random() < np.exp(-(c - best_c) / temp):
            # accept move, update best if better
            if c < best_c:
                best_c = c
                best_h = h_cur.copy()
        else:
            # revert
            h_cur[i] = 1; h_cur[j] = 0
        temp *= temp_decay
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# globals for soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def binary_hill_climb(h, rng, max_iters=2000, temp_start=1e-3, temp_end=1e-6):
    """Swap‑based hill‑climb on binary vectors with simulated annealing."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    temp = temp_start
    decay = (temp_end / temp_start) ** (1.0 / max_iters)
    for it in range(max_iters):
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        if c < best_c or rng.random() < np.exp(-(c - best_c) / max(temp, 1e-12)):
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        temp *= decay
    return best_h, best_c

    candidate_ns = [96, 108, 120, 144, 168, 180, 200]
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.94:
            break

        # ----- binary search phase -----
        binary_attempts = 20 if n <= 120 else 10
        local_best_h = None
        local_best_c = np.inf
        for _ in range(binary_attempts):
            if time.time() - start_time > budget_s * 0.94:
                break
            hb = np.zeros(n)
            hb[rng.choice(n, n // 2, replace=False)] = 1.0
            hb_opt, cb = binary_hill_climb(hb, rng, max_iters=1500)
            if cb < local_best_c:
                local_best_c = cb
                local_best_h = hb_opt.copy()

        # ----- continuous random start phase -----
        cont_attempts = 5 if n <= 120 else 3
        for _ in range(cont_attempts):
            if time.time() - start_time > budget_s * 0.94:
                break
            h0 = random_feasible_vector(n, rng)
            h_opt, c_opt = hill_climb(h0, rng, max_iters=1500, start_step=0.5)
            if c_opt < local_best_c:
                local_best_c = c_opt
                local_best_h = h_opt.copy()

        # ----- refinement of the best candidate for this n -----
        if local_best_h is not None:
            # if this candidate beats global best, record it early
            if local_best_c < best_c:
                best_h, best_c, best_n = local_best_h.copy(), local_best_c, n

            # SLSQP passes with increasing β
            h_ref = local_best_h.copy()
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                h_ref = slsqp_optimize(h_ref, beta, rng, max_iter=300)
                c_ref = compute_c5(h_ref)
                if c_ref < local_best_c:
                    local_best_c = c_ref
                    local_best_h = h_ref.copy()
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref.copy(), c_ref, n

            # Gradient refinement
            if time.time() - start_time < budget_s * 0.97:
                h_ref = gradient_refine(local_best_h, rng, step=0.02, max_iter=6)
                c_ref = compute_c5(h_ref)
                if c_ref < local_best_c:
                    local_best_c = c_ref
                    local_best_h = h_ref.copy()
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref.copy(), c_ref, n

            # Final short hill‑climb
            if time.time() - start_time < budget_s * 0.98:
                h_ref, c_ref = hill_climb(local_best_h, rng, max_iters=1200, start_step=0.3)
                if c_ref < local_best_c:
                    local_best_c = c_ref
                    local_best_h = h_ref.copy()
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref.copy(), c_ref, n

    # ------------------------------------------------------------------
    # Optional up‑sampling refinement
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        if best_n * 2 <= 200:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            # local refinement on the up‑sampled vector
            h_up, _ = hill_climb(h_up, rng, max_iters=1500, start_step=0.3)
            for beta in (5.0, 20.0, 80.0):
                if time.time() - start_time > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # Final feasibility enforcement
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
