def binary_initial(n, rng):
    # half ones
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5_binary(h): returns C5.

def anneal(h, rng, max_iter, start_temp, end_temp):
    # temperature schedule log or linear.
    best_h = h.copy()
    best_c = compute_c5_binary(best_h)
    cur_c = best_c
    n = h.size
    for it in range(max_iter):
        # generate neighbor by swapping a random 1 and 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5_binary(h)
        # acceptance probability
        # temperature depends on iteration
        T = start_temp + (end_temp - start_temp) * (it / max_iter)
        # Actually we want decreasing, start high, end low
        T = max(end_temp, start_temp * (0.99**it)) ; many forms.
        # Accept if better, else accept with prob exp(-(new_c - cur_c) / T)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # if T>0:
            if T > 0 and rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Simple hill‑climb that accepts only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Iterative gradient step on the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k]) for i< n-k
            g[k:] -= h[:upper]                # - h[i] for i>=k
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # exact sum correction
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run one SLSQP optimisation on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (small correction)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def binary_anneal_start(n, rng, max_iter=5000, init_temp=0.005, final_temp=1e-6):
    """Simple simulated‑annealing binary start (half ones)."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(max_iter):
        T = max(final_temp, init_temp * (1.0 - it / max_iter))
        i, j = rng.integers(0, n, size=2)
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]      # swap
        new_c = compute_c5(cur_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # revert
    return best_h, best_c


def generate_start_h(n, rng):
    """Diverse initial vectors (binary anneal, binary, or continuous)."""
    r = rng.random()
    if r < 0.35:
        # binary annealed start
        max_iter = int(3000 + n * 8)
        h, _ = binary_anneal_start(n, rng, max_iter=max_iter)
        return h.astype(float)
    elif r < 0.6:
        # plain binary half‑ones
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        return h
    else:
        # continuous random feasible
        return random_feasible_vector(n, rng)


def optimise_at_n(n, rng, time_limit):
    """Run a series of stochastic searches for a given resolution n."""
    best_local_h = None
    best_local_c = np.inf
    attempts = 0
    # adaptive attempts: more for small n, less for larger n
    max_attempts = 30 if n <= 96 else 20
    while attempts < max_attempts and time.time() < time_limit:
        attempts += 1
        start_h = generate_start_h(n, rng)

        # 1) hill‑climb (coarse)
        h, c = hill_climb(start_h, rng, max_iters=2500, start_step=0.5)
        if c < best_local_c:
            best_local_h, best_local_c = h.copy(), c

        # 2) gradient refinement
        h = gradient_refine(h, rng, step=0.02, max_iter=6)
        c = compute_c5(h)
        if c < best_local_c:
            best_local_h, best_local_c = h.copy(), c

        # 3) SLSQP passes (soft‑max surrogate)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
            if time.time() >= time_limit:
                break
            max_it = 250 if beta < 100 else 500
            h = slsqp_optimize(h, beta, rng, max_iter=max_it)
            c = compute_c5(h)
            if c < best_local_c:
                best_local_h, best_local_c = h.copy(), c

        # 4) final short hill‑climb
        if time.time() < time_limit:
            h, c = hill_climb(h, rng, max_iters=1500, start_step=0.3)
            if c < best_local_c:
                best_local_h, best_local_c = h.copy(), c
    return best_local_h, best_local_c


def upsample_and_refine(h, rng, time_limit):
    """Upsample by 2 (repeat) and run a brief refinement."""
    n = h.size
    if n * 2 > 240 or time.time() >= time_limit:
        return h, compute_c5(h)
    h_up = np.repeat(h, 2)
    h_up, c = hill_climb(h_up, rng, max_iters=2000, start_step=0.3)
    for beta in (5.0, 20.0, 80.0):
        if time.time() >= time_limit:
            break
        h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
    c = compute_c5(h_up)
    return h_up, c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Use supplied parent solution if present
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target0 = n0 / 2.0
        diff = target0 - h0.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n0)
                if diff > 0:
                    add = min(diff, 1.0 - h0[i])
                else:
                    add = max(diff, -h0[i])
                h0[i] += add
                diff = target0 - h0.sum()
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # Candidate discretisations (focus on even sizes for binary work)
    candidate_ns = [48, 60, 72, 84, 96, 108, 120, 144, 168, 192]
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    # Main optimisation loop
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.94:
            break
        # allocate a slice of the remaining budget for this n
        time_for_n = start_time + budget_s * (0.94) - time.time()
        h_n, c_n = optimise_at_n(n, rng, time_for_n)
        if h_n is not None and c_n < best_c:
            best_h, best_c, best_n = h_n.copy(), c_n, n

    # Optional up‑sampling refinement if time allows
    if best_h is not None and time.time() - start_time < budget_s * 0.98:
        h_up, c_up = upsample_and_refine(best_h, rng, start_time + budget_s * 0.99)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, h_up.size

    # Final feasibility enforcement (clip + exact sum)
    if best_h is None:
        best_n = 48
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
