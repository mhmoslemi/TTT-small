import time, numpy as np

def full_from_half(h_half):
    return np.concatenate([h_half, 1 - h_half[::-1]])

def compute_c5(full_h):
    corr = np.correlate(full_h, 1 - full_h, mode='full')
    return np.max(corr) * (2.0 / full_h.size)

def compute_c5_half(h_half):
    return compute_c5(full_from_half(h_half))

def random_half(m, rng):
    return rng.random(m)

def mutate_transfer(h, rng, scale=0.2):
    # same as perturb but preserve sum (transfer)
    m = h.shape[0]
    i = rng.integers(m)
    j = rng.integers(m)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def tournament_select(pop, scores, rng):
    i, j = rng.integers(len(pop), size=2)
    return pop[i] if scores[i] < scores[j] else pop[j]

def hill_climb(full_h, rng, max_iters=3000, start_step=0.3):
    # use perturb_h as defined earlier (transfer)
    # compute c5 each iteration.
    def compute(full):
        return compute_c5(full)
    h = full_h.copy()
    best_h = h.copy()
    best_c = compute(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)  # we need perturb_h defined for full vector
        new_c = compute(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay_interval ==0:
            step = max(step*0.7, 1e-4)
    return best_h, best_c

    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:

    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale

    new_h[i] -= eps
    new_h[j] += eps
    return new_h

        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    new_vec[j] += eps

CURRENT_BETA = None


def sum_eq_constraint(x):
    """Sum constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate of max_k Σ_i x_i (1-x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h using SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5_full(h):
    """Exact C₅ for a full‑length step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def full_from_half(h_half):
    """Create a symmetric full vector from a half‑vector (h(x+1)=1‑h(x))."""
    return np.concatenate([h_half, 1 - h_half[::-1]])


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j while staying in [0,1] (preserves sum)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(full_h, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = full_h.copy()
    best_h = h.copy()
    best_c = compute_c5_full(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5_full(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def mutate_transfer(vec, rng, scale=0.2):
    """Same as perturb_h but works on half‑vectors."""
    n = vec.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return vec.copy()
    max_eps = min(vec[i], 1.0 - vec[j])
    if max_eps <= 0:
        return vec.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_vec = vec.copy()
    new_vec[i] -= eps
    new_vec[j] += eps
    return new_vec


def tournament_select(pop, scores, rng):
    """Return the better of two randomly chosen individuals."""
    i, j = rng.integers(len(pop), size=2)
    return pop[i] if scores[i] < scores[j] else pop[j]


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -----------------------------------------------------------------
    # Determine size and possibly load the parent construction
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        init_full = np.array(initial_h_values, dtype=float)
        n = init_full.size
    else:
        n = 180
        init_full = None
    if n % 2 != 0:               # enforce even length for symmetry
        n -= 1
    m = n // 2                    # length of the half‑vector

    # -----------------------------------------------------------------
    # Helper to compute C₅ from a half‑vector
    # -----------------------------------------------------------------
    def c5_half(v_half):
        return compute_c5_full(full_from_half(v_half))

    # -----------------------------------------------------------------
    # Initialise best candidates from the parent (if any)
    # -----------------------------------------------------------------
    best_full = None
    best_c = np.inf
    if init_full is not None:
        c0 = compute_c5_full(init_full)
        best_full = init_full.copy()
        best_c = c0
        # symmetric version built from the first half of the parent
        sym_full = full_from_half(init_full[:m])
        c_sym = compute_c5_full(sym_full)
        if c_sym < best_c:
            best_c = c_sym
            best_full = sym_full.copy()

    # -----------------------------------------------------------------
    # Evolutionary search on half‑vectors (enforces h(x+1)=1‑h(x))
    # -----------------------------------------------------------------
    pop_size = 80
    elite_size = 5
    mut_prob = 0.3
    mut_scale = 0.25

    population = []
    scores = []

    # seed the population with the (possibly symmetric) best half‑vector
    if best_full is not None:
        population.append(best_full[:m].copy())
        scores.append(c5_half(population[-1]))

    while len(population) < pop_size:
        h_half = rng.random(m)                     # uniform [0,1]
        population.append(h_half)
        scores.append(c5_half(h_half))

    # track the globally best half‑vector
    best_idx = int(np.argmin(scores))
    global_best_half = population[best_idx].copy()
    global_best_c = scores[best_idx]

    # main EA loop
    while time.time() - start_time < budget_s * 0.90:
        # elitism
        order = np.argsort(scores)
        new_pop = [population[i].copy() for i in order[:elite_size]]
        new_scores = [scores[i] for i in order[:elite_size]]

        # generate offspring
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, scores, rng)
            p2 = tournament_select(population, scores, rng)
            mask = rng.random(m) < 0.5
            child = np.where(mask, p1, p2)
            if rng.random() < mut_prob:
                child = mutate_transfer(child, rng, scale=mut_scale)
            child = np.clip(child, 0.0, 1.0)
            new_pop.append(child)
            new_scores.append(c5_half(child))

        population = new_pop
        scores = new_scores

        # update global best
        cur_best_idx = int(np.argmin(scores))
        if scores[cur_best_idx] < global_best_c:
            global_best_c = scores[cur_best_idx]
            global_best_half = population[cur_best_idx].copy()

        # occasional random immigrant to keep diversity
        if rng.random() < 0.2:
            worst_idx = int(np.argmax(scores))
            rnd = rng.random(m)
            population[worst_idx] = rnd
            scores[worst_idx] = c5_half(rnd)

    # -----------------------------------------------------------------
    # Build the full‑length candidate from the best half‑vector
    # -----------------------------------------------------------------
    best_half_full = full_from_half(global_best_half)

    # Choose the better of the symmetric and the (possibly non‑symmetric) parent
    if best_full is not None and best_c < global_best_c:
        final_full = best_full
        final_c = best_c
    else:
        final_full = best_half_full
        final_c = global_best_c

    # -----------------------------------------------------------------
    # Local hill‑climb refinement (if time permits)
    # -----------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.96:
        hc_h, hc_c = hill_climb(final_full, rng, max_iters=4000, start_step=0.4)
        if hc_c < final_c:
            final_full, final_c = hc_h, hc_c

    # -----------------------------------------------------------------
    # Smooth‑surrogate SLSQP refinement (few β values)
    # -----------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.99:
        target = final_full.size / 2.0
        # tiny sum correction before SLSQP (should already be correct)
        diff = target - final_full.sum()
        if abs(diff) > 1e-12:
            n_full = final_full.size
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n_full)
                add = min(diff, 1.0 - final_full[i]) if diff > 0 else max(diff, -final_full[i])
                final_full[i] += add
                diff = target - final_full.sum()
        final_full = np.clip(final_full, 0.0, 1.0)

        for beta in (20.0, 80.0, 200.0, 500.0):
            if time.time() - start_time > budget_s:
                break
            try:
                final_full = slsqp_optimize(final_full, beta, max_iter=400)
            except Exception:
                continue
            final_c = compute_c5_full(final_full)

    # -----------------------------------------------------------------
    # Final clean‑up (ensure feasibility) and return
    # -----------------------------------------------------------------
    final_full = np.clip(final_full, 0.0, 1.0)
    target = final_full.size / 2.0
    diff = target - final_full.sum()
    if abs(diff) > 1e-12:
        n_full = final_full.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n_full)
            add = min(diff, 1.0 - final_full[i]) if diff > 0 else max(diff, -final_full[i])
            final_full[i] += add
            diff = target - final_full.sum()

    final_c = compute_c5_full(final_full)
    return final_full.tolist(), float(final_c), int(final_full.size)
