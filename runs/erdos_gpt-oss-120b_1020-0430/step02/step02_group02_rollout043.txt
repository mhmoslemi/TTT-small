def project_onto_bounds_sum(v, target, tol=1e-9, max_iter=100):
    lo = -1.0
    hi = 1.0
    # Actually λ can be as large as max(v), but we can compute bounds: 
    # For λ large negative, v_i - λ > 1 -> clip to 1, giving sum = n.
    # For λ large positive, v_i - λ < 0 -> clip to 0, giving sum = 0.
    # So we can binary search λ in [ -max(v), max(v) ] maybe.
    lo = -max(v) - 1
    hi = max(v) + 1
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # Guarantee sum close to target; adjust minor residual via adding to entries not at bounds
    residual = target - x.sum()
    if abs(residual) > 1e-8:
        # distribute residual uniformly to free entries
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt > 0:
            x[free] += residual / cnt
            # clip again to ensure bounds
            x = np.clip(x, 0.0, 1.0)
    return x

import time, numpy as np

def _c5_from_h(h):
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def _construct_h(n, a):
    # a integer number of zeros and ones at ends
    a = int(a)
    ramp_len = n - 2*a
    if ramp_len < 0:
        raise ValueError
    if ramp_len == 0:
        # block function: zeros then ones
        h = np.concatenate([np.zeros(a), np.ones(a)])
    else:
        ramp = (np.arange(ramp_len) + 0.5) / ramp_len
        h = np.concatenate([np.zeros(a), ramp, np.ones(a)])
    return h

def _symmetrize(h):
    # enforce h[i] + h[-i-1] = 1
    return 0.5 * (h + (1 - h[::-1]))

def _hill_climb_symmetric(h, max_steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    while max_steps > 0 and time.time() < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # compute max epsilon respecting bounds before symmetrization
        # we will just propose change on the full h, then symmetrize
        eps = np.random.rand() * 0.1  # small step size
        cand = best_h.copy()
        # reduce at i, increase at j
        # ensure after change values stay in [0,1]; clip after symmetrization
        cand[i] = np.clip(cand[i] - eps, 0.0, 1.0)
        cand[j] = np.clip(cand[j] + eps, 0.0, 1.0)
        cand = _symmetrize(cand)
        cand = np.clip(cand, 0.0, 1.0)
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
        max_steps -= 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    best_c = float('inf')
    best_h = None
    best_n = None

    n_list = [80,96,112,128,160,192,224,256,320,384,512]
    for n in n_list:
        if time_left() < 2:
            break
        max_a = n // 2
        # precompute ramp lengths for each a
        for a in range(0, max_a + 1):
            # optional early break if current best less than achievable lower bound? Not needed.
            ramp_len = n - 2 * a
            if ramp_len == 0:
                # all zeros then ones, C5 = 1, skip
                continue
            ramp = (np.arange(ramp_len) + 0.5) / ramp_len
            h = np.concatenate([np.zeros(a), ramp, np.ones(a)])
            c = _c5_from_h(h)
            if c < best_c:
                best_c = c
                best_h = h
                best_n = n
                # if already below target, can break outer loops maybe
                if best_c < 0.3808:
                    break
        if best_c < 0.3808:
            break

    # Local search: hill climb with symmetrization
    if best_h is not None and time_left() > 5:
        # set step limit proportional to remaining time
        steps = int(min(20000, max(1000, time_left() * 30)))  # e.g., 30 per second
        limit = start + budget_s - 0.1
        best_h, best_c = _hill_climb_symmetric(best_h, steps, limit)

    # Ensure final h is list
    if best_h is None:
        n = 128
        best_h = (np.arange(n) + 0.5) / n
        best_c = _c5_from_h(best_h)
        best_n = n

import time
import numpy as np

# --------------------------------------------------------------
# Core helpers
# --------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a given step vector h."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _symmetrize(h: np.ndarray) -> np.ndarray:
    """Force h_i + h_{n-1-i} = 1 (preserves sum = n/2)."""
    return 0.5 * (h + (1.0 - h[::-1]))


def _hill_climb_symmetric(h0: np.ndarray, max_steps: int, time_limit: float):
    """
    Small random exchanges with symmetrisation.
    Returns the improved (h, C5) pair.
    """
    n = h0.size
    best_h = h0.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount we can move from i to j
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0.0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = _symmetrize(cand)
        cand = np.clip(cand, 0.0, 1.0)
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


def _construct_h(n: int, a: int) -> np.ndarray:
    """
    Build a symmetric step function:
    - a zeros on the left,
    - linear ramp of length n-2a,
    - a ones on the right.
    """
    a = int(a)
    ramp_len = n - 2 * a
    if ramp_len <= 0:
        # degenerate block (won't be optimal)
        return np.concatenate([np.zeros(a), np.ones(a)])
    ramp = (np.arange(ramp_len) + 0.5) / ramp_len
    return np.concatenate([np.zeros(a), ramp, np.ones(a)])

# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h achieving a low C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # baseline from parent construction if provided
    try:
        init_h = np.array(initial_h_values, dtype=float)
        init_c = _c5_from_h(init_h)
    except NameError:
        init_h = None
        init_c = float("inf")

    best_h = init_h
    best_c = init_c
    best_n = init_h.size if init_h is not None else None

    # Grid sizes to explore (larger sizes give finer resolution)
    n_candidates = [80, 96, 112, 128, 160, 192, 224, 256, 320, 384, 512]

    for n in n_candidates:
        if time_left() < 5.0:
            break

        max_a = n // 2
        # Exhaustive scan over plateau length a
        for a in range(0, max_a + 1):
            h = _construct_h(n, a)
            c = _c5_from_h(h)
            if c < best_c:
                best_h, best_c, best_n = h, c, n
                if best_c < 0.3808:
                    break
        if best_c < 0.3808:
            break

        # Local refinement with a cheap hill‑climb
        if best_h is not None and time_left() > 5.0:
            # allocate a few thousand steps proportional to remaining time
            remaining = time_left()
