def simulated_annealing(h, rng, time_limit):
    start=time.time()
    n=len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    # parameters
    init_T = 0.01
    final_T = 1e-5
    max_iter = 1000000  # unlimited, stop by time
    it = 0
    while time.time() - start < time_limit:
        it += 1
        progress = it / max_iter
        T = init_T * (1 - progress) + final_T * progress  # linear cooling
        # also scale epsilon
        scale = 0.5 * (1 - progress) + 0.02 * progress  # reduce over time
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # direction already i->j
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        eps = rng.random() * max_transfer * scale
        if eps <= 0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c

            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # optionally time break
    return best_h, best_c

def genetic_optimize(n, rng, time_budget, pop_size=30, elite_frac=0.2, mut_rate=0.3):
    start = time.time()
    # init population
    pop = [random_feasible_vector(n, rng) for _ in range(pop_size)]
    scores = [compute_c5(ind) for ind in pop]
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_c = scores[best_idx]
    while time.time() - start < time_budget:
        # sort population
        idx_sorted = np.argsort(scores)
        elite_num = max(1, int(pop_size * elite_frac))
        elites = [pop[i] for i in idx_sorted[:elite_num]]
        # generate new pop
        new_pop = elites.copy()
        # fill rest
        while len(new_pop) < pop_size:
            # select two parents
            parents = rng.choice(elites, size=2, replace=False)
            # crossover: convex combination
            alpha = rng.random()
            child = alpha * parents[0] + (1 - alpha) * parents[1]
            # mutation: perturb_h several times maybe
            if rng.random() < mut_rate:
                child = perturb_h(child, rng, scale=0.3)
            # repair sum, clip
            child = np.clip(child, 0.0, 1.0)
            target = n/2.0
            diff = target - child.sum()
            if abs(diff) > 1e-12:
                for _ in range(200):
                    i = rng.integers(0,n)
                    if diff > 0:
                        add = min(diff, 1.0 - child[i])
                    else:
                        add = max(diff, -child[i])
                    child[i] += add
                    diff = target - child.sum()
            new_pop.append(child)
        pop = new_pop
        scores = [compute_c5(ind) for ind in pop]
        # update best
        cur_best_idx = np.argmin(scores)
        cur_best_c = scores[cur_best_idx]
        if cur_best_c < best_c:
            best_c = cur_best_c
            best_h = pop[cur_best_idx].copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----- global placeholders for SLSQP surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max of the correlation via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)

        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def simulated_annealing(h, rng, time_budget):
    """Simple SA that respects the sum constraint by moving mass between two coordinates."""
    start = time.time()
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    max_T, min_T = 0.02, 1e-6
    start_scale, final_scale = 0.5, 0.005

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max_T * (1 - frac) + min_T * frac
        scale = start_scale * (1 - frac) + final_scale * frac

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            continue
        eps = rng.random() * max_transfer * scale
        if eps <= 0.0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c


def genetic_optimize(n, rng, time_budget, pop_size=30, elite_frac=0.2, mut_rate=0.3):
    """Very lightweight GA on the feasible simplex."""
    start = time.time()
    # initial population
    pop = [random_feasible_vector(n, rng) for _ in range(pop_size)]
    scores = [compute_c5(ind) for ind in pop]
    best_idx = int(np.argmin(scores))
    best_h, best_c = pop[best_idx].copy(), scores[best_idx]

    elite_num = max(1, int(pop_size * elite_frac))

    while time.time() - start < time_budget:
        # sort by fitness (lower C5 is better)
        order = np.argsort(scores)
        elites = [pop[i] for i in order[:elite_num]]

        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            p1, p2 = rng.choice(elites, size=2, replace=False)
            alpha = rng.random()
            child = alpha * p1 + (1.0 - alpha) * p2
            if rng.random() < mut_rate:
                child = perturb_h(child, rng, scale=0.3)
            child = np.clip(child, 0.0, 1.0)

            target = n / 2.0
            diff = target - child.sum()
            if abs(diff) > 1e-12:
                for _ in range(200):
                    i = rng.integers(0, n)
                    if diff > 0:
                        add = min(diff, 1.0 - child[i])
                    else:
                        add = max(diff, -child[i])
                    child[i] += add
                    diff = target - child.sum()
                    if abs(diff) < 1e-12:
                        break
            new_pop.append(child)

        pop = new_pop
        scores = [compute_c5(ind) for ind in pop]
        cur_best = int(np.argmin(scores))
        cur_c = scores[cur_best]
        if cur_c < best_c:
            best_c = cur_c
            best_h = pop[cur_best].copy()
    return best_h, best_c
