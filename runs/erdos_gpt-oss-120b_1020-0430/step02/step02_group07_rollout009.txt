    start = time.time()
    while time.time() - start < time_limit:
        # compute O array (full correlation)
        O = np.correlate(h, 1 - h, mode='full')
        idx = int(np.argmax(O))  # index of max O
        lag = idx - (n - 1)
        # find list of candidate positions:
        if lag >= 0:

            partner = inds - lag
        else:
            lag_abs = -lag
            inds = np.arange(0, n - lag_abs)
            partner = inds + lag_abs

def greedy_swap_search(h, rng, time_limit):
    # h is binary (0/1 floats)
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    # maintain O array each iteration
    while time.time() - start < time_limit:
        # compute O
        O = np.correlate(h, 1 - h, mode='full')
        idx = int(np.argmax(O))
        lag = idx - (n - 1)  # integer lag
        # compute indices where swap can improve
        if lag >= 0:
            inds = np.arange(lag, n)
            partner = inds - lag
        else:
            lag_abs = -lag
            inds = np.arange(0, n - lag_abs)
            partner = inds + lag_abs
        # mask arrays for h at inds, partner
        hi = h[inds]
        hp = h[partner]
        # define bad_one mask: hi == 1 and hp == 0
        bad_one_mask = (hi > 0.5) & (hp < 0.5)
        # define bad_zero mask: hi == 0 and hp == 1
        bad_zero_mask = (hi < 0.5) & (hp > 0.5)
        if not np.any(bad_one_mask) or not np.any(bad_zero_mask):
            # cannot improve via single swap for this lag; random perturb
            # attempt random swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if h[i] == h[j]:
                continue
            # swap
            h[i], h[j] = h[j], h[i]
            new_c = compute_c5(h)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            else:
                # revert
                h[i], h[j] = h[j], h[i]

        # pick random from each mask

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- basic utilities --------------------

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Create a binary vector of length n with exactly n/2 ones (rounded down)."""
    target = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=target, replace=False)
    h[idx] = 1.0
    return h


# -------------------- local search helpers --------------------

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    no_improve = 0
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
            no_improve = 0
        else:
            no_improve += 1
        # step size schedule
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
        if no_improve > max_iters // 4:
            break
    return best_h, best_c


# -------------------- soft‑max surrogate for SLSQP --------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # repair exact sum (numeric drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# -------------------- up‑sampling --------------------

def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


# -------------------- targeted greedy swap --------------------

def greedy_swap_search(h, rng, time_limit):
    """
    Greedy improvement for binary vectors:
    repeatedly swap a (1,0) pair that hurts the worst lag.
    """
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    no_improve = 0
    max_no_improve = 2000

    while time.time() - start < time_limit:
        # correlation (unnormalised)
        O = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(O))
        lag = idx - (n - 1)  # integer lag

        if lag >= 0:
            inds = np.arange(lag, n)
            partner = inds - lag
        else:
            lag_abs = -lag
            inds = np.arange(0, n - lag_abs)
            partner = inds + lag_abs

        hi = h[inds]
        hp = h[partner]

        bad_one_mask = (hi > 0.5) & (hp < 0.5)   # contributes +1 to O
        bad_zero_mask = (hi < 0.5) & (hp > 0.5)  # would contribute +1 if swapped

        if np.any(bad_one_mask) and np.any(bad_zero_mask):
            i = rng.choice(inds[bad_one_mask])
            j = rng.choice(inds[bad_zero_mask])
            h[i], h[j] = h[j], h[i]
            new_c = compute_c5(h)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                # revert – the swap did not improve overall max
                h[i], h[j] = h[j], h[i]
                no_improve += 1
        else:
            # fallback: try a random opposite‑value swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or h[i] == h[j]:
                no_improve += 1
                continue
            h[i], h[j] = h[j], h[i]
            new_c = compute_c5(h)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1

        if no_improve > max_no_improve:
            break
    return best_h, best_c


# -------------------- main optimisation routine --------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial binary vector ---------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).clip(0.0, 1.0)
        n = h0.size
        target = n / 2.0
        # simple binary rounding with exact sum
        bin_h = (h0 >= 0.5).astype(float)
        cur = bin_h.sum()
        diff = int(target - cur)
        if diff != 0:
            zeros = np.where(bin_h == 0)[0]
            ones = np.where(bin_h == 1)[0]
            if diff > 0:
                idx = rng.choice(zeros, size=diff, replace=False)
                bin_h[idx] = 1.0
            else:
                idx = rng.choice(ones, size=-diff, replace=False)
                bin_h[idx] = 0.0
        h_start = bin_h
    else:
        n = 90
        h_start = random_binary_vector(n, rng)

    # ----- Phase 1 : many random restarts with greedy swaps -----
    best_h, best_c = greedy_swap_search(h_start, rng, time_limit=0.2)
    time_elapsed = time.time() - start_time
    phase1_budget = budget_s * 0.4
    while time.time() - start_time < phase1_budget:
        h_rand = random_binary_vector(h_start.size, rng)
        h_opt, c_opt = greedy_swap_search(h_rand, rng, time_limit=0.15)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ----- Phase 2 : up‑sample and sophisticated refinement -----
    time_elapsed = time.time() - start_time
    if time_elapsed < budget_s * 0.8:
        # up‑sample once (factor 2)
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        best_h_up = h_up
        best_c_up = compute_c5(best_h_up)

        # SLSQP ladder with increasing β
        betas = (5.0, 20.0, 80.0, 200.0, 500.0, 2000.0)
        for beta in betas:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                trial = slsqp_optimize(best_h_up, beta=beta, max_iter=300)
            except Exception:
                continue
            c_trial = compute_c5(trial)
            if c_trial < best_c_up:
                best_h_up, best_c_up = trial, c_trial

        # aggressive hill‑climb on the continuous result
        if time.time() - start_time < budget_s * 0.95:
            h_hc, c_hc = hill_climb(best_h_up, rng, max_iters=4000, start_step=0.4)
            if c_hc < best_c_up:
                best_h_up, best_c_up = h_hc, c_hc

        # binary greedy refinement on a rounded version
        if time.time() - start_time < budget_s * 0.95:
            bin_up = (best_h_up >= 0.5).astype(float)
            # enforce exact sum
            diff = int(n_up / 2 - bin_up.sum())
            if diff != 0:
                zeros = np.where(bin_up == 0)[0]
                ones = np.where(bin_up == 1)[0]
                if diff > 0:
                    idx = rng.choice(zeros, size=diff, replace=False)
                    bin_up[idx] = 1.0
                else:
                    idx = rng.choice(ones, size=-diff, replace=False)
                    bin_up[idx] = 0.0
            # short greedy run
            bin_up_opt, bin_up_c = greedy_swap_search(bin_up, rng, time_limit=budget_s * 0.03)
            if bin_up_c < best_c_up:
                best_h_up, best_c_up = bin_up_opt, bin_up_c

        best_h, best_c = best_h_up, best_c_up

    # ----- Final clean‑up ----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
