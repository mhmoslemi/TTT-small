def anneal(h, rng, max_iters, init_temp, cooling_rate):
   cur = h; cur_c = compute_c5(cur)
   best = cur; best_c = cur_c
   t = init_temp
   for it in range(max_iters):
        new_h = perturb_h(cur, rng, scale=0.5)
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / t):
            cur, cur_c = new_h, new_c
            if new_c < best_c:
                best, best_c = new_h, new_c
        t *= cooling_rate
   return best, best_c

def compute_c5(h):
    n = h.size
    # zero pad to next power of 2 for speed
    n_fft = 1 << (int(np.ceil(np.log2(2*n-1))))
    fa = np.fft.rfft(h, n=n_fft)
    fb = np.fft.rfft((1-h)[::-1], n=n_fft)
    corr = np.fft.irfft(fa*fb, n=n_fft)[:2*n-1]
    return np.max(corr) * (2.0/n)

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project vector v onto [lower,upper] with exact sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Generate a random feasible vector (0≤h≤1, sum = n/2)."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a random amount from i to j respecting bounds."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def compute_corr_fft(h):
    """Full correlation of h and (1-h) using FFT (length 2n-1)."""
    n = h.size
    n_fft = 1 << (int(np.ceil(np.log2(2 * n - 1))))
    fa = np.fft.rfft(h, n=n_fft)
    fb = np.fft.rfft((1 - h)[::-1], n=n_fft)
    corr_full = np.fft.irfft(fa * fb, n=n_fft)[: 2 * n - 1]
    return corr_full


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    n = h.size
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb using perturb_h and exact C5 evaluation."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        # occasional uphill acceptance to escape shallow traps
        elif rng.random() < 0.001:
            h = nh
        if (it + 1) % max(1, max_iters // 6) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth max surrogate."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in beta_sched:
        step = 0.6
        for _ in range(steps_per_beta):
            corr = compute_corr_fft(h)                     # length 2n-1
            C = corr * (2.0 / n)                           # scaled overlaps
            # soft‑max weights
            mb = np.max(beta * C)
            w = np.exp(beta * C - mb)
            w /= w.sum()
            # gradient of smooth max (ignoring constant term)
            g = np.zeros(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g[: n - k] -= wk * h[k:]
                    g[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g[kk:]    -= wk * h[: n - kk]

            g *= (2.0 / n)  # scale to match C’s scaling

                h = nh
            else:
                # small chance to move uphill
                if rng.random() < 0.002:
                    h = nh
            step *= 0.997
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = compute_corr_fft(x)               # length 2n-1
    C = corr * (2.0 / x.size)
    bc = CURRENT_BETA * C
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    """Single SLSQP refinement using soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size

    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
