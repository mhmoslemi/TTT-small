import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    # clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # pick random index
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    target = n / 2.0
    x = rng.random(n)
    # scale to meet target sum
    total = x.sum()
    if total > 0:
        x = x * (target / total)
    else:
        x = np.full(n, target / n)
    x = np.clip(x, 0.0, 1.0)
    x = enforce_sum(x, target, rng)
    return x

def perturb_h(h, rng, scale=0.3):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # fix tiny sum drift
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def simulated_annealing(h0, rng, max_iters, T0):
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h = h.copy()
    best_c = c_cur
    n = h.size
    for it in range(max_iters):
        T = T0 * (1.0 - it / max_iters)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5  # up to half possible transfer
        if eps <= 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        delta = c_new - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            h = new_h
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = rng.integers(120, 181)  # moderate size
        h0 = random_feasible_vector(n, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    beta_list = [10.0, 30.0, 80.0, 200.0, 500.0, 1000.0]
    # main optimisation loop
    while time.time() - start_time < budget_s * 0.83:
        # choose start point
        if rng.random() < 0.7:

        else:

        # simulated annealing

        rem = budget_s - (time.time() - start_time)
        max_sa_iters = 40000
        if rem < 30:

        cur_h, cur_c = simulated_annealing(start, rng, max_sa_iters, T0=0.02)
        if cur_c < best_c:
            best_h, best_c = cur_h.copy(), cur_c
        # SLSQP refinement
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.90:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=300)
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined.copy(), c_ref
            except Exception:
                continue
    # optional up‑sampling
    if time.time() - start_time < budget_s * 0.96:
        n_up = best_h.size * 2
        if n_up <= 380:
            h_up = np.repeat(best_h, 2)
            h_up = enforce_sum(h_up, n_up / 2.0, rng)
            # anneal on finer grid
            cur_h_up, cur_c_up = simulated_annealing(h_up, rng, 60000, T0=0.015)
            # SLSQP refinement on finer grid
            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                try:
                    cur_h_up = slsqp_optimize(cur_h_up, beta, max_iter=300)

                if c_up < best_c:

    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the smooth surrogate for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    target = n / 2.0
    x = rng.random(n)
    total = x.sum()
    if total > 0:
        x = x * (target / total)
    else:
        x = np.full(n, target / n)
    x = np.clip(x, 0.0, 1.0)
    return enforce_sum(x, target, rng)


def perturb_h(h, rng, scale=0.3):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct possible tiny drift in the sum constraint
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def simulated_annealing(h0, rng, max_iters, T0):
    """Simple SA that preserves the sum by moving ε between two coordinates."""
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h = h.copy()
    best_c = c_cur
    n = h.size
    for it in range(max_iters):
        T = T0 * (1.0 - it / max_iters)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5  # at most half of the admissible amount
        if eps <= 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        delta = c_new - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            h = new_h
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -----------------------------------------------------------------
    # initialise a feasible h
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = rng.integers(120, 181)          # moderate resolution
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    beta_list = [10.0, 30.0, 80.0, 200.0, 500.0, 1000.0]

    # -----------------------------------------------------------------
    # main optimisation: repeated SA + SLSQP refinement
    # -----------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.83:
        # pick a starting point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.25)
        else:
            start = random_feasible_vector(best_h.size, rng)

        # simulated annealing
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        cur_h, cur_c = simulated_annealing(start, rng, max_iters=40000, T0=0.02)
        if cur_c < best_c:
            best_h, best_c = cur_h.copy(), cur_c

        # local SLSQP refinement with increasing β
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.90:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=300)
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined.copy(), c_ref
            except Exception:
                continue

    # -----------------------------------------------------------------
    # optional up‑sampling to a finer grid
    # -----------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.96:
        n_up = best_h.size * 2
        if n_up <= 380:                     # keep the problem reasonably small
            h_up = np.repeat(best_h, 2)
            h_up = enforce_sum(h_up, n_up / 2.0, rng)

            # SA on the finer grid
            cur_h_up, cur_c_up = simulated_annealing(h_up, rng, max_iters=60000, T0=0.015)

            # SLSQP refinement on the finer grid
            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                try:
                    cur_h_up = slsqp_optimize(cur_h_up, beta, max_iter=300)
                except Exception:
                    pass
                c_up = compute_c5(cur_h_up)
                if c_up < best_c:
                    best_h, best_c = cur_h_up.copy(), c_up
                    n_up = cur_h_up.size
            # update n if we kept the up‑sampled version
            if best_h.size == n_up:
                n = n_up

    # -----------------------------------------------------------------
    # final clean‑up
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
