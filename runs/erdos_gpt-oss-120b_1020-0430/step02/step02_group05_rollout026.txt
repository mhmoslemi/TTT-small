def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n ∩ {sum = target} using a binary search for λ."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)

            break

        # Build deterministic base candidates for this resolution.
        base_candidates = []
        if init_h is not None and init_h.size == n:
            base_candidates.append(init_h.copy())
        elif init_h is not None:
            base_candidates.append(_upscale_h(init_h, n))
        else:
            base_candidates.append(_tent_h(n))

        base_candidates.append(_tent_h(n))
        base_candidates.append(_sine_h(n, amp=0.45))

        # Deterministic refinements.
        for base_h in base_candidates:
            if time_left() < 5.0:
                break
            h_opt, c5 = _solve_h(base_h, max_iter=800)
            h_opt = _local_refine(h_opt, steps=6, lr=0.025)
            c5 = _c5_from_h(h_opt)
            if c5 < best_c5:
                best_h, best_c5, best_n = h_opt, c5, n
                if best_c5 < target:
                    return best_h.tolist(), float(best_c5), int(best_n)

        # Stochastic restarts.
        attempts = 80
        sigma = 0.25
        for _ in range(attempts):
            if time_left() < 5.0:
                break
            r = np.random.rand()
            if r < 0.3:
                start = _binary_start(n)
            elif r < 0.6:
                start = base_candidates[0] + sigma * np.random.randn(n)
            else:
                start = np.random.rand(n)
            start = _project_to_simplex_box(start, n / 2.0)

            h_opt, c5 = _solve_h(start, max_iter=800)
            h_opt = _local_refine(h_opt, steps=5, lr=0.02)
            c5 = _c5_from_h(h_opt)
            if c5 < best_c5:
                best_h, best_c5, best_n = h_opt, c5, n
                if best_c5 < target:
                    return best_h.tolist(), float(best_c5), int(best_n)
            sigma *= 0.96

    # Fallback if nothing succeeded (unlikely).
    if best_h is None:
        n = 128
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
