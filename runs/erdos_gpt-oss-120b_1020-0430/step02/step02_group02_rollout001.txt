import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Projection onto the feasible polytope: 0<=h<=1, sum(h)=n/2
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Constraints for the continuous SLSQP optimisation
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


# ----------------------------------------------------------------------
# Continuous refinement using SLSQP (non‑convex but works well in practice)
# ----------------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Small greedy hill‑climb (continuous epsilon exchange)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps      # allow full range
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Linear up‑sampling of a coarse step function
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Simulated annealing on binary vectors (mass swap moves)
# ----------------------------------------------------------------------
def _binary_anneal(start_h: np.ndarray, time_limit: float) -> tuple:
    h = start_h.copy()
    n = h.size
    cur_c = _c5_from_h(h)
    best_h = h.copy()
    best_c = cur_c

    T = 0.05                # initial temperature
    decay = 0.99995

    # maintain index lists for O(1) sampling
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])

    while time.time() < time_limit:
        if not ones or not zeros:
            break
        i_pos = np.random.randint(len(ones))
        j_pos = np.random.randint(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = _c5_from_h(h)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            diff = new_c - cur_c
            if T > 0 and np.random.rand() < np.exp(-diff / T):
                accept = True

        if accept:
            cur_c = new_c
            # update best
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            # update index lists
            ones[i_pos] = j
            zeros[j_pos] = i
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

        T *= decay
        # occasional re‑thermalisation to avoid freezing completely
        if T < 1e-9:
            T = 0.01

    return best_h, best_c

            # random binary
            start_h = np.zeros(n)
            ones_idx = np.random.choice(n, n // 2, replace=False)
            start_h[ones_idx] = 1.0

        # ----- binary simulated annealing -----
        limit_anneal = time.time() + min(0.6 * time_left(), 20.0)
        # ensure exact half‑ones
        if start_h.sum() != n / 2:
            # simple fix: randomise again
            start_h = np.zeros(n)
            start_h[np.random.choice(n, n // 2, replace=False)] = 1.0

        bin_h, bin_c = _binary_anneal(start_h, limit_anneal)

        # ----- continuous refinement -----
        if time_left() > 2.0:
            h_ref, c_ref = _solve_h(bin_h, max_iter=1200)
            if c_ref < bin_c:
                bin_h, bin_c = h_ref, c_ref

        if time_left() > 2.0:
            limit_hc = time.time() + min(0.2 * time_left(), 8.0)
            bin_h, bin_c = _hill_climb(bin_h, max_steps=3000, time_limit=limit_hc)

        # ----- record best -----
        if bin_c < best_c5:
            best_h, best_c5, best_n = bin_h.copy(), bin_c, n
            best_global = bin_h.copy()

        # early exit if record beaten
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Fallback
    # ------------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    # recompute final bound to be safe
    final_c5 = _c5_from_h(np.array(best_h, dtype=float))

    return best_h.tolist(), float(final_c5), int(best_n)
