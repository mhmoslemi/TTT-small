def smooth_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1-h, mode='full')
            C = corr * (2.0 / n)
            bc = beta * C
            max_bc = np.max(bc)
            # weights via softmax
            w = np.exp(bc - max_bc)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient for smooth max
            g = np.zeros(n)
            for idx, wk in enumerate(w):
                if wk == 0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    i_range = slice(0, n - k)
                    j_range = slice(k, n)
                else:
                    kk = -k
                    i_range = slice(kk, n)
                    j_range = slice(0, n - kk)
                # contribution: wk * (1 - h[j]) to g[i]; -wk * h[i] to g[j]
                g[i_range] += wk * (1 - h[j_range])
                g[j_range] -= wk * h[i_range]
            grad = (2.0 / n) * g
            # step update with projection
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)

            # evaluate
            c_new = compute_c5(h_new)

                best_c, best_h = c_new, h_new.copy()
                h = h_new
                # maybe increase step a bit?
                step *= 1.02
            else:
                # no improvement, reduce step
                step *= 0.95
            # optionally cap step

    return best_h, best_c

def subgradient_descent_optimize(h0, steps, step_size, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    step = step_size
    for it in range(steps):
        corr = np.correlate(h, 1-h, mode='full')
        C = corr * (2.0 / n)
        max_val = np.max(C)
        mask = np.isclose(C, max_val, atol=1e-12)  # can use relative tolerance?
        wk = mask.astype(float) / mask.sum()
        g = np.zeros(n)
        for idx, w in enumerate(wk):
            if w == 0:
                continue
            k = idx - (n - 1)
            if k >= 0:
                i_range = slice(0, n - k)
                j_range = slice(k, n)
            else:
                kk = -k
                i_range = slice(kk, n)
                j_range = slice(0, n - kk)
            g[i_range] += w * (1 - h[j_range])
            g[j_range] -= w * h[i_range]
        grad = (2.0 / n) * g
        # step
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, n/2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # update step (maybe reduce)
        step *= 0.9995
        h = h_new
    return best_h, best_c

import time
import numpy as np

# -------------------------------------------------------
# Basic utilities
# -------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0≤h≤1, Σh=n/2). n is assumed even."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    h = np.concatenate([x, x[::-1]])
    return h


def perturb_h(h, rng, scale=0.2):
    """Move a random amount from i to j keeping the sum unchanged."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill climbing using pairwise perturbations."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        h_new = perturb_h(h, rng, scale=step)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# -------------------------------------------------------
# Smooth max gradient descent
# -------------------------------------------------------
def smooth_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth max (log‑sum‑exp) surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    target = n / 2.0

    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            C = corr * (2.0 / n)          # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            g = np.zeros(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    i_range = slice(0, n - k)
                    j_range = slice(k, n)
                else:
                    kk = -k
                    i_range = slice(kk, n)
                    j_range = slice(0, n - kk)
                # gradient contributions for this shift
                g[i_range] += wk * (1.0 - h[j_range])
                g[j_range] -= wk * h[i_range]

            grad = (2.0 / n) * g
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            # enforce symmetry and sum
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = project_onto_box_sum(h_new, target)

            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                h = h_new
                step = min(step * 1.05, 1.0)
            else:
                step = max(step * 0.95, 1e-4)
    return best_h, best_c


# -------------------------------------------------------
# Subgradient descent on the true max
# -------------------------------------------------------
def subgradient_descent_optimize(h0, steps, step_size, rng):
    """Projected sub‑gradient descent on the exact max overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    target = n / 2.0
    step = step_size

    for it in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        C = corr * (2.0 / n)
        max_val = np.max(C)
        mask = np.isclose(C, max_val, atol=1e-12, rtol=0.0)
        wk = mask.astype(float)
        if wk.sum() == 0:
            wk = np.ones_like(wk) / wk.size
        else:
            wk /= wk.sum()

        g = np.zeros(n)
        for idx, w in enumerate(wk):
            if w == 0.0:
                continue
            k = idx - (n - 1)
            if k >= 0:
                i_range = slice(0, n - k)
                j_range = slice(k, n)
            else:
                kk = -k
                i_range = slice(kk, n)
                j_range = slice(0, n - kk)
            g[i_range] += w * (1.0 - h[j_range])
            g[j_range] -= w * h[i_range]

        grad = (2.0 / n) * g
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = (h_new + h_new[::-1]) * 0.5
        h_new = project_onto_box_sum(h_new, target)

        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # slowly decay step size
        step *= 0.9995
        h = h_new
    return best_h, best_c

    if h_small.size == N:

    xt = np.linspace(0.0, 2.0, N, endpoint=False)
    return np.interp(xt, xs, h_small)
