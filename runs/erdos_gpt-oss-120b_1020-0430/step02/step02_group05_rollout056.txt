        # If shift is 0, the correlation is sum h*(1-h) which might be nonzero if h not binary.

def _hill_climb(h, steps=200, epsilon=0.02):
    n = h.size
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        # Compute shift
        shift = max_idx - (n - 1)
        if shift == 0:
            # cannot improve via swap; break
            break
        # Define i range:
        if shift > 0:
            i_start = 0
            i_end = n - 1 - shift
        else:
            i_start = -shift
            i_end = n - 1
        # vectorized compute feasible indices
        idx = np.arange(i_start, i_end + 1)

        gain = ((1 - hj) + hi) * feasible
        i_best = idx[np.argmax(gain)]
        delta = min(epsilon, float(h[i_best]), float(1 - h[i_best + shift]))
        if delta <= 0:
            break
        h[i_best] -= delta

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """C5 for discrete h on [0,2] (dx = 2/n)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_h(start_h: np.ndarray, max_iter: int = 400):
    """Refine a feasible h with SLSQP on the max‑correlation formulation."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    constraints = [{"type": "ineq", "fun": ineq},
                   {"type": "eq", "fun": eq}]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1],
                            x0,
                            method="SLSQP",
                            bounds=bounds,
                            constraints=constraints,
                            options=opts)
    h_opt = h0 if not res.success else res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape, then projected to the simplex."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _hill_climb(h: np.ndarray, steps: int = 250, epsilon: float = 0.02) -> np.ndarray:
    """Local greedy improvement: move mass from 1‑positions to 0‑positions
    across the shift giving the current maximal correlation."""
    n = h.size
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        if shift == 0:
            break
        if shift > 0:
            i0, i1 = 0, n - 1 - shift
        else:
            i0, i1 = -shift, n - 1
        idx = np.arange(i0, i1 + 1)
        hi = h[idx]
        hj = h[idx + shift]
        feasible = (hi > 1e-12) & (hj < 1.0 - 1e-12)
        if not np.any(feasible):
            break
        gain = ((1.0 - hj) + hi) * feasible
        i_best = idx[np.argmax(gain)]
        delta = min(epsilon, float(h[i_best]), float(1.0 - h[i_best + shift]))
        if delta <= 0:
            break
        h[i_best] -= delta
        h[i_best + shift] += delta
    # tiny sum correction
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) > 1e-9:
        h = _project_to_simplex_box(h, target)
    return h


def _fourier_h(gamma: np.ndarray, n: int) -> np.ndarray:
    """Build h = 0.5 + Σ (a_j cos + b_j sin), clip & project."""
    m = len(gamma) // 2
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for j in range(1, m + 1):
        a = gamma[j - 1]
        b = gamma[m + j - 1]
        h += a * np.cos(np.pi * j * xs) + b * np.sin(np.pi * j * xs)
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_simplex_box(h, n / 2.0)
    return h


def _obj_gamma(gamma: np.ndarray, n: int) -> float:
    """Objective for Fourier parameters."""
    h = _fourier_h(gamma, n)
    return _c5_from_h(h)
