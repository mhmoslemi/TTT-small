def compute_c5_fft(h):
    n = h.shape[0]
    m = 1 << ((2*n-1).bit_length())
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)

    B = np.fft.rfft(b, m)
    C = np.fft.irfft(A * B, m)[:2*n-1]

    for _ in range(max_iter):

        if mask.any():
            proj[mask] += diff / mask.sum()

    return proj

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

# ---------- C5 computation (FFT for speed) ----------
def compute_c5(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    a = h
    b = (1.0 - h)[::-1]
    A = np.fft.rfft(a, m)
    B = np.fft.rfft(b, m)
    C = np.fft.irfft(A * B, m)[: 2 * n - 1]
    return np.max(C) * (2.0 / n)

        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- random feasible ----------
def random_feasible(n, rng):
    v = rng.random(n)
    v = project_onto_box_sum(v, n / 2.0)
    return v


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise (upsample if we have a parent) -----
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n_parent = parent.size
        target_n = 500  # chosen resolution
        x_old = np.linspace(0, 2, n_parent, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h0 = np.interp(x_new, x_old, parent)
    else:
        target_n = 500
        h0 = random_feasible(target_n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_sum(h0, target_n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing (dominant phase) -----
    sa_budget = budget_s * 0.78
    cur_h = best_h.copy()
    cur_c = best_c
    T_start = 0.5
    T_end = 1e-6
    it = 0
    while time.time() - start < sa_budget:
        elapsed = time.time() - start
        frac = (elapsed) / sa_budget
        T = T_start * (1.0 - frac) + T_end * frac

        i, j = rng.integers(0, target_n, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * (rng.random() ** 2)
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        it += 1
        # occasional jitter around best to escape plateaus
        if it % 100000 == 0:
            cur_h = best_h + rng.normal(scale=0.02, size=target_n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = project_onto_box_sum(cur_h, target_n / 2.0)
            cur_c = compute_c5(cur_h)

    # ----- greedy hill‑climb refinement if time permits -----
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0:
        max_iters = int(min(8000, remaining * 1500))
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=max_iters, start_step=0.5)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ----- final SLSQP polishing -----
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        for beta in (200.0, 500.0, 1000.0):
            try:
                best_h = slsqp_optimize(best_h, beta=beta, max_iter=200)
                best_c = compute_c5(best_h)
            except Exception:
                pass

    # ----- final safety projection -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
