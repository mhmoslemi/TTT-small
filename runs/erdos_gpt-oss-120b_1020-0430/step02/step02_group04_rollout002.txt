def max_shift_grad(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    idxs = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            # term1
            term1 = np.empty(n)
            term1[:n-k] = 1 - h[k:]
            term1[n-k:] = 0.0
            # term2
            term2 = np.empty(n)
            term2[:k] = 0.0
            term2[k:] = h[:n-k]
        else:
            kk = -k
            term1 = np.empty(n)
            term1[:kk] = 0.0
            term1[kk:] = 1 - h[:n-kk]
            term2 = np.empty(n)
            term2[:n-kk] = h[kk:]
            term2[n-kk:] = 0.0
        grad += term1 - term2
    grad /= len(idxs)  # average
    return grad, max_val

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Basic utilities
# -------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto the box [lower,upper] while fixing sum(v)=target."""
    lo = -1e9
    hi = 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Return a symmetric feasible vector of length n (n even)."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def perturb_h(h, rng, scale=0.2):
    """Move a random amount between two entries while preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb using perturbations."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# Gradient of the max‑overlap (sub‑gradient)
# -------------------------------------------------
def max_shift_grad(h):
    """Return gradient of max_k corr_k and the max value (unnormalized)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            # term1 = 1 - h[i+k] for i < n-k, else 0
            term1 = np.empty(n)
            term1[: n - k] = 1.0 - h[k:]
            term1[n - k :] = 0.0
            # term2 = h[i-k] for i >= k, else 0
            term2 = np.empty(n)
            term2[:k] = 0.0
            term2[k:] = h[: n - k]
        else:
            kk = -k
            term1 = np.empty(n)
            term1[:kk] = 0.0
            term1[kk:] = 1.0 - h[: n - kk]
            term2 = np.empty(n)
            term2[: n - kk] = h[kk:]
            term2[n - kk :] = 0.0
        grad += term1 - term2
    grad /= max(1, len(idxs))
    return grad, max_val

# Soft‑max surrogate + SLSQP (optional refinement)
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP minimisation of the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

# Adam optimiser for the true max‑overlap objective
# -------------------------------------------------
def adam_optimize(h_start, max_iter, lr):
    n = h_start.size
    dx = 2.0 / n
    h = h_start.copy()
    m_t = np.zeros_like(h)
    v_t = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iter + 1):
        grad_raw, _ = max_shift_grad(h)
        grad = grad_raw * dx  # gradient of C5
        # Adam moments
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * (grad * grad)
        m_hat = m_t / (1 - beta1 ** t)
        v_hat = v_t / (1 - beta2 ** t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step_vec
        # project onto box & sum
        h_new = project_onto_box_sum(h_new, n / 2.0)
        # enforce symmetry
        h_new = (h_new + h_new[::-1]) * 0.5
        h = h_new
    return h, compute_c5(h)
