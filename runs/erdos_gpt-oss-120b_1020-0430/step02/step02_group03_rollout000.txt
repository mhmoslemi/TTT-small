import time, math, numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx=2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that sum(h)=target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate random h with 0≤h≤1 and sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def biased_perturb_h(h, rng, scale=0.2):
    """Bias transfer away from positions contributing most to the max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    shift = k - (n - 1)
    i0 = max(0, -shift)
    i1 = min(n, n - shift)
    idx = np.arange(i0, i1)
    prod = h[idx] * (1 - h[idx + shift])
    if prod.size == 0:
        return perturb_h(h, rng, scale)
    # pick high‑product source, low‑product destination
    thr_hi = np.quantile(prod, 0.70)
    thr_lo = np.quantile(prod, 0.30)
    high = idx[prod > thr_hi]
    low = idx[prod < thr_lo]
    if high.size == 0 or low.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(high)
    # destination chosen outside the high‑product set (could be low or any)
    j = rng.choice(low)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    cons = {"type": "eq", "fun": sum_eq_constraint}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- Simulated annealing ----------
def simulated_annealing(h0, rng, max_iter, init_temp, final_temp):
    """Simple SA on C5 (lower is better)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    # geometric temperature schedule
    log_ratio = math.log(final_temp / init_temp) if init_temp > 0 else 0.0
    for it in range(1, max_iter + 1):
        # temperature
        T = init_temp * math.exp(log_ratio * it / max_iter)
        # biased or uniform proposal
        if rng.random() < 0.7:
            cand = biased_perturb_h(h, rng, scale=0.3)
        else:
            cand = perturb_h(h, rng, scale=0.3)
        cand_c = compute_c5(cand)
        diff = cand_c - cur_c
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
    return best_h, best_c

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------
    # initialise vector
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------
    # main optimisation loop (SA + SLSQP refinements)
    # ------------------------------------------------------------
    # allocate ~90% of time for SA, remaining for final refinements
    sa_time = budget_s * 0.86
    while time.time() - start_time < sa_time:
        # choose start point: perturb best or fresh random
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # allocate iterations based on remaining time (roughly 0.5 ms per iter)
        remaining = max(0.0, sa_time - (time.time() - start_time))
        # aim for about 20000 iterations if time permits
        max_iter = int(min(40000, remaining * 2000))
        max_iter = max(2000, max_iter)

        cand_h, cand_c = simulated_annealing(start, rng,
                                            max_iter=max_iter,
                                            init_temp=0.02,
                                            final_temp=1e-6)

        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c

        # occasional SLSQP refinement (cheap beta list)
        for beta in (50, 200, 800):
            if time.time() - start_time > sa_time:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=250)
                rc = compute_c5(refined)
                if rc < best_c:
                    best_h, best_c = refined, rc
            except Exception:
                continue

    # ------------------------------------------------------------
    # final high‑beta SLSQP polishing (use remaining budget)
    # ------------------------------------------------------------
    for beta in (2000, 8000, 20000):
        if time.time() - start_time > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=200)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    # final clip & sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
