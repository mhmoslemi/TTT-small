def c5_fast(h):
    n = h.size
    b = 1 - h
    # Compute full convolution via FFT
    N = 1 << (2*n - 1).bit_length()
    ha = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    hb = np.fft.rfft(np.concatenate([b[::-1], np.zeros(N - n)]))
    conv = np.fft.irfft(ha * hb, n=N)
    corr = conv[:2*n - 1]  # length 2n-1
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

    n = h.size
    b = 1 - h
    N = 1 << (2*n-1).bit_length()
    ha = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))

    conv = np.fft.irfft(ha * hb, n=N)

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to target
    # Use water-filling
    # We'll compute diff = h.sum() - target and adjust uniformly among free variables
    diff = h.sum() - target_sum
    if abs(diff) < 1e-12:
        return h
    # We'll iteratively adjust
    for _ in range(50):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:
            break
    # final correction if still off
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute small amount
        for i in range(len(h)):
            if diff > 0 and h[i] < 1 - 1e-12:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 1e-12:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    t0 = np.max(corr0)

                             bounds=bounds, constraints=constraints,

import time
import numpy as np
from scipy import optimize

# --------------------- feasibility helpers ---------------------
def _project_to_feasible(h: np.ndarray, target_sum: float, tol: float = 1e-9) -> np.ndarray:
    """Clip to [0,1] and adjust sum to target_sum by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    for _ in range(50):
        if abs(diff) <= tol:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target_sum
    # tiny final correction
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        for i in range(len(h)):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    x_coarse = np.arange(coarse.size) / coarse.size * 2.0
    x_fine = np.arange(n_fine) / n_fine * 2.0
    return np.interp(x_fine, x_coarse, coarse)


# --------------------- evaluation helpers ---------------------
def _c5_exact(h: np.ndarray) -> float:
    """Exact C5 using np.correlate (same as the reference evaluator)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _max_corr_and_lag_fft(h: np.ndarray):
    """Return (max correlation value, lag) for h*(1‑h) using FFT."""
    n = h.size
    b = 1.0 - h
    N = 1 << (2 * n - 1).bit_length()
    ha = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    hb = np.fft.rfft(np.concatenate([b[::-1], np.zeros(N - n)]))
    conv = np.fft.irfft(ha * hb, n=N)
    conv = conv[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx - (n - 1)


# --------------------- local optimisation ---------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 400):
    """SLSQP refinement for a given starting h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = float(np.max(corr0))

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n / 2.0)
    return h_opt, _c5_exact(h_opt)


def _gradient_descent(h: np.ndarray, t_end: float):
    """Projected gradient descent on max h*(1‑h) (sub‑gradient of max shift)."""
    n = h.size
    target = n / 2.0
    best_h, best_c5 = h.copy(), _c5_exact(h)
    step = 0.5
    while time.time() < t_end:
        max_corr, lag = _max_corr_and_lag_fft(h)
        g = np.zeros_like(h)
        if lag >= 0:
            g[: n - lag] = 1.0 - h[lag:]
            g[lag:] -= h[: n - lag]
        else:
            lag_abs = -lag
            g[lag_abs:] = 1.0 - h[: n - lag_abs]
            g[: n - lag_abs] -= h[lag_abs:]
        pos = g > 0
        neg = g < 0
        alphas = []
        if np.any(pos):
            alphas.append(np.min(h[pos] / g[pos]))
        if np.any(neg):
            alphas.append(np.min((1.0 - h[neg]) / (-g[neg])))
        max_alpha = min(alphas) if alphas else 0.0
        if max_alpha <= 0:
            break
        step = min(step, max_alpha * 0.99)
        if step < 1e-8:
            break
        h_new = h - step * g
        h_new = _project_to_feasible(h_new, target)
        c5_new = _c5_exact(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new.copy(), c5_new
            h = h_new
            step *= 1.02
        else:
            step *= 0.5
    return best_h, best_c5


def _hill_climb(h: np.ndarray, max_steps: int, t_limit: float):
    """Simple random pair moves, accepting improvements."""
    n = h.size
    best_h, best_c5 = h.copy(), _c5_exact(h)
    step = 0
    while step < max_steps and time.time() < t_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * 0.5 * np.random.rand()
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_exact(cand)
        if c < best_c5:
            best_h, best_c5 = cand, c
        step += 1
    return best_h, best_c5


# --------------------- main entry ---------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left(): return budget_s - (time.time() - start)

    # target grid
    n = 256
    target_sum = n / 2.0

    # initialise from parent if present
    try:
        base = np.array(initial_h_values, dtype=float)
        h0 = _upscale_h(base, n)
    except NameError:
        h0 = np.random.rand(n)

    h0 = _project_to_feasible(h0, target_sum)
    best_h, best_c5 = h0, _c5_exact(h0)

    # main stochastic loop
    while time_left() > 7.0:
        # random start (perturb current best a little)
        h_start = best_h + 0.2 * np.random.randn(n)
        h_start = _project_to_feasible(h_start, target_sum)

        # gradient descent phase (≈5 s)
        t_end = time.time() + min(5.0, time_left() - 2.0)
        h_gd, c5_gd = _gradient_descent(h_start, t_end)

        if c5_gd < best_c5:
            best_h, best_c5 = h_gd, c5_gd
            # quick SLSQP polish
            if time_left() > 2.0:
                h_slsqp, c5_slsqp = _solve_h(best_h, max_iter=300)
                if c5_slsqp < best_c5:
                    best_h, best_c5 = h_slsqp, c5_slsqp

        # short hill‑climb from best
        if time_left() > 2.0:
            t_hc = time.time() + min(2.0, time_left() - 0.5)
            best_h, best_c5 = _hill_climb(best_h, max_steps=3000, t_limit=t_hc)

    # final polish with SLSQP if time permits
    if time_left() > 2.0:
        best_h, best_c5 = _solve_h(best_h, max_iter=500)

    final_c5 = _c5_exact(best_h)
    return best_h.tolist(), float(final_c5), int(n)
