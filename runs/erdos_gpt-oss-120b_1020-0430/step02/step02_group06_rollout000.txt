def max_corr_of_h(h):

    # correlation h * (1-h) but we need full mode.
    # correlation (full) can be done via fft of h and (1-h) reversed.
    a = np.fft.rfft(h, n*2-1)

def anneal_binary(h_initial, n, time_left_func, target_max_corr=None):
    best_h = h_initial.copy()
    best_val = max_corr_val(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    # define temperature schedule based on initial val.
    T = (cur_val) * 0.5  # initial temperature maybe half of current val.
    step = 0
    while time_left_func() > 0.10: # minimal time left to allow iteration
        # pick random swap
        # get one indices and zero indices
        ones_idx = np.flatnonzero(cur_h)
        zeros_idx = np.flatnonzero(1 - cur_h)  # zeros
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_val = max_corr_val(new_h)
        d = new_val - cur_val
        if d < 0 or np.random.rand() < np.exp(-d / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            # update ones_idx and zeros_idx maybe recompute each time
        if cur_val < best_val:
            best_h, best_val = cur_h.copy(), cur_val
            if target_max_corr is not None and best_val <= target_max_corr:
                break
        # cooling
        step += 1
        # exponential decay: T = T * 0.9999 maybe
        T *= 0.99995
        if T < 1e-8:
            T = 1e-8
    return best_h, best_val

import time
import numpy as np

def _max_corr_binary(h: np.ndarray) -> int:
    """Maximum correlation of binary h with its complement (full mode)."""
    # convolution of h with reversed complement gives full correlation
    corr = np.convolve(h, (1 - h)[::-1], mode='full')
    return int(corr.max())

def _binary_from_continuous(hc: np.ndarray) -> np.ndarray:
    """Return binary array with exactly half ones, using largest values of hc."""
    n = hc.size
    half = n // 2
    idx = np.argpartition(hc, -half)[-half:]
    bin_arr = np.zeros(n, dtype=int)
    bin_arr[idx] = 1
    return bin_arr

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of coarse h onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _anneal_binary(
    start_h: np.ndarray,
    time_left,
    target_max_corr: int = None,
    max_no_improve: int = 2000,
) -> tuple:
    """Simulated‑annealing on binary h (exactly n/2 ones)."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_val = _max_corr_binary(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    # initial temperature roughly half the current max_corr
    T = max(0.5 * cur_val, 0.1)
    it = 0
    no_improve = 0

    while time_left() > 0.05:
        # pick a random 1 and a random 0 to swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_val = _max_corr_binary(new_h)

        delta = new_val - cur_val
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
                no_improve = 0
                if target_max_corr is not None and best_val <= target_max_corr:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # cooling
        it += 1
        T *= 0.99995
        if T < 1e-8:
            T = 1e-8
        if no_improve >= max_no_improve:
            # re‑heat a little to escape possible stagnation
            T = max(T, 0.5)
            no_improve = 0

    return best_h, best_val
