def max_corr_fft(h):
    f = 1 - h
    n = len(h)
    size = 2 * n
    # Use rfft for real signals
    H = np.fft.rfft(h, n=size)
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(H * F, n=size)
    # conv length size; only first 2n-1 entries relevant
    # conv has small floating errors; round
    max_val = int(np.rint(conv[:2*n-1].max()))
    return max_val

def _max_corr_fft(h: np.ndarray) -> int:
    """Maximum of h * (1‑h) correlation (full mode) for binary h."""
    f = 1 - h

    H = np.fft.rfft(h, n=size)
    F = np.fft.rfft(f, n=size)

    return int(np.rint(conv[: 2 * n - 1].max()))
