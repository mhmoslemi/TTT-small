def _solve_h(start_h, max_iter=400):
    n = start_h.size
    # project to feasible
    h0 = _project_to_simplex_box(start_h, n/2)
    corr0 = _full_corr(h0)
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {'type': 'ineq', 'fun': _ineq_constraint},
        {'type': 'eq', 'fun': _eq_constraint},
    ]

    res = optimize.minimize(_objective, x0, method='SLSQP', bounds=bounds,
                            constraints=constraints,
                            options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    # final projection
    h_opt = _project_to_simplex_box(h_opt, n/2)
    return h_opt, _c5_from_h(h_opt)

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum(x)=target}."""
    lo = np.min(v) - 1.0   # force projection to all-ones when lambda <= lo
    hi = np.max(v)         # force projection to all-zeros when lambda >= hi
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny repair for residual error
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1.0 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _full_corr(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h) using FFT (equivalent to np.correlate)."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    corr = np.fft.irfft(H * G, L)
    return corr.real


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discrete step function h."""
    n = h.size
    max_corr = np.max(_full_corr(h))
    return max_corr * (2.0 / n)


def _objective(x: np.ndarray) -> float:
    """Objective: minimise auxiliary variable t."""
    return float(x[-1])


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """Inequality constraints: t - correlation >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = _full_corr(h)
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Equality constraint: sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return float(h.sum() - n / 2.0)


def _solve_h(start_h: np.ndarray, max_iter: int = 500) -> tuple[np.ndarray, float]:
    """Run SLSQP from a feasible start and return refined h and its C5."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(_full_corr(h0))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points, projected onto the feasible set."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)
