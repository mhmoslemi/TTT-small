def hill_climbing_binary(h, end_time):
    n = len(h)
    sz = 2*n - 1
    ov = np.correlate(h, h, mode='full')
    half = n // 2
    corr = half - ov
    max_corr = corr.max()
    # maintain ones & zeros lists
    ones = list(np.flatnonzero(h))
    zeros = list(np.where(h == 0)[0])
    best_h = h.copy()
    best_max_corr = max_corr
    # Pre allocate delta buffer
    delta_buf = np.empty(sz, dtype=np.int32)
    batch_size = 30 if n <= 200 else 15  # adapt
    attempts_without_improve = 0
    max_no_improve = 2000
    while time.time() < end_time and attempts_without_improve < max_no_improve:
        # sample batch of swaps
        best_new_max = max_corr
        best_swap = None
        best_delta = None
        for _ in range(batch_size):
            i = random.choice(ones)
            j = random.choice(zeros)
            delta_buf.fill(0)
            # arr_i_plus
            start = n - 1 - i
            delta_buf[start:start+n] -= h
            delta_buf[i:i+n] -= h[::-1]
            # arr_j_plus
            startj = n - 1 - j
            delta_buf[startj:startj+n] += h
            delta_buf[j:j+n] += h[::-1]
            # conv(d,d)
            delta_buf[n-1] += 2
            delta_buf[i - j + n - 1] -= 1
            delta_buf[j - i + n - 1] -= 1
            new_corr = corr - delta_buf
            new_max = new_corr.max()
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j)
                best_delta = delta_buf.copy()
        if best_swap is None:
            attempts_without_improve += 1
            continue
        # apply best_swap
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        # update ones zeros lists
        # replace i with j in ones list
        # assume i in ones list.
        idx_i = ones.index(i)   # O(k)
        ones[idx_i] = j
        idx_j = zeros.index(j)
        zeros[idx_j] = i
        # update corr
        corr -= best_delta
        max_corr = best_new_max
        attempts_without_improve = 0
        # record best
        if max_corr < best_max_corr:
            best_h = h.copy()
            best_max_corr = max_corr
    # final C5:
    c5 = max_corr * (2.0 / n)
    return best_h, c5

def projected_gradient_descent(h0, max_iters, lr, time_end):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2
    for it in range(max_iters):
        if time.time() >= time_end:
            break
        # compute ov
        ov = np.correlate(h, h, mode='full')
        corr = target_sum - ov
        max_idx = np.argmax(corr)
        s = max_idx - (n - 1)  # shift
        # compute gradient
        grad = np.zeros_like(h)
        # indices for shift s: for k where k-s valid
        if s >= 0:
            # h_{k-s} valid for k >= s
            grad[s:] -= h[:n-s]
            # h_{k+s} valid for k < n-s
            grad[:n-s] -= h[s:]
        else:
            # s negative
            sp = -s
            # h_{k-s} = h_{k+sp} valid for k <= n-1-sp
            grad[:n-sp] -= h[sp:]
            # h_{k+s} = h_{k-sp} valid for k >= sp
            grad[sp:] -= h[:n-sp]
        # gradient descent step
        h_new = h - lr * grad
        # projection
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust sum
        h_new = _project_to_simplex_box(h_new, target_sum)
        # check for convergence (change small)
        if np.max(np.abs(h_new - h)) < 1e-8:
            break
        h = h_new
        # optionally adjust lr schedule
        lr *= 0.99  # decay
    # final c5
    ov = np.correlate(h, h, mode='full')
    corr = target_sum - ov
    max_corr = np.max(corr)
    c5 = max_corr * 2.0 / n
    return h, c5

import time
import random
import numpy as np


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n with sum(v)=target."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _init_binary_h(n: int, base: np.ndarray | None) -> np.ndarray:
    """Return a binary (0/1) vector with exactly n/2 ones."""
    if base is not None:
        if base.size != n:
            # linear interpolation to the wanted size
            x_src = (np.arange(base.size) + 0.5) * (2.0 / base.size)
            x_tgt = (np.arange(n) + 0.5) * (2.0 / n)
            base = np.interp(x_tgt, x_src, base)
        idx = np.argpartition(base, -n // 2)[-n // 2 :]
        h = np.zeros(n, dtype=np.int8)
        h[idx] = 1
        return h
    # random binary vector
    h = np.zeros(n, dtype=np.int8)
    h[np.random.choice(n, n // 2, replace=False)] = 1
    return h


def _proj_grad_descent(h0: np.ndarray, end_time: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    lr = 0.5
    best_h = h.copy()
    # initial max correlation
    ov = np.correlate(h, h, mode="full")
    corr = target - ov
    max_corr = np.max(corr)
    best_corr = max_corr
    it = 0
    while time.time() < end_time and it < 1500:
        it += 1
        ov = np.correlate(h, h, mode="full")
        corr = target - ov
        idx = np.argmax(corr)
        s = idx - (n - 1)

        grad = np.zeros_like(h)
        if s >= 0:
            grad[s:] -= h[: n - s]
            grad[: n - s] -= h[s:]
        else:
            sp = -s
            grad[: n - sp] -= h[sp:]
            grad[sp:] -= h[: n - sp]

        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_simplex_box(h_new, target)

        ov_new = np.correlate(h_new, h_new, mode="full")
        corr_new = target - ov_new
        max_corr_new = np.max(corr_new)

        # simple acceptance rule with occasional uphill moves
        if max_corr_new < max_corr - 1e-12:
            h, max_corr = h_new, max_corr_new
            if max_corr_new < best_corr:
                best_corr, best_h = max_corr_new, h_new.copy()
        else:
            T = max(0.01, 0.4 * (1.0 - it / 1500.0))
            if np.random.rand() < np.exp(-(max_corr_new - max_corr) / T):
                h, max_corr = h_new, max_corr_new

        lr *= 0.995
    bound = max_corr * (2.0 / n)
    return best_h, float(bound)


def _binary_hill_climb(h0: np.ndarray, end_time: float) -> tuple[np.ndarray, float]:
    """Local search on binary vectors using best‑of‑batch swaps."""
    h = h0.copy()
    n = h.size
    sz = 2 * n - 1
    half = n // 2
    ov = np.correlate(h, h, mode="full")
    corr = half - ov
    max_corr = int(corr.max())
    best_h = h.copy()
    best_max = max_corr

    ones = list(np.flatnonzero(h))
    zeros = list(np.where(h == 0)[0])
    rev_h = h[::-1]
    delta = np.empty(sz, dtype=np.int32)

    batch = 30 if n <= 224 else 20
    no_improve = 0
    while time.time() < end_time and no_improve < 800:
        # evaluate a batch and keep the best improving swap
        cur_best = max_corr
        best_swap = None
        best_delta = None
        for _ in range(batch):
            i = random.choice(ones)
            j = random.choice(zeros)
            delta.fill(0)
            delta[n - 1 - i : n - 1 - i + n] -= h
            delta[i : i + n] -= rev_h
            delta[n - 1 - j : n - 1 - j + n] += h
            delta[j : j + n] += rev_h
            delta[n - 1] += 2
            delta[i - j + n - 1] -= 1
            delta[j - i + n - 1] -= 1
            new_corr = corr - delta
            new_max = int(new_corr.max())
            if new_max < cur_best:
                cur_best = new_max
                best_swap = (i, j)
                best_delta = delta.copy()
        if best_swap is None:
            no_improve += 1
            continue

        i, j = best_swap
        # apply swap
        h[i] = 0
        h[j] = 1
        # update index lists (O(n) but swaps are few)
        idx_i = ones.index(i)
        ones[idx_i] = j
        idx_j = zeros.index(j)
        zeros[idx_j] = i
        corr -= best_delta
        max_corr = cur_best
        no_improve = 0

        if max_corr < best_max:
            best_max = max_corr
            best_h = h.copy()
    bound = best_max * (2.0 / n)
    return best_h.astype(float), float(bound)

    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = float("inf")
    best_n = None
    target = 0.3808

    # possible resolutions
    n_cands = [192, 224, 256, 320, 384, 448]
    try:
        parent = np.array(initial_h_values, dtype=float)
        if parent.size not in n_cands:
            n_cands.append(parent.size)
    except NameError:
        parent = None
    n_cands = sorted(set(n for n in n_cands if n <= 500 and n % 2 == 0))

    for n in n_cands:
        if budget_s - (time.time() - start) < 10.0:
            break
        # number of restarts per resolution
        restarts = 4 if n <= 256 else 3
        for r in range(restarts):
            now = time.time()
            time_left = budget_s - (now - start)
            if time_left < 8.0:
                break
            # allocate time for this restart
            max_restart = min(30.0, time_left * 0.45)
            end_restart = now + max_restart

            # initialise
            if r == 0 and parent is not None and parent.size == n:
                h0 = _project_to_simplex_box(parent.copy(), n / 2.0)
            else:
                rand = np.random.rand(n)
                h0 = _project_to_simplex_box(rand, n / 2.0)

            # gradient descent (about 60% of allocated time)
            mid = now + max_restart * 0.6
            h_pg, c5_pg = _proj_grad_descent(h0, mid)
            if c5_pg < best_c5:
                best_h, best_c5, best_n = h_pg, c5_pg, n
                if best_c5 <= target:
                    return best_h.tolist(), float(best_c5), int(best_n)

            # binary hill‑climb from thresholded version
            if time.time() < end_restart:
                bin_h = np.zeros(n, dtype=np.int8)
                idx = np.argpartition(h_pg, -n // 2)[-n // 2 :]
                bin_h[idx] = 1
                h_bin, c5_bin = _binary_hill_climb(bin_h, end_restart)
                if c5_bin < best_c5:
                    best_h, best_c5, best_n = h_bin, c5_bin, n
                    if best_c5 <= target:
                        return best_h.tolist(), float(best_c5), int(best_n)

    # fallback (unlikely)

        n = 256
        h = np.zeros(n, dtype=int)
        h[np.random.choice(n, n // 2, replace=False)] = 1
        best_h = h.astype(float)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    return best_h.tolist(), float(best_c5), int(best_n)
