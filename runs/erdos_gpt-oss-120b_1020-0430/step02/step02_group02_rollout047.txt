def _ineq_constraint(x):

def _solve_h(initial_h, max_iter=300):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type':'ineq','fun':_ineq_constraint},
                   {'type':'eq','fun':_eq_constraint}]
    opts = {'maxiter': max_iter, 'ftol':1e-9, 'disp': False}
    res = optimize.minimize(lambda x: x[-1], x0,
                            method='SLSQP',
                            bounds=bounds,
                            constraints=constraints,
                            options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt

def _gradient_for_shift(h, shift):
    n = h.size
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        d = shift
        # source part
        src = 1.0 - h[d:]
        g[:n-d] += src
        # target part subtract
        g[d:] -= h[:n-d]
    else:
        d = -shift
        # source part (i where i+d < n)
        src = 1.0 - h[:n-d]
        g[d:] += src
        # target part subtract
        g[:n-d] -= h[d:]
    return g

import time

# -------------------------------------------------------------
# Feasibility projection (clip to [0,1] and enforce sum = n/2)
# -------------------------------------------------------------
def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h):
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _max_corr_and_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    return corr[idx], shift, corr

def _gradient_for_shift(h, shift):
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        d = shift
        g[: n - d] += 1.0 - h[d:]
        g[d:] -= h[: n - d]
    else:
        d = -shift
        g[d:] += 1.0 - h[: n - d]
        g[: n - d] -= h[d:]
    return g

def _gd_optimize(h0, end_time):
    n = h0.size
    h = h0.copy()
    best_c = _c5_from_h(h)
    step = 0.2
    while time.time() < end_time:
        _, shift, _ = _max_corr_and_shift(h)
        g = _gradient_for_shift(h, shift)
        cur = step
        improved = False
        for _ in range(10):
            trial = _project_to_feasible(h - cur * g, n)
            c = _c5_from_h(trial)
            if c < best_c - 1e-12:
                h, best_c = trial, c
                step = cur * 1.2
                improved = True
                break

                c = _c5_from_h(trial)
                if c < best_c:
                    h, best_c = trial, c
                step = 0.2
    return h, best_c

def _hill_climb_targeted(h, max_steps, time_limit):

    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
