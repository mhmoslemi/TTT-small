def anneal_binary(initial_h, n_iters, start_temp, end_temp, time_left, target_bound):
   best_h = initial_h.copy()
   best_val = max_corr_fft(best_h)
   best_c5 = (2.0 / best_h.size) * best_val
   cur_h = best_h.copy()
   cur_val = best_val
   cur_c5 = best_c5

   T = start_temp
   T_end = end_temp
   # schedule: exponential decreasing per iteration
   # compute cooling factor = (T_end/T_start)^(1/max_iters)
   if n_iters <= 0:
        return best_h, best_c5
   alpha = (T_end / T_start) ** (1.0 / n_iters)
   for it in range(n_iters):
        if time_left() < 0.5:
            break
        # propose swap
        # choose random indices of 1 and 0
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_val = max_corr_fft(cur_h)
        d = new_val - cur_val
        if d <= 0 or np.random.rand() < np.exp(-d / T):
           # accept
           cur_val = new_val
           cur_c5 = (2.0 / cur_h.size) * cur_val
           if cur_val < best_val:
               best_val = cur_val
               best_h = cur_h.copy()
               best_c5 = cur_c5
               if best_c5 < target_bound:
                   break
        else:
           # revert swap
           cur_h[i] = 1
           cur_h[j] = 0
        # cooling
        T *= alpha
   return best_h, best_c5

def anneal_binary(init_h, n_iters, start_temp, end_temp, time_left_func, target_bound):
    h = init_h.copy()
    cur_val = max_corr(h)
    cur_c5 = (2.0 / h.size) * cur_val
    best_h = h.copy()
    best_val = cur_val
    best_c5 = cur_c5
    if cur_c5 <= target_bound:
        return best_h, best_c5
    # temperature schedule
    if n_iters <= 0:
        return best_h, best_c5
    alpha = (end_temp / start_temp) ** (1.0 / n_iters) if start_temp > 0 else 0.0
    T = start_temp
    n = h.size
    half = n // 2
    for it in range(n_iters):
        if time_left_func() < 0.5:
            break
        # choose random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_val = max_corr(h)
        d = new_val - cur_val
        if d <= 0 or np.random.rand() < np.exp(-d / T):
            # accept
            cur_val = new_val
            cur_c5 = (2.0 / n) * cur_val

                if best_c5 <= target_bound:
                    break
        else:
            # revert
            h[i] = 1
            h[j] = 0
        # cooling
        T *= alpha
        # optional early break if T becomes very small
        if T < 1e-8:
            T = 0.0
    return best_h, best_c5

import time
import numpy as np


def _max_corr(h, L):
    """Maximum of h * (1‑h) correlation via FFT (any h∈[0,1])."""
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[: 2 * h.size - 1].max()


def _corr_array(h, L):
    """Full correlation array h·(1‑h)."""
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[: 2 * h.size - 1]


def _contrib_indices(h, shift):
    """Indices i where h[i]=1 and h[i+shift]=0 (valid range)."""
    n = h.size
    idx = np.arange(n)
    idx_s = idx + shift
    valid = (idx_s >= 0) & (idx_s < n)
    idx = idx[valid]
    idx_s = idx_s[valid]
    return idx[(h[idx] == 1) & (h[idx_s] == 0)]


def _anneal_binary(start_h, iters, t0, tf, time_left, target_c5):
    """Simulated annealing on binary vectors (exactly n/2 ones)."""
    n = start_h.size
    L = 1 << ((2 * n - 1).bit_length())
    h = start_h.copy()
    cur = _max_corr(h, L)
    best_h, best_val = h.copy(), cur
    if (2.0 / n) * best_val <= target_c5:
        return best_h, best_val
    if t0 <= 0:
        return best_h, best_val
    alpha = (tf / t0) ** (1.0 / max(iters, 1))
    T = t0
    for _ in range(iters):
        if time_left() < 0.5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0
        h[j] = 1
        new = _max_corr(h, L)
        d = new - cur
        if d <= 0 or (T > 0 and np.random.rand() < np.exp(-d / T)):
            cur = new
            if new < best_val:
                best_val = new
                best_h = h.copy()
                if (2.0 / n) * best_val <= target_c5:
                    break
        else:
            h[i] = 1
            h[j] = 0
        T *= alpha
        if T < 1e-12:
            T = 0.0
    return best_h, best_val


def _local_optimize_bin(h, max_allowed, attempts, time_left):
    """Greedy swaps aiming to push max correlation below max_allowed."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    for _ in range(attempts):
        if time_left() < 0.5:
            break
        corr = _corr_array(h, L)
        cur = corr.max()
        if cur <= max_allowed:
            break
        idxs = np.where(corr == cur)[0]
        shift = int(idxs[0] - (n - 1))
        contrib = _contrib_indices(h, shift)
        if contrib.size:
            i = int(np.random.choice(contrib))
        else:
            i = int(np.random.choice(np.where(h == 1)[0]))
        j = int(np.random.choice(np.where(h == 0)[0]))
        h[i] = 0
        h[j] = 1
        new = _max_corr(h, L)
        if new < cur:
            continue
        h[i] = 1
        h[j] = 0
    return h, _max_corr(h, L)
