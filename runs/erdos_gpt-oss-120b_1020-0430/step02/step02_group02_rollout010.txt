def _solve_slsqp(h0, max_iter=200):
    n = h0.size
    h0 = _project_to_feasible(h0, n)
    # initial t = max correlation for start
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # define constraints
    def ineq(x):
        h = x[:n]
        t = x[n]
        return t - np.correlate(h, 1.0 - h, mode='full')
    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    constraints = [{'type': 'ineq', 'fun': ineq},
                   {'type': 'eq', 'fun': eq}]
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = optimize.minimize(lambda x: x[n], x0, method='SLSQP', bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt

import time, numpy as np
from scipy import optimize

# -------------------------------------------------
# Basic utilities
# -------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust to achieve sum(h)=n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 = max ∫h(1−h(·+k)) using discrete correlation."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    if n_fine == n_coarse:
        return coarse_h.copy()
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse_h)


def _random_feasible_h(n: int) -> np.ndarray:
    """Random vector in [0,1] projected to the feasible affine subspace."""
    h = np.random.rand(n)
    return _project_to_feasible(h, n)


# -------------------------------------------------
# Local optimisation via SLSQP
# -------------------------------------------------
def _solve_slsqp(start_h: np.ndarray, max_iter: int = 200) -> np.ndarray:
    n = start_h.size
    h0 = _project_to_feasible(start_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[n]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": ineq},
            {"type": "eq",   "fun": eq}]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[n], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    return _project_to_feasible(h_opt, n)


# -------------------------------------------------
# Simulated annealing on the sub‑space
# -------------------------------------------------
def _anneal_h(start_h: np.ndarray, deadline: float,
              start_temp: float = 0.02, decay: float = 0.997) -> tuple:
    n = start_h.size
    best_h = start_h.copy()
    best_c = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    T = start_temp
    while time.time() < deadline and T > 1e-12:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = cur_h[i]
        max_up = 1.0 - cur_h[j]
        max_eps = min(max_down, max_up)
        if max_eps == 0:
            continue
        # step size scales with temperature
        scale = 0.5 + 0.5 * (T / start_temp)
        eps = (np.random.rand() * 2 - 1) * max_eps * scale
        eps = np.clip(eps, -max_eps, max_eps)
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = _c5_from_h(cand)
        d = cand_c - cur_c
        if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        T *= decay
    return best_h, best_c

def _optimise_n(n: int, base_h: np.ndarray, deadline: float) -> tuple:

    best_h, best_c = None, float("inf")
