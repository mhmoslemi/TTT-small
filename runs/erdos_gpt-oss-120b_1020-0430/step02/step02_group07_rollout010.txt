import time, numpy as np

def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final clip
    np.clip(h, 0.0, 1.0, out=h)
    return h

def random_feasible_vector(n, rng):
    target = n / 2.0
    # start with random uniform values
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        # fill target entries with 1: impossible if fractional, but just fill floor
        # use uniform positions
        # Use definition from earlier: same as previous code.
        # implement simple approach: set first floor(target) entries to 1
        k = int(np.floor(target))
        idx = rng.choice(n, k, replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def compute_c5(h):
    n = h.size
    # reverse h
    ar = h[::-1]
    b = 1.0 - h
    N = 2 * n
    H = np.fft.rfft(ar, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(H * B, n=N)
    corr = conv[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def perturb_move(cur, rng, scale):
    n = cur.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    max_eps = min(cur[i], 1.0 - cur[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(cur[j], 1.0 - cur[i])
        if max_eps <= 0:
            return cur, 0.0, None, None  # no move
        i, j = j, i
    eps = rng.random() * max_eps * scale
    new_h = cur.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h, eps, i, j

def hill_climb(h_start, rng, max_iters=2000, start_step=0.1):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
    # optional step decay
    # we can skip for simplicity
    return best_h, best_c

def simulated_annealing_limited(start_h, rng, time_limit, init_temp=0.02, decay=0.9995, scale=0.3):
    cur = start_h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    temp = init_temp
    n = cur.size
    start_temp = init_temp
    # deterministic loop with time check
    while time.time() < time_limit:
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute maximum epsilon allowed
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            max_eps = min(cur[j], 1.0 - cur[i])
            if max_eps <= 0:
                continue
            i, j = j, i
        eps = rng.random() * max_eps * scale
        new_h = cur.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / (temp + 1e-15)):
            cur = new_h
            cur_c = new_c
            if new_c < best_c:
                best = new_h.copy()
                best_c = new_c
        temp *= decay
    return best, best_c

def upsample_h(h, factor, rng):
    n = h.size
    # define positions normalized
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target, rng)
    return h_up

import time
import numpy as np

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keep 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def random_feasible_vector(n, rng):
    """Random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0.0:
        h[:] = 0.0
        k = int(np.floor(target))
        if k > 0:
            idx = rng.choice(n, k, replace=False)
            h[idx] = 1.0
    else:
        h *= target / s
    np.clip(h, 0.0, 1.0, out=h)
    return enforce_sum(h, target, rng)

def compute_c5(h):
    """Exact C5 using FFT‑based linear correlation."""
    n = h.size
    ar = h[::-1]               # reversed h
    b = 1.0 - h
    N = 2 * n
    H = np.fft.rfft(ar, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(H * B, n=N)
    corr = conv[: 2 * n - 1]   # linear correlation
    return np.max(corr) * (2.0 / n)

def hill_climb(h_start, rng, max_iters=3000, start_step=0.05):
    """Deterministic local improvement (transfer ε from i to j)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing_limited(start_h, rng, time_limit,
                                init_temp=0.02, decay=0.9995, scale=0.3):
    """SA run bounded by a wall‑clock time limit."""
    cur = start_h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    temp = init_temp
    start_temp = init_temp
    n = cur.size
    while time.time() < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0.0:
            max_eps = min(cur[j], 1.0 - cur[i])
            if max_eps <= 0.0:
                continue
            i, j = j, i
        cur_scale = scale * (temp / start_temp)
        eps = rng.random() * max_eps * cur_scale
        new_h = cur.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / (temp + 1e-15)):
            cur = new_h
            cur_c = new_c
            if new_c < best_c:
                best = new_h.copy()
                best_c = new_c
        temp *= decay
    return best, best_c

def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    np.clip(h_up, 0.0, 1.0, out=h_up)
    target = (n * factor) / 2.0
    return enforce_sum(h_up, target, rng)
