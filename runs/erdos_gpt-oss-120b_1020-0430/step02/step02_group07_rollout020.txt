def isotonic_regression_decreasing(v):
    # Input v array.
    # We'll perform PAVA for monotone decreasing.
    n = len(v)
    # We'll store block start indices, lengths, averages.
    avg = v.astype(float).copy()
    weight = np.ones_like(v, dtype=float)
    # We'll treat decreasing: we need avg[i] >= avg[i+1]. Since we process forward, maintain groups.
    # Implementation: use list of groups (start, end, sum, weight)
    groups = []
    for val in avg:
        groups.append([val, 1]) # sum, weight
        # Merge while decreasing condition violated: i.e., previous group's avg < current group's avg (since we want decreasing)
        while len(groups) >= 2 and groups[-2][0] / groups[-2][1] < groups[-1][0] / groups[-1][1]:
            # merge last two
            s1,w1 = groups[-2]
            s2,w2 = groups[-1]
            groups[-2] = [s1 + s2, w1 + w2]
            groups.pop()
    # Now assign avg as block average
    out = np.empty_like(v)
    idx = 0
    for s, w in groups:
        avg_val = s / w
        out[idx: idx+w] = avg_val
        idx += w
    return out

def subgradient_descent(h, target, rng, max_iters=2000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    for it in range(max_iters):
        # compute correlation & shift(s) achieving max
        corr = np.correlate(h, 1 - h, mode="full")
        maxc = corr.max()
        max_shifts = np.flatnonzero(corr == maxc) - (n-1)
        # compute subgradient
        g = np.zeros(n)
        for k in max_shifts:
            # overlapping region indices
            start_i = max(0, -k)
            end_i = min(n-1, n-1 - k)
            # slices
            i_slice = slice(start_i, end_i+1)
            j_slice = slice(start_i + k, end_i + k + 1)
            hi = h[i_slice]
            hj = h[j_slice]   # corresponds to h_{i+k}
            g[i_slice] += (1 - hj)
            g[j_slice] -= hi
        # step size
        alpha = 0.5 / np.sqrt(it+1) # or 0.1
        # update
        h_new = h - alpha * g
        # project onto [0,1] and sum
        h_new = project_bounds_and_sum(h_new, target)
        # optional monotonic projection using cummax (if desired)
        # compute C5
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # accept move if improvement else maybe accept with probability decreasing (like SA)
        h = h_new
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the SLSQP surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log‑sum‑exp (β scaling)."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    """Stochastic search with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (numeric drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def quick_refine(h, rng):
    """Fast refinement: a few β values and a short hill‑climb."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=800, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def deep_refine(h, rng):
    """Intensive refinement: larger β ladder and longer local searches."""
    n = h.size
    max_iter_slsqp = 200 if n > 200 else 300
    hill_iters = 800 if n > 200 else 1500
    anneal_iters = 2000 if n > 200 else 3000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter_slsqp)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    trial_sa, c_sa = simulated_annealing(
        best_h, rng, max_iters=anneal_iters, init_temp=0.03, decay=0.998, scale=0.2
    )
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def project_bounds_and_sum(v, target, max_iter=40):
    """Project onto [0,1] with exact sum = target via bisection on a shift."""
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)


def subgradient_descent(h0, target, rng, max_iters=4000):
    """Sub‑gradient descent on the exact max‑overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        maxc = corr.max()
        # all shifts achieving the max
        max_idxs = np.flatnonzero(corr == maxc) - (n - 1)
        g = np.zeros_like(h)
        for k in max_idxs:
            start_i = max(0, -k)
            end_i = min(n - 1, n - 1 - k)
            i_slice = slice(start_i, end_i + 1)
            j_slice = slice(start_i + k, end_i + k + 1)
            hi = h[i_slice]
            hj = h[j_slice]
            g[i_slice] += (1 - hj)
            g[j_slice] -= hi
        # step size schedule
        alpha = 0.5 / np.sqrt(it + 1)
        h = project_bounds_and_sum(h - alpha * g, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def long_simulated_annealing(h_start, rng, max_iters=8000, init_temp=0.05,
                             decay=0.998, scale=0.2):
    """Extended SA with more iterations and a slower cooling schedule."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise base solution
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = quick_refine(h0, rng)

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts with quick refinement
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.35:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample and deep refinement (up to 720 points)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.85 and n < 720:
        factor = 2
        h_up = upsample_h(best_h, factor=factor, rng=rng)
        n_up = h_up.size
        # deeper deterministic / stochastic refinement
        h_ref, c_ref = deep_refine(h_up, rng)
        # optional long SA for extra polishing
        if time.time() - start_time < budget_s * 0.90:
            h_ref, c_ref = long_simulated_annealing(h_ref, rng,
                                                    max_iters=6000,
                                                    init_temp=0.04,
                                                    decay=0.998,
                                                    scale=0.15)
        # optional sub‑gradient descent for fine tuning
        if time.time() - start_time < budget_s * 0.95:
            target_up = n_up / 2.0
            h_sg, c_sg = subgradient_descent(h_ref, target_up, rng,
                                             max_iters=3000)
            if c_sg < c_ref:
                h_ref, c_ref = h_sg, c_sg
        if c_ref < best_c:
            best_h, best_c = h_ref.copy(), c_ref
            n = n_up

    # ------------------------------------------------------------------
    # Final clean‑up (ensure feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
