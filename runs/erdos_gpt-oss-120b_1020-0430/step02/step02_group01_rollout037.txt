def worst_correlation_and_lag(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_corr = corr[max_idx]
    lag = max_idx - (len(h) - 1)  # lag = index shift
    return max_corr, lag

def gradient_descent_step(h, lag, step_factor=0.5):
    g = gradient_of_lag(h, lag)
    # find i_pos with max g, i_neg with min g

    i_neg = np.argmin(g)   # most negative

    max_delta = min(h[i_pos], 1 - h[i_neg])

    new_h[i_neg] += delta
    return new_h, True

def run(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # define candidate grid sizes
    grid_sizes = [48, 60, 72, 84, 96, 108, 120] # moderate
    # Possibly add 150 if time.

    # if initial_h_values defined, treat that as n0 and initial solution.
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0, 1)
        # adjust sum to exactly n0/2
        adjust_sum(h0, n0/2, rng)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_c = c0
        best_n = n0

    # Function to adjust sum:
    def adjust_sum(x, target, rng):
        diff = target - x.sum()
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, len(x))
            if diff > 0:
                add = min(diff, 1 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()

    # Define random feasible vector
    def random_feasible(n):
        # similar to previous
        h = rng.random(n)
        scale = (n/2) / h.sum()
        h *= scale
        h = np.clip(h, 0, 1)
        adjust_sum(h, n/2, rng)
        return h

    # Define gradient optimization:
    def gradient_optimize(h, max_iter=200, step_factor=0.6):
        n = h.size
        target = n / 2.0
        for it in range(max_iter):
            max_corr, lag = worst_correlation_and_lag(h)
            # compute gradient of worst lag
            g = gradient_of_lag(h, lag)
            # find pos, neg indices
            i_pos = np.argmax(g)
            i_neg = np.argmin(g)
            if g[i_pos] <= 0 or g[i_neg] >= 0:
                break
            # max possible delta
            max_delta = min(h[i_pos], 1 - h[i_neg])
            if max_delta <= 1e-12:
                break
            # Use step_factor (decreasing)
            delta = step_factor * max_delta
            # try several deltas: e.g., fractions 0.2,0.5,1.0 of max_delta to find best improvement
            best_local_h = h
            best_local_c = max_corr
            improved = False
            for frac in (0.2, 0.5, 1.0):
                d = frac * max_delta
                if d < 1e-12:
                    continue
                new_h = h.copy()
                new_h[i_pos] -= d
                new_h[i_neg] += d
                cnew = compute_c5(new_h)
                if cnew < best_local_c - 1e-12:
                    best_local_h = new_h
                    best_local_c = cnew
                    improved = True
                    # update i_pos,i_neg for next iteration?
                    # break early maybe
            if not improved:
                # reduce step_factor and continue?
                step_factor *= 0.5
                if step_factor < 1e-3:
                    break
            else:
                h = best_local_h
                # recompute target sum check (should be preserved)
                # but due to floating errors may be slight diff; we can adjust
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    # adjust small diff to a random index within bounds
                    i = rng.integers(0, n)
                    # add diff respecting bounds
                    if diff > 0:
                        add = min(diff, 1 - h[i])
                    else:
                        add = max(diff, -h[i])
                    h[i] += add
        return h

    # Define hill climb with random moves, but accept if improves.
    def hill_climb2(h, max_iters=2000, init_step=0.3):
        n = h.size
        step = init_step
        best_h = h.copy()
        best_c = compute_c5(h)

            i = rng.integers(0, n)

            max_eps = min(best_h[i], 1 - best_h[j])

    # final check: upscale if possible

        if best_n * 2 <= 200:

    # final feasibility correction

            else:

            best_h[i] += add

def gradient_of_lag(h, lag):
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        if L < n:
            g[:n - L] += 1 - h[L:]
            g[L:] -= h[:n - L]
    else:
        d = -lag
        if d < n:
            g[d:] += 1 - h[:n - d]
            g[:n - d] -= h[d:]
    return g

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_correlation_and_lag(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    max_corr = corr[idx]
    lag = idx - (h.size - 1)
    return max_corr, lag

def gradient_of_lag(h, lag):
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        if L < n:
            g[:n - L] += 1 - h[L:]
            g[L:] -= h[:n - L]
    else:
        d = -lag
        if d < n:
            g[d:] += 1 - h[:n - d]
            g[:n - d] -= h[d:]
    return g
