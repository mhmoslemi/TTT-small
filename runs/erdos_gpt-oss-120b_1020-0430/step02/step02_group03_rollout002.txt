def project_to_feasible(h, target):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Iteratively adjust sum
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # compute free indices where we can move in direction of diff
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        # apply update but ensure we don't overshoot bounds
        # for each free index: new = h[i] + delta, then clip
        h[free] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # correction for tiny numerical errors
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        # distribute diff to random free indices
        rng = np.random.default_rng()
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, len(h))
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
    return h
```

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def project_to_feasible(h, target):
    # clip then adjust sum iteratively
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # free indices where we can move towards diff
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        # apply delta but keep within bounds
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        rng = np.random.default_rng()
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(len(h))
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
    return h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # final sum fix
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_local(h0, rng, max_iters=3000, start_scale=0.3):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = start_scale
    decay = 0.9995
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        # slowly decay scale
        scale = max(scale * decay, 1e-4)
    return best_h, best_c

def de_step(pop, scores, F, CR, target, rng):
    pop_new = pop.copy()
    scores_new = scores.copy()
    pop_size, n = pop.shape
    for i in range(pop_size):
        # select three distinct individuals distinct from i
        # we can sample until uniqueness
        # simple method: choose three random indices, if any equals i replace with another
        idxs = rng.choice(pop_size, 3, replace=False)
        while i in idxs:
            idxs = rng.choice(pop_size, 3, replace=False)
        a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
        mutant = a + F * (b - c)
        mutant = np.clip(mutant, 0.0, 1.0)
        cross = rng.random(n) < CR
        if not np.any(cross):
            cross[rng.integers(n)] = True
        trial = np.where(cross, mutant, pop[i])
        trial = project_to_feasible(trial, target)
        trial_score = compute_c5(trial)
        if trial_score < scores[i]:
            pop_new[i] = trial
            scores_new[i] = trial_score
    return pop_new, scores_new

        vec = project_to_feasible(vec, target)
        pop[i] = vec

    # Additional diverse seeds (sinusoids, ramps) – add directly to population if space
    extra = []

    # linear ramps
    extra.append(project_to_feasible(xs / 2.0, target))
    extra.append(project_to_feasible(1.0 - xs / 2.0, target))

        if refined_c < best_c:
            best_h, best_c = refined_h, refined_c

    beta_list = [5.0, 20.0, 80.0, 200.0, 500.0]
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.95:

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def project_to_feasible(h, target):
    """Clip to [0,1] and adjust the sum to the required target."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        rng = np.random.default_rng()
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(len(h))
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
    return h


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine using the smooth log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_local(h0, rng, max_iters=3000, start_scale=0.3):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = start_scale
    decay = 0.9995
    for _ in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        scale = max(scale * decay, 1e-4)
    return best_h, best_c


def de_step(pop, scores, F, CR, target, rng):
    pop_new = pop.copy()
    scores_new = scores.copy()
    pop_size, n = pop.shape
    for i in range(pop_size):
        idxs = rng.choice(pop_size, 3, replace=False)
        while i in idxs:
            idxs = rng.choice(pop_size, 3, replace=False)
        a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
        mutant = a + F * (b - c)
        mutant = np.clip(mutant, 0.0, 1.0)
        cross = rng.random(n) < CR
        if not np.any(cross):
            cross[rng.integers(n)] = True
        trial = np.where(cross, mutant, pop[i])
        trial = project_to_feasible(trial, target)
        trial_score = compute_c5(trial)
        if trial_score < scores[i]:
            pop_new[i] = trial
            scores_new[i] = trial_score
    return pop_new, scores_new
