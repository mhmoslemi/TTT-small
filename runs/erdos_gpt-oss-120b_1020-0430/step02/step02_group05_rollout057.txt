import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C₅ from a feasible h (sum(h)=len/2)."""
    n = h.size
    q = np.correlate(h, h, mode="full")
    c = (n * 0.5 - q) * (2.0 / n)
    return float(c.max())


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    q = np.correlate(h, h, mode="full")
    corr = n * 0.5 - q
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return float(h.sum() - n * 0.5)


def _objective(x: np.ndarray) -> float:
    """Objective is the auxiliary variable t."""
    return float(x[-1])


def _solve_h(start_h: np.ndarray, max_iter: int = 500):
    """Refine a feasible h using SLSQP on the t‑formulation."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n * 0.5)
    q0 = np.correlate(h0, h0, mode="full")
    t0 = (n * 0.5 - q0).max()
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    h_opt = h0 if not res.success else res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n * 0.5)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape, then projected to the feasible simplex."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n * 0.5)


def _param_shape(n: int, a: float) -> np.ndarray:
    """
    One‑parameter family:
        a = length of zero side (0<a<0.5)
        support: [a,2-a] with linear ramps of length l=1-2a,
        plateau of length p=2a at height 1.
    """
    a = max(0.0, min(a, 0.5 - 1e-12))
    l = 1.0 - 2.0 * a
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.zeros_like(x)

    # left ramp
    mask = (x >= a) & (x < 1.0 - a)
    h[mask] = (x[mask] - a) / l

    # plateau
    mask = (x >= 1.0 - a) & (x < 1.0 + a)
    h[mask] = 1.0

    # right ramp
    mask = (x >= 1.0 + a) & (x < 2.0 - a)
    h[mask] = (2.0 - a - x[mask]) / l

    return _project_to_simplex_box(h, n * 0.5)
