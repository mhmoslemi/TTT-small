def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # logsumexp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0) for _ in range(n)]
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust randomly
        # similar to enforce_sum.
        for _ in range(2000):
            if abs(diff) < 1e-12: break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
```

def binary_hill_climb(h, rng, max_iters=5000):
    n = h.size
    target_ones = n // 2
    # ensure exactly target_ones ones
    h = np.rint(h).astype(int)
    cur = h.sum()
    if cur > target_ones:
        # turn random ones to zero
        ones_idx = np.where(h == 1)[0]
        to_zero = rng.choice(ones_idx, size=cur-target_ones, replace=False)
        h[to_zero] = 0
    elif cur < target_ones:
        zeros_idx = np.where(h == 0)[0]
        to_one = rng.choice(zeros_idx, size=target_ones-cur, replace=False)
        h[to_one] = 1
    best = h.copy()
    best_c = compute_c5(best)
    for it in range(max_iters):
        # pick random swap
        ones_idx = np.where(best == 1)[0]
        zeros_idx = np.where(best == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        new = best.copy()
        new[i] = 0
        new[j] = 1
        c = compute_c5(new)
        if c < best_c - 1e-9:  # small improvement tolerance
            best = new
            best_c = c
        # maybe also accept if equal random for exploration? ; but skip.
    return best.astype(float), best_c

def simulated_annealing_binary(h_start, rng, max_iters=8000, init_temp=0.05, decay=0.9995):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    temp = init_temp
    n = len(h)
    for it in range(max_iters):
        # pick swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        new = h.copy()
        new[i] = 0
        new[j] = 1
        new_c = compute_c5(new)
        if new_c < best_c:
            best_h = new.copy()
            best_c = new_c
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new
            cur_c = new_c
        temp *= decay
        if temp < 1e-8:
            temp = init_temp
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust continuous h in‐place to hit exactly target sum, keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def enforce_sum_binary(h, target_ones, rng):
    """Force a binary float vector to have exactly target_ones entries equal to 1."""
    cur = int(h.sum())
    if cur > target_ones:
        idx = np.where(h == 1)[0]
        to_zero = rng.choice(idx, size=cur - target_ones, replace=False)
        h[to_zero] = 0
    elif cur < target_ones:
        idx = np.where(h == 0)[0]
        to_one = rng.choice(idx, size=target_ones - cur, replace=False)
        h[to_one] = 1
    return h


def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def random_binary_vector(n, rng):
    """Create a random binary vector with exactly n/2 ones."""
    k = n // 2
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=k, replace=False)] = 1.0
    return h

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP optimisation of the soft‑max surrogate."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # small repair of the equality constraint
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def hill_climb(h_start, rng, max_iters=1200, start_step=0.4):
    """Simple continuous hill‑climb (move ε between coordinates)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def continuous_refine(h0, rng):
    """Series of SLSQP β‑ladders followed by a short hill‑climb."""
    best_h, best_c = h0.copy(), compute_c5(h0)
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
        try:
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=300)
        except Exception:
            continue
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:

    if c_hc < best_c:

    return best_h, best_c

# Fast binary local search (O(n) per swap)

def binary_hill_climb_fast(h0, rng, max_iters=4000):
    """Binary hill‑climb using O(n) correlation updates per swap."""
    n = h0.size
    h = h0.astype(float).copy()
    # initial correlation
    corr = np.correlate(h, 1 - h, mode="full")
    best_corr = corr.copy()
    best_h = h.copy()
    best_c = np.max(best_corr) * (2.0 / n)

    for _ in range(max_iters):
        # pick a random 1 and a random 0
        while True:
            i = rng.integers(0, n)
            if h[i] == 1:
                break
        while True:
            j = rng.integers(0, n)
            if h[j] == 0:
                break

        # simulate swap on a copy of corr
        new_corr = corr.copy()
        # removal at i (δ = -1)
        delta = -1.0
        for q in range(n):
            if q == i:
                continue
            s = q - i
            new_corr[n - 1 - s] += delta * (1.0 - h[q])
            new_corr[n - 1 + s] += -delta * h[q]
        # addition at j (δ = +1)
        delta = 1.0
        for q in range(n):
            if q == j:
                continue
            s = q - j
            new_corr[n - 1 - s] += delta * (1.0 - h[q])
            new_corr[n - 1 + s] += -delta * h[q]

        new_c = np.max(new_corr) * (2.0 / n)
        if new_c + 1e-12 < best_c:
            # accept swap
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            best_c = new_c
            best_h = h.copy()
    return best_h, best_c

def upsample_h(h, factor):
    """Linear up‑sampling preserving shape (no periodic wrap)."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    return np.clip(h_up, 0.0, 1.0)
