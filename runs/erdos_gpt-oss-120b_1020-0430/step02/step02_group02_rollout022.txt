def _targeted_hill_climb(h, max_iters, time_limit):
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    n = len(h)
    for t in range(max_iters):
        if time.time()>time_limit: break
        # compute correlation vector
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n-1)  # shift
        # compute candidate i set
        if k >= 0:
            i_range = np.arange(0, n - k)
            j_range = i_range + k
        else:
            # shift negative: i_range = -k..n-1 ; j=i+k within bounds
            i_range = np.arange(-k, n)
            j_range = i_range + k
        # i_set: i where h[i]=1 and h[j]=0
        i_set = i_range[(best_h[i_range] > 0.5) & (best_h[j_range] < 0.5)]
        # j_set: i where h[i]=0 and h[j]=1 (these are positions to swap)
        j_set = i_range[(best_h[i_range] < 0.5) & (best_h[j_range] > 0.5)]

        if i_set.size == 0 or j_set.size == 0:
            # fallback: random swap
            i, j = np.random.randint(n, size=2)
            while best_h[i] == best_h[j]:
                i, j = np.random.randint(n, size=2)
        else:
            i = np.random.choice(i_set)
            j = np.random.choice(j_set)
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]  # swap

        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
        # else maybe accept based on probability? Omit for simplicity
    return best_h, best_c

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Feasibility projection (bounds [0,1] and sum = n/2)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Linear constraints for SLSQP
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    """One SLSQP run starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Interpolation to a finer grid
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Random binary vector with exactly n/2 ones
# ----------------------------------------------------------------------
def _binary_random_h(n: int) -> np.ndarray:
    ones = n // 2
    zeros = n - ones
    arr = np.array([1] * ones + [0] * zeros, dtype=float)
    np.random.shuffle(arr)
    return arr


# ----------------------------------------------------------------------
# Targeted hill‑climb: swap a 1 with a 0 that improves the worst shift
# ----------------------------------------------------------------------
def _targeted_hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)

    for step in range(max_steps):
        if time.time() > time_limit:
            break

        # correlation vector and worst shift
        corr_full = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = int(np.argmax(corr_full))
        k = max_idx - (n - 1)

        # overlapping index ranges for this shift
        if k >= 0:
            i_idx = np.arange(0, n - k)
            j_idx = i_idx + k
        else:
            i_idx = np.arange(-k, n)
            j_idx = i_idx + k

        # positions i (value 1) with partner 0, and i (value 0) with partner 1
        mask_i = (best_h[i_idx] > 0.5) & (best_h[j_idx] < 0.5)
        mask_j = (best_h[i_idx] < 0.5) & (best_h[j_idx] > 0.5)

        i_candidates = i_idx[mask_i]
        j_base_candidates = i_idx[mask_j]  # partner index will be +k

        if i_candidates.size and j_base_candidates.size:
            i_swap = int(np.random.choice(i_candidates))
            j_swap = int(np.random.choice(j_base_candidates) + k)
            # guarantee i_swap != j_swap
            if i_swap == j_swap:
                continue
        else:
            # fallback: random opposite‑valued pair
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i_swap = int(np.random.choice(ones))
            j_swap = int(np.random.choice(zeros))

        cand = best_h.copy()
        cand[i_swap], cand[j_swap] = cand[j_swap], cand[i_swap]
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c

    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Initial construction (if supplied)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    init_n = init_h.size if init_h is not None else None

    # ------------------------------------------------------------------
    # Grid sizes to explore
    # ------------------------------------------------------------------
    n_candidates = []
    if init_n is not None:
        n_candidates.append(init_n)
    for mul in (1, 2, 3, 4):
        if init_n is not None:
            n = init_n * mul
        else:
            n = 64 * mul
        if n <= 384:
            n_candidates.append(n)
    n_candidates.extend([96, 112, 128, 160, 192, 224, 256, 320])
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Search over grid sizes
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Build a small collection of starting vectors for this n
        start_vectors = []

        if init_h is not None:
            if init_h.size != n:
                base = _upscale_h(init_h, n)
            else:
                base = init_h.copy()
            start_vectors.append(base)

        # a uniform half‑half vector
        start_vectors.append(np.full(n, 0.5))

        # two random binary vectors
        for _ in range(2):
            start_vectors.append(_binary_random_h(n).astype(float))

        for base_h in start_vectors:
            if time_left() < 2.0:
                break

            base_h = _project_to_feasible(base_h, n)

            is_binary = np.all((base_h == 0) | (base_h == 1))

            # --------------------------------------------------------------
            # Continuous optimisation (skip if already binary)
            # --------------------------------------------------------------
            if not is_binary:
                max_iter = 300
                if time_left() < 30:
                    max_iter = 50
                h_opt, c_val = _solve_h(base_h, max_iter=max_iter)
            else:
                h_opt = base_h
                c_val = _c5_from_h(h_opt)

            # --------------------------------------------------------------
            # Record best so far
            # --------------------------------------------------------------
            if c_val < best_c:
                best_h, best_c, best_n = h_opt, c_val, n

            # --------------------------------------------------------------
            # Targeted hill‑climb on a binary version of the solution
            # --------------------------------------------------------------
            # Produce a binary copy (exact half‑ones)
            bin_h = (h_opt >= 0.5).astype(float)
            cur_ones = bin_h.sum()
            target_ones = n // 2
            if cur_ones > target_ones:
                idx = np.where(bin_h == 1)[0]
                to_zero = np.random.choice(idx, int(cur_ones - target_ones), replace=False)
                bin_h[to_zero] = 0
            elif cur_ones < target_ones:
                idx = np.where(bin_h == 0)[0]
                to_one = np.random.choice(idx, int(target_ones - cur_ones), replace=False)
                bin_h[to_one] = 1

            limit = start + budget_s - 0.2
            bin_h, bin_c = _targeted_hill_climb(bin_h, max_steps=8000, time_limit=limit)

            if bin_c < best_c:
                best_h, best_c, best_n = bin_h, bin_c, n

            if best_c < 0.3808:
                break

        if best_c < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final sanity projection and evaluation
    # ------------------------------------------------------------------
    if best_h is not None:
        best_h = _project_to_feasible(best_h, best_n)
        best_c = _c5_from_h(best_h)
    else:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
