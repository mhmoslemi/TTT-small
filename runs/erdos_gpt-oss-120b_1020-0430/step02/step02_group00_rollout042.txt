def _solve_h(initial_h, max_iter=300):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr
    def eq(x):
        return np.sum(x[:n]) - n/2.0
    cons = [{'type':'ineq','fun':ineq},

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter':max_iter, 'ftol':1e-9, 'disp':False})

        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)

import time
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
# Projection onto feasible set: 0≤h≤1 and sum(h)=n/2
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction (rare)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 value for a (already feasible) step function h
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# SLSQP refinement (minimize the auxiliary variable t = max correlation)
# ----------------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    res = opt.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Linear up‑sampling from a coarse grid
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Trapezoid (two ramps + plateau) parametrised by ramp width r∈[0,2/3]
# ----------------------------------------------------------------------
def _trapezoid_h(n: int, r: float) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    if r <= 1e-12:
        h = np.where((x >= 0.5) & (x <= 1.5), 1.0, 0.0)
    else:
        a = 0.5 - 0.75 * r          # left boundary of support
        p = 1.0 - r / 2.0           # length of the top plateau
        h = np.zeros_like(x)
        left = (x >= a) & (x < a + r)
        h[left] = (x[left] - a) / r
        plateau = (x >= a + r) & (x <= a + r + p)
        h[plateau] = 1.0
        right = (x > a + r + p) & (x <= 2.0 - a)
        h[right] = (2.0 - a - x[right]) / r
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_feasible(h, n)
    return h


# ----------------------------------------------------------------------
# Grid search for the best ramp width r for a given resolution n
# ----------------------------------------------------------------------
def _best_r_for_n(n: int) -> tuple:

    best_c5 = float("inf")
    # coarse grid
    for r in np.linspace(0.0, 2.0 / 3.0, 31):

            if c5 < best_c5:

                best_r = r
    return best_r, best_c5


# ----------------------------------------------------------------------
# Projected gradient descent that reduces the current maximal correlation
# ----------------------------------------------------------------------
def _gradient_descent_optimize(h0: np.ndarray, max_iter: int = 300, step0: float = 0.2, patience: int = 20) -> tuple:
    n = h0.size
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_val = _c5_from_h(h)
    no_improve = 0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        L = max_idx - (n - 1)  # integer shift
        grad = np.zeros_like(h)
        if L >= 0:
            grad[: n - L] = 1.0 - h[L:]
            grad[L:] -= h[: n - L]
        else:
            p = -L
            grad[p:] = 1.0 - h[: n - p]
            grad[: n - p] -= h[p:]
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_feasible(h_new, n)
        val = _c5_from_h(h_new)
        if val < best_val - 1e-9:
            best_val = val
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve >= patience:
            break
    return best_h, best_val

        best_c5 = _c5_from_h(best_h)

    else:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
