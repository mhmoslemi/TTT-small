def local_search_binary(h, max_iters, time_left):
    n = len(h)
    best_corr = np.correlate(h, 1-h, mode='full')
    best_max = best_corr.max()
    best_h = h.copy()
    while max_iters>0 and time_left() > 0:
        # generate random candidate swaps
        num_candidates = min(200, n//2) # e.g., 200
        candidates = []
        for _ in range(num_candidates):
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            candidates.append((i,j))
        # evaluate each candidate
        improvements = []
        for i,j in candidates:
            delta = compute_delta_corr(i, j, h)
            new_max = (best_corr + delta).max()
            improvements.append((new_max, i, j, delta))
        # choose best improvement (minimum new_max)
        best_candidate = min(improvements, key=lambda x: x[0])
        new_max, i, j, delta = best_candidate
        if new_max < best_max - 1e-9:
            # accept improvement
            h[i], h[j] = 0, 1
            # update corr
            best_corr = best_corr + delta
            best_max = new_max
            # update ones_idx etc.
        else:
            # no improvement: maybe accept random uphill with probability
            # else break
            pass
        max_iters -= 1
    return best_h, best_max

def binary_from_initial(initial_h):
    h = np.array(initial_h, dtype=float)
    n = h.size
    # threshold at 0.5
    binary = (h >= 0.5).astype(int)
    ones = binary.sum()
    target = n // 2
    if ones > target:
        # need to turn off (ones-target) ones, choose those with smallest h value
        surplus = ones - target
        # sort indices of ones by h value ascending
        idx = np.where(binary == 1)[0]
        order = idx[np.argsort(h[idx])]
        binary[order[:surplus]] = 0
    elif ones < target:
        deficit = target - ones
        idx = np.where(binary == 0)[0]
        order = idx[np.argsort(-h[idx])]  # descending h for zeros
        binary[order[:deficit]] = 1
    return binary
```

def minmax_overlap_sa(h, time_left, max_iters=100000):
    n = len(h)
    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    best_corr = corr.copy()
    # maintain lists
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    # define temperature schedule
    # parameters
    iter_count = 0
    init_temp = 5.0
    final_temp = 0.01
    total_iters_est = 200000
    while time_left() > 0.5 and iter_count < total_iters_est:
        # temperature linear decay
        t = init_temp + (final_temp - init_temp) * (iter_count / total_iters_est)
        # propose random swap
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # compute delta
        delta = _delta_for_swap(i, j, h)
        new_corr = corr + delta
        new_max = new_corr.max()
        diff = new_max - cur_max
        if diff <= 0 or np.random.rand() < np.exp(-diff / max(t, 1e-9)):
            # accept
            h[i] = 0
            h[j] = 1
            # update indices arrays efficiently: swap
            # remove i from ones, add to zeros; similarly for j.
            ones_idx = ones_idx[ones_idx != i]
            zeros_idx = zeros_idx[zeros_idx != j]
            # add i to zeros and j to ones
            zeros_idx = np.append(zeros_idx, i)
            ones_idx = np.append(ones_idx, j)
            # update corr and cur_max
            corr = new_corr
            cur_max = new_max
            # track best
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                best_corr = corr.copy()
        # maybe random restart

            corr = best_corr.copy()
            cur_max = best_max
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
        iter_count += 1
    return best_h, best_max

def _delta_for_swap(i, j, h):
    n = len(h)
    pad = np.concatenate([np.zeros(n, dtype=int), h, np.zeros(n, dtype=int)])

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # indices that can be moved in the needed direction
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a (already feasible) step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift (vectorised)."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    # correlation h * (1-h) (full linear correlation)
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Objective: minimise auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    """Run a single SLSQP optimisation from a feasible start."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
