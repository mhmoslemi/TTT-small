    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.2  # maybe small
        for _ in range(steps_per_beta):
            # compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient raw
            g_raw = np.ones(n)
            # iterate over k
            # compute shift index mapping: for idx in range(len(w)), shift = idx - (n - 1)
            # can vectorize? but loop maybe fine.
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    # subtract wk * h shifted forward/backward
                    g_raw[:n - k] -= wk * h[k:]
                    g_raw[k:] -= wk * h[:n - k]
                else:
                    kk = -k
                    g_raw[kk:] -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            # project
            new_h = project_onto_box_sum(new_h, n / 2.0)
            # enforce bounds
            new_h = np.clip(new_h, 0.0, 1.0)
            c_new = compute_c5(new_h)
            # acceptance: if better, accept; else maybe accept with small prob
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # simulated annealing acceptance based on temperature = 1/(beta? maybe)
                # Use simple acceptance probability decreasing with beta.
                if rng.random() < np.exp(-(c_new - best_c) * 10):  # tune factor
                    h = new_h
            # step decay
            step *= 0.995
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with sum exactly target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


# ---------- smooth‑max gradient descent ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)            # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient via convolutions (vectorised)
            conv_f = np.convolve(w[::-1], h, mode="full")
            conv_b = np.convolve(w, h, mode="full")
            mid = n - 1
            sum_f = conv_f[mid : mid + n]
            sum_b = conv_b[mid : mid + n]
            g_raw = 1.0 - sum_f - sum_b
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # occasional uphill move
                if rng.random() < 0.005:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP pass using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- simple hill‑climb on the exact objective ----------
def simple_hill_climb(h, rng, iters=800):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for _ in range(iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
    return best_h, best_c


# ---------- binary refinement ----------
def binary_refine(h_cont, rng, max_iters=2000):
    """Round to 0/1 and improve by swaps."""
    n = h_cont.size
    binary = (h_cont >= 0.5).astype(float)
    target_ones = n // 2
    cur = int(binary.sum())
    diff = target_ones - cur
    if diff != 0:
        # order by closeness to 0.5
        dist = np.abs(h_cont - 0.5)
        order = np.argsort(dist)
        if diff > 0:
            zeros = np.where(binary == 0)[0]
            cand = order[np.isin(order, zeros)][:diff]
            binary[cand] = 1.0
        else:
            ones = np.where(binary == 1)[0]
            cand = order[np.isin(order, ones)][: -diff]
            binary[cand] = 0.0
    best = binary.copy()
    best_c = compute_c5(best)
    idx_one = np.where(best == 1)[0]
    idx_zero = np.where(best == 0)[0]
    for _ in range(max_iters):
        if len(idx_one) == 0 or len(idx_zero) == 0:
            break
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        best[i], best[j] = 0.0, 1.0
        c = compute_c5(best)
        if c < best_c:
            best_c = c
            idx_one = np.where(best == 1)[0]
            idx_zero = np.where(best == 0)[0]
        else:
            # revert most of the time
            if rng.random() < 0.01:
                idx_one = np.where(best == 1)[0]
                idx_zero = np.where(best == 0)[0]
            else:
                best[i], best[j] = 1.0, 0.0
    return best, best_c
