def gradient_descent(h0, max_iter, time_limit):

    h = h0.copy()
    target_sum = len(h)/2
    best_c5 = c5_from_h(h)
    best_h = h.copy()
    start_time = time.time()
    α = 0.5  # initial step size

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break

        # compute active shift
        corr, max_corr, L = compute_corr_and_max(h)
        # gradient
        g = gradient_from_L(h, L)

        # try step
        improved = False
        α_try = α
        for _ in range(10):
            h_candidate = project_to_feasible(h - α_try * g, target_sum)
            c5_candidate = c5_from_h(h_candidate)
            if c5_candidate < best_c5 - 1e-12:
                improved = True
                break
            α_try *= 0.5
        if not improved:
            # reduce step size for future?
            α *= 0.5
            if α < 1e-6:
                # random perturb to escape
                noise = np.random.randn(len(h)) * 0.01
                h_candidate = project_to_feasible(h + noise, target_sum)
                h = h_candidate
                best_c5 = c5_from_h(h)  # maybe not improved
                α = 0.5
                continue
            else:
                continue
        # accept
        h = h_candidate

def project_to_feasible(h, target_sum, tol=1e-10, max_iter=50):
    # clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) <= tol:
        return h
    # binary search for lambda
    lo = -1.0
    hi = 1.0
    # ensure hi large enough:
    while np.clip(h + hi, 0.0, 1.0).sum() < target_sum:
        hi *= 2.0
    while np.clip(h + lo, 0.0, 1.0).sum() > target_sum:
        lo *= 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        s_mid = np.clip(h + mid, 0.0, 1.0).sum()
        if s_mid < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi)/2.0, 0.0, 1.0)
    return h_proj

def gradient_descent(h, max_iter, time_limit):
    start = time.time()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = c5_from_h(best_h)
    # pre-calc maybe gradient per iteration spontaneously
    α = 0.5
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # compute correlation and active L
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        L = max_idx - (n - 1)
        # compute gradient
        g = np.zeros(n)
        if L >= 0:
            g[:n - L] += (1 - best_h[L:])
            g[L:] -= best_h[:n - L]
        else:
            Lneg = -L
            g[Lneg:] += (1 - best_h[:n - Lneg])
            g[:n - Lneg] -= best_h[Lneg:]
        # attempt step
        improved = False
        α_try = α
        for _ in range(10):
            h_new = project_to_feasible(best_h - α_try * g, target_sum)
            c5_new = c5_from_h(h_new)
            if c5_new < best_c5 - 1e-12:
                improved = True
                break
            α_try *= 0.5
        if not improved:
            # reduce α
            α *= 0.5
            if α < 1e-6:
                # random perturbation
                noise = np.random.randn(n) * 0.01
                best_h = project_to_feasible(best_h + noise, target_sum)
                best_c5 = c5_from_h(best_h)
                α = 0.5
            continue
        # accept
        best_h = h_new
        best_c5 = c5_new
        # optionally increase step
        α = α_try * 1.1
        # optional occasional random jitter to escape flat region
        if it % 200 == 0 and it > 0:
            # small random shake
            shake = np.random.randn(n) * 0.005
            best_h = project_to_feasible(best_h + shake, target_sum)
            best_c5 = c5_from_h(best_h)
    return best_h, best_c5

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to exactly `target` by a uniform shift."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) <= tol:
        return h
    lo, hi = -1.0, 1.0
    # expand bounds if needed
    while np.clip(h + lo, 0.0, 1.0).sum() > target:
        lo *= 2.0
    while np.clip(h + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        s_mid = np.clip(h + mid, 0.0, 1.0).sum()
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for a feasible step function."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _interpolate(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n = h.size
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)


# ----------------------------------------------------------------------
# Gradient‑descent optimiser (projected sub‑gradient on the max‑overlap)
# ----------------------------------------------------------------------
def _gradient_descent(h0: np.ndarray, max_iter: int, time_limit: float) -> tuple:
    """Projected sub‑gradient descent on max_{k}∫h(1−h(·+k))."""
    start_time = time.time()
    n = h0.size
    target = n / 2.0
    h = _project_to_feasible(h0, target)
    best_c5 = _c5_from_h(h)
    best_h = h.copy()
    step = 0.5
    stagnation = 0

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break

        # correlation and active shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        L = idx - (n - 1)          # signed shift

        # gradient of the active term (sub‑gradient of the max)
        g = np.zeros(n)
        if L >= 0:
            g[: n - L] += 1.0 - h[L:]
            g[L:]     -= h[: n - L]
        else:
            Lm = -L
            g[Lm:]    += 1.0 - h[: n - Lm]
            g[: n - Lm] -= h[Lm:]

        # try a step with back‑tracking
        improved = False
        cur_step = step
        for _ in range(12):
            h_candidate = _project_to_feasible(h - cur_step * g, target)
            c5_candidate = _c5_from_h(h_candidate)
            if c5_candidate < best_c5 - 1e-14:
                h, best_c5, best_h = h_candidate, c5_candidate, h_candidate.copy()
                improved = True
                break
            cur_step *= 0.5

        if not improved:
            step *= 0.5
            stagnation += 1
            if step < 1e-6 or stagnation > 200:
                # random shake to escape flat regions
                h = _project_to_feasible(h + np.random.randn(n) * 0.01, target)
                best_c5 = _c5_from_h(h)
                best_h = h.copy()
                step = 0.5
                stagnation = 0
            continue

        # successful step – modestly increase step size
        step = cur_step * 1.1
        stagnation = 0

        # occasional jitter (helps avoid tiny cycles)
        if (it + 1) % 500 == 0:
            h = _project_to_feasible(h + np.random.randn(n) * 0.005, target)
            best_c5 = _c5_from_h(h)
            best_h = h.copy()

    return best_h, best_c5
