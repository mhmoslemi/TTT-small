            max_eps = min(h[target], 1.0 - h[source])
            if max_eps <= 0: continue

        eps = max_eps * rng.random() * (0.1 + 0.9 * T / T0)  # scale

        h_new = h.copy()

            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        # temperature decay
        T *= decay
        iter += 1
        # optionally adjust after many steps
    return best_h, best_c

def anneal_optimize(h, rng, time_left):
    n = h.size
    target_sum = n/2
    best_h = h.copy()
    best_c = compute_c5(best_h)
    current_c = best_c
    T0 = best_c * 0.5
    T = T0
    decay = 0.9994
    start = time.time()
    while time.time() - start < time_left:
        # get source and target
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Determine direction
        # We'll try both directions randomly
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0:
            # try opposite direction
            src, dst = dst, src
            max_eps = min(h[src], 1.0 - h[dst])
            if max_eps <= 0:
                continue
        # step size scaling
        eps = max_eps * rng.random() * (0.05 + 0.95 * (T / T0))
        # apply
        h[src] -= eps
        h[dst] += eps
        new_c = compute_c5(h)
        delta = new_c - current_c
        accept = False
        if delta < 0:
            accept = True
        else:
            p = np.exp(-delta / (T + 1e-12))
            if rng.random() < p:
                accept = True
        if accept:
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[src] += eps
            h[dst] -= eps
        # update temperature
        T *= decay
        if T < 1e-12:
            T = 1e-12
    return best_h, best_c, current_c
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 8) -> np.ndarray:
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for _ in range(500):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def anneal_optimize(h_start: np.ndarray, rng: np.random.Generator,
                    time_limit: float) -> tuple[np.ndarray, float, float]:
    """
    Simulated‑annealing style optimisation.
    Returns (best_h, best_c, current_c) after `time_limit` seconds.
    """
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    current_c = best_c

    # temperature parameters
    T0 = max(best_c * 3.0, 1e-3)   # start fairly hot
    T = T0
    decay = 0.9995

    start = time.time()
    while time.time() - start < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # pick move direction randomly
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0.0:
            # try opposite direction
            src, dst = dst, src
            max_eps = min(h[src], 1.0 - h[dst])
            if max_eps <= 0.0:
                continue
        # step size scaled by temperature (large early, small later)
        scale = 0.05 + 0.95 * (T / T0)   # between 0.05 and 1
        eps = max_eps * rng.random() * scale
        # apply move
        h[src] -= eps
        h[dst] += eps
        new_c = compute_c5(h)
        delta = new_c - current_c
        if delta < 0.0:
            accept = True

            accept = rng.random() < prob
        if accept:
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[src] += eps
            h[dst] -= eps
        T *= decay
        if T < 1e-12:
            T = 1e-12
    return best_h, best_c, current_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise from parent if available, otherwise random binary-like start.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
        best_h, best_c, best_n = h, compute_c5(h), n
    else:
        best_n = 120
        best_h = random_feasible_vector(best_n, rng)
        best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1 – long annealing on the current best.
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 0:
        phase_time = min(time_left * 0.55, 600.0)
        best_h, best_c, _ = anneal_optimize(best_h.copy(), rng, phase_time)
        # brief gradient polishing
        if time.time() - start_time < budget_s * 0.6:
            best_h = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=12)
            best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 2 – random restarts (if budget permits).
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 120 or best_c <= 0.3806:
            break
        slice_time = min(150.0, time_left * 0.4)

        # start from a fresh random vector
        h0 = random_feasible_vector(best_n, rng)

        # anneal this candidate
        h_cand, c_cand, _ = anneal_optimize(h0, rng, slice_time)

        # short gradient polishing
        if time.time() - start_time < budget_s * 0.9:
            h_cand = gradient_refine(h_cand, rng, step=0.02, max_iter=6)
            c_cand = compute_c5(h_cand)

        if c_cand < best_c:
            best_h, best_c = h_cand, c_cand

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to double resolution (if time permits).
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if best_n * 2 <= 240 and time_left > 30:
        up_h = np.repeat(best_h, 2)
        up_n = up_h.size
        up_time = min(time_left * 0.5, 200.0)
        up_h, up_c, _ = anneal_optimize(up_h, rng, up_time)
        up_h = gradient_refine(up_h, rng, step=0.02, max_iter=8)
        up_c = compute_c5(up_h)
        if up_c < best_c:
            best_h, best_c, best_n = up_h, up_c, up_n

    # ------------------------------------------------------------------
    # Final feasibility enforcement (tiny numerical drift may appear).
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
