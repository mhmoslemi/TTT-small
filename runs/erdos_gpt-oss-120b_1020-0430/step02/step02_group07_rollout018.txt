    cur_h = h_init.copy()
    cur_c = compute_c5_bin(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = start_temp
    start = time.time()

        # swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        new_h[j] = 1

        T *= cooling

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5_bin(h: np.ndarray) -> float:
    """Exact C5 for a binary step function h on [0,2] (dx = 2/n)."""
    # cast to small int to avoid overflow, correlation returns int
    corr = np.correlate(h.astype(np.int16), (1 - h).astype(np.int16), mode="full")
    return int(corr.max()) * (2.0 / h.size)


def round_to_binary(arr: np.ndarray) -> np.ndarray:
    """Greedy rounding to a binary vector with exactly n//2 ones."""
    n = arr.size
    half = n // 2
    idx = np.argsort(-arr)  # descending
    bin_vec = np.zeros(n, dtype=np.uint8)
    bin_vec[idx[:half]] = 1
    return bin_vec


def generate_random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random binary vector with exactly n//2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=np.uint8)
    h[rng.choice(n, size=half, replace=False)] = 1
    return h


# ----------------------------------------------------------------------
# Local deterministic improvement: focus on the worst shift
# ----------------------------------------------------------------------
def targeted_hill_climb_bin(
    h: np.ndarray, rng: np.random.Generator, max_iters: int = 800
) -> tuple[np.ndarray, float]:
    cur = h.copy()
    cur_c = compute_c5_bin(cur)
    n = cur.size
    for _ in range(max_iters):
        corr = np.correlate(cur.astype(np.int16), (1 - cur).astype(np.int16), mode="full")
        max_idx = int(np.argmax(corr))
        max_val = int(corr[max_idx])
        if max_val == 0:
            break
        k = max_idx - (n - 1)  # shift (positive → right)
        if k >= 0:
            # mismatches: 1 at i, 0 at i+k
            mism_i = np.where((cur[: n - k] == 1) & (cur[k:] == 0))[0]
        else:
            kk = -k
            mism_i = np.where((cur[kk:] == 1) & (cur[: n - kk] == 0))[0] + kk
        if mism_i.size == 0:
            break
        i = rng.choice(mism_i)          # a 1 that causes a mismatch
        zeros = np.where(cur == 0)[0]    # any 0
        if zeros.size == 0:
            break
        j = rng.choice(zeros)           # position to receive the 1
        new = cur.copy()
        new[i] = 0
        new[j] = 1
        new_c = compute_c5_bin(new)
        if new_c < cur_c:
            cur, cur_c = new, new_c
    return cur, cur_c


# ----------------------------------------------------------------------
# Simulated annealing on binary vectors (swap moves)
# ----------------------------------------------------------------------
def simulated_annealing_bin(
    h_start: np.ndarray,
    rng: np.random.Generator,
    time_limit: float,
    start_temp: float = 0.01,
    cooling: float = 0.9995,
) -> tuple[np.ndarray, float]:
    cur = h_start.copy()
    cur_c = compute_c5_bin(cur)
    best = cur.copy()
    best_c = cur_c
    T = max(start_temp, 1e-8)
    start = time.time()
    n = cur.size
    # pre‑allocate index arrays to avoid repeated allocation
    while time.time() - start < time_limit:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = cur.copy()
        new[i] = 0
        new[j] = 1
        new_c = compute_c5_bin(new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur, cur_c = new, new_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
        T *= cooling
        if T < 1e-9:                     # prevent temperature from vanishing
            T = start_temp
    return best, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Choose problem size and initial vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.size
        if n % 2 != 0:
            # odd length – fall back to a fresh even size (360)
            n = 360
            cur_bin = generate_random_binary(n, rng)
        else:
            cur_bin = round_to_binary(init_arr)
    else:
        n = 360
        cur_bin = generate_random_binary(n, rng)

    # initial local improvement
    cur_bin, cur_c = targeted_hill_climb_bin(cur_bin, rng, max_iters=1200)

    best_h = cur_bin.copy()
    best_c = cur_c

    # --------------------------------------------------------------
    # Target C5 (integer mismatches) for early stopping
    # --------------------------------------------------------------
    target_m = int(np.floor(0.3808 * n / 2 + 1e-12))
    target_c = 2.0 * target_m / n

    # --------------------------------------------------------------
    # Main search loop (restarts + annealing)
    # --------------------------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed > budget_s * 0.93:
            break

        # ----- restart with a fresh random binary vector -----
        trial = generate_random_binary(n, rng)
        trial, trial_c = targeted_hill_climb_bin(trial, rng, max_iters=800)
        if trial_c < best_c:
            best_h, best_c = trial, trial_c

        # ----- short simulated annealing on the current best -----
        remaining = budget_s - (time.time() - start_time)
        # we reserve a small buffer for final clean‑up
        if remaining < 2.0:
            break
        sa_time = min(remaining * 0.12, 5.0)  # at most 5 s per SA call
        if sa_time < 0.5:
            break
        trial, trial_c = simulated_annealing_bin(
            best_h,
            rng,
            time_limit=sa_time,
            start_temp=best_c * 0.02,
            cooling=0.9995,
        )
        if trial_c < best_c:
            best_h, best_c = trial, trial_c

        if best_c <= target_c + 1e-12:
            break

    # --------------------------------------------------------------
    # Final fine‑grained deterministic refinement
    # --------------------------------------------------------------
    best_h, best_c = targeted_hill_climb_bin(best_h, rng, max_iters=3000)

    # --------------------------------------------------------------
    # Return results (list of floats, C5 value, number of points)
    # --------------------------------------------------------------
    return best_h.astype(float).tolist(), float(best_c), int(n)
