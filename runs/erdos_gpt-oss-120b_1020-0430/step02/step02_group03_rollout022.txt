def build_h_from_u(u):
    """Construct symmetric step function h on [0,2] from half‑vector u."""
    n2 = u.size
    n = n2 * 2
    h = np.empty(n, dtype=np.float64)
    h[:n2] = u
    h[n2:] = 1.0 - u[::-1]
    return h


def compute_c5(h):
    """Exact discrete C5 for a full‑resolution h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def surrogate_val(u, beta):
    """Smooth log‑sum‑exp surrogate of the max overlap (no dx factor)."""
    h = build_h_from_u(u)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_terms = np.exp(bc - m)
    return (m + np.log(np.sum(exp_terms))) / beta


def surrogate_grad(u, beta):
    """Gradient of the surrogate w.r.t. the half‑vector u."""
    h = build_h_from_u(u)
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_terms = np.exp(bc - m)
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp                     # softmax weights, length 2n‑1

    grad_h = np.zeros_like(h)                  # gradient w.r.t. full h
    offset = n - 1

    # loop over all shifts k = idx - offset
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            # (1 - h[p + k]) part
            grad_h[: n - k] += wk * (1.0 - h[k:])
            # -h[p - k] part
            grad_h[k:] -= wk * h[: n - k]
        else:

            # (1 - h[p + k]) part (p >= d)
            grad_h[d:] += wk * (1.0 - h[: n - d])
            # -h[p - k] part (p <= n-d-1)
            grad_h[: n - d] -= wk * h[d:]

    n2 = n // 2
    grad_u = grad_h[:n2] - grad_h[n2:][::-1]
    return grad_u


def perturb_u(u, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = u.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:

    max_eps = min(u[i], 1.0 - u[j])

    new_u = u.copy()
    new_u[i] -= eps
    new_u[j] += eps
    return new_u

    best_u = u.copy()
    best_c = compute_c5(build_h_from_u(u))
    n = u.size
    decay = max(1, max_iters // 10)
    step = step_scale
    for it in range(max_iters):
        new_u = perturb_u(u, rng, scale=step)
        c = compute_c5(build_h_from_u(new_u))
        if c < best_c:
            best_c = c
            best_u = new_u.copy()
            u = new_u
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_u, best_c


# ---------- main run function ----------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- determine problem dimension -----
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
        n = h_init.size
        if n % 2 == 1:           # enforce even length
            n -= 1
            h_init = h_init[:n]
    else:
        n = 180
        h_init = None
    n2 = n // 2
    bounds = [(0.0, 1.0)] * n2

    # ----- obtain a symmetric starting half‑vector -----
    if h_init is not None:
        u0_sym = (h_init[:n2] + (1.0 - h_init[-n2:][::-1])) * 0.5
        u0_sym = np.clip(u0_sym, 0.0, 1.0)
    else:
        u0_sym = None

    # ----- optimisation helpers -----
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0]

    def local_optimize(u_start):
        """Run the β‑ladder of L‑BFGS‑B starting from u_start."""
        u = u_start.copy()
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.94:
                break
            res = opt.minimize(
                fun=surrogate_val,
                x0=u,
                args=(beta,),
                jac=surrogate_grad,
                bounds=bounds,
                method="L-BFGS-B",
                options={"maxiter": 120, "ftol": 1e-12, "disp": False},
            )
            u = np.clip(res.x, 0.0, 1.0)
        return u

    best_u = None
    best_c = float("inf")

    # ----- main stochastic search loop -----
    while time.time() - start_time < budget_s * 0.90:
        # choose a start point
        if u0_sym is not None and rng.random() < 0.25:
            u_start = u0_sym.copy()
            # add a small random perturbation to diversify
            u_start = perturb_u(u_start, rng, scale=0.1)
        else:
            u_start = rng.random(n2)            # uniform in [0,1]

        # deterministic hill‑climb to get a reasonable seed
        u_hc, c_hc = hill_climb_u(u_start, rng, max_iters=1200, step_scale=0.4)

        # L‑BFGS‑B refinement with smoothing
        u_opt = local_optimize(u_hc)

        # optional short hill‑climb from the refined point
        if time.time() - start_time < budget_s * 0.92:
            u_opt, c_opt = hill_climb_u(u_opt, rng, max_iters=800, step_scale=0.25)
        else:
            c_opt = compute_c5(build_h_from_u(u_opt))

        if c_opt < best_c:
            best_c = c_opt
            best_u = u_opt.copy()

    # ----- final polishing if time permits -----
    if best_u is not None and time.time() - start_time < budget_s * 0.97:
        best_u, best_c = hill_climb_u(best_u, rng, max_iters=3000, step_scale=0.2)

    # fallback in the extremely unlikely case we have no candidate
    if best_u is None:
        best_u = u0_sym if u0_sym is not None else rng.random(n2)

    final_h = build_h_from_u(best_u)
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(n)
