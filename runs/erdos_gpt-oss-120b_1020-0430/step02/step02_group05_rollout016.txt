def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the intersection of the box [0,1]^n and the hyperplane sum=v=target."""
    lo = -1.0  # low bound for Lagrange multiplier λ
    hi = 1.0   # high bound (sufficient for 0‑1 box)
    # Expand interval until it brackets the solution
    for _ in range(20):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny correction for accumulated floating error
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C₅ = maxₖ ∫ h(x)(1‑h(x+k))dx for the discrete step function."""
    n = h.size
    dx = 2.0 / n
    # autocorrelation Sₖ = Σ h_i h_{i+k}
    corr = np.correlate(h, h, mode="full")
    S = corr[n - 1 : n - 1 + n]       # k = 0,…,n‑1
    minS = np.min(S)
    return 1.0 - dx * minS


def _grad_smooth_min_autocorr(h: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of a smooth approximation of minₖ Sₖ where Sₖ = Σ h_i h_{i+k}.
    Uses soft‑min wₖ ∝ exp(‑Sₖ/τ).
    """
    n = h.size
    full = np.correlate(h, h, mode="full")
    S = full[n - 1 : n - 1 + n]      # length n
    # soft‑min weights
    exps = -S / max(tau, 1e-12)
    mx = np.max(exps)
    w = np.exp(exps - mx)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    # backward part: Σₖ wₖ h_{j‑k}
    conv_b = np.convolve(w, h, mode="full")[:n]
    # forward part: Σₖ wₖ h_{j+k}
    conv_f_full = np.convolve(w, h[::-1], mode="full")
    conv_f = conv_f_full[n - 1 :: -1]   # indices n‑1‑j, j=0…n‑1
    return conv_b + conv_f


def _gradient_ascent(
    h0: np.ndarray,
    max_iter: int = 2000,
    step0: float = 0.5,
    tau_decay: float = 0.99,
    tau_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Run a projected gradient ascent on the smooth‑min objective."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    # initialise temperature from the initial spread of Sₖ
    init_corr = np.correlate(h, h, mode="full")
    S0 = init_corr[n - 1 : n - 1 + n]
    tau = max(S0.max() - S0.min(), 1e-6) * tau_factor
    τ_min = 1e-6

    for it in range(max_iter):
        # gradient of smooth‑min
        g = _grad_smooth_min_autocorr(h, tau)
        norm = np.linalg.norm(g)
        if norm > 0:
            step = step0 * (1.0 - it / max_iter)  # linear decay
            h = _project_to_simplex_box(h + step * (g / norm), target)

        tau = max(tau * tau_decay, τ_min)

        if (it & 31) == 0:      # evaluate every 32 steps
            c5 = _c5_from_h(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def _solve_h(start_h: np.ndarray, max_iter: int = 400) -> tuple[np.ndarray, float]:
    """Refine a feasible h via SLSQP on the original max‑correlation formulation."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    # initialise auxiliary variable t = max correlation with 1‑h
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse h onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape projected onto the feasible set."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] achieving a low C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    left = lambda: budget_s - (time.time() - start_time)

    # obtain parent construction if present
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # candidate resolutions (prefer the parent size)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (96, 128, 160, 192, 224, 256, 320, 384):
        n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = np.inf
    best_n = None

    # main search over resolutions
    for n in n_candidates:
        if left() < 5.0:
            break

        # base shape for this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        # --- deterministic gradient ascent from the base shape ---
        h_opt, c5_val = _gradient_ascent(
            base_h,
            max_iter=1800,
            step0=0.6,
            tau_decay=0.985,
            tau_factor=1.0,
        )
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < 0.3808:
                break

        # fine‑tune with SLSQP
        h_ref, c5_ref = _solve_h(h_opt, max_iter=300)
        if c5_ref < best_c5:
            best_h, best_c5, best_n = h_ref, c5_ref, n
            if best_c5 < 0.3808:
                break

        # --- stochastic restarts ---
        attempts = 8
        sigma = 0.12
        for _ in range(attempts):
            if left() < 5.0:
                break
            # random perturbation of the base shape
            if np.random.rand() < 0.7:
                start = base_h + sigma * np.random.randn(n)
            else:
                start = np.random.rand(n)
            start = _project_to_simplex_box(start, n / 2.0)

            h_tmp, c5_tmp = _gradient_ascent(
                start,
                max_iter=1200,
                step0=0.5,
                tau_decay=0.99,
                tau_factor=0.8,
            )
            # optional SLSQP polish
            h_tmp2, c5_tmp2 = _solve_h(h_tmp, max_iter=250)
            if c5_tmp2 < best_c5:
                best_h, best_c5, best_n = h_tmp2, c5_tmp2, n
                if best_c5 < 0.3808:
                    break
            sigma *= 0.88

        if best_c5 < 0.3808:
            break

    # fallback if nothing succeeded
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
