def enforce_sum(h, target, rng):
    diff = target - h.sum()
    # while diff > tolerance, adjust random entries
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

import time
import numpy as np

# ----------------------------------------------------------------------
# Core evaluation
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Utility to keep the linear constraint Σh = n/2
# ----------------------------------------------------------------------
def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator):
    """Adjust h (0≤h≤1) so that Σh = target, keeping bounds."""
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ----------------------------------------------------------------------
# Binary utilities
# ----------------------------------------------------------------------
def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones (n assumed even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def binarize_and_fix(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Convert h to binary (threshold 0.5) and fix the sum to n/2."""
    n = h.size
    target = n // 2
    b = (h > 0.5).astype(float)
    cur = int(b.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(b == 0)[0]
        idx = rng.choice(zeros, size=diff, replace=False)
        b[idx] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        idx = rng.choice(ones, size=-diff, replace=False)
        b[idx] = 0.0
    return b


# ----------------------------------------------------------------------
# Simulated annealing on binary vectors
# ----------------------------------------------------------------------
def binary_anneal(
    n: int,
    rng: np.random.Generator,
    end_time: float,
    start_h: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """Anneal a binary vector of length n until end_time (seconds)."""
    # initialise
    if start_h is None:
        h = random_binary_vector(n, rng)
    else:
        # ensure binary and correct sum
        h = binarize_and_fix(start_h, rng)
    c = compute_c5(h)
    best_h, best_c = h.copy(), c

    T = 0.02               # initial temperature
    decay = 0.99996        # exponential decay factor

    while time.time() < end_time:
        # pick two different indices with opposite values
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while i == j or h[i] == h[j]:
            i = rng.integers(0, n)
            j = rng.integers(0, n)

        # swap
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h)
        delta = c_new - c

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

        T = max(T * decay, 1e-8)

    return best_h, best_c


# ----------------------------------------------------------------------
# Small continuous hill‑climb (transfer ε between two coordinates)
# ----------------------------------------------------------------------
def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 3000,
    start_step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)

    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)

    return best_h, best_c

    # ------------------------------------------------------------------

    if "initial_h_values" in globals():

    best_h = enforce_sum(best_h, target, rng)
