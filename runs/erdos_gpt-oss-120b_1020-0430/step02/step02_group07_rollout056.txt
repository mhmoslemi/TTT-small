def compute_c5_fast(h):
    n = h.size
    # next power of two
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    # zero pad
    a = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    b = np.fft.rfft(np.concatenate([1 - h, np.zeros(N - n)]))
    # correlation via irfft of a*conj(b)
    corr = np.fft.irfft(a * np.conj(b), n=N)
    # get first L entries
    corr = corr[:L]
    # max
    return np.max(corr) * (2.0 / n)

import time, numpy as np

Define functions:

def compute_c5_fast(h):
    # compute max overlap integral (C5) for binary h
    n = h.shape[0]
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    a = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    b = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(N - n)]))
    corr = np.fft.irfft(a * np.conj(b), n=N)
    # real part due to numerical errors
    max_corr = np.max(corr[:L])
    return max_corr * (2.0 / n)

def neighbor_swap(h, rng):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h
    # swap values
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def random_binary(n, rng):
    k = n // 2
    idx = rng.permutation(n)[:k]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def upsample_float(arr, factor):
    n = arr.size
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, n * factor, endpoint=False)
    return np.interp(x_new, x_old, arr)

def float_to_binary(arr):
    n = arr.size
    k = n // 2
    # Get indices of largest k entries
    idx = np.argpartition(-arr, k)[:k]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def hill_climb_binary(h, rng, max_iter=2000):
    cur_h = h.copy()
    cur_c = compute_c5_fast(cur_h)
    best_h = cur_h
    best_c = cur_c
    n = h.size
    for _ in range(max_iter):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5_fast(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

import time
import numpy as np

# ---------- Utility functions -------------------------------------------------
def compute_c5_and_shift(h: np.ndarray):
    """Return C5 value and the shift (lag) where the maximum occurs."""
    n = h.shape[0]
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()          # smallest power of two ≥ L
    a = np.fft.rfft(h, n=N)
    b = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(a * np.conj(b), n=N)[:L]
    idx = int(np.argmax(corr))
    max_corr = corr[idx]
    shift = idx - (n - 1)                  # lag = index - (n-1)
    return max_corr * (2.0 / n), shift


def compute_c5(h: np.ndarray):
    """C5 value only (fast wrapper)."""
    return compute_c5_and_shift(h)[0]


def neighbor_swap(h: np.ndarray, rng: np.random.Generator):
    """Swap two random positions (preserves number of ones)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h


def targeted_swap(h: np.ndarray, shift: int, rng: np.random.Generator):
    """
    Perform a swap that directly reduces the contribution for the given shift.
    If no suitable pair exists, fall back to a random swap.
    """
    n = h.size
    if shift >= 0:
        lim = n - shift
        i_cand = np.where((h[:lim] == 1) & (h[shift:] == 0))[0]
        j_cand = np.where((h[:lim] == 0) & (h[shift:] == 1))[0]
        if i_cand.size == 0 or j_cand.size == 0:
            return neighbor_swap(h, rng)
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)
    else:
        s = -shift
        lim = n - s
        i_cand = np.where((h[s:] == 1) & (h[:lim] == 0))[0] + s
        j_cand = np.where((h[s:] == 0) & (h[:lim] == 1))[0] + s
        if i_cand.size == 0 or j_cand.size == 0:
            return neighbor_swap(h, rng)
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)

    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h


def random_binary(n: int, rng: np.random.Generator):
    """Uniform random binary vector with exactly n/2 ones (n must be even)."""
    k = n // 2
    idx = rng.permutation(n)[:k]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def float_to_binary(a: np.ndarray):
    """Convert a real vector to a binary one by taking the largest half entries."""
    n = a.size
    k = n // 2
    idx = np.argpartition(-a, k)[:k]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def upsample_float(arr: np.ndarray, target_n: int):
    """Linear up‑sampling of a float vector to length target_n."""
    n0 = arr.size
    x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


# ---------- Local deterministic refinement ------------------------------------
def deterministic_climb(h: np.ndarray, rng: np.random.Generator, max_no_improve=5000):
    """Hill‑climb using targeted swaps; stops after max_no_improve failed attempts."""
    cur_h = h.copy()
    cur_c, cur_shift = compute_c5_and_shift(cur_h)
    stagn = 0
    while stagn < max_no_improve:
        cand_h = targeted_swap(cur_h, cur_shift, rng)
        cand_c, cand_shift = compute_c5_and_shift(cand_h)
        if cand_c < cur_c - 1e-12:
            cur_h, cur_c, cur_shift = cand_h, cand_c, cand_shift
            stagn = 0
            continue
        # fallback random swap
        cand_h = neighbor_swap(cur_h, rng)
        cand_c, cand_shift = compute_c5_and_shift(cand_h)
        if cand_c < cur_c - 1e-12:
            cur_h, cur_c, cur_shift = cand_h, cand_c, cand_shift
            stagn = 0
        else:
            stagn += 1
    return cur_h, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- problem size -------------------------------------------------------
    n_target = 720                     # even, ≤ 1000 as recommended

    # ---- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        init_f = np.array(initial_h_values, dtype=np.float64)
        init_up = upsample_float(init_f, n_target)
        cur_h = float_to_binary(init_up)
    else:
        cur_h = random_binary(n_target, rng)

    best_h, best_c = cur_h.copy(), compute_c5(cur_h)

    # ---- simulated‑annealing with targeted moves ---------------------------
    target_prob = 0.65                 # probability of using a targeted swap
    while time.time() - start < budget_s * 0.80:
        # start point for this SA run
        if rng.random() < 0.25:
            h_cur = best_h.copy()
        else:
            h_cur = random_binary(n_target, rng)

        cur_c, cur_shift = compute_c5_and_shift(h_cur)

        # iteration budget for this run (≈300 iters / s)
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(200000, remaining * 300))

        # temperature schedule (simple exponential cooling)
        temp = max(best_c * 0.4, 0.001)
        decay = 0.9995

        for _ in range(max_iters):
            temp = max(temp * decay, 1e-7)

            # propose neighbour
            if rng.random() < target_prob:
                h_new = targeted_swap(h_cur, cur_shift, rng)
            else:
                h_new = neighbor_swap(h_cur, rng)

            new_c, new_shift = compute_c5_and_shift(h_new)
            delta = new_c - cur_c
            if delta < 0.0 or rng.random() < np.exp(-delta / temp):
                h_cur, cur_c, cur_shift = h_new, new_c, new_shift
                if new_c < best_c:
                    best_h, best_c = h_new.copy(), new_c

            if time.time() - start > budget_s * 0.80:
                break

        # short deterministic climb to polish the current best
        polished_h, polished_c = deterministic_climb(best_h, rng, max_no_improve=1500)
        if polished_c < best_c:
            best_h, best_c = polished_h, polished_c

    # ---- final deterministic polishing ---------------------------------------
    best_h, best_c = deterministic_climb(best_h, rng, max_no_improve=5000)

    # ---- strict feasibility check (binary, exact sum) ------------------------
    target_ones = n_target // 2
    current_ones = int(best_h.sum())
    if current_ones != target_ones:
        diff = target_ones - current_ones
        if diff > 0:
            zeros = np.where(best_h == 0)[0]
            change = rng.choice(zeros, diff, replace=False)
            best_h[change] = 1.0
        else:
            ones = np.where(best_h == 1)[0]
            change = rng.choice(ones, -diff, replace=False)
            best_h[change] = 0.0

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_target)
