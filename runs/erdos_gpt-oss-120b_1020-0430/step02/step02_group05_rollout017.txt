import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum(x)=target} via binary search on λ."""
    lo = np.min(v) - 1.0  # ensure projection at lo gives all 1's → sum = n
    hi = np.max(v)        # projection at hi gives all 0's → sum = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)

    # tiny fix for numerical drift
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C₅ = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _max_corr_and_shift(h: np.ndarray):
    """Return (max_corr, shift) where shift∈[−(n‑1),n‑1] gives the max."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx], idx - (n - 1)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric triangle on [0,2] sampled with n points, projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse grid to fine grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """Inequality: t - correlation ≥ 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Equality: sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 500):
    """Refine h via SLSQP on the exact constrained formulation."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]

    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _subgrad_optimize(base_h: np.ndarray,
                      max_iter: int,
                      alpha0: float,
                      restarts: int,

    """Projected sub‑gradient descent on max_{k} corr, with occasional noise."""
    n = base_h.size
    target = n / 2.0
    best_h = None

            max_corr, shift = _max_corr_and_shift(h)

            s = shift

    # load parent if present
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (favor parent size)
    cand_n = [64, 96, 128, 192, 256, 384, 448]
    if init_h is not None:
        cand_n.append(init_h.size)
    cand_n = sorted(set(cand_n))

    for n in cand_n:
        if time_left() < 5.0:
            break

        # construct base h at this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        # parameters scaling with n
        max_iter = int(2000 + n * 5)          # ~4000 at n=400
        restarts = 6 if n <= 128 else 4
        alpha0 = 0.7 if n <= 128 else 0.5

        # global sub‑gradient phase
        h_sg, c5_sg = _subgrad_optimize(base_h,
                                        max_iter=max_iter,
                                        alpha0=alpha0,
                                        restarts=restarts,
                                        time_left=time_left)

        # keep best so far
        if h_sg is not None and c5_sg < best_c5:
            best_h, best_c5, best_n = h_sg, c5_sg, n
            if best_c5 < target_bound:
                break

        # local SLSQP refinement (few iterations)
        if h_sg is not None:
            h_ref, c5_ref = _solve_h(h_sg, max_iter=400)
            if c5_ref < best_c5:
                best_h, best_c5, best_n = h_ref, c5_ref, n
                if best_c5 < target_bound:
                    break

        # occasional extra SLSQP from random starts if time permits
        attempts = 4
        for _ in range(attempts):
            if time_left() < 4.0:
                break
            rnd = np.random.rand(n)
            rnd = _project_to_simplex_box(rnd, n / 2.0)
            h_rand, c5_rand = _solve_h(rnd, max_iter=300)
            if c5_rand < best_c5:
                best_h, best_c5, best_n = h_rand, c5_rand, n
                if best_c5 < target_bound:
                    break

        if best_c5 < target_bound:
            break

    # final fallback
    if best_h is None:
        n = 128
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
