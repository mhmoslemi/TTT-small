import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Core evaluation
# ------------------------------------------------------------
def _c5_of_h(h: np.ndarray) -> float:
    """Return the C5 value for a (feasible) step function h."""
    n = h.size
    # correlation of h with (1-h) using zero padding (full mode)
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ------------------------------------------------------------
# Feasibility helpers
# ------------------------------------------------------------
def _binary_feasible(h: np.ndarray) -> np.ndarray:
    """Turn a real vector into a binary one with exactly n/2 ones."""
    n = h.size
    # simple rounding then correction
    h_bin = np.rint(h).astype(int)
    h_bin = np.clip(h_bin, 0, 1)
    cur = h_bin.sum()
    target = n // 2
    if cur > target:
        # turn some ones to zero
        ones = np.where(h_bin == 1)[0]
        np.random.shuffle(ones)
        h_bin[ones[: cur - target]] = 0
    elif cur < target:
        zeros = np.where(h_bin == 0)[0]
        np.random.shuffle(zeros)
        h_bin[zeros[: target - cur]] = 1
    return h_bin.astype(float)


def _initial_binary(n: int, init_vals=None) -> np.ndarray:
    """Create an initial binary vector of length n."""
    if init_vals is not None:
        # use the given values as a seed, round to binary
        h = np.array(init_vals, dtype=float)
        if h.size != n:
            # resize via interpolation
            x_src = (np.arange(h.size) + 0.5) * (2.0 / h.size)
            x_tgt = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(x_tgt, x_src, h)
        h_bin = _binary_feasible(h)
        return h_bin
    # fresh random binary with exact half ones
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h

# Simple hill‑climbing / simulated‑annealing on binary vectors
# ------------------------------------------------------------
def _hill_climb_binary(start_h: np.ndarray, budget_s: float) -> tuple:
    """Improves a binary vector by random swaps while time permits."""
    n = start_h.size
    best_h = start_h.copy()
    best_c5 = _c5_of_h(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    # temperature schedule (linear decay)
    temp_start = 0.02 * best_c5  # scale with initial value
    no_improve = 0

    # Pre‑compute index lists
    def _indices(arr):
        return np.where(arr == 1)[0], np.where(arr == 0)[0]

    ones, zeros = _indices(cur_h)

    while time.time() - start_time < budget_s:
        # pick a random 1 and a random 0 and swap them
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # swap

        new_c5 = _c5_of_h(cur_h)

        # acceptance rule
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        temp = temp_start * (1.0 - frac)  # cooling

        accept = False
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            # probabilistic uphill move
            if temp > 0:
                prob = np.exp(-(new_c5 - cur_c5) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_c5 = new_c5
            # update index lists after a successful swap
            # i became 0, j became 1
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            no_improve = 0
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                # early exit if we beat the target bound
                if best_c5 < 0.3808:
                    break
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

            no_improve += 1
            if no_improve >= 4000:
                # restart from best so far to escape local plateau
                cur_h = best_h.copy()
                cur_c5 = best_c5
                ones, zeros = _indices(cur_h)
                no_improve = 0

    return best_h, best_c5

# Optional final polishing with SLSQP (continuous refinement)
# ------------------------------------------------------------
def _project_feasible_continuous(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 via water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny clean‑up
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _solve_continuous(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """Run a single SLSQP optimisation starting from a feasible point."""
    n = initial_h.size
    h0 = _project_feasible_continuous(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def _obj(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    res = optimize.minimize(
        _obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_feasible_continuous(h_opt, n)
    c5_val = _c5_of_h(h_opt)
    return h_opt, c5_val

# Main driver
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init_vals = np.array(initial_h_values, dtype=float)
        n0 = init_vals.size
    except NameError:
        init_vals = None
        n0 = 256  # default size

    # Ensure an even size (required for sum = n/2 integer)
    if n0 % 2 == 1:
        n0 += 1

    # Build initial binary vector
    h_start = _initial_binary(n0, init_vals)

    # ------------------------------------------------------------------
    # Phase 1: binary hill‑climbing / simulated annealing
    # ------------------------------------------------------------------
    # Allocate most of the budget to this cheap phase
    hc_budget = max(0.85 * budget_s, 5.0)
    best_h, best_c5 = _hill_climb_binary(h_start, hc_budget)

    # ------------------------------------------------------------------
    # Phase 2: optional continuous polishing (if time permits)
    # ------------------------------------------------------------------
    if time_left() > 15.0:
        # Use the best binary as a starting point for SLSQP
        h_cont, c5_cont = _solve_continuous(best_h, max_iter=800)
        if c5_cont < best_c5:
            best_h, best_c5 = h_cont, c5_cont

    # ------------------------------------------------------------------
    # Final sanity check (project again) and return
    # ------------------------------------------------------------------
    best_h = _project_feasible_continuous(best_h, best_h.size)
    best_c5 = _c5_of_h(best_h)
    n_points = int(best_h.size)

    return best_h.tolist(), float(best_c5), n_points
