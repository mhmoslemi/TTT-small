def gradient_of_f(h, shift):
    n = len(h)
    g = np.ones(n)  # from Σ_i h_i term derivative gives 1 each.
    if shift >= 0:
        # h_i * h_{i+shift}: indices i from 0 to n-1-shift
        # For each i, gradient w.r.t. h_i includes -h_{i+shift}
        # For each j = i+shift, gradient w.r.t. h_j includes -h_{i}
        # So we subtract h_{i+shift} from g[i], and also subtract h_i from g[i+shift].
        g[:n-shift] -= h[shift:]  # g[i] - = h[i+shift]
        g[shift:] -= h[:n-shift]  # g[i+shift] - = h[i]
    else:
        # shift negative: treat similarly with shift_abs = -shift.
        s = -shift
        # term: h_i * h_{i - s}
        # indices i from s to n-1
        # For i, gradient w.r.t. h_i includes -h_{i - s}
        # For j = i - s, gradient w.r.t. h_j includes -h_i
        g[s:] -= h[:n-s]  # g[i] - = h[i-s]
        g[:n-s] -= h[s:]  # g[i-s] - = h[i]
    return g

def _corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    # Using rfft for real inputs
    h_fft = np.fft.rfft(h, n=L)
    rev = np.flip(1.0 - h)
    rev_fft = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(h_fft * rev_fft, n=L)[:2*n-1]
    return conv

def _c5_from_corr(corr, n):
    return np.max(corr) * (2.0/n)

def _gradient_f(h, shift):
    n = h.shape[0]
    g = np.ones(n)
    if shift >= 0:
        s = shift
        g[:n-s] -= h[s:]
        g[s:] -= h[:n-s]
    else:
        s = -shift
        g[s:] -= h[:n-s]
        g[:n-s] -= h[s:]
    return g

def _subgradient_descent(h0, time_left_func, max_iter=5000):
    h = h0.copy()
    n = h.shape[0]
    target = n/2.0
    L = 1 << ((2*n-1).bit_length())
    best_h = h.copy()
    best_c5 = np.inf
    α0 = 2.0/n  # initial step size scaling

    for it in range(max_iter):
        if not time_left_func():
            break
        # compute correlation via FFT
        # we could compute FFT each iteration, but we also need rev
        h_fft = np.fft.rfft(h, n=L)
        rev = np.flip(1.0 - h)
        rev_fft = np.fft.rfft(rev, n=L)
        corr = np.fft.irfft(h_fft * rev_fft, n=L)[:2*n-1]
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        c5 = max_val * (2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # if c5 below target bound maybe break, but not mandatory
        shift = max_idx - (n-1)
        grad = _gradient_f(h, shift)
        step = α0 / np.sqrt(it+1)
        h = h - step * grad
        # project onto constraints
        h = _project_to_simplex_box(h, target)

    return best_h, best_c5

def _project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import time
import numpy as np


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x | 0 ≤ x ≤ 1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape h(x)=min(x,2-x) sampled on n points and projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (returns same ordering as np.correlate)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    h_fft = np.fft.rfft(h, n=L)
    rev = np.flip(1.0 - h)
    rev_fft = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(h_fft * rev_fft, n=L)[: 2 * n - 1]
    return conv  # length 2n-1, index 0 ↔ shift -(n-1)


def _c5_from_h_fft(h: np.ndarray) -> float:
    """C5 value from FFT correlation."""
    n = h.shape[0]
    corr = _corr_fft(h)
    return np.max(corr) * (2.0 / n)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value using numpy.correlate (fallback)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _gradient_f(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of f_shift(h)= Σ h_i - Σ h_i h_{i+shift} w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n)
    if shift >= 0:
        s = shift
        g[: n - s] -= h[s:]        # -h_{i+shift}
        g[s:] -= h[: n - s]        # -h_{i-shift}
    else:
        s = -shift
        g[s:] -= h[: n - s]        # -h_{i-shift}
        g[: n - s] -= h[s:]        # -h_{i+shift}
    return g


def _subgradient_descent(
    h0: np.ndarray, time_left_func, max_iter: int = 3000, step0: float = 0.1
):
    """Subgradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _c5_from_h_fft(h)
    for it in range(max_iter):
        if not time_left_func():
            break
        corr = _corr_fft(h)
        k = int(np.argmax(corr))
        c5 = corr[k] * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        shift = (n - 1) - k  # convert convolution index to correlation shift
        grad = _gradient_f(h, shift)
        step = step0 / np.sqrt(it + 1.0)
        h = h - step * grad
        h = _project_to_simplex_box(h, target)
    return best_h, best_c5


def _local_search(
    h0: np.ndarray, time_left_func, max_moves: int = 20000, delta0: float = 0.1
):
    """Pairwise random swaps (mass exchange) to fine‑tune C5."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h_fft(best_h)
    delta = delta0
    for m in range(max_moves):
        if not time_left_func():
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # increase i, decrease j
        up_max = 1.0 - best_h[i]
        down_max = best_h[j]
        if up_max <= 0 or down_max <= 0:
            continue
        d = delta * np.random.rand()
        d = min(d, up_max, down_max)
        if d <= 0:
            continue
        new_h = best_h.copy()
        new_h[i] += d
        new_h[j] -= d
        c5 = _c5_from_h_fft(new_h)
        if c5 < best_c5:
            best_h = new_h
            best_c5 = c5
    return best_h, best_c5
