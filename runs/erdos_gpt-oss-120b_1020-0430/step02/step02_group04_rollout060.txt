def gradient_max_shift(h):
    n=len(h)
    conv = np.correlate(h, 1-h, mode='full')
    c = conv * (2.0/n)
    maxc = c.max()
    # indices where c close to max
    idxs = np.where(c >= maxc - 1e-12)[0]   # tolerance
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n-1)
        if shift >= 0:
            # term1: (1 - h[i+shift]) for i where i+shift < n
            term1 = np.empty(n)
            term1[:shift] = 0.0
            term1[shift:] = 1 - h[: n - shift]
            # term2: h[i-shift] for i where i-shift >=0
            term2 = np.empty(n)
            term2[: n - shift] = h[shift:]   # Wait: term2[j] = h[j-shift]; for j from shift..n-1, j-shift >=0 => term2[j] = h[j-shift]
            # Actually term2[j] = h[j - shift]; So for j>=shift: term2[j] = h[j - shift] = h[0:n-shift]; for j<shift: term2[j]=0
            term2[:shift] = 0.0
            term2[shift:] = h[: n - shift]    # careful: j-shift index from 0 to n-shift-1
        else:
            s = -shift
            term1 = np.empty(n)
            term1[:s] = 0.0
            term1[s:] = 1 - h[: n - s]
            term2 = np.empty(n)
            term2[: n - s] = h[s:]
            term2[n - s:] = 0.0
        g += term1 - term2
    g /= len(idxs)  # average if multiple
    return g

def max_subgradient(h):
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    c = conv * (2.0 / n)
    maxc = c.max()
    # tolerance for floating point
    idxs = np.where(c >= maxc - 1e-12)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            # term1: (1 - h[i+shift]) where i+shift < n
            term1 = np.empty(n)
            term1[:shift] = 0.0
            term1[shift:] = 1 - h[: n - shift]
            # term2: h[i-shift] where i-shift >=0
            term2 = np.empty(n)
            term2[:shift] = 0.0
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.empty(n)
            term1[:s] = 0.0
            term1[s:] = 1 - h[: n - s]
            term2 = np.empty(n)
            term2[: n - s] = h[s:]
            term2[n - s:] = 0.0
        g += (2.0 / n) * (term1 - term2)
    if len(idxs) > 1:
        g /= len(idxs)
    return g

def perturb_h_pairwise(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    # max possible transfer from i to j while staying in [0,1]
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_transfer * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h_pairwise(h, rng, scale=step)
        c_new = compute_c5(new_h)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 5) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Projection onto the box [lower,upper] with exact sum = target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric vector in [0,1] with sum = n/2."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def triangle_initial(n):
    """Triangular shape h(x)=max(0,1-|x-1|) sampled on n points and projected."""
    dx = 2.0 / n
    x = np.arange(n) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    h = project_onto_box_sum(h, n / 2.0)
    return h


def enforce_symmetry(v):
    v = (v + v[::-1]) * 0.5
    v = project_onto_box_sum(v, v.size / 2.0)
    return v

# subgradient of the max‑overlap functional
# ------------------------------------------------------------
def max_subgradient(h):
    """Sub‑gradient of max_k C(k) w.r.t. h (projected onto the box‑sum later)."""
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode="full")
    c = conv * (2.0 / n)
    maxc = c.max()
    idxs = np.where(c >= maxc - 1e-12)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            term1 = np.empty(n)
            term1[:shift] = 0.0
            term1[shift:] = 1 - h[: n - shift]
            term2 = np.empty(n)
            term2[:shift] = 0.0
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.empty(n)
            term1[:s] = 0.0
            term1[s:] = 1 - h[: n - s]
            term2 = np.empty(n)
            term2[: n - s] = h[s:]
            term2[n - s:] = 0.0
        g += (2.0 / n) * (term1 - term2)
    if len(idxs) > 1:
        g /= len(idxs)
    return g


def subgradient_descent(
    h0,
    rng,
    max_iters=15000,
    init_lr=0.6,
    decay=5e-4,
    enforce_sym=True,
    time_limit=None,
):
    """Projected sub‑gradient descent on max_k C(k)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        g = max_subgradient(h)
        step = init_lr / (1.0 + decay * it)
        new_h = h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, target)
        if enforce_sym:
            new_h = enforce_symmetry(new_h)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
        else:
            # keep current h (monotone descent)
            pass
    return best_h, best_c

# simple hill‑climb (two‑index exchange)
# ------------------------------------------------------------
def perturb_h_pairwise(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h_pairwise(h, rng, scale=step)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 5) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# Simulated annealing refinement

def simulated_annealing(
    h_start, rng, max_iters=20000, init_temp=0.02, final_temp=1e-6, enforce_sym=True
):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    target = n / 2.0
    temp = init_temp
    decay = (final_temp / init_temp) ** (1.0 / max_iters)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        if enforce_sym:
            new_h = enforce_symmetry(new_h)
        else:
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_onto_box_sum(new_h, target)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
        # Metropolis acceptance
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(temp, 1e-12)):
            h = new_h
            cur_c = c_new
        temp *= decay
        if temp < 1e-12:
            break
    return best_h, best_c

    # candidate discretisations (all even)
