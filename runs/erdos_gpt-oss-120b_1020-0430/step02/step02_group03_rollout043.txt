def perturb_a(a, rng, scale):
    a_new = a.copy()
    # randomly choose number of positions to modify? Use one pair for simplicity
    i = rng.integers(0, len(a))
    j = rng.integers(0, len(a))
    if i == j:
        # small perturb to a[i] only
        eps = (rng.random() - 0.5) * 2 * scale
        a_new[i] = np.clip(a[i] + eps, 0.0, 1.0)
        return a_new
    # Transfer epsilon from i to j while staying within bounds
    max_transfer = min(a[i], 1 - a[j])
    max_transfer_neg = min(1 - a[i], a[j]) # if we move from j to i
    if max_transfer + max_transfer_neg == 0:
        return a_new
    # Choose direction randomly
    if rng.random() < 0.5 and max_transfer > 0:
        # move from i to j
        eps = (rng.random() - 0.5) * 2 * max_transfer * scale
        # ensure we don't exceed max_transfer
        eps = np.clip(eps, -max_transfer, max_transfer)
        a_new[i] -= eps
        a_new[j] += eps
    else:
        # move from j to i
        eps = (rng.random() - 0.5) * 2 * max_transfer_neg * scale
        eps = np.clip(eps, -max_transfer_neg, max_transfer_neg)
        a_new[j] -= eps
        a_new[i] += eps
    return a_new
```

def smooth_obj_from_a(a, beta):
    h = construct_h_from_half(a)
    corr = np.correlate(h, 1-h, mode='full')
    # compute log-sum-exp scaled by beta
    # using max-c trick
    max_corr = np.max(corr)
    # Use exp(beta * (corr - max_corr)) maybe
    # Actually we want approx of max, can use logsumexp:
    # smooth = (max_corr + np.log(np.sum(np.exp(beta * (corr - max_corr)))))/beta
    # Equivalent.
    exps = np.exp(beta * (corr - max_corr))
    return (max_corr + np.log(np.sum(exps))) / beta
```

Note: This returns smooth surrogate of corr (no dx scaling). We'll scale by dx later for final C5.

This function is smooth; we can compute gradient approximations; SLSQP approximates gradients automatically via finite diff.

def smooth_obj_from_a(a, beta):
    h = construct_h_from_a(a)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # compute log-sum-exp
    diff = corr - max_corr
    # clip for stability
    tmp = np.clip(beta * diff, -700, 700)
    exp_sum = np.exp(tmp).sum()
    return (max_corr + np.log(exp_sum)) / beta
```

import time
import numpy as np
from scipy.optimize import minimize

# Global variable for the current grid size; will be set inside run()
n = 0


def construct_h_from_a(a):
    """Build the full symmetric step function h from half‑vector a."""
    n_local = n
    half = n_local // 2
    h = np.empty(n_local, dtype=float)
    h[:half] = a
    h[half:] = 1.0 - a[::-1]
    return h


def compute_c5_full(h):
    """Exact discrete C₅ for a full h (dx = 2 / n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_from_a(a):
    """C₅ for a half‑vector a (uses symmetry)."""
    return compute_c5_full(construct_h_from_a(a))


def smooth_obj_from_a(a, beta):
    """Log‑sum‑exp surrogate of the max correlation."""
    h = construct_h_from_a(a)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    diff = corr - max_corr
    tmp = np.clip(beta * diff, -700.0, 700.0)  # avoid overflow
    exp_sum = np.exp(tmp).sum()
    return (max_corr + np.log(exp_sum)) / beta


class SmoothObjective:
    """Callable wrapper for smooth_obj_from_a to avoid lambdas."""

    def __init__(self, beta):
        self.beta = beta

    def __call__(self, x):
        return smooth_obj_from_a(x, self.beta)


def l_bfgs_refine(a, beta, max_iter=200):
    """Refine a using L‑BFGS‑B on the smooth surrogate."""
    bounds = [(0.0, 1.0)] * len(a)
    obj = SmoothObjective(beta)
    res = minimize(
        obj,
        a,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12},
    )
    return np.clip(res.x, 0.0, 1.0)


def perturb_a(a, rng, scale):
    """Small random perturbation of a single coordinate (keeps 0≤a≤1)."""
    a_new = a.copy()
    i = rng.integers(0, a.shape[0])
    delta = (rng.random() * 2.0 - 1.0) * scale
    a_new[i] = np.clip(a[i] + delta, 0.0, 1.0)
    return a_new


def hill_climb_a(a_start, rng, end_time, max_iters=50000):
    """Greedy hill‑climb on a (accept only improvements)."""
    a = a_start.copy()
    best_a = a.copy()
    best_c = compute_c5_from_a(a)
    current_c = best_c
    scale = 0.5
    it = 0
    while time.time() < end_time and it < max_iters:
        if it % 5000 == 0 and it > 0:
            scale = max(scale * 0.7, 1e-4)
        a_new = perturb_a(a, rng, scale)
        c_new = compute_c5_from_a(a_new)
        if c_new < current_c:
            a = a_new
            current_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_a = a_new.copy()
        it += 1
    return best_a, best_c


def linear_piecewise_candidate(half, rng):
    """Search simple 0‑ramp‑1 patterns for a good initial a."""
    N = half
    best_a = None
    best_c = float("inf")
    # limit the zero block to at most a third of the half‑interval
    max_zero = N // 3
    for L0 in range(0, max_zero + 1):
        max_ramp = N - L0 - 1
        for Lr in range(1, max_ramp + 1):
            Lc = N - L0 - Lr
            ramp = np.arange(1, Lr + 1) / (Lr + 1)  # values in (0,1)
            a = np.concatenate([np.zeros(L0), ramp, np.ones(Lc)])
            c = compute_c5_from_a(a)
            if c < best_c:
                best_c = c
                best_a = a.copy()
    # If nothing found (should not happen), fall back to random
    if best_a is None:
        best_a = rng.random(half)
        best_c = compute_c5_from_a(best_a)
    return best_a, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.98  # keep a small safety margin

    # -------------------------------------------------------------
