    n = b.size
    best = b.copy()
    best_c = _c5_from_h(best.astype(float))
    ones = np.flatnonzero(best)

def local_continuous(h, max_steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time()>time_limit: break
        i,j = np.random.randint(n, size=2)
        if i==j: continue
        # Compute max epsilon that can be moved from i to j while staying in [0,1]
        max_down = best_h[i]
        max_up = 1 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0: continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        # ensure sum unchanged (by construction)
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

def _project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    # adjust linearly: find indices where we can adjust
    if diff > 0:
        # need to reduce sum
        # compute possible reduction per element: h_i
        reducible = h.copy()
        # reduce proportionally maybe?
        # simpler: iterate decreasing from largest.
        idx = np.argsort(-reducible)
        for i in idx:
            dec = min(reducible[i], diff)
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    else:
        # need to increase sum
        inc_needed = -diff
        incable = 1.0 - h
        idx = np.argsort(-incable)
        for i in idx:
            inc = min(incable[i], inc_needed)
            h[i] += inc
            inc_needed -= inc
            if inc_needed <= 1e-12:
                break
    # final clip
    return np.clip(h, 0.0, 1.0)
```

    n = h.size
    target_sum = n/2
    step = 1.0
    best_h = h.copy()
    best_c = _c5_from_h(h)
    for it in range(max_iters):
        if time.time() > time_limit: break
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        eps = 1e-12
        # find indices with value within tolerance of max
        mask = corr >= max_val - 1e-9
        # Determine shift indices from lag index to shift k = lag - (n-1)
        lags = np.where(mask)[0]
        # Compute gradient as sum over these shifts
        g = np.zeros(n)
        for lag in lags:
            k = lag - (n - 1)
            if k >= 0:
                # shift forward
                # for i 0..n-1
                # h_{i+k} defined for i+k < n else 0
                # h_{i - k} defined for i - k >=0 else 0
                # vector: for i in range(n):
                # h_i+k: shift left by -k? we can compute using slicing
                # Using np.roll with fill? easier: compute shift arrays via zeros.
                shift_pos = np.zeros(n)
                if k < n:
                    shift_pos[:n - k] = h[k:]  # h[i+k] for i=0...n-k-1
                shift_neg = np.zeros(n)
                if k > 0:
                    shift_neg[k:] = h[:n - k]  # h[i - k] for i=k...n-1
                # gradient term
                g += (1 - (shift_pos + shift_neg))
            else:
                # k negative, treat similarly
                kk = -k
                shift_pos = np.zeros(n)
                if kk < n:
                    shift_pos[kk:] = h[:n - kk]
                shift_neg = np.zeros(n)
                if kk > 0:
                    shift_neg[:n - kk] = h[kk:]
                g += (1 - (shift_pos + shift_neg))
        # average gradient if multiple shifts
        if lags.size > 1:
            g = g / lags.size
        # Take step in direction of -g to reduce max f_k?
        # Since objective is max g, subgradient points in direction of gradient of worst.
        # So we move opposite gradient.
        # step size schedule: sqrt decreasing
        step = 0.5 / np.sqrt(it + 1)
        h = h - step * g
        # project to feasible
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

            # adjust uniformly via water-filling or simple scaling
            h -= diff / n  # shift all equally; then clip, not perfect.
            h = np.clip(h, 0.0, 1.0)
            # re-adjust sum
            s = h.sum()
            diff = s - target_sum
            # final adjustment via adding/subtracting minimal amounts
            if abs(diff) > 1e-9:

                for i in range(n):
                    if diff > 0 and h[i] > 0:

def solve_with_slsqp(h0, max_iter=800):
    n = h0.size
    h_initial = _project_to_feasible(h0, n)
    corr0 = np.correlate(h_initial, 1.0 - h_initial, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h_initial, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{"type":"ineq","fun":_ineq_constraint},
                   {"type":"eq","fun":_eq_constraint}]
    opts = {"maxiter": max_iter, "ftol":1e-9, "disp":False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h_initial
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5
```

    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

def _project_to_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = s - target
    # simple correction loop
    if abs(diff) > 1e-12:
        # attempt to fix by adjusting entries towards bounds
        if diff > 0:
            # need to reduce sum
            # prioritize reducing from entries >0
            idx = np.where(h > 0)[0]
            if idx.size > 0:
                dec = diff / idx.size
                h[idx] -= dec
        else:
            # need to increase sum
            idx = np.where(h < 1)[0]
            if idx.size > 0:
                inc = -diff / idx.size
                h[idx] += inc
        h = np.clip(h, 0.0, 1.0)
        # final tweak if still off
        s = h.sum()
        diff = s - target
        if abs(diff) > 1e-12:
            # distribute remaining diff one by one
            for i in range(n):
                if diff > 0 and h[i] > 0:
                    sub = min(h[i], diff)
                    h[i] -= sub
                    diff -= sub
                elif diff < 0 and h[i] < 1:
                    add = min(1 - h[i], -diff)
                    h[i] += add
                    diff += add
                if abs(diff) < 1e-12:
                    break
    return np.clip(h, 0.0, 1.0)

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode='full')

def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective_t(x):
    return x[-1]

def _solve_h(initial_h, max_iter=800):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective_t, x0, method="SLSQP",
                            bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

def hill_climb_continuous(h, max_steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

def hill_climb_binary(b, max_iters, time_limit):
    n = b.size
    best = b.copy()
    best_c = _c5_from_h(best)
    T0 = 0.2
    for step in range(max_iters):
        if time.time() > time_limit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if best[i] == best[j]:
            continue
        cand = best.copy()
        cand[i], cand[j] = best[j], best[i]
        c = _c5_from_h(cand)
        if c < best_c:
            best, best_c = cand, c
        else:
            T = T0 * (1.0 - step / max_iters)
            if T < 1e-12:
                continue
            prob = np.exp(-(c - best_c) / T)
            if np.random.rand() < prob:
                best, best_c = cand, c
    return best, best_c

def find_best_binary(n, time_budget):
    start = time.time()
    best_h = None
    best_c = float('inf')
    while time.time() - start < time_budget:
        # random binary start with exactly n/2 ones
        b = np.zeros(n)
        ones = np.random.choice(n, n // 2, replace=False)
        b[ones] = 1.0
        # local search (simulated annealing)
        remaining = time_budget - (time.time() - start)
        iters = int(5e3 * (remaining / time_budget + 0.1))  # scale with leftover
        iters = max(500, min(iters, 5000))
        h_bin, c_bin = hill_climb_binary(b, max_iters=iters, time_limit=start + time_budget)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin
            if best_c < 0.3808:
                break
    return best_h, best_c

def optimize_n(n, time_budget):
    start = time.time()
    best_h = None
    best_c = float('inf')
    # parameter grid for p
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    for p in np.linspace(0.5, 2.0, 15):
        h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0) ** p)
        h = _project_to_feasible(h, n)
        c = _c5_from_h(h)
        if c < best_c:
            best_h, best_c = h, c
    # random continuous restarts with quick hill climb
    while time.time() - start < time_budget * 0.3:
        h_rand = np.random.rand(n)
        h_rand = _project_to_feasible(h_rand, n)
        limit = start + time_budget
        h_rand, c_rand = hill_climb_continuous(h_rand, max_steps=800, time_limit=limit)
        if c_rand < best_c:
            best_h, best_c = h_rand, c_rand
    # apply SLSQP on the current best
    elapsed = time.time() - start
    remaining = time_budget - elapsed
    if remaining > 3.0 and best_h is not None:
        max_iter = max(200, int(1200 * (remaining / time_budget)))
        h_opt, c_opt = _solve_h(best_h, max_iter=max_iter)

    # final fine‑grained hill climb
    if best_h is not None and time.time() - start < time_budget:
        limit = start + time_budget
        best_h, best_c = hill_climb_continuous(best_h, max_steps=4000, time_limit=limit)
    return best_h, best_c
