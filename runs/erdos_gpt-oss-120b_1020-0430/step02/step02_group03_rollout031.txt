def hill_climb_v(v, rng, iters=5000, step=0.05):
    best_v = v.copy()
    best_c = compute_c5_h(build_h_from_v(v))
    n = len(v)
    for _ in range(iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        max_eps = min(best_v[i], 1 - best_v[j])
        if max_eps <= 0: continue
        eps = (rng.random() - 0.5) * 2 * max_eps * step
        new_v = best_v.copy()

        c = compute_c5_h(build_h_from_v(new_v))

            best_c = c
            best_v = new_v

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# globals used by the smooth surrogate and DE callback
CURRENT_BETA = None
DE_TIME_LIMIT = None


def build_h_from_v(v):
    """Construct full symmetric step function from half vector v."""
    return np.concatenate([v, 1 - v[::-1]])


def compute_c5_h(h):
    """Exact discrete C5 for a full vector h (dx = 2/len)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_v(v):
    """C5 for a half‑vector v via symmetry."""
    return compute_c5_h(build_h_from_v(v))


def c5_obj_v(v):
    """Objective for DE: exact C5 (to be minimised)."""
    h = build_h_from_v(v)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def softmax_obj_v(v):
    """Log‑sum‑exp smooth surrogate of max_k Σ h(x)(1‑h(x+k))."""
    beta = CURRENT_BETA
    h = build_h_from_v(v)
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def de_callback(xk, convergence):
    """Stop DE when the global time budget for DE is exhausted."""
    if time.time() > DE_TIME_LIMIT:
        return True
    return False


def run_de(start_v, rng, start_time, total_budget, seed):
    """Run a single DE segment, starting from start_v."""
    half_n = start_v.size
    popsize = 7
    init_pop = np.empty((popsize, half_n))

    for i in range(1, popsize):
        init_pop[i] = rng.random(half_n)

    global DE_TIME_LIMIT

    DE_TIME_LIMIT = start_time + total_budget * 0.85

        c5_obj_v,

        maxiter=500,
        init=init_pop,
        polish=False,
        seed=seed,
        callback=de_callback,
        disp=False,
    )
    return result.x

    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

            softmax_obj_v,
            v,
            method="SLSQP",
            bounds=bounds,
            options={"maxiter": 400, "ftol": 1e-12, "disp": False},
        )
        v = np.clip(res.x, 0.0, 1.0)
    return v


def hill_climb(v, rng, time_limit):
    """Local search by moving ε between two coordinates."""
    best_v = v.copy()
    best_c = compute_c5_v(best_v)
    step = 0.5
    n = v.size
    while time.time() < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_v[i], 1.0 - best_v[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_v = best_v.copy()
        new_v[i] -= eps
        new_v[j] += eps
        # values stay inside [0,1] by construction
        c = compute_c5_v(new_v)
        if c < best_c:

            best_v = new_v
            step = 0.5
        else:

            if step < 1e-5:

    return best_v, best_c
