def _optimize_h_smooth(h0, n, time_left_func):

    best_c5 = _c5_from_h(h)
    best_h = h.copy()
    # schedule
    alphas = [5, 10, 20, 40, 80] if n <= 128 else [5, 10, 20, 40]
    # maybe also adapt steps
    for alpha in alphas:
        step = 0.1
        for it in range(2000):
            if time_left_func() <= 5:
                break
            # compute gradient
            corr = np.correlate(h, 1.0 - h, mode="full")
            w = np.exp(alpha * corr - np.max(alpha * corr))  # subtract max for stability

            if w_sum == 0:

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Project onto the set {0≤h≤1, sum(h)=n/2} using water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 value from the discrete correlation."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse grid to a finer one (or down‑sample)."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Simple pairwise epsilon move hill‑climb, respecting feasibility."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]          # can take at most this from i
        max_up = 1.0 - best_h[j]      # can add at most this to j
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


def _log_sum_exp(a: np.ndarray) -> float:
    """Stable log‑sum‑exp."""
    m = np.max(a)
    return m + np.log(np.exp(a - m).sum())


def _optimize_h_smooth(h0: np.ndarray, n: int, time_left) -> tuple:
    """
    Minimise a smooth approximation of max‑overlap using projected gradient descent.
    Returns the best (h, C5) found before the timeout.
    """
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    # schedule of smoothing parameters (larger -> tighter max)
    alpha_seq = [5, 10, 20, 40, 80] if n <= 128 else [5, 10, 20, 40]
    for alpha in alpha_seq:
        if time_left() <= 5:
            break
        # initial values for this alpha
        corr = np.correlate(h, 1.0 - h, mode="full")
        f_curr = _log_sum_exp(alpha * corr) / alpha
        # normalized weights
        w = np.exp(alpha * corr - np.max(alpha * corr))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        step = 0.05
        for _ in range(500):
            if time_left() <= 5:
                break
            # gradient of the smoothed objective
            conv = np.convolve(w, h, mode="full")
            conv_rev = np.convolve(w[::-1], h, mode="full")
            start = n - 1
            grad = 1.0 - (conv[start:start + n] + conv_rev[start:start + n])

            # back‑tracking line search
            improved = False
            for _ in range(10):
                h_new = h - step * grad
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = _project_to_feasible(h_new, n)
                corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
                f_new = _log_sum_exp(alpha * corr_new) / alpha
                if f_new <= f_curr + 1e-12:
                    # accept
                    h = h_new
                    corr = corr_new
                    f_curr = f_new
                    w = np.exp(alpha * corr - np.max(alpha * corr))
                    w_sum = w.sum()
                    w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)
                    c5 = _c5_from_h(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    improved = True
                    break
                else:
                    step *= 0.5
            if not improved:
                break
            step = min(step * 1.2, 0.3)   # gently increase step size when successful
    # final discrete refinement
    if time_left() > 5:
        limit = time.time() + time_left() - 0.1
        best_h, best_c5 = _hill_climb(best_h, max_steps=5000, time_limit=limit)
    return best_h, best_c5


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # possibly supplied construction
    try:
        GLOBAL_INITIAL_H = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL_H = None

    # grid sizes to explore (small → medium, all even for convenience)
    n_candidates = [80, 96, 112, 128, 160, 192, 224, 256]
    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in n_candidates:
        if time_left() <= 5:
            break

        # build a feasible starting vector for this n
        if GLOBAL_INITIAL_H is not None:
            base_h = _upscale_h(GLOBAL_INITIAL_H, n)
        else:
            # random half‑ones
            base_h = np.zeros(n)
            base_h[: n // 2] = 1.0
            np.random.shuffle(base_h)

        base_h = _project_to_feasible(base_h, n)

        # number of independent starts per size
        starts = 4 if n <= 128 else 2
        for _ in range(starts):
            if time_left() <= 5:
                break
            # perturb a bit to diversify
            h0 = base_h + 0.07 * np.random.randn(n)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = _project_to_feasible(h0, n)

            h_opt, c5_val = _optimize_h_smooth(h0, n, time_left)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_val, n
                # early exit if we already beat the target
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # fallback – should never happen
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
