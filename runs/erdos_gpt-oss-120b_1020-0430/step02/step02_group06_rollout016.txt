def grad_descent(h, max_iter=2000, step0=0.1, decay=0.99, min_step=1e-4):
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = c5_from_h(best_h)
    step = step0
    for it in range(max_iter):
        c = np.correlate(h, 1-h, mode='full')
        # compute max and shift
        idx = np.argmax(c)
        max_corr = c[idx]
        # optional early exit if max_corr*dx < best bound etc.
        if max_corr* (2.0/n) < best_c5:
            best_h = h.copy()
            best_c5 = max_corr * (2.0/n)
        k = idx - (n-1)  # shift
        # compute gradient
        if k >= 0:
            h_plus = np.concatenate([h[k:], np.zeros(k)])
            h_minus = np.concatenate([np.zeros(k), h[:n-k]])
        else:
            k2 = -k
            h_plus = np.concatenate([np.zeros(k2), h[:n-k2]])
            h_minus = np.concatenate([h[k2:], np.zeros(k2)])
        grad = 1.0 - h_plus - h_minus
        # update
        h = h - step * grad
        # projection
        h = project_to_feasible(h, n)
        # decay step
        step = max(step * decay, min_step)
        # optionally stop if step small
        if step <= min_step and it > 5:
            break
    return best_h
```

import time
import numpy as np

# ---------- feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9) -> np.ndarray:
    """Clip h to [0,1] and adjust sum to n/2 via a simple bisection (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) <= tol:
        return h
    # lambda such that sum(clip(h - lam,0,1)) = target
    lo, hi = -1.0, 1.0  # lo gives sum>=target, hi gives sum<=target
    for _ in range(60):
        lam = (lo + hi) / 2.0
        adj = np.clip(h - lam, 0.0, 1.0)
        s_adj = adj.sum()
        if s_adj > target:
            lo = lam
        else:
            hi = lam
        if abs(s_adj - target) <= tol:
            break
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


# ---------- C5 evaluation ----------
def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a step function h discretised on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- gradient descent on the worst shift ----------
def _grad_descent(
    h0: np.ndarray,
    max_iter: int = 1500,
    step0: float = 0.2,
    decay: float = 0.995,
    min_step: float = 1e-4,
) -> np.ndarray:
    n = h0.size
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    step = step0

    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_c5 = corr[idx] * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = idx - (n - 1)  # shift corresponding to the max correlation

        # compute gradient = 1 - h_{i+k} - h_{i-k}
        if k >= 0:
            # forward shift (i+k)
            h_fwd = np.concatenate((h[k:], np.zeros(k)))
            # backward shift (i-k)
            h_bwd = np.concatenate((np.zeros(k), h[: n - k]))
        else:
            kk = -k
            h_fwd = np.concatenate((np.zeros(kk), h[: n - kk]))
            h_bwd = np.concatenate((h[kk:], np.zeros(kk)))
        grad = 1.0 - h_fwd - h_bwd

        # gradient step and projection
        h = h - step * grad
        h = _project_to_feasible(h, n)

        # step decay
        step = max(step * decay, min_step)

        if step <= min_step and abs(cur_c5 - best_c5) < 1e-10:
            break

    return best_h
