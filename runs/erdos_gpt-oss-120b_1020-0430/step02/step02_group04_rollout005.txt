    n = h.size

        for _ in range(steps_per_beta):

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        # adjust free variables
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        # enforce symmetry and constraints
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    x = (x + x[::-1]) * 0.5
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- gradient descent on smooth max ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:] -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:] -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            new_h = (new_h + new_h[::-1]) * 0.5
            new_h = project_onto_box_sum(new_h, n / 2.0)

            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, n / 2.0)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 180
        h0 = random_feasible_symmetric(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- quick hill‑climb (30% time) ----
    if time.time() - start < budget_s * 0.3:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2500, start_step=0.5)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---- main optimisation loop (≈70% time) ----
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    steps_per_beta = 250
    target_c = 0.3808
    while time.time() - start < budget_s * 0.85:
        # occasional blend with current best
        if rng.random() < 0.15:
            h_start = (random_feasible_symmetric(n, rng) + best_h) * 0.5
        else:
            h_start = random_feasible_symmetric(n, rng)
        h_start = project_onto_box_sum(h_start, n / 2.0)
        h_start = (h_start + h_start[::-1]) * 0.5
        h_start = project_onto_box_sum(h_start, n / 2.0)

        # gradient descent
        h_opt, c_opt = grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng)

        # short hill‑climb refinement
        h_opt, c_opt = hill_climb(h_opt, rng, max_iters=1200, start_step=0.4)

        # SLSQP polish
        for beta in (200.0, 500.0, 1000.0):
            try:
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=250)
            except Exception:
                pass

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            if best_c <= target_c:
                break

    # ---- optional up‑sampling (if time permits) ----
    if time.time() - start < budget_s * 0.97 and best_h.size * 2 <= 500:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up = project_onto_box_sum(h_up, n_up / 2.0)
        h_up = (h_up + h_up[::-1]) * 0.5
        h_up = project_onto_box_sum(h_up, n_up / 2.0)
        h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.4)
        for beta in beta_sched:
            if time.time() - start > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta=beta, max_iter=300)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up, c_up
            n = n_up

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = (best_h + best_h[::-1]) * 0.5
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
