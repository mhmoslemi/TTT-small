def anneal_optimize(h0, rng, time_limit):
    start = time.time()
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    h = h0.copy()
    curr_c = best_c
    # initial temperature based on spread
    T = 0.01  # maybe adjust
    # dec per iteration
    decay = 0.99995
    while time.time() - start < time_limit:
        # choose i,j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            continue
        # epsilon sample
        eps = rng.random() * max_eps
        if eps <= 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        improve = new_c < curr_c
        if improve or rng.random() < np.exp(-(new_c - curr_c) / T):
            h = new_h
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        T = max(T * decay, 1e-6)
    return best_h, best_c

from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

    eps = max(-max_eps, min(eps, max_eps))

    new_h[i] -= eps
    new_h[j] += eps
    return new_h

    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=8):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:

        else:

            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]

        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0

        if abs(diff) > 1e-12:

                    break

                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def anneal_optimize(h0, rng, time_limit):
    start = time.time()

    h = h0.copy()
    curr_c = compute_c5(h)
    best_h = h.copy()

                best_h = new_h.copy()

    return best_h, best_c

                h_start += (rng.random(n) - 0.5) * 0.01
                h_start = np.clip(h_start, 0.0, 1.0)
                diff = n / 2.0 - h_start.sum()
                for _ in range(2000):
                    if abs(diff) < 1e-12:
                        break
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h_start[i]) if diff > 0 else max(diff, -h_start[i])
                    h_start[i] += add
                    diff = n / 2.0 - h_start.sum()
            else:
                h_start = random_feasible_vector(n, rng)

        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0:
            break
        anneal_time = min(remaining * 0.4, 30.0)
        h_ann, c_ann = anneal_optimize(h_start, rng, anneal_time)
        if c_ann < best_c:
            best_h, best_c, best_n = h_ann.copy(), c_ann, n

        for beta in (10.0, 50.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() - start_time > budget_s * 0.96:
                break
            try:
                h_s = slsqp_optimize(h_ann, beta, rng, max_iter=800)
            except Exception:
                continue
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_h, best_c, best_n = h_s.copy(), c_s, n
                h_ann = h_s

        if time.time() - start_time < budget_s * 0.95:
            h_g = gradient_refine(h_ann, rng, step=0.02, max_iter=6)
            c_g = compute_c5(h_g)
            if c_g < best_c:
                best_h, best_c, best_n = h_g.copy(), c_g, n

    # optional up‑sampling refinement
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        if best_n * 2 <= 480:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up, _ = hill_climb(h_up, rng, max_iters=2000, start_step=0.3)
            for beta in (10.0, 50.0, 200.0, 500.0):
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, rng, max_iter=500)
                except Exception:
                    pass
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
