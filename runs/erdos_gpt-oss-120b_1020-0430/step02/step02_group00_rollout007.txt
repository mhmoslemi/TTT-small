import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a feasible step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum(h)=n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # adjust iteratively – usually one or two passes are enough
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Sub‑gradient descent for the minimax problem
# ----------------------------------------------------------------------
def _subgradient_descent(
    h0: np.ndarray,
    n: int,
    time_left,
    max_iter: int = 2000,
    step0: float = 0.5,
    step_decay: float = 0.995,
    noise_every: int = 200,
    noise_scale: float = 0.02,
):
    """Projected sub‑gradient descent on  max_k Σ h_i(1‑h_{i+k})."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    for it in range(max_iter):
        if time_left() < 0.5:
            break

        # correlation and current C5
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        c5 = max_corr * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # pick a shift attaining the maximum (randomly if several)
        max_idxs = np.flatnonzero(corr >= max_corr - 1e-12)
        k_idx = np.random.choice(max_idxs)
        k = k_idx - (n - 1)

        # gradient of the selected shift
        i = np.arange(n)
        idx_plus = i + k
        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        term1 = np.ones(n)
        term1[mask_plus] = 1.0 - h[idx_plus[mask_plus]]

        idx_minus = i - k
        mask_minus = (idx_minus >= 0) & (idx_minus < n)
        term2 = np.zeros(n)
        term2[mask_minus] = h[idx_minus[mask_minus]]

        grad = term1 - term2

        # step and projection
        step = step0 * (step_decay ** it)
        h = h - step * grad
        h = _project_to_feasible(h, n)

        # occasional random jitter to escape flat regions
        if noise_every and (it + 1) % noise_every == 0:
            h = h + noise_scale * np.random.randn(n)
            h = _project_to_feasible(h, n)

    # final check
    final_c5 = _c5_from_h(best_h)
    return best_h, final_c5
