def max_corr_and_gradient(h):
    n = h.size
    # compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    max_idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]  # exact equality may be fine.
    # compute gradient sum
    grad = np.zeros_like(h)
    for idx in max_idxs:
        lag = idx - (n - 1)
        if lag >= 0:
            # g[0:n-lag] += 1 - h[lag:]
            grad[:n-lag] += (1.0 - h[lag:])
            # g[lag:] -= h[:n-lag]
            grad[lag:] -= h[:n-lag]
        else:
            lag = -lag
            # for negative lag: gradient contributions same but with index swapped.
            # correlation for negative lag is sum_{i=0}^{n-lag-1} h[i+lag] * (1 - h[i])
            # So gradient w.r.t. h[i+lag] (i.e., index j=i+lag): (1 - h[i])
            # for j in lag..n-1: grad[j] += (1 - h[j-lag])
            # gradient w.r.t. h[i] (i.e., index i): -h[i+lag] for i in 0..n-lag-1
            # => grad[:n-lag] -= h[lag:]
            grad[lag:] += (1.0 - h[:n-lag])  # since j = i+lag -> j from lag..n-1. Use slice
            grad[:n-lag] -= h[lag:]
    # average if multiple maxima
    grad /= len(max_idxs)
    return max_val, grad

def _subgradient_descent(h0, max_iters, step0, time_check):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    # schedule for step decreasing
    step = step0
    for it in range(1, max_iters+1):
        if time_check() < 0:
            break
        max_val, grad = _max_corr_and_grad(h)
        # step size schedule: step = step0 / sqrt(it)
        step = step0 / np.sqrt(it)
        # update
        h_new = h - step * grad
        # project
        h_new = _project_to_simplex_box(h_new, n/2.0)
        # evaluate
        c5_new = _c5_from_h(h_new)
        # simple acceptance: if c5_new < best_c5, keep; also if c5_new < current c5 maybe accept
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # replace h with new (maybe accept always)
        h = h_new
    # return best
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for the discrete step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _max_corr_and_grad(h: np.ndarray):
    """Compute max correlation value and a subgradient of the max."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    eps = 1e-12
    max_idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in max_idxs:
        lag = idx - (n - 1)
        if lag >= 0:
            L = int(lag)
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] -= h[:n - L]
        else:
            d = -lag
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
    grad /= len(max_idxs)
    return max_val, grad


def _subgradient_descent(h0: np.ndarray, max_iters: int, step0: float, time_check):
    """Projected subgradient descent on max correlation."""
    n = h0.size
    h = _project_to_simplex_box(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    no_improve = 0
    for it in range(1, max_iters + 1):
        if time_check() <= 0:
            break
        _, grad = _max_corr_and_grad(h)
        if np.linalg.norm(grad) < 1e-10:
            break
        step = step0 / np.sqrt(it)
        h_new = h - step * grad
        h_new = _project_to_simplex_box(h_new, n / 2.0)
        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve >= 500:
            break
    return best_h, best_c5


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points, then projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Refine h using SLSQP on the auxiliary variable t."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)
