def hill_climb(h_start, rng, max_iters=5000, step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step_size = step
    for it in range(max_iters):
        # propose
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        # ensure i != j
        if i == j: continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step_size
        # ensure in bound via clipping eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # ensure bounds
        new_h[i] = min(max(new_h[i], 0.0), 1.0)
        new_h[j] = min(max(new_h[j], 0.0), 1.0)
        # adjust sum may drift slightly due to clipping; we will enforce sum afterwards via repair (enforce_sum)

            h = new_h

# global variables for surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # correlation
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):

    step = start_step

    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        # pick two indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        # compute max allowed eps

        max_neg = min(h[j], 1.0 - h[i])
        if max_pos + max_neg == 0:
            continue
        # epsilon uniformly in [-max_neg, max_pos]
        eps = (rng.random() * (max_pos + max_neg)) - max_neg
        # scale by step size
        eps *= step

            h = new_h

    return best_h, best_c

CURRENT_TARGET = None

def _sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    # Optional up‑sampling to a finer grid (if time permits)
    # --------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.94 and n * 2 <= 360:
        h_up = np.repeat(global_best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        # refine on the finer grid
        h_up_opt, c_up_opt = hill_climb(h_up, rng, max_iters=5000, start_step=0.3)
        for beta in beta_list:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up_opt = slsqp_optimize(h_up_opt, beta, max_iter=500)
            except Exception:
                pass
            c_up_opt = compute_c5(h_up_opt)
            if c_up_opt < global_best_c:
                global_best_h, global_best_c, n = h_up_opt.copy(), c_up_opt, n_up

        # final fine‑tune on up‑sampled best
        if time.time() - start_time < budget_s * 0.99:
            h_fine_up, c_fine_up = hill_climb(global_best_h, rng, max_iters=5000, start_step=0.2)
            if c_fine_up < global_best_c:
                global_best_h, global_best_c = h_fine_up.copy(), c_fine_up

    # --------------------------------------------------------------
    # Final clean‑up (clip + exact sum enforcement)
    # --------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
