import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Feasibility projection (box [0,1] and sum = target)
# ----------------------------------------------------------------------
def project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Linear up‑sampling of a step function
# ----------------------------------------------------------------------
def upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Local optimisation using SLSQP (minimise auxiliary t)
# ----------------------------------------------------------------------
def solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    n = initial_h.size
    h0 = project_to_feasible(initial_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr  # must be >= 0

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def obj(x):
        return x[-1]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}],
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Gradient‑descent refinement on the worst shift
# ----------------------------------------------------------------------
def gradient_descent_refine(h: np.ndarray, max_iter: int = 2000) -> tuple:
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_val = c5_from_h(best_h)
    eta = 0.5
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        grad = np.ones_like(h)
        if shift >= 0:
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] -= h[: n - s]
            grad[: n - s] -= h[s:]

        h_new = h - eta * grad
        h_new = project_to_feasible(h_new, target)
        val_new = c5_from_h(h_new)

        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()

        # simple step‑size adaptation
        if val_new > best_val:
            eta *= 0.5
        else:
            eta = min(eta * 1.02, 1.0)

        h = h_new
    return best_h, best_val
