We can write a fast deterministic projection:

```
def _project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    s = h.sum()
    if s < target:
        # need to add (target-s) to some entries < 1
        deficit = target - s
        # distribution: fill in order of increasing h
        idx = np.argsort(h)  # increasing
        for i in idx:
            space = 1.0 - h[i]
            inc = min(space, deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    elif s > target:
        excess = s - target
        idx = np.argsort(-h) # decreasing h
        for i in idx:
            avail = h[i]  # can subtract up to h[i]
            dec = min(avail, excess)
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    return h
```

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1] and adjust to make sum exactly n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) <= 1e-12:
        return h
    if s < target:
        deficit = target - s
        # fill from smallest values upwards
        idx = np.argsort(h)
        for i in idx:
            space = 1.0 - h[i]
            inc = min(space, deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    else:  # s > target
        excess = s - target
        idx = np.argsort(-h)  # largest first
        for i in idx:
            avail = h[i]
            dec = min(avail, excess)
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    # tiny numerical correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny diff to the first non‑boundary element
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a feasible step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _de_obj(x: np.ndarray, n: int) -> float:
    """Objective for differential evolution (minimise C5)."""
    h = _project_to_feasible(x, n)
    return _c5_from_h(h)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _ineq_constraint(x: np.ndarray):
    """t - correlation >= 0 for every shift (inequality constraints)."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray):
    """Enforce sum(h) = n/2 (equality constraint)."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Objective for SLSQP: minimise auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 300):
    """Local refinement of a feasible step function using SLSQP."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val
