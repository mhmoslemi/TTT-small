def solve_slsqp(initial_h, max_iter=800, tol=1e-9):

    n = initial_h.size

    h0 = project_to_feasible(initial_h.copy(), n/2)

    corr0 = np.correlate(h0, 1-h0, mode='full')

    t0 = np.max(corr0) * 1.0

    x0 = np.concatenate([h0, [t0]])

    bounds = [(0,1)]*n + [(0,None)]

    def ineq(x):

        h = x[:n]; t = x[-1]

        return t - np.correlate(h, 1-h, mode='full')

    def eq(x):

        return np.sum(x[:n]) - n/2

    constraints = [{'type':'ineq','fun':ineq}, {'type':'eq','fun':eq}]

    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter':max_iter,'ftol':tol, 'disp':False})

    h_opt = res.x[:n] if res.success else h0

    h_opt = project_to_feasible(h_opt, n/2)

    c5 = _c5_from_h(h_opt)

    return h_opt, c5

def _simulated_annealing(start_h, end_time, rng, start_temp=1e-4, end_temp=1e-9, max_iters=5_000_000):
    h = start_h.copy()
    n = h.size
    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5
    iter_count = 0
    # Pre-calc total allowed iterations: maybe compute as max_iters but we break earlier.
    while time.time() < end_time and iter_count < max_iters:
        # exponential schedule
        frac = iter_count / max_iters
        T = start_temp * ((end_temp/start_temp) ** frac) if start_temp > 0 else 0.0
        # choose two indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # eps magnitude maybe proportional to max_eps and temperature: allow larger moves early.
        # Use factor = min(1.0, max_eps) * rng.random() * 0.5
        # Potentially scale by (start_temp / max(start_temp, T))? Simpler: random uniform.
        eps = max_eps * rng.random() * 0.5
        h[i] -= eps
        h[j] += eps
        new_c5 = _c5_from_h(h)

def _project_to_feasible(h, target_sum, tol=1e-9, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target_sum
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to reduce sum
            idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            h[idx] -= delta
        else:
            idx = np.where(h < 1)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size  # diff negative => delta negative, removing negative amount => addition?
            # Actually diff negative => need to increase sum => subtract diff/len? Let's compute needed increase = -diff
            # but we can just set delta = diff / idx.size (negative). Then h[idx] -= delta (subtract negative) => add.
            h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target_sum - h.sum()
    if abs(residual) > tol:
        # add/sub small amounts to some entries
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - tol:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > tol:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) <= tol:
                break
    return np.clip(h, 0.0, 1.0)

def _project_to_feasible(h, target_sum, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target_sum
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            h[idx] -= delta
        else:
            idx = np.where(h < 1)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size  # diff negative
            h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > tol:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - tol:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > tol:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) <= tol:
                break
    return np.clip(h, 0.0, 1.0)

    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

    h = x[:n]

def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-9,
                         max_iter: int = 500) -> np.ndarray:
    """Clip h to [0,1] and adjust its sum to the target using water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
