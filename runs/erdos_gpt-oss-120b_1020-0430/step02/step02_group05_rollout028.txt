import time
import numpy as np
from scipy import optimize

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n ∩ {sum = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a discrete step function on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise auxiliary variable t."""
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300) -> tuple:
    """Run SLSQP from a feasible start and return refined h and its C5."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points, then projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _sine_h(n: int) -> np.ndarray:
    """Scaled sine shape with integral 1, then projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    raw = (np.pi / 4.0) * np.sin(np.pi * x / 2.0)   # max ≈0.785 < 1
    return _project_to_simplex_box(raw, n / 2.0)


def _refine_h(h: np.ndarray, max_steps: int, sigma: float, deadline: float) -> tuple:
    """Simple hill‑climbing: keep sum, stay in [0,1], accept only improvements."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    for _ in range(max_steps):
        if time.time() >= deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # direction: try to push i up, j down
        max_up_i = min(sigma, 1.0 - best_h[i])
        max_down_j = min(sigma, best_h[j])
        if max_up_i > 0 and max_down_j > 0:
            delta = np.random.rand() * min(max_up_i, max_down_j)
            cand = best_h.copy()
            cand[i] += delta
            cand[j] -= delta
            c5 = _c5_from_h(cand)
            if c5 < best_c5:
                best_c5, best_h = c5, cand
                continue

        # opposite direction: i down, j up
        max_down_i = min(sigma, best_h[i])
        max_up_j = min(sigma, 1.0 - best_h[j])
        if max_down_i > 0 and max_up_j > 0:
            delta = np.random.rand() * min(max_down_i, max_up_j)
            cand = best_h.copy()
            cand[i] -= delta
            cand[j] += delta
            c5 = _c5_from_h(cand)
            if c5 < best_c5:
                best_c5, best_h = c5, cand

    # final exact projection to meet the sum constraint precisely
    best_h = _project_to_simplex_box(best_h, n / 2.0)
    best_c5 = _c5_from_h(best_h)
    return best_h, best_c5
