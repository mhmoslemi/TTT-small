def gradient_step(h, eta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # compute plus/minus arrays
    plus = np.zeros_like(h)
    minus = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            plus[shift:] = h[:n-shift]
            minus[:n-shift] = h[shift:]
    else:
        s = -shift
        if s < n:
            plus[:n-s] = h[s:]
            minus[s:] = h[:n-s]
    grad = 1.0 - (plus + minus)
    # step
    h_new = h - eta * grad
    # project
    h_new = _project_to_feasible(h_new, n)
    return h_new
```

def hill_climb(h, best_c5, max_iter):
    n = h.size
    for it in range(max_iter):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # ensure we can move from i to j
        if h[i] <= 1e-9 or h[j] >= 1-1e-9:
            continue
        eps = min(0.02, h[i], 1 - h[j]) * np.random.rand()
        if eps < 1e-6:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # projection not needed because sum unchanged and bounds preserved (if eps respects)
        new_c5 = _c5_from_h(h_new)
        if new_c5 < best_c5 - 1e-9:
            h = h_new
            best_c5 = new_c5
            best_h = h_new.copy()
        # optionally break early if no improvement after many trials
        if time_left() < 3:
            break
    return h, best_c5, best_h

import time
import numpy as np

# --------------------------------------------------------------
# Feasibility projection (clip to [0,1] and enforce sum = n/2)
# --------------------------------------------------------------
def _project_to_feasible(h, tol=1e-12, max_iter=20):
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        diff = h.sum() - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0.0
        else:
            free = h < 1.0
        if not np.any(free):
            break
        delta = diff / np.sum(free)
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny fix to hit the target exactly
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        # add/subtract residual from the first non‑boundary entry
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if idx.size:
            i = idx[0]
            h[i] = np.clip(h[i] + residual, 0.0, 1.0)
    return h


# --------------------------------------------------------------
# C5 evaluation (max correlation * 2/n)
# --------------------------------------------------------------
def _c5_from_h(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# --------------------------------------------------------------
# Upscale by linear interpolation
# --------------------------------------------------------------
def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# --------------------------------------------------------------
# Single gradient step for the worst shift
# --------------------------------------------------------------
def _gradient_step(h, eta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)               # integer shift producing the worst overlap

    plus = np.zeros_like(h)
    minus = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            plus[shift:] = h[: n - shift]
            minus[: n - shift] = h[shift:]
    else:
        s = -shift
        if s < n:
            plus[: n - s] = h[s:]
            minus[s:] = h[: n - s]

    grad = 1.0 - (plus + minus)        # sub‑gradient of the active term
    h_new = h - eta * grad
    return _project_to_feasible(h_new)


# --------------------------------------------------------------
# Simple hill‑climbing using pairwise mass transfers
# --------------------------------------------------------------
def _hill_climb(h, cur_c5, max_trials, time_left):
    n = h.size
    best_h = h.copy()
    best_val = cur_c5
    for _ in range(max_trials):
        if time_left() < 0.5:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if best_h[i] <= 1e-12 or best_h[j] >= 1.0 - 1e-12:
            continue
        eps = min(0.02, best_h[i], 1.0 - best_h[j]) * np.random.rand()
        if eps < 1e-10:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        val = _c5_from_h(cand)
        if val + 1e-12 < best_val:
            best_h, best_val = cand, val
    return best_h, best_val


# --------------------------------------------------------------
# Main optimisation driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Load optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is not None:
        best_h = _project_to_feasible(init_h)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size
    else:
        n0 = 256
        best_h = _project_to_feasible(np.random.rand(n0))
        best_c5 = _c5_from_h(best_h)
        best_n = n0

    # ------------------------------------------------------------------
    # Candidate grid sizes (even, up to 512)
    # ------------------------------------------------------------------
    n_candidates = [128, 160, 200, 256, 320, 400, 512]
    n_candidates = [n for n in n_candidates if n <= 512]
    n_candidates = sorted(set(n_candidates))

    # ------------------------------------------------------------------
    # Main multi‑resolution search
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # starting vector for this resolution
        if best_n == n:
            cur_h = best_h.copy()
        else:
            cur_h = _upscale_h(best_h, n)
        cur_h = _project_to_feasible(cur_h)

        # --------------------------------------------------------------
        # Gradient descent on the worst shift
        # --------------------------------------------------------------
        max_iter = 250 if n <= 256 else 180
        eta0 = 0.5
        cur_c5 = _c5_from_h(cur_h)
        for it in range(max_iter):
            if time_left() < 1.5:
                break
            eta = eta0 / np.sqrt(it + 1.0)
            new_h = _gradient_step(cur_h, eta)
            new_c5 = _c5_from_h(new_h)

            # accept only if non‑worsening (guarantees monotone descent)
            if new_c5 <= cur_c5 + 1e-12:
                cur_h, cur_c5 = new_h, new_c5
                if cur_c5 < best_c5 - 1e-12:
                    best_h, best_c5, best_n = cur_h.copy(), float(cur_c5), n

            # early exit if we already beat target
            if best_c5 < 0.3808:
                break

        # --------------------------------------------------------------
        # Local hill‑climbing fine‑tuning
        # --------------------------------------------------------------
        if time_left() > 2.0:
            trials = 3000 if n <= 256 else 2000
            cur_h, cur_c5 = _hill_climb(cur_h, cur_c5, trials, time_left)
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5, best_n = cur_h.copy(), float(cur_c5), n

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final safeguard projection & evaluation
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_h.size)
