def simulated_annealing_optimize(h_start, rng, elapsed, budget):
    target = h_start.size / 2.0
    h = h_start.copy()
    c_cur = compute_c5(h)
    best_h = h.copy()
    best_c = c_cur
    # allocate max time fraction: remaining = budget - elapsed; want to spend <= (#) sec
    max_time = min(30.0, budget - elapsed)  # limit to some seconds per SA call
    start = time.time()
    max_iters = 20000
    # schedule parameters
    T0 = 0.01
    Tf = 1e-5
    for it in range(max_iters):
        # time check
        if time.time() - start > max_time:
            break
        # compute temperature schedule
        t = T0 * ((Tf / T0) ** (it / max_iters))
        # random move
        n = h.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0≤h≤1, Σh=n/2)."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def symmetrize(v):
    return (v + v[::-1]) * 0.5


# ---------- hill‑climb ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_h = symmetrize(new_h)
        new_h = project_onto_box_sum(new_h, h.size / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth‑max gradient descent ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()

            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            new_h = symmetrize(new_h)
            new_h = project_onto_box_sum(new_h, n / 2.0)

            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # occasional uphill to escape plateaus
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr * (2.0 / x.size)
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

def simulated_annealing_optimize(h_start, rng, elapsed, budget):
    """Simple SA with time‑bound."""
    target = h_start.size / 2.0
    h = h_start.copy()
    c_cur = compute_c5(h)
    best_h, best_c = h.copy(), c_cur

    # give this SA call at most 30 s or the remaining budget portion
    max_time = min(30.0, budget - elapsed)
    start = time.time()
    max_iters = 20000
    T0, Tf = 0.02, 1e-5

    for it in range(max_iters):
        if time.time() - start > max_time:
            break
        t = T0 * ((Tf / T0) ** (it / max_iters))
        n = h.size
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        step_scale = max(0.02, t)      # larger steps early, tiny later
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step_scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = symmetrize(new_h)
        new_h = project_onto_box_sum(new_h, target)
        c_new = compute_c5(new_h)
        delta = c_new - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            h, c_cur = new_h, c_new
            if c_new < best_c:
                best_c, best_h = c_new, new_h.copy()
    return best_h, best_c
