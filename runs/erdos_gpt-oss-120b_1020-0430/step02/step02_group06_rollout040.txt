def _binary_sa(initial_h, max_iter, time_left):
    h = initial_h.copy()
    n = len(h)
    cur_c5 = _c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # maintain index lists
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    # temperature schedule
    T0 = 0.02
    T_end = 0.0005
    for i in range(max_iter):
        if time_left() < 0.03:
            break
        # linear cooling
        T = T0 + (T_end - T0) * (i / max_iter)
        # pick swap indices
        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]
        # perform swap
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = _c5_fft(h)
        delta = new_c5 - cur_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            # update lists
            # replace i1 with i0 in ones
            idx1 = ones.index(i1)
            ones[idx1] = i0
            # replace i0 with i1 in zeros
            idx0 = zeros.index(i0)
            zeros[idx0] = i1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i1] = 1.0
            h[i0] = 0.0
        # optional early exit if best_c5 already below target
    # optional greedy refinement
    # try a few random swaps that always accept if improve
    for _ in range(200):
        if time_left() < 0.01:
            break
        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = _c5_fft(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            # update lists as before
            idx1 = ones.index(i1)
            ones[idx1] = i0
            idx0 = zeros.index(i0)
            zeros[idx0] = i1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return best_h, best_c5

def _upscale_h(coarse_h, n_fine):

def _solve_continuous(initial_h, max_iter=600):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    def ineq(x):
        h = x[:n]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full')
        return t - c
    def eq(x):
        return np.sum(x[:n]) - n/2.0
    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]
    opts = {'maxiter':max_iter, 'ftol':1e-9, 'disp':False}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                            constraints=cons, options=opts)
    if res.success and res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_fft(h_opt)
    return h_opt, c5

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy import optimize

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_fft(h):
    n = h.size
    rev = (1.0 - h)[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    R = np.fft.rfft(rev, N)
    conv = np.fft.irfft(H * R, N)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _solve_continuous(initial_h, max_iter=800):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full')
        return t - c

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP',
                            bounds=bounds, constraints=cons, options=opts)
    if res.success and res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_fft(h_opt)
    return h_opt, c5

def _binary_sa(initial_h, max_iter, start_time, budget_s):
    h = initial_h.copy()
    n = h.size
    cur_c5 = _c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    # linear temperature schedule
    T0 = 0.02
    T_end = 0.0005
    for i in range(max_iter):
        if budget_s - (time.time() - start_time) < 0.02:
            break
        T = T0 + (T_end - T0) * (i / max_iter)
        # pick a random swap
        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]
        # perform swap
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = _c5_fft(h)
        delta = new_c5 - cur_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            # update index lists
            idx1 = ones.index(i1)
            ones[idx1] = i0
            idx0 = zeros.index(i0)
            zeros[idx0] = i1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # reject
            h[i1] = 1.0
            h[i0] = 0.0
    # small greedy refinement
    for _ in range(200):
        if budget_s - (time.time() - start_time) < 0.01:
            break
        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = _c5_fft(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            idx1 = ones.index(i1)
            ones[idx1] = i0
            idx0 = zeros.index(i0)
            zeros[idx0] = i1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return best_h, best_c5

def _subgradient_opt(h, steps, start_time, budget_s):
    h_opt = h.copy()
    n = h_opt.size
    best_c5 = _c5_fft(h_opt)
    for step in range(steps):
        if budget_s - (time.time() - start_time) < 0.01:
            break
        corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n - 1)
        g = np.zeros_like(h_opt)
        if shift >= 0:
            s = shift
            g[:n - s] += 1.0 - h_opt[s:]
            g[s:] -= h_opt[:n - s]
        else:
            s = -shift
            g[s:] += 1.0 - h_opt[:n - s]
            g[:n - s] -= h_opt[s:]
        alpha = 0.1 / (step + 1)
        h_new = h_opt - alpha * g
        h_new = _project_to_feasible(h_new, n)
        new_c5 = _c5_fft(h_new)
        if new_c5 < best_c5:
            h_opt = h_new
            best_c5 = new_c5
    return h_opt, best_c5

import time

from scipy import optimize

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_fft(h):
    n = h.size
    rev = (1.0 - h)[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    R = np.fft.rfft(rev, N)
    conv = np.fft.irfft(H * R, N)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _solve_continuous(initial_h, max_iter=800):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full')
        return t - c

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP',
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success and res.x is not None else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_fft(h_opt)

def _subgradient_opt(h, steps, start_time, budget_s):
    h_opt = h.copy()
    n = h_opt.size
    best_c5 = _c5_fft(h_opt)
    for step in range(steps):
        if budget_s - (time.time() - start_time) < 0.01:
            break
        corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n - 1)
        g = np.zeros_like(h_opt)
        if shift >= 0:
            s = shift
            g[:n - s] += 1.0 - h_opt[s:]
            g[s:] -= h_opt[:n - s]
        else:
            s = -shift
            g[s:] += 1.0 - h_opt[:n - s]
            g[:n - s] -= h_opt[s:]
        alpha = 0.1 / (step + 1)
        h_new = h_opt - alpha * g
        h_new = _project_to_feasible(h_new, n)
        new_c5 = _c5_fft(h_new)
        if new_c5 < best_c5:
            h_opt = h_new
            best_c5 = new_c5
    return h_opt, best_c5
