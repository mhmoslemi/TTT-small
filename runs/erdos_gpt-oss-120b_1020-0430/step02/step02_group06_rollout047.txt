def hill_climb(h, best_h, best_c5, n, time_left):
    factor = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    current_c5 = corr[max_idx] * factor
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    it = 0
    max_iters = 100000  # but break earlier by time
    while time_left() > 0.2 and it < max_iters:
        # shift
        max_idx = np.argmax(corr)
        shift = max_idx - (n-1)
        # Determine overlap
        if shift >= 0:
            i_start = 0
            i_end = n - shift
        else:
            i_start = -shift
            i_end = n
        length = i_end - i_start
        # selection
        if length > 0:
            idx = np.arange(i_start, i_end)
            contrib = h[idx] * (1.0 - h[idx + shift])
            # top indices
            top_k = min(5, length)
            # random selection among top_k
            # using argpartition
            top_local = np.argpartition(-contrib, top_k-1)[:top_k]
            i = int(idx[top_local[np.random.randint(len(top_local))]])
        else:
            # degenerate, random
            i = np.random.randint(n)
            shift = 0
        j = i + shift
        # decide delta
        max_delta = min(0.04, h[i], 1 - h[j])
        if max_delta <= 1e-12:
            # try opposite direction
            max_delta = min(0.04, h[j], 1 - h[i])
            if max_delta <= 1e-12:
                # can't move, continue
                it += 1
                continue
            # swap i,j
            i, j = j, i
        delta = max_delta * np.random.rand()
        # apply move
        h[i] -= delta
        h[j] += delta
        # compute new correlation
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(new_corr) * factor
        if new_c5 + 1e-12 < current_c5:

                    return best_h, best_c5, True
        else:

import time
import numpy as np
from scipy import optimize

# ---------- feasibility utilities ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


# ---------- SLSQP optimisation ----------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)
    h_opt = _project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse_h)


# ---------- hill‑climbing refinement ----------
def _hill_climb(h: np.ndarray, best_h: np.ndarray, best_c5: float,
                time_left, target=0.3808) -> tuple:
    """Move mass from a high‑value cell to a low‑value cell in the shift that attains max."""
    n = h.size
    factor = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_c5 = np.max(corr) * factor
    if cur_c5 < best_c5:
        best_h, best_c5 = h.copy(), cur_c5
    max_iters = 20000
    it = 0
    while time_left() > 0.3 and it < max_iters:
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # overlapping interval for this shift
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n
        length = i_end - i_start
        if length <= 0:
            i = np.random.randint(n)
            j = i
        else:
            idx = np.arange(i_start, i_end)
            contrib = h[idx] * (1.0 - h[idx + shift])
            top_k = min(5, length)
            top_local = np.argpartition(-contrib, top_k - 1)[:top_k]
            i = int(idx[top_local[np.random.randint(top_local.size)]])
            j = i + shift
        # decide delta respecting bounds
        max_delta = min(0.04, h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            # try opposite direction
            max_delta = min(0.04, h[j], 1.0 - h[i])
            if max_delta <= 1e-12:
                it += 1
                continue
            i, j = j, i
        delta = max_delta * np.random.rand()
        h[i] -= delta
        h[j] += delta
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_c5 = np.max(new_corr) * factor
        # accept only if improvement
        if new_c5 + 1e-12 < cur_c5:
            corr = new_corr
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = h.copy(), cur_c5
                if best_c5 < target:
                    return best_h, best_c5, True
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        it += 1
    return best_h, best_c5, False
