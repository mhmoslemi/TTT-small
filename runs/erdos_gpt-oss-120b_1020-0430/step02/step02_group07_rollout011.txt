def compute_c5(h):
    n = h.size
    # zero-pad to length 2n
    m = 2 * n
    # compute convolution of h with reversed (1-h)
    # correlation c[k] = sum_i h[i] * (1-h)[i+k] where i and i+k index valid (i from 0..n-1, i+k from 0..n-1)
    # This is same as conv(h, rev(1-h)) [valid indices correspond].
    # let a = h, b = (1-h)[::-1]   # reversed
    a = h
    b = (1 - h)[::-1]
    # compute via FFT
    A = np.fft.rfft(a, n=m)
    B = np.fft.rfft(b, n=m)
    c = np.fft.irfft(A * B, n=m)
    # c length = m = 2n. The correlation values for shift k = -(n-1) .. (n-1) correspond to indices n-1+k
    # So we need max over indices 0..2n-2? Actually correlation length 2n-1; but FFT result length 2n includes extra trailing zero. That's fine; we can ignore last element.
    corr = c[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()

    best_c = compute_c5(h)

def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02, decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        # accept?
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(temp, 1e-12)):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

def softmax_obj(x):
    corr = full_corr(x)

    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# globals used by the soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def full_corr(x):
    """Full (non‑circular) correlation of x with (1‑x) using FFT.
    Returns length 2·n‑1 array where n = len(x)."""
    n = x.shape[0]
    a_rev = x[::-1]
    b = 1.0 - x
    m = 2 * n
    A = np.fft.rfft(a_rev, n=m)
    B = np.fft.rfft(b, n=m)
    c = np.fft.irfft(A * B, n=m)
    return c[: 2 * n - 1]


def compute_c5(h):
    """Exact discrete C₅ for a step function h on [0,2]."""
    n = h.size
    corr = full_corr(h)
    return np.max(corr) * (2.0 / n)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of maxₖ Σᵢ xᵢ (1‑xᵢ₊ₖ) using log‑sum‑exp."""
    corr = full_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # gradual step reduction
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(temp, 1e-12)):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def quick_refine(h, rng):
    """Fast refinement with a few β values and a short hill‑climb."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=1200, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def deep_refine(h, rng, time_limit=None):
    """Intensive refinement: several β values, hill‑climb, then SA."""
    start = time.time()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    slsqp_iter = 200 if n > 200 else 300
    hill_iters = 1500 if n > 200 else 2500
    anneal_iters = 3000 if n > 200 else 5000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 2000.0)

    for beta in betas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=slsqp_iter)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        if time_limit is not None and time.time() - start > time_limit:
            break
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h

    if time_limit is None or (time.time() - start) < (time_limit or 0):
        trial_sa, c_sa = simulated_annealing(
            best_h,
            rng,
            max_iters=anneal_iters,
            init_temp=0.03,
            decay=0.998,
            scale=0.2,
        )
        if c_sa < best_c:
            best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    target = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target, rng)
    return h_up


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C₅ overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initialise
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = quick_refine(h0, rng)

    # ----------------------------------------------------------------------
    # Phase 1 – many random restarts with quick refinement
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.60:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----------------------------------------------------------------------
    # Phase 2 – up‑sample and deep refinement
    # ----------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > budget_s * 0.15 and n * 3 <= 1200:
        # one up‑sampling step (factor 3)
        h_up = upsample_h(best_h, factor=3, rng=rng)
        n_up = h_up.size
        deep_time = remaining * 0.90
        h_up_ref, c_up_ref = deep_refine(h_up, rng, time_limit=deep_time)
        if c_up_ref < best_c:
            best_h, best_c, n = h_up_ref, c_up_ref, n_up

    # ----------------------------------------------------------------------
    # Final feasibility clean‑up
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
