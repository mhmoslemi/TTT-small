def improve_worst_shift(h, rng):
    while True:
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        worst_c = corr[max_idx]
        # compute shift
        k = max_idx - (len(h) - 1)
        # compute gradient for that shift: g_i = (1 - h_{i+k}) - h_{i - k} if both indexes in range; edges handle.
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2 * h
        else:
            if k > 0:
                # g[i] = (1 - h[i+k]) for i in [0, n-k)
                g[:len(h)-k] = 1 - h[k:]
                # g[i] -= h[i-k] for i in [k, n)
                g[k:] -= h[:len(h)-k]
            else:
                k_abs = -k
                g[k_abs:] = 1 - h[:len(h)-k_abs]
                g[:len(h)-k_abs] -= h[k_abs:]
        # Transfer along gradient direction:
        # find i_pos = argmax(g) (most positive)
        # find i_neg = argmin(g) (most negative)
        # if g[i_pos] <= 0 or g[i_neg] >= 0 => cannot improve linear approx; break
        i_pos = np.argmax(g)
        i_neg = np.argmin(g)
        if g[i_pos] <= 0 or g[i_neg] >= 0:
            break
        # max epsilon
        max_eps = min(h[i_neg], 1 - h[i_pos])
        if max_eps <= 0:
            break
        # step size as fraction of max_eps, maybe random between 0.1 and 1
        eps = max_eps * rng.random()
        # Apply transfer
        h_new = h.copy()
        h_new[i_pos] += eps
        h_new[i_neg] -= eps
        # Evaluate new C5; if improved, accept, else maybe try smaller step?
        new_c = compute_c5(h_new)
        if new_c < compute_c5(h) - 1e-9:
            h = h_new
        else:
            # try smaller step
            # reduces eps gradually
            eps = max_eps * (rng.random() * 0.5)
            if eps < 1e-12:
                break
            h_new[i_pos] += eps
            h_new[i_neg] -= eps
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
            else:
                break
    return h
```

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
    else:  # k < 0
        km = -k
        g[km:] = 1.0 - h[: n - km]
        g[: n - km] -= h[km:]
    return g, corr[idx], k


def gradient_transfer_move(h, rng):
    """Transfer mass from a most negative‑gradient coordinate to a most positive‑gradient one."""
    g, _, _ = worst_shift_grad(h)
    i_pos = int(np.argmax(g))
    i_neg = int(np.argmin(g))
    if g[i_pos] <= 0 or g[i_neg] >= 0:
        return h
    max_eps = min(h[i_neg], 1.0 - h[i_pos])
    if max_eps <= 0:
        return h
    eps = max_eps * (rng.random() * 0.5 + 0.5)   # 0.5 – 1.0 of the allowable amount
    h_new = h.copy()
    h_new[i_pos] += eps
    h_new[i_neg] -= eps
    return h_new


def swap_move(h, rng):
    """Swap two entries."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    if h[i] == h[j]:
        return h
    h_new = h.copy()
    h_new[i], h_new[j] = h[j], h[i]
    return h_new


def perturb_move(h, rng, scale=0.1):
    """Small epsilon transfer between two random coordinates."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new


# ---------- local search ----------
def local_search(h0, rng, max_iters, max_no_improve=200):
    """Simple stochastic descent that accepts improving moves."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    no_improve = 0

    for _ in range(max_iters):
        move_type = rng.random()
        if move_type < 0.6:
            h_new = gradient_transfer_move(h, rng)
        elif move_type < 0.9:
            h_new = swap_move(h, rng)
        else:
            h_new = perturb_move(h, rng, scale=0.08)

        c_new = compute_c5(h_new)

        if c_new + 1e-12 < best_c:
            best_c, best_h = c_new, h_new.copy()
            h = h_new
            no_improve = 0
        elif c_new + 1e-12 < compute_c5(h):
            h = h_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            h = best_h.copy()
            no_improve = 0

    return best_h, best_c


# ---------- soft‑max surrogate for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stabilised
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """One SLSQP pass using the smooth approximation of max overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (fast projection)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) <= 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ----- use supplied parent solution -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target0 = n0 / 2.0
        diff = target0 - h0.sum()
        for _ in range(2000):
            if abs(diff) <= 1e-12:
                break
            i = rng.integers(0, n0)
            if diff > 0:
                add = min(diff, 1.0 - h0[i])
            else:
                add = max(diff, -h0[i])
            h0[i] += add
            diff = target0 - h0.sum()
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # ----- grid sizes to try -----
    cand_ns = [72, 84, 96, 108, 120, 144, 168]
    if best_n is not None and best_n not in cand_ns:
        cand_ns = [best_n] + cand_ns
    cand_ns = [n for n in cand_ns if n % 2 == 0]

    for n in cand_ns:
        if time.time() - start > budget_s * 0.95:
            break

        # number of restarts per size (bounded by time)
        max_restarts = 12
        base_iters = 6000
        max_iters = max(2000, int(base_iters * 100.0 / n))

        for attempt in range(max_restarts):
            if time.time() - start > budget_s * 0.96:
                break

            if best_n == n and best_h is not None and attempt == 0:
                h_start = best_h.copy()
            else:
                if rng.random() < 0.5:
                    h_start = binary_feasible(n, rng)
                else:
                    h_start = continuous_feasible(n, rng)

            h_opt, c_opt = local_search(h_start, rng, max_iters)
            if c_opt < best_c - 1e-12:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n

            # ----- short SLSQP polishing on this candidate -----
            if time.time() - start < budget_s * 0.97:
                for beta in (5.0, 20.0, 80.0):
                    if time.time() - start > budget_s * 0.98:
                        break
                    h_opt = slsqp_optimize(h_opt, beta, rng, max_iter=250)
                    c_opt = compute_c5(h_opt)
                    if c_opt < best_c - 1e-12:
                        best_h, best_c, best_n = h_opt.copy(), c_opt, n

    # ----- optional up‑sampling refinement -----
    if best_h is not None and time.time() - start < budget_s * 0.99:
        if best_n * 2 <= 240:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, best_n * 2, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            target_up = (best_n * 2) / 2.0
            diff = target_up - h_up.sum()
            for _ in range(2000):
                if abs(diff) <= 1e-12:
                    break
                i = rng.integers(0, h_up.size)
                if diff > 0:
                    add = min(diff, 1.0 - h_up[i])
                else:
                    add = max(diff, -h_up[i])
                h_up[i] += add
                diff = target_up - h_up.sum()
            # refine
            h_up, c_up = local_search(h_up, rng, max_iters // 2)
            if c_up < best_c - 1e-12:
                best_h, best_c, best_n = h_up, c_up, h_up.size

    # ----- final SLSQP polishing on the global best -----
    if best_h is not None and time.time() - start < budget_s * 0.995:
        for beta in (5.0, 20.0, 80.0, 200.0):
            if time.time() - start > budget_s * 0.998:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)

    # ----- final feasibility check -----
    if best_h is None:
        # fallback: binary 80‑point construction
        best_n = 80
        best_h = binary_feasible(best_n, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    for _ in range(5000):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, best_n)
        if diff > 0:
            add = min(diff, 1.0 - best_h[i])
        else:
            add = max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
