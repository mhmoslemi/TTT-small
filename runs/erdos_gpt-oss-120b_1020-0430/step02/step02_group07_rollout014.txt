```
def compute_c5_fft(h):
    n = h.size
    m = 1 << ((2*n - 1)-1).bit_length()
    H = np.fft.fft(h, m)
    G = np.fft.fft(1-h, m)
    corr = np.fft.ifft(H * G).real[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

def upsample_float_to_n(h_parent, n):
    m = h_parent.size
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, n, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_parent)

    np.clip(h_interp, 0, 1, out=h_interp)

def binary_from_continuous(h_cont, k, rng):
    # h_cont in [0,1]. Choose k positions with highest values as 1.
    # If tie, break randomly.
    idx = np.argsort(-h_cont)  # descending
    # Determine threshold
    threshold = h_cont[idx[k-1]]
    # All above threshold set to 1
    binary = np.zeros_like(h_cont, dtype=np.float64) # maybe int
    binary[idx[:k]] = 1.0
    # If exactly k entries have threshold equal and more duplicates, we need random tie resolution.
    # However, simple approach might overshoot duplicates: we could handle ties.
    # Let's implement:
    ties = np.where(np.isclose(h_cont, threshold))[0]
    if ties.size > (k - np.sum(h_cont > threshold)):
        # need random choose among ties
        need = k - np.sum(h_cont > threshold)
        chosen = rng.choice(ties, size=need, replace=False)
        binary[h_cont > threshold] = 1.0
        binary[chosen] = 1.0
    else:
        binary[h_cont > threshold] = 1.0
        binary[ties] = 1.0
    return binary.astype(float)

def binary_from_continuous(vals, k, rng):
    # Random tie-breaking: shuffle before argsort
    perm = rng.permutation(len(vals))
    shuffled_vals = vals[perm]
    order = np.argsort(-shuffled_vals, kind='mergesort')
    selected = perm[order[:k]]
    binary = np.zeros(len(vals), dtype=np.float64)
    binary[selected] = 1.0
    return binary
```

def hill_climb_binary(h, rng, max_no_improve=1000):
    n = len(h)
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(1-h).astype(int))
    cur_c = compute_c5_fft(h)
    no_improve = 0
    while no_improve < max_no_improve:
        # try a random swap
        pos_i = rng.integers(len(ones))
        i = ones[pos_i]
        pos_j = rng.integers(len(zeros))
        j = zeros[pos_j]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        ones[pos_i] = j
        zeros[pos_j] = i
        new_c = compute_c5_fft(h)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            no_improve = 0
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            ones[pos_i] = i
            zeros[pos_j] = j
            no_improve += 1
    return h, cur_c
```

def binary_from_continuous(vals, k, rng):
    perm = rng.permutation(len(vals))
    shuffled = vals[perm]
    order = np.argsort(-shuffled, kind='mergesort')
    selected = perm[order[:k]]
    binary = np.zeros(len(vals), dtype=np.float64)
    binary[selected] = 1.0
    return binary
```

def hill_climb_binary(h0, rng, max_no_improve=500, deadline=None):
    h = h0.copy()
    n = h.size
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(1 - h).astype(int))
    cur_c = corr_max(h)
    best_h = h.copy()

    no_improve = 0
    while no_improve < max_no_improve:
        if deadline is not None and time.time() > deadline:

        i_pos = rng.integers(len(ones))

            no_improve = 0

            # revert
            h[i] = 1.0
            h[j] = 0.0
            ones[i_pos] = i
            zeros[j_pos] = j

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum if needed (should not happen)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple correction: add diff to first coordinate within bounds
        for i in range(n):
            adj = min(max(diff, -x[i]), 1.0 - x[i])
            x[i] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            adj = min(max(diff, -x[i]), 1.0 - x[i])
            x[i] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    return x


def corr_max(h):
    """Exact discrete C₅ for a step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def random_binary_vector(n, k, rng):
    """Binary vector of length n with exactly k ones."""
    idx = rng.choice(n, size=k, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def upsample_float_to_n(arr, n):
    """Linear interpolation of a float array to length n."""
    m = arr.size
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    interp = np.interp(x_new, x_old, arr)
    np.clip(interp, 0.0, 1.0, out=interp)
    return interp


def binary_from_continuous(vals, k, rng):
    """Select k positions with highest values, random tie‑break, produce binary."""
    perm = rng.permutation(len(vals))
    shuffled = vals[perm]
    order = np.argsort(-shuffled, kind="mergesort")
    chosen = perm[order[:k]]
    bin_vec = np.zeros(len(vals), dtype=np.float64)
    bin_vec[chosen] = 1.0
    return bin_vec


def hill_climb_binary(h0, rng, max_no_improve=500, deadline=None):
    """Greedy descent on binary vectors (swap 1↔0)."""
    h = h0.copy()
    n = h.size
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(1 - h).astype(int))
    cur_c = corr_max(h)
    best_h, best_c = h.copy(), cur_c
    no_improve = 0
    while no_improve < max_no_improve:
        if deadline is not None and time.time() > deadline:
            break
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i, j = ones[i_pos], zeros[j_pos]

        # swap
        h[i] = 0.0
        h[j] = 1.0
        ones[i_pos] = j
        zeros[j_pos] = i

        new_c = corr_max(h)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            best_h, best_c = h.copy(), cur_c
            no_improve = 0
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            ones[i_pos] = i
            zeros[j_pos] = j
            no_improve += 1
    return best_h, best_c


def simulated_annealing_binary(h0, rng, max_iters, init_temp, decay, deadline):
    """SA that swaps a random 1 with a random 0, respecting the sum constraint."""
    h = h0.copy()
    n = h.size
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(1 - h).astype(int))
    cur_c = corr_max(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i, j = ones[i_pos], zeros[j_pos]

        h[j] = 1.0

        zeros[j_pos] = i

        if accept:

        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            ones[i_pos] = i
