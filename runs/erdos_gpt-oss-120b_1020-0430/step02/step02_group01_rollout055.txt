def binary_swap_hill_climb(h, rng, max_iters=5000):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = len(h)
    for it in range(max_iters):
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if best_h[i]==best_h[j]:
            continue
        # ensure i is 1 and j is 0 or opposite; swap
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
    return best_h, best_c
```

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    z = CURRENT_BETA * corr
    m = np.max(z)
    # log-sum-exp stable
    return (m + np.log(np.sum(np.exp(z - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h0, rng, max_iters=2000, start_step=0.5):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h0, rng, step=0.02, max_iter=8):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        # repair sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_swap_hill_climb(h0, rng, max_iters=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:

        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
    return best_h, best_c

def enforce_feasibility(h, rng):

            if c_cont < best_c:
                best_h, best_c, best_n = h_cont.copy(), c_cont, best_n

    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    best_h = enforce_feasibility(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

                best_h, best_c, best_n = h_cont.copy(), c_cont, best_n
        if time.time() - start < budget_s * 0.97:
            h_cont, c_cont = hill_climb(h_cont, rng, max_iters=2000, start_step=0.3)
            if c_cont < best_c:
                best_h, best_c, best_n = h_cont.copy(), c_cont, best_n

    # optional up‑sampling refinement
    if best_h is not None and time.time() - start < budget_s * 0.98:
        if best_n * 2 <= 240:
            h_up = np.repeat(best_h, 2)
            h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.3)
            for beta in (5, 20, 80):
                if time.time() - start > budget_s * 0.99:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, h_up.size

    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    best_h = enforce_feasibility(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
