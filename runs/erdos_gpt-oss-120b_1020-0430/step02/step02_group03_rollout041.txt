import time, math, numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def propose_move(h, rng, scale=0.5):
    """Transfer a random amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # decide direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        return new_h
    return h.copy()


# --------------------------------------------------------------
# SLSQP surrogate optimizer (smooth log‑sum‑exp of max)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # small drift correction
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Main optimisation routine
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 150
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing ---------------------------------------
    T0, T_end = 0.05, 1e-6
    scale_max, scale_min = 0.5, 0.01
    cur_h, cur_c = best_h.copy(), best_c

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.94:
            break
        progress = elapsed / (budget_s * 0.94)
        T = T0 * (1 - progress) + T_end * progress
        scale = scale_max * (1 - progress) + scale_min * progress

        new_h = propose_move(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)

        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c

        # occasional random restart to retain diversity
        if rng.random() < 0.001:
            if rng.random() < 0.5:
                cur_h, cur_c = best_h.copy(), best_c
            else:
                cur_h = random_feasible_vector(n, rng)
                cur_c = compute_c5(cur_h)

    # ----- local refinement with smooth surrogate --------------------
    beta_seq = [5.0, 20.0, 80.0, 200.0]
    for beta in beta_seq:
        if time.time() - start_time > budget_s * 0.98:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=400)
        except Exception:
            continue
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_h, best_c = refined, refined_c

    # ----- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
