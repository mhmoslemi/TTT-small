def propose_move(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(n, size=2)
    if i == j:
        return h
    # choose epsilon
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # apply
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

CURRENT_TARGET = None; CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    scaled = CURRENT_BETA * corr
    m = np.max(scaled)
    # log-sum-exp stable
    return (m + np.log(np.sum(np.exp(scaled - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum if deviates
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff using simple loop
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    scaled = CURRENT_BETA * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def repair_sum_and_bounds(h, target_sum, rng):
    """Adjust h so that sum(h)=target_sum while keeping 0≤h_i≤1."""
    diff = target_sum - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff -= add
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff += sub
    # final tiny uniform correction if needed
    if abs(diff) >= 1e-12:
        uniform = diff / n
        h += uniform
        h = np.clip(h, 0.0, 1.0)

        diff = target_sum - h.sum()

def resample_h(h, n_target):
    """Linear interpolation of h onto a new grid of size n_target."""
    n_src = h.size
    xs_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xs_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(xs_tgt, xs_src, h)
    return np.clip(h_new, 0.0, 1.0)

                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Initialise from supplied parent if present
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = repair_sum_and_bounds(h0, n0 / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # ------------------------------------------------------------------
    # Candidate discretisations (keep size modest for speed)
    # ------------------------------------------------------------------
    candidates = [96, 108, 120, 144, 168, 192, 216, 240]
    if best_n is not None and best_n in candidates:
        candidates = [best_n] + [n for n in candidates if n != best_n]

    # ------------------------------------------------------------------
    # Main optimisation over different n
    # ------------------------------------------------------------------
    for idx, n in enumerate(candidates):
        if time.time() - start_time > budget_s * 0.98:
            break

        # Number of independent attempts for this n
        max_attempts = 2
        for attempt in range(max_attempts):
            if time.time() - start_time > budget_s * 0.98:
                break

            # ------------------------------------------------------------------
            # Initialise a feasible vector for this n
            # ------------------------------------------------------------------
            if best_h is not None and best_n != n:
                h = resample_h(best_h, n)
            elif best_h is not None and best_n == n and attempt == 0:
                h = best_h.copy()
            else:
                # random start
                h = random_feasible_vector(n, rng)

            h = repair_sum_and_bounds(h, n / 2.0, rng)

            # ------------------------------------------------------------------
            # Time budgeting for this attempt
            # ------------------------------------------------------------------
            remaining = budget_s - (time.time() - start_time)
            # allocate roughly half to annealing, remainder to refinements
            anneal_time = min(remaining * 0.5, 30.0)  # cap to keep schedule sane
            h_anneal, c_anneal = anneal_optimize(h, rng, anneal_time,
                                                start_temp=0.02,
                                                end_temp=1e-7,
                                                scale=0.3)

            if c_anneal < best_c:
                best_h, best_c, best_n = h_anneal.copy(), c_anneal, n

            # ------------------------------------------------------------------
            # Local hill‑climb refinement
            # ------------------------------------------------------------------
            h_hc, c_hc = hill_climb(h_anneal, rng, max_iters=2000, start_step=0.3)
            if c_hc < best_c:
                best_h, best_c, best_n = h_hc.copy(), c_hc, n

            # ------------------------------------------------------------------
            # Gradient‑based fine tuning
            # ------------------------------------------------------------------
            h_gr = gradient_refine(best_h, rng, step=0.02, max_iter=6)
            c_gr = compute_c5(h_gr)
            if c_gr < best_c:
                best_h, best_c, best_n = h_gr.copy(), c_gr, n

            # ------------------------------------------------------------------
            # SLSQP surrogate optimisation with a few betas
            # ------------------------------------------------------------------
            if time.time() - start_time < budget_s * 0.97:
                for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                    if time.time() - start_time > budget_s * 0.99:
                        break
                    try:
                        h_slsqp = slsqp_optimize(best_h, beta, rng, max_iter=400)
                    except Exception:
                        continue
                    c_slsqp = compute_c5(h_slsqp)
                    if c_slsqp < best_c:
                        best_h, best_c, best_n = h_slsqp.copy(), c_slsqp, n

            # ------------------------------------------------------------------
            # Quick final hill‑climb (few iterations) to polish
            # ------------------------------------------------------------------
            if time.time() - start_time < budget_s * 0.995:
                h_final, c_final = hill_climb(best_h, rng, max_iters=1500, start_step=0.2)
                if c_final < best_c:
                    best_h, best_c, best_n = h_final.copy(), c_final, n

    # ------------------------------------------------------------------
    # Final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_sum_and_bounds(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
