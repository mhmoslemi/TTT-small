# Projection onto box [lower,upper] with exact sum = target
def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1 : many short gradient runs from random starts
    phase1_time = budget_s * 0.55
    while time.time() - start_time < phase1_time:
        # random start (occasionally blend with best)
        if rng.random() < 0.2:
            h = (random_feasible_symmetric(n, rng) + best_h) * 0.5
            h = project_onto_box_sum(h, n / 2.0)
        else:
            h = random_feasible_symmetric(n, rng)

        # a modest number of gradient steps with decaying step size
        step = 0.12
        max_iters = 1800
        for it in range(max_iters):
            if time.time() - start_time > phase1_time:
                break
            h = gradient_step_worst_shift(h, step, rng)
            step *= 0.9998
            if (it + 1) % 400 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                if best_c < 0.3808:
                    break
        # quick hill‑climb on the outcome
        h_opt, c_opt = hill_climb_local(h, rng, max_iters=900, start_step=0.4)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        if best_c < 0.3808:
            break

    # ------------------------------------------------------------------
    # Phase 2 : soft‑max refinement on the current best
    beta_schedule = [80.0, 200.0, 500.0, 1200.0]
    for beta in beta_schedule:
        if time.time() - start_time > budget_s * 0.9:
            break
        try:
            best_h = softmax_optimize(best_h, beta=beta, max_iter=250)
            best_c = compute_c5(best_h)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 3 : optional up‑sampling if time permits
    if time.time() - start_time < budget_s * 0.95 and best_h.size * 2 <= 400:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # up‑sampled gradient descent (shorter schedule)
        step = 0.09
        max_iters = 1500
        for it in range(max_iters):
            if time.time() - start_time > budget_s * 0.97:
                break
            h_up = gradient_step_worst_shift(h_up, step, rng)
            step *= 0.9997
            if (it + 1) % 300 == 0:
                c = compute_c5(h_up)
                if c < best_c:
                    best_c = c
                    best_h = h_up.copy()
        # final local hill‑climb on up‑sampled version
        h_up_opt, c_up_opt = hill_climb_local(h_up, rng, max_iters=1200, start_step=0.35)
        if c_up_opt < best_c:
            best_h = h_up_opt.copy()
            best_c = c_up_opt

    # ------------------------------------------------------------------
    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
