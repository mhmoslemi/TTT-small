def _gd_optimize(h0, n, time_budget):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    alpha = 5.0
    lr = 0.5
    beta = 0.9
    v = np.zeros_like(h)
    start = time.time()
    iter = 0
    while time.time() - start < time_budget and iter < 2000:
        grad = _smooth_grad(h, alpha)
        v = beta * v - lr * grad
        h = h + v
        h = _project_to_feasible(h, n)
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # adapt parameters
        if iter % 100 == 0 and iter>0:
            alpha = min(30.0, alpha * 1.2)
            lr = max(0.01, lr * 0.8)
        iter += 1
    return best_h, best_c5

import time
import numpy as np

# ---------- basic utilities ----------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 300) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (deterministic)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 evaluation used by the harness."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- smooth max and its gradient ----------

def _smooth_weights(h: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights approximating the max over all shifts."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    z = alpha * corr
    m = np.max(z)
    expz = np.exp(z - m)
    return expz / expz.sum()


def _smooth_grad(h: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of (1/alpha) log‑sum‑exp(alpha * corr) w.r.t. h.
    """
    n = h.size
    g = 1.0 - h                     # length n
    w = _smooth_weights(h, alpha)  # length 2n-1
    g_rev = g[::-1]                 # reverse for convolution
    conv_full = np.convolve(w, g_rev, mode="full")
    offset = n - 1
    grad = conv_full[offset:offset + n]
    return grad


# ---------- projected gradient descent on the smooth surrogate ----------

def _gd_optimize(h0: np.ndarray, n: int, time_budget: float) -> tuple:
    """
    Run a short projected gradient descent on the smooth surrogate.
    Returns (best_h, best_c5) seen during the run.
    The routine respects the supplied time_budget (seconds).
    """
    start = time.time()
    h = _project_to_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    # hyper‑parameters (hand‑tuned)
    alpha = 8.0          # smoothness
    lr = 0.7             # learning rate
    beta = 0.85          # momentum
    v = np.zeros_like(h)

    it = 0
    while True:
        if time.time() - start > time_budget:
            break
        if it >= 1500:  # safe upper bound
            break

        grad = _smooth_grad(h, alpha)
        v = beta * v - lr * grad
        h = h + v
        h = _project_to_feasible(h, n)

        # occasional sharpening of the surrogate
        if it % 300 == 0 and it > 0:
            alpha = min(30.0, alpha * 1.3)
            lr = max(0.02, lr * 0.8)

        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        it += 1

    # --- binary rounding refinement (often beneficial) ---
    if n % 2 == 0:
        bin_h = np.zeros_like(best_h)
        idx = np.argpartition(-best_h, n // 2)[: n // 2]
        bin_h[idx] = 1.0
        c5_bin = _c5_from_h(bin_h)
        if c5_bin < best_c5:
            best_h = bin_h
            best_c5 = c5_bin

    return best_h, best_c5
