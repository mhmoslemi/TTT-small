import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# Feasibility projection (water-filling)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

# C5 evaluation (fast via FFT) and slow fallback
# ----------------------------------------------------------------------
def _c5_from_h_fast(h: np.ndarray) -> float:
    """Fast C5 using FFT based linear correlation."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * G, N)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def _c5_from_h(h: np.ndarray) -> float:
    """Accurate (but slower) C5 using numpy correlate."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# SLSQP refinement (same as the previous solution)
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = minimize(_objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


# ----------------------------------------------------------------------
# Upscaling (linear interpolation)
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Differential Evolution driver (global search)
# ----------------------------------------------------------------------
def _de_optimize(n: int, seed: int, maxiter: int = 150, popsize: int = 12) -> tuple:
    bounds = [(0.0, 1.0)] * n

    def obj(y):
        h = _project_to_feasible(y, n)
        return _c5_from_h_fast(h)

    result = differential_evolution(
        obj,
        bounds,
        strategy="best1bin",
        maxiter=maxiter,
        popsize=popsize,
        seed=seed,
        polish=False,
        disp=False,
    )
    h = _project_to_feasible(result.x, n)
    c5 = _c5_from_h_fast(h)
    return h, c5


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initial construction (if supplied)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except NameError:
        n = 256
        init_h = np.zeros(n, dtype=float)
        init_h[: n // 2] = 1.0
        np.random.shuffle(init_h)

    init_h = _project_to_feasible(init_h, n)
    best_h = init_h.copy()
    best_c5 = _c5_from_h(best_h)

    # ------------------------------------------------------------------
    # Stage 1: global search at current resolution
    # ------------------------------------------------------------------
    if time_left() > 20:
        # modest DE settings to stay well within budget
        h_de, c5_de = _de_optimize(
            n, seed=seed + 1, maxiter=150, popsize=12
        )
        if c5_de < best_c5:
            best_h, best_c5 = h_de, c5_de

    # ------------------------------------------------------------------
    # Stage 2: progressive upscaling and additional DE passes
    # ------------------------------------------------------------------
    while time_left() > 60 and n * 2 <= 2048:
        n2 = n * 2
        # upscale current best as a warm start
        up_h = _upscale_h(best_h, n2)
        up_h = _project_to_feasible(up_h, n2)

        # allocate a few DE iterations (fewer for larger n)
        maxiter = max(60, int(180 * 256 / n2))
        h_de, c5_de = _de_optimize(
            n2, seed=seed + n2, maxiter=maxiter, popsize=10
        )
        if c5_de < best_c5:
            best_h, best_c5 = h_de, c5_de
            n = n2  # adopt the finer grid
        else:
            # even if not better we still move to finer grid for next loop
            n = n2
        # continue loop – will stop after one or two upscales given time
        if n >= 1024:
            break

    # ------------------------------------------------------------------
    # Final local polish (SLSQP) if time permits
    # ------------------------------------------------------------------
    if time_left() > 10:
        h_opt, c5_opt = _solve_h(best_h, max_iter=800)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # ------------------------------------------------------------------
    # Final feasibility check and return
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)
    return best_h.tolist(), float(best_c5), int(best_h.size)
