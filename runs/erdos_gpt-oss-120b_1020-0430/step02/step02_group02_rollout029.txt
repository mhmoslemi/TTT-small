def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water-filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx for discrete samples."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Simple pair‑exchange hill climbing (small epsilon moves)."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Binary simulated annealing
# ----------------------------------------------------------------------
def _binary_sa(start_h: np.ndarray, time_limit: float) -> tuple:
    """
    Simulated annealing on binary vectors maintaining exactly n/2 ones.
    Returns the best binary vector and its C5.
    """
    n = start_h.size
    cur_h = start_h.astype(float)  # ensure float for correlation
    best_h = cur_h.copy()
    cur_c = best_c = _c5_from_h(cur_h)

    # maintain Python lists of indices
    ones = list(np.flatnonzero(cur_h > 0.5))
    zeros = list(np.flatnonzero(cur_h < 0.5))

    # schedule parameters (linear cooling)
    start_T = max(best_c * 0.5, 1e-4)
    end_T = best_c * 1e-4 + 1e-6

    # estimate iteration budget from remaining time
    est_iter = max(1, int((time_limit - time.time()) / 0.0015))
    # cap to a reasonable max
    est_iter = min(est_iter, 200_000)

    for it in range(est_iter):
        if time.time() > time_limit:
            break
        T = start_T + (end_T - start_T) * (it / est_iter)

        # pick random indices to swap
        i_pos = np.random.randint(len(ones))
        j_pos = np.random.randint(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]

        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = _c5_from_h(cur_h)
        delta = new_c - cur_c

        accept = delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12))
        if accept:
            cur_c = new_c
            # update index lists
            ones[i_pos] = j
            zeros[j_pos] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_c


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load any provided initial construction
    # ------------------------------------------------------------------
    try:
        GLOBAL_INITIAL_H = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL_H = None

    # ------------------------------------------------------------------
    # Candidate grid sizes (prefer multiples of any provided size)
    # ------------------------------------------------------------------
    n_candidates = []
    if GLOBAL_INITIAL_H is not None:
        n_candidates.append(GLOBAL_INITIAL_H.size)
    # reasonable range
    base_sizes = [80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256, 288, 320]
    n_candidates.extend([s for s in base_sizes if s <= 512])
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Main loop over grid sizes
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 15.0:
            break

        # ----- Prepare initial binary vector ---------------------------------
        if GLOBAL_INITIAL_H is not None:
            base_h = _upscale_h(GLOBAL_INITIAL_H, n)
            # threshold then adjust to exact half‑ones
            binary_h = (base_h >= 0.5).astype(float)
            cur_sum = binary_h.sum()
            target = n // 2
            diff = int(target - cur_sum)
            if diff > 0:
                zero_idx = np.where(binary_h == 0)[0]
                # choose zeros with largest base_h values
                order = np.argsort(-base_h[zero_idx])
                add_idx = zero_idx[order[:diff]]
                binary_h[add_idx] = 1.0
            elif diff < 0:
                one_idx = np.where(binary_h == 1)[0]
                order = np.argsort(base_h[one_idx])
                rem_idx = one_idx[order[: -diff]]
                binary_h[rem_idx] = 0.0
        else:
            binary_h = np.zeros(n)
            binary_h[: n // 2] = 1.0
            np.random.shuffle(binary_h)

        # ----- Simulated annealing on binary space ---------------------------
        sa_deadline = time.time() + max(1.0, 0.6 * time_left())
        bin_h, bin_c5 = _binary_sa(binary_h, sa_deadline)

        # ----- Local SLSQP refinement ----------------------------------------
        refined_h, refined_c5 = _solve_h(bin_h, max_iter=300)

        # ----- Optional hill‑climb refinement --------------------------------
        if time_left() > 5.0:
            refined_h, refined_c5 = _hill_climb(
                refined_h,
                max_steps=2000,
                time_limit=time.time() + time_left() - 0.5,
            )

        # ----- Record best ---------------------------------------------------
        if refined_c5 < best_c5:
            best_c5, best_h, best_n = refined_c5, refined_h, n

        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
