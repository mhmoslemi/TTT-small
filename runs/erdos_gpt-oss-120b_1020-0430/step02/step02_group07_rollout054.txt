def continuous_move(h, rng, scale=0.1):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    max_transfer = min(h[i], 1 - h[j])
    if max_transfer <= 0:
        return h
    eps = (rng.random() - 0.5) * 2 * max_transfer * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def continuous_move(h, rng, scale=0.2):
    """Transfer a random amount from a 1‑position to a 0‑position (preserves sum)."""
    n = h.size
    i = rng.integers(n)
    while h[i] < 1e-12:               # need source >0
        i = rng.integers(n)
    j = rng.integers(n)
    while h[j] > 1 - 1e-12 or j == i:  # need target <1
        j = rng.integers(n)
    max_eps = min(h[i], 1.0 - h[j])
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing_bin_fixed(h0, rng, max_iters=5000):
    """Binary SA with a fixed number of iterations (swap 1 ↔ 0)."""
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    T0 = cur_c * 0.3 + 0.01
    for it in range(max_iters):
        # pick a random 1 and a random 0
        i = rng.integers(n)
        while cur_h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] > 0.5 or j == i:
            j = rng.integers(n)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        T = T0 * (1.0 - it / max_iters)
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def simulated_annealing_continuous(h0, rng, end_time):
    """Continuous SA (mass transfer) running until a given wall‑time."""
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T0 = cur_c * 0.3 + 0.01
    stage_start = time.time()
    while time.time() < end_time:
        new_h = continuous_move(cur_h, rng, scale=0.3)
        new_c = compute_c5(new_h)
        elapsed = time.time() - stage_start
        total = max(end_time - stage_start, 1e-12)
        T = T0 * (1.0 - elapsed / total)
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def hill_climb_bin(h, rng, max_iters=5000):
    """Deterministic hill‑climb on binary vectors (swap only if improves)."""
    cur_h = h.copy()

def hill_climb_continuous(h, rng, max_iters=5000):

def enforce_sum(h, target, rng):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)
