import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Projection onto the feasible polytope (box + sum = n/2)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # simple water‑filling loop
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        if not np.any(mask):
            break
        delta = diff / mask.sum()
        h[mask] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (rare)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))

        {"type": "eq", "fun": _eq_constraint},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,

    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)

    neigh_i = np.zeros_like(L, dtype=float)
    neigh_i[mask] = h[idx[mask]]
    idx = i - L
    mask = (idx >= 0) & (idx < n)
    neigh_i[mask] += h[idx[mask]]

    # neighbor contributions for j
    idx = j + L
    mask = (idx >= 0) & (idx < n)
    neigh_j = np.zeros_like(L, dtype=float)
    neigh_j[mask] = h[idx[mask]]
    idx = j - L
    mask = (idx >= 0) & (idx < n)
    neigh_j[mask] += h[idx[mask]]

    delta_A = di * neigh_i + dj * neigh_j

    # quadratic corrections
    k0 = n - 1
    delta_A[k0] += di * di + dj * dj
    k1 = (j - i) + (n - 1)
    k2 = (i - j) + (n - 1)
    delta_A[k1] += di * dj
    delta_A[k2] += di * dj

    A += delta_A
    h[i] -= eps
    h[j] += eps


# ----------------------------------------------------------------------
# Greedy maximisation of the minimal autocorrelation
# ----------------------------------------------------------------------
def _greedy_optimize(h_start, time_limit):
    h = h_start.copy()
    n = h.size
    A = _autocorr_full(h)
    best_min = A.min()
    best_h = h.copy()
    L = np.arange(-n + 1, n, dtype=int)

    while time.time() < time_limit:
        # worst lag
        k_min = int(np.argmin(A))
        lag = k_min - (n - 1)

        # neighbour sum for the worst lag (vectorised)
        idx_plus = np.arange(n) + lag
        idx_minus = np.arange(n) - lag
        neigh = np.zeros(n, dtype=float)
        mask = (idx_plus >= 0) & (idx_plus < n)
        neigh[mask] += h[idx_plus[mask]]
        mask = (idx_minus >= 0) & (idx_minus < n)
        neigh[mask] += h[idx_minus[mask]]

        # source / target candidates
        src_cand = np.where(h > 1e-12)[0]
        tgt_cand = np.where(h < 1 - 1e-12)[0]
        if src_cand.size == 0 or tgt_cand.size == 0:
            break

        src = src_cand[np.argmin(neigh[src_cand])]
        tgt = tgt_cand[np.argmax(neigh[tgt_cand])]

        # fallback to random pair if gradient offers no gain
        if neigh[src] >= neigh[tgt]:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            src, tgt = i, j

        max_eps = min(h[src], 1.0 - h[tgt])
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.9 + 1e-9

        # backup
        A_old = A.copy()
        h_old = h.copy()

        _move_update(h, A, src, tgt, eps, L)

        new_min = A.min()
        if new_min > best_min:
            best_min = new_min
            best_h = h.copy()
        else:
            # undo move
            h[:] = h_old
            A[:] = A_old

    return best_h, best_min


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # grid sizes to explore (even numbers, modest size)
    grid_sizes = [80, 96, 112, 128, 160, 192, 224, 256]

    for n in grid_sizes:
        if time_left() < 5.0:
            break

        # number of independent restarts for this grid
        restarts = 6
        for r in range(restarts):
            if time_left() < 2.0:
                break

            # start vector
            if r == 0:
                h0 = _initial_h(n)
            else:
                # fresh random feasible vector
                h0 = np.random.rand(n)
                h0 = _project_to_feasible(h0, n)

            # greedy optimisation (fills up the remaining time for this phase)
            phase_end = start_time + budget_s - 0.5
            h_opt, min_A = _greedy_optimize(h0, phase_end)

            # compute C5 estimate from min_A
            c5_est = 1.0 - 2.0 * min_A / n

            # optional SLSQP polishing if time permits
            if time_left() > 1.0:
                h_ref, c5_ref = _solve_h(h_opt, max_iter=600)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    best_n = n
            else:
                if c5_est < best_c5:
                    best_c5 = c5_est
                    best_h = h_opt.copy()
                    best_n = n

            if best_c5 < 0.3808:
                return best_h.tolist(), float(best_c5), int(best_n)

    # fallback – use whatever we have or a simple construction
    if best_h is None:
        try:
            fallback = np.array(initial_h_values, dtype=float)
        except NameError:
            fallback = np.full(128, 0.5)
        n = fallback.size
        best_h = _project_to_feasible(fallback, n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
