    for it in range(max_iters):

        ind = corr.argmax()
        k = ind - (n - 1)
        # compute start;
        start = max(0, -k)
        end = n - max(0, k)
        if start >= end:
            break
        i_range = np.arange(start, end)
        i_shift = i_range + k
        # indices where h[i]==1 and h[i_shift]==0
        mask1 = (h[i_range] > 0.5) & (h[i_shift] < 0.5)
        # mask where h[i]==0 and h[i_shift]==1
        mask2 = (h[i_range] < 0.5) & (h[i_shift] > 0.5)

def local_search_swaps(h, rng, max_iters=2000):
    n = len(h)
    best = h.copy()
    best_c = compute_c5(best)
    for it in range(max_iters):
        # pick random i (where h[i]==1) and random j (where h[j]==0)
        # to maintain sum
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        c = compute_c5(new_h)
        if c < best_c:
            best = new_h
            best_c = c
    return best, best_c

    # for source candidates:
    src_mask = (h[i_range] > 1e-12) & ((1 - h[i_shift]) > 1e-12)
    src_idx = i_range[src_mask]
    # for sink candidates:

    sink_idx = i_range[sink_mask]

    j = rng.choice(sink_idx)

    eps = max_eps * rng.random() * eps_scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine base size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 360
    # Helper to compute max correlation bound
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / h.size)
    # Simple deterministic binary reduction
    def greedy_binary_reduce(h, max_iters=2000):
        h = h.copy()
        n = h.size
        for _ in range(max_iters):
            corr = np.correlate(h, 1 - h, mode='full')
            cmax = corr.max()
            if cmax <= 0:
                break
            ind = corr.argmax()
            k = ind - (n - 1)
            start_idx = max(0, -k)
            end_idx = n - max(0, k)
            if start_idx >= end_idx:
                break
            i_range = np.arange(start_idx, end_idx)
            i_shift = i_range + k
            src = (h[i_range] == 1) & (h[i_shift] == 0)
            sink = (h[i_range] == 0) & (h[i_shift] == 1)
            src_idx = i_range[src]
            sink_idx = i_range[sink]
            if src_idx.size == 0 or sink_idx.size == 0:
                break
            i_swap = src_idx[0]
            j_swap = sink_idx[0]
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]  # swap 1 and 0
        return h, compute_c5(h)
    # Local random swap
    def local_swaps(h, max_iters=2000):
        h = h.copy()
        n = h.size
        best_c = compute_c5(h)
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        for _ in range(max_iters):
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            new_h = h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0
            c = compute_c5(new_h)
            if c < best_c:
                h = new_h
                best_c = c
                # update indices
                ones = np.flatnonzero(h == 1)
                zeros = np.flatnonzero(h == 0)
        return h, best_c
    # Random binary initial vector
    def random_binary(n):
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        return h
    # Function to upsample by integer factor using linear interpolation
    def upsample_h(h, factor):
        n = h.size
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
        h_new = np.interp(x_new, x_old, h)
        # Clip and enforce sum exactly n*factor/2
        h_new = np.clip(h_new, 0.0, 1.0)
        target = (n * factor) / 2.0
        # Adjust sum via simple random addition/subtraction
        diff = target - h_new.sum()
        for _ in range(10000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, h_new.size)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    # Initialise best solution
    best_h = None
    best_c = float('inf')
    # If parent exists, convert to binary and improve
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        bin_init = (init > 0.5).astype(float)
        # ensure exact half ones
        cur_sum = bin_init.sum()
        need = n // 2 - cur_sum
        if need > 0:
            zeros = np.flatnonzero(bin_init == 0)
            idx = rng.choice(zeros, int(need), replace=False)
            bin_init[idx] = 1.0
        elif need < 0:
            ones = np.flatnonzero(bin_init == 1)
            idx = rng.choice(ones, -int(need), replace=False)
            bin_init[idx] = 0.0
        bin_init = np.clip(bin_init, 0.0, 1.0)
        bin_init, c_init = greedy_binary_reduce(bin_init, max_iters=2000)
        if c_init < best_c:
            best_h, best_c = bin_init, c_init
    # Random binary restarts
    while time.time() - start < 0.5 * budget_s:
        h = random_binary(n)
        h, c = greedy_binary_reduce(h, max_iters=2000)
        if c < best_c:
            best_h, best_c = h, c
    # Local swaps on best after time
    if best_h is not None and time.time() - start < 0.7 * budget_s:
        best_h, best_c = local_swaps(best_h, max_iters=3000)
    # Upsample and refine if time permits
    if time.time() - start < 0.9 * budget_s:
        factor = 2
        h_up = upsample_h(best_h, factor)
        n_up = h_up.size
        # binary threshold and adjust
        bin_up = (h_up > 0.5).astype(float)
        need = n_up // 2 - bin_up.sum()
        if need > 0:
            zeros = np.flatnonzero(bin_up == 0)
            idx = rng.choice(zeros, int(need), replace=False)
            bin_up[idx] = 1.0
        elif need < 0:
            ones = np.flatnonzero(bin_up == 1)
            idx = rng.choice(ones, -int(need), replace=False)
            bin_up[idx] = 0.0
        bin_up = np.clip(bin_up, 0.0, 1.0)
        bin_up, c_up = greedy_binary_reduce(bin_up, max_iters=4000)
        if c_up < best_c:
            best_h, best_c = bin_up, c_up
        # final local swaps on upsampled
        if time.time() - start < budget_s:
            best_h, best_c = local_swaps(best_h, max_iters=4000)
    # Ensure feasibility and compute final bound
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact sum
    target = best_h.size / 2.0
    # adjust sum using simple operations
    cur_sum = best_h.sum()
    diff = target - cur_sum
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, best_h.size)
        if diff > 0:
            add = min(diff, 1.0 - best_h[i])
        else:
            add = max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def hill_climb_targeted(h, max_iters=2000):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur)
    eps_scale = 0.5
    for it in range(max_iters):
        # produce swap aimed at worst shift
        corr = np.correlate(h_cur, 1 - h_cur, mode='full')
        ind = np.argmax(corr)
        k = ind - (h_cur.size - 1)
        start_i = max(0, -k)
        end_i = h_cur.size - max(0, k)
        if start_i >= end_i:
            # fallback random perturb
            i = rng.integers(0, h_cur.size)
            j = rng.integers(0, h_cur.size)
            if i == j:
                continue
            max_eps = min(h_cur[i], 1 - h_cur[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * eps_scale
            new_h = h_cur.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            i_range = np.arange(start_i, end_i)
            i_shift = i_range + k
            src = (h_cur[i_range] > 1e-12) & ((1 - h_cur[i_shift]) > 1e-12)
            sink = (h_cur[i_range] < 1 - 1e-12) & ((1 - h_cur[i_shift]) < 1e-12)
            src_idx = i_range[src]
            sink_idx = i_range[sink]
            if src_idx.size == 0 or sink_idx.size == 0:
                # fallback random perturb
                i = rng.integers(0, h_cur.size)
                j = rng.integers(0, h_cur.size)
                if i == j:
                    continue
                max_eps = min(h_cur[i], 1 - h_cur[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * eps_scale
                new_h = h_cur.copy()
                new_h[i] -= eps
                new_h[j] += eps
            else:
                i_src = rng.choice(src_idx)
                i_sink = rng.choice(sink_idx)
                max_eps = min(h_cur[i_src], 1 - h_cur[i_sink])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * eps_scale
                new_h = h_cur.copy()
                new_h[i_src] -= eps
                new_h[i_sink] += eps
        # compute c
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h_cur = new_h
        # gradually reduce eps_scale
        if (it+1) % (max_iters // 10 + 1) == 0:
            eps_scale = max(eps_scale * 0.7, 0.02)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Even n: vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample_h(h, factor, rng):
    """Linear up‑sampling by integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target = (n * factor) / 2.0
    return enforce_sum(h_up, target, rng)


# ---------- binary greedy reduction ----------
def greedy_binary_reduce(h, max_iters=2000):
    """Swap a 1 with a 0 that reduces the current worst shift."""
    h = h.copy()
    n = h.size
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        cmax = np.max(corr)
        if cmax <= 0:
            break
        ind = np.argmax(corr)
        k = ind - (n - 1)
        start = max(0, -k)
        end = n - max(0, k)
        if start >= end:
            break
        i_range = np.arange(start, end)
        i_shift = i_range + k
        src = (h[i_range] == 1) & (h[i_shift] == 0)
        sink = (h[i_range] == 0) & (h[i_shift] == 1)
        src_idx = i_range[src]
        sink_idx = i_range[sink]
        if src_idx.size == 0 or sink_idx.size == 0:
            break
        i_swap = src_idx[0]
        j_swap = sink_idx[0]
        h[i_swap], h[j_swap] = h[j_swap], h[i_swap]  # swap 1 ↔ 0
    return h, compute_c5(h)


def local_swaps(h, rng, max_iters=2000):
    """Random 1↔0 swaps that improve the bound."""
    h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        c = compute_c5(new_h)
        if c < best_c:
            h = new_h
            best_c = c
    return h, best_c

    n = h_cur.size

            sink = (h_cur[i_range] < 1 - 1e-12) & ((1 - h_cur[i_shift]) < 1e-12)

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
