def project_onto_bounds_sum(x, target):
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # water-filling
    # Use while loop
    diff = target - np.sum(x)
    # tolerance
    eps = 1e-12
    if abs(diff) < eps:
        return x
    while abs(diff) > 1e-12:
        # indexes not at bounds
        not_fixed = (x > 0.0+1e-12) & (x < 1.0-1e-12)
        count = np.count_nonzero(not_fixed)
        if count == 0:
            # nothing to adjust; break (should not happen because diff !=0)
            break
        delta = diff / count
        # add delta to those entries
        x[not_fixed] += delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
        diff = target - np.sum(x)
    return x

def gradient_of_max_shift(h, k):
    n = h.size
    g = np.zeros(n)
    idx = np.arange(n)
    low = max(0, -k)
    high = min(n-1, n-1 - k)
    mask = (idx >= low) & (idx <= high)
    g[mask] = 1.0
    if k >= 0:
        # subtract forward shift
        g[:n-k] -= h[k:]
        # subtract backward shift
        g[k:] -= h[:n-k]
    else:
        kk = -k
        # forward (for negative shift, forward is shift left)
        g[kk:] -= h[:n-kk]
        # backward
        g[:n-kk] -= h[kk:]
    return g

def project_onto_bounds_sum(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Simple iterative water-filling
    eps = 1e-12
    diff = target - x.sum()
    # We will iteratively adjust using vector of free indices.
    # Use while loop with max iterations maybe 1000.
    for _ in range(1000):
        if abs(diff) <= eps:
            break
        # free indices not at bounds
        free = (x > 0 + eps) & (x < 1 - eps)
        count = free.sum()
        if count == 0:
            # adjust by adding diff to random index that is not at bound sign
            # Choose random index i and clip after addition.
            i = np.random.randint(len(x))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
            diff = target - x.sum()
            continue
        delta = diff / count
        x[free] += delta
        # Clip after addition
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return x

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_onto_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto 0≤x≤1 and Σx = target by water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:  # should not happen, but break to avoid division by zero
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return x


def random_feasible(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random vector with 0≤h≤1 and Σh = n/2."""
    x = rng.random(n)
    return project_onto_bounds_sum(x, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def gradient_of_max_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the worst‑shift term for current h."""
    n = h.size
    g = np.zeros_like(h)
    idx = np.arange(n)
    low = max(0, -k)
    high = min(n - 1, n - 1 - k)
    mask = (idx >= low) & (idx <= high)
    g[mask] = 1.0
    if k >= 0:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    else:
        kk = -k
        g[kk:] -= h[: n - kk]
        g[: n - kk] -= h[kk:]
    return g

# Local optimisation primitives
# ----------------------------------------------------------------------
def hill_climb(h0: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.3):
    """Deterministic hill‑climb (only downhill moves)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def simulated_annealing(h0: np.ndarray, max_iters: int,
                        rng: np.random.Generator, time_deadline: float):
    """Metropolis SA with mass‑transfer moves."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0 = 0.01
    scale0 = 0.5
    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        T = T0 * (0.995 ** it)
        scale = scale0 * (0.999 ** it)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
    return best_h, best_c


def projected_gradient_descent(h0: np.ndarray, max_iters: int,
                              rng: np.random.Generator, time_deadline: float):
    """Gradient descent on the worst‑shift term with projection onto the feasible set."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    target = n / 2.0
    step0 = 0.5
    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr) - (n - 1)
        g = gradient_of_max_shift(h, k)
        alpha = step0 / np.sqrt(it + 1.0)
        new_h = h - alpha * g
        new_h = project_onto_bounds_sum(new_h, target)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < compute_c5(h):
            h = new_h
    return best_h, best_c

def optimise_one_n(n: int, rng: np.random.Generator,
                  deadline: float, init_h: np.ndarray | None = None):
    best_h = None
    best_c = np.inf
    attempts = 0
    max_attempts = 5
    while attempts < max_attempts and time.time() < deadline:
        attempts += 1
        if init_h is not None and attempts == 1 and init_h.size == n:
            h_start = init_h.copy()
        else:
            h_start = random_feasible(rng, n)

        # -------- simulated annealing ----------
        rem = deadline - time.time()
        if rem <= 0:
            break
        sa_dead = time.time() + rem * 0.4
        h_sa, c_sa = simulated_annealing(h_start, max_iters=15000,
                                         rng=rng, time_deadline=sa_dead)

        # -------- projected gradient descent ----------
        rem = deadline - time.time()
        if rem <= 0:
            break
        gd_dead = time.time() + rem * 0.35
        h_gd, c_gd = projected_gradient_descent(h_sa, max_iters=2000,
                                                rng=rng, time_deadline=gd_dead)

        # -------- final hill‑climb ----------
        rem = deadline - time.time()
        if rem <= 0:
            break
        h_hc, c_hc = hill_climb(h_gd, rng, max_iters=2000, start_step=0.3)

        for h_cand, c_cand in ((h_sa, c_sa), (h_gd, c_gd), (h_hc, c_hc)):
            if c_cand < best_c:
                best_c, best_h = c_cand, h_cand.copy()
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise from provided parent if any
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = h_parent.size
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_onto_bounds_sum(h_parent, n_parent / 2.0)
        best_h = h_parent.copy()
        best_c = compute_c5(best_h)
        best_n = n_parent
    else:
        # fallback uniform 0.5 (feasible for any even n)
        best_n = 120
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Candidate discretisations (keep them modest)
    # ------------------------------------------------------------------
    candidates = [96, 108, 120, 144, 168, 192, 216]
    if best_n not in candidates:
        candidates = [best_n] + candidates
    # preserve order, remove duplicates
    seen = set()
    cand_ns = [x for x in candidates if not (x in seen or seen.add(x))]

    # ------------------------------------------------------------------
    # Optimise each size within time budget
    # ------------------------------------------------------------------
    for idx, n in enumerate(cand_ns):
        if time.time() - start > budget_s * 0.95:
            break
        # allocate a share of the remaining time
        remaining = budget_s - (time.time() - start)
        remaining_ns = len(cand_ns) - idx
        deadline = time.time() + remaining * 0.9 / max(remaining_ns, 1)

        init_for_n = best_h if best_n == n else None
        h_opt, c_opt = optimise_one_n(n, rng, deadline, init_h=init_for_n)

        if c_opt < best_c:
            best_c, best_h, best_n = c_opt, h_opt.copy(), n

    # ------------------------------------------------------------------
    # Optional up‑sampling refinement (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < budget_s * 0.98:
        if best_n * 2 <= 240:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            deadline = time.time() + (budget_s - (time.time() - start)) * 0.9
            h_up_opt, c_up_opt = optimise_one_n(n_up, rng, deadline, init_h=h_up)
            if c_up_opt < best_c:
                best_c, best_h, best_n = c_up_opt, h_up_opt.copy(), n_up

    # ------------------------------------------------------------------
    # Final feasibility enforcement & return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
