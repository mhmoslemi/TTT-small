def simulated_annealing(start_h, rng, time_limit):
    best_h = start_h.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    T0 = best_c * 0.2 + 1e-6
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # propose
        new_h = perturb_transfer(cur_h, rng, sigma=0.1) # sigma maybe variable?
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T0 * 0.99**it, 1e-8)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        it += 1
    return best_h, best_c

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh == target while 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(20000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 0:
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_transfer(h: np.ndarray, rng: np.random.Generator, sigma: float = 0.1) -> np.ndarray:
    """Transfer ε from i to j (preserves sum). ε drawn from N(0,σ)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    eps = rng.normal(0.0, sigma)
    max_eps = min(h[i], 1.0 - h[j])
    min_eps = -min(h[j], 1.0 - h[i])
    if eps > max_eps:
        eps = max_eps
    elif eps < min_eps:
        eps = min_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def simulated_annealing(start_h: np.ndarray, rng: np.random.Generator, time_limit: float):
    """Simple SA that respects sum and bounds."""
    best_h = start_h.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    start = time.time()
    it = 0
    T0 = best_c * 0.2 + 1e-9
    while time.time() - start < time_limit:
        sigma = 0.2 * (0.995 ** it)
        new_h = perturb_transfer(cur_h, rng, sigma)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        T = T0 * (0.9995 ** it)
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        it += 1
    return best_h, best_c


def hill_climb_exact(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2500, step_scale: float = 0.1):
    """Deterministic hill‑climb accepting only improvements (exact C₅)."""
    h = h_start.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay_it = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_transfer(h, rng, sigma=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, h = c, new_h.copy()
        if (it + 1) % decay_it == 0:
            scale = max(scale * 0.7, 1e-4)
    return h, best_c


# ---------- surrogate + SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, safe
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 400):
    """Refine via SLSQP on smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
