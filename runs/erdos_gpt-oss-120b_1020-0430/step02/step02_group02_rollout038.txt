import time

# ----------------------------------------------------------------------
# Helper utilities

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a discretised step function (integral is already enforced)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    """SLSQP refinement of a feasible h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Small epsilon exchanges between two coordinates, accepting improvements."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]          # amount we can take from i
        max_up = 1.0 - best_h[j]      # amount we can give to j
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


def _binary_hc_targeted(x: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Targeted hill‑climb on a binary vector – reduce the current worst shift."""
    n = x.size
    best = x.copy()
    best_c = _c5_from_h(best)

    def creates_contrib(j, shift, h):
        idx = j + shift
        if 0 <= idx < n:
            return h[idx] == 0
        return False

    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        corr = np.correlate(best, 1 - best, mode="full")
        max_idx = np.argmax(corr)
        max_c = corr[max_idx]
        if max_c < best_c:
            best_c = max_c
        shift = max_idx - (n - 1)

        # indices that contribute to the maximal shift
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        i_range = np.arange(i_start, i_end)
        contrib_mask = (best[i_range] == 1) & (best[i_range + shift] == 0)
        contrib_idx = i_range[contrib_mask]
        if contrib_idx.size == 0:
            break
        i = np.random.choice(contrib_idx)

        zeros = np.where(best == 0)[0]

        candidates = zeros[good]

            j = np.random.choice(zeros)

            j = np.random.choice(candidates)
