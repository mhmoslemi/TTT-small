    else:

    h_opt = _project_to_simplex_box(h_opt, n/2)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

import time
import numpy as np
import scipy.optimize as opt


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x in [0,1]^n : sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _conv_hh_fft(h: np.ndarray) -> np.ndarray:
    """Linear convolution of h with itself using FFT (length 2*len-1)."""
    n = h.size
    size = 2 * n
    H = np.fft.rfft(h, size)
    conv = np.fft.irfft(H * H, size)[: 2 * n - 1]
    return np.maximum(conv, 0.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute the C5 value for a feasible step function h."""
    n = h.size
    sum_h = np.sum(h)
    min_conv = np.min(_conv_hh_fft(h))
    max_corr = sum_h - min_conv
    return max_corr * (2.0 / n)


def _gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of Σ_i h_i h_{i+shift} w.r.t. h for a given |shift| = k."""
    n = h.size
    if k == 0:
        return 2.0 * h
    grad = np.empty_like(h)
    grad.fill(0.0)
    grad[: n - k] += h[k:]
    grad[k:] += h[: n - k]
    return grad


def _gradient_descent(
    h0: np.ndarray, max_iter: int, step0: float, decay: float
) -> tuple[np.ndarray, float]:
    """Projected gradient ascent on min‑convolution (equivalently descent on C5)."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    step = step0
    for _ in range(max_iter):
        conv = _conv_hh_fft(h)
        idx_min = np.argmin(conv)
        shift = idx_min - (n - 1)
        k = abs(shift)
        grad = _gradient_for_shift(h, k)
        h = h + step * grad
        h = _project_to_simplex_box(h, n / 2.0)
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= decay
        if step < 1e-12:
            break
    return best_h, best_c5


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation(k) >= 0 for every shift k."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    sum_h = np.sum(h)
    conv = _conv_hh_fft(h)
    corr = sum_h - conv
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce Σ h = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 400) -> tuple[np.ndarray, float]:
    """Run SLSQP from a feasible start and return refined h and its C5."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    sum_h0 = np.sum(h0)
    min_conv0 = np.min(_conv_hh_fft(h0))
    t0 = sum_h0 - min_conv0
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=constraints, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape h(x)=min(x,2-x) sampled on n points, projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)
