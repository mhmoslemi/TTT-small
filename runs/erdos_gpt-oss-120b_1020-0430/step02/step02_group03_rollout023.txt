import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def compute_c5_x(x):
    h = np.concatenate([x, 1 - x])
    return compute_c5(h)

def smooth_obj(x, beta):
    h = np.concatenate([x, 1 - x])
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize(x0, beta, max_iter=300):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: smooth_obj(x, beta), x0, method='SLSQP',
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    return np.clip(res.x, 0.0, 1.0)

def hill_climb_x(x_start, rng, max_iters=2000, step=0.2):
    x = x_start.copy()
    best_x = x.copy()
    best_c = compute_c5_x(x)
    cur_step = step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, x.size)
        eps = (rng.random() - 0.5) * 2 * cur_step
        new_val = best_x[i] + eps
        if new_val < 0.0:
            new_val = 0.0
        elif new_val > 1.0:
            new_val = 1.0
        if new_val == best_x[i]:
            continue
        new_x = best_x.copy()
        new_x[i] = new_val
        new_c = compute_c5_x(new_x)
        if new_c < best_c:
            best_c, best_x = new_c, new_x
        if (it + 1) % decay_interval == 0:

        arr[:n//2] = 1

        # schedule temperature decreasing
        temperature = max(0.001, 1.0 - it / max_iters)  # linear cooling
        # pick random swap
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)

        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1

        delta = new_c - cur_c
        # accept if better or by probability

            # check update best
            if cur_c < best_c:
                best_c, best = cur_c, cur.copy()
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0
    return best.astype(float), best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities ------------------- #
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for a step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_x(x: np.ndarray) -> float:
    """C₅ for a half‑vector x using the symmetry h = [x, 1‑x]."""
    h = np.concatenate([x, 1.0 - x])
    return compute_c5(h)


def smooth_obj(x: np.ndarray, beta: float) -> float:
    """Smooth log‑sum‑exp surrogate of max_k Σ h(i)(1‑h(i+k))."""
    h = np.concatenate([x, 1.0 - x])
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(x0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """SLSQP refinement of x for a given beta."""
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda v: smooth_obj(v, beta), x0,
                   method="SLSQP", bounds=bounds,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    return np.clip(res.x, 0.0, 1.0)


def hill_climb_x(x_start: np.ndarray, rng: np.random.Generator,
                 max_iters: int = 2000, step: float = 0.2):
    """Hill‑climb on the half‑vector x (modifies a single coordinate)."""
    best_x = x_start.copy()
    best_c = compute_c5_x(best_x)
    cur_step = step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        i = rng.integers(0, best_x.size)
        eps = (rng.random() - 0.5) * 2.0 * cur_step
        val = best_x[i] + eps
        val = 0.0 if val < 0.0 else (1.0 if val > 1.0 else val)
        if val == best_x[i]:
            continue
        new_x = best_x.copy()
        new_x[i] = val
        new_c = compute_c5_x(new_x)
        if new_c < best_c:
            best_c, best_x = new_c, new_x
        if (it + 1) % decay == 0:
            cur_step = max(cur_step * 0.7, 1e-4)
    return best_x, best_c


def perturb_full(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb_full(h_start: np.ndarray, rng: np.random.Generator,
                    max_iters: int = 2000, step: float = 0.3):
    """Hill‑climb on the full vector h (preserves total integral)."""
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    cur_step = step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_full(best_h, rng, scale=cur_step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h
        if (it + 1) % decay == 0:
            cur_step = max(cur_step * 0.7, 1e-4)
    return best_h, best_c


def binary_anneal(rng: np.random.Generator, n: int, time_budget: float):
    """Simulated‑annealing on binary vectors with exactly n/2 ones."""
    half = n // 2
    b = np.zeros(n, dtype=int)
    b[:half] = 1
    rng.shuffle(b)
    best = b.copy()
    best_c = compute_c5(b.astype(float))
    cur = b.copy()
    cur_c = best_c
    start = time.time()
    while time.time() - start < time_budget:
        # linear temperature schedule
        t = max(0.01, 1.0 - (time.time() - start) / time_budget)
        # pick a random 1 and a random 0 to swap
        i = rng.integers(0, n)
        if cur[i] == 1:
            j = rng.integers(0, n)
            while cur[j] == 1:
                j = rng.integers(0, n)
        else:
            j = rng.integers(0, n)
            while cur[j] == 0:
                j = rng.integers(0, n)
        cur[i], cur[j] = cur[j], cur[i]          # perform swap
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / t):
            cur_c = new_c
            if cur_c < best_c:
                best_c, best = cur_c, cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]      # revert swap
        # early exit if already under target
        if best_c <= 0.3808:
            break
    return best.astype(float), best_c
