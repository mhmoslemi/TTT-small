def anneal_binary(h0, rng, time_limit):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = len(h)
    init_temp = 1.0
    temp = init_temp
    decay = 0.9995
    while time.time() < time_limit:
        # propose swap
        i = rng.integers(0, n)
        while h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1:
            j = rng.integers(0, n)
        # swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        temp *= decay
        if temp < 1e-6:
            temp = 1e-6
    return best_h, best_c

def hill_climb_binary(h, rng, max_iters):
    n = len(h)
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    for _ in range(max_iters):
        # pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c
```

    return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / beta

def perturb_continuous(h, rng, scale):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def enforce_sum(x, target, rng, max_iter=1000):
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the SLSQP surrogate (soft‑max of the overlap)
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(x, target, rng, max_iter=2000):
    """Adjust x in‑place so that Σx = target while keeping 0≤x≤1."""
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Binary (0/1) vector with exactly n/2 ones (n must be even)."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, half, replace=False)
    h[idx] = 1.0
    return h


def upsample_linear(h, factor):
    """Linear up‑sampling by an integer factor (keeps the integral)."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    return h_up


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # small corrective pass to enforce the sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_continuous(h, rng, scale):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb on the exact C5 (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_continuous(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def refine(h0, rng, betas, slsqp_iters, hill_iters):
    """Apply a ladder of SLSQP surrogates followed by hill‑climbing."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        h = slsqp_optimize(best_h, beta=beta, max_iter=slsqp_iters)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        h, c = hill_climb_continuous(best_h, rng, max_iters=hill_iters, start_step=0.4)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    max_n = 720                     # upper resolution (still < 1 000)

    # ----------------------------------------------------------------------
    # Initialise base solution
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 60
        h0 = random_binary_vector(n, rng)

    best_h, best_c = refine(
        h0,
        rng,
        betas=(5.0, 20.0, 80.0, 200.0, 500.0, 2000.0),
        slsqp_iters=250,
        hill_iters=2500,
    )

    # ----------------------------------------------------------------------
    # Additional random restarts (quick" refinements only)
    # ----------------------------------------------------------------------
    restarts = 8
    while time.time() - start_time < budget_s * 0.45 and restarts > 0:
        h_start = random_binary_vector(n, rng)
        h_ref, c_ref = refine(
            h_start,
            rng,
            betas=(5.0, 20.0, 80.0, 200.0, 500.0),
            slsqp_iters=200,
            hill_iters=1500,
        )
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref
        restarts -= 1

    # ----------------------------------------------------------------------
    # Upsample ladder (gradually increase resolution)
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.90 and n * 2 <= max_n:
        h_up = upsample_linear(best_h, factor=2)
        n = h_up.size
        h_up = enforce_sum(h_up, n / 2.0, rng)
        h_up, c_up = refine(
            h_up,
            rng,
            betas=(5.0, 20.0, 80.0, 200.0, 500.0, 2000.0, 5000.0),
            slsqp_iters=300,
            hill_iters=3000,
        )
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # ----------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
