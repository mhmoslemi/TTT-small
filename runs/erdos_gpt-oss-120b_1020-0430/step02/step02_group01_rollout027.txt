    n = h.size

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stabilized
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        # move a small amount between two coordinates (preserving sum)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-6)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] = 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # exact sum repair
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    assert n % 2 == 0
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def simulated_annealing_binary(n, rng, time_limit):
    """Binary simulated annealing (swap a 1 with a 0)."""
    cur_h = random_binary(n, rng)
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    T = 0.01
    decay = 0.9995
    min_T = 1e-9
    while time.time() - start < time_limit:
        # propose a swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if cur_h[i] == cur_h[j]:
            continue
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, min_T)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        T = max(T * decay, min_T)
    return best_h, best_c


def targeted_improve(h, rng, max_steps=2000):
    """Greedy swaps that strictly improve the worst shift."""
    n = h.size
    for _ in range(max_steps):
        old_c = compute_c5(h)
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        if k < 0:
            k = -k
        if k == 0 or k >= n:
            break
        # mismatched positions for this shift
        left = h[: n - k]
        right = h[k:]
        miss_one_zero = np.where((left == 1) & (right == 0))[0]
        miss_zero_one = np.where((left == 0) & (right == 1))[0]
        if miss_one_zero.size and miss_zero_one.size:
            i = miss_one_zero[rng.integers(0, miss_one_zero.size)]
            j = miss_zero_one[rng.integers(0, miss_zero_one.size)]
            # perform swap
            h[i], h[j] = h[j], h[i]
            new_c = compute_c5(h)
            if new_c < old_c:
                continue  # keep the improvement
            # revert if not better
            h[i], h[j] = h[j], h[i]
            break
        else:
            # no direct improvement possible for this shift – try random swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            if compute_c5(h) < old_c:
                continue
            h[i], h[j] = h[j], h[i]
            break
    return h
