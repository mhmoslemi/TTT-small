def _subgradient_optimize(h0, max_iter, time_left):
    n = h0.size
    target = n / 2.0
    h = _project_to_simplex_box(h0, target)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    alpha = 0.5
    for t in range(1, max_iter+1):
        if not time_left():
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = corr.argmax()
        shift = max_idx - (n - 1)
        if shift == 0:
            g = 1.0 - 2.0 * h
        elif shift > 0:
            g = np.ones(n)
            g[:n-shift] -= h[shift:]
            g[shift:] -= h[:n-shift]
        else:
            s = -shift
            g = np.ones(n)
            g[s:] -= h[:n-s]
            g[:n-s] -= h[s:]
        step = alpha / np.sqrt(t)
        h_new = h - step * g
        h_new = _project_to_simplex_box(h_new, target)
        c_new = _c5_from_h(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _solve_h(start_h, max_iter=500):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n/2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": lambda x: _ineq_constraint(x)},
        {"type": "eq", "fun": lambda x: _eq_constraint(x)},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                                constraints=constraints, options=opts)
    except Exception:
        res = None
    if res is not None and res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_simplex_box(h_opt, n/2.0)
    return h_opt, _c5_from_h(h_opt)

import time
import numpy as np
from scipy import optimize

# ----- projection onto [0,1]^n intersect sum(x)=target -----------------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# ----- C5 evaluation --------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


# ----- subgradient descent on max‑correlation -------------------------
def _subgradient_optimize(h0: np.ndarray, max_iter: int, time_ok) -> tuple:
    n = h0.size
    target = n / 2.0
    h = _project_to_simplex_box(h0, target)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    alpha = 0.5
    for t in range(1, max_iter + 1):
        if not time_ok():
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = corr.argmax()
        shift = max_idx - (n - 1)
        if shift == 0:
            g = 1.0 - 2.0 * h
        elif shift > 0:
            g = np.ones(n)
            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]
        else:  # shift < 0
            s = -shift
            g = np.ones(n)
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
        step = alpha / np.sqrt(t)
        h = _project_to_simplex_box(h - step * g, target)
        c = _c5_from_h(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


# ----- SLSQP refinement ------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 600):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(_objective, x0, method="SLSQP",
                                bounds=bounds, constraints=cons, options=opts)
        h_opt = res.x[:n] if res.success else h0
    except Exception:
        h_opt = h0
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


# ----- simple constructions --------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

    # initial construction if supplied
