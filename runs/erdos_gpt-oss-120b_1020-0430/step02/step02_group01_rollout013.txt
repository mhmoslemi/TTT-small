def simulated_annealing(h0, rng, max_iter, init_temp, cooling):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    temp = init_temp
    for it in range(max_iter):
        scale = temp / init_temp
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        temp *= cooling
    return best_h, best_c
```

def hill_climb(h_start, rng, max_iters=4000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters//10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a (scaled) amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=5000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def simulated_annealing(h0, rng, max_iter, init_temp, cooling):
    """Simple SA that preserves the sum constraint."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iter):
        scale = temp / init_temp
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / (temp + 1e-15)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        temp = max(temp * cooling, 1e-12)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=15):
    """Gradient steps targeting the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    x0 = np.clip(h0, 0.0, 1.0)

        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:

            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

            # iteration budget depends on remaining time
            time_left = deadline - time.time()
            max_iter_sa = int(min(20000, max(2000, time_left * 30)))
            init_temp = 0.025
            cooling = 0.9996
            h_sa, c_sa = simulated_annealing(
                h, rng, max_iter=max_iter_sa, init_temp=init_temp, cooling=cooling
            )
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n

            # ----- gradient refinement -----------------------------------------
            h = gradient_refine(h_sa, rng, step=0.025, max_iter=12)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ----- SLSQP passes with increasing beta ---------------------------
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                if time.time() > deadline:
                    break
                try:
                    h_opt = slsqp_optimize(h, beta, rng, max_iter=600 if beta < 100 else 900)
                except Exception:
                    continue
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c, best_n = h_opt.copy(), c_opt, n
                # quick gradient polish after each SLSQP
                h_opt = gradient_refine(h_opt, rng, step=0.02, max_iter=6)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c, best_n = h_opt.copy(), c_opt, n

            # ----- final short hill‑climb ------------------------------------
            if time.time() < deadline:
                h_fin, c_fin = hill_climb(best_h, rng, max_iters=2500, start_step=0.3)
                if c_fin < best_c:
                    best_h, best_c, best_n = h_fin.copy(), c_fin, n

    # ----------------------------------------------------------------------
    # Optional up‑sampling of the best solution (if time permits)
    # ----------------------------------------------------------------------
    if best_h is not None and time.time() < deadline:
        if best_n * 2 <= 300:
            # linear interpolation to double resolution
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, best_n * 2, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            target = (best_n * 2) / 2.0
            diff = target - h_up.sum()
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, h_up.size)
                if diff > 0:
                    add = min(diff, 1.0 - h_up[i])
                else:
                    add = max(diff, -h_up[i])
                h_up[i] += add
                diff = target - h_up.sum()
            # refinement at higher resolution
            h_up, c_up = hill_climb(h_up, rng, max_iters=3000, start_step=0.4)
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=8)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, h_up.size

    # ----------------------------------------------------------------------
    # Final long SA on the current best (if time permits)
    # ----------------------------------------------------------------------
    if best_h is not None and time.time() < deadline:
        # allocate remaining time to a final SA run
        remaining_iters = int((deadline - time.time()) * 35)
        if remaining_iters > 0:
            best_h, best_c = simulated_annealing(
                best_h,
                rng,
                max_iter=remaining_iters,
                init_temp=0.02,
                cooling=0.9997,
            )
            # final polish
            best_h = gradient_refine(best_h, rng, step=0.015, max_iter=10)
            best_c = compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Final feasibility enforcement
    # ----------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, best_n)
        if diff > 0:
            add = min(diff, 1.0 - best_h[i])
        else:
            add = max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
