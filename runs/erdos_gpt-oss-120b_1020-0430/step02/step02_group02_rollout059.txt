```
def gradient_wrt_h(h, L):
    n = h.size
    grad = np.ones(n)
    if L >= 0:
        grad[:n-L] -= h[L:]  # subtract h[i+L] for i in 0..n-L-1
        grad[L:] -= h[:n-L]  # subtract h[i-L] for i in L..n-1
    else:
        # negative L: analogous; we can just use absolute
        L_abs = -L
        grad[L_abs:] -= h[:n-L_abs]
        grad[:n-L_abs] -= h[L_abs:]
    # for positions where an index is out of bounds, no subtraction occurs (since we only subtract where defined)
    return grad
```

import numpy as np
import time

def _project_to_feasible(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _c5_from_h(h):
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _subgradient_descent(h, max_iter, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    current = h.copy()
    eta0 = 0.1
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation to find worst lag
        corr = np.correlate(current, 1.0 - current, mode='full')
        max_val = corr.max()
        # find lag indices where max achieved
        lag_inds = np.where(corr == max_val)[0]
        lag_index = np.random.choice(lag_inds)
        L = lag_index - (n - 1)   # shift value
        # gradient of g_L = sum_i h_i - sum_i h_i h_{i+L}
        grad = np.ones(n)
        if L >= 0:
            grad[:n-L] -= current[L:]   # subtract h_{i+L}
            grad[L:] -= current[:n-L]   # subtract h_{i-L}
        else:
            L_abs = -L
            grad[L_abs:] -= current[:n-L_abs]
            grad[:n-L_abs] -= current[L_abs:]
        # step
        eta = eta0 / (1.0 + 1e-3 * it)
        cand = current - eta * grad
        cand = _project_to_feasible(cand, n)
        c_cand = _c5_from_h(cand)
        # accept always for subgradient descent (move may increase)
        current = cand
        if c_cand < best_c:
            best_h, best_c = cand.copy(), c_cand
    return best_h, best_c

    current = start_h.copy()

    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C₅ value for a step function sampled on a uniform grid."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse sample onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _subgradient_descent(h0: np.ndarray, max_iter: int, time_limit: float):
    """Projected sub‑gradient descent on the worst lag."""
    n = h0.size
    best_h, best_c = h0.copy(), _c5_from_h(h0)
    cur = h0.copy()
    eta0 = 0.1
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        lag_idx = np.argmax(corr)
        L = lag_idx - (n - 1)  # shift producing the current maximum
        grad = np.ones(n)
        if L >= 0:
            grad[: n - L] -= cur[L:]
            grad[L:] -= cur[: n - L]
        else:
            a = -L
            grad[a:] -= cur[: n - a]
            grad[: n - a] -= cur[a:]
        eta = eta0 / (1.0 + 1e-3 * it)
        cand = cur - eta * grad
        cand = _project_to_feasible(cand, n)
        c_cand = _c5_from_h(cand)
        cur, _ = cand, c_cand
        if c_cand < best_c:
            best_h, best_c = cand.copy(), c_cand
    return best_h, best_c


def _hill_climb(start_h: np.ndarray, max_steps: int, time_limit: float):
    """Greedy + simulated‑annealing hill climb (epsilon moves and swaps)."""
    n = start_h.size
    cur = start_h.copy()
    cur_c = _c5_from_h(cur)
    best, best_c = cur.copy(), cur_c
    T0 = 0.02
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        if np.random.rand() < 0.7:                     # epsilon transfer
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_down = cur[i]
            max_up = 1.0 - cur[j]
            eps = np.random.rand() * min(max_down, max_up)
            if eps <= 0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps
        else:                                          # swap two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cur[j], cur[i]
        cand_c = _c5_from_h(cand)
        # accept if better, otherwise probabilistically
        if cand_c < cur_c:
            cur, cur_c = cand, cand_c
        else:
            T = T0 * (1.0 - step / max_steps)
            if T > 0 and np.random.rand() < np.exp(-(cand_c - cur_c) / T):
                cur, cur_c = cand, cand_c
        if cand_c < best_c:
            best, best_c = cand.copy(), cand_c
    return best, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Load optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # ------------------------------------------------------------------
    # Grid sizes to explore (favor small, but allow refinement)
    # ------------------------------------------------------------------
    sizes = [80, 96, 112, 128, 160, 192, 224, 256, 320, 384, 512]
    if init_h is not None and init_h.size not in sizes:
        sizes.append(init_h.size)
    sizes = sorted(set(sizes))

    best_h = None
    best_c = float("inf")
    best_n = None
    target = 0.3808

    # ------------------------------------------------------------------
    # Main optimisation loop over grid sizes
    # ------------------------------------------------------------------
    for n in sizes:
        if time_left() < 5.0:
            break

        # base vector for this resolution
        if init_h is not None:
            base_h = _upscale_h(init_h, n) if init_h.size != n else init_h.copy()
        else:
            base_h = np.full(n, 0.5)
        base_h = _project_to_feasible(base_h, n)

        # 1) sub‑gradient descent (quick coarse improvement)
        sg_iters = min(3000, int(time_left() * 10))
        limit_sg = start + budget_s - 0.5
        h_sg, c_sg = _subgradient_descent(base_h, sg_iters, limit_sg)
        if c_sg < best_c:
            best_h, best_c, best_n = h_sg.copy(), c_sg, n

        # 2) intensive hill climbing
        steps = int(min(50000, time_left() * 80))
        limit_hc = start + budget_s - 0.2
        h_hc, c_hc = _hill_climb(h_sg, steps, limit_hc)
        if c_hc < best_c:
            best_h, best_c, best_n = h_hc.copy(), c_hc, n

        # 3) a few perturbation‑restart rounds
        rest = 0
        while time_left() > 5 and rest < 4:
            rest += 1
            sigma = 0.15
            pert = sigma * np.random.randn(n)
            start_h = best_h + pert
            start_h = _project_to_feasible(start_h, n)
            steps2 = int(min(20000, time_left() * 40))
            limit2 = start + budget_s - 0.2
            h2, c2 = _hill_climb(start_h, steps2, limit2)
            if c2 < best_c:
                best_h, best_c, best_n = h2.copy(), c2, n

        if best_c < target:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c), int(best_n)
