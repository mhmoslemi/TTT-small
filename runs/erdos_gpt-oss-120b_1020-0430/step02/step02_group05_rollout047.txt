def _c5_from_h(h):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def _max_corr_and_shifts(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxv = corr.max()
    ks = np.where(corr == maxv)[0] - (h.size - 1)
    return maxv, ks

def _gradient_max_corr(h):
    _, ks = _max_corr_and_shifts(h)
    n = h.size
    g = np.zeros_like(h)
    for k in ks:
        if k >= 0:
            g[:n - k] += 1.0 - h[k:]
            g[k:] -= h[:n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[:n - kk]
            g[:n - kk] -= h[kk:]
    # average over ks
    if ks.size > 1:
        g /= ks.size
    return g

def _gradient_descent(h, max_iters, step0, decay):
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    step = step0
    for it in range(max_iters):
        g = _gradient_max_corr(best_h)

        h_new = best_h - step * g
        # project
        h_new = _project_to_simplex_box(h_new, best_h.size / 2.0)

def _pairwise_sa(h, max_iters, temp0, decay):
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    T = temp0
    for it in range(max_iters):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi = cur_h[i]
        hj = cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if high <= low:
            continue
        delta = np.random.uniform(low, high)
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        # optionally ensure within bounds after rounding errors.
        # Compute C5.
        c5_new = _c5_from_h(new_h)
        if c5_new < cur_c5 or np.random.rand() < np.exp(-(c5_new - cur_c5) / T):
            cur_h = new_h
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = new_h
                best_c5 = c5_new
        T *= decay
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector onto [0,1]^n intersect sum(v)=target."""
    lo = np.min(v) - 1.0  # lower bound for binary search lambda
    hi = np.max(v)        # upper bound
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny correction for residual rounding error
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for discrete h on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for each shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Objective is the auxiliary variable t."""
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Run SLSQP from a feasible start and return refined h and its C5."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Triangular (tent) shape on [0,2] sampled with n points, then projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _max_corr_and_shifts(h: np.ndarray):
    """Return max correlation value and the list of shift indices attaining it."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxv = corr.max()
    ks = np.where(corr == maxv)[0] - (n - 1)  # convert to shift range [-n+1, n-1]
    return maxv, ks


def _gradient_max_corr(h: np.ndarray):
    """Sub‑gradient of max_k corr_k at h (average over active shifts)."""
    n = h.size
    _, ks = _max_corr_and_shifts(h)
    g = np.zeros_like(h)
    for k in ks:
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    if ks.size > 1:
        g /= ks.size
    return g


def _gradient_descent(h: np.ndarray, max_iters: int, step0: float, decay: float):
    """Projected gradient descent on the max‑overlap objective."""
    n = h.size
    cur = h.copy()
    cur_c5 = _c5_from_h(cur)
    step = step0
    for _ in range(max_iters):
        g = _gradient_max_corr(cur)
        if np.all(g == 0):
            break
        nxt = cur - step * g
        nxt = _project_to_simplex_box(nxt, n / 2.0)
        nxt_c5 = _c5_from_h(nxt)
        if nxt_c5 < cur_c5:
            cur, cur_c5 = nxt, nxt_c5
        else:
            step *= 0.5  # back‑off if no improvement
        step *= decay
        if step < 1e-12:
            break
    return cur, cur_c5


def _pairwise_sa(h: np.ndarray, max_iters: int, temp0: float, decay: float):
    """Simple pairwise simulated annealing preserving sum(h)=n/2."""
    n = h.size
    cur = h.copy()
    cur_c5 = _c5_from_h(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T = temp0
    for _ in range(max_iters):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if high <= low:
            continue
        delta = np.random.uniform(low, high)
        nxt = cur.copy()
        nxt[i] = hi + delta
        nxt[j] = hj - delta
        nxt_c5 = _c5_from_h(nxt)
        accept = False
        if nxt_c5 < cur_c5:
            accept = True
        else:
            prob = np.exp(-(nxt_c5 - cur_c5) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur, cur_c5 = nxt, nxt_c5
            if nxt_c5 < best_c5:
                best, best_c5 = nxt, nxt_c5
        T *= decay
        if T < 1e-12:
            T = 0.0
    return best, best_c5

    # Grid sizes to try.
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (256, 320, 384, 448):
        n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Build starting vector for this resolution.
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        base_h = _project_to_simplex_box(base_h, n / 2.0)

        # ---- 1) optional SLSQP refinement ----
        cur_h = base_h
        cur_c5 = _c5_from_h(cur_h)
        if time_left() > 20.0:
            # heuristic iteration count depending on n
            max_iter = max(200, int(500 * (200.0 / n)))
            cur_h, cur_c5 = _solve_h(cur_h, max_iter=max_iter)

        # ---- 2) Gradient descent ----
        if time_left() > 10.0:
            # allocate iterations based on remaining budget
            iters = int(min(8000, max(2000, time_left() * 5)))
            cur_h, cur_c5 = _gradient_descent(
                cur_h, max_iters=iters, step0=0.04, decay=0.9997
            )

        # ---- 3) Pairwise simulated annealing ----
        if time_left() > 10.0:
            moves = int(min(50000, max(8000, time_left() * 40)))
            cur_h, cur_c5 = _pairwise_sa(
                cur_h, max_iters=moves, temp0=0.02, decay=0.9999
            )

        # Record if it's the best so far.
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h, cur_c5, n
            if best_c5 < 0.3808:
                break

    # Fallback if nothing succeeded.
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
