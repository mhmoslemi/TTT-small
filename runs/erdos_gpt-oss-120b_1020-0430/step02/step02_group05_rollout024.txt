import time
import numpy as np
from scipy import optimize

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x_i<=1, sum x_i = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Refine h with SLSQP and return (h_opt, C5)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape h(x)=min(x,2-x) sampled on n points and projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _shape_h(p: float, n: int) -> np.ndarray:
    """Family h(x)=α·(1-|x-1|^p) with exact area 1 (α=(p+1)/(2p))."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = np.abs(x - 1.0)
    base = 1.0 - t**p
    alpha = (p + 1.0) / (2.0 * p)
    h = alpha * base
    h = np.clip(h, 0.0, 1.0)
    # projection fixes the tiny discretisation error in the integral
    return _project_to_simplex_box(h, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # obtain parent if any
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate grid sizes (prefer the parent size, then a modest spread)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (192, 256, 384):
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 8.0:
            break

        # base h: either the parent (rescaled) or a tent
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        # -----------------------------------------------------------------
        # 1) explore parametric family h(p) on this grid
        # -----------------------------------------------------------------
        # minimise C5 over p∈[1,5] using bounded scalar optimisation
        def obj_p(p):
            h = _shape_h(p, n)
            return _c5_from_h(h)

        # coarse bounded optimisation
        res_p = optimize.minimize_scalar(
            obj_p, bounds=(1.0, 5.0), method="bounded", options={"xatol": 1e-3}
        )
        p_best = res_p.x
        h_best = _shape_h(p_best, n)
        c5_val = _c5_from_h(h_best)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_best, c5_val, n
            if best_c5 < target_bound:
                break

        # -----------------------------------------------------------------
        # 2) local refinement of the best parametric shape with SLSQP
        # -----------------------------------------------------------------
        h_opt, c5_val = _solve_h(h_best, max_iter=1200)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target_bound:
                break

        # -----------------------------------------------------------------
        # 3) stochastic restarts from this refined shape
        # -----------------------------------------------------------------
        attempts = 15
        sigma = 0.12
        for _ in range(attempts):
            if time_left() < 6.0:
                break
            if np.random.rand() < 0.7:
                start = best_h + sigma * np.random.randn(n)
            else:
                start = np.random.rand(n)
            start = _project_to_simplex_box(start, n / 2.0)
            h_opt, c5_val = _solve_h(start, max_iter=800)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 < target_bound:
                    break
            sigma *= 0.90

        if best_c5 < target_bound:
            break

    # final fallback (unlikely)
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
