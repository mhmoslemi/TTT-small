def adam_optimize(h0, beta_sched, steps_per_beta, rng, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for beta in beta_sched:
        for _ in range(steps_per_beta):
            t += 1
            # compute gradient of smooth max (as before)
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * (2.0 / h.size)
            bc = beta * C
            mmax = np.max(bc)
            w = np.exp(bc - mmax)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.zeros_like(w)
            else:
                w /= w_sum
            # compute gradient of smooth max wrt h. Derive formula:
            # For each shift k where weight wk, derivative of C_k = (2/n) * d/dh sum_i h_i * (1 - h_{i+k})
            # d/dh_i of C_k = (2/n)*( (1 - h_{i+k}) - h_{i-k} ) depending on i and k. Because sum_i h_i * (1 - h_{i+k}) = sum_i h_i - sum_i h_i * h_{i+k} .
            # derivative of sum_i h_i is 1 for each i, derivative of - sum_i h_i h_{i+k} is -(h_{i+k} + h_{i-k}) if both defined.
            # So gradient g_i = (2/n)*(1 - (h_{i+k} + h_{i-k}) ) over each k, weighted.
            # but careful for boundaries where indices out of range; treat missing terms as zero.
            # Let's implement loops using weight w_k to accumulate contributions.
            g = np.zeros_like(h)
            n = h.size
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)  # shift offset
                # base term: 1 for each i where i range 0..n-1 => g += wk * 1
                # We'll add wk for all entries
                g += wk
                # subtract wk * h_{i+k} for i where i+k in range
                # For i from 0..n-1-k (if k>=0) else i from -k..n-1
                if k >= 0:
                    # i in [0, n-1-k]
                    g[:n - k] -= wk * h[k:]
                    # also subtract wk * h_{i-k} (the other side) => for i where i-k >=0 => i>=k
                    g[k:] -= wk * h[:n - k]
                else:
                    kk = -k
                    # i in [kk, n-1] -> i - kk >=0
                    g[kk:] -= wk * h[:n - kk]
                    # i in [0, n-1-kk] for second term
                    g[:n - kk] -= wk * h[kk:]
            # Multiply by scaling factor (2/n)
            g *= (2.0 / n)

            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            # projection
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, n / 2.0)
            # enforce symmetry
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = project_onto_box_sum(h_new, n / 2.0)
            # evaluate
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new

    return best_h, best_c

    n = h.size

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h = h.copy()

    for beta in beta_sched:
        for _ in range(steps_per_beta):
            t += 1
            grad, _ = smooth_max_and_gradient(h, beta)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

import time

from scipy.optimize import minimize

# ----------------- basic utilities -----------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while fixing the sum = target."""
    lo = -1e9
    hi = 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible vector (0≤h≤1, Σh=n/2)."""
    if n % 2:
        n += 1
    half = n // 2
    target_half = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])

def perturb_h(h, rng, scale=0.2):
    i = rng.integers(0, h.size)
    j = rng.integers(0, h.size)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the smoothed max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ----------------- smooth max gradient -----------------
def smooth_grad(h, beta):
    """Gradient of the β‑soft‑max approximation to max C(k)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    C = corr * (2.0 / n)               # scaled overlaps
    bc = beta * C
    mx = np.max(bc)
    w = np.exp(bc - mx)
    w /= w.sum()                       # soft‑max weights

    # term_i = Σ_k w_k (h_{i+k} + h_{i‑k}), out‑of‑range h assumed 0
    conv1 = np.correlate(w, h, mode="full")
    conv2 = np.correlate(w[::-1], h, mode="full")
    off = n - 1
    term = conv1[off : off + n] + conv2[off : off + n]

    grad = (2.0 / n) * (1.0 - term)
    return grad, np.max(C)


# ----------------- Adam optimizer -----------------
def adam_optimize(h0, beta_sched, steps_per_beta, rng,
                  lr=0.02, b1=0.9, b2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in beta_sched:
        for _ in range(steps_per_beta):
            t += 1
            grad, _ = smooth_grad(h, beta)
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            nh = h - step
            nh = np.clip(nh, 0.0, 1.0)
            nh = project_onto_box_sum(nh, target)
            # enforce symmetry
            nh = (nh + nh[::-1]) * 0.5
            nh = project_onto_box_sum(nh, target)

            c = compute_c5(nh)
            if c < best_c:
                best_c = c
                best_h = nh.copy()
            h = nh
    return best_h, best_c


# ----------------- main run -----------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial construction ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 120
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        tri = np.minimum(xs, 2.0 - xs)          # triangular shape
        h0 = project_onto_box_sum(tri, n / 2.0)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- quick hill‑climb if budget permits ----
    if time.time() - start < budget_s * 0.1:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=3000, start_step=0.5)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---- gradient‑descent (Adam) restarts ----
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0]
    steps_per_beta = 250
    max_restarts = 30
    restarts = 0
    while (time.time() - start) < budget_s * 0.78 and restarts < max_restarts:
        restarts += 1
        h_start = random_feasible_symmetric(n, rng)
        if rng.random() < 0.15:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)
            h_start = (h_start + h_start[::-1]) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)

        h_opt, c_opt = adam_optimize(h_start, beta_sched, steps_per_beta, rng)

        # short SLSQP refinement passes
        for bet in (200.0, 500.0):
            try:
                h_opt = slsqp_optimize(h_opt, beta=bet, max_iter=200)
            except Exception:
                pass

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- optional up‑sampling ----
    if (time.time() - start) < budget_s * 0.95 and best_h.size * 2 <= 200:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size

        # hill‑climb on up‑sampled vector
        h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.4)

        # fine‑tune with Adam
        beta_up = [5.0, 20.0, 80.0, 200.0]
        h_up, _ = adam_optimize(h_up, beta_up, 150, rng)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
