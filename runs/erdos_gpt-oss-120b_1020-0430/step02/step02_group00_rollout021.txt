def project_to_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # If sum equals target within tol, return
    s = h.sum()

        return h

    diff = target - s

    # We'll adjust free variables proportionally.

def _project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Use simple water-filling: adjust up/down in loop
    max_iter = 100
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass to those not at 1
            free = np.where(h < 1 - 1e-12)[0]
            if free.size == 0:
                break
            add = diff / free.size
            # ensure not exceed 1
            possible_add = np.minimum(add, 1 - h[free])
            h[free] += possible_add
        else:
            # need to subtract from those >0
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            sub = -diff / free.size
            possible_sub = np.minimum(sub, h[free])
            h[free] -= possible_sub
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # final adjustment within tolerance
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # distribute diff uniformly to all entries without violating bounds
        # simple approach: adjust first entry within bound
        for i in range(n):
            if diff > 0 and h[i] < 1 - 1e-12:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 1e-12:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import time
import numpy as np

# --------------------------------------------------------------
# Basic helpers

def _project_to_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1 - 1e-12)[0]
            if free.size == 0:
                break
            add = diff / free.size
            inc = np.minimum(add, 1 - h[free])
            h[free] += inc
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            sub = -diff / free.size
            dec = np.minimum(sub, h[free])
            h[free] -= dec
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _correlation_fft(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h and (1‑h) using FFT (full mode)."""
    n = h.shape[0]
    L = 2 * n - 1
    A = np.fft.rfft(h, L)
    B = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(A * np.conj(B), L)
    return corr.real


def _c5_from_h(h: np.ndarray) -> float:
    """C₅ value for a feasible step function."""
    n = h.size
    corr = _correlation_fft(h)
    return np.max(corr) * (2.0 / n)


def _gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the correlation term with shift k."""
    n = h.size
    g = np.ones(n)
    if k >= 0:
        # subtract h_{i+k}
        g[: n - k] -= h[k:]
        # subtract h_{i‑k}
        g[k:] -= h[: n - k]
    else:
        kk = -k
        # subtract h_{i+k} = h_{i‑kk}
        g[kk:] -= h[: n - kk]
        # subtract h_{i‑k} = h_{i+kk}
        g[: n - kk] -= h[kk:]
    return g


def _random_binary_h(n: int) -> np.ndarray:
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def _upscale_h(h_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_small = h_small.size
    xs = (np.arange(n_small) + 0.5) * (2.0 / n_small)
    xb = (np.arange(n_big) + 0.5) * (2.0 / n_big)
    return np.interp(xb, xs, h_small)

# Core optimiser (time‑bounded)
# --------------------------------------------------------------
def _optimize_h(start_h: np.ndarray, time_budget: float) -> tuple:
    """Local search (gradient + random swaps) limited by `time_budget` seconds."""
    n = start_h.size
    h_cur = _project_to_feasible(start_h, n)
    c5_cur = _c5_from_h(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    start = time.time()
    it = 0
    target = 0.3808  # early‑stop threshold
    while time.time() - start < time_budget:
        if best_c5 < target:
            break
        # ------------------------------------------------------------------
        # Choose move type
        # ------------------------------------------------------------------
        if np.random.rand() < 0.25:                     # random transfer
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h_cur[i] < 1e-12 or h_cur[j] > 1 - 1e-12:
                continue
            max_delta = min(h_cur[i], 1.0 - h_cur[j])
            if max_delta <= 0:
                continue
            delta = max_delta * np.random.rand()
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:                                            # gradient step
            corr = _correlation_fft(h_cur)
            idx_max = int(np.argmax(corr))
            k = idx_max - (n - 1)
            grad = _gradient_for_shift(h_cur, k)
            eta = 0.5 / np.sqrt(it + 1)                  # diminishing step
            h_new = h_cur - eta * grad
            h_new = _project_to_feasible(h_new, n)
        # ------------------------------------------------------------------
        c5_new = _c5_from_h(h_new)
        if c5_new + 1e-12 < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new + 1e-12 < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
        it += 1
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    t0 = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - t0))

    target_c5 = 0.3808

    # ------------------------------------------------------------------
    # Initial construction (provided or fall‑back)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        # golden‑ratio binary pattern, length 256
        n0 = 256
        phi = (np.sqrt(5) - 1) / 2
        idx = np.arange(n0)
        fr = (idx * phi) % 1.0
        order = np.argsort(fr)
        init_h = np.zeros(n0, dtype=float)
        init_h[order[: n0 // 2]] = 1.0
    n0 = init_h.size
    init_h = _project_to_feasible(init_h, n0)

    # ------------------------------------------------------------------
    # Primary optimisation on the original resolution
    # ------------------------------------------------------------------
    primary_budget = time_left() * 0.55
    best_h, best_c5 = _optimize_h(init_h, primary_budget)
    best_n = n0

    # ------------------------------------------------------------------
    # Random restarts while time permits
    # ------------------------------------------------------------------
    while time_left() > 12.0 and best_c5 > target_c5:
        remaining = time_left()
        # each restart gets at most 20 % of the remaining time, capped at 100 s
        restart_budget = min(remaining * 0.2, 100.0)
        start_h = _random_binary_h(best_n)
        h_opt, c5_opt = _optimize_h(start_h, restart_budget)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if best_c5 < target_c5:
            break

    # ------------------------------------------------------------------
    # If still not good enough, upscale and refine
    # ------------------------------------------------------------------
    if time_left() > 6.0 and best_c5 > target_c5:
        n_up = best_n * 2
        h_up = _upscale_h(best_h, n_up)
        h_up = _project_to_feasible(h_up, n_up)
        h_opt, c5_opt = _optimize_h(h_up, time_left())
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_up

    # ------------------------------------------------------------------
    # Final safety projection & evaluation
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_h.size)
