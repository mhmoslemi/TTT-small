def _c5_fast(h: np.ndarray) -> float:
    n = h.size

    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

# ----------------------------------------------------------------------
# C5 evaluation (exact – uses np.correlate)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

# ----------------------------------------------------------------------
# Fast C5 evaluation (FFT) – used inside local searches
# ----------------------------------------------------------------------
def _c5_fast(h: np.ndarray) -> float:
    n = h.size
    m = 2 * n
    f = np.fft.rfft(h, n=m)
    g = np.fft.rfft((1.0 - h)[::-1], n=m)
    conv = np.fft.irfft(f * g, n=m)[: 2 * n - 1]
    return conv.max() * (2.0 / n)

# ----------------------------------------------------------------------
# SLSQP helper (used for small grids)
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")

def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)

def _objective(x: np.ndarray) -> float:
    return x[-1]

def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

# ----------------------------------------------------------------------
# Upscaling (linear interpolation)
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# ----------------------------------------------------------------------
# Convert a continuous vector to a binary one with exactly n/2 ones
# ----------------------------------------------------------------------
def _binary_from_continuous(fp: np.ndarray, n: int) -> np.ndarray:
    idx = np.argsort(-fp, kind='stable')
    h = np.zeros(n, dtype=float)
    h[idx[: n // 2]] = 1.0
    return h

# ----------------------------------------------------------------------
# Fractional annealing (preserves sum, respects bounds)
# ----------------------------------------------------------------------
def _fractional_anneal(h0: np.ndarray, max_iters: int, time_left_func) -> tuple:
    n = h0.size
    h = h0.copy()
    score = _c5_fast(h)
    best_h = h.copy()
    best_score = score
    step = 0.2
    for it in range(max_iters):
        if time_left_func() < 0.5:
            break
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = h[i], h[j]
        delta_low = max(-hi, hj - 1.0)
        delta_up = min(1.0 - hi, hj)
        if delta_low >= delta_up:
            continue
        delta = (np.random.rand() * 2.0 - 1.0) * step
        delta = np.clip(delta, delta_low, delta_up)
        h[i] = hi + delta
        h[j] = hj - delta
        new_score = _c5_fast(h)
        if new_score < score:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step *= 0.7
    return best_h, best_score

# ----------------------------------------------------------------------
# Binary hill climbing (swaps a 1 with a 0)
# ----------------------------------------------------------------------
def _binary_hill_climb(h0: np.ndarray, max_iters: int, time_left_func) -> tuple:
    n = h0.size
    h = h0.copy()
    score = _c5_fast(h)
    best_h = h.copy()
    best_score = score
    for it in range(max_iters):
        if time_left_func() < 0.5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0, 1
        new_score = _c5_fast(h)
        if new_score < score:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_score

# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # Stage 1 – small‑grid SLSQP optimisation
    # ------------------------------------------------------------------
    if init_h is not None:
        n0 = init_h.size
        h_opt, c5_val = _solve_h(init_h, max_iter=400)
        if c5_val < best_score:
            best_h, best_score, best_n = h_opt, c5_val, n0
        # a few random perturbations
        for _ in range(3):
            if time_left() < 2.0:
                break
            pert = 0.15 * np.random.randn(n0)
            start = _project_to_feasible(init_h + pert, n0)
            h_opt, c5_val = _solve_h(start, max_iter=300)
            if c5_val < best_score:
                best_h, best_score, best_n = h_opt, c5_val, n0
    else:
        # fallback binary half‑ones at size 256
        n0 = 256
        init_h = np.zeros(n0, dtype=float)
        init_h[: n0 // 2] = 1.0
        np.random.shuffle(init_h)
        h_opt, c5_val = _solve_h(init_h, max_iter=400)
        best_h, best_score, best_n = h_opt, c5_val, n0

    # ------------------------------------------------------------------
    # Early exit if time is exhausted
    # ------------------------------------------------------------------
    if time_left() < 10.0:
        best_h = _project_to_feasible(best_h, best_h.size)
        return best_h.tolist(), float(_c5_from_h(best_h)), int(best_h.size)

    # ------------------------------------------------------------------
    # Stage 2 – increase resolution, fractional annealing + binary climb
    # ------------------------------------------------------------------
    for n in (512, 768, 1024):
        if time_left() < 15.0:
            break
        # upscale the current best (may have different size)
        up_h = _upscale_h(best_h, n)
        up_h = _project_to_feasible(up_h, n)

        # fractional annealing
        max_iters = int(min(50000, time_left() * 30))
        h_frac, score_frac = _fractional_anneal(up_h, max_iters, time_left)
        if score_frac < best_score:
            best_h, best_score, best_n = h_frac, score_frac, n

        # binary version + hill climbing
        bin_h = _binary_from_continuous(up_h, n)
        max_iters_bin = int(min(40000, time_left() * 30))
        h_bin, score_bin = _binary_hill_climb(bin_h, max_iters_bin, time_left)
        if score_bin < best_score:
            best_h, best_score, best_n = h_bin, score_bin, n
        if best_score < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final SLSQP polish (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 6.0:
        h_opt, c5_val = _solve_h(best_h, max_iter=800)
        if c5_val < best_score:
            best_h, best_score = h_opt, c5_val
            best_n = best_h.size

    # ------------------------------------------------------------------
    # Ensure feasibility and compute final bound with exact routine
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_score = _c5_from_h(best_h)

    return best_h.tolist(), float(best_score), int(best_h.size)
