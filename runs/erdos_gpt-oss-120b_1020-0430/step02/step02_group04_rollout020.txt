def grad_adam_optimize(h0, beta_sched, steps_per_beta, rng):

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0

    # hyperparameters
    lr = 0.1
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    for beta in beta_sched:
        for _ in range(steps_per_beta):
            t += 1
            # correlation
            corr = np.correlate(h, 1 - h, mode='full')
            bc = beta * corr
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w /= w_sum

            # gradient via convolution
            conv = np.convolve(h, w, mode='full')
            conv_rev = np.convolve(h, w[::-1], mode='full')
            start = n - 1
            u = conv[start:start + n]
            v2 = conv_rev[start:start + n]
            grad = 1.0 - (u + v2)

            # Adam update
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            h_new = h - step

            # projection
            h_new = project_onto_box_sum(h_new, n / 2.0)
            # symmetrize
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = project_onto_box_sum(h_new, n / 2.0)

            # evaluate
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                # sometimes accept uphill with probability decreasing
                # For simplicity, accept if random < 0.005
                if rng.random() < 0.005:
                    h = h_new
                # else keep old h
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# basic utilities
# -------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while fixing the sum == target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric vector h with sum = n/2 and values in [0,1]."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    h = np.concatenate([x, x[::-1]])
    return h


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def symmetrize(h):
    h = (h + h[::-1]) * 0.5
    return h


# -------------------------------------------------
# smooth‑max / SLSQP utilities
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# -------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# -------------------------------------------------
def grad_adam_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected Adam optimiser on smooth‑max of C_k."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    # Adam state
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    lr = 0.1
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    for beta in beta_sched:
        for _ in range(steps_per_beta):
            t += 1
            corr = np.correlate(h, 1.0 - h, mode="full")
            bc = beta * corr
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w /= w_sum

            # gradient via convolution
            conv_fwd = np.convolve(h, w, mode="full")
            conv_rev = np.convolve(h, w[::-1], mode="full")
            start = n - 1
            u = conv_fwd[start:start + n]
            v2 = conv_rev[start:start + n]
            grad = 1.0 - (u + v2)

            # Adam update
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad * grad)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            h_new = h - step
            h_new = project_onto_box_sum(h_new, n / 2.0)
            h_new = symmetrize(h_new)
            h_new = project_onto_box_sum(h_new, n / 2.0)

            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                if rng.random() < 0.005:  # occasional uphill move
                    h = h_new
    return best_h, best_c


# -------------------------------------------------
# main driver
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise from parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        h0 = symmetrize(h0)
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 120
        h0 = random_feasible_symmetric(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # quick hill‑climb if time allows
    if time.time() - start < budget_s * 0.2:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2000, start_step=0.5)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # main stochastic optimisation
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0]
    steps_per_beta = 250

    while time.time() - start < budget_s * 0.85:
        h_start = random_feasible_symmetric(best_h.size, rng)
        # occasional mixing with current best
        if rng.random() < 0.2:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, best_h.size / 2.0)
            h_start = symmetrize(h_start)
            h_start = project_onto_box_sum(h_start, best_h.size / 2.0)

        h_opt, c_opt = grad_adam_optimize(h_start, beta_sched, steps_per_beta, rng)

        # short SLSQP refinements
        try:
            for beta in (80.0, 200.0):
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=250)
        except Exception:
            pass

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # optional up‑sampling for finer resolution
    if time.time() - start < budget_s * 0.95 and best_h.size * 2 <= 400:
        h_up = np.repeat(best_h, 2)
        h_up = project_onto_box_sum(h_up, h_up.size / 2.0)
        h_up = symmetrize(h_up)
        h_up = project_onto_box_sum(h_up, h_up.size / 2.0)

        up_beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0]
        up_steps = 200
        h_up_opt, c_up_opt = grad_adam_optimize(h_up, up_beta_sched, up_steps, rng)

        try:
            for beta in (200.0, 500.0):
                h_up_opt = slsqp_optimize(h_up_opt, beta=beta, max_iter=300)
        except Exception:
            pass

        c_up_opt = compute_c5(h_up_opt)
        if c_up_opt < best_c:
            best_h, best_c = h_up_opt, c_up_opt

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
