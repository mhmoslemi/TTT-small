import time, numpy as np

def _project_to_feasible(h, n, tol=1e-12):
    # clip and water-filling to sum = n/2
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # simple water-filling using sorting
    # compute excess or deficit
    sum_h = h.sum()
    diff = sum_h - target
    if abs(diff) < tol:
        return h
    # If sum too high, need to reduce some values; if too low, increase
    if diff > 0:
        # reduce values: sort indices by h (largest first)
        idx = np.argsort(-h)  # descending
        for i in idx:
            amount = min(diff, h[i])
            h[i] -= amount
            diff -= amount
            if diff <= tol:
                break
    else:
        # increase values: sort by (1-h) descending (i.e., smallest h first)
        idx = np.argsort(h)  # ascending
        diff = -diff
        for i in idx:
            amount = min(diff, 1.0 - h[i])
            h[i] += amount
            diff -= amount
            if diff <= tol:
                break
    # final clip
    return np.clip(h, 0.0, 1.0)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _c5_from_corr(max_corr, n):
    return (2.0 / n) * max_corr

import time
import numpy as np

# ---------- feasibility projection (water‑filling) ---------- #
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

# ---------- linear up‑sampling ---------- #
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# ---------- C5 from a correlation maximum ---------- #
def _c5_from_max_corr(max_corr: float, n: int) -> float:
    return (2.0 / n) * max_corr

# ------------------- main driver ------------------- #
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ---- starting construction ------------------------------------------------
    try:
        parent = np.array(initial_h_values, dtype=float)
    except NameError:
        parent = None

    # target grid size – 256 points give enough resolution while staying fast
    target_n = 256
    if parent is not None:
        if parent.size > target_n:
            target_n = parent.size + (parent.size % 2)  # make even
    n = target_n

    if parent is not None and parent.size != n:
        h = _upscale_h(parent, n)
    elif parent is not None:
        h = parent.copy()
    else:
        h = np.full(n, 0.5)

    h = _project_to_feasible(h, n)

    # ---- initial correlation --------------------------------------------------
    f = np.correlate(h, 1.0 - h, mode="full")
    max_corr = f.max()
    best_h = h.copy()
    best_corr = max_corr
    best_c5 = _c5_from_max_corr(best_corr, n)

    # auxiliary vectors
    X = 1.0 - h               # 1‑h
    Y = h.copy()              # h

    # ---- simulated annealing with pairwise mass transfers ---------------------
    total_steps = int(5e5)      # generous upper bound, real stop is time limit
    step = 0
    T0 = 5e-4
    scale_start = 0.5
    scale_end = 0.02

    while time_left() > 0.5 and step < total_steps:
        frac = step / total_steps
        T = T0 * max(0.0, 1.0 - frac)          # linear cooling
        step_scale = scale_start * (1.0 - frac) + scale_end * frac

        nonzero = np.where(h > 1e-12)[0]
