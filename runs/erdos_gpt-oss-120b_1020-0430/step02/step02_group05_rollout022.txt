def _smooth_max_grad(h, alpha):
    n = h.size
    C = np.correlate(h, 1 - h, mode='full')
    # compute smooth max
    max_alphaC = np.max(alpha * C)
    exp_term = np.exp(alpha * C - max_alphaC)
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp  # normalized weights over shifts.
    # compute convolution gradients
    L = 1 << ((len(w) + n - 1).bit_length())
    # Use real FFT for speed
    fft_w = np.fft.rfft(w, n=L)
    fft_w_rev = np.fft.rfft(w[::-1], n=L)
    fft_h = np.fft.rfft(h, n=L)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=L)
    conv_h = np.fft.irfft(fft_w * fft_h, n=L)
    conv_one_minus_h_rev = np.fft.irfft(fft_w_rev * fft_one_minus_h, n=L)
    # slice
    start = n - 1
    grad = conv_one_minus_h_rev[start:start + n] - conv_h[start:start + n]
    # objective (smooth max) actual value
    smooth_val = (max_alphaC + np.log(sum_exp)) / alpha
    return smooth_val, grad

import time
import numpy as np
from scipy import optimize

# ---------- utilities ----------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C₅ for a discrete h on [0,2]."""
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / h.size)


def _smooth_max_grad(h: np.ndarray, alpha: float):
    """Smooth approximation of max correlation and its gradient."""
    n = h.size
    C = np.correlate(h, 1.0 - h, mode="full")
    max_aC = np.max(alpha * C)
    exp_term = np.exp(alpha * C - max_aC)
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp

    L = 1 << ((len(w) + n - 1).bit_length())
    fft_w = np.fft.rfft(w, n=L)
    fft_w_rev = np.fft.rfft(w[::-1], n=L)
    fft_h = np.fft.rfft(h, n=L)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=L)

    conv_h = np.fft.irfft(fft_w * fft_h, n=L)
    conv_one_minus_h_rev = np.fft.irfft(fft_w_rev * fft_one_minus_h, n=L)

    start = n - 1
    grad = conv_one_minus_h_rev[start:start + n] - conv_h[start:start + n]
    smooth_val = (max_aC + np.log(sum_exp)) / alpha
    return smooth_val, grad


def _adam_optimize(start_h: np.ndarray, max_iters: int, deadline: float):
    """Projected Adam on the smooth‑max surrogate."""
    n = start_h.size
    h = start_h.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    alpha_start, alpha_end = 2.0, 80.0

    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    for i in range(max_iters):
        if time.time() > deadline:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (i / max_iters)
        _, grad = _smooth_max_grad(h, alpha)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        t = i + 1
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = _project_to_simplex_box(h - step, n / 2.0)

        if (i & 15) == 0:  # every 16 iterations
            c5 = _c5_from_h(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    best_h = _project_to_simplex_box(best_h, n / 2.0)
    return best_h, best_c5


def _ineq_constraint(x: np.ndarray):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray):
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 1500):
    """Local refinement with SLSQP."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": _ineq_constraint},
            {"type": "eq", "fun": _eq_constraint}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = h0 if not res.success else res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse to fine grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points, then projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)
