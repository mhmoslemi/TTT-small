def simulated_annealing(h0, rng, max_iter, start_temp, end_temp):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    # Precompute indices of ones and zeros to pick quickly
    for it in range(max_iter):
        T = start_temp + (end_temp - start_temp) * (it / max_iter)  # linear cooling
        # select i among ones, j among zeros
        ones = np.where(h > 0.5)[0]  # binary approx
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # optionally, after certain number of iterations adjust T
    return best_h, best_c
```

def simulated_annealing_binary(h0, rng, max_iters, start_temp, end_temp, time_limit):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    # store indices of ones/zeros as python lists for fast random choice (update after each swap)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    start_time = time.time()
    for it in range(max_iters):
        # time check
        if time.time() - start_time > time_limit:
            break
        # linear cooling schedule
        T = start_temp + (end_temp - start_temp) * (it / max_iters)
        # choose random i,j
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # update ones/zeros lists
        # after swap, i becomes zero, j becomes one
        # we need to update lists accordingly
        # remove i from ones, add to zeros, etc.
        # To maintain O(1) we can use indices to replace; but n small we can recompute each iteration.
        # Simpler: after evaluating we revert if not accepted; only update lists if accepted.
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            # update lists
            # i removed from ones, added to zeros
            # j removed from zeros, added to ones
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # optional periodic restart if stagnation
        # could track no improvement for many steps
    return best_h, best_c

def simulated_annealing_continuous(h0, rng, max_iters, start_temp, end_temp):
    h = h0.astype(float).copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    move_scale = 0.5
    for it in range(max_iters):
        # compute temperature
        T = start_temp + (end_temp - start_temp) * (it / max_iters)
        # choose two indices i != j
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute max possible delta
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move * move_scale
        # modify
        h[i] -= delta
        h[j] += delta
        new_c = compute_c5(h)
        delta_c = new_c - cur_c
        accept = False
        if delta_c < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta_c / T)
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_c

def gradient_refine_simple(h, rng, max_iter=10, step=0.02):
    h = h.copy()
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple proportional correction
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            # final random repair if needed
            if abs(diff) > 1e-12:
                for _ in range(200):
                    i = rng.integers(0, n)
                    if diff > 0:
                        add = min(diff, 1.0 - h[i])
                    else:
                        add = max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
    return h
```

import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def repair_feasible(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_binary_feasible(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def simulated_annealing_continuous(h0, rng, max_iters, start_temp, end_temp):
    h = h0.astype(float).copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    move_scale = 0.5  # scale of transfer per move
    for it in range(max_iters):
        T = start_temp + (end_temp - start_temp) * (it / max_iters)
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move * move_scale
        h[i] -= delta
        h[j] += delta
        new_c = compute_c5(h)
        delta_c = new_c - cur_c
        if delta_c < 0 or (T > 0 and rng.random() < np.exp(-delta_c / T)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_c

def gradient_refine_simple(h, rng, max_iter=5, step=0.02):
    h = h.copy()
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple proportional correction
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                for _ in range(200):
                    i = rng.integers(0, n)
                    if diff > 0:
                        add = min(diff, 1.0 - h[i])
                    else:
                        add = max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
    return h

def optimize_on_grid(n, rng, start_time, budget_s):
    target = n / 2.0
    best_h = None
    best_c = np.inf
    # use parent if present
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            h0 = repair_feasible(parent, target, rng)
            # small SA

            if iters > 0:

                h_ref = gradient_refine_simple(h_sa, rng, max_iter=5)

            # sum is preserved by repetition, but enforce precision
            target = n_up / 2.0
            diff = target - h_up.sum()
            if abs(diff) > 1e-12:
                h_up += diff / n_up
                h_up = np.clip(h_up, 0.0, 1.0)
                diff = target - h_up.sum()
                for _ in range(2000):
                    if abs(diff) < 1e-12:
                        break
                    i = rng.integers(0, n_up)
                    if diff > 0:
                        add = min(diff, 1.0 - h_up[i])
                    else:
                        add = max(diff, -h_up[i])
                    h_up[i] += add
                    diff = target - h_up.sum()
            h_up, c_up = optimise_grid(n_up, rng, start, budget_s)
            if c_up < c_best:
                h_best, c_best = h_up, c_up
    # final feasibility correction
    n_fin = h_best.size
    target = n_fin / 2.0
    h_best = np.clip(h_best, 0.0, 1.0)
    diff = target - h_best.sum()
    if abs(diff) > 1e-12:
        h_best += diff / n_fin
        h_best = np.clip(h_best, 0.0, 1.0)
        diff = target - h_best.sum()
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n_fin)
            if diff > 0:
                add = min(diff, 1.0 - h_best[i])
            else:
                add = max(diff, -h_best[i])
            h_best[i] += add
            diff = target - h_best.sum()
    final_c = compute_c5(h_best)
    return h_best.tolist(), float(final_c), int(n_fin)
