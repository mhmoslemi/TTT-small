def _hill_climb(h, max_steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    step = 0
    while step < max_steps and time.time() < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # compute max epsilon we can move from i to j while staying in [0,1]
        max_down = best_h[i]          # amount we can take from i
        max_up = 1.0 - best_h[j]      # amount we can give to j
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        # choose epsilon: sometimes full move, sometimes random fraction
        if np.random.rand() < 0.2:
            eps = max_eps   # full move
        else:
            eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        # ensure feasibility (should be)
        c = _c5_from_h(cand)
        if c < best_c - 1e-12:
            best_h, best_c = cand, c
        step += 1
    return best_h, best_c

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:

            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h):
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

# ----------------------------------------------------------------------
# Constraints for SLSQP (minimise the auxiliary variable t)
# ----------------------------------------------------------------------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")

def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)

# ----------------------------------------------------------------------
# Local optimisation with SLSQP
# ----------------------------------------------------------------------
def _solve_h(initial_h, max_iter=800):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    res = opt.minimize(lambda x: x[-1], x0, method="SLSQP",
                       bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

# ----------------------------------------------------------------------
# Differential Evolution (global) – uses projection to enforce the sum
# ----------------------------------------------------------------------
def _de_objective(x, n):
    h = _project_to_feasible(x, n)
    return _c5_from_h(h)

def _de_optimize(start_h, max_iter=50, popsize=12, seed=0):
    n = start_h.size
    bounds = [(0.0, 1.0)] * n
    result = opt.differential_evolution(
        func=_de_objective,
        bounds=bounds,
        args=(n,),
        strategy="best1bin",
        maxiter=max_iter,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        init="random",
        updating="deferred",
    )
    h = _project_to_feasible(result.x, n)
    return h, _c5_from_h(h)

# ----------------------------------------------------------------------
# Simple hill‑climb (pairwise epsilon exchange) – larger moves allowed
# ----------------------------------------------------------------------
def _hill_climb(h, max_steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    step = 0
    while step < max_steps and time.time() < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        # occasionally take full epsilon, otherwise a modest random fraction
        if np.random.rand() < 0.2:
            eps = max_eps
        else:
            eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c - 1e-12:
            best_h, best_c = cand, c
        step += 1
    return best_h, best_c

# ----------------------------------------------------------------------
# Upscale a coarse step function by linear interpolation
# ----------------------------------------------------------------------
def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load any provided initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = np.full(80, 0.5)  # uniform feasible start
    base_n = init_h.size

    # ------------------------------------------------------------------
    # Candidate grid sizes (few multiples of the base size)
    # ------------------------------------------------------------------
    n_candidates = [base_n]
    if base_n * 2 <= 256:
        n_candidates.append(base_n * 2)
    for m in (96, 112, 128, 160, 192, 224, 256):
        if m not in n_candidates and m >= base_n:
            n_candidates.append(m)

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Optimisation loop over grid sizes
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Build base h for this grid size
        if n == base_n:
            base_h = init_h.copy()
        else:
            base_h = _upscale_h(init_h, n)
        base_h = _project_to_feasible(base_h, n)

        # Baseline evaluation
        c0 = _c5_from_h(base_h)
        if c0 < best_c5:
            best_h, best_c5, best_n = base_h, c0, n

        # --------------------------------------------------------------
        # Global search with DE (if time permits)
        # --------------------------------------------------------------
        if time_left() > 20.0:
            try:
                h_de, c_de = _de_optimize(base_h, max_iter=50, popsize=12, seed=seed)
                if c_de < best_c5:
                    best_h, best_c5, best_n = h_de, c_de, n
            except Exception:
                pass

        # --------------------------------------------------------------
        # Local refinement with SLSQP
        # --------------------------------------------------------------
        if time_left() > 5.0:
            try:
                h_loc, c_loc = _solve_h(best_h, max_iter=800)
                if c_loc < best_c5:
                    best_h, best_c5, best_n = h_loc, c_loc, n
            except Exception:
                pass

        # --------------------------------------------------------------
        # Final hill‑climb search (time‑controlled)
        # --------------------------------------------------------------
        remaining = time_left()
        if remaining > 2.0:
            limit = time.time() + min(remaining, 30.0)
            best_h, best_c5 = _hill_climb(best_h, max_steps=20000, time_limit=limit)

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final safety projection and evaluation
    # ------------------------------------------------------------------
    if best_h is not None:
        best_h = _project_to_feasible(best_h, best_n)
        best_c5 = _c5_from_h(best_h)
    else:
        best_n = base_n
        best_h = _project_to_feasible(init_h, best_n)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
