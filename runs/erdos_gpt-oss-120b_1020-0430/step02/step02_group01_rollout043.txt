def gradient_logsumexp(h, beta):
    n = h.size
    y = 1 - h
    corr = np.correlate(h, y, mode='full')
    # weights via softmax
    max_corr = np.max(corr)
    # compute exponentials with stability
    exps = np.exp(beta * (corr - max_corr))
    sum_exps = np.sum(exps)
    w = exps / sum_exps  # shape (2n-1)
    grad = np.zeros_like(h)
    # loop over k from -(n-1) to (n-1)
    # use slicing
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k]
        if wk == 0:
            continue
        if k >= 0:
            # A contribution: i from 0 to n - k - 1 inclusive
            i_end = n - k
            if i_end > 0:

def perturb_h_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j: return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0: return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def gradient_logsumexp(h, beta):

    corr = np.correlate(h, y, mode='full')

    # w = exps / np.sum(exps)
    # but to avoid overflow/underflow: they are stable due to subtraction of maxc.
    exps = np.exp(beta * (corr - maxc))
    w = exps / exps.sum()
    grad = np.zeros_like(h)

import time
import numpy as np

# ---------- basic utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def project_capped_simplex(v: np.ndarray, target: float, max_iter: int = 60, tol: float = 1e-10) -> np.ndarray:
    """Projection onto {0≤x≤1, Σx = target} using bisection on the threshold."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
        if hi - lo < tol:
            break
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the capped simplex."""
    v = rng.random(n)
    return project_capped_simplex(v, n / 2.0)


def binary_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample_h(parent_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a larger size."""
    m = parent_h.size
    t_parent = np.arange(m) / m
    t_new = np.arange(new_n) / new_n
    return np.interp(t_new, t_parent, parent_h)


def perturb_transfer(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount from one coordinate to another (keeps sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_local(h0: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 1200, step_start: float = 0.5):
    """Simple hill‑climber that accepts only improvements."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_start
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_transfer(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift attaining the max)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift
    k = idx
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1 - h[k:]
        grad[k:]      -= h[: n - k]
    else:
        kp = -k
        grad[kp:]     += 1 - h[: n - kp]
        grad[: n - kp] -= h[kp:]
    return grad


def gradient_logsumexp(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth log‑sum‑exp surrogate of the max overlap."""
    n = h.size
    y = 1 - h
    corr = np.correlate(h, y, mode="full")
    maxc = np.max(corr)
    exps = np.exp(beta * (corr - maxc))
    w = exps / exps.sum()                      # weights for each shift
    grad = np.zeros_like(h)

    for k in range(-(n - 1), n):
        wk = w[n - 1 + k]
        if wk == 0.0:
            continue
        if k >= 0:
            if n - k > 0:
                grad[: n - k] += wk * (1 - h[k:])
                grad[k:]      -= wk * h[: n - k]
        else:
            kp = -k
            if n - kp > 0:
                grad[kp:]     += wk * (1 - h[: n - kp])
                grad[: n - kp] -= wk * h[kp:]
    return grad

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # optional parent
    parent_avail = "initial_h_values" in globals()
    parent_h = np.array(initial_h_values, dtype=float) if parent_avail else None

    # candidate discretisations (increase a little for finer control)
    candidate_ns = [120, 144, 168, 192, 240]

    for n in candidate_ns:
        target = n / 2.0
        # pre‑compute an up‑sampled parent if available
        if parent_avail:
            parent_up = upsample_h(parent_h, n)
            parent_up = project_capped_simplex(parent_up, target)
        else:
            parent_up = None

        # we will do as many restarts as time permits (max 30 per n)
        restarts = 0
        while True:
            elapsed = time.time() - start
            if elapsed > budget_s * 0.94:
                break
            if restarts >= 30:
                break
            restarts += 1

            # ----------- initial vector ------------------------------------
            if best_n == n and best_h is not None and restarts == 1:
                h = best_h.copy()                     # reuse the current best
            else:
                r = rng.random()
                if r < 0.33:
                    h = binary_feasible(n, rng)
                elif r < 0.66:
                    h = random_feasible(n, rng)
                else:
                    # use up‑sampled parent with a touch of noise
                    if parent_up is not None:
                        h = parent_up + 0.02 * rng.standard_normal(n)
                        h = np.clip(h, 0.0, 1.0)
                        h = project_capped_simplex(h, target)
                    else:
                        h = random_feasible(n, rng)

            # ----------- smooth‑max descent --------------------------------
            schedule = [
                (5.0, 300, 0.09),
                (20.0, 600, 0.06),
                (80.0, 800, 0.04),
                (200.0, 800, 0.025),
                (500.0, 500, 0.015)
            ]
            for beta, iters, lr0 in schedule:
                for it in range(iters):
                    if time.time() - start > budget_s * 0.94:
                        break
                    grad = gradient_logsumexp(h, beta)
                    # simple decay of step within this beta stage
                    step = lr0 * (0.9 ** (it / iters))
                    h = h - step * grad
                    h = project_capped_simplex(h, target)
                # keep track of exact C5 after each beta stage
                c_now = compute_c5(h)
                if c_now < best_c:
                    best_h, best_c, best_n = h.copy(), c_now, n

            # ----------- refinement on the actual worst shift ---------------
            for it in range(6):
                if time.time() - start > budget_s * 0.94:
                    break
                grad_w = worst_shift_gradient(h)
                step = 0.02 * (0.5 ** it)
                h = h - step * grad_w
                h = project_capped_simplex(h, target)
                c_now = compute_c5(h)
                if c_now < best_c:
                    best_h, best_c, best_n = h.copy(), c_now, n

            # ----------- short local hill‑climb -----------------------------
            if time.time() - start < budget_s * 0.96:
                h_hc, c_hc = hill_climb_local(h, rng, max_iters=1200, step_start=0.3)
                if c_hc < best_c:
                    best_h, best_c, best_n = h_hc.copy(), c_hc, n

    # -----------------------------------------------------------------------
    # final sanity check and return
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_capped_simplex(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
