def max_corr_fft(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    N = 2*n - 1
    Nfft = 1 << (N-1).bit_length()
    fa = np.fft.rfft(h, n=Nfft)
    fr = np.fft.rfft(rev, n=Nfft)
    conv = np.fft.irfft(fa * fr, n=Nfft)
    # take the part from 0 to N-1
    conv = conv[:N]
    return np.max(conv) * (2.0 / n)
```

def hill_climb(h, rng, Nfft, max_iters=2000, start_step=0.5):
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5_fft(best_h, Nfft)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(best_h, rng, scale=step)
        c = compute_c5_fft(new_h, Nfft)
        if c < best_c:
            best_c = c
            best_h = new_h
        if (it+1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

def genetic_optimize(n, rng, budget_seconds, initial_vectors=None):
    target = n/2.0
    Nfft = 1 << ((2*n-2).bit_length())
    pop_size = 30
    elite_size = 5
    # Build initial population
    pop = []
    if initial_vectors:
        for vec in initial_vectors:
            if len(vec)==n:
                pop.append(vec.copy())
    # Fill up to pop_size
    while len(pop) < pop_size:
        if rng.random() < 0.5:
            # binary
            h = np.zeros(n)
            idx = rng.choice(n, n//2, replace=False)
            h[idx] = 1.0
        else:
            h = random_feasible_vector(n, rng)
        pop.append(h)
    # Ensure feasibility for all
    pop = [make_feasible(h, target, rng) for h in pop]
    # Main loop
    best_h = None
    best_c = np.inf
    start = time.time()
    generation = 0
    while time.time() - start < budget_seconds:
        generation += 1
        # evaluate
        scores = np.empty(len(pop))
        for i, h in enumerate(pop):
            scores[i] = compute_c5_fft(h, Nfft)
        # update global best
        min_idx = np.argmin(scores)
        if scores[min_idx] < best_c:
            best_c = scores[min_idx]
            best_h = pop[min_idx].copy()
        # sort by scores
        sorted_idx = np.argsort(scores)
        elites = [pop[i].copy() for i in sorted_idx[:elite_size]]
        # generate new population
        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            # select two parents from elites
            p1, p2 = rng.choice(elites, size=2, replace=False)
            alpha = rng.random()
            child = alpha * p1 + (1 - alpha) * p2
            # mutation: add Gaussian noise to some entries
            if rng.random() < 0.8:
                mask = rng.random(n) < 0.1  # 10% mutated
                noise = rng.normal(0.0, 0.05, size=n)
                child[mask] += noise[mask]
            # make feasible
            child = make_feasible(child, target, rng)
            new_pop.append(child)
        pop = new_pop
    return best_h, best_c
```

def make_feasible(x, target, rng, max_iter=1000):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # identify indices that can be increased if diff>0 or decreased if diff<0
        if diff > 0:
            # need to increase some entries
            # choose indices where x < 1
            idx = rng.choice(np.where(x < 1.0)[0]) if np.any(x < 1.0) else None
            if idx is None:
                break
            add = min(diff, 1.0 - x[idx])
        else:
            idx = rng.choice(np.where(x > 0.0)[0]) if np.any(x > 0.0) else None
            if idx is None:
                break
            add = max(diff, -x[idx])  # diff negative
        x[idx] += add
        diff = target - x.sum()
    return x

def genetic_optimize(n, rng, time_budget, initial_vectors=None):
    target = n/2.0
    Nfft = 1 << ((2*n - 2).bit_length())
    pop_size = 30
    elite_size = 5
    # Initialize population
    pop = []
    if initial_vectors:
        for vec in initial_vectors:
            if vec.shape[0] == n:
                pop.append(vec.copy())
    # Fill with random feasible
    while len(pop) < pop_size:
        if rng.random() < 0.5:
            h = np.zeros(n)
            idx = rng.choice(n, n//2, replace=False)
            h[idx] = 1.0
        else:
            h = random_feasible_vector(n, rng)
        pop.append(h)
    # Ensure feasibility
    pop = [make_feasible(h, target, rng) for h in pop]

    best_h = None
    best_c = np.inf
    start = time.time()
    generation = 0
    while time.time() - start < time_budget:
        generation += 1
        scores = np.empty(len(pop))
        for i, h in enumerate(pop):
            scores[i] = compute_c5_fft(h, Nfft)
        # Update best
        min_idx = int(np.argmin(scores))
        if scores[min_idx] < best_c:
            best_c = scores[min_idx]
            best_h = pop[min_idx].copy()
        # sort elites
        sorted_idx = np.argsort(scores)
        elites = [pop[i].copy() for i in sorted_idx[:elite_size]]
        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            p1, p2 = rng.choice(elites, size=2, replace=False)
            alpha = rng.random()
            child = alpha * p1 + (1 - alpha) * p2
            # mutation
            if rng.random() < 0.9:
                # perturb some entries
                mask = rng.random(n) < 0.2
                noise = rng.normal(0.0, 0.05, size=n)
                child[mask] += noise[mask]
            # clip and fix sum
            child = make_feasible(child, target, rng)
            new_pop.append(child)
        pop = new_pop
    # after loop, ensure best_h feasible (should be)
    best_h = make_feasible(best_h, target, rng)
    return best_h, best_c

def compute_c5_fft(h, Nfft):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    fa = np.fft.rfft(h, n=Nfft)
    fr = np.fft.rfft(rev, n=Nfft)
    conv = np.fft.irfft(fa * fr, n=Nfft)
    conv = conv[:2*n-1]
    return np.max(conv) * (2.0 / n)
```

import time
import numpy as np

def make_feasible(x, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust sum to target by moving mass between entries."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(x < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - x[i])
            x[i] += add
        else:
            cand = np.where(x > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -x[i])          # diff is negative
            x[i] += add
        diff = target - x.sum()
    return x

def random_feasible_vector(n, rng):
    """Generate a random feasible step function of length n."""
    return make_feasible(rng.random(n), n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    """Move a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5_fft(h, Nfft):
    """Fast C5 evaluation using FFT based convolution (full linear correlation)."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    fa = np.fft.rfft(h, n=Nfft)
    fr = np.fft.rfft(rev, n=Nfft)
    conv = np.fft.irfft(fa * fr, n=Nfft)
    conv = conv[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)

def compute_c5_exact(h):
    """Exact C5 using numpy correlate (used for final reporting)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def hill_climb(h, rng, Nfft, max_iters=3000, start_step=0.5):
    """Simple deterministic hill‑climb (accepts only improvements)."""
    best_h = h.copy()
    best_c = compute_c5_fft(best_h, Nfft)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(best_h, rng, scale=step)
        c = compute_c5_fft(new_h, Nfft)
        if c < best_c - 1e-12:
            best_c, best_h = c, new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

    """Population based global search (simple GA) respecting the constraints."""
    target = n / 2.0
    Nfft = 1 << ((2 * n - 2).bit_length())
    pop_size = 30
    elite_sz = 5

    pop = []
    if initial_vectors:

                pop.append(vec.copy())
    while len(pop) < pop_size:

            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0

    best_h, best_c, best_n = None, np.inf, None

    for i, n in enumerate(cand_ns):
