Pseudo:

def compute_c5(h):
    n = h.size
    a = h
    b = 1 - h
    L = 2*n - 1
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(fa * np.conj(fb), n=L)
    max_corr = np.max(corr)

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Feasibility projection (clip to [0,1] and enforce sum = n/2)
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1.0 - 1e-12:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# --------------------------------------------------------------
# C5 value using FFT based correlation (fast)
# --------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    a = h
    b = 1.0 - h
    L = 2 * n - 1
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(fa * np.conj(fb), n=L)
    return np.max(corr) * (2.0 / n)


# --------------------------------------------------------------
# Upscale via linear interpolation
# --------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    xs_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xs_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xs_fine, xs_coarse, coarse_h)


# --------------------------------------------------------------
# Simple simulated annealing with pairwise mass transfer
# --------------------------------------------------------------
def _anneal_h(start_h: np.ndarray, iterations: int,
              temp_start: float = 0.6, temp_end: float = 1e-4,
              delta_start: float = 0.4, delta_end: float = 0.02) -> tuple:
    n = start_h.size
    cur = start_h.copy()
    cur_val = _c5_from_h(cur)
    best_h = cur.copy()
    best_val = cur_val
    for it in range(iterations):
        # temperature schedule
        t = temp_start * (temp_end / temp_start) ** (it / iterations)
        # delta magnitude schedule
        delta_max = delta_start + (delta_end - delta_start) * (it / iterations)
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]
        # propose transfer delta from i to j (i loses, j gains)
        delta = (np.random.rand() * 2.0 - 1.0) * delta_max
        # clamp to keep within [0,1]
        delta = max(-hi, min(delta, 1.0 - hj))
        if abs(delta) < 1e-14:
            continue
        cur[i] = hi - delta
        cur[j] = hj + delta
        new_val = _c5_from_h(cur)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_val - cur_val) / max(t, 1e-12)):
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
        else:
            # revert
            cur[i] = hi
            cur[j] = hj
    return best_h, best_val


# --------------------------------------------------------------
# Optional final SLSQP refinement (few iterations)
# --------------------------------------------------------------
def _refine_slsqp(h: np.ndarray, max_iter: int = 400) -> tuple:
    n = h.size
    h0 = _project_to_feasible(h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        hvec = x[:n]
        t = x[-1]
        return t - np.correlate(hvec, 1.0 - hvec, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": ineq},
                     {"type": "eq", "fun": eq}],
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        return h0, _c5_from_h(h0)
    h_opt = _project_to_feasible(res.x[:n], n)
    return h_opt, _c5_from_h(h_opt)


# --------------------------------------------------------------
# Main driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left(): return budget_s - (time.time() - start_time)

    # initial construction if supplied
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # candidate grid sizes (even only)
    cand_n = [256, 384, 512, 768, 1024]
    if init_h is not None:
        cand_n.append(init_h.size)
    cand_n = sorted(set([n for n in cand_n if n % 2 == 0 and n <= 1024]))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper to possibly update the global best
    # ------------------------------------------------------------------
    def _update(h, val, n):
        nonlocal best_h, best_c5, best_n
        if val < best_c5:
            best_h, best_c5, best_n = h.copy(), float(val), n

    # ------------------------------------------------------------------
    # Process each resolution
    # ------------------------------------------------------------------
    for idx, n in enumerate(cand_n):
        if time_left() < 5.0:
            break

        # build starting vector
        if init_h is not None and init_h.size == n:
            base = init_h.copy()
        elif best_h is not None:
            base = _upscale_h(best_h, n)
        else:
            # random half‑ones binary start
            base = np.zeros(n, dtype=float)
            base[: n // 2] = 1.0
            np.random.shuffle(base)
        base = _project_to_feasible(base, n)

        # decide annealing budget
        remaining = time_left()
        remaining_levels = len(cand_n) - idx
        iters = int(min(150000, max(20000, remaining * 2500 / remaining_levels)))
        iters = max(iters, 20000)

        # run annealing
        h_anneal, c5_anneal = _anneal_h(base, iters,
                                        temp_start=0.7, temp_end=5e-4,
                                        delta_start=0.4, delta_end=0.02)
        _update(h_anneal, c5_anneal, n)

        # optional short SLSQP refinement if time permits
        if time_left() > 12.0:
            h_ref, c5_ref = _refine_slsqp(h_anneal, max_iter=400)
            _update(h_ref, c5_ref, n)

        # early exit if target achieved
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final polishing on global best (if any time left)
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 10.0:
        h_final, c5_final = _refine_slsqp(best_h, max_iter=800)
        _update(h_final, c5_final, best_n)

    # fallback uniform if nothing found
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    else:
        best_h = _project_to_feasible(best_h, best_n)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
