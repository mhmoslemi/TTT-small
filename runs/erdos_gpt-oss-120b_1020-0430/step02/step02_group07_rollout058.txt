def compute_c5_fft(h):
    n = h.size
    f = h.astype(np.float64)
    g = (1 - h)[::-1].astype(np.float64)
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, size)
    G = np.fft.rfft(g, size)
    conv = np.fft.irfft(F * G, size)
    # conv length = size
    # we need first (2n-1) values
    conv = conv[: 2 * n - 1]
    # correlation index offset = n - 1
    max_corr = conv.max()
    # Multiply by dx = 2/n
    return max_corr * (2.0 / n)

def simulated_annealing(h_start, time_limit):
    rng = np.random.default_rng()
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = 0.02
    T = T0
    decay = 0.9999
    while time.time() < time_limit:
        # propose by transferring epsilon
        i, j = rng.integers(0,n,2)
        if i == j:
            continue
        # decide direction: move from i to j
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.5  # scale
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / T):
            cur_h, cur_c = new_h, new_c
        T *= decay
    return best_h, best_c

    if abs(diff) < 1e-12:
        return h
    # Clip active indices that can accept modification
    if diff > 0:
        # need to increase some entries not at 1
        idx = np.where(h < 1 - 1e-12)[0]
        if idx.size == 0:
            # nothing to adjust
            return h

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0≤z≤1, sum(z)=target} using a simple binary search."""
    lo = np.min(v) - 1.0      # at lo all entries clip to 1 → sum = len(v)
    hi = np.max(v)            # at hi all entries clip to 0 → sum = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(v - mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) (full, no scaling)."""
    n = h.size
    f = h.astype(np.float64)
    g = (1.0 - h)[::-1].astype(np.float64)
    size = _next_pow2(2 * n - 1)
    F = np.fft.rfft(f, size)
    G = np.fft.rfft(g, size)
    conv = np.fft.irfft(F * G, size)[: 2 * n - 1]
    return conv

def _c5_value(h: np.ndarray) -> float:
    """Exact C₅ for a step function represented by h."""
    n = h.size
    return _corr_fft(h).max() * (2.0 / n)

def _subgradient_optimize(h0: np.ndarray, target: float, time_limit: float):
    """Projected sub‑gradient descent on the true max‑overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_value(h)
    n = h.size
    step = 0.12               # initial step size
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = _corr_fft(h)
        idx = int(corr.argmax())
        shift = idx - (n - 1)
        # gradient for this shift
        grad = np.zeros_like(h)
        if shift >= 0:
            i = np.arange(0, n - shift)
            grad[i] += 1.0 - h[i + shift]
            grad[i + shift] -= h[i]
        else:
            s = -shift
            i = np.arange(0, n - s)
            grad[i + s] += 1.0 - h[i]
            grad[i] -= h[i + s]
        # projected step
        h = _project_capped_simplex(h - step * grad, target)
        # occasional best‑update
        if (it & 31) == 0:
            cur_c = _c5_value(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        # decay step size slowly
        step *= 0.9997
        it += 1
    return best_h, best_c

def _simulated_annealing(h_start: np.ndarray, target: float, time_limit: float):
    """Very light SA using epsilon transfers (preserves the sum)."""
    rng = np.random.default_rng()
    cur_h = h_start.copy()
    cur_c = _c5_value(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    T = 0.02
    decay = 0.9999
    start = time.time()
    while time.time() - start < time_limit:
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = _c5_value(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / T):
            cur_h, cur_c = new_h, new_c
        T *= decay
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initialise feasible h (use the supplied parent if present)
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = _project_capped_simplex(h0, target)
    else:
        n = 360
        target = n / 2.0
        h0 = rng.random(n)
        h0 = _project_capped_simplex(h0, target)

    best_h, best_c = h0.copy(), _c5_value(h0)

    # ----------------------------------------------------------------------
    # Phase 1 – deep sub‑gradient refinement (≈60 % of the budget)
    # ----------------------------------------------------------------------
    time_remaining = budget_s - (time.time() - start_time)
    phase1_time = min(time_remaining * 0.6, 600.0)
    refined_h, refined_c = _subgradient_optimize(best_h, target, phase1_time)
    if refined_c < best_c:
        best_h, best_c = refined_h, refined_c

    # ----------------------------------------------------------------------
    # Phase 2 – random restarts with short sub‑gradient bursts
    # ----------------------------------------------------------------------
    time_remaining = budget_s - (time.time() - start_time)
    while time_remaining > 5.0:
        # allocate a short burst (≈5 % of remaining time)
        burst = min(time_remaining * 0.05, 30.0)
        # random feasible start
        h_rand = rng.random(n)
        h_rand = _project_capped_simplex(h_rand, target)
        h_trial, c_trial = _subgradient_optimize(h_rand, target, burst)
        if c_trial < best_c:
            best_h, best_c = h_trial, c_trial
        time_remaining = budget_s - (time.time() - start_time)

    # ----------------------------------------------------------------------
    # Phase 3 – final simulated annealing clean‑up (remaining budget)
    # ----------------------------------------------------------------------
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 1.0:
        h_sa, c_sa = _simulated_annealing(best_h, target, time_remaining)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ----------------------------------------------------------------------
    # Final feasibility enforcement and return
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target)
    final_c = _c5_value(best_h)

    return best_h.tolist(), float(final_c), int(n)
