def compute_c5_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # compute FFT of h and reversed complement
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(H * G, L)[:2*n-1]
    max_corr = conv.max()
    return max_corr * (2.0 / n), conv

def compute_c5_and_corr(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(H * G, L)[:2*n-1]
    max_corr = conv.max()
    c5 = max_corr * (2.0 / n)
    return c5, conv

def hill_climb_float(h_start, rng, max_iters=1000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        # perturb h by moving epsilon from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step

        new_h[i] -= eps

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Global placeholders for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


# ------------------------------------------------------------
# Utility functions
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def compute_c5_and_corr(h):
    """Return (C5, full linear correlation) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(H * G, L)[: 2 * n - 1]
    max_corr = conv.max()
    c5 = max_corr * (2.0 / n)
    return c5, conv


def compute_c5(h):
    """C5 for a step function (float or binary)."""
    c5, _ = compute_c5_and_corr(h)
    return c5


def random_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=int)
    arr[rng.choice(n, n // 2, replace=False)] = 1
    return arr

# Binary simulated‑annealing / hill‑climb
def binary_sa(h0, rng, max_iters=2000, init_temp=0.5, decay=0.9995):
    """SA on binary vectors (swap a 1 with a 0)."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    n = h.size
    for _ in range(max_iters):
        # current worst lag
        _, conv = compute_c5_and_corr(h)
        idx = conv.argmax()
        k = idx - (n - 1)

        # indices that contribute to the worst lag
        if k >= 0:
            i_range = np.arange(0, n - k)
            i_vals = h[i_range]
            j_vals = h[i_range + k]
        else:
            off = -k
            i_range = np.arange(off, n)
            i_vals = h[i_range]
            j_vals = h[i_range + k]
        mask = (i_vals == 1) & (j_vals == 0)
        if np.any(mask):
            i_swap = i_range[mask][rng.integers(0, mask.sum())]
        else:
            i_swap = rng.choice(np.where(h == 1)[0])

        j_swap = rng.choice(np.where(h == 0)[0])

        h_new = h.copy()
        h_new[i_swap] = 0
        h_new[j_swap] = 1
        new_c = compute_c5(h_new)

        if new_c < best_c:
            best_c, best_h = new_c, h_new.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = h_new, new_c
        temp *= decay
    return best_h, best_c


def upsample_linear(h, factor, rng):
    """Linear up‑sampling by integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = (n * factor) / 2.0
    h_new = enforce_sum(h_new, target, rng)
    return h_new

        # decide start vector
        if best_h_bin is None:
            h_start = h0_bin if h0_bin is not None else random_binary(n, rng)
        else:
            if rng.random() < 0.1:
                h_start = best_h_bin.copy()
            else:
                h_start = random_binary(n, rng)

        remaining = budget_s - (time.time() - start_time)
        # at least a few hundred iterations, more if time permits
        max_iters = int(min(4000, max(500, remaining * 8)))
        cand_h, cand_c = binary_sa(h_start, rng, max_iters=max_iters,
                                   init_temp=0.5, decay=0.9995)
        if cand_c < best_c_bin:
            best_c_bin, best_h_bin = cand_c, cand_h.copy()
        # early exit if we already beat the target
        if best_c_bin < 0.3808:
            break

    # ---- up‑sample and continuous refinement (remaining budget) ----
    time_used = time.time() - start_time
    time_left = budget_s - time_used

    if best_h_bin is None:  # safety fallback
        best_h_bin = random_binary(n, rng)

    final_h = best_h_bin.astype(float)
    final_c = best_c_bin
    n_final = n

    # try a linear up‑sample if we have enough time
    if time_left > budget_s * 0.25 and n * 2 <= 800:
        h_up = upsample_linear(final_h, factor=2, rng=rng)
        h_up_ref, c_up_ref = quick_refine_float(h_up, rng)
        if c_up_ref < final_c:
            final_h, final_c, n_final = h_up_ref, c_up_ref, h_up_ref.size
    else:
        # a quick float hill‑climb on the binary vector can still help
        h_hc, c_hc = hill_climb_float(final_h, rng, max_iters=2000, start_step=0.3)
        if c_hc < final_c:
            final_h, final_c = h_hc, c_hc

    # final polishing loop (up to ~97 % of budget)
    while time.time() - start_time < budget_s * 0.97:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5:
            break
        # longer hill‑climb for refinement
        h2, c2 = hill_climb_float(final_h, rng, max_iters=4000, start_step=0.2)
        if c2 < final_c:
            final_h, final_c = h2, c2
        # occasional extra up‑sampling (factor 2) if still time and size permits
        if final_h.size * 2 <= 800 and time_left > 20:
            h_big = upsample_linear(final_h, factor=2, rng=rng)
            h_big_ref, c_big_ref = quick_refine_float(h_big, rng)
            if c_big_ref < final_c:
                final_h, final_c = h_big_ref, c_big_ref

    # enforce exact feasibility before returning
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = enforce_sum(final_h, final_h.size / 2.0, rng)
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(final_h.size)
