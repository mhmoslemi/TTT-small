    low = -1.0
    high = 1.0 + np.max(v)  # ensures sum at low > target, at high = 0 < target.

            return x
        if s > target:
            low = mid
        else:
            high = mid

def _corr_full(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 1 << ((2*n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)
    return conv[:2*n-1]
```

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Projection onto box [0,1] with exact sum constraint (∑h = target)
# ----------------------------------------------------------------------
def _proj_box_sum(v: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    low = -1.0               # sum will be n (>target)
    high = 1.0 + np.max(v)   # sum will be 0 (<target)
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    mid = (low + high) * 0.5
    return np.clip(v - mid, 0.0, 1.0)


def _project_to_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=len(h)/2."""
    n = h.size
    return _proj_box_sum(np.clip(h, 0.0, 1.0), n / 2.0)


# ----------------------------------------------------------------------
# Correlation (full) using FFT – linear convolution of h with rev(1-h)
# ----------------------------------------------------------------------
def _full_corr(h: np.ndarray) -> np.ndarray:
    n = h.size
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)
    return conv[: 2 * n - 1]


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible step function h."""
    n = h.size
    corr = _full_corr(h)
    return np.max(corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Gradient of the active (max) correlation lag
# ----------------------------------------------------------------------
def _grad_max_corr(h: np.ndarray, lag: int) -> np.ndarray:
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        # region where i and i+lag both inside
        i1 = slice(0, n - lag)
        g[i1] += 1.0 - h[i1 + lag]
        i2 = slice(lag, n)
        g[i2] -= h[i2 - lag]
    else:
        l = -lag
        i1 = slice(l, n)
        g[i1] += 1.0 - h[i1 - l]
        i2 = slice(0, n - l)
        g[i2] -= h[i2 + l]
    return g


# ----------------------------------------------------------------------
# Sub‑gradient descent on the max‑correlation objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    h0: np.ndarray, max_iter: int = 4000, alpha0: float = 0.6
) -> tuple:
    """Return (best_h, best_c5) after sub‑gradient descent."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    for t in range(1, max_iter + 1):
        corr = _full_corr(h)
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        # current C5
        cur_c5 = corr[idx] * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # gradient w.r.t. the active lag
        g = _grad_max_corr(h, lag)
        step = alpha0 / np.sqrt(t)
        h = _proj_box_sum(h - step * g, target)
    # final check
    final_c5 = _c5_from_h(best_h)
    return best_h, final_c5


# ----------------------------------------------------------------------
# Simple SLSQP wrapper (as in the previous solution)
# ----------------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    """Single SLSQP optimisation starting from initial_h (feasible or not)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h)
    t0 = np.max(_full_corr(h0))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        corr = _full_corr(h)
        return t - corr  # t >= all correlations

    def _eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Upscaling / interpolation
# ----------------------------------------------------------------------
def _upscale(h: np.ndarray, n_new: int) -> np.ndarray:
    n_old = h.size
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)
