def _solve_slsqp(h0, max_iter=400):
    n = h0.size
    h0 = project_feasible(h0, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {'type': 'ineq', 'fun': lambda x: x[-1] - np.correlate(x[:n], 1.0 - x[:n], mode='full')},
        {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n/2.0}
    ]
    res = scipy.optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=constraints,
                                  options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_feasible(h_opt, n)
    return h_opt, c5_val(h_opt)
```

def _de_optimize(m, n_fine, max_iter, seed):
    # precompute x arrays
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    x_coarse = (np.arange(m) + 0.5) * (2.0 / m)

    def objective(a):
        h = np.interp(x_fine, x_coarse, a)
        h = project_feasible(h, n_fine)
        return c5_val(h)

    bounds = [(0.0, 1.0)] * m
    result = scipy.optimize.differential_evolution(objective, bounds, maxiter=max_iter, popsize=12,
                                                  polish=False, seed=seed, updating='deferred')
    # Build fine h
    a_opt = result.x
    h_opt = np.interp(x_fine, x_coarse, a_opt)
    h_opt = project_feasible(h_opt, n_fine)
    return h_opt, c5_val(h_opt)
```

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce Σh = n/2 (integral = 1) by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a step function h (already feasible)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _solve_slsqp(h_start: np.ndarray, max_iter: int = 500) -> tuple:

        return x[-1] - np.correlate(x[:n], 1.0 - x[:n], mode="full")

        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n]
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


def _de_optimize(m: int, n_fine: int, max_iter: int, seed: int):
    """Differential Evolution on a coarse block representation."""
    # pre‑compute sampling points
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    x_coarse = (np.arange(m) + 0.5) * (2.0 / m)

    def obj(a):
        h = np.interp(x_fine, x_coarse, a)
        h = _project_to_feasible(h, n_fine)
        return _c5_from_h(h)

    bounds = [(0.0, 1.0)] * m
    result = opt.differential_evolution(
        obj,
        bounds,
        maxiter=max_iter,
        popsize=12,
        polish=False,
        seed=seed,
        updating="deferred",
        disp=False,
    )
    a_opt = result.x
    h_opt = np.interp(x_fine, x_coarse, a_opt)
    h_opt = _project_to_feasible(h_opt, n_fine)
    return h_opt, _c5_from_h(h_opt)
