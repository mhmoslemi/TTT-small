def compute_c5_bin(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    conv = np.fft.irfft(H * np.conj(H), L)
    # take first 2*n-1 elements
    C11 = conv[:2*n-1]
    O = n/2 - C11
    max_O = O.max()
    return max_O * (2.0 / n)

def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0

    # objective
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    x = np.clip(res.x, 0.0, 1.0)
    # final projection
    x = project_onto_box_sum(x, target)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
def compute_c5(h):
    """Exact C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_balanced_binary(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1
    return h


def compute_c5_bin(h):
    """C5 for a binary (0/1) vector using FFT (fast)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    hf = h.astype(np.float64)
    a_fft = np.fft.rfft(hf, L)
    b_fft = np.fft.rfft((1.0 - hf)[::-1], L)
    conv = np.fft.irfft(a_fft * b_fft, L)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def sa_optimize_binary(n, max_iters, rng, time_limit=None, start_h=None):
    """Simulated annealing on binary vectors (exact half ones)."""
    if start_h is not None:
        h = start_h.copy()
    else:
        h = random_balanced_binary(n, rng)

    cur_c = compute_c5_bin(h)
    best_h, best_c = h.copy(), cur_c

    T_start, T_end = 0.5, 1e-6
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)

        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if not ones_idx.size or not zeros_idx.size:
            break

        i1 = rng.choice(ones_idx)
        i2 = rng.choice(zeros_idx)

        h_new = h.copy()
        h_new[i1] = 0
        h_new[i2] = 1

        new_c = compute_c5_bin(h_new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h, cur_c = h_new, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using a soft‑max surrogate of the max‑overlap."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, target)
    return x


def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()

            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:] -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:] -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)

            if c_new < best_c:
                best_c, best_h, h = c_new, new_h.copy(), new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c
