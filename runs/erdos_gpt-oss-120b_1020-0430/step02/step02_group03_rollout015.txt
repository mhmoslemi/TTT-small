def project_onto_bounds_and_sum(v, target, eps=1e-10, max_iter=200):
    low = -np.max(v) - 1.0   # ensure all v_i+low <=0
    high = 1 - np.min(v) + 1.0  # ensure all v_i+high >= 1.
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= eps:
            return h
        if s < target:
            low = mid
        else:
            high = mid
    # after loop, return final h
    h = np.clip(v + (low+high)/2.0, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------------------------------------------------------
#  Helper functions
# ---------------------------------------------------------

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_bounds_and_sum(v, target, eps=1e-9, max_iter=200):
    """
    Projection onto the set {0<=x_i<=1, sum x_i = target}.
    Solves for λ in  Σ_i clip(v_i+λ,0,1) = target  (monotone in λ).
    """
    low = -np.max(v) - 1.0          # guarantees all clip to 0
    high = 1.0 - np.min(v) + 1.0    # guarantees all clip to 1
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s < target:
            low = mid
        else:
            high = mid
    # final fallback
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector projected onto feasible simplex with bounds."""
    v = rng.random(n)
    target = n / 2.0
    return project_onto_bounds_and_sum(v, target)


def softmax_obj_and_grad(h, beta):
    """
    Smooth surrogate of max_{k} Σ h_i (1 - h_{i+k}) and its gradient.
    Uses log-sum-exp on the correlation values.
    """
    n = h.size
    sum_h = h.sum()
    # auto‑correlation O_k = Σ_i h_i h_{i+k}
    O = np.correlate(h, h, mode="full")
    c = sum_h - O                     # c_k = Σ h_i - O_k
    # log‑sum‑exp in a numerically stable way
    bc = beta * c
    max_bc = np.max(bc)
    exp_terms = np.exp(bc - max_bc)
    Z = exp_terms.sum()
    w = exp_terms / Z                 # weights, sum to 1
    # objective value
    f = (np.log(Z) + max_bc) / beta

    # gradient: ∇f = Σ w_k ∇c_k  ,  ∇c_k = 1 - (h_{i+k}+h_{i-k})
    term = np.zeros(n)
    shift_offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - shift_offset
        if shift == 0:
            term += 2.0 * wk * h
        elif shift > 0:
            term[: n - shift] += wk * h[shift:]
            term[shift:]      += wk * h[: n - shift]
        else:  # shift < 0
            s = -shift
            term[s:]         += wk * h[: n - s]
            term[: n - s]    += wk * h[s:]
    grad = np.ones(n) - term
    return f, grad


def optimize_h_adam(h0, beta_seq, steps_per_beta, lr, rng):
    """Projected Adam optimiser on the smooth surrogate."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for beta in beta_seq:
        for _ in range(steps_per_beta):
            t += 1
            _, g = softmax_obj_and_grad(h, beta)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)
            m_hat = m / (1.0 - 0.9 ** t)
            v_hat = v / (1.0 - 0.999 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)
            h_new = h - step
            h = project_onto_bounds_and_sum(h_new, target)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_improve(h_start, rng, max_iters=1500, start_step=0.3):
    """Simple hill‑climb (accept only improving moves)."""
    n = h_start.size
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(best_h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def slsqp_refine(h0, beta, rng):
    """One‑shot SLSQP refinement using analytic gradient."""
    n = h0.size
    target = n / 2.0

    def fun(x):
        f, _ = softmax_obj_and_grad(x, beta)
        return f

    def jac(x):
        _, g = softmax_obj_and_grad(x, beta)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        fun,
        h0,
        method="SLSQP",
        jac=jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final projection onto the exact feasible set
    x = project_onto_bounds_and_sum(x, target)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---------------------------------------
    # Determine dimension and target integral
    # ---------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_bounds_and_sum(h0, n / 2.0)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0

    # ---------------------------------------
    # Build initial pool of candidates
    # ---------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    for f in (1, 2, 3):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(project_onto_bounds_and_sum(np.clip(h, 0.0, 1.0), target))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(project_onto_bounds_and_sum(np.clip(h, 0.0, 1.0), target))

    # linear ramps
    h = xs / 2.0
    candidates.append(project_onto_bounds_and_sum(np.clip(h, 0.0, 1.0), target))
    h = 1.0 - xs / 2.0
    candidates.append(project_onto_bounds_and_sum(np.clip(h, 0.0, 1.0), target))

    # random vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ---------------------------------------
    # Optimisation parameters
    # ---------------------------------------
    beta_seq = [5.0, 20.0, 80.0, 200.0, 500.0]
    steps_per_beta = 80          # 80 gradient steps per β
    lr = 0.25                    # Adam base step size

    best_h = None
    best_c = float("inf")

    # ---------------------------------------
    # Optimise each candidate (time‑capped)
    # ---------------------------------------
    for h_start in candidates:
        if time.time() - start_time > budget_s * 0.65:
            break
        h_opt = optimize_h_adam(h_start, beta_seq, steps_per_beta, lr, rng)
        # optional local hill‑climb for fine tuning
        h_opt, _ = hill_climb_improve(h_opt, rng, max_iters=800, start_step=0.2)
        # final SLSQP refinement with highest β
        h_opt = slsqp_refine(h_opt, beta_seq[-1], rng)
        c_val = compute_c5(h_opt)
        if c_val < best_c:
            best_c, best_h = c_val, h_opt.copy()

    # ---------------------------------------
    # Main stochastic loop (restarts + refinement)
    # ---------------------------------------
    while time.time() - start_time < budget_s * 0.96:
        # generate start point
        if best_h is not None and rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.25)
        else:
            start = random_feasible_vector(n, rng)
        # Adam optimisation
        h_opt = optimize_h_adam(start, beta_seq, steps_per_beta, lr, rng)
        # brief hill‑climb
        h_opt, _ = hill_climb_improve(h_opt, rng, max_iters=500, start_step=0.15)
        # SLSQP refinement
        h_opt = slsqp_refine(h_opt, beta_seq[-1], rng)
        c_val = compute_c5(h_opt)
        if c_val < best_c:
            best_c, best_h = c_val, h_opt.copy()
        # occasional up‑sampling if time permits
        if time.time() - start_time > budget_s * 0.85 and n * 2 <= 300:
            # double resolution of the current best
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            target_up = n_up / 2.0
            h_up = project_onto_bounds_and_sum(h_up, target_up)
            beta_seq_up = beta_seq
            h_up_opt = optimize_h_adam(
                h_up, beta_seq_up, steps_per_beta, lr, rng
            )
            h_up_opt, _ = hill_climb_improve(
                h_up_opt, rng, max_iters=800, start_step=0.15
            )
            h_up_opt = slsqp_refine(h_up_opt, beta_seq_up[-1], rng)
            c_up = compute_c5(h_up_opt)
            if c_up < best_c:
                best_c, best_h, n, target = c_up, h_up_opt.copy(), n_up, target_up
                # reset parameters for finer grid
                steps_per_beta = max(30, steps_per_beta - 10)
            break  # after up‑sampling we exit main loop

    # ---------------------------------------
    # Final clean‑up
    # ---------------------------------------
    if best_h is None:          # fallback (should never happen)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
