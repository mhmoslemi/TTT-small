def subgrad_descent(h0, rng, max_iter=500, step0=0.5):
    h = h0.copy()
    target = h.size/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        C = corr * (2.0 / h.size)
        max_c = np.max(C)
        active = np.isclose(C, max_c, atol=1e-12)
        # number of active shifts
        num_active = np.count_nonzero(active)
        if num_active == 0:
            break
        w = np.zeros_like(C)
        w[active] = 1.0 / num_active
        # gradient raw
        g_raw = np.ones(h.size)
        for idx, wk in enumerate(w):
            if wk == 0:
                continue
            k = idx - (h.size - 1)
            if k >= 0:
                g_raw[: h.size - k] -= wk * h[k:]
                g_raw[k:]      -= wk * h[: h.size - k]
            else:
                kk = -k
                g_raw[kk:]    -= wk * h[: h.size - kk]
                g_raw[: h.size - kk] -= wk * h[kk:]
        grad = (2.0 / h.size) * g_raw
        h_new = h - step * grad
        # symmetrize & project
        h_new = (h_new + h_new[::-1]) * 0.5
        h_new = project_onto_box_sum(h_new, target)
        c_new = compute_c5(h_new)
        # adapt step: if improved, maybe keep step else reduce

            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.02, 0.5)
        else:
            # occasional uphill acceptance
            if rng.random() < 0.01:
                h = h_new
            step *= 0.98

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    C = corr * (2.0 / x.size)
    bc = CURRENT_BETA * C
    m = np.max(bc)
    # compute log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities ------------------- #
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto the box [lower,upper] while fixing the sum = target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Return a random feasible symmetric vector of length n."""
    if n % 2 == 0:
        half = n // 2
        target_half = n / 4.0
        x = rng.random(half)
        x = project_onto_box_sum(x, target_half)
        h = np.concatenate([x, x[::-1]])
    else:
        half = n // 2
        target_half = n / 4.0
        x = rng.random(half)
        x = project_onto_box_sum(x, target_half)
        cen = rng.random()
        h = np.concatenate([x, [cen], x[::-1]])
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, n / 2.0)
    h = (h + h[::-1]) * 0.5
    h = project_onto_box_sum(h, n / 2.0)
    return h


def perturb_h(h, rng, scale=0.2):
    """Move a random epsilon from i to j keeping the sum constant."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Simple pairwise hill‑climb with occasional step reduction."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, target)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 7) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ------------------- smooth‑max gradient descent ------------------- #
def grad_descent_smooth(h0, rng, beta_sched, steps_per_beta):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    target = n / 2.0
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            s = w.sum()
            if s > 0:
                w /= s
            else:
                w[:] = 0.0

            g_raw = np.ones(n)
            # vectorised slice updates (fast enough for n ≤ 300)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:] -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:] -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            h_new = h - step * grad
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = project_onto_box_sum(h_new, target)
            c_new = compute_c5(h_new)

            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                step = min(step * 1.02, 0.5)
            else:
                if rng.random() < 0.02:
                    h = h_new
                step *= 0.99
    return best_h, best_c

def subgrad_descent(h0, rng, max_iter=800, step0=0.5):
    """Projected sub‑gradient descent on the exact max‑overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    target = n / 2.0
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        C = corr * (2.0 / n)
        max_c = np.max(C)
        # treat all maximizing shifts as active
        active = np.isclose(C, max_c, atol=1e-12)
        num_active = active.sum()
        if num_active == 0:
            break
        w = np.zeros_like(C)
        w[active] = 1.0 / num_active

        g_raw = np.ones(n)
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            k = idx - (n - 1)
            if k >= 0:
                g_raw[: n - k] -= wk * h[k:]
                g_raw[k:] -= wk * h[: n - k]
            else:
                kk = -k
                g_raw[kk:] -= wk * h[: n - kk]
                g_raw[: n - kk] -= wk * h[kk:]
        grad = (2.0 / n) * g_raw
        h_new = h - step * grad
        h_new = (h_new + h_new[::-1]) * 0.5
        h_new = project_onto_box_sum(h_new, target)
        c_new = compute_c5(h_new)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.02, 0.5)
        else:
            if rng.random() < 0.01:
                h = h_new
            step *= 0.98
    return best_h, best_c


# ------------------- SLSQP refinement (soft‑max) ------------------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def _eq_sum(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    C = corr * (2.0 / x.size)
    bc = CURRENT_BETA * C
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP refinement iteration using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _eq_sum}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
