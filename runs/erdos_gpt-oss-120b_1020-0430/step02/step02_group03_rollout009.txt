    low = -np.max(x) - 1.0

import numpy as np
import time

def project_onto_sum_bounds(x, target):
    # binary search lambda
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    proj = np.clip(x - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # simple adjustment
        for i in range(len(proj)):
            if diff > 0:
                add = min(diff, 1.0 - proj[i])
            else:
                add = max(diff, -proj[i])
            proj[i] += add
            diff = target - proj.sum()
            if abs(diff) <= 1e-12:
                break
    return proj
```

```
def max_corr(h):
    n = h.size
    L = 1
    while L < 2*n-1:
        L <<= 1
    rev = (1 - h)[::-1]
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    c = np.fft.irfft(H * R, n=L)[:2*n-1]
    return c.max(), c.argmax()
```

def compute_c5_from_h(h):

import time
import numpy as np

def _project_sum_bounds(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n with Σx = target using a bisection on the shift λ."""
    low = -np.max(x) - 1.0          # makes x-λ > 1 for all entries → sum = n
    high = 1.0 - np.min(x) + 1.0   # makes x-λ < 0 for all entries → sum = 0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    proj = np.clip(x - lam, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        for i in range(proj.size):
            if diff > 0.0:
                add = min(diff, 1.0 - proj[i])
            else:
                add = max(diff, -proj[i])
            proj[i] += add
            diff = target - proj.sum()
            if abs(diff) <= 1e-12:
                break
    return proj


def _max_corr(h: np.ndarray):
    """Return (max_correlation, index_of_max) for h with (1‑h) using FFT."""
    n = h.size
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _c5_from_h(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2]."""
    maxc, _ = _max_corr(h)
    return maxc * (2.0 / h.size)


def _c5_from_v(v: np.ndarray) -> float:
    """C5 using the symmetric extension of v (full h = [v, rev(v)])."""
    h = np.concatenate([v, v[::-1]])
    return _c5_from_h(h)


def _gradient_best_shift(v: np.ndarray):
    """Sub‑gradient of the worst‑shift correlation w.r.t. the half‑vector v."""
    h = np.concatenate([v, v[::-1]])
    n = h.size
    _, idx = _max_corr(h)
    t = idx - (n - 1)                     # shift offset (positive → right)
    if t >= 0:
        term1 = np.concatenate([np.zeros(t), 1.0 - h[: n - t]])
        term2 = np.concatenate([h[t:], np.zeros(t)])
    else:
        l = -t
        term1 = np.concatenate([np.zeros(l), 1.0 - h[: n - l]])
        term2 = np.concatenate([h[l:], np.zeros(l)])
    g_full = term1 - term2                # gradient for full h
    m = v.size
    # each v_i appears twice symmetrically
    g_v = (g_full[:m] + g_full[-1 : -(m + 1) : -1]) * 0.5
    return g_v


def _subgradient_optimize(v0: np.ndarray, target: float,
                          time_limit: float, start_time: float,
                          rng: np.random.Generator):
    """Gradient‑search with stochastic restarts; returns best (v, C5)."""
    best_v = v0.copy()
    best_c = _c5_from_v(best_v)
    cur_v = v0.copy()
    cur_c = best_c
    last_imp = 0
    it = 0
    while True:
        if time.time() - start_time >= time_limit:
            break
        # deterministic sub‑gradient step
        g = _gradient_best_shift(cur_v)
        step = 0.5 / np.sqrt(it + 1)
        cand_v = cur_v - step * g
        cand_v = _project_sum_bounds(cand_v, target)
        cand_c = _c5_from_v(cand_v)

        # acceptance
        if cand_c < cur_c:
            cur_v, cur_c = cand_v, cand_c
            if cand_c < best_c - 1e-12:
                best_v, best_c = cand_v.copy(), cand_c
                last_imp = it
        else:
            # Metropolis criterion with a temperature that cools to ~0
            remaining = max(1e-12, time_limit - (time.time() - start_time))
            T = remaining / time_limit
            if rng.random() < np.exp(-(cand_c - cur_c) / max(T, 0.001)):
                cur_v, cur_c = cand_v, cand_c

        # occasional random jitter to escape plateaus
        if it % 1000 == 0 and rng.random() < 0.2:
            cur_v = _project_sum_bounds(cur_v + rng.normal(scale=0.01, size=v0.shape), target)
            cur_c = _c5_from_v(cur_v)

        # restart if stagnant for many iterations
        if it - last_imp > 4000:
            cur_v = _project_sum_bounds(best_v + rng.normal(scale=0.05, size=v0.shape), target)
            cur_c = _c5_from_v(cur_v)
            last_imp = it

        it += 1
    return best_v, best_c
