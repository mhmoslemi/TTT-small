def project_onto_capped_simplex(v, target, ub=1.0, tol=1e-12, max_iter=100):
    # v is original array
    lo = np.min(v - ub) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, ub)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, ub)

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def project_onto_capped_simplex(v: np.ndarray, target: float, ub: float = 1.0,
                               tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x : 0 <= x_i <= ub, sum(x)=target}.
    Uses a bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0   # ensures v - lo >= ub
    hi = np.max(v) + 1.0    # ensures v - hi <= 0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, ub)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, ub)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    raw = rng.random(n)
    return project_onto_capped_simplex(raw, n / 2.0, ub=1.0)


def gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the overlap for a given integer shift.
    shift can be positive, zero, or negative.
    """
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        upper = n - k
        if upper > 0:
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
    else:
        k = -shift
        upper = n - k
        if upper > 0:
            g[k:] += 1.0 - h[:upper]
            g[:upper] -= h[k:]
    return g


def subgradient_descent(h0: np.ndarray, rng: np.random.Generator,
                        max_iter: int = 5000, step0: float = 0.5) -> tuple:
    """
    Projected sub‑gradient descent on the max‑overlap objective.
    Returns (best_h, best_c5) where best_c5 is the *unscaled* max correlation.
    """
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        maxval = np.max(corr)
        if maxval < best_corr:
            best_corr = maxval
            best_h = h.copy()
        # indices achieving the maximum (allow tiny tolerance)
        idxs = np.where(corr >= maxval - 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n - 1)
            grad += gradient_of_shift(h, shift)
        if len(idxs) > 1:
            grad /= len(idxs)
        # step size schedule
        step = step0 / np.sqrt(it + 1.0)
        h -= step * grad
        # projection back to feasible set
        h = project_onto_capped_simplex(h, target, ub=1.0)
        # occasional jitter to escape plateaus
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            eps = (rng.random() - 0.5) * 0.08
            h[i] = np.clip(h[i] + eps, 0.0, 1.0)
            h[j] = np.clip(h[j] - eps, 0.0, 1.0)
            h = project_onto_capped_simplex(h, target, ub=1.0)
    return best_h, best_corr * (2.0 / n)


def perturb_transfer(h: np.ndarray, rng: np.random.Generator,
                    scale: float = 0.3) -> np.ndarray:
    """Transfer a small amount ε between two coordinates, preserving the sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.5) -> tuple:
    """Simple deterministic hill‑climb (accept only strict improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_transfer(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def binarize_top_half(h: np.ndarray) -> np.ndarray:
    """Create a 0/1 vector keeping the indices with largest values (exactly n/2 ones)."""
    n = h.size
    ones = n // 2
    idx = np.argpartition(-h, ones)[:ones]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out
