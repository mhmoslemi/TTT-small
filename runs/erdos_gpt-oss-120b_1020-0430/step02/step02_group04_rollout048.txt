def grad_smooth_max(full, beta):

    corr = np.correlate(full, 1-full, mode='full')

            g_raw[: N - k] -= wk * full[k:]
            g_raw[k:]      -= wk * full[: N - k]
        else:
            kk = -k
            g_raw[kk:]    -= wk * full[: N - kk]
            g_raw[: N - kk] -= wk * full[kk:]
    grad = (2.0 / N) * g_raw
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# basic utilities
def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while forcing sum(v)=target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def compute_c5(h):
    """Maximum overlap for a full symmetric step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sym_from_half(half):
    """Create a full symmetric array from its first half."""
    return np.concatenate([half, half[::-1]])


def compute_c5_half(half):
    """Maximum overlap using the half representation."""
    full = sym_from_half(half)
    return compute_c5(full)


def random_feasible_half(m, rng):
    target = m / 2.0
    x = rng.random(m)
    return project_onto_box_sum(x, target)


def perturb_half(half, rng, scale=0.2):
    """Transfer a small amount epsilon from i to j (keeps sum)."""
    m = half.size
    i = rng.integers(0, m)
    j = rng.integers(0, m)
    if i == j:
        return half
    max_eps = min(half[i], 1.0 - half[j])
    if max_eps <= 0.0:
        return half
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = half.copy()
    new[i] -= eps
    new[j] += eps
    return new

# smooth‑max gradient utilities (full representation)
def grad_smooth_max(full, beta):
    """Gradient of the beta‑softmax of overlaps w.r.t. full h."""
    N = full.size
    corr = np.correlate(full, 1 - full, mode="full")
    C = corr * (2.0 / N)                # scaled overlaps
    bc = beta * C
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= w.sum()
    g_raw = np.ones(N)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (N - 1)
        if k >= 0:
            g_raw[: N - k] -= wk * full[k:]
            g_raw[k:]      -= wk * full[: N - k]
        else:
            kk = -k
            g_raw[kk:]    -= wk * full[: N - kk]
            g_raw[: N - kk] -= wk * full[kk:]
    return (2.0 / N) * g_raw


def grad_descent_optimize_half(half_start, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth‑max surrogate (half domain)."""
    half = half_start.copy()
    best_half = half.copy()
    best_c = compute_c5_half(best_half)
    m = half.size
    target = m / 2.0
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            full = sym_from_half(half)
            grad_full = grad_smooth_max(full, beta)
            # combine symmetric contributions
            grad_half = grad_full[:m] + grad_full[-1:-m-1:-1]
            new_half = half - step * grad_half
            new_half = project_onto_box_sum(new_half, target)
            c_new = compute_c5_half(new_half)
            if c_new < best_c:
                best_c = c_new
                best_half = new_half.copy()
                half = new_half
            else:
                if rng.random() < 0.01:
                    half = new_half
            step *= 0.9995
    return best_half, best_c

# simulated annealing on half representation
def anneal_optimize_half(half_start, rng, time_budget, start_time):
    half = half_start.copy()
    best_half = half.copy()
    best_c = compute_c5_half(half)
    cur_c = best_c
    m = half.size
    target = m / 2.0
    total_iters = 500_000
    deadline = start_time + time_budget
    for it in range(total_iters):
        if time.time() >= deadline:
            break
        # exponential temperature from 0.02 down to 0.001
        t = 0.02 * ((0.001 / 0.02) ** (it / total_iters))
        # epsilon scale from 0.5 down to 0.02
        scale = 0.5 * ((0.02 / 0.5) ** (it / total_iters))
        new_half = perturb_half(half, rng, scale=scale)
        # projection only to correct tiny drift
        new_half = project_onto_box_sum(new_half, target)
        new_c = compute_c5_half(new_half)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(t, 1e-12)):
            half, cur_c = new_half, new_c
            if new_c < best_c:
                best_c, best_half = new_c, new_half.copy()
        # occasional diversification
        if (it + 1) % 100_000 == 0:
            noise = rng.normal(scale=0.02, size=m)
            trial = best_half + noise
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, target)
            trial_c = compute_c5_half(trial)
            if trial_c < cur_c:
                half, cur_c = trial, trial_c
                if trial_c < best_c:
                    best_c, best_half = trial_c, trial.copy()
    return best_half, best_c

# simple hill‑climb on full representation (keeps feasibility)
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# SLSQP refinement on the smooth‑max surrogate (full)
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Single SLSQP refinement using soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
