def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            # compute corr
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # compute gradient: earlier code used loops:
            g_raw = np.ones(n)
            # vectorized processing:
            # For each shift k, weight w_k applied.
            # Implementation using loops as earlier.
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    # Overlap region length n - k
                    # subtract wk * h[k:] from first part (g_raw[:n-k])
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            # gradient step
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n/2.0)
            new_h = (new_h + new_h[::-1]) * 0.5  # ensure symmetry

                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # maybe accept occasional uphill move?
                if rng.random() < 0.01:
                    h = new_h

    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 using linear correlation (FFT for speed)."""
    a = h.astype(np.float64)
    b = (1 - h).astype(np.float64)
    conv = fftconvolve(a, b[::-1], mode="full")
    return conv.max() * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto the box [lower,upper] while enforcing Σv = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Continuous feasible symmetric vector (0≤h≤1, Σh=n/2)."""
    m = n // 2
    half = rng.random(m)
    half = project_onto_box_sum(half, m / 2.0)
    full = np.concatenate([half, half[::-1]])
    full = project_onto_box_sum(full, n / 2.0)
    return full


# ---------- binary helpers ----------
def random_binary_half(m, rng):
    """Generate a binary half‑vector of length m with exactly m/2 ones."""
    half = np.zeros(m, dtype=np.float64)
    ones_needed = m // 2
    idx = rng.choice(m, size=ones_needed, replace=False)
    half[idx] = 1.0
    return half


def hill_climb_binary_half(half, rng, max_iters=4000):
    """Swap‑based hill climbing on a symmetric binary half‑vector."""
    m = half.size
    best_half = half.copy()
    best_c = compute_c5(np.concatenate([half, half[::-1]]))
    for _ in range(max_iters):
        ones = np.flatnonzero(best_half == 1)
        zeros = np.flatnonzero(best_half == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_half = best_half.copy()
        new_half[i] = 0.0
        new_half[j] = 1.0
        full = np.concatenate([new_half, new_half[::-1]])
        c = compute_c5(full)
        if c < best_c:
            best_c = c
            best_half = new_half
    return best_half, best_c


# ---------- continuous helpers ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb on continuous vectors (preserves symmetry)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, n / 2.0)
        # enforce symmetry
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth‑max gradient descent ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            new_h = (new_h + new_h[::-1]) * 0.5
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- SLSQP with soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
