def _subgradient_optimize(h, n_iters=4000, step0=0.6):
    n = h.size
    for t in range(n_iters):
        # compute correlation and max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # find indices where corr == max_corr (within tolerance)
        # Use tolerance small relative to values e.g., 1e-12
        idxs = np.where(corr >= max_corr - 1e-12)[0]
        # compute gradient
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                # part1
                grad[:n - k] += 1 - h[k:]
                # part2
                grad[k:] -= h[:n - k]
            else:
                kp = -k
                grad[kp:] += 1 - h[:n - kp]
                grad[:n - kp] -= h[kp:]
        # average if multiple idx
        if len(idxs) > 1:
            grad /= len(idxs)
        # step size
        alpha = step0 / np.sqrt(t + 1)
        # gradient descent step (minimize max_corr => subtract grad)
        h = h - alpha * grad
        # projection
        h = _project_to_feasible(h, n)
    return h

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Feasibility projection (water filling)
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# --------------------------------------------------------------
# Objective evaluation
# --------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# --------------------------------------------------------------
# Interpolation (coarse → fine)
# --------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# --------------------------------------------------------------
# SLSQP local optimisation (same formulation as the reference)
# --------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5


# --------------------------------------------------------------
# Simple sub‑gradient descent for the max‑correlation objective
# --------------------------------------------------------------
def _subgradient_optimize(h_start: np.ndarray, n_iters: int = 4000, step0: float = 0.6) -> np.ndarray:
    h = h_start.copy()
    n = h.size
    for t in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        idxs = np.where(corr >= max_corr - 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                # i in [0, n‑k)
                grad[: n - k] += 1.0 - h[k:]
                # i in [k, n)
                grad[k:] -= h[: n - k]
            else:
                kp = -k
                # i in [kp, n)
                grad[kp:] += 1.0 - h[: n - kp]
                # i in [0, n‑kp)
                grad[: n - kp] -= h[kp:]
        if len(idxs) > 1:
            grad /= len(idxs)
        alpha = step0 / np.sqrt(t + 1.0)
        h -= alpha * grad
        h = _project_to_feasible(h, n)
    return h
