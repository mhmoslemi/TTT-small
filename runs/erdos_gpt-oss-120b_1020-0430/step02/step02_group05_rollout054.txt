Define function compute_c5_fft(h):

    n = h.shape[0]
    # compute linear cross-correlation of h and (1-h) (full)
    a = h
    b = 1.0 - h
    # reverse b for correlation
    b_rev = b[::-1]
    # length for linear convolution = 2*n -1
    L = 1 << (2*n -1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(A * B, n=L)[:2*n-1]  # real
    max_corr = conv.max()
    return max_corr * (2.0 / n)

Thus each compute_c5_fft is O(L log L) with L near next pow2 of 2*n-1. For n=384, 2*n-1=767; next pow2 = 1024. So each evaluation ~ 1024*log2(1024)=1024*10=10k op plus overhead. That's fine.

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x_i≤1, sum x_i = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 using linear correlate (np.correlate)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _c5_fast(h: np.ndarray) -> float:
    """Fast C5 via FFT (same result, used in local search)."""
    n = h.shape[0]
    a = h
    b_rev = (1.0 - h)[::-1]
    L = 1 << (2 * n - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(A * B, n=L)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Run SLSQP from a feasible start."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse vector onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Tent shape min(x,2-x) projected onto the simplex."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _pairwise_refine(
    h: np.ndarray, max_iter: int = 3000, init_step: float = 0.2, final_step: float = 0.001
):
    """Local pairwise adjustments keeping sum fixed."""
    n = h.size
    cur = h.copy()
    cur_c = _c5_fast(cur)
    best = cur.copy()
    best_c = cur_c
    for it in range(max_iter):
        # step size annealing
        step = init_step * (final_step / init_step) ** (it / max_iter)
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = (np.random.rand() - 0.5) * 2.0 * step
        hi = cur[i] + delta
        hj = cur[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        cur[i] = hi
        cur[j] = hj
        new_c = _c5_fast(cur)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert
            cur[i] = hi - delta
            cur[j] = hj + delta
    best = _project_to_simplex_box(best, n / 2.0)
    return best, best_c
