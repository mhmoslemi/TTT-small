def hill_climb(h_start, rng, max_iters=2000, start_scale=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = start_scale
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_transfer(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it+1) % decay_interval == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_at_max_shift(h):
    """Gradient of the max‑overlap objective at the shift attaining the max."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_idx = corr.argmax()
    k = k_idx - (n - 1)          # shift value
    g = np.ones(n)
    if k >= 0:
        g[: n - k] -= h[k:]
    else:
        kk = -k
        g[kk:] -= h[: n - kk]
    if k <= 0:
        kk = -k
        g[: n - kk] -= h[kk:]
    else:
        g[k:] -= h[: n - k]
    return g, k


def gradient_hill_climb(h_start, rng, max_iters=1500, start_scale=0.4):
    """Hill‑climb using the sub‑gradient of the max‑overlap."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = start_scale
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        g, _ = gradient_at_max_shift(h)
        donor = int(np.argmax(g))
        receiver = int(np.argmin(g))
        max_eps = min(h[donor], 1.0 - h[receiver])
        if max_eps > 0.0:
            eps = min(max_eps, scale)
            new_h = h.copy()
            new_h[donor] -= eps
            new_h[receiver] += eps
        else:
            new_h = perturb_transfer(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c


def hill_climb(h_start, rng, max_iters=1500, start_scale=0.5):
    """Simple hill‑climb with random transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = start_scale
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_transfer(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.exp(bc - m).sum())) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=800):
    """Refine h using SLSQP on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    

def upsample(h):
    """Duplicate each entry – keeps the integral exact."""
    return np.repeat(h, 2)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        np.clip(h0, 0.0, 1.0, out=h0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 150
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = hill_climb(h0, rng, max_iters=2000, start_scale=0.5)

    # ----- main optimisation loop ------------------------------------
    beta_seq = [5.0, 20.0, 80.0, 200.0, 500.0]
    while time.time() - start_time < budget_s * 0.88:
        # starting point
        if rng.random() < 0.6:
            start = perturb_transfer(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # gradient guided hill‑climb
        h1, c1 = gradient_hill_climb(start, rng, max_iters=1500, start_scale=0.4)

        # finishing random hill‑climb
        h2, c2 = hill_climb(h1, rng, max_iters=1500, start_scale=0.3)

        # SLSQP ladder
        h_cur = h2
        for beta in beta_seq:
            if time.time() - start_time > budget_s * 0.88:
                break
            try:
                h_cur = slsqp_optimize(h_cur, beta, max_iter=800)
            except Exception:
                continue
        cur_c = compute_c5(h_cur)
        if cur_c < best_c:
            best_c, best_h = cur_c, h_cur.copy()

    # ----- optional up‑sampling ---------------------------------------
    if time.time() - start_time < budget_s * 0.96 and n * 2 <= 360:
        h_up = upsample(best_h)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)
        # coarse hill‑climb on the finer grid
        h_up, _ = gradient_hill_climb(h_up, rng, max_iters=2000, start_scale=0.3)
        h_up, _ = hill_climb(h_up, rng, max_iters=2000, start_scale=0.2)
        for beta in beta_seq:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, max_iter=800)
            except Exception:
                pass
        up_c = compute_c5(h_up)
        if up_c < best_c:
            best_c, best_h, n = up_c, h_up.copy(), n_up

    # ----- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
