def _max_corr_fast(h):
    n = len(h)
    N = 1 << (2*n - 1).bit_length()  # Next power of two >= 2n
    # But we can just set N = 2*n
    N = 2*n
    # Compute FFT of h and reversed (1-h)
    h_pad = np.zeros(N)
    h_pad[:n] = h
    rev = np.zeros(N)
    rev[:n] = (1 - h)[::-1]  # reversed
    H = np.fft.rfft(h_pad)
    R = np.fft.rfft(rev)
    conv = np.fft.irfft(H * R, n=N)
    # conv length N = 2n, but correlation for full length is conv[:2*n-1]
    max_corr = np.max(conv[:2*n-1])
    # rounding errors: max_corr may be slightly negative small; clip
    if max_corr < 0:
        max_corr = 0.0
    return max_corr

import time
import math
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Basic feasibility projection (box [0,1] + sum = n/2)
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9) -> np.ndarray:
    """Euclidean projection of h onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    # clip once to have a feasible starting point for the bisection
    h = np.clip(h, 0.0, 1.0)

    # Bisection on lambda so that sum(clip(h - lambda,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(h - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(h - lam, 0.0, 1.0)

    # tiny residual fix (should be extremely small)
    resid = target - x.sum()
    if abs(resid) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            x[free[0]] += resid
            x = np.clip(x, 0.0, 1.0)
    return x


# --------------------------------------------------------------
# Fast correlation / C5 evaluation (FFT based)
# --------------------------------------------------------------
def _max_corr_fast(h: np.ndarray) -> float:
    """Maximum of full correlation of h with (1‑h) using FFT."""
    n = h.size
    N = 2 * n
    h_pad = np.zeros(N, dtype=float)
    rev_pad = np.zeros(N, dtype=float)
    h_pad[:n] = h
    rev_pad[:n] = (1.0 - h)[::-1]
    H = np.fft.rfft(h_pad)
    R = np.fft.rfft(rev_pad)
    conv = np.fft.irfft(H * R, n=N)
    max_corr = np.max(conv[: 2 * n - 1])
    if max_corr < 0:
        max_corr = 0.0
    return max_corr


def _c5_fast(h: np.ndarray) -> float:
    """Fast C5 (uses _max_corr_fast)."""
    n = h.size
    return _max_corr_fast(h) * (2.0 / n)


def _c5_from_h(h: np.ndarray) -> float:
    """Accurate C5 using np.correlate (the official evaluator)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# --------------------------------------------------------------
# Local SLSQP refinement (same as previous program)
# --------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    """SLSQP local optimisation starting from a feasible vector."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

# Simple simulated‑annealing optimiser (real‑valued h)
# --------------------------------------------------------------
def _simulated_annealing(n: int, start_h: np.ndarray, time_limit: float):
    """Run SA for at most *time_limit* seconds, return best (h, C5)."""
    cur_h = start_h.copy()
    cur_c5 = _c5_fast(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    max_iters = int(min(200_000, time_limit * 400))  # rough 400 it/s estimate
    T0, Tmin = 0.02, 0.0005
    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = it / max_iters
        T = T0 * (1.0 - frac) + Tmin * frac
        sigma = 0.18 * (1.0 - frac) + 0.02

        proposal = cur_h + sigma * np.random.randn(n)
        proposal = _project_to_feasible(proposal, n)
        new_c5 = _c5_fast(proposal)

        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c5 = proposal, new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5

        # occasional random restart to avoid stagnation
        if np.random.rand() < 0.0004:
            cur_h = _project_to_feasible(np.random.rand(n), n)
            cur_c5 = _c5_fast(cur_h)

        # early exit if we already beat the target
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

        return budget_s - (time.time() - start_time)
