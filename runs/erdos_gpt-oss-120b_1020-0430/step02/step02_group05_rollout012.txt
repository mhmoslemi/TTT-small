def _c5_from_h(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return 2.0 / n * np.max(corr)

def _softmax_weights(corr, alpha):
    # stable softmax
    maxc = np.max(corr)
    exp = np.exp(alpha * (corr - maxc))
    exp_sum = exp.sum()
    return exp / exp_sum

def _grad_softmax_max_corr(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = _softmax_weights(corr, alpha)
    g = np.ones(n)
    # shifts from -(n-1) to n-1
    for idx, s in enumerate(range(-n+1, n)):
        ws = w[idx]
        if ws == 0.0:

        if s >= 0:
            # t + s part
            g[:n-s] -= ws * h[s:]

import time

# ---------- basic utilities ----------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)

# ---------- SLSQP refinement ----------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x: np.ndarray) -> float:
    return x[-1]

def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

# ---------- symmetry helpers ----------
def _v_to_h(v: np.ndarray) -> np.ndarray:
    """From half‑vector v (size m) build full h of size 2m with h[i]+h[2m-1-i]=1."""
    m = v.size
    n = m * 2
    h = np.empty(n, dtype=np.float64)
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h

def _grad_softmax_max_corr(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max_k ∫h(1−h(·+k))."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    # stable soft‑max
    maxc = np.max(corr)
    w = np.exp(alpha * (corr - maxc))
    w /= w.sum()
    g = np.ones(n)
    # shifts from -(n-1) … (n-1)
    for idx, s in enumerate(range(-n + 1, n)):
        ws = w[idx]
        if ws == 0.0:
            continue
        if s >= 0:
            g[:n - s] -= ws * h[s:]
            g[s:] -= ws * h[:n - s]
        else:
            ss = -s
            g[ss:] -= ws * h[:n - ss]
            g[:n - ss] -= ws * h[ss:]
    return g

def _grad_h_to_v(grad_h: np.ndarray) -> np.ndarray:
    n = grad_h.size
    m = n // 2
    return grad_h[:m] - grad_h[-1:-m - 1:-1]

    target_bound = 0.3808
    best_overall_h = None
    best_overall_c5 = float("inf")
    best_overall_n = None

    # candidate grid sizes (even numbers, limited to ≈500‑800)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (128, 192, 256, 320, 384, 448, 512, 640):
        if n % 2 != 0:
            n += 1
        n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 4.0:
            break
        h_n, c5_n = _optimise_at_n(n, init_h, time_left, target_bound)
        if h_n is not None and c5_n < best_overall_c5:
            best_overall_h, best_overall_c5, best_overall_n = h_n, c5_n, n
            if best_overall_c5 < target_bound:
                break

    # fallback simple construction if nothing succeeded
    if best_overall_h is None:
        n = 256
        h0 = _tent_h(n)
        best_overall_h, best_overall_c5 = _solve_h(h0, max_iter=500)
        best_overall_n = n

    return best_overall_h.tolist(), float(best_overall_c5), int(best_overall_n)
