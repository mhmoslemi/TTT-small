def reduce_max_corr(h, rng, max_iter):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        # compute corr and find argmax
        corr = np.correlate(h, 1 - h, mode='full')
        # shift index of max
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n-1)
        # Determine indices that contribute: where h[i]==1 and i+max_shift in range and (1-h[i+max_shift]) == 1 => h[i+max_shift]==0
        # For simplicity, produce list of bad ones and candidate zeros not causing contribution.
        bad_one_idxs = []
        if max_shift >= 0:
            for i in range(n - max_shift):
                if h[i] == 1 and h[i + max_shift] == 0:
                    bad_one_idxs.append(i)
        else:
            s = -max_shift
            for i in range(s, n):
                if h[i] == 1 and h[i - s] == 0: # shift left
                    bad_one_idxs.append(i)
        # If no bad ones (shouldn't happen when max corr >0)
        # Choose random bad i and a random zero j not in bad list, maybe also not causing new contributions?
        if not bad_one_idxs:
            # fallback random swap
            i = rng.choice(np.where(h == 1)[0])
            j = rng.choice(np.where(h == 0)[0])
        else:
            i = rng.choice(bad_one_idxs)
            # choose any zero index j not equal i
            j = rng.choice(np.where(h == 0)[0])
        # propose swap
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h
            h = new_h
        else:
            # maybe accept with probability
            if rng.random() < 0.01:
                h = new_h
    return best_h, best_c
```

def binary_hill_climb(h, rng, max_iter):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # Precompute indices of ones and zeros.
    for it in range(max_iter):
        # compute correlation and max shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (best_h.size - 1)
        # find candidates: bad ones that contribute to max shift.
        n = best_h.size
        if max_shift >= 0:
            # contributions where i in [0, n - max_shift - 1]
            i_vals = np.arange(n - max_shift)
            # boolean mask
            mask = (best_h[i_vals] == 1) & (best_h[i_vals + max_shift] == 0)
            bad_one_idxs = i_vals[mask]
        else:
            s = -max_shift
            i_vals = np.arange(s, n)
            mask = (best_h[i_vals] == 1) & (best_h[i_vals - s] == 0)
            bad_one_idxs = i_vals[mask]
        if bad_one_idxs.size == 0:
            # no contributions? continue random
            i = rng.choice(np.where(best_h == 1)[0])
        else:
            i = rng.choice(bad_one_idxs)
        # choose zero j not equal i
        zero_idxs = np.where(best_h == 0)[0]
        if zero_idxs.size == 0:
            break
        j = rng.choice(zero_idxs)

        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def project_onto_box_sum(v: np.ndarray, target: float,
                         lower: float = 0.0, upper: float = 1.0,
                         tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Project v onto [lower,upper]^n with exact sum = target (box‑sum projection)."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def binary_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                      max_iter: int = 15000) -> tuple[np.ndarray, float]:
    """Hill‑climb on a binary vector (0/1) preserving the weight."""
    cur = h0.astype(np.float64).copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    n = cur.size

    for _ in range(max_iter):
        # correlation and the worst shift
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # indices of ones that actually contribute to this shift
        if shift >= 0:
            i_vals = np.arange(n - shift)
            mask = (cur[i_vals] == 1) & (cur[i_vals + shift] == 0)
            bad = i_vals[mask]
        else:
            s = -shift
            i_vals = np.arange(s, n)
            mask = (cur[i_vals] == 1) & (cur[i_vals - s] == 0)
            bad = i_vals[mask]

        if bad.size == 0:
            # fallback: any one
            i = rng.choice(np.where(cur == 1)[0])
        else:
            i = rng.choice(bad)

        zero_idxs = np.where(cur == 0)[0]
        if zero_idxs.size == 0:
            break
        j = rng.choice(zero_idxs)

        # propose swap
        new = cur.copy()
        new[i] = 0.0
        new[j] = 1.0
        new_c = compute_c5(new)

        # accept if improvement, else tiny uphill probability
        accept = False
        if new_c < cur_c - 1e-12:
            accept = True
        elif rng.random() < 0.001:
            accept = True

        if accept:
            cur = new
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best = cur.copy()
    return best, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial size and pattern -----
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)

    else:
        n0 = 180
        base_pat = np.zeros(n0, dtype=float)

    best_h = None
    best_c = np.inf

    # ----- several restarts on the base size -----
    restarts = 5
    for r in range(restarts):
        if time.time() - start > budget_s * 0.85:

        else:

        h_opt, c_opt = binary_hill_climb(h_start, rng, max_iter=18000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # ----- progressive up‑sampling (doubling) -----
    max_n = 500  # safety cap
    while best_h is not None and time.time() - start < budget_s * 0.95:
        cur_n = best_h.size
        if cur_n * 2 > max_n:
            break
        # repeat each entry twice -> binary, weight doubles correctly
        up = np.repeat(best_h, 2)
        # hill‑climb on the larger grid

        h_opt, c_opt = binary_hill_climb(up, rng, max_iter=iters)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        else:
            # no improvement – stop further up‑sampling
            break

    # ----- final cleanup -----
    if best_h is None:
        # fallback: uniform half‑value
        n_f = n0
        best_h = np.full(n_f, 0.5, dtype=float)

        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_h.size)
