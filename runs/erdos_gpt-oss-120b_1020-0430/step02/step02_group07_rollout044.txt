       # pick i from ones, j from zeros

       # But we can randomly pick i and j until they are opposite values.
       i = rng.integers(0, n)

           cur_H[i], cur_H[j] = cur_H[j], cur_H[i]

       temp *= decay
       it += 1
   return best_H, best_c

def float_to_binary_feasible(h, rng):
    # threshold at 0.5 to get integer 0/1, then adjust sum to n/2 exactly.
    n = h.size
    H = (h > 0.5).astype(int)
    target = n // 2
    current = H.sum()
    diff = target - current
    if diff > 0:
        # need to turn diff zeros to ones
        zero_indices = np.where(H == 0)[0]
        choices = rng.choice(zero_indices, size=diff, replace=False)
        H[choices] = 1
    elif diff < 0:
        # need to turn -diff ones to zeros
        one_indices = np.where(H == 1)[0]
        choices = rng.choice(one_indices, size=-diff, replace=False)
        H[choices] = 0
    return H

   best_h, best_c = quick_refine(h0, rng)
   # More robust search with random restarts while time permits (0.6*budget)
   while time.time() - start < budget_s * 0.5:

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Fast C5 using FFT (linear correlation)."""
    n = h.shape[0]
    N = 2 * n
    fh = np.fft.fft(h, N)
    fg = np.fft.fft(1 - h, N)
    corr = np.fft.ifft(fh * np.conj(fg)).real
    max_val = np.max(corr[n - 1 : 2 * n - 1])
    return max_val * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02, decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for _ in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def quick_refine(h, rng):
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=800, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def deep_refine(h, rng):
    n = h.size
    max_iter_slsqp = 200 if n > 200 else 300
    hill_iters = 800 if n > 200 else 1500
    anneal_iters = 2000 if n > 200 else 3000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter_slsqp)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    trial_sa, c_sa = simulated_annealing(
        best_h, rng, max_iters=anneal_iters, init_temp=0.03, decay=0.998, scale=0.2
    )
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def random_binary_feasible(n, rng):
    H = np.zeros(n, dtype=int)
    ones = rng.choice(n, n // 2, replace=False)
    H[ones] = 1
    return H


def float_to_binary_feasible(h, rng):
    n = h.size
    H = (h > 0.5).astype(int)
    target = n // 2
    cur = H.sum()
    diff = target - cur
    if diff > 0:
        zero_idx = np.where(H == 0)[0]
        choose = rng.choice(zero_idx, diff, replace=False)
        H[choose] = 1
    elif diff < 0:
        one_idx = np.where(H == 1)[0]
        choose = rng.choice(one_idx, -diff, replace=False)
        H[choose] = 0
    return H


def binary_sa_time_budget(H0, rng, budget):
    start = time.time()
    H = H0.copy()
    cur_c = compute_c5(H.astype(float))
    best_H = H.copy()
    best_c = cur_c
    temp = 0.01
    decay = 0.9995
    n = H.size
    while time.time() - start < budget:
        i = rng.integers(0, n)
        if H[i] == 1:
            j = rng.integers(0, n)
            while H[j] != 0:
                j = rng.integers(0, n)
        else:
            j = rng.integers(0, n)
            while H[j] != 1:
                j = rng.integers(0, n)
            i, j = j, i
        H[i] = 0
        H[j] = 1
        new_c = compute_c5(H.astype(float))
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_H = H.copy()
        else:
            H[i] = 1
            H[j] = 0
        temp *= decay
    return best_H, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # continuous optimization phase (≈40% of budget)
    best_h, best_c = quick_refine(h0, rng)
    while time.time() - start < budget_s * 0.4:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # up‑sample and deep refinement if time permits
    if time.time() - start < budget_s * 0.7:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        h_up_ref, c_up_ref = deep_refine(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = n_up

    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), n / 2.0, rng)
    best_c = compute_c5(best_h)

    # binary phase: start from binary version of best continuous result
    H_bin = float_to_binary_feasible(best_h, rng)
    bin_c = compute_c5(H_bin.astype(float))
    if bin_c < best_c:
        best_h = H_bin.astype(float)
        best_c = bin_c

    # allocate remaining time to binary simulated annealing
    remaining = budget_s - (time.time() - start)
    if remaining > 0.2:
        H_sa, c_sa = binary_sa_time_budget(H_bin, rng, remaining * 0.7)
        if c_sa < best_c:
            best_h = H_sa.astype(float)
            best_c = c_sa

    # occasional random binary restarts with short SA bursts
    while time.time() - start < budget_s - 0.05:
        H_rand = random_binary_feasible(n, rng)
        # give a brief budget (≈0.1 s) per restart
        H_res, c_res = binary_sa_time_budget(H_rand, rng, 0.1)
        if c_res < best_c:
            best_h = H_res.astype(float)
            best_c = c_res

    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
