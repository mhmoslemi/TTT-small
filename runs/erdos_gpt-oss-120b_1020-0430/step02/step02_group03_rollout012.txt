import time
import numpy as np
from scipy.optimize import minimize

def half_to_full(half):
    """Construct symmetric full vector of length 2*m from half."""
    return np.concatenate([half, 1.0 - half[::-1]])

def compute_c5_full(full):
    """Exact discrete C5 for a full vector."""
    corr = np.correlate(full, 1.0 - full, mode="full")
    return np.max(corr) * (2.0 / full.size)

def enforce_sum(full, target, rng):
    """Adjust a full vector to have exact sum while staying in [0,1]."""
    diff = target - full.sum()
    n = full.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - full[i]) if diff > 0 else max(diff, -full[i])
        full[i] += add
        diff = target - full.sum()
    return full

def hill_climb_half(half, rng, max_iters=3000, step=0.25):
    """Simple hill‑climb on the half‑vector (preserves symmetry)."""
    m = half.size
    best = half.copy()
    best_c = compute_c5_full(half_to_full(best))
    for it in range(max_iters):
        i, j = rng.integers(0, m, size=2)
        if i == j:
            continue
        max_eps = min(best[i], 1.0 - best[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        eps = max(-max_eps, min(eps, max_eps))
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        c = compute_c5_full(half_to_full(cand))
        if c < best_c:
            best = cand
            best_c = c
    return best, best_c

def slsqp_refine_half(half, beta, max_iter=300):
    """SLSQP refinement of the half‑vector using a smooth soft‑max surrogate."""
    m = half.size
    def softmax_obj(x):
        full = half_to_full(x)
        corr = np.correlate(full, 1.0 - full, mode="full")
        bc = beta * corr
        mval = np.max(bc)
        return (mval + np.log(np.sum(np.exp(bc - mval)))) / beta
    bounds = [(0.0, 1.0)] * m
    res = minimize(
        softmax_obj,
        half,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)

    start_time = time.time()

    # -------------------------------------------------------------

    # -------------------------------------------------------------
    if "initial_h_values" in globals():

        n = n0
        # possibly resample to a convenient even size (keep original if even)
        if n0 != init_full.size:

    else:
        n = 180               # default choice
        init_full = None

    m = n // 2
    target_full_sum = n / 2.0

    # -------------------------------------------------------------
    # 2. GA initial population (half‑vectors)
    # -------------------------------------------------------------
    pop_size = max(60, min(120, m))  # modest size
    elite_sz = 2
    pop = rng.random((pop_size, m))  # uniform in [0,1]

    # insert a seed derived from the provided construction if available
    if init_full is not None:
        seed_half = init_full[:m]
        pop[0] = np.clip(seed_half, 0.0, 1.0)

    # -------------------------------------------------------------
    # 3. Fitness evaluation
    # -------------------------------------------------------------
    def fitness_half(half):
        return compute_c5_full(half_to_full(half))

    fitness = np.empty(pop_size)

    best_c = np.inf
    for i in range(pop_size):

        if fitness[i] < best_c:

            a, b = rng.integers(0, pop_size, size=2)
            p1 = pop[a] if fitness[a] < fitness[b] else pop[b]

    full = half_to_full(best_half)

    full = enforce_sum(full, target_full_sum, rng)

    final_c = compute_c5_full(full)

    return full.tolist(), float(final_c), int(n)
