import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global placeholders for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    # smooth approximation of max_k Σ_i x_i (1-x_{i+k})
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def _slsqp_optimize(x0, beta, max_iter=400):
    """Run a single SLSQP refinement for a given smoothing beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny numerical drift fix
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- Utility functions ----------
def _compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def _repair_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def _random_feasible(n, rng):
    """Generate a random feasible vector (0≤h_i≤1, Σh_i = n/2)."""
    h = rng.random(n)
    h = _repair_sum(h, n / 2.0, rng)
    return h

def _perturb(h, rng, scale):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    new_h = h.copy()
    moves = rng.integers(1, 4)          # 1‑3 elementary transfers
    for _ in range(moves):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(new_h[i], 1.0 - new_h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        eps = max(-max_eps, min(eps, max_eps))
        new_h[i] -= eps
        new_h[j] += eps
    return new_h

    # use 60 % of total budget for SA
    sa_budget = time_limit * 0.60
    T0 = 0.015           # initial temperature
    T_end = 1e-8         # final temperature
    target_sum = n / 2.0
    cur_h = best_h.copy()
    cur_c = best_c
    it = 0
    while time.time() - start_time < sa_budget:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / sa_budget)
        # exponential cooling
        T = T0 * (T_end / T0) ** frac
        # scale of transfer shrinks with temperature, but stays ≥0.05
        scale = 0.05 + 0.45 * (T / T0)

        cand = _perturb(cur_h, rng, scale)
        # clipping + repair (tiny drift possible)
        cand = _repair_sum(cand, target_sum, rng)

        cand_c = _compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c

    # ---- SLSQP refinement -------------------------------------------
    # use increasing smoothing; allocate up to 30 % of total time
    slsqp_budget = time_limit * 0.30
    beta_seq = [5, 20, 80, 200, 800, 3200]
    for beta in beta_seq:
        if time.time() - start_time > slsqp_budget:
            break
        try:
            refined = _slsqp_optimize(best_h, beta, max_iter=400)
        except Exception:
            continue
        refined_c = _compute_c5(refined)
        if refined_c < best_c:
            best_h, best_c = refined.copy(), refined_c

    # ---- optional up‑sampling ---------------------------------------
    # if time permits and we can stay ≤300 points
    up_budget = time_limit * 0.07
    if time.time() - start_time < up_budget and n * 2 <= 300:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = _repair_sum(h_up, target_up, rng)

        # short SA on the finer grid
        sa_up_budget = up_budget * 0.5
        cur_h = h_up.copy()
        cur_c = _compute_c5(cur_h)
        it = 0
        while time.time() - start_time < up_budget - sa_up_budget * 0.5:
            it += 1
            elapsed = time.time() - start_time
            frac = min(1.0, elapsed / sa_up_budget)
            T = T0 * (T_end / T0) ** frac
            scale = 0.05 + 0.45 * (T / T0)
            cand = _perturb(cur_h, rng, scale)
            cand = _repair_sum(cand, target_up, rng)
            cand_c = _compute_c5(cand)
            delta = cand_c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
                cur_h, cur_c = cand, cand_c
                if cand_c < best_c:
                    best_h, best_c = cand.copy(), cand_c

        # SLSQP on the upsampled version
        for beta in beta_seq:
            if time.time() - start_time > time_limit:
                break
            try:
                refined = _slsqp_optimize(best_h, beta, max_iter=300)
            except Exception:
                continue
            refined_c = _compute_c5(refined)
            if refined_c < best_c:
                best_h, best_c = refined.copy(), refined_c
        n = best_h.size

    # ---- final clean‑up ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, n / 2.0, rng)
    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
