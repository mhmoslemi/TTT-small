import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# Projection onto the intersection of the box [0,1]^n and the hyperplane
# sum(x) = target.  Used to enforce the constraints h_i ∈ [0,1] and
# ∑h_i = n/2.
# ----------------------------------------------------------------------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0  # lower bound for lambda
    hi = np.max(v)        # upper bound for lambda
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

# ----------------------------------------------------------------------
# Compute C5 for a discrete step function h on [0,2].
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

# ----------------------------------------------------------------------
# SLSQP refinement (same as the previous program, but kept compact)
# ----------------------------------------------------------------------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x):
    return x[-1]

def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = minimize(_objective, x0, method="SLSQP", bounds=bounds,
                   constraints=constraints, options=opts)
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

# ----------------------------------------------------------------------
# Simple tent shape (fallback)
# ----------------------------------------------------------------------
def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

# ----------------------------------------------------------------------
# Decode a low‑dimensional parameter vector into a full‑resolution h.
# The parameters are interpreted as heights at equally spaced distances
# from the centre (symmetry is built in).
# ----------------------------------------------------------------------
def _decode_to_h(params: np.ndarray, n: int) -> np.ndarray:
    m = params.size
    # distances from centre for each grid point
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    d = np.abs(xs - 1.0)
    # linear interpolation of the parameter values
    raw = np.interp(d, np.linspace(0.0, 1.0, m), params)
    # enforce sum = n/2 and box constraints
    h = _project_to_simplex_box(raw, n / 2.0)
    return h

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    target_bound = 0.3808
    best_h = None
    best_c5 = np.inf
    best_n = None

    # Candidate resolutions (coarse -> fine)
    n_candidates = [192, 256, 384]

    for n in n_candidates:
        if time_left() < 20.0:
            break

        # Dimensionality of the latent representation.
        m = max(6, n // 32)           # 6 for n=192, 8 for 256, 12 for 384

        # Run a modest differential evolution search.
        bounds = [(0.0, 1.0)] * m
        maxgen = 100                  # generations
        popsize = 15                  # individuals per dimension
        de = differential_evolution(
            lambda p, n=n: _c5_from_h(_decode_to_h(p, n)),
            bounds,
            maxiter=maxgen,
            popsize=popsize,
            seed=seed + n,
            polish=False,
            tol=1e-6,
            disp=False,
        )
        h_candidate = _decode_to_h(de.x, n)
        c5_candidate = _c5_from_h(h_candidate)

        if c5_candidate < best_c5:
            best_h, best_c5, best_n = h_candidate, c5_candidate, n
            if best_c5 < target_bound:
                break

        # Local refinement with SLSQP (if time permits)
        if time_left() > 30.0:
            h_refined, c5_refined = _solve_h(h_candidate, max_iter=500)
            if c5_refined < best_c5:
                best_h, best_c5, best_n = h_refined, c5_refined, n
                if best_c5 < target_bound:
                    break

        # Also try refining a pure tent shape at this resolution
        if time_left() > 30.0:
            h_tent = _tent_h(n)
            h_refined, c5_refined = _solve_h(h_tent, max_iter=500)
            if c5_refined < best_c5:
                best_h, best_c5, best_n = h_refined, c5_refined, n
                if best_c5 < target_bound:
                    break

    # In the unlikely case nothing was produced, fall back to a simple tent.
    if best_h is None:
        best_n = 192
        best_h = _tent_h(best_n)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
