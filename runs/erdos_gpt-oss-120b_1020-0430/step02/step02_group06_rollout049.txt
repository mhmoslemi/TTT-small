def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = np.sum(h)
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        count = np.sum(free)
        if count == 0:
            break
        delta = diff / count
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction:
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _solve_h(initial_h, max_iter=800):
    n = initial_h.size
    # project feasible start
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # constraints functions that accept x vector
    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode='full')
    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)
    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]
    # Use SLSQP from scipy.optimize
    from scipy import optimize
    opts = {'maxiter': max_iter, 'ftol': 1e-10, 'disp': False}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                            constraints=cons, options=opts)
    if res.success and res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

import time
import numpy as np
from scipy import optimize

# ---------------------- Helper utilities ---------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction for residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_h(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Run SLSQP to minimise the auxiliary variable t."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    if res.success and res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tournament_select(pop, fitness, size=3):
    idx = np.random.choice(len(pop), size=size, replace=False)
    best = idx[np.argmin([fitness[i] for i in idx])]
    return pop[best]


def _two_point_crossover(p1, p2):
    n = p1.shape[0]
    a, b = sorted(np.random.choice(n + 1, 2, replace=False))
    child = np.empty(n)
    child[:a] = p1[:a]
    child[a:b] = p2[a:b]
    child[b:] = p1[b:]
    return child


def _mutate(h, sigma):
    n = h.shape[0]
    h = h + sigma * np.random.randn(n)
    return _project_to_feasible(h, n)


def _ga_optimize(n, init_h, time_left, target_bound):
    """Simple GA that returns a promising h and its C5."""
    pop_size = 30
    elite = 2
    mut_rate = 0.7
    sigma = 0.2
    sigma_decay = 0.99

    pop = []
    fit = []
    for i in range(pop_size):
        if i == 0:
            h = _project_to_feasible(init_h.copy(), n)
        else:
            h = _project_to_feasible(np.random.rand(n), n)
        pop.append(h)
        fit.append(_c5_from_h(h))

    best_idx = int(np.argmin(fit))
    best_h = pop[best_idx].copy()
    best_c5 = fit[best_idx]

    gen = 0
    while time_left() > 2.0:
        gen += 1
        # elitism
        order = np.argsort(fit)
        new_pop = [pop[i].copy() for i in order[:elite]]
        # offspring
        while len(new_pop) < pop_size:
            p1 = _tournament_select(pop, fit)
            p2 = _tournament_select(pop, fit)
            child = _two_point_crossover(p1, p2)
            if np.random.rand() < mut_rate:
                child = _mutate(child, sigma)
            else:
                child = _project_to_feasible(child, n)
            new_pop.append(child)
        pop = new_pop
        fit = [_c5_from_h(ind) for ind in pop]

        cur_idx = int(np.argmin(fit))
        if fit[cur_idx] < best_c5 - 1e-12:
            best_c5 = fit[cur_idx]
            best_h = pop[cur_idx].copy()
            if best_c5 < target_bound:
                break
        sigma *= sigma_decay
        if gen >= 5000:
            break
    return best_h, best_c5

    start = time.time()
