def optimize_binary(n, rng, time_limit):
    target_ones = n // 2
    # start with random binary vector with exactly target ones
    h = np.zeros(n, dtype=np.int8)
    ones_idx = rng.choice(n, size=target_ones, replace=False)
    h[ones_idx] = 1
    curr_c = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c = curr_c
    # Pre-calc list of indices for ones and zeros
    ones = set(np.where(h==1)[0])
    zeros = set(np.where(h==0)[0])
    # annealing parameters
    start_temp = 0.1  # on C scale maybe
    end_temp = 1e-4
    iter = 0
    # compute total iterations based on time budget
    while time.time() < time_limit:
        # Determine temperature decreasing schedule
        # maybe linear or exponential based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / (time_limit - start_time)
        T = start_temp * (1 - frac) + end_temp * frac
        # pick random swap
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # swap
        h[i], h[j] = 0, 1
        # update sets
        ones.remove(i); zeros.add(i)
        zeros.remove(j); ones.add(j)
        new_c = compute_c5(h.astype(float))
        delta = new_c - curr_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
            ones.add(i); zeros.remove(i)
            zeros.add(j); ones.remove(j)
    return best_h.astype(float), best_c

        zeros_list = [i for i in range(n) if h[i] == 0]

        iters_since_improve = 0
        total_iters = 0

        while True:
            if time.time() - start_time >= total_budget * 0.95:
                break
            # maybe stop this inner loop after many attempts without improvement?
            if iters_since_improve > 50000:
                # break to restart
                break

            # temperature schedule based on total elapsed
            elapsed = time.time() - start_time
            fraction = elapsed / (total_budget * 0.95)
            T = 0.05 * (1 - fraction) + 0.001 * fraction

            # pick random swap
            idx_i = rng.integers(len(ones_list))
            i = ones_list[idx_i]
            idx_j = rng.integers(len(zeros_list))
            j = zeros_list[idx_j]

            h[i] = 0
            h[j] = 1

            ones_list[idx_i] = j
            pos_in_ones[j] = idx_i
            pos_in_ones[i] = -1

            zeros_list[idx_j] = i
            pos_in_zeros[i] = idx_j
            pos_in_zeros[j] = -1

            new_c = compute_c5(h)
            delta = new_c - curr_c
            accept = False
            if delta < 0:
                accept = True
            else:
                if T > 0 and rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                curr_c = new_c
                iters_since_improve = 0 if delta < 0 else iters_since_improve + 0
                if new_c < best_c:

def hill_climb_binary(h, max_no_improve=5000):

    # maintain lists like before
    # For simplicity, we can convert to arrays and recompute sets each step (since hill climbing runs few steps)
    best_c = compute_c5(h)
    no_improve = 0
    while no_improve < max_no_improve:
        # pick random ones and zeros
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h_tmp = h.copy()
        h_tmp[i], h_tmp[j] = 0, 1
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c:
            h = h_tmp
            best_c = c_tmp
            no_improve = 0
        else:
            no_improve += 1
    return h, best_c

        best_h = project_onto_box_sum(best_h, target_sum)
    final_c = float(final_c)

import time
import numpy as np

# -------------------------------------------------
# basic overlap computation (discrete)
def compute_c5(arr):
    """C5 for a discretized step function (arr may be int or float)."""
    n = arr.size
    corr = np.correlate(arr, 1 - arr, mode="full")
    return np.max(corr) * (2.0 / n)

# -------------------------------------------------
def anneal_binary(n, rng, seconds):
    """Simulated‑annealing on a binary vector of length n with n/2 ones."""
    start = time.time()
    target_ones = n // 2
    best_h = None
    best_c = np.inf

    # temperature schedule (linear)
    T_start, T_end = 0.10, 0.001

    while time.time() - start < seconds:
        # ----- random feasible start -----
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, size=target_ones, replace=False)
        h[ones] = 1
        cur_c = compute_c5(h)

        # no‑improvement counter – restart after long stagnation
        stagnant = 0

        while True:
            elapsed = time.time() - start
            if elapsed >= seconds:
                break

            # stop inner loop if we have been stagnant for a long time
            if stagnant > 50000:
                break

            # temperature (linear decay)
            frac = elapsed / seconds
            T = T_start * (1 - frac) + T_end * frac

            # ---- identify worst shift ----
            corr = np.correlate(h, 1 - h, mode="full")
            max_idx = int(np.argmax(corr))
            k = max_idx - (n - 1)          # shift value giving the current maximum

            # ---- positions that contribute to the worst overlap ----
            if k >= 0:
                left = h[: n - k]
                right = h[k:]
                bad = np.where((left == 1) & (right == 0))[0]
                good = np.where((left == 0) & (right == 1))[0]
            else:
                k1 = -k
                left = h[k1:]                # corresponds to i positions
                right = h[: n - k1]
                bad = np.where((left == 1) & (right == 0))[0] + k1
                good = np.where((left == 0) & (right == 1))[0] + k1

            # ---- propose a swap ----
            if bad.size and good.size:
                i = rng.choice(bad)
                j = rng.choice(good)
            else:
                # fallback to a completely random 1↔0 swap
                ones_idx = np.where(h == 1)[0]
                zeros_idx = np.where(h == 0)[0]
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)

            # perform tentative swap
            h[i], h[j] = 0, 1
            new_c = compute_c5(h)
            delta = new_c - cur_c

            # acceptance test
            if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
                cur_c = new_c
                stagnant = 0 if delta < 0 else stagnant + 1
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = 1, 0
                stagnant += 1

    # ensure we have something to return
    if best_h is None:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, size=target_ones, replace=False)] = 1
        best_h, best_c = h.astype(float), compute_c5(h)
    return best_h.astype(float), float(best_c)


# -------------------------------------------------
def hill_climb_binary(h, rng, seconds):
    """Deterministic improvement (only downhill moves) after annealing."""
    start = time.time()
    n = h.size
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    no_improve = 0

    while time.time() - start < seconds:
        if no_improve > 30000:
            break

        # worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        if k >= 0:
            left = h[: n - k]
            right = h[k:]
            bad = np.where((left == 1) & (right == 0))[0]
            good = np.where((left == 0) & (right == 1))[0]
        else:
            k1 = -k
            left = h[k1:]
            right = h[: n - k1]
            bad = np.where((left == 1) & (right == 0))[0] + k1
            good = np.where((left == 0) & (right == 1))[0] + k1

        if bad.size and good.size:
            i = rng.choice(bad)
            j = rng.choice(good)
        else:
            # random swap as a last resort
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

        h[i], h[j] = 0, 1
        new_c = compute_c5(h)
        if new_c < best_c:
            best_c, best_h = new_c, h.copy()
            cur_c = new_c
            no_improve = 0
        else:
            h[i], h[j] = 1, 0
            no_improve += 1

    return best_h, float(best_c)


# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_global = time.time()
    deadline = start_global + budget_s

    best_h, best_c, best_n = None, np.inf, None

    # try a couple of even lengths; the constant is independent of n,
    # but different n give different discretisation granularity.
    # 180 and 200 are fast enough and already known to approach the optimum.
    for n in (180, 200):
        # time left for this n (split remaining time roughly equally)
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        # allocate half of the remaining budget for this length
        alloc = remaining * 0.5

        # ---- simulated annealing ----
        h, c = anneal_binary(n, rng, alloc * 0.85)   # keep a bit for refinement
        # ---- deterministic hill‑climb refinement ----
        refinement_time = max(0.0, alloc - (time.time() - now) - 0.1)
        if refinement_time > 0.0:
            h, c = hill_climb_binary(h, rng, refinement_time)

        if c < best_c:
            best_h, best_c, best_n = h, c, n

    # final safety check – recompute C5 with the official evaluator if present
    try:
        final_c = float(evaluate_erdos_solution(best_h))
    except Exception:               # fallback to our own computation
        final_c = compute_c5(best_h)

    # enforce exact constraints (tiny numeric drift may appear)
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_h.size / 2.0
    current_sum = best_h.sum()
    if abs(current_sum - target_sum) > 1e-12:
        # simple projection onto the box and exact sum
        diff = target_sum - current_sum
        # move the required mass from the largest entries that are not at the bound
        idx = np.argsort(best_h)[-int(abs(diff) * best_h.size):]
        best_h[idx] += diff / len(idx)
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h.tolist(), float(final_c), int(best_n)
