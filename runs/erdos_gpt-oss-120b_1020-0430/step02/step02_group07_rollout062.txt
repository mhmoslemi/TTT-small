def simulated_annealing(h_start, rng, max_iters, init_temp, decay, scale):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        # generate move
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        # decide acceptance
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h

    if max_eps <= 0:
        return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    if rng.random() < 0.5:
        max_transfer = min(h[i], 1.0 - h[j])
        eps = rng.random() * max_transfer * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    else:
        max_transfer = min(h[j], 1.0 - h[i])
        eps = rng.random() * max_transfer * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()

        # optional step decay
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    # scale to the correct sum
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    # fix any tiny residue
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # add/subtract the residue to a coordinate that can absorb it
        if diff > 0:
            idx = np.argmax(1.0 - h)          # place with most room
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmin(h)                # smallest entry
            sub = min(-diff, h[idx])
            h[idx] -= sub
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    if rng.random() < 0.5:
        max_transfer = min(h[i], 1.0 - h[j])
        eps = rng.random() * max_transfer * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    else:
        max_transfer = min(h[j], 1.0 - h[i])
        eps = rng.random() * max_transfer * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters, init_temp, decay, scale):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def upsample_double(h):
    """Double the resolution by repeating each sample once."""
    return np.repeat(h, 2)
