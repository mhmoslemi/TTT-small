def binary_hill_climb(h_bin, max_swaps=5000):
    h = h_bin.copy()
    n = len(h)
    ones_idx = np.where(h==1)[0]
    zeros_idx = np.where(h==0)[0]
    # compute Q using convolution initially.
    Q_full = np.correlate(h.astype(float), h.astype(float), mode='full')
    best_minQ = Q_full.min()
    best_h = h.copy()
    # iterate
    for t in range(max_swaps):
        # randomly pick one from ones and one from zeros
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        h[i] = 0
        h[j] = 1
        # recompute Q quickly? use full conv each iteration may be okay for n up to 256 and 5000 iterations => 5000*256*log= trivial.
        Q_full = np.correlate(h.astype(float), h.astype(float), mode='full')
        cur_minQ = Q_full.min()
        if cur_minQ > best_minQ:
            best_minQ = cur_minQ
            best_h = h.copy()
            # update indices
            ones_idx = np.where(best_h==1)[0]
            zeros_idx = np.where(best_h==0)[0]
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h.astype(float)

import time
import numpy as np

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / free.sum()
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # tiny correction via a single entry
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
