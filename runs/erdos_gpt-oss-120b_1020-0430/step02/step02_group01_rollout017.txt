def softmax_obj_and_grad(h, beta):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # compute weights
    maxc = np.max(corr)
    # to prevent overflow: subtract max
    expb = np.exp(beta * (corr - maxc))
    sumexp = np.sum(expb)
    obj = (maxc + np.log(sumexp)) / beta
    # weights
    w = expb / sumexp
    # gradient
    grad = np.zeros(N)
    # iterate over d
    offset = N - 1
    for d in range(-N+1, N):
        wd = w[d + offset]
        if abs(wd) < 1e-16:
            continue
        # compute contributions
        # for i where i+d in [0,N):
        #   grad[i] += wd * (1 - h[i+d])
        # for i where i-d in [0,N):

def repair_sum(h, target, rng):
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:
        i = rng.integers(0, h.size)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0:
                attempts += 1
                continue
            delta = min(cap, diff)
        else:
            cap = -h[i]
            if cap >= 0:
                attempts += 1
                continue
            delta = max(cap, diff)  # diff negative
        h[i] += delta
        diff = target - h.sum()
        attempts += 1
    return h

def softmax_obj_and_grad(h, beta):
    N = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(corr)
    # subtract max for stability
    expb = np.exp(beta * (corr - maxc))
    sumexp = np.sum(expb)
    obj = (maxc + np.log(sumexp)) / beta
    w = expb / sumexp
    # compute gradient using loops
    grad = np.zeros_like(h, dtype=float)
    offset = N - 1
    for d in range(-N+1, N):
        wd = w[d + offset]
        if wd == 0.0:
            continue
        if d >= 0:
            # forward part
            end = N - d
            grad[:end] += wd * (1 - h[d:])
            grad[d:] -= wd * h[:end]
        else:
            dd = -d
            end = N - dd
            grad[dd:] += wd * (1 - h[:end])
            grad[:end] -= wd * h[dd:]
    return obj, grad

def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Smooth max (log‑sum‑exp) of the overlap correlates and its gradient."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")

        if d >= 0:
            end = N - d
            grad[:end] += wd * (1.0 - h[d:])   # (1‑h_{i+d})
            grad[d:]  -= wd * h[:end]          # -h_{i‑d}
        else:
            dd = -d
            end = N - dd
            grad[dd:] += wd * (1.0 - h[:end])  # (1‑h_{i‑dd})
            grad[:end] -= wd * h[dd:]          # -h_{i+dd}
    return obj, grad


def project_sum_bounds(y: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector y onto the box [0,1]^n while forcing Σx_i = target.
    Uses a simple bisection on the common shift λ where x = clip(y+λ,0,1).
    """
    lo = -np.max(y) - 1.0          # guarantees all clipped to 0
    hi = 1.0 - np.min(y) + 1.0     # guarantees all clipped to 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-13:
            break
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def gradient_descent_surrogate(
    h0: np.ndarray,
    beta: float,
    max_iter: int,
    rng: np.random.Generator,
    step_scale: float = 0.5,
):
    """Projected gradient descent on the soft‑max surrogate."""
    h = h0.copy()
    N = h.size
    target = N / 2.0
    obj, _ = softmax_obj_and_grad(h, beta)
    for _ in range(max_iter):
        obj, grad = softmax_obj_and_grad(h, beta)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        step = step_scale / norm
        for _ in range(5):               # simple back‑tracking
            h_tmp = h - step * grad
            h_proj = project_sum_bounds(h_tmp, target)
            new_obj, _ = softmax_obj_and_grad(h_proj, beta)
            if new_obj <= obj + 1e-14:
                h = h_proj
                obj = new_obj
                break
            step *= 0.5
    return h, obj


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(
    h_start: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 3000,
    start_step: float = 0.5,
):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


# ------------------------------------------------------------
# Main optimisation driver
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # --------------------------------------------------------
    # Parent solution (if supplied)
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = h_parent.size
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_sum_bounds(h_parent, n_parent / 2.0)
        c_parent = compute_c5(h_parent)
        best_h, best_c, best_n = h_parent.copy(), c_parent, n_parent

    # --------------------------------------------------------
    # Candidate grid sizes
    # --------------------------------------------------------
    cand_ns = []
    if best_n is not None:
        cand_ns.append(best_n)
        if best_n * 2 <= 240:
            cand_ns.append(best_n * 2)
    # a few extra modest sizes
    for n in (96, 108, 120, 144, 168, 192, 240):
        if n not in cand_ns:
            cand_ns.append(n)

    # --------------------------------------------------------
    # Main search loop
    # --------------------------------------------------------
    for n in cand_ns:
        if time.time() - start_time > budget_s * 0.92:
            break
        # number of random restarts tailored to size
        restarts = 8 if n <= 120 else 5
        for _ in range(restarts):
            if time.time() - start_time > budget_s * 0.95:
                break

            # ----- initialise a feasible vector -----------------
            if best_n == n and best_h is not None and _ == 0:
                h = best_h.copy()
            elif best_n and n == best_n * 2:
                # integer up‑sampling via repeat
                factor = n // best_n
                h = np.repeat(best_h, factor)
                h = project_sum_bounds(h, n / 2.0)
            else:
                # random binary half‑ones or continuous start
                if rng.random() < 0.5:
                    h = np.zeros(n)
                    idx = rng.choice(n, n // 2, replace=False)
                    h[idx] = 1.0
                else:
                    h = rng.random(n)
                    h = project_sum_bounds(h, n / 2.0)

            # ----- coarse optimisation with surrogate ------------
            for beta in (5.0, 20.0, 80.0, 200.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                max_iter = 250 if n <= 120 else 200
                h, _ = gradient_descent_surrogate(
                    h, beta, max_iter=max_iter, rng=rng, step_scale=0.5
                )
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ----- final local refinement -----------------------
            if time.time() - start_time < budget_s * 0.98:
                h, c = hill_climb(h, rng, max_iters=1500, start_step=0.3)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # --------------------------------------------------------
    # Final feasibility enforcement and return
    # --------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
