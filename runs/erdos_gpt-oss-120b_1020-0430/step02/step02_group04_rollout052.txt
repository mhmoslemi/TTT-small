                step *= 0.9995

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
#  Basic utilities
# -------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / h.size)
    return float(np.max(corr))


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto the box [lower,upper] with exact sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    # fix tiny residual
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible vector (0≤h≤1, sum=n/2)."""
    m = n // 2
    half_target = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, half_target)
    h = np.concatenate([x, x[::-1]])
    return h


def enforce_symmetry_sum(h, target):
    """Enforce mirror symmetry and exact sum."""
    h = project_onto_box_sum(h, target)
    h = (h + h[::-1]) * 0.5
    h = project_onto_box_sum(h, target)
    return h

#  Hill‑climb utilities (simple local search)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# -------------------------------------------------
#  SLSQP refinement (soft‑max surrogate)
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

#  Smooth‑max gradient descent (fast convolution)

def smooth_max_descent(
    h0, betas, steps_per_beta, step0=0.5, mu=0.9, rng=None, target=None
):
    """Projected accelerated gradient descent on the log‑sum‑exp surrogate."""
    n = h0.size
    if target is None:
        target = n / 2.0
    h = h0.copy()
    v = np.zeros_like(h)
    for beta in betas:
        step = step0
        for _ in range(steps_per_beta):
            # scaled overlaps O(k)
            O = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
            xb = beta * O
            m = np.max(xb)
            w = np.exp(xb - m)
            p = w / w.sum()  # soft‑max probabilities, sum to 1
            # convolution p * h
            conv = np.convolve(p, h, mode="full")
            conv_center = conv[n - 1 : n - 1 + n]  # length n
            grad = 1.0 - 2.0 * conv_center  # gradient of (un‑scaled) smooth max
            v = mu * v - step * grad
            h = h + v
            # keep feasibility and symmetry
            h = enforce_symmetry_sum(h, target)
            step *= 0.9995
    return h
