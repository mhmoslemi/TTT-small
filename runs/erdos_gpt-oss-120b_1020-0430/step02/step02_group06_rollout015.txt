def _project_to_feasible(h, target_sum=None):
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    if target_sum is None:
        target_sum = n / 2.0
    # water-filling: adjust values while preserving bounds.
    # Use iterative approach similar to earlier but possibly faster.
    # We'll do a simple while loop.
    for _ in range(100):
        s = h.sum()
        diff = s - target_sum
        if abs(diff) < 1e-12:
            break
        # find indices not at bound in direction of diff
        if diff > 0: # need to decrease sum
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # distribute one by one
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy import optimize

# -------------------- Feasibility projection -------------------- #
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# -------------------- Correlation & C5 -------------------- #
def _c5_from_h(h: np.ndarray) -> float:
    """Compute C₅ using FFT‑based convolution (fast for many evaluations)."""
    n = h.size
    a = h.astype(np.float64)
    b = 1.0 - a
    b_rev = b[::-1]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b_rev, m)
    conv = np.fft.irfft(fa * fb, m)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


# -------------------- Local SLSQP refinement -------------------- #
def _solve_h(initial_h: np.ndarray, max_iter: int = 600):
    """Run SLSQP starting from `initial_h`. Returns refined (h, C5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def obj(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = optimize.minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.x is not None else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5


# -------------------- Interpolation helper -------------------- #
def _interpolate_h(src: np.ndarray, target_n: int) -> np.ndarray:
    src_n = src.size
    x_src = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_tgt = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    return np.interp(x_tgt, x_src, src)
