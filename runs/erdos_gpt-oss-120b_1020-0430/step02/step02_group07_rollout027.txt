def binary_simulated_annealing(h0, rng, time_limit):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    it = 0
    temp = init_temp = 0.5
    while time.time() - start < time_limit and it < max_iters:
        # random swap
        # sample indices with value1 and value0
        idx1 = rng.choice(np.where(h==1)[0])
        idx0 = rng.choice(np.where(h==0)[0])
        h_new = h.copy()
        h_new[idx1] = 0
        h_new[idx0] = 1
        new_c = compute_c5(h_new)
        if new_c < best_c:
            best_h, best_c = h_new.copy(), new_c
        # accept if better or with probability
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = h_new, new_c
        # cooling
        temp *= decay
        it += 1
    return best_h, best_c
```

        while cur_h[idx1] == 0:

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    target = n // 2
    idx = rng.permutation(n)[:target]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def binarize_vector(v: np.ndarray) -> np.ndarray:
    """Set the largest n/2 entries of v to 1, others to 0."""
    n = v.size
    target = n // 2
    # argpartition gives unordered top‑target indices efficiently
    idx = np.argpartition(v, -target)[-target:]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


# ----------------------------------------------------------------------
# Continuous local search (same as previous solution)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # repair tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def deep_refine(h, rng):
    """Intensive refinement: ramped β, hill‑climb and annealing."""
    n = h.size
    max_iter_slsqp = 200 if n > 200 else 300
    hill_iters = 800 if n > 200 else 1500
    anneal_iters = 2000 if n > 200 else 3000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter_slsqp)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    trial_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iters=anneal_iters,
                                         init_temp=0.03,
                                         decay=0.998,
                                         scale=0.2)
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


# ----------------------------------------------------------------------
# Binary annealing (new component)
# ----------------------------------------------------------------------
def binary_anneal(h0, rng, time_limit):
    """Simulated annealing on binary vectors (swap 1↔0)."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    start = time.time()
    temp = 0.5
    decay = 0.9995
    while time.time() - start < time_limit:
        # pick a random 1 and a random 0
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = cur_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_c = compute_c5(h_new)
        if new_c < best_c:
            best_h, best_c = h_new.copy(), new_c
        # Metropolis acceptance
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            cur_h, cur_c = h_new, new_c
        temp *= decay
        # occasional random restart to avoid dead‑ends
        if rng.random() < 0.001:
            cur_h = random_feasible_binary(n, rng)
            cur_c = compute_c5(cur_h)
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Initialise base vector (binary) – use supplied parent if any
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        n = h_init.size
        h_bin_start = binarize_vector(h_init)
    else:
        n = 360  # reasonable default resolution
        h_bin_start = random_feasible_binary(n, rng)

    # --------------------------------------------------------------
    # Stage 1 – binary simulated annealing (≈30 % of budget)
    # --------------------------------------------------------------
    t_stage1_end = start_time + budget_s * 0.30
    time_left = max(0.0, t_stage1_end - time.time())
    h_best_bin, c_best_bin = binary_anneal(h_bin_start, rng, time_left)

    # --------------------------------------------------------------
    # Stage 2 – continuous refinement using the previous pipeline
    # --------------------------------------------------------------
    # Convert to float, add a tiny random perturbation to leave binary basin
    h_cont = h_best_bin.astype(float)
    h_cont = perturb_h(h_cont, rng, scale=0.01)
    h_cont = enforce_sum(h_cont, n / 2.0, rng)

    # Deep refinement (SLSQP ladder + hill‑climb + annealing)
    h_cur, c_cur = deep_refine(h_cont, rng)

    # --------------------------------------------------------------
    # Stage 3 – up‑sample and re‑refine if time permits
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.85:
            break
        if n * 2 > 800:   # keep size reasonable for the time budget
            break
        # up‑sample by factor 2
        h_up = upsample_h(h_cur, factor=2, rng=rng)
        n_up = h_up.size
        h_up_ref, c_up_ref = deep_refine(h_up, rng)
        if c_up_ref < c_cur:
            h_cur, c_cur, n = h_up_ref, c_up_ref, n_up
        else:
            break

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    h_cur = np.clip(h_cur, 0.0, 1.0)
    h_cur = enforce_sum(h_cur, n / 2.0, rng)
    final_c = compute_c5(h_cur)

    return h_cur.tolist(), float(final_c), int(n)
