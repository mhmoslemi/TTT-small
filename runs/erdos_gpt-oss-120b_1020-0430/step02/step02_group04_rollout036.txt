def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=50):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    # residual correction
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj
```

def subgrad_optimize(h0, rng, max_iters=5000, step0=0.4):
    h = symmetrize(h0)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute gradient
        inds = np.arange(n)
        h_plus = h[(inds + k) % n]
        h_minus = h[(inds - k) % n]
        grad = 1.0 - (h_plus + h_minus)
        # move opposite gradient
        h_new = h - step * grad
        # small random jitter occasional
        if rng.random() < 0.01:
            h_new += (rng.random(n) - 0.5) * 0.01
        # projection
        h_new = project_onto_box_sum(h_new, n/2.0)
        # symmetrize
        h_new = symmetrize(h_new)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step *= 0.99  # reduce step on improvement
        else:
            step *= 0.995
            # occasional uphill accept
            if rng.random() < 0.01:
                h = h_new
    return best_h, best_c

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] with uniform spacing."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def project_onto_box_sum(v: np.ndarray, target: float,
                         lower: float = 0.0, upper: float = 1.0,
                         tol: float = 1e-12, max_iter: int = 50) -> np.ndarray:
    """Project v onto the intersection of [lower,upper]^n and the affine hyperplane sum=v_target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def symmetrize(h: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry h(x)=h(2-x)."""
    return (h + h[::-1]) * 0.5


def random_feasible_symmetric(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random symmetric feasible vector (0≤h≤1, Σh=n/2)."""
    half = n // 2
    target_half = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target_half)
    if n % 2 == 0:
        h = np.concatenate([x, x[::-1]])
    else:
        # odd length: add a middle element to meet the total sum
        sum_half = x.sum()
        middle = n / 2.0 - 2 * sum_half
        middle = float(np.clip(middle, 0.0, 1.0))
        h = np.concatenate([x, [middle], x[::-1]])
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another while staying in [0,1]."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 5000, start_step: float = 0.5) -> tuple:
    """Simple pairwise hill‑climb with decreasing step size."""
    h = symmetrize(h_start)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_h = symmetrize(new_h)
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def subgrad_optimize(h0: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 5000, step0: float = 0.3) -> tuple:
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = symmetrize(h0)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    inds = np.arange(n)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)                     # shift achieving the current max
        # gradient of the offending shift
        h_plus = h[(inds + k) % n]
        h_minus = h[(inds - k) % n]
        grad = 1.0 - (h_plus + h_minus)       # ∂f_k/∂h
        h_new = h - step * grad
        if rng.random() < 0.01:               # occasional jitter
            h_new += (rng.random(n) - 0.5) * 0.01
        h_new = project_onto_box_sum(h_new, n / 2.0)
        h_new = symmetrize(h_new)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step *= 0.99
        else:
            step *= 0.995
            if rng.random() < 0.01:           # occasional uphill move
                h = h_new
    return best_h, best_c


def cos_initial(n: int) -> np.ndarray:
    """Analytic cosine‑bell start: h(x)=½(1+cos(π(x‑1))) sampled uniformly."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx          # cell centres
    h = 0.5 * (1.0 + np.cos(np.pi * (xs - 1.0)))
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, n / 2.0)
    return h
