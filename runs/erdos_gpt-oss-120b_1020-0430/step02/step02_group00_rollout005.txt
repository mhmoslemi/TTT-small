def local_refine(h, max_iter=2000):

    n = len(h); target = n/2

    current = h.copy()

    current_c5 = _c5_from_h(current)

    for _ in range(max_iter):

        if time_left() < small:

            break

        # choose two distinct indices i, j
        i, j = np.random.choice(n, size=2, replace=False)

        # compute allowed delta to keep within [0,1] and sum constant

        # we will move delta from j to i: new_i = h_i + delta, new_j = h_j - delta

        # max delta = min(1 - h_i, h_j) if moving from j to i; also can be negative.

        # Let's sample delta uniformly in [-max_neg, max_pos]

        max_pos = min(1 - current[i], current[j])
        max_neg = min(current[i], 1 - current[j])
        # choose direction randomly
        if max_pos + max_neg == 0:
            continue
        delta = (np.random.rand() * (max_pos + max_neg)) - max_neg  # uniform in [-max_neg, max_pos]

        new_h = current.copy()
        new_h[i] += delta
        new_h[j] -= delta

        # Clip not needed as delta ensures within bounds
        new_c5 = _c5_from_h(new_h)

        if new_c5 < current_c5:
            current, current_c5 = new_h, new_c5

    return current, current_c5

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a feasible step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse discretisation onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _solve_h(start_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Run SLSQP from a feasible start and return (h_opt, c5)."""
    n = start_h.size
    t0 = np.max(np.correlate(start_h, 1 - start_h, mode="full"))
    x0 = np.concatenate([start_h, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    if res.success:
        h_opt = res.x[:n]
    else:
        h_opt = start_h.copy()
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)
