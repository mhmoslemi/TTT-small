def _project_to_feasible(h, n, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # If too high, reduce from indices > 0; if too low, increase at indices < 1
        if diff > 0:
            idx = h > 0
        else:
            idx = h < 1
        if not np.any(idx):
            break
        # Distribute diff uniformly across these indices
        delta = diff / np.count_nonzero(idx)
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust smallest possible
        # find index that can be increased or decreased
        if residual > 0:
            # increase
            for i in range(n):
                cap = 1.0 - h[i]
                add = min(cap, residual)
                h[i] += add
                residual -= add
                if residual <= 0:
                    break
        else:
            # decrease
            for i in range(n):
                cap = h[i]
                sub = min(cap, -residual)
                h[i] -= sub
                residual += sub
                if residual >= 0:
                    break
    return np.clip(h, 0.0, 1.0)
```

import time
import numpy as np
from scipy import signal

# ----------------------------------------------------------------------
# Projection onto the feasible set (0<=h<=1, sum(h)=n/2)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = h > 0
        else:
            idx = h < 1
        if not np.any(idx):
            break
        delta = diff / np.count_nonzero(idx)
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 value for a given h (dx factor = 2/n)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Linear up‑sampling of a coarse step function
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Simple hill‑climb improvement (pairwise epsilon transfers)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    rng = np.random.default_rng()
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Gradient‑descent optimisation with soft‑max approximation of the max
# ----------------------------------------------------------------------
def _gradient_descent(h0: np.ndarray, n: int, time_limit: float, rng) -> tuple:
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    momentum = np.zeros_like(h)
    beta = 0.9
    max_iters = 5000
    target_sum = n / 2.0

    for it in range(max_iters):
        if time.time() > time_limit:
            break

        # correlation (raw, without dx)
        C = np.correlate(h, 1.0 - h, mode="full")
        maxC_raw = C.max()
        c_val = maxC_raw * (2.0 / n)

        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

        # soft‑max weights for the max
        p = 4.0 + 24.0 * (it / max_iters)          # from 4 to 28
        w = np.exp(p * (C - maxC_raw))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via FFT convolutions
        a = 1.0 - h
        w_rev = w[::-1]
        conv1 = signal.fftconvolve(w_rev, a, mode="full")
        conv2 = signal.fftconvolve(w, h, mode="full")
        offset = len(w) - 1  # = 2n-2
        g1 = conv1[offset:offset + n]
        g2 = conv2[offset:offset + n]
        grad = g1 - g2

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # learning‑rate schedule (0.1 → 0.001)
        lr = 0.1 * (1.0 - it / max_iters) + 0.001

        h = h - lr * momentum
        h = _project_to_feasible(h, n, tol=1e-10, max_iter=30)

    # final local refinement with hill‑climb (still respecting time budget)
    limit = time_limit
    best_h, best_c = _hill_climb(best_h, max_steps=2000, time_limit=limit)
    return best_h, best_c

# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load any provided initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is None:
        n0 = 80
        init_h = np.full(n0, 0.5)
    else:
        n0 = init_h.size

    best_h = None
    best_c = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Coarse optimisation on the initial grid size
    # ------------------------------------------------------------------
    coarse_time = min(0.25 * budget_s, time_left())
    limit_abs = start_time + coarse_time
    h_opt, c_opt = _gradient_descent(init_h, n0, limit_abs, rng)
    if c_opt < best_c:
        best_h, best_c, best_n = h_opt, c_opt, n0

    # ------------------------------------------------------------------
    # Upscale and refine on larger grids
    # ------------------------------------------------------------------
    for n in (128, 192, 256, 384, 512):
        if time_left() < 5.0:
            break
        base = _upscale_h(best_h, n)
        base = _project_to_feasible(base, n)

        sigma = 0.4
        for attempt in range(3):
            if time_left() < 5.0:
                break
            perturb = sigma * rng.standard_normal(n)
            start_h = base + perturb
            start_h = _project_to_feasible(start_h, n)

            remaining = time_left()
            # give at most 20 s or 40 % of what’s left to this run
            alloc = min(remaining * 0.4, 20.0)
            limit_abs = time.time() + alloc
            h_opt, c_opt = _gradient_descent(start_h, n, limit_abs, rng)

            if c_opt < best_c:
                best_h, best_c, best_n = h_opt, c_opt, n
            sigma *= 0.7

    # ------------------------------------------------------------------
    # Final hill‑climb on the overall best solution (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 2.0:
        limit_abs = time.time() + min(time_left() - 0.5, 10.0)
        best_h, best_c = _hill_climb(best_h, max_steps=5000, time_limit=limit_abs)

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
