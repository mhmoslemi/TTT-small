import numpy as np

def _max_count(h: np.ndarray) -> int:
    """Maximum overlap count = max_k Σ h_i (1-h_{i+k}) for binary h."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))

def _binary_from_initial(init_h: np.ndarray, n: int, target: int, rng: np.random.Generator) -> np.ndarray:
    """Convert a possibly fractional parent to a feasible binary vector."""
    bin_h = (init_h >= 0.5).astype(int)
    cur = int(bin_h.sum())
    diff = target - cur
    if diff > 0:
        # turn some zeros into ones – pick zeros with largest original values
        zeros = np.where(bin_h == 0)[0]
        if zeros.size:
            vals = init_h[zeros]
            order = np.argsort(-vals)  # descending
            to_flip = zeros[order[:diff]]
            bin_h[to_flip] = 1
    elif diff < 0:
        # turn some ones into zeros – pick ones with smallest original values
        ones = np.where(bin_h == 1)[0]
        if ones.size:
            vals = init_h[ones]
            order = np.argsort(vals)  # ascending
            to_flip = ones[order[: -diff]]
            bin_h[to_flip] = 0
    return bin_h

def _rand_swap(h: np.ndarray, rng: np.random.Generator):
    n = h.size
    i = rng.integers(n)
    if h[i]:
        zeros = np.where(h == 0)[0]
        if zeros.size == 0:
            return i, i
        j = rng.choice(zeros)
    else:
        ones = np.where(h == 1)[0]
        if ones.size == 0:
            return i, i
        j = rng.choice(ones)
    return int(i), int(j)

    """Swap a 1 that participates in the current worst shift with any 0."""
    n = h.size
    max_idx = int(np.argmax(cur_corr))
    k = max_idx - (n - 1)                # shift value (may be negative)

    if k >= 0:
        i_cand = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
    else:
        kk = -k
        i_cand = np.where((h[kk:] == 1) & (h[:n - kk] == 0))[0]

    if i_cand.size:
        i = rng.choice(i_cand)
        zeros = np.where(h == 0)[0]
        if zeros.size == 0:
            i, i = int(i), int(i)      # dummy – will be ignored
        else:
            j = rng.choice(zeros)
            return int(i), int(j)

    # fall back to a random feasible swap
    return _rand_swap(h, rng)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function (binary vector) on [0,2] with integral 1
    that minimizes the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Problem size and target bound
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 360                     # a comfortable size for the budget
    target_sum = n // 2
    dx = 2.0 / n
    target_c = 0.3808
    target_cnt = int(math.floor(target_c / dx + 1e-12))

    # ----------------------------------------------------------------------
    # Helper to keep the current best solution
    # ----------------------------------------------------------------------
    best_h = None
    best_cnt = 10 ** 9

    # ----------------------------------------------------------------------
    # Initialise from parent if available
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n:
            init_arr = np.resize(init_arr, n)
        h0 = _binary_from_initial(init_arr, n, target_sum, rng)
        cnt0 = _max_count(h0)
        best_h, best_cnt = h0.copy(), cnt0

    # ----------------------------------------------------------------------
    # Core simulated‑annealing search
    # ----------------------------------------------------------------------
    def _anneal(start_vec: np.ndarray, start_cnt: int, time_limit: float):
        """Return a locally improved pair (vector, count)."""
        h = start_vec.copy()
        cur_cnt = start_cnt
        cur_corr = np.correlate(h, 1 - h, mode="full")
        T = max(1.0, cur_cnt * 0.02)   # initial temperature
        decay = 0.9995

        it = 0
        while True:
            it += 1
            if time.time() - start >= time_limit:
                break

            # choose swap type
            if rng.random() < 0.65:
                i, j = _targeted_swap(h, cur_corr, rng)
            else:
                i, j = _rand_swap(h, rng)

            if i == j:
                continue

            # propose
            h[i], h[j] = h[j], h[i]
            new_corr = np.correlate(h, 1 - h, mode="full")
            new_cnt = int(np.max(new_corr))
            delta = new_cnt - cur_cnt

            if delta <= 0 or rng.random() < math.exp(-delta / T):
                # accept
                cur_cnt = new_cnt
                cur_corr = new_corr
                if new_cnt < best_cnt:
                    # update outer best
                    nonlocal best_h, best_cnt
                    best_h = h.copy()
                    best_cnt = new_cnt
                    if best_cnt <= target_cnt:
                        return h, cur_cnt
            else:
                # reject – revert
                h[i], h[j] = h[j], h[i]

            T *= decay
        return h, cur_cnt

    # ----------------------------------------------------------------------
    # Repeated restarts until time runs out or target reached
    # ----------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # random start
        perm = rng.permutation(n)
        vec = np.zeros(n, dtype=int)
        vec[perm[:target_sum]] = 1
        cnt = _max_count(vec)

        # anneal this start
        _ = _anneal(vec, cnt, budget_s * 0.95)

        if best_cnt <= target_cnt:
            break

    # ----------------------------------------------------------------------
    # Small deterministic hill climbing on the current best (pure improvements)
    # ----------------------------------------------------------------------
    if best_h is not None:
        h = best_h.copy()
        cur_cnt = best_cnt
        improved = True
        while improved and time.time() - start < budget_s * 0.99:
            improved = False
            for _ in range(2000):
                i, j = _rand_swap(h, rng)
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                new_cnt = _max_count(h)
                if new_cnt < cur_cnt:
                    cur_cnt = new_cnt
                    best_h = h.copy()
                    best_cnt = cur_cnt
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]

    # ----------------------------------------------------------------------
    # Final feasibility check and reporting
    # ----------------------------------------------------------------------
    if best_h is None:
        # fallback – a simple random feasible vector
        perm = rng.permutation(n)
        best_h = np.zeros(n, dtype=int)
        best_h[perm[:target_sum]] = 1
        best_cnt = _max_count(best_h)

    final_c = best_cnt * dx
    return best_h.astype(float).tolist(), float(final_c), int(n)
