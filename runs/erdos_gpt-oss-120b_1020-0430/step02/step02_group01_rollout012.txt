import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Random feasible h (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h_{i+k}
        grad[k:]       -= h[: n - k]    # subtract h_{i‑k}
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h_start, rng, max_iters=3000, step_scale=0.5):
    """Hill‑climb using gradient‑directed transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def enforce_sum(h, rng):
    """Adjust a vector to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

        # local gradient‑directed hill‑climb
        # ------------------------------------------------------------------
        h, c = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # increasing‑β SLSQP refinement
        # ------------------------------------------------------------------
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() - start > 0.96 * budget_s:
                break
            maxit = 600 if beta >= 200 else 400
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # fine gradient refinement
        # ------------------------------------------------------------------
        if time.time() - start < 0.97 * budget_s:
            h = gradient_refine(h, rng, step=0.02, max_iter=8)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # ------------------------------------------------------------------
        # final short guided hill‑climb
        # ------------------------------------------------------------------
        if time.time() - start < 0.98 * budget_s:
            h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.3)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # optional up‑sampling refinement (if time left)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.99 * budget_s:
        if best_n * 2 <= 240:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2000, step_scale=0.3)
            for beta in (5.0, 20.0, 80.0, 200.0):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # final local search with tiny random transfers (remaining budget)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 84
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
