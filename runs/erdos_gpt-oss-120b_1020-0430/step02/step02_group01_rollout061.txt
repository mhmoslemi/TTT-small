def softmax_obj_and_grad(h, beta):
    n = h.size
    bc = np.correlate(h, 1 - h, mode='full')
    max_bc = np.max(bc)
    exp_vals = np.exp(beta * (bc - max_bc))
    sum_exp = np.sum(exp_vals)
    obj = (max_bc + np.log(sum_exp)) / beta
    # compute weights
    w = exp_vals / sum_exp  # shape (2n-1,)
    # gradient
    grad = np.zeros_like(h)
    # shift index mapping
    # for shift idx from 0 to 2*n-2, corresponding s = idx - (n-1)
    for idx, s in enumerate(range(-(n-1), n)):
        weight = w[idx]
        if weight == 0:
            continue
        if s >= 0:
            # part1: i from 0 to n-1-s where i+s < n
            i_start = 0
            i_end = n - s  # exclusive
            # add weight * (1 - h[i + s]) to grad[i]
            grad[i_start:i_end] += weight * (1 - h[i_start + s:i_end + s])
            # part2: i from s to n, where i - s >=0
            i_start2 = s
            i_end2 = n
            grad[i_start2:i_end2] -= weight * h[i_start2 - s:i_end2 - s]
        else:
            # s negative
            # s = -t, t>0
            t = -s
            # part1: i from t to n where i+s >=0
            i_start = t
            i_end = n
            grad[i_start:i_end] += weight * (1 - h[i_start + s:i_end + s])  # i_start+s = i_start - t
            # part2: i from 0 to n - t where i - s < n
            i_start2 = 0
            i_end2 = n - t
            grad[i_start2:i_end2] -= weight * h[i_start2 - s:i_end2 - s]  # i - s = i + t
        # Note: careful with slice indices; we used same h indexing.
    # Because obj = (max_bc + log sum exp)/beta, gradient is (1/beta)*sum w_s * grad_bc_s
    grad = grad / beta
    return obj, grad

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n-1)
    k = idx

    g = np.zeros_like(h)
    if k >= 0:
        # part from first term: i from 0 to n-k-1: derivative w.r.t h[i] is 1 - h[i+k]
        g[:n-k] += 1 - h[k:]
        # part from second term: derivative w.r.t h[i] for i >= k: -h[i - k]
        g[k:] -= h[:n-k]
    else:
        t = -k
        # i from t to n-1: derivative (1 - h[i - t])
        g[t:] += 1 - h[:n-t]
        # i from 0 to n-t-1: derivative -h[i + t]
        g[:n-t] -= h[t:]
    return k, g

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def worst_gradient(h):
    """Return (k, g) where k is index of worst shift and g = ∇ O(k)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    g = np.zeros_like(h)
    if k >= 0:
        # first part: i in [0, n-k)
        g[: n - k] += 1 - h[k:]
        # second part: i in [k, n)
        g[k:] -= h[: n - k]
    else:
        t = -k
        # i in [t, n)
        g[t:] += 1 - h[: n - t]
        # i in [0, n-t)
        g[: n - t] -= h[t:]
    return k, g


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5, guided=False):
    """Hill‑climb (accept only improvements)."""
    h = h_start.copy()
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)

    for it in range(max_iters):
        if guided:
            _, g = worst_gradient(h)
            pos = np.where(g > 1e-12)[0]
            neg = np.where(g < -1e-12)[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = step * max_eps * rng.random()
                    new_h = h.copy()
                    new_h[i] -= eps
                    new_h[j] += eps
                else:
                    new_h = perturb_h(h, rng, scale=step)
            else:
                new_h = perturb_h(h, rng, scale=step)
        else:
            new_h = perturb_h(h, rng, scale=step)

        new_c = compute_c5(new_h)
        # accept if improvement over current position
        if new_c < curr_c:
            h = new_h
            curr_c = new_c
        # remember best ever seen
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()

        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)

    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        k, g = worst_gradient(h)
        # take step opposite to gradient of O(k)
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def upsample_linear(h):
    """Double the resolution by linear interpolation."""
    n = h.size
    h2 = np.empty(2 * n)
    h2[::2] = h
    for idx in range(1, 2 * n, 2):
        left = h[idx // 2]
        right = h[idx // 2 + 1] if (idx // 2 + 1) < n else h[idx // 2]
        h2[idx] = (left + right) / 2.0
    return h2


def enforce_sum(h, rng):
    """Clip to [0,1] and repair the linear‑sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Use supplied parent solution if available
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # Candidate grid sizes (keep them modest for speed)
    candidate_ns = [96, 120, 144, 168, 192, 240]
    if best_n is not None:

    else:
        order_ns = candidate_ns

    # ------------------------------------------------------------------
    # Main optimisation loop over different discretisations
    # ------------------------------------------------------------------
    for n in order_ns:
        if time.time() - start > budget_s * 0.93:
            break
        attempts = 0
        max_attempts = 15 if n <= 120 else 8
        while attempts < max_attempts and time.time() - start < budget_s * 0.93:
            attempts += 1

            # ---- initialise a feasible vector --------------------------------
            if best_n == n and best_h is not None and attempts == 1:
                h = best_h.copy()
            else:
                if rng.random() < 0.4:

        best_h = np.full(best_n, 0.5)
