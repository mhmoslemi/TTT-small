import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Global variable for SLSQP constraints (set before each call)
# ----------------------------------------------------------------------
_current_n = None

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9,
                         max_iter: int = 500) -> np.ndarray:
    """
    Clip h to [0,1] and enforce sum(h) = n/2 by a water‑filling scheme.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)

    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a step function h (already feasible)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

# ----------------------------------------------------------------------
# SLSQP helpers (use global _current_n)
# ----------------------------------------------------------------------
def _objective(x: np.ndarray) -> float:
    return float(x[-1])

def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = _current_n
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x: np.ndarray) -> float:
    n = _current_n
    return float(np.sum(x[:n]) - (n / 2.0))

def _solve_slsqp(initial_h: np.ndarray, max_iter: int = 1200) -> tuple:
    """Run SLSQP on a feasible (or not) starting point."""
    global _current_n
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    _current_n = n
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP",
                            bounds=bounds, constraints=constraints, options=opts)
    _current_n = None
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

# ----------------------------------------------------------------------
# Upscaling utility
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# ----------------------------------------------------------------------
# Binary hill‑climbing (swap based)
# ----------------------------------------------------------------------
def _binary_hill_climb(h_bin: np.ndarray, time_left_func, target=0.3808) -> tuple:
    """Random swap hill climbing on a binary vector."""
    n = h_bin.size
    best_h = h_bin.copy()
    best_c5 = _c5_from_h(best_h)
    # pre‑compute positions
    ones = np.where(best_h == 1.0)[0]
    zeros = np.where(best_h == 0.0)[0]
    it = 0
    while time_left_func() > 1.0 and best_c5 > target and it < 20000:
        it += 1
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        best_h[i] = 0.0
        best_h[j] = 1.0
        new_c5 = _c5_from_h(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            ones = np.where(best_h == 1.0)[0]
            zeros = np.where(best_h == 0.0)[0]
        else:
            # revert
            best_h[i] = 1.0
            best_h[j] = 0.0
    return best_h, best_c5

def _binarize_h(h: np.ndarray) -> np.ndarray:
    """Return a binary vector with exactly half ones (choose largest entries)."""
    n = h.size
    idx = np.argsort(-h)[: n // 2]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    except NameError:

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target_c5 = 0.3808

    def update(h_vec, val):
        nonlocal best_h, best_c5, best_n
        if val < best_c5:
            best_h = h_vec.copy()
            best_c5 = float(val)
            best_n = h_vec.size

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n_coarse = 128
