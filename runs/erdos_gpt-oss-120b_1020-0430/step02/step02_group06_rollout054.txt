def project_to_feasible(h):
    n = len(h)
    target = n/2
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # water-filling: adjust to target
    # We'll use simple loop iteration with max_iter 200 to converge.
    for _ in range(200):
        sum_h = h.sum()
        diff = sum_h - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to reduce sum; subtract equally from all entries > 0
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust a few entries
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

# ---------- basic evaluation ----------

def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.size
    m = 2 * n - 1                     # length of linear correlation
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(B), n=m)
    return corr.max() * (2.0 / n)


# ---------- feasibility projection ----------

def _project_to_feasible(h: np.ndarray, max_iter: int = 300) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- scaling of a coarse construction ----------

def _upscale_h(coarse_h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_target = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_target, x_coarse, coarse_h)


# ---------- gradient (sub‑gradient) refinement ----------

def _refine_gradient(h: np.ndarray, iters: int, lr0: float = 0.6, decay: float = 0.998):
    """Sub‑gradient descent on the worst overlap shift."""
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(best_h)

    for it in range(iters):
        m = 2 * n - 1
        H = np.fft.rfft(h, n=m)
        B = np.fft.rfft(1.0 - h, n=m)
        corr = np.fft.irfft(H * np.conj(B), n=m)
        k = int(np.argmax(corr) - (n - 1))          # worst shift (can be negative)

        g = np.ones(n)
        if k > 0:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        elif k < 0:
            kk = -k
            g[: n - kk] -= h[kk:]
            g[kk:] -= h[: n - kk]
        else:                     # k == 0
            g -= 2 * h

        lr = lr0 * (decay ** it)
        h = _project_to_feasible(h - lr * g)

        val = _c5_from_h(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val


# ---------- binary helpers ----------

def _binary_from_continuous(h: np.ndarray) -> np.ndarray:
    """Set the top half of entries to 1, rest to 0 (preserves the mass constraint)."""
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h


def _binary_hill_climb(h_bin: np.ndarray, swaps: int, t_start: float = 0.02, t_end: float = 1e-4):
    """Simulated‑annealing hill‑climb on binary vectors (preserving exactly n/2 ones)."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_val = _c5_from_h(cur)

    best = cur.copy()
    best_val = cur_val

    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur < 0.5)[0].tolist())

    for s in range(swaps):
        if not ones or not zeros:
            break
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))

        # perform tentative swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = _c5_from_h(cur)

        # temperature schedule (linear)
        T = t_start + (t_end - t_start) * (s / swaps)

        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            # keep the swap, update sets and current value
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1.0
            cur[j] = 0.0

    return best, best_val
