    v = np.clip(v, 0.0, 1.0)
    # iterative water-filling
    max_iter = 1000
    for _ in range(max_iter):
        sum_v = v.sum()
        diff = target - sum_v
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (v > 0.0) & (v < 1.0)
        if not np.any(free):
            # all at bounds, cannot adjust; break
            break
        delta = diff / free.sum()
        # add delta but ensure we keep within bounds
        v[free] += delta
        # clip after addition
        v = np.clip(v, 0.0, 1.0)
    # final rounding for small errors
    v = np.clip(v, 0.0, 1.0)
    # sometimes due to rounding sum not exactly target; we can adjust using deterministic loop small epsilon (like previous).
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # adjust last element
        for i in range(v.size):
            if (diff > 0 and v[i] < 1.0) or (diff < 0 and v[i] > 0.0):
                add = min(abs(diff), 1.0 - v[i]) if diff > 0 else -min(abs(diff), v[i])
                v[i] += add
                diff = target - v.sum()
                if abs(diff) < 1e-12:
                    break

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def build_h_from_a(a: np.ndarray) -> np.ndarray:
    """Construct the full symmetric h from the first half a (n even)."""
    d = a.size
    n = d * 2
    h = np.empty(n, dtype=float)
    h[:d] = a
    h[d:] = 1.0 - a[::-1]
    return h


def compute_c5_from_a(a: np.ndarray) -> float:
    """C5 computed via the symmetric representation."""
    return compute_c5(build_h_from_a(a))


def symmetrize_h_to_a(h: np.ndarray) -> np.ndarray:
    """Given any h, return a symmetric a (first half) by averaging opposite pairs."""
    n = h.size
    d = n // 2
    a = np.empty(d, dtype=float)
    for i in range(d):
        a[i] = (h[i] + (1.0 - h[n - 1 - i])) * 0.5
    # numerical safety
    return np.clip(a, 0.0, 1.0)


# ---------- local search on a ----------
def perturb_a(a: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε between two coordinates (preserves sum)."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a.copy()
    max_eps = min(a[i], 1.0 - a[j])
    if max_eps <= 0.0:
        return a.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    b = a.copy()
    b[i] -= eps
    b[j] += eps
    return b


def hill_climb_a(a_start: np.ndarray, rng: np.random.Generator,
                 max_iters: int = 2000, start_step: float = 0.5):
    """Simple deterministic hill‑climb on a (accept only improvements)."""
    a = a_start.copy()
    best_a = a.copy()
    best_c = compute_c5_from_a(a)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_a(a, rng, scale=step)
        c = compute_c5_from_a(cand)
        if c < best_c:
            best_c = c
            best_a = cand.copy()
            a = cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_a, best_c


# ---------- smooth surrogate & SLSQP ----------
CURRENT_BETA = None  # will be set before each SLSQP run


def softmax_obj_a(a: np.ndarray) -> float:
    """Log‑sum‑exp surrogate of max_{k} Σ_i h_i (1‑h_{i+k}) for symmetric h."""
    h = build_h_from_a(a)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_a(a0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * a0.size
    res = minimize(
        softmax_obj_a,
        a0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    a_opt = np.clip(res.x, 0.0, 1.0)
    return a_opt


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- determine dimension and initial a ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        if n % 2 == 1:               # enforce even length for symmetry
            n -= 1
            h0 = h0[:n]
        d = n // 2
        a0 = symmetrize_h_to_a(h0)
    else:
        n = 180
        d = n // 2
        a0 = rng.random(d)            # uniform start
    a0 = np.clip(a0, 0.0, 1.0)

    # ---- differential evolution on a ----
    bounds = [(0.0, 1.0)] * d
    popsize = 5
    init_pop = rng.random((popsize * d, d))
    init_pop[0] = a0

    def obj_de(x):
        return compute_c5_from_a(x)

    # callback to respect overall time budget
    def de_callback(xk, convergence):

    result = differential_evolution(

    beta_list = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    final_h = build_h_from_a(global_best_a)
    final_h = np.clip(final_h, 0.0, 1.0)       # safety
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(n)
