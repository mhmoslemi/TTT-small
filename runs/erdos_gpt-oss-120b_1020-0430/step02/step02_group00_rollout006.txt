def greedy_descent(h_int, max_iters, time_check_interval=100):
    n = h_int.size
    best_val = compute_c5(h_int)
    best_h = h_int.copy()
    for it in range(max_iters):
        # if time exceeded abort
        # compute correlation
        h_float = h_int.astype(np.float64)
        corr = np.correlate(h_float, 1.0 - h_float, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # compute bad positions i where pair contributes
        if lag >= 0:
            i_candidates = np.where((h_int[:n - lag] == 1) & (h_int[lag:] == 0))[0]
        else:
            lag_abs = -lag
            i_candidates = np.where((h_int[lag_abs:] == 1) & (h_int[:n - lag_abs] == 0))[0] + lag_abs
        if i_candidates.size == 0:
            break
        i = np.random.choice(i_candidates)
        # compute good positions j where swapping reduces
        if lag >= 0:
            j_candidates = np.where((h_int[:n - lag] == 0) & (h_int[lag:] == 1))[0]
        else:
            lag_abs = -lag
            j_candidates = np.where((h_int[lag_abs:] == 0) & (h_int[:n - lag_abs] == 1))[0] + lag_abs
        if j_candidates.size > 0:
            j = np.random.choice(j_candidates)
        else:

            j = np.random.choice(zero_idxs)
        # perform swap
        h_int[i], h_int[j] = h_int[j], h_int[i]

def upscale_binary(h_prev, n):
    n_prev = h_prev.size
    x_prev = (np.arange(n_prev) + 0.5) / n_prev * 2.0
    x_new = (np.arange(n) + 0.5) / n * 2.0
    h_interp = np.interp(x_new, x_prev, h_prev)
    # set top half to 1
    idx = np.argpartition(h_interp, -n//2)[-n//2:]
    h_bin = np.zeros(n, dtype=np.int8)
    h_bin[idx] = 1
    return h_bin

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h_int: np.ndarray) -> float:
    """Return C5 for a binary (0/1) step function."""
    h = h_int.astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h_int.size)


def upscale_binary(prev_vals: np.ndarray, n: int) -> np.ndarray:
    """Interpolate previous (float) values onto a finer grid and keep the top half."""
    n0 = prev_vals.size
    x0 = (np.arange(n0) + 0.5) / n0 * 2.0
    x1 = (np.arange(n) + 0.5) / n * 2.0
    interp = np.interp(x1, x0, prev_vals)
    # keep exactly n/2 ones (largest values)
    idx = np.argpartition(interp, -n // 2)[-n // 2 :]
    out = np.zeros(n, dtype=np.int8)
    out[idx] = 1
    return out


def binary_from_float(arr: np.ndarray) -> np.ndarray:
    """Make a binary vector with exactly half ones, taking the largest entries."""
    n = arr.size
    idx = np.argpartition(arr, -n // 2)[-n // 2 :]
    out = np.zeros(n, dtype=np.int8)
    out[idx] = 1
    return out


# ----------------------------------------------------------------------
# Greedy descent – tries to eliminate the worst lag
# ----------------------------------------------------------------------
def greedy_descent(h_int: np.ndarray, max_iters: int, time_limit: float) -> tuple:
    """
    Local greedy improvement.
    Returns (best_h, best_val).
    """
    n = h_int.size
    best_h = h_int.copy()
    best_val = compute_c5(best_h)
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        # correlation and worst lag
        h_float = best_h.astype(np.float64)
        corr = np.correlate(h_float, 1.0 - h_float, mode="full")
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)

        # collect positions that contribute to this lag
        if lag >= 0:
            i_range = np.arange(0, n - lag)
            bad = i_range[(best_h[i_range] == 1) & (best_h[i_range + lag] == 0)]
            good = i_range[(best_h[i_range] == 0) & (best_h[i_range + lag] == 1)]
        else:
            d = -lag
            i_range = np.arange(d, n)
            bad = i_range[(best_h[i_range] == 1) & (best_h[i_range - d] == 0)]
            good = i_range[(best_h[i_range] == 0) & (best_h[i_range - d] == 1)]

        if bad.size == 0:        # nothing to improve for this lag
            break

        i = int(np.random.choice(bad))
        if good.size > 0:
            j = int(np.random.choice(good))
        else:
            # fallback: any zero position
            zero_idxs = np.flatnonzero(best_h == 0)
            j = int(np.random.choice(zero_idxs))

        # perform swap
        best_h[i], best_h[j] = best_h[j], best_h[i]
        new_val = compute_c5(best_h)

        if new_val < best_val:
            best_val = new_val
        else:
            # revert – greedy step only keeps improvements
            best_h[i], best_h[j] = best_h[j], best_h[i]

    return best_h, best_val


# ----------------------------------------------------------------------
# Simulated annealing – occasional uphill moves
# ----------------------------------------------------------------------
def anneal(h_int: np.ndarray, time_budget: float,
           start_temp: float = 0.5, end_temp: float = 0.001) -> tuple:
    """
    Simple simulated‑annealing on binary vectors.
    Returns (best_h, best_val).
    """
    n = h_int.size
    cur_h = h_int.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = start_temp * (1 - frac) + end_temp * frac

        # choose a random 1 and a random 0 to swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))

        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = compute_c5(cur_h)
        delta = new_val - cur_val

        if delta < 0 or np.random.rand() < np.exp(-delta / (temp + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    return best_h, best_val

# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start_time))

    cand_n = [256, 512, 768, 1024]
    if init_h_float is not None and init_h_float.size not in cand_n:
        cand_n.insert(0, init_h_float.size)
    cand_n = [n for n in cand_n if n % 2 == 0]  # must be even

    best_h = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Main optimisation loop over resolutions
    # ------------------------------------------------------------------
    for n in cand_n:
        if time_left() < 5.0:
            break

        # ----- initialise a binary vector for this n -----
        if init_h_float is not None and init_h_float.size == n:
            h_bin = binary_from_float(init_h_float)
        elif best_h is not None:
            # upscale previous best (which is binary) and keep half ones
            h_bin = upscale_binary(best_h.astype(float), n)
        else:
            # fresh random balanced vector
            h_bin = np.zeros(n, dtype=np.int8)
            h_bin[: n // 2] = 1
            np.random.shuffle(h_bin)

        cur_sum = int(h_bin.sum())
        target = n // 2
        if cur_sum != target:
            diff = target - cur_sum
            if diff > 0:
                zero_idx = np.flatnonzero(h_bin == 0)
                choose = np.random.choice(zero_idx, diff, replace=False)
                h_bin[choose] = 1
            else:
                one_idx = np.flatnonzero(h_bin == 1)
                choose = np.random.choice(one_idx, -diff, replace=False)
                h_bin[choose] = 0

        # ----- greedy refinement -----
        # allocate a modest portion of remaining budget for greedy steps
        greedy_time = min(0.35 * time_left(), 30.0)
        # translate time budget into iteration count (heuristic)
        iters = int(min(2000, max(200, (greedy_time * 20))))  # ~20 iters per second
