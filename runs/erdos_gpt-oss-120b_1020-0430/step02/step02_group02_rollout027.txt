def de_optimize(base_h, n, time_limit, max_gen=200, popsize=20, F=0.8, CR=0.7):

    # generate initial population:
    pop = np.empty((popsize, n), dtype=float)
    for i in range(popsize):
        # random uniform then project
        rand = np.random.rand(n)
        pop[i] = _project_to_feasible(rand, n)
    # also insert base_h as one individual
    pop[0] = _project_to_feasible(base_h, n)
    # evaluate
    pop_c = np.empty(popsize, dtype=float)
    for i in range(popsize):
        pop_c[i] = _c5_from_h(pop[i])
    best_idx = np.argmin(pop_c)
    best_h = pop[best_idx].copy()
    best_c = pop_c[best_idx]

    gen = 0
    while gen < max_gen and time.time() < time_limit:
        gen += 1
        for i in range(popsize):
            # select a,b,c distinct from i
            idxs = np.arange(popsize)
            idxs = idxs[idxs != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            # mutation: best + F*(pop[a] - pop[b])
            mutant = best_h + F * (pop[a] - pop[b])
            # clip bounds [0,1] then project to sum
            mutant = _project_to_feasible(mutant, n)
            # crossover
            trial = pop[i].copy()
            j_rand = np.random.randint(n)
            # generate mask
            cross_mask = np.random.rand(n) < CR
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            # project trial again to feasible
            trial = _project_to_feasible(trial, n)
            # evaluate
            trial_c = _c5_from_h(trial)
            if trial_c <= pop_c[i]:
                pop[i] = trial
                pop_c[i] = trial_c
                if trial_c < best_c:
                    best_c = trial_c
                    best_h = trial.copy()
        # time check
        if time.time() >= time_limit:
            break
    return best_h, best_c

def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # Determine indices that can be adjusted (not at bound that would be violated by moving in the sign of diff)
        if diff > 0:
            # need to reduce sum: reduce from indices above 0
            free = np.where(h > 0)[0]
        else:
            # need to increase sum: increase indices below 1
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction (tiny)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation (max overlap integral)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a given feasible h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Linear interpolation to a finer grid
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Differential Evolution (population based global search)
# ----------------------------------------------------------------------
def _de_optimize(base_h: np.ndarray, n: int, deadline: float,
                 max_gen: int = 200, popsize: int = 20,
                 F: float = 0.8, CR: float = 0.7) -> tuple:
    """Simple DE that respects the feasibility constraints via projection."""
    # initialise population
    pop = np.empty((popsize, n), dtype=float)
    for i in range(popsize):
        pop[i] = _project_to_feasible(np.random.rand(n), n)
    pop[0] = _project_to_feasible(base_h, n)      # ensure one good seed

    # evaluate
    pop_c = np.empty(popsize, dtype=float)
    for i in range(popsize):
        pop_c[i] = _c5_from_h(pop[i])

    best_idx = int(np.argmin(pop_c))
    best_h = pop[best_idx].copy()
    best_c = pop_c[best_idx]

    gen = 0
    while gen < max_gen and time.time() < deadline:
        gen += 1
        for i in range(popsize):
            # choose three distinct indices different from i
            idxs = np.arange(popsize)
            idxs = idxs[idxs != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)

            # mutation (best/1)
            mutant = best_h + F * (pop[a] - pop[b])
            mutant = _project_to_feasible(mutant, n)

            # crossover
            trial = pop[i].copy()
            j_rand = np.random.randint(n)
            cross_mask = np.random.rand(n) < CR
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = _project_to_feasible(trial, n)

            # evaluation
            trial_c = _c5_from_h(trial)
            if trial_c <= pop_c[i]:
                pop[i] = trial
                pop_c[i] = trial_c
                if trial_c < best_c:
                    best_c = trial_c
                    best_h = trial.copy()
        # early exit if time exhausted
        if time.time() >= deadline:
            break
    return best_h, best_c


# ----------------------------------------------------------------------
# Simple hill‑climb (local improvement)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, deadline: float, max_steps: int = 50000) -> tuple:
    """Transfer a small amount of mass between two coordinates if it improves C5."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    steps = 0
    while steps < max_steps and time.time() < deadline:
        steps += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount we can move from i to j while staying feasible
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5  # up to half of the possible transfer
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        # no need to re‑project: sum unchanged and bounds respected by construction
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Load any provided initial construction
    # ------------------------------------------------------------------
    try:
        GLOBAL_INITIAL_H = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL_H = None

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # Candidate grid sizes
    # ------------------------------------------------------------------
    n_candidates = []
    if GLOBAL_INITIAL_H is not None:
        n_candidates.append(GLOBAL_INITIAL_H.size)
    # multiples of the initial size (or of 80 if none)
    base_n = GLOBAL_INITIAL_H.size if GLOBAL_INITIAL_H is not None else 80
    for mul in (1, 2, 3, 4, 5, 6):
        n = base_n * mul
        if n <= 512:
            n_candidates.append(n)
    # hand‑picked extra even sizes
    n_candidates.extend([96, 112, 128, 144, 160, 192, 224, 256, 320, 384, 448, 512])
    n_candidates = sorted(set(n_candidates))

    # ------------------------------------------------------------------
    # Optimisation loop over grid sizes
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time.time() >= deadline:
            break

        # Build starting h for this resolution
        if GLOBAL_INITIAL_H is not None:
            if GLOBAL_INITIAL_H.size != n:
                base_h = _upscale_h(GLOBAL_INITIAL_H, n)
            else:
                base_h = GLOBAL_INITIAL_H.copy()
        else:
            base_h = np.full(n, 0.5)

        # Global search (DE)
        h_de, c5_de = _de_optimize(base_h, n, deadline,
                                   max_gen=250, popsize=25,
                                   F=0.9, CR=0.7)

        # Local refinement
        if time.time() < deadline:
            h_loc, c5_loc = _hill_climb(h_de, deadline, max_steps=40000)
        else:
            h_loc, c5_loc = h_de, c5_de

        # Record improvement
        if c5_loc < best_c5:
            best_h, best_c5, best_n = h_loc, c5_loc, n

        # Early stop if we already beat the target
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Optional upscaling of the best found solution for finer resolution
    # ------------------------------------------------------------------
    if best_h is not None and best_n * 2 <= 512 and time.time() < deadline:
        n_fine = best_n * 2
        h_fine = _upscale_h(best_h, n_fine)
        h_fine = _project_to_feasible(h_fine, n_fine)
        h_fine, c5_fine = _hill_climb(h_fine, deadline, max_steps=50000)
        if c5_fine < best_c5:
            best_h, best_c5, best_n = h_fine, c5_fine, n_fine

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
