import numpy as np
from scipy import optimize

def _project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid=(lo+hi)*0.5
        proj=np.clip(v-mid,0.0,1.0)
        s=proj.sum()
        if s>target:
            lo=mid
        else:
            hi=mid
    proj=np.clip(v-lo,0.0,1.0)
    diff=proj.sum()-target
    if abs(diff)>1e-12:
        free=np.where((proj>1e-9)&(proj<1-1e-9))[0]
        if free.size:
            proj[free]-=diff/free.size
            proj=np.clip(proj,0.0,1.0)
    return proj

def _c5_from_h(h):
    n=h.size
    return np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n)

def _solve_h(start_h, max_iter=300):
    n=start_h.size
    h0=_project_to_simplex_box(start_h, n/2.0)
    t0=np.max(np.correlate(h0, 1.0-h0, mode='full'))
    x0=np.concatenate([h0, [t0]])
    bounds=[(0.0,1.0)]*n+[(0.0,None)]

    def ineq(x):
        h=x[:n]
        t=x[-1]
        corr=np.correlate(h,1.0-h,mode='full')
        return t - corr
    def eq(x):
        h=x[:n]
        return np.sum(h) - n/2.0
    cons=[{'type':'ineq','fun':ineq},{'type':'eq','fun':eq}]
    opts={'maxiter':max_iter,'ftol':1e-9,'disp':False}
    res=optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options=opts)
    if not res.success:
        h_opt=h0
    else:
        h_opt=res.x[:n]
    h_opt=_project_to_simplex_box(h_opt, n/2.0)
    return h_opt, _c5_from_h(h_opt)

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Helper: projection onto the simplex {0<=x_i<=1, sum x_i = target}
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# --------------------------------------------------------------
# C5 evaluation for a discrete step function h on [0,2]
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

# Local refinement via SLSQP (same as previous solution)
def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

# Parametric symmetric piecewise‑linear shape
def _shape_from_params(params: np.ndarray, n: int, K: int):
    """Return (h, C5) built from params.
    params = [u_0,…,u_{K-1}, a] with u_i>=0, a∈[0,1].
    The half‑interval [0,1] is split into K equal segments.
    """
    u = np.clip(params[:K], 1e-12, None)
    a = np.clip(params[K], 0.0, 1.0)
    total_u = u.sum()
    if total_u <= 0.0:
        return None, None

    # monotone heights on [0,1]
    cum = np.cumsum(u)
    y = a * cum / total_u            # y_i at x_i = i/K, i=1..K
    x_half = np.linspace(0.0, 1.0, K + 1)   # K+1 nodes, including 0 and 1
    y_half = np.concatenate([[0.0], y])    # value at x=0 is 0

    # sample full interval [0,2]
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xm = np.where(xs <= 1.0, xs, 2.0 - xs)   # mirror to [0,1]
    h_raw = np.interp(xm, x_half, y_half)

    dx = 2.0 / n
    area = h_raw.sum() * dx
    max_raw = h_raw.max()
    if area <= 0.0 or max_raw > area:
        return None, None

    h = h_raw / area   # scale to make total area = 1, respects 0≤h≤1
    c5 = _c5_from_h(h)
    return h, c5


def _shape_obj(params: np.ndarray, n: int, K: int) -> float:
    _, val = _shape_from_params(params, n, K)
    return 10.0 if val is None else val
