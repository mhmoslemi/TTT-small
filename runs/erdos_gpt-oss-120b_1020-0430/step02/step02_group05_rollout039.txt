def shape(p, n):
    # generate points x
    x = (np.arange(n) + 0.5) * (2.0 / n)
    half_w = (1 - p) / 2.0
    T = (1 + p) / 2.0
    d = np.abs(x - 1.0)
    lead = np.maximum(d - half_w, 0.0)
    h = np.where(d <= half_w, 1.0, 1.0 - (lead / T) ** p)
    return h
```

import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} by a simple bisection on the Lagrange multiplier."""
    lo = np.min(v) - 1.0   # ensures projection at lo gives sum=n
    hi = np.max(v)         # ensures projection at hi gives sum=0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny correction if needed
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discrete step function using FFT."""
    n = h.size
    g = 1.0 - h
    # correlation = convolution of h with reversed g
    g_rev = g[::-1]
    N = 2 * n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    G = np.fft.rfft(np.concatenate([g_rev, np.zeros(n)]))
    conv = np.fft.irfft(H * G)
    # conv length = 2*n, linear part is first 2*n-1 entries
    max_corr = np.max(conv[: 2 * n - 1])
    return max_corr * (2.0 / n)


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Refine a candidate using SLSQP on the original max‑correlation formulation."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    constraints = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


# ------------------------------------------------------------
# Parametric family of shapes (convex tails + possible plateau)
# ------------------------------------------------------------
def _shape(p: float, n: int) -> np.ndarray:
    """
    Symmetric shape with a central plateau of length w = 1-p (p in (0,1]).
    The tails decrease as 1 - (lead/T)^p where lead is distance from plateau.
    """
    x = (np.arange(n) + 0.5) * (2.0 / n)          # mid‑points
    half_w = (1.0 - p) / 2.0
    T = (1.0 + p) / 2.0
    d = np.abs(x - 1.0)                           # distance from centre
    lead = np.maximum(d - half_w, 0.0)
    # avoid division by zero when p==1 (T==1)
    with np.errstate(divide="ignore", invalid="ignore"):
        tail = 1.0 - (lead / T) ** p
    tail = np.where(d <= half_w, 1.0, tail)
    # guarantee exact bounds
    tail = np.clip(tail, 0.0, 1.0)
    return tail

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    Tries a low‑dimensional parametric family, refines a few best candidates
    with SLSQP, and falls back to a simple triangle if needed.
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # candidate resolutions (include parent if it exists)
    n_candidates = []
    if "initial_h_values" in globals():
        n_candidates.append(len(initial_h_values))
    n_candidates.extend([128, 192, 256, 384, 512, 768, 1024])
    n_candidates = sorted(set([n for n in n_candidates if n <= 1024]))

    target = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # coarse grid over p
        p_grid = np.linspace(0.6, 1.0, 41)   # 0.6 .. 1.0 step 0.01
        local_best_p = None
        local_best_c5 = float("inf")

        for p in p_grid:
            if time_left() < 2.0:
                break
            h = _shape(p, n)
            h = _project_to_simplex_box(h, n / 2.0)
            c5 = _c5_from_h(h)
            if c5 < local_best_c5:
                local_best_c5, local_best_p = c5, p

        # refine p with a bounded scalar optimisation
        if time_left() > 5.0 and local_best_p is not None:
            def eval_c5(p):
                h = _shape(p, n)
                h = _project_to_simplex_box(h, n / 2.0)
                return _c5_from_h(h)

            res = minimize_scalar(
                eval_c5,
                bounds=(0.01, 1.0),
                method="bounded",
                options={"xatol": 1e-4, "maxiter": 30},
            )
            if res.success and res.fun < local_best_c5:
                local_best_c5, local_best_p = res.fun, res.x

        # shape built from the best p for this resolution
        if local_best_p is None:
            continue
        h_best = _shape(local_best_p, n)
        h_best = _project_to_simplex_box(h_best, n / 2.0)
        c5_best = _c5_from_h(h_best)

        # local refinement with SLSQP (a few iterations only)
        if time_left() > 5.0:
            h_ref, c5_ref = _solve_h(h_best, max_iter=350)
            if c5_ref < c5_best:
                h_best, c5_best = h_ref, c5_ref

        # a couple of stochastic restarts
        sigma = 0.08
        for _ in range(4):
            if time_left() < 5.0:
                break
            start = h_best + sigma * np.random.randn(n)
            start = _project_to_simplex_box(start, n / 2.0)
            h_ref, c5_ref = _solve_h(start, max_iter=250)
            if c5_ref < c5_best:
                h_best, c5_best = h_ref, c5_ref
                sigma *= 0.85

        # update global best
        if c5_best < best_c5:
            best_c5, best_h, best_n = c5_best, h_best.copy(), n
            if best_c5 < target:
                break

    # final fallback (unlikely to be needed)
    if best_h is None:
        n = 256
        best_h = _shape(1.0, n)          # plain triangle
        best_h = _project_to_simplex_box(best_h, n / 2.0)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
