def anneal_optimize(h0, time_left_func, target_bound):
    best_h = h0.copy()
    n = h0.size
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    # schedule parameters
    sigma = 0.1
    temp = 0.01
    sigma_decay = 0.9995
    temp_decay = 0.9995
    # evaluate once at start
    while time_left_func() > 0.2:
        # propose
        prop = cur_h + sigma * np.random.randn(n)
        # project
        prop = project_to_simplex_box(prop, n/2.0)
        prop_val = compute_c5(prop)
        if prop_val < best_val:
            best_h, best_val = prop.copy(), prop_val
        # accept rule
        if prop_val < cur_val or np.random.rand() < np.exp(-(prop_val - cur_val) / max(temp,1e-12)):
            cur_h, cur_val = prop, prop_val
        # schedule update
        sigma *= sigma_decay
        temp *= temp_decay
        # Occasionally reset sigma if too low
        if sigma < 1e-5:
            sigma = 0.05
        if temp < 1e-6:
            temp = 0.001
        # early break when already below target
        if best_val < target_bound:
            break
    return best_h, best_val

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1.0 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C5 value for a step function h on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 250) -> tuple[np.ndarray, float]:
    """Run a single SLSQP refinement from a feasible start."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=constraints, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse vector onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points and projected to meet the constraints."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _anneal_optimize(h0: np.ndarray, time_left_func, target: float = 0.3808) -> tuple[np.ndarray, float]:
    """
    Simple simulated annealing on the feasible set.
    h0 must already satisfy the constraints.
    """
    best_h = h0.copy()
    best_val = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    n = h0.size

    sigma = 0.15          # step size for proposals
    temp = 0.008          # temperature for Metropolis acceptance

    sigma_decay = 0.9995
    temp_decay = 0.9995
    min_sigma = 1e-5
    min_temp = 5e-7

    while time_left_func() > 0.3:
        # propose a new feasible point
        prop = cur_h + sigma * np.random.randn(n)
        prop = _project_to_simplex_box(prop, n / 2.0)
        prop_val = _c5_from_h(prop)

        if prop_val < best_val:
            best_h, best_val = prop.copy(), prop_val

        # Metropolis acceptance
        if prop_val < cur_val or np.random.rand() < np.exp(-(prop_val - cur_val) / max(temp, 1e-12)):
            cur_h, cur_val = prop, prop_val

        # schedule
        sigma = max(min_sigma, sigma * sigma_decay)
        temp = max(min_temp, temp * temp_decay)

        if best_val < target:
            break

    return best_h, best_val
