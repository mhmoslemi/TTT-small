    else:
        forward[-k:] = h[:n + k]  # because k negative; -k positive
    # backward = h shifted by +k
    backward = np.zeros_like(h)
    if k <= 0:
        # if k negative, i - k = i + |k|
        # So backward[i] = h_{i - k} = h[i+|k|] (shift left by |k|)
        backward[:n + k] = h[-k:]  # because -k = |k|
    else:
        # k positive
        backward[k:] = h[:n - k]
    # Actually need to verify correctness.
    # Let's do systematic:
    #
    # For each index i:
    #   forward[i] = h[i + k] if 0 <= i+k < n else 0
    #   backward[i] = h[i - k] if 0 <= i-k < n else 0

import time
import numpy as np

# ----------------------------------------------------------------------
# Feasibility projection (clip to [0,1] and enforce sum = n/2)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation (max cross‑correlation, scaled)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Gradient of the active correlation shift
# ----------------------------------------------------------------------
def _grad_for_shift(h: np.ndarray, shift_idx: int) -> np.ndarray:
    n = h.size
    k = shift_idx - (n - 1)          # signed shift
    forward = np.zeros_like(h)
    backward = np.zeros_like(h)
    if k >= 0:
        forward[: n - k] = h[k:]
        backward[k:] = h[: n - k]
    else:
        kk = -k
        forward[kk:] = h[: n - kk]
        backward[: n - kk] = h[kk:]
    return 1.0 - (forward + backward)


# ----------------------------------------------------------------------
# Simple projected gradient descent on the max‑correlation objective
# ----------------------------------------------------------------------
def _gradient_descent(start_h: np.ndarray, time_budget: float) -> tuple:
    n = start_h.size
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_from_h(h)
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * (2.0 / n)
        grad = _grad_for_shift(h, max_idx)
        step = 0.4 / (1.0 + 0.001 * it)          # decaying step size
        h_new = h - step * grad
        h_new = _project_to_feasible(h_new, n)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_val = corr_new.max() * (2.0 / n)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        if new_val <= cur_val:
            h = h_new
        it += 1
        if best_val < 0.3808:
            break
    return best_h, best_val

# Local refinement with SLSQP (auxiliary variable t for max correlation)
# ----------------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    # auxiliary variable t = max correlation
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load optional starting construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is not None:
        n = init_h.size
        cur_h = _project_to_feasible(init_h, n)
    else:
        n = 256
        cur_h = np.random.rand(n)
        cur_h = _project_to_feasible(cur_h, n)

    best_h = cur_h.copy()
    best_c5 = _c5_from_h(best_h)

    # ------------------------------------------------------------------
    # Optional quick SLSQP refinement of the initial guess
    # ------------------------------------------------------------------
    if time_left() > 30:
        h_ref, c5_ref = _solve_h(best_h, max_iter=300)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    # ------------------------------------------------------------------
    # Main exploration loop
    # ------------------------------------------------------------------
    while time_left() > 5.0:
        # choose a starting point: perturb current best or fresh random
        if np.random.rand() < 0.7:
            start = best_h + 0.12 * np.random.randn(n)
        else:
            start = np.random.rand(n)
        start = _project_to_feasible(start, n)

        # gradient descent phase (allocate a slice of remaining time)
        gd_budget = min(6.0, time_left() * 0.2)
        h_gd, c5_gd = _gradient_descent(start, gd_budget)

        if c5_gd < best_c5:
            best_h, best_c5 = h_gd, c5_gd

        # occasional SLSQP polish
        if time_left() > 8.0 and c5_gd < best_c5 + 1e-5:
            h_pol, c5_pol = _solve_h(best_h, max_iter=200)
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final high‑precision SLSQP polish if time allows
    # ------------------------------------------------------------------
    if time_left() > 6.0:
        h_fin, c5_fin = _solve_h(best_h, max_iter=800)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_h.size)
