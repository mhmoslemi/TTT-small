   sigma = sigma0

   start = time.time()

       c5 = _c5_from_h(trial)

           best_h, best_c5 = trial, c5

        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break

            # need to reduce sum
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break

        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)

    residual = target - h.sum()

                residual += sub

import time
import numpy as np

# ---- Helper functions -------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete approximation of C5."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _hill_climb(start_h: np.ndarray, timelimit: float, sigma0: float = 0.3,
                decay: float = 0.9995) -> tuple:
    """Simple stochastic hill‑climbing with feasibility projection."""
    n = start_h.size
    best_h = _project_to_feasible(start_h, n)
    best_c5 = _c5_from_h(best_h)
    sigma = sigma0
    start = time.time()
    stagnant = 0

    while time.time() - start < timelimit:
        trial = best_h + sigma * np.random.randn(n)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 + 1e-12 < best_c5:
            best_h, best_c5 = trial, c5
            stagnant = 0
        else:
            stagnant += 1
        sigma *= decay
        if sigma < 1e-7:
            sigma = sigma0
        if stagnant > 2000:
            # random restart to escape a flat region
            trial = np.random.rand(n)
            trial = _project_to_feasible(trial, n)
            c5 = _c5_from_h(trial)
            if c5 < best_c5:
                best_h, best_c5 = trial, c5
                stagnant = 0
            sigma = sigma0
    return best_h, best_c5
