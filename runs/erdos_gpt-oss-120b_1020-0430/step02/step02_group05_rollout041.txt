def shape_a(n, a):
    c = 1.0/(2 - a)
    # compute x positions as midpoints.
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.empty_like(x)
    left_mask = x < a
    right_mask = x > 2 - a
    middle_mask = ~(left_mask | right_mask)
    # left linear: h = (c/a) * x
    h[left_mask] = (c / a) * x[left_mask]
    # middle constant:
    h[middle_mask] = c
    # right linear: h = (c / a) * (2 - x)
    h[right_mask] = (c / a) * (2 - x[right_mask])
    # Ensure sum constraint satisfaction and clipping.
    h = _project_to_simplex_box(h, n / 2)
    return h

import time
import numpy as np
from scipy import optimize

# ---------------------------------------------------------------
# Helper utilities (simplex+box projection, C5 evaluation, etc.)
# ---------------------------------------------------------------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} via a simple bisection on λ."""
    lo = np.min(v) - 1.0  # lower bound for λ
    hi = np.max(v)        # upper bound for λ
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny residual correction
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C5 using full linear correlation."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------------------------------------------------------------
# Local optimiser (SLSQP) that handles the auxiliary variable t
# ---------------------------------------------------------------
def _solve_h(start_h: np.ndarray, max_iter: int = 500) -> tuple[np.ndarray, float]:
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


# ---------------------------------------------------------------
# Simple up‑sampling (linear) of a coarse step function
# ---------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


# ---------------------------------------------------------------
# One‑parameter family: symmetric trapezoid with a flat top.
# a = length of each sloping side (0<a<=1). Height c = 1/(2-a).
# ---------------------------------------------------------------
def _shape_a(n: int, a: float) -> np.ndarray:
    if a <= 0.0:
        a = 1e-9
    if a > 1.0:
        a = 1.0
    c = 1.0 / (2.0 - a)  # ensures total area = 1 analytically
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.empty_like(x)

    left = x < a
    right = x > 2.0 - a
    middle = ~(left | right)

    # left ramp
    h[left] = (c / a) * x[left]
    # flat top
    h[middle] = c
    # right ramp
    h[right] = (c / a) * (2.0 - x[right])

    # project to the exact feasible simplex‑box (discretisation error)
    return _project_to_simplex_box(h, n / 2.0)

# Simple greedy local improvement (pair­wise swaps)
# ---------------------------------------------------------------
def _greedy_improve(base_h: np.ndarray, iters: int, step: float,
                    time_left) -> tuple[np.ndarray, float]:
    n = base_h.size
    cur = base_h.copy()
    cur_c5 = _c5_from_h(cur)
    best_h, best_c5 = cur.copy(), cur_c5
    for _ in range(iters):
        if time_left() < 0.5:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # choose a delta respecting bounds
        delta_min = max(-step, -cur[i], cur[j] - 1.0)
        delta_max = min(step, 1.0 - cur[i], cur[j])
        if delta_min >= delta_max:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        cur[i] += delta
        cur[j] -= delta
        new_c5 = _c5_from_h(cur)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = cur.copy()
                best_c5 = new_c5
        else:
            # revert
            cur[i] -= delta
            cur[j] += delta
    return best_h, best_c5


# ---------------------------------------------------------------
# Main driver
# ---------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ----------------------------------------------------------------
    # Load parent construction if available
    # ----------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ----------------------------------------------------------------
    # Candidate resolutions (favor the parent size, then a few larger ones)
    # ----------------------------------------------------------------
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (64, 96, 128, 160, 192, 224, 256, 300, 384, 448):
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    # ----------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # base shape (parent or simple up‑sample)
        if init_h is not None and init_h.size == n:
            base = init_h.copy()
        elif init_h is not None:
            base = _upscale_h(init_h, n)
        else:
            # default starting shape: the tent (a=1)
            base = _shape_a(n, 1.0)

        # ----------------------------------------------------------------
        # Explore the one‑parameter family _shape_a
        # ----------------------------------------------------------------
        a_grid = np.linspace(0.2, 0.99, 30)
        for a in a_grid:
            if time_left() < 5.0:
                break
            cand = _shape_a(n, a)
            c5 = _c5_from_h(cand)
            if c5 < best_c5:
                best_h, best_c5, best_n = cand.copy(), c5, n
                if best_c5 < target_bound:
                    break

        # ----------------------------------------------------------------
        # Refine the current best for this n with SLSQP
        # ----------------------------------------------------------------
        if best_h is not None and best_n == n:
            refined, c5_ref = _solve_h(best_h, max_iter=800)
            if c5_ref < best_c5:
                best_h, best_c5 = refined, c5_ref

        # ----------------------------------------------------------------
        # Additional stochastic greedy improvement on the refined shape
        # ----------------------------------------------------------------
        if best_h is not None and best_n == n:
            # modest number of greedy passes; step size adapts with n
            step_size = 0.12  # absolute change allowed per pair
            greedy_h, greedy_c5 = _greedy_improve(
                best_h, iters=8000, step=step_size, time_left=time_left
            )
            if greedy_c5 < best_c5:
                best_h, best_c5 = greedy_h, greedy_c5

        # ----------------------------------------------------------------
        # Random restarts with SLSQP (few attempts)
        # ----------------------------------------------------------------
        attempts = 5
        for _ in range(attempts):
            if time_left() < 5.0:
                break
            start = np.random.rand(n)
            start = _project_to_simplex_box(start, n / 2.0)
            opt_h, opt_c5 = _solve_h(start, max_iter=500)
            if opt_c5 < best_c5:
                best_h, best_c5 = opt_h, opt_c5
                if best_c5 < target_bound:
                    break

        if best_c5 < target_bound:
            break

    # ----------------------------------------------------------------
    # Final fallback (unlikely)
    # ----------------------------------------------------------------
    if best_h is None:
        n = 192
        best_h = _shape_a(n, 1.0)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    # Ensure the returned array satisfies the constraints (tiny numerical drift)
    best_h = _project_to_simplex_box(np.asarray(best_h, dtype=float), best_n / 2.0)

    return best_h.tolist(), float(best_c5), int(best_n)
