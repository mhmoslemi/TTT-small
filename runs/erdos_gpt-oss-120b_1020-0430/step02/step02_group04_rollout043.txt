def swap_delta(i, j, h):
    N = len(h)
    ones = h
    zeros = 1 - h
    delta = np.zeros(2*N-1, dtype=int)
    # i first factor - zeros
    start_i = N-1 - i
    delta[start_i:start_i+N] -= zeros
    # j first factor + zeros
    start_j = N-1 - j
    delta[start_j:start_j+N] += zeros
    # i second factor + ones
    idx = np.arange(N)
    p_i = (i - idx) + (N-1)
    delta[p_i] += ones
    # j second factor - ones
    p_j = (j - idx) + (N-1)
    delta[p_j] -= ones
    return delta

def compute_delta(i, j, h):
    N = len(h)
    ones = h
    zeros = 1 - h
    delta = np.zeros(2*N-1, dtype=int)
    delta[N-1 - i : N-1 - i + N] -= zeros
    delta[N-1 - j : N-1 - j + N] += zeros
    delta[i : i + N] += ones[::-1]
    delta[j : j + N] -= ones[::-1]
    return delta

    N = len(x)
    h = (x > 0.5).astype(np.int8)
    current_sum = h.sum()
    target = N//2
    diff = target - current_sum

def compute_delta_swap(i, j, h):
    N = h.shape[0]
    ones = h
    zeros = 1 - h
    delta = np.zeros(2*N-1, dtype=int)
    delta[N-1 - i : N-1 - i + N] -= zeros
    delta[N-1 - j : N-1 - j + N] += zeros
    delta[i : i + N] += ones[::-1]
    delta[j : j + N] -= ones[::-1]
    return delta

def binary_anneal(N, rng, time_left):
    # set per loop time limit
    # Start with random binary
    h = np.zeros(N, dtype=np.int8)
    ones_idx = rng.choice(N, N//2, replace=False)
    h[ones_idx] = 1
    # compute counts
    counts = np.correlate(h, 1 - h, mode='full')
    cur_max = counts.max()
    best_h = h.copy()
    best_c = (2.0/N) * cur_max
    start = time.time()
    while time.time() - start < time_left:
        # choose lag with max
        max_lags = np.flatnonzero(counts == cur_max)
        lag_idx = rng.choice(max_lags)
        lag = lag_idx - (N-1)
        A, B = mismatched_positions(h, lag)
        if A.size > 0 and B.size > 0:
            i = rng.choice(A)
            j = rng.choice(B)
        else:
            # random swap
            ones_i = np.flatnonzero(h == 1)
            zeros_j = np.flatnonzero(h == 0)
            i = rng.choice(ones_i)
            j = rng.choice(zeros_j)
        delta = compute_delta_swap(i, j, h)
        new_counts = counts + delta
        new_max = new_counts.max()
        # Accept if improves or with small probability
        if new_max <= cur_max or rng.random() < 0.001:
            # perform swap
            h[i] = 0
            h[j] = 1
            counts = new_counts
            cur_max = new_max
            c_val = (2.0/N) * cur_max
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()
        # occasionally random restart: if no improvement for many iterations
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# basic helpers
# -------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Generate a random feasible h (0≤h≤1, Σh=n/2)."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Small random move that keeps sum exact (box‑sum projection)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=5000, start_step=0.5):
    """Simple hill‑climb on the exact C5 (discrete eval)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# smooth‑max surrogate + SLSQP
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP pass using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                     # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()                              # soft‑max weights

            # gradient of the smooth max (raw, before scaling)
            g_raw = np.zeros(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)

            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# -------------------------------------------------
# binary hill‑climb (exact integer updates)
# -------------------------------------------------
def binary_hill_climb(N, rng, time_budget):
    """Search a binary {0,1} step function for low C5."""
    start = time.time()
    # random start
    h = np.zeros(N, dtype=np.int8)
    h[rng.choice(N, N // 2, replace=False)] = 1
    counts = np.correlate(h, 1 - h, mode="full")
    cur_max = int(counts.max())
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0

    while time.time() - start < time_budget:
        # choose a shift attaining the current maximum
        max_idxs = np.flatnonzero(counts == cur_max)
        lag_idx = rng.choice(max_idxs)
        lag = lag_idx - (N - 1)

        # indices where the current max shift contributes mismatches
        if lag >= 0:
            i_range = np.arange(0, N - lag)
        else:
            i_range = np.arange(-lag, N)

        a = i_range[(h[i_range] == 1) & (h[i_range + lag] == 0)]
        b = i_range[(h[i_range] == 0) & (h[i_range + lag] == 1)]

        if a.size and b.size:
            i = int(rng.choice(a))
            j = int(rng.choice(b))
        else:
            i = int(rng.choice(np.flatnonzero(h == 1)))
            j = int(rng.choice(np.flatnonzero(h == 0)))

        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_counts = np.correlate(new_h, 1 - new_h, mode="full")
        new_max = int(new_counts.max())

        # accept if improves or with a tiny probability
        if new_max < cur_max or rng.random() < 0.001:
            h = new_h
            counts = new_counts
            cur_max = new_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1

        if no_improve >= 5000:
            # restart to avoid stagnation
            h = np.zeros(N, dtype=np.int8)
            h[rng.choice(N, N // 2, replace=False)] = 1
            counts = np.correlate(h, 1 - h, mode="full")
            cur_max = int(counts.max())
            no_improve = 0

    return best_h.astype(float), best_max * 2.0 / N


# -------------------------------------------------
# main driver
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    tic = time.time()

    # start from parent if present, otherwise random feasible
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, h0.size / 2.0)
    else:
        n0 = 180
        h0 = random_feasible(n0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # -------------------------------------------------
    # Phase 1: quick hill‑climb on the original resolution
    # -------------------------------------------------
    if time.time() - tic < budget_s * 0.25:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=8000, start_step=0.5)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # -------------------------------------------------
    # Phase 2: smooth‑max gradient descent with restarts
    # -------------------------------------------------
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 800.0]
    steps_per_beta = 1200
    while time.time() - tic < budget_s * 0.6:
        h_start = random_feasible(best_h.size, rng)
        # occasional blend with current best
        if rng.random() < 0.2:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, h_start.size / 2.0)
        h_opt, c_opt = grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng)

        # short SLSQP refinements
        for beta in (200.0, 500.0, 800.0):
            try:
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=300)
            except Exception:
                pass
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # -------------------------------------------------
    # Phase 3: binary search on larger grids
    # -------------------------------------------------
    if time.time() - tic < budget_s * 0.85:
        # try a few larger sizes; allocate remaining time equally
        remaining = budget_s - (time.time() - tic)
        sizes = [240, 300, 360, 420, 480]
        time_per = remaining / len(sizes)
        for N in sizes:
            h_bin, c_bin = binary_hill_climb(N, rng, time_per * 0.9)
            if c_bin < best_c:
                best_h, best_c = h_bin, c_bin

    # -------------------------------------------------
    # Phase 4: final fine‑tune (optional up‑sampling)
    # -------------------------------------------------
    if time.time() - tic < budget_s * 0.95:
        # double resolution of current best and hill‑climb a bit
        h_up = np.repeat(best_h, 2)
        h_up, c_up = hill_climb(h_up, rng, max_iters=4000, start_step=0.4)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
