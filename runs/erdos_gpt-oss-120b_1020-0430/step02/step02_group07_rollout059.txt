def binary_hill_climb(h, rng, max_iters=2000):
    n = h.size
    w = 1 - h
    c_pos = compute_corr_pos(h, w)
    c_neg = compute_corr_pos(w, h)
    cur_max = max(c_pos.max(), c_neg.max())
    best_h = h.copy()
    best_max = cur_max
    # Precompute ones and zeros
    for it in range(max_iters):
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(w)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        delta_pos = delta_corr(h, w, i, j)
        delta_neg = delta_corr(w, h, i, j)
        new_c_pos = c_pos + delta_pos

Implementation of binary_hill_climb:

def binary_hill_climb(h, rng, max_iters=20000, init_temp=5.0, decay=0.9995, fast_stop=5000):
   n = h.size
   w = 1 - h
   c_pos = compute_corr(h, w)
   c_neg = compute_corr(w, h)
   cur_max = int(max(c_pos.max(), c_neg.max()))
   best_h = h.copy()
   best_max = cur_max
   temp = init_temp
   no_improve_counter = 0
   for it in range(max_iters):
       # Choose i,j
       # random 1 positions
       ones = np.flatnonzero(h)
       zeros = np.flatnonzero(w)
       if len(ones)==0 or len(zeros)==0:
           break
       i = rng.choice(ones)
       j = rng.choice(zeros)
       delta_pos = delta_corr(h, w, i, j)
       delta_neg = delta_corr(w, h, i, j)
       new_c_pos = c_pos + delta_pos
       new_c_neg = c_neg + delta_neg
       new_max = int(max(new_c_pos.max(), new_c_neg.max()))
       accept = False
       if new_max < cur_max:
           accept = True
       else:
           # probability
           prob = np.exp((cur_max - new_max) / temp)
           if rng.random() < prob:
               accept = True
       if accept:
           # swap
           h[i], h[j] = 0, 1
           w[i], w[j] = 1, 0
           c_pos = new_c_pos
           c_neg = new_c_neg
           cur_max = new_max
           if cur_max < best_max:
               best_max = cur_max
               best_h = h.copy()
               no_improve_counter = 0
           else:
               no_improve_counter += 1
       else:
           no_improve_counter += 1
       temp = max(temp * decay, 1e-4)
       # optional break if no improve for many iterations.
       if no_improve_counter > fast_stop:
           break
   return best_h, best_max

import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def perturb_h(h, rng, scale=0.3):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=1500, start_step=0.4):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=3000, init_temp=0.03,
                        decay=0.998, scale=0.2):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(nh)
        if new_c < best_c:
            best_c = new_c
            best_h = nh.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = nh
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def quick_refine(h, rng):
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        trial = slsqp_optimize(best_h, beta=beta, max_iter=250)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=1200, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def deep_refine(h, rng):
    n = h.size
    max_slsqp = 250 if n > 200 else 300
    hill_iters = 1200 if n > 200 else 1800
    anneal_iters = 2500 if n > 200 else 3500
    betas = (5.0, 20.0, 80.0, 200.0, 500.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        trial = slsqp_optimize(best_h, beta=beta, max_iter=max_slsqp)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    trial_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iters=anneal_iters,
                                         init_temp=0.03,
                                         decay=0.998,
                                         scale=0.2)
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def compute_corr(src, dst):
    """Positive‑lag correlation Σ_i src[i]·dst[i+k] for k=0…n‑1."""
    corr_full = np.correlate(src, dst, mode="full")
    return corr_full[len(src) - 1 :]


def delta_corr(src, dst, i, j):
    """
    Incremental change of Σ_i src[i]·dst[i+k] when entries i and j are swapped
    (src and dst are complements, i.e. dst = 1‑src).
    Returns an int array of length n (the delta for each lag k≥0).
    """
    n = src.size
    sign = int(dst[j] - dst[i])  # equals src[j] - src[i], ±1
    # term1: sign·(dst[i+k]‑dst[j+k])
    term1 = np.zeros(n, dtype=int)
    term1[: n - i] += sign * dst[i:]
    term1[: n - j] -= sign * dst[j:]
    # term2: sign·(‑src[i‑k] + src[j‑k])
    term2 = np.zeros(n, dtype=int)
    term2[: i + 1] -= sign * src[i :: -1]
    term2[: j + 1] += sign * src[j :: -1]
    # term3: fixed contribution
    term3 = np.zeros(n, dtype=int)
    term3[0] = -2
    term3[abs(j - i)] += 1
    return term1 + term2 + term3


def binary_hill_climb(h_start, rng, max_iters=20000, init_temp=5.0,
                      decay=0.9995, stop_patience=4000):
    """
    Simulated‑annealing style local search on binary vectors.
    Returns the best binary vector seen and its raw maximal overlap count.
    """
    h = h_start.copy()
    w = 1 - h
    c_pos = compute_corr(h, w)
    c_neg = compute_corr(w, h)
    cur_max = int(max(c_pos.max(), c_neg.max()))
    best_h, best_max = h.copy(), cur_max
    temp = init_temp
    patience = 0
    n = h.size
    for it in range(max_iters):
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(w)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        dpos = delta_corr(h, w, i, j)
        dneg = delta_corr(w, h, i, j)
        new_c_pos = c_pos + dpos
        new_c_neg = c_neg + dneg
        new_max = int(max(new_c_pos.max(), new_c_neg.max()))
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp((cur_max - new_max) / temp):
                accept = True
        if accept:
            h[i], h[j] = 0, 1
            w[i], w[j] = 1, 0
            c_pos, c_neg = new_c_pos, new_c_neg
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
        temp = max(temp * decay, 1e-4)
        if patience > stop_patience:
            break
    return best_h, best_max


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function on [0,2] to minimise the Erdős C₅ bound.
    Returns (list_of_h, C5_value, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initialise a binary start vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # convert to binary by taking the largest half of the values
        idx = np.argsort(-h0)
        h_bin = np.zeros(n, dtype=int)
        h_bin[idx[: n // 2]] = 1
    else:
        n = 360
        perm = rng.permutation(n)
        h_bin = np.zeros(n, dtype=int)
        h_bin[perm[: n // 2]] = 1

    best_bin_h, best_bin_max = binary_hill_climb(h_bin.copy(), rng, max_iters=20000)
    best_bin_c5 = best_bin_max * (2.0 / n)

    # ------------------------------------------------------------------
    # Random binary restarts (30 % of the time budget)
    # ------------------------------------------------------------------
    time_limit_bin = start + budget_s * 0.30
    while time.time() < time_limit_bin:
        perm = rng.permutation(n)
        h_rand = np.zeros(n, dtype=int)
        h_rand[perm[: n // 2]] = 1
        h_loc, max_loc = binary_hill_climb(h_rand, rng, max_iters=8000)
        if max_loc < best_bin_max:
            best_bin_h, best_bin_max = h_loc, max_loc
            best_bin_c5 = max_loc * (2.0 / n)

    # ------------------------------------------------------------------
    # Convert best binary to float and perform continuous refinement
    # ------------------------------------------------------------------
    h_float = best_bin_h.astype(float)
    best_h, best_c = quick_refine(h_float, rng)

    # ------------------------------------------------------------------
    # Up‑sample and deep refinement if time permits (remaining budget)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.90:
        if n * 2 > 720:  # limit size for speed
            break
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        h_up_ref, c_up_ref = deep_refine(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = n_up
        else:
            break

    # ------------------------------------------------------------------
    # Final feasibility clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
