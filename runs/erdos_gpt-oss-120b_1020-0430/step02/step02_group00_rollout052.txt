def _local_gradient_descent(h0, max_iters, time_left):
    n = h0.size
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_val = _c5_from_h(h)
    step0 = 0.2
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # compute correlation array
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute gradient g = 1 - (h_{j+k}+h_{j-k})
        g = np.ones(n)
        if k >= 0:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        else:
            g[-k:] -= h[:n + k]
            g[:n + k] -= h[-k:]
        # step size decreasing linearly
        step = step0 * (1.0 - it / max_iters) + 0.001
        h_new = h - step * g
        h_new = _project_to_feasible(h_new, n)
        val_new = _c5_from_h(h_new)
        if val_new < best_val - 1e-12:
            best_h, best_val = h_new.copy(), val_new
            h = h_new
        else:
            # occasional random move
            if it % 50 == 0:
                pert = 0.05 * np.random.randn(n)
                h = _project_to_feasible(h + pert, n)
    return best_h, best_val

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 using water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction (rare)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a feasible h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _active_shift(h: np.ndarray):
    """Return the shift index achieving the maximal correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    return idx - (n - 1), corr[idx]


def _gradient_of_active_shift(h: np.ndarray, k: int):
    """Gradient of the active correlation term w.r.t. h."""
    n = h.size
    g = np.ones(n)
    if k >= 0:
        g[:n - k] -= h[k:]
        g[k:] -= h[:n - k]
    else:
        g[-k:] -= h[:n + k]
        g[:n + k] -= h[-k:]
    return g


def _local_gradient_descent(h0: np.ndarray, max_iters: int, time_left):
    """Projected gradient descent on the max‑correlation objective."""
    n = h0.size
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_val = _c5_from_h(h)
    step0 = 0.2

    for it in range(max_iters):
        if time_left() <= 0:
            break
        k, _ = _active_shift(h)
        g = _gradient_of_active_shift(h, k)
        step = step0 * (1.0 - it / max_iters) + 0.001
        h_new = h - step * g
        h_new = _project_to_feasible(h_new, n)
        val_new = _c5_from_h(h_new)
        if val_new < best_val - 1e-12:
            best_h, best_val = h_new.copy(), val_new
            h = h_new
        else:
            # occasional random jitter to escape plateaus
            if it % 50 == 0:
                h = _project_to_feasible(h + 0.05 * np.random.randn(n), n)
    return best_h, best_val


def _binary_sa(h_real: np.ndarray, time_left):
    """Simulated annealing on a binary version of the current real solution."""
    n = h_real.size
    # initialise binary vector by taking the largest entries
    idx = np.argsort(-h_real)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    best_h = h.copy()
    best_val = _c5_from_h(best_h)

    # temperature schedule
    T = 0.2
    decay = 0.999
    min_T = 1e-6

    while time_left() > 0.3 and T > min_T:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        val = _c5_from_h(new_h)
        delta = val - best_val
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            best_h = new_h
            best_val = val
        T *= decay
    return best_h, best_val

        if n != best_n:

        else:
