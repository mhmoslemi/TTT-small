def _c5_from_h(h):
    n = len(h)
    # compute convolution using FFT
    # but np.fft has overhead but okay.
    # We'll compute correlation of h and 1-h.
    # Using np.fft.rfft for real input.
    a = np.fft.rfft(h, n*2)  # zero pad to length 2n
    b = np.fft.rfft(1.0 - h, n*2)
    corr = np.fft.irfft(a * np.conj(b), n*2)[:2*n-1]  # correlation full
    max_corr = corr.max()

def _project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1.0-1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _local_search(h, cur_c5, time_left):
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = cur_c5
    step_factor = 0.2
    no_improve = 0
    while time_left() > 0.15:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * step_factor * np.random.rand()
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5 = _c5_from_h(h_new)
        if new_c5 + 1e-12 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improve = 0
            step_factor = min(step_factor * 1.05, 0.5)
        else:
            no_improve += 1
            if no_improve % 4000 == 0:
                step_factor *= 0.7
        # occasional gradient step every 50000 iterations
        if no_improve and no_improve % 50000 == 0:
            # gradient descent for active shift
            # compute correlation array
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            k = np.argmax(corr) - (n - 1)  # shift value (negative allowed)
            # compute gradient
            grad = np.ones_like(best_h)
            # subtract h[i+k] where defined
            if k >= 0:
                grad[:n-k] -= best_h[k:]
                grad[k:] -= best_h[:n-k]
            else:
                kk = -k
                grad[kk:] -= best_h[:n-kk]
                grad[:n-kk] -= best_h[kk:]
            # projection onto simplex box after step
            alpha = step_factor * 0.5
            h_tmp = best_h - alpha * grad
            h_tmp = _project_to_simplex_box(h_tmp, n/2.0)
            new_c5 = _c5_from_h(h_tmp)
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = h_tmp, new_c5
                step_factor = min(step_factor * 1.05, 0.5)

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {x∈[0,1]^n : sum(x)=target} via binary search."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1.0 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Fast C5 via FFT based correlation."""
    n = h.shape[0]
    a = np.fft.rfft(h, n=2 * n)
    b = np.fft.rfft(1.0 - h[::-1], n=2 * n)
    conv = np.fft.irfft(a * b, n=2 * n)
    max_corr = conv[: 2 * n - 1].max()
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Refine h with SLSQP; returns feasible h and its C5."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    old_n = old_h.size
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(x_new, x_old, old_h)
