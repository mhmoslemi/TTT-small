def gradient_smooth_min(h, beta):
    n = len(h)
    prod = np.correlate(h, h, mode='full')
    w = np.exp(-beta * prod)
    w /= w.sum()
    grad = np.zeros(n)
    # shift indices: idx 0-> -(n-1), idx 2n-2 -> (n-1)
    for idx, w_i in enumerate(w):
        shift = idx - (n-1)
        if w_i == 0:
            continue
        # plus side: j such that j+shift in [0,n)
        if shift >= 0:
            # j from 0 to n-1-shift
            j_start = 0
            j_end = n - shift
            grad[j_start:j_end] += w_i * h[j_start+shift:j_end+shift]
            # minus side: j from shift to n-1
            j_start2 = shift
            j_end2 = n
            grad[j_start2:j_end2] += w_i * h[j_start2-shift:j_end2-shift]  # j-shift indexes

def _gradient_optimize(h0, n, time_end, max_iters=5000, init_alpha=0.2):
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    alpha = init_alpha
    it = 0
    while time.time() < time_end and it < max_iters:
        # compute gradient
        g = _min_shift_gradient(h)
        if np.all(g == 0):
            # random perturb
            h = _project_to_feasible(h + 0.01*(np.random.rand(n)-0.5), n)
            continue
        # step
        h_candidate = h + alpha * g
        h_candidate = _project_to_feasible(h_candidate, n)
        c5_new = _c5_from_h(h_candidate)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_candidate.copy()
        if c5_new < _c5_from_h(h):  # improvement

                h = _project_to_feasible(best_h + 0.1*(np.random.rand(n)-0.5), n)
                alpha = init_alpha

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 using a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Integral C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# --------------------------------------------------------------
# Gradient based optimizer (smooth‑min of self‑correlation)
# --------------------------------------------------------------

def _smooth_min_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of a smoothed minimum of the self‑correlation of h."""
    n = h.size
    prod = np.correlate(h, h, mode="full")
    # weights larger for smaller products
    w = np.exp(-beta * prod)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    grad = np.zeros_like(h)
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            if s < n:
                grad[: n - s] += wi * h[s:]
                grad[s:] += wi * h[: n - s]
        else:
            s = -shift
            if s < n:
                grad[s:] += wi * h[: n - s]
                grad[: n - s] += wi * h[s:]
    return grad


def _gradient_optimize(
    h0: np.ndarray,
    n: int,
    start_time: float,
    budget_s: float,
    max_iters: int = 3000,
    init_alpha: float = 0.2,
    beta: float = 5.0,
) -> tuple:
    """Projected gradient ascent on the smooth‑min of self‑correlation."""
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    alpha = init_alpha
    current_c5 = best_c5

    for _ in range(max_iters):
        if time.time() - start_time > budget_s - 0.2:
            break
        grad = _smooth_min_gradient(h, beta)
        if not np.any(grad):
            # stagnated – perturb a little
            h = _project_to_feasible(h + 0.02 * (np.random.rand(n) - 0.5), n)
            continue
        cand = _project_to_feasible(h + alpha * grad, n)
        cand_c5 = _c5_from_h(cand)

        if cand_c5 < best_c5:
            best_c5, best_h = cand_c5, cand.copy()
        if cand_c5 < current_c5:
            h, current_c5 = cand, cand_c5
            alpha = min(alpha * 1.07, 1.0)
        else:
            alpha *= 0.6
            if alpha < 1e-4:
                # reset around the current best
                h = _project_to_feasible(best_h + 0.05 * (np.random.rand(n) - 0.5), n)
                alpha = init_alpha
    return best_h, best_c5


# --------------------------------------------------------------
# Simple epsilon hill‑climb (continuous swaps)
# --------------------------------------------------------------

def _hill_climb_swap(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Random epsilon exchanges between two coordinates, keeping sum fixed."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c
