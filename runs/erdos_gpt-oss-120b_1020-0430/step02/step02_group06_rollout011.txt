import time
import numpy as np

# -------------------------------------------------------------
#  Projection onto the capped simplex {0≤h≤1, sum(h)=n/2}
# -------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny fix‑up
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

# -------------------------------------------------------------
#  Convolution / correlation via FFT (full, linear)
# -------------------------------------------------------------
def _corr_full_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    f = h.astype(np.float64)
    g = 1.0 - f
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    F = np.fft.rfft(f, L)
    G = np.fft.rfft(g[::-1], L)           # reversed g
    conv = np.fft.irfft(F * G, L)[: 2 * n - 1]
    return conv.real

# -------------------------------------------------------------
#  C5 value from a discretised function
# -------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(_corr_full_fft(h))
    return max_corr * (2.0 / n)

# -------------------------------------------------------------
#  Simple stochastic refinement around a candidate
# -------------------------------------------------------------
def _local_refine(h: np.ndarray, attempts: int = 20, delta: float = 0.02) -> np.ndarray:
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h

# -------------------------------------------------------------
#  Upscale a coarse step function by linear interpolation
# -------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# -------------------------------------------------------------
#  Gradient of the worst‑shift correlation (sub‑gradient)
# -------------------------------------------------------------
def _grad_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    n = h.size
    g = np.ones_like(h)
    if s == 0:
        g -= 2.0 * h
    elif s > 0:
        g[s:] -= h[:-s]               # term h[i‑s]
        g[: n - s] -= h[s:]           # term h[i+s]
    else:  # s < 0
        a = -s
        g[: n - a] -= h[a:]           # term h[i‑s] = h[i+a]
        g[a:] -= h[: n - a]           # term h[i+s] = h[i‑a]
    return g

# -------------------------------------------------------------
#  Main optimisation routine for a single starting point
# -------------------------------------------------------------
def _optimise_start(h0: np.ndarray, time_left, target_bound) -> tuple:
    n = h0.size
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    it = 0
    while True:
        it += 1
        conv = _corr_full_fft(h)
        s = int(np.argmax(conv)) - (n - 1)    # shift giving max correlation
        grad = _grad_for_shift(h, s)
        eta = 0.5 / np.sqrt(it)
        h = h - eta * grad
        h = _project_to_feasible(h, n)

        if it % 30 == 0:
            c5 = _c5_from_h(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
                if best_c5 < target_bound:
                    break

        if time_left() < 4.0:
            break
    # final stochastic polishing
    best_h = _local_refine(best_h, attempts=30, delta=0.015)
    best_c5 = _c5_from_h(best_h)
    return best_h, best_c5
