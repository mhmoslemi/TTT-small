def compute_c5_bin(x):
    n = len(x)
    n_fft = 1 << ((2*n-1).bit_length())
    fx = np.fft.rfft(x, n_fft)
    conv = np.fft.irfft(fx * np.conj(fx), n_fft)
    # conv length n_fft
    # linear autocorrelation: conv[:2*n-1]
    autocorr = conv[:2*n-1].round().astype(np.int64) # due to floating errors
    # central index = n-1
    min_inner = autocorr[:n-1].min()
    # also consider autocorr[n:] (positive lags)
    min_inner = min(min_inner, autocorr[n:].min())
    c5 = 1.0 - (2.0 / n) * min_inner
    return c5

def compute_min_inner(x):
    n = x.shape[0]
    n_fft = 1 << ((2*n - 1).bit_length())
    fx = np.fft.rfft(x, n_fft)
    conv = np.fft.irfft(fx * np.conj(fx), n_fft)
    autocorr = conv[:2*n-1]
    # since values are integer, we can round .
    # np.rint maybe.

    # get min over non-zero lags
    # The lag zero index = n-1
    left = autocorr[:n-1]
    right = autocorr[n:]
    # If empty (n=1) not case.
    min_val = int(min(left.min() if left.size else np.inf, right.min() if right.size else np.inf))
    return min_val

def binary_sa(n, rng, time_limit):
    # returns best binary array and best C5
    start = time.time()
    # initial random state
    half = n // 2
    cur = np.zeros(n, dtype=np.int8)
    cur[:half] = 1
    rng.shuffle(cur)
    cur_c = compute_c5(cur.astype(float))
    best_h = cur.copy()
    best_c = cur_c
    # temperature schedule
    start_temp = 2.0  # maybe set
    while time.time() - start < time_limit:
        # propose swap
        # choose random 1
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        T = start_temp * (1 - t_frac) + 1e-6
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            # update best
            if new_c < best_c:
                best_c = new_c
                best_h = cur.copy()
        else:
            # revert
            cur[i] = 1
            cur[j] = 0
    return best_h.astype(float), best_c

import time
import numpy as np

# ------------------- core utilities -------------------
def compute_c5(h):
    """Maximum overlap integral for step vector h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Box‑sum projection: clip to [lower,upper] while forcing sum == target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


# ------------------- continuous helper -------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5(best)
    step = start_step
    cur_c = best_c
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < cur_c:
            h, cur_c = nh, c
            if c < best_c:
                best, best_c = nh.copy(), c
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best, best_c


# ----- soft‑max surrogate for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best = h.copy()
    best_c = compute_c5(best)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            C = corr * (2.0 / n)                     # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()
            # gradient of smoothed max (raw, before scaling)
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            nh = h - step * grad
            nh = project_onto_box_sum(nh, n / 2.0)
            cnew = compute_c5(nh)
            if cnew < best_c:
                best_c, best = cnew, nh.copy()
                h = nh
            else:
                if rng.random() < 0.01:
                    h = nh
            step *= 0.9995
    return best, best_c


# ------------------- binary simulated annealing -------------------
def binary_sa(n, rng, time_limit, init_state=None):
    """Simple SA on a balanced 0/1 vector."""
    start = time.time()
    # initialise
    if init_state is not None:
        cur = init_state.astype(np.int8).copy()
    else:
        cur = np.zeros(n, dtype=np.int8)
        cur[: n // 2] = 1
        rng.shuffle(cur)

    cur_c = compute_c5(cur.astype(float))
    best_h = cur.copy()
    best_c = cur_c
    start_temp = 0.3

    while time.time() - start < time_limit:
        # pick a random 1 and a random 0
        i_candidates = np.where(cur == 1)[0]
        j_candidates = np.where(cur == 0)[0]
        if i_candidates.size == 0 or j_candidates.size == 0:
            break
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)

        # swap
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        elapsed = time.time() - start
        T = start_temp * (1.0 - elapsed / time_limit) + 1e-12
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, cur.copy()
        else:
            # revert
            cur[i] = 1
            cur[j] = 0

        # occasional full restart if stuck
        if rng.random() < 0.001:
            cur[:] = 0
            cur[: n // 2] = 1
            rng.shuffle(cur)
            cur_c = compute_c5(cur.astype(float))

    return best_h.astype(float), best_c


# ------------------- main driver -------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise size and a binary start -----
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        n = h_parent.size
        # binary start: top half entries become 1
        k = n // 2
        top_idx = np.argpartition(h_parent, -k)[-k:]
        binary_start = np.zeros(n, dtype=np.int8)
        binary_start[top_idx] = 1
    else:
        n = 180
        binary_start = None

    # ----- phase 1: binary simulated annealing -----
    binary_time = budget_s * 0.45
    best_h, best_c = binary_sa(n, rng, binary_time, init_state=binary_start)

    # keep the best continuous parent if it beats our binary result
    if "initial_h_values" in globals():
        parent_c = compute_c5(h_parent)
        if parent_c < best_c:
            best_h = h_parent.copy()
            best_c = parent_c

    # ----- early exit if already good enough -----
    target = 0.3808
    if best_c <= target:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        final_c = compute_c5(best_h)
        return best_h.tolist(), float(final_c), int(best_h.size)

    # ----- phase 2: continuous refinement (hill‑climb + smooth max) -----
    remaining = budget_s - (time.time() - start_time)
    # quick hill‑climb
    if remaining > budget_s * 0.15:
        hc_h, hc_c = hill_climb(best_h.copy(), rng, max_iters=3000, start_step=0.5)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c
        remaining = budget_s - (time.time() - start_time)

    # gradient‑descent on smooth max
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0]
    steps_per_beta = 300
    while time.time() - start_time < budget_s * 0.78:
        gd_h, gd_c = grad_descent_optimize(best_h.copy(), beta_sched,
                                           steps_per_beta, rng)
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c

    # final SLSQP polishing (few passes)
    for beta in (200.0, 500.0):
        try:
            best_h = slsqp_optimize(best_h, beta=beta, max_iter=250)
        except Exception:
            pass
        best_c = compute_c5(best_h)

    # ----- optional up‑sampling if time permits -----
    if time.time() - start_time < budget_s * 0.95 and n * 2 <= 300:
        up_h = np.repeat(best_h, 2)
        up_h, up_c = hill_climb(up_h, rng, max_iters=1500, start_step=0.4)
        for beta in beta_sched:
            if time.time() - start_time > budget_s:
                break
            up_h = slsqp_optimize(up_h, beta=beta, max_iter=250)
        up_c = compute_c5(up_h)
        if up_c < best_c:
            best_h, best_c, n = up_h, up_c, up_h.size

    # ----- final cleanup -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
