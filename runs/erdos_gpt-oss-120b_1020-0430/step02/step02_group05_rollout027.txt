def _grad_smooth_max(h, w):
    # h shape (n,), w shape (2n-1,)
    n = h.shape[0]
    grad = np.zeros_like(h)
    # offset to map idx to shift
    offset = n - 1
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        shift = idx - offset
        if shift >= 0:
            i = shift
            # forward part
            if i < n:
                grad[:n-i] += weight * (1 - h[i:])
                # backward part
                grad[i:] -= weight * h[:n-i]
        else:
            k = -shift
            # forward part: grad[k:] += weight * (1 - h[:n-k])
            grad[k:] += weight * (1 - h[:n-k])
            # backward part: grad[:n-k] -= weight * h[k:]
            grad[:n-k] -= weight * h[k:]
    return grad

def _local_refine(h, max_iter=2000, step_scale=0.02):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for it in range(max_iter):
        if time_left() < 0.1: break
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # compute max delta that can be added while staying in bounds.
        # we propose random delta in [-step_scale, step_scale]
        delta = (np.random.rand() - 0.5) * 2 * step_scale
        # clip to feasible bounds
        # ensure new_i = best_h[i] + delta in [0,1] and new_j = best_h[j] - delta in [0,1].
        max_inc_i = 1 - best_h[i]
        max_dec_i = -best_h[i]
        max_inc_j = 1 - best_h[j]
        max_dec_j = -best_h[j]
        # allowed delta range limited by both.
        delta_low = max(max_dec_i, -max_inc_j)
        delta_high = min(max_inc_i, -max_dec_j)
        if delta_low > delta_high:
            continue
        delta = np.clip(delta, delta_low, delta_high)
        if abs(delta) < 1e-9:
            continue
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # sum unchanged, values within [0,1] by construction.
        new_c5 = _c5_from_h(new_h)
        if new_c5 < best_c5 - 1e-7:  # improvement threshold
            best_h = new_h
            best_c5 = new_c5
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x | 0 <= x_i <= 1, sum x_i = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 value for discrete h on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift (used by SLSQP)."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise auxiliary variable t."""
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 400):
    """Refine h using SLSQP on the true max‑correlation objective."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points, then projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _grad_smooth_max(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α) * log Σ exp(α * corr_i) w.r.t. h,
    where w are the softmax weights.
    """
    n = h.size
    grad = np.zeros_like(h)
    offset = n - 1
    for idx in range(2 * n - 1):
        weight = w[idx]
        if weight == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            i = shift
            # forward contribution
            if i < n:
                grad[: n - i] += weight * (1.0 - h[i:])
                grad[i:] -= weight * h[: n - i]
        else:
            k = -shift
            grad[k:] += weight * (1.0 - h[: n - k])
            grad[: n - k] -= weight * h[k:]
    return grad


def _grad_descent_h(
    start_h: np.ndarray,
    n_iter: int,
    time_budget: float,
    start_time: float,
    max_iter_limit: int = 800,
):
    """
    Projected gradient descent on a smooth approximation of the max correlation.
    Returns a refined h (still feasible).
    """
    n = start_h.size
    target = n / 2.0
    h = start_h.copy()
    # optional symmetry enforcement
    h = 0.5 * (h + h[::-1])
    for it in range(n_iter):
        # time check
        if time_budget - (time.time() - start_time) < 0.3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        # linear schedules for α and step size
        alpha = 5.0 + 25.0 * (it / n_iter)          # 5 → 30
        step = 0.08 * (1.0 - it / n_iter) + 0.002  # 0.08 → 0.002
        # soft‑max weights
        exp_vals = np.exp(alpha * (corr - max_corr))
        w = exp_vals / exp_vals.sum()
        grad = _grad_smooth_max(h, w)
        # gradient step + projection + symmetrisation
        h = _project_to_simplex_box(h - step * grad, target)
        h = 0.5 * (h + h[::-1])
    return h


def _local_refine_h(h: np.ndarray, max_iters: int, time_budget: float, start_time: float):
    """
    Simple hill‑climbing: move a tiny amount of mass between two points.
    Keeps the sum exactly n/2 and respects box constraints.
    """
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    # scale for the random moves – start relatively small
    sigma = 0.02

    for it in range(max_iters):
        if time_budget - (time.time() - start_time) < 0.2:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # permissible delta so both entries stay in [0,1]
        lo = max(-best_h[i], -sigma)
        hi = min(1.0 - best_h[i], sigma)
        lo2 = max(-best_h[j], -sigma)
        hi2 = min(1.0 - best_h[j], sigma)
        # we want delta to be added to i and subtracted from j
        delta_low = max(lo, -hi2)
        delta_high = min(hi, -lo2)
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # projection not required – construction respects bounds and sum
        new_c5 = _c5_from_h(new_h)
        if new_c5 + 1e-9 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            # shrink step size when progress is good
            sigma = max(sigma * 0.99, 0.005)
    return best_h, best_c5
