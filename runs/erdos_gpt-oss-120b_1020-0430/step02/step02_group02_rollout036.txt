def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(50):
        diff = h.sum() - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = h > 0.0
        else:
            mask = h < 1.0
        free = np.where(mask)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        i = 0
        while abs(residual) > 1e-14 and i < n:
            if residual > 0 and h[i] < 1.0:

                h[i] += add

def _optimal_eps_and_value(C, b, eps_max):
    """Given current C_k and slopes b_k, find epsilon in [0,eps_max] that minimises max(C-εb)."""
    # keep a small set of candidate epsilons
    eps_cand = [0.0, eps_max]

    # top 8 worst (largest) C values
    top_k = np.argpartition(-C, 8)[:8]
    # indices with negative slopes (lines that can rise)
    neg_k = np.where(b < 0)[0]
    if neg_k.size > 12:
        # keep the most negative
        neg_k = neg_k[np.argsort(b[neg_k])[:12]]

    for i in top_k:
        for j in neg_k:
            if b[i] == b[j]:
                continue
            eps = (C[i] - C[j]) / (b[i] - b[j])
            if 0.0 < eps < eps_max:
                eps_cand.append(eps)

    eps_arr = np.asarray(eps_cand)
    # evaluate max for each candidate (vectorised)
    vals = C[:, None] - eps_arr * b[:, None]
    max_vals = np.max(vals, axis=0)
    idx = np.argmin(max_vals)
    return eps_arr[idx], max_vals[idx]


def _pairwise_opt_move(h, C_raw):
    """Try one random i→j move, compute exact optimal epsilon and apply if improving."""
    n = h.size
    i, j = np.random.randint(n, size=2)
    if i == j:
        return False, C_raw
    eps_max = min(h[i], 1.0 - h[j])
    if eps_max <= 1e-12:
        return False, C_raw

    # shift range for full correlation
    s = np.arange(-n + 1, n)  # length 2n-1
    # neighbour sums for i
    idx = i + s
    f_i = np.where((idx >= 0) & (idx < n), h[idx], 0.0)
    idx = i - s
    b_i = np.where((idx >= 0) & (idx < n), h[idx], 0.0)
    neigh_i = f_i + b_i

    # neighbour sums for j
    idx = j + s
    f_j = np.where((idx >= 0) & (idx < n), h[idx], 0.0)
    idx = j - s
    b_j = np.where((idx >= 0) & (idx < n), h[idx], 0.0)
    neigh_j = f_j + b_j

    b = neigh_j - neigh_i  # slopes for each correlation entry

    eps_opt, new_max = _optimal_eps_and_value(C_raw, b, eps_max)

    if new_max < C_raw.max() - 1e-12:

        # recompute correlations from scratch (fast for small n)
        C_raw = np.correlate(h, 1.0 - h, mode="full")
        return True, C_raw
    return False, C_raw

                h = best_h + 0.02 * np.random.randn(*h.shape)
                h = _project_to_feasible(h, h.size)
                C_raw = np.correlate(h, 1.0 - h, mode="full")
                no_imp = 0
    return best_h, best_raw


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ----------------------------------------------------------------
    # Load optional parent construction
    # ----------------------------------------------------------------
    try:
        GLOBAL_INITIAL = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL = None

    # ----------------------------------------------------------------
    # Candidate grid sizes (prefer multiples of the parent size)
    # ----------------------------------------------------------------
    n_candidates = []
    if GLOBAL_INITIAL is not None:
        n_candidates.append(GLOBAL_INITIAL.size)
    for mul in (1, 2, 3, 4):
        if GLOBAL_INITIAL is not None:
            n = int(GLOBAL_INITIAL.size * mul)
        else:
            n = 64 * mul
        if n <= 256:
            n_candidates.append(n)
    n_candidates.extend([80, 96, 112, 128, 160, 192, 224, 256])
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ----------------------------------------------------------------
    # Search over grid sizes with a time budget per size
    # ----------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 8.0:
            break

        # initialise h on this grid
        if GLOBAL_INITIAL is not None:
            if GLOBAL_INITIAL.size == n:
                h0 = GLOBAL_INITIAL.copy()
            else:
                # linear interpolation of the parent onto the finer grid
                x_src = (np.arange(GLOBAL_INITIAL.size) + 0.5) * (2.0 / GLOBAL_INITIAL.size)
                x_tgt = (np.arange(n) + 0.5) * (2.0 / n)
                h0 = np.interp(x_tgt, x_src, GLOBAL_INITIAL)
        else:
            # simple sinusoidal start (integral 1 after projection)
            xs = (np.arange(n) + 0.5) * (2.0 / n)
            h0 = 0.5 + 0.4 * np.sin(np.pi * xs)
        h0 = _project_to_feasible(h0, n)

        # allocate a slice of the remaining time
        per = min(120.0, time_left() * 0.5)
        deadline = time.time() + per

        h_opt, C_raw_max = _local_search(h0, deadline)
        c5_val = C_raw_max * (2.0 / n)

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

        # stop early if we already beat the target record
        if best_c5 < 0.3808:
            break

    # fallback (should never happen)
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    # final safety projection (tiny numerical drift)
    best_h = _project_to_feasible(best_h, best_n)

    return best_h.tolist(), float(best_c5), int(best_n)
