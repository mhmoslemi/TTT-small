def project_feasible(h, target_sum, max_iter=200):
    h = np.clip(h, 0, 1)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # distributed equally among free variables
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # no free vars; adjust by clipping the largest/smallest
            # break
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0, 1)
    # final fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # adjust a single element if possible
        idx = np.argmin(np.abs(h - 0.5))  # maybe any
        h[idx] = np.clip(h[idx] + residual, 0, 1)
    return h

def pairwise_descent(h, max_iters=2000, step_scale=0.1, temp=0.0):
    n = h.shape[0]; target_sum = n/2.0
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    for it in range(max_iters):
        i, j = np.random.choice(n, 2, replace=False)
        if cur_h[i] > 1e-12 and cur_h[j] < 1-1e-12:
            # choose delta direction random sign: moving from i to j? We'll always move from i to j to stay feasible of sum.
            # delta positive reduces i, increases j.
            max_delta = min(cur_h[i], 1 - cur_h[j])
        else:
            max_delta = 0.0
        if max_delta <= 0:
            continue
        # delta magnitude decays
        delta = step_scale * max_delta * (np.random.rand() - 0.5) * 2  # uniform [-step*max, step*max]
        # ensure within bounds
        delta = np.clip(delta, -cur_h[i], 1 - cur_h[j])
        if abs(delta) < 1e-12:
            continue
        trial = cur_h.copy()
        trial[i] -= delta
        trial[j] += delta
        # ensure within bounds
        # (by construction)
        trial_c5 = compute_c5(trial)
        accept = False
        if trial_c5 < cur_c5:
            accept = True
        elif temp > 0:
            diff = trial_c5 - cur_c5
            if np.random.rand() < np.exp(-diff / temp):
                accept = True
        if accept:
            cur_h = trial
            cur_c5 = trial_c5
        # decay step_scale and temp
        step_scale *= 0.9995
        temp *= 0.9995
    return cur_h, cur_c5

import time

from scipy import optimize

# ---------- Feasibility projection ----------
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust sum to target using a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- Utility functions ----------
def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a discretised step function."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- Pairwise greedy descent ----------
def _pairwise_descent(h: np.ndarray, max_iters: int = 4000, step_scale: float = 0.2, temp: float = 0.0):
    """Greedy (or simulated‑annealing) descent that moves mass between two coordinates."""
    n = h.size
    cur_h = h.copy()
    cur_c5 = _c5_from_h(cur_h)
    for it in range(max_iters):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = (np.random.rand() * 2 - 1) * step_scale * max_delta
        delta = np.clip(delta, -cur_h[i], 1.0 - cur_h[j])
        if abs(delta) < 1e-12:
            continue
        trial = cur_h.copy()
        trial[i] -= delta
        trial[j] += delta
        trial_c5 = _c5_from_h(trial)
        accept = False
        if trial_c5 < cur_c5:
            accept = True
        elif temp > 0.0:
            diff = trial_c5 - cur_c5
            if np.random.rand() < np.exp(-diff / temp):
                accept = True
        if accept:
            cur_h, cur_c5 = trial, trial_c5
        # decay parameters gently
        step_scale = max(step_scale * 0.9999, 1e-5)
        temp = max(temp * 0.9999, 0.0)
    return cur_h, cur_c5


# ---------- SLSQP refinement ----------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400):
    """Run SLSQP starting from `initial_h`."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.x is not None else h0
    h_opt = _project_to_feasible(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)
