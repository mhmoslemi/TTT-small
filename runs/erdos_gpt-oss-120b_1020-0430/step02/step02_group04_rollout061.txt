def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        target = n0 / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        # Ensure symmetry
        h0 = (h0 + h0[::-1])*0.5
        h0 = project_onto_box_sum(h0, target)
    else:
        n0 = 180  # default
        h0 = random_feasible_symmetric(n0, rng)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    
    # Phase 1: multiple restarts with gradient descent + hill climb
    while time.time() - start < budget_s * 0.6:
        # create start candidate
        if rng.random() < 0.2:
            start_h = (best_h + random_feasible_symmetric(best_h.size, rng))*0.5
        else:
            start_h = random_feasible_symmetric(best_h.size, rng)
        # gradient descent

        # SLSQP refinement

def correl_fft(h):
    n = h.size
    a = h.astype(np.float64)
    b = (1 - h).astype(np.float64)
    n_fft = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    corr = np.fft.irfft(A * np.conj(B), n=n_fft)[:2*n-1]
    return corr

def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n/2.0
    # define constraints as equality sum = target
    bounds = [(0.0, 1.0)] * n
    def constr(x):
        return np.sum(x) - target
    cons = {'type': 'eq', 'fun': constr}
    # we can use objective: smooth max (like earlier)
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        c = corr * (2.0 / n)
        bc = beta * c
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = opt.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
              options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # project onto box sum to correct small errors
    x = project_onto_box_sum(x, target)
    return x

def compute_c5(h):
    n = h.size
    a = h.astype(np.float64)
    b = (1.0 - h).astype(np.float64)
    # length for linear correlation via FFT
    m = 2 * n - 1
    n_fft = 1 << (m - 1).bit_length()   # next power of 2 >= m
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    corr = np.fft.irfft(A * np.conj(B), n=n_fft)[:m]
    # scaling factor dx = 2.0 / n
    return np.max(corr) * (2.0 / n)

def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            # compute gradient
            corr = correl_fft(h)   # uses FFT
            C = corr * (2.0 / n)
            bc = beta * C
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w_sum = np.sum(w)
            if w_sum == 0:
                w = np.ones_like(w) / len(w)
            else:
                w /= w_sum
            # compute gradient
            g = np.ones(n)
            m = n - 1
            # loop over shifts
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - m
                if k >= 0:
                    g[: n - k] -= wk * h[k:]
                    g[k:] -= wk * h[: n - k]
                else:
                    kk = -k
                    g[kk:] -= wk * h[: n - kk]
                    g[: n - kk] -= wk * h[kk:]
            g = (2.0 / n) * g
            # step
            new_h = h - step * g
            new_h = project_onto_box_sum(new_h, n/2.0)
            new_h = enforce_symmetry(new_h)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c, best_h = c_new, new_h.copy()
                h = new_h
            else:
                # accept occasionally
                if rng.random() < 0.005:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = 0.5 * (lo + hi)
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    proj = np.clip(v + lam, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


# ---------- symmetry ----------
def enforce_symmetry(v):
    return (v + v[::-1]) * 0.5


# ---------- random feasible ----------
def random_feasible_symmetric(n, rng):
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    h = np.concatenate([x, x[::-1]])
    h = project_onto_box_sum(h, n / 2.0)
    return h


# ---------- FFT correlation ----------
def correl_fft(h):
    n = h.size
    a = h.astype(np.float64)
    b = (1.0 - h).astype(np.float64)
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    corr = np.fft.irfft(A * np.conj(B), n=n_fft)[:m].real
    return corr


def compute_c5(h):
    n = h.size
    corr = correl_fft(h)
    return np.max(corr) * (2.0 / n)


# ---------- hill climbing ----------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = enforce_symmetry(new_h)
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 10) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- simulated annealing ----------
def anneal_optimize(h0, rng, time_limit):
    start = time.time()
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    target = cur_h.size / 2.0
    T0 = 0.02
    while time.time() - start < time_limit:
        n = cur_h.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = enforce_symmetry(new_h)
        new_h = project_onto_box_sum(new_h, target)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        else:
            elapsed = time.time() - start
            frac = elapsed / time_limit
            T = T0 * (1.0 - frac)
            if T > 1e-8 and rng.random() < np.exp(-(new_c - cur_c) / T):
                cur_h, cur_c = new_h, new_c
        if rng.random() < 0.001:
            cur_h, cur_c = best_h.copy(), best_c
    return best_h, best_c


# ---------- smooth‑max gradient descent ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = correl_fft(h)
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            g = np.ones(n)
            mid = n - 1
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - mid
                if k >= 0:
                    g[: n - k] -= wk * h[k:]
                    g[k:] -= wk * h[: n - k]
                else:
                    kk = -k
                    g[kk:] -= wk * h[: n - kk]
                    g[: n - kk] -= wk * h[kk:]
            g = (2.0 / n) * g
            new_h = h - step * g
            new_h = project_onto_box_sum(new_h, n / 2.0)
            new_h = enforce_symmetry(new_h)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c, best_h, h = c_new, new_h.copy(), new_h
            else:
                if rng.random() < 0.005:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- SLSQP (smooth‑max surrogate) ----------
def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        C = corr * (2.0 / n)
        bc = beta * C
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def eq_cons(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_cons}
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, target)
    return x
