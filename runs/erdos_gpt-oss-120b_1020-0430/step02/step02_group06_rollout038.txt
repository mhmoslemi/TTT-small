def c5_from_h(h):
    n = h.shape[0]
    a = h
    b = 1 - h
    # linear correlation: conv of a and reversed b.
    # Use FFT: pad to L = 2**int(np.ceil(np.log2(2*n-1))).
    L = 1 << (2*n-1).bit_length()
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b[::-1], L)   # reversed b
    corr = np.fft.irfft(A * B, L)[:2*n-1]
    # Because correlation defined as sum_i a_i * b_{i+k} for shift k, k from -(n-1) to n-1.
    # That's exactly conv of a and reversed b.
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def hill_climb(h, time_end):
    n = len(h)
    cur = h.copy()
    cur_c5 = c5_from_h(cur)
    best = cur.copy()
    best_c5 = cur_c5
    start = time.time()
    # parameters
    sigma = 0.3
    while time.time() < time_end:
        # degrade sigma gradually
        elapsed = time.time() - start
        total = time_end - start
        # exponential cooling
        sigma = 0.3 * (0.01 ** (elapsed / total))  # from 0.3 to 0.003
        # choose small subset indices
        k = max(1, int(0.05 * n))
        idx = np.random.choice(n, size=k, replace=False)
        trial = cur.copy()
        trial[idx] += sigma * np.random.randn(k)
        trial = project_to_feasible(trial, n)
        trial_c5 = c5_from_h(trial)
        if trial_c5 < cur_c5:
            cur = trial
            cur_c5 = trial_c5
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
    return best, best_c5

def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    current_sum = h.sum()
    # if already correct return
    if abs(current_sum - target) < 1e-12:
        return h
    # Determine direction
    # Use water-filling algorithm:
    # While not close, adjust free variables.
    # We'll use simple approach: compute diff = current_sum - target.
    # If diff > 0: need to reduce sum, so subtract diff equally from all entries >0.
    # If diff <0: need to increase sum, add diff equally to all entries <1.
    # Iterate few times.
    max_iter = 100
    for _ in range(max_iter):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        # subtract if diff>0, else add.
        h[free_idx] -= delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # Final minor correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # add/subtract to/from first free variable
        if residual > 0:
            # need to increase sum
            idx = np.where(h < 1 - 1e-12)[0]

                sub = min(-residual, h[idx[0]])

import time
import numpy as np

# ---------- Utility functions ----------

def _project_to_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h)=n/2 via water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # iterative water‑filling
    for _ in range(100):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size:
                add = min(residual, 1 - h[idx[0]])
                h[idx[0]] += add
                residual -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size:
                sub = min(-residual, h[idx[0]])
                h[idx[0]] -= sub
                residual += sub
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute the C5 value for a discretised function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _subgradient(h: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k Σ_i h_i(1-h_{i+k}) w.r.t. h.
    Uses all shifts attaining the maximal correlation.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # tolerance to collect ties
    idxs = np.where(np.abs(corr - max_val) < 1e-9)[0]
    shifts = idxs - (n - 1)            # convert index to shift
    total = np.zeros(n)
    for K in shifts:
        g = np.ones(n)
        if K >= 0:
            g[: n - K] -= h[K:]
            g[K:] -= h[: n - K]
        else:
            L = -K
            g[L:] -= h[: n - L]
            g[: n - L] -= h[L:]
        total += g
    return total / len(shifts)


def _hill_climb(start_h: np.ndarray, time_end: float) -> tuple:
    """
    Greedy hill climbing with occasional gradient steps.
    Returns best (h, C5) found before time_end.
    """
    n = start_h.size
    cur = _project_to_feasible(start_h, n)
    cur_c5 = _c5_from_h(cur)
    best = cur.copy()
    best_c5 = cur_c5

    sigma_start = 0.30
    sigma_end = 0.001
    t0 = time.time()
    while time.time() < time_end:
        # schedule decreasing sigma (geometric)
        prog = (time.time() - t0) / (time_end - t0)
        sigma = sigma_start * (sigma_end / sigma_start) ** prog

        # 20% of the time try a gradient step, otherwise random perturbation
        if np.random.rand() < 0.2:
            trial = cur - sigma * _subgradient(cur)
        else:
            k = max(1, int(0.05 * n))                     # modify ~5% of entries
            idx = np.random.choice(n, size=k, replace=False)
            trial = cur.copy()
            trial[idx] += sigma * np.random.randn(k)

        trial = _project_to_feasible(trial, n)
        trial_c5 = _c5_from_h(trial)

        if trial_c5 < cur_c5:
            cur, cur_c5 = trial, trial_c5
            if trial_c5 < best_c5:
                best, best_c5 = trial.copy(), trial_c5
    return best, best_c5

            # random perturbation of the base as starting point
            start = base_h + 0.35 * np.random.randn(n)
            start = _project_to_feasible(start, n)
            h_opt, c5_opt = _hill_climb(start, time.time() + restart_budget)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                if best_c5 < target:
                    break
            # shrink perturbation radius a bit for the next restart
            # (handled implicitly by the random scale used above)

        if best_c5 < target:
            break

    # final polishing if time permits
    if best_h is not None and time_left() > 5.0:
        final_h, final_c5 = _hill_climb(best_h, time.time() + min(8.0, time_left()))
        if final_c5 < best_c5:
            best_h, best_c5, best_n = final_h, final_c5, best_h.size

    # safety fallback
    if best_h is None:
        best_n = 96
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
