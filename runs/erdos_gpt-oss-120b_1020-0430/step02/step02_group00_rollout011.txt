def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    xs_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xs_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xs_f, xs_c, coarse_h)


def _triangular_h(n: int) -> np.ndarray:
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.where(xs < 1.0, xs, 2.0 - xs)
    return _project_to_feasible(h, n)


def _gradient_refine(h: np.ndarray, steps: int, base_alpha: float = 0.015) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h.size
    best_h, best_val = h.copy(), _c5_from_h(h)
    for it in range(steps):
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        d = abs(k)
        grad = np.ones(n)
        if d:
            grad[: n - d] -= h[d:]
            grad[d:] -= h[: n - d]
        alpha = base_alpha / np.sqrt(it + 1)
        h = h - alpha * grad
        h = _project_to_feasible(h, n)
        val = _c5_from_h(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
    return best_h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial construction (if provided)
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None

    def update(h):
        nonlocal best_h, best_c5, best_n
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), float(c5), h.size

    if init_h is not None:
        init_h = _project_to_feasible(init_h, init_h.size)
        update(init_h)

    # candidate sizes (include some coarse and finer grids)
    cand_sizes = [128, 160, 200, 240, 256, 300, 360, 400, 512]
    if best_n:
        cand_sizes.append(best_n)
    cand_sizes = sorted(set(cand_sizes))

    for n in cand_sizes:
        if time_left() < 12.0:
            break

        # base start vector for this resolution
        if best_h is not None and best_n == n:
            base = best_h.copy()
        elif best_h is not None:
            base = _upscale_h(best_h, n)
        elif init_h is not None and init_h.size == n:
            base = init_h.copy()
        else:
            base = _triangular_h(n)

        base = _project_to_feasible(base, n)

        # allocate attempts proportionally to remaining time
        remaining = time_left()
        attempts = max(6, int(remaining // (len(cand_sizes) * 0.8)))
        attempts = min(attempts, 30)

        for att in range(attempts):
            if time_left() < 8.0:
                break

            # generate start point
            if att == 0:
                start_h = base.copy()
            elif att % 5 == 0:
                start_h = np.random.rand(n)
                start_h = _project_to_feasible(start_h, n)
            else:
                sigma = 0.18 * (0.85 ** att)
                start_h = base + sigma * np.random.randn(n)
                start_h = _project_to_feasible(start_h, n)

            # local SLSQP refinement
            max_iter = int(500 + 150 * np.log2(n / 128 + 1))
            max_iter = min(max_iter, 2500)
            h_opt, c5_opt = _solve_h(start_h, max_iter=max_iter)
            update(h_opt)

            # optional gradient polishing (small budget)
            if time_left() > 4.0:
                steps = 800 if n <= 256 else 1500
                h_ref = _gradient_refine(h_opt, steps=steps)
                update(h_ref)

            # shift base if improvement
            if c5_opt < best_c5 * 0.999:
                base = h_opt

            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # final intensive polish on the best solution
    if best_h is not None and time_left() > 6.0:
        h_opt, c5_opt = _solve_h(best_h, max_iter=3000)
        update(h_opt)
        if time_left() > 4.0:
            h_ref = _gradient_refine(best_h, steps=2000)
            update(h_ref)

    # safety fallback
    if best_h is None:
        best_h = _triangular_h(256)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)
    return best_h.tolist(), float(best_c5), int(best_h.size)
