def projected_gradient_descent(h, rng, steps=200, step_size=0.1):
    n=len(h)
    target=n/2.0
    for it in range(steps):
        # compute gradient for max shift
        corr = np.correlate(h, 1-h, mode='full')
        k_idx = np.argmax(corr) - (n - 1)
        k = k_idx
        g = np.zeros_like(h)
        if k == 0:
            g = 1 - 2*h
        else:
            # positive k
            if k > 0:
                # i from 0 to n-1-k
                g[:n-k] += 1 - h[k:]
                # i from k to n-1  (i-k >=0)
                g[k:] -= h[:n-k]
            else: # k < 0
                kk = -k
                # similar
                g[kk:] += 1 - h[:n-kk]
                g[:n-kk] -= h[kk:]
        # step size schedule
        alpha = step_size * (0.95**it)
        h = h - alpha * g
        # project
        np.clip(h, 0.0, 1.0, out=h)
        h = repair_sum(h, target, rng)
   return h

def hill_climb_swap_binary(h, rng, max_iters=5000):
    n=len(h)
    target=n//2
    # assert binary
    best=h.copy()
    best_c=compute_c5(best)
    ones = np.flatnonzero(best==1)
    zeros = np.flatnonzero(best==0)
    for it in range(max_iters):
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new = best.copy()
        new[i]=0
        new[j]=1
        c=compute_c5(new)
        if c < best_c - 1e-12: # accept improvement
            best = new
            best_c = c
            # update ones/zeros
            ones = np.flatnonzero(best==1)
            zeros = np.flatnonzero(best==0)
    return best, best_c

def compute_c5(h):

    return np.max(corr) * (2.0 / n)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)

            h = new_h

    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')

import time
import numpy as np
import scipy.optimize as opt

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log‑sum‑exp (scaled by CURRENT_BETA)."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # (log‑sum‑exp)/beta  with numerical stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def repair_sum(x, target, rng, max_iter=2000):
    """Adjust x to satisfy Σx = target while keeping 0≤x≤1."""
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, x.size)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = repair_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb accepting only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def hill_climb_swap_binary(h, rng, max_iters=2000):
    """Binary hill‑climb: swap a 1 with a 0 if it improves C5."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_iters):
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
    return best_h, best_c


def gradient_max_shift(h):
    """Gradient of the worst shift (max overlap) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    g = np.zeros_like(h)
    if k == 0:
        g = 1 - 2 * h
    elif k > 0:
        g[: n - k] += 1 - h[k:]
        g[k:] -= h[: n - k]
    else:  # k < 0
        kk = -k
        g[kk:] += 1 - h[: n - kk]
        g[: n - kk] -= h[kk:]
    return g, k


def projected_gradient_descent(h, rng, steps=200, step_size=0.1):
    """Projected sub‑gradient descent on the exact max‑overlap objective."""
    n = h.size
    target = n / 2.0
    for it in range(steps):
        g, _ = gradient_max_shift(h)
        alpha = step_size * (0.94 ** it)
        h = h - alpha * g
        np.clip(h, 0.0, 1.0, out=h)
        h = repair_sum(h, target, rng, max_iter=500)
    return h


def slsqp_optimize(h0, beta, rng, max_iter=250):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = repair_sum(x, target, rng, max_iter=2000)
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Take parent solution if provided
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = repair_sum(h0, n0 / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # ------------------------------------------------------------------
    # Candidate discretisations (keep modest for speed)
    # ------------------------------------------------------------------
    cand_ns = [120, 180, 240]
    if best_n is not None and best_n not in cand_ns:
        cand_ns.insert(0, best_n)
    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for n in cand_ns:
        if time.time() - start_time > budget_s * 0.90:
            break
        target_sum = n / 2.0
        # Prepare a handful of diverse initial vectors
        init_vecs = []
        if best_n == n and best_h is not None:
            init_vecs.append(best_h.copy())
        # random binary starts
        for _ in range(3):
            h_bin = np.zeros(n)
            idx = rng.choice(n, int(target_sum), replace=False)
            h_bin[idx] = 1.0
            init_vecs.append(h_bin)
        # random continuous starts
        for _ in range(2):
            init_vecs.append(random_feasible_vector(n, rng))

        for h0 in init_vecs:
            if time.time() - start_time > budget_s * 0.95:
                break

            # ------------------------------------------------------------------
            # Local optimisation depending on binary / continuous nature
            # ------------------------------------------------------------------
            if np.all(np.logical_or(h0 == 0.0, h0 == 1.0)):
                # binary hill‑climb
                h_bc, c_bc = hill_climb_swap_binary(h0, rng, max_iters=2500)
                # gradient refinement from binary point
                h_gd = projected_gradient_descent(h_bc.copy(), rng,
                                                   steps=400, step_size=0.2)
                c_gd = compute_c5(h_gd)
                h_cur, c_cur = (h_bc, c_bc) if c_bc < c_gd else (h_gd, c_gd)
            else:
                # continuous hill‑climb
                h_hc, c_hc = hill_climb_continuous(h0, rng,
                                                   max_iters=2500, start_step=0.5)
                # gradient refinement
                h_gd = projected_gradient_descent(h_hc.copy(), rng,
                                                   steps=400, step_size=0.15)
                c_gd = compute_c5(h_gd)
                h_cur, c_cur = (h_hc, c_hc) if c_hc < c_gd else (h_gd, c_gd)

            if c_cur < best_c:
                best_h, best_c, best_n = h_cur.copy(), c_cur, n

            # ------------------------------------------------------------------
            # SLSQP refinement with increasing beta
            # ------------------------------------------------------------------
            if time.time() - start_time < budget_s * 0.97:
                for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                    if time.time() - start_time > budget_s * 0.98:
                        break
                    try:
                        h_opt = slsqp_optimize(h_cur.copy(), beta, rng,
                                               max_iter=300 if beta < 100 else 500)
                    except Exception:
                        continue
                    c_opt = compute_c5(h_opt)
                    if c_opt < best_c:
                        best_h, best_c, best_n = h_opt.copy(), c_opt, n
                        h_cur = h_opt  # keep for possible further refinement
                # one extra gradient pass after SLSQP
                if time.time() - start_time < budget_s * 0.99:
                    h_cur = projected_gradient_descent(best_h.copy(), rng,
                                                       steps=200, step_size=0.1)
                    c_cur = compute_c5(h_cur)
                    if c_cur < best_c:
                        best_h, best_c, best_n = h_cur.copy(), c_cur, n

            # Early exit if target achieved
            if best_c < 0.3808:
                break

    # ------------------------------------------------------------------
    # Optional up‑sampling refinement (if budget permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        if best_n * 2 <= 300:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            # refine at higher resolution
            h_up = projected_gradient_descent(h_up, rng,
                                              steps=600, step_size=0.09)
            h_up, c_up = hill_climb_continuous(h_up, rng,
                                               max_iters=2000, start_step=0.3)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final feasibility enforcement
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_sum(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
