        b = h[k:]

import time, math, numpy as np

# functions definitions
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time, math
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Generate random h with 0≤h≤1 and Σh = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, scale=0.2):
    """Move mass from the donor side of the worst shift to its partner."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # shift giving the maximal overlap
    if k == 0:
        return h
    if k > 0:
        a = h[: n - k]
        b = h[k:]
        weight = a * (1 - b)
        i = int(np.argmax(weight))
        donor = i
        receiver = i + k
    else:
        k_abs = -k
        a = h[k_abs:]
        b = h[: n - k_abs]
        weight = a * (1 - b)
        i = int(np.argmax(weight))
        donor = i + k_abs
        receiver = i
    max_eps = min(h[donor], 1.0 - h[receiver], scale * weight[i])
    if max_eps <= 0:
        return perturb_h(h, rng, scale=scale)
    eps = rng.random() * max_eps
    new_h = h.copy()
    new_h[donor] -= eps
    new_h[receiver] += eps
    return new_h

def targeted_hill_climb(h0, rng, max_iters=2000, start_scale=0.5, min_scale=1e-4):
    """Deterministic hill‑climb using targeted moves."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    cur_c = best_c
    scale = start_scale
    for it in range(max_iters):
        new_h = targeted_perturb(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
        if c < cur_c:
            h, cur_c = new_h, c
        if (it + 1) % (max_iters // 5 + 1) == 0:
            scale = max(scale * 0.7, min_scale)
    return best_h, best_c

    """SA with targeted moves."""
