def gradient_descent_step(h, rng, steps=100, step_size=0.01):
    h = h.copy()
    n = h.size
    for _ in range(steps):
        # compute correlation, find max shift index
        corr = np.correlate(h, 1-h, mode='full')
        k_idx = np.argmax(corr)  # index in full array
        # convert index to shift offset: shift = k_idx - (len(h)-1)
        shift = k_idx - (n-1)
        # compute gradient vector g: g_i = (1 - h_{i+shift}) - h_{i-shift}
        # where out-of-bounds indices have h=0.
        # Use vectorized with np.roll and mask for out-of-bounds
        g = np.zeros_like(h)
        # compute forward term
        if shift >= 0:
            # i+shift within bound for i <= n-1-shift

            g[backward_idx] -= h[backward_idx - shift]
        else:

            backward_idx = np.arange(0, n+shift)  # n+shift < n
            g[backward_idx] -= h[backward_idx - shift]
        # Take step
        h_new = h - step_size * g

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    h = x
    corr = np.correlate(h, 1 - h, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤hi≤1, Σhi=n/2)."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # 1. Choose discretisation size
    # -------------------------------------------------------------
    # Prefer a moderately fine grid but still cheap to evaluate.
    n = 240                     # must be even for symmetry, < 1000
    dx = 2.0 / n
    target_sum = n / 2.0

    # -------------------------------------------------------------
    # 2. Build a pool of diverse initial candidates
    # -------------------------------------------------------------
    candidates = []

    # raised cosine (full period) – already satisfies the integral
    xs = (np.arange(n) + 0.5) * dx - 1.0
    h_cos = 0.5 * (1.0 + np.cos(np.pi * xs))
    candidates.append(enforce_sum(h_cos, target_sum, rng))

    # simple sine / cosine families
    xs_pos = (np.arange(n) + 0.5) * dx
    for f in (1, 2, 3, 4):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs_pos)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), target_sum, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs_pos)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), target_sum, rng))

    # linear ramps
    candidates.append(enforce_sum(xs_pos / 2.0, target_sum, rng))
    candidates.append(enforce_sum(1.0 - xs_pos / 2.0, target_sum, rng))

    # a few random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # incorporate the provided construction if it matches our size
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)

    # -------------------------------------------------------------
    # 3. Pick the best of the pool
    # -------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    global_best_h = best_h
    global_best_c = best_c

    # -------------------------------------------------------------
    # 4. Main optimisation loop (hill‑climb + SLSQP ladder)
    # -------------------------------------------------------------
    beta_list = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    while time.time() - start_time < budget_s * 0.85:
        # start from a perturbed best or a fresh random point
        if rng.random() < 0.6:
            start = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # local hill‑climb (more iterations than before)
        h_opt, c_opt = hill_climb(start, rng, max_iters=3000, start_step=0.4)

        # SLSQP refinement with increasingly tight β
        for beta in beta_list:

                break
            try:
                h_opt = slsqp_optimize(h_opt, beta, max_iter=500)
            except Exception:
                continue
            c_opt = compute_c5(h_opt)
            if c_opt < global_best_c:
                global_best_h, global_best_c = h_opt.copy(), c_opt

        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

    # -------------------------------------------------------------
    # 5. Final clean‑up (clip + exact sum)
    # -------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, target_sum, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
