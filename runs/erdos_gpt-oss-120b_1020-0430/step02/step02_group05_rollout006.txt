def hill_climb(h, max_iter, time_left):
    n = len(h)
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    while iter < max_iter and time_left() > 2.0:
        # compute corr
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = corr.max()
        lag = np.argmax(corr) - (n - 1)
        # gradient for this lag
        g = np.ones(n)  # base 1
        # subtract h[i+lag] where valid
        if lag >= 0:
            g[:n-lag] -= best_h[lag:]
            g[lag:] -= best_h[:n-lag]
        else: # lag < 0
            L = -lag
            g[L:] -= best_h[:n-L]
            g[:n-L] -= best_h[L:]
        # we want direction to reduce max_corr: d = -g (since derivative of corr)
        # but need sum(d)=0; for simplicity, choose increase at negative g, decrease at positive g.
        # Determine indices sorted by g
        idx_pos = np.where(g > 0)[0]
        idx_neg = np.where(g < 0)[0]
        if idx_pos.size == 0 or idx_neg.size == 0:
            # random perturb small
            delta = 0.02 * (np.random.rand(n) - 0.5)
        else:
            # choose some indices
            k = min(5, idx_pos.size, idx_neg.size)
            # pick largest positive g for decrease
            dec_idx = idx_pos[np.argsort(-g[idx_pos])[:k]]
            # pick most negative g for increase
            inc_idx = idx_neg[np.argsort(g[idx_neg])[:k]]
            # compute possible delta
            # max decrease amount each dec_idx can give: h[dec_idx] (to zero)
            max_dec = h[dec_idx].min()
            # max increase each inc_idx can accept: 1 - h[inc_idx]
            max_inc = (1 - h[inc_idx]).min()
            step = min(0.01, max_dec, max_inc)
            delta = np.zeros(n)
            delta[dec_idx] -= step
            delta[inc_idx] += step
        # apply delta
        new_h = best_h + delta
        # project to feasible region
        new_h = _project_to_simplex_box(new_h, n/2.0)
        new_val = _c5_from_h(new_h)
        if new_val < best_val - 1e-9: # improvement
            best_h, best_val = new_h, new_val
        else:
            # maybe accept with some probability (simulated annealing)
            # but we can just skip.
            pass
        iter += 1

def _hill_climb(h, time_left, max_iter=20000):
    n = len(h)
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    T0 = 0.005
    for it in range(max_iter):
        if time_left() < 0.5:
            break
        # propose random move: pick i,j distinct
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # compute allowed delta
        max_up = 1.0 - cur_h[i]
        max_down = cur_h[j]
        delta_max = min(max_up, max_down, 0.02)
        if delta_max <= 0:
            continue
        delta = delta_max * np.random.rand()
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # maintain sum automatically (since we transferred)
        # Ensure within [0,1] - already satisfied by delta
        new_val = _c5_from_h(new_h)
        # accept if improved or probabilistically
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
        else:
            # annealing
            T = T0 * (1 - it / max_iter)
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
        # Occasionally do a gradient-guided move to speed up
        if it % 1000 == 0:
            # gradient on max lag
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            lag = np.argmax(corr) - (n - 1)
            # compute gradient g = 1 - (h[i+lag] + h[i-lag])
            g = np.ones(n)
            if lag >= 0:
                g[:n-lag] -= cur_h[lag:]
                g[lag:] -= cur_h[:n-lag]
            else:
                L = -lag
                g[L:] -= cur_h[:n-L]
                g[:n-L] -= cur_h[L:]
            # choose top pos and neg
            pos_idx = np.where(g > 0)[0]
            neg_idx = np.where(g < 0)[0]
            if pos_idx.size and neg_idx.size:
                # move small mass from pos to neg
                i = pos_idx[np.argmax(g[pos_idx])]
                j = neg_idx[np.argmin(g[neg_idx])]
                step = min(0.01, cur_h[i], 1 - cur_h[j])
                if step > 1e-6:
                    new_h = cur_h.copy()
                    new_h[i] -= step
                    new_h[j] += step
                    new_val = _c5_from_h(new_h)
                    if new_val < cur_val:
                        cur_h, cur_val = new_h, new_val
                        if cur_val < best_val:
                            best_h, best_val = cur_h.copy(), cur_val

import time
import numpy as np
from scipy import optimize

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x | 0<=x_i<=1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x):
    return x[-1]

def _solve_h(start_h: np.ndarray, max_iter: int = 400):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds,
        constraints=cons, options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

    cur_val = best_val
    T0 = 0.006
    for it in range(max_iter):

        if i == j:

        up = 1.0 - cur_h[i]
        down = cur_h[j]
        step = min(up, down, 0.015) * np.random.rand()
        if step <= 1e-9:

        new_h = cur_h.copy()
        new_h[i] += step
        new_h[j] -= step
        new_val = _c5_from_h(new_h)
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
        else:
            T = T0 * (1.0 - it / max_iter)
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
        # every 800 steps try a gradient‑guided transfer on the current worst lag
        if it % 800 == 0:
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            lag = np.argmax(corr) - (n - 1)
