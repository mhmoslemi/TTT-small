def anneal_binary(h, max_iter, time_limit):
    n = h.size
    best_h = h.copy()
    best_corr, best_c5 = max_corr_from_h(best_h)
    cur_h = best_h.copy()
    cur_corr = best_corr
    T0 = 5.0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # temperature schedule
        T = T0 * (1 - it / max_iter)
        # compute correlation array to identify max shift
        corr_arr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr_arr)
        max_corr = corr_arr[max_idx]
        # compute shift
        shift = max_idx - (n - 1)
        # get mismatching pair indices for this shift
        if shift >= 0:
            i_vals = np.arange(0, n - shift)
            j_vals = i_vals + shift
        else:
            i_vals = np.arange(-shift, n)
            j_vals = i_vals + shift
        # mismatches where cur_h[i]==1 and cur_h[j]==0
        mism_i = i_vals[(cur_h[i_vals] == 1) & (cur_h[j_vals] == 0)]
        if mism_i.size > 0:
            i = np.random.choice(mism_i)
            j = i + shift
        else:
            # fallback: pick random 1 index
            ones = np.where(cur_h == 1)[0]
            i = np.random.choice(ones)
            j = None

        # pick random other index p = 0 where cur_h[p]==0 and p!=j
        zeros = np.where(cur_h == 0)[0]
        if zeros.size == 0:
            break
        p = np.random.choice(zeros)
        # Ensure we are swapping i (1) with p (0)
        # If i==p skip
        if p == i:
            continue
        # create candidate by swapping
        cand = cur_h.copy()
        cand[i], cand[p] = cand[p], cand[i]  # swap
        cand_corr, cand_c5 = max_corr_from_h(cand)
        delta = cand_corr - cur_corr
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_h = cand
            cur_corr = cand_corr
            # update best
            if cur_corr < best_corr:
                best_h = cur_h.copy()
                best_corr = cur_corr
        # optionally early break if best bound reached
        if best_corr * (2.0 / n) < 0.3808:
            break
    return best_h, best_corr

def anneal_binary(h, max_iter, time_limit):
    n = h.size
    best_h = h.copy()
    best_corr, _ = max_corr_from_h(best_h)
    cur_h = best_h.copy()
    cur_corr = best_corr
    temperature = 5.0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # cooling
        T = temperature * (1 - it / max_iter)
        # compute correlation array
        corr_arr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr_arr)
        shift = max_idx - (n - 1)
        # mismatching indices for shift
        if shift >= 0:
            i_vals = np.arange(0, n - shift)
            j_vals = i_vals + shift
        else:
            i_vals = np.arange(-shift, n)
            j_vals = i_vals + shift
        # mismatches set
        mask = (cur_h[i_vals] == 1) & (cur_h[j_vals] == 0)
        mism_i = i_vals[mask]
        # Choose a swap candidate
        if mism_i.size > 0:
            i = int(np.random.choice(mism_i))
            # choose p from zeros not equal i
            zeros = np.where(cur_h == 0)[0]
            p = int(np.random.choice(zeros))
            # ensure p != i, but i is 1 so not zero
        else:
            # fallback random 1 and 0
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i = int(np.random.choice(ones))
            p = int(np.random.choice(zeros))
        # make swap
        cand = cur_h.copy()
        cand[i], cand[p] = cand[p], cand[i]
        cand_corr, _ = max_corr_from_h(cand)
        delta = cand_corr - cur_corr
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_h = cand
            cur_corr = cand_corr
            if cur_corr < best_corr:
                best_h = cur_h.copy()
                best_corr = cur_corr
        # optional early break
        if best_corr * (2.0 / n) < 0.3808:
            break
    return best_h, best_corr

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Core helpers (same as previous solution, slightly trimmed)

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given step vector (integral already satisfied)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """Run a single SLSQP optimisation starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)  # the objective (raw correlation)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    constraints = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    x_fine = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    return np.interp(x_fine, x_coarse, coarse_h)


def binary_from_real(r: np.ndarray) -> np.ndarray:
    """Round a real vector to 0/1 while keeping exactly half ones."""
    n = r.size
    h = (r >= 0.5).astype(float)
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        ones = np.where(h == 1)[0]
        flip = np.random.choice(ones, cur - target, replace=False)
        h[flip] = 0.0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        flip = np.random.choice(zeros, target - cur, replace=False)
        h[flip] = 1.0
    return h


def random_binary(n: int) -> np.ndarray:
    """Uniform random 0/1 vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def simulated_annealing_h(h: np.ndarray, max_iter: int, time_limit: float) -> tuple:
    """Annealing on a feasible vector (preserves sum, box constraints)."""
    n = h.size
    cur = h.copy()
    cur_c5 = _c5_from_h(cur)
    best = cur.copy()
    best_c5 = cur_c5
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # temperature schedule (starts around 1.0, cools to ~0)
        T = max(0.01, 1.0 * (1.0 - it / max_iter))
        # pick i with h_i>0 and j with h_j<1
        i = np.random.randint(n)
        while cur[i] <= 1e-12:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while cur[j] >= 1 - 1e-12:
            j = np.random.randint(n)
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c5 = _c5_from_h(cand)
        delta = cand_c5 - cur_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            cur = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3808:
                    break
    return best, best_c5
