def simple_proj(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo+hi)/2
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # fix any small residual
    residual = target - proj.sum()
    if abs(residual) > 1e-12:
        # adjust first element within bounds
        for i in range(len(proj)):
            if residual > 0 and proj[i] < 1 - 1e-12:
                add = min(residual, 1 - proj[i])
                proj[i] += add
                residual -= add
            elif residual < 0 and proj[i] > 1e-12:
                sub = min(-residual, proj[i])
                proj[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return proj

def ineq_jac(x):

    # Build gradient matrix of shape (2*n-1, n+1)
    m = 2 * n - 1
    grad = np.empty((m, n+1), dtype=float)
    # derivative w.r.t t is 1
    grad[:, -1] = 1.0
    # Compute gradient w.r.t h
    # Use loops over shifts for speed
    shifts = np.arange(-n+1, n)  # length m
    i = np.arange(n)

        ip = i + s

        grad[idx, :n] = -term1 + term2
    return grad

def solve_h_grad(initial_h, max_iter=500):
    n = initial_h.size
    h0 = simple_proj(initial_h, n/2.0)
    corr0 = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr0)  # initial guess for t
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type":"ineq", "fun": ineq_fun, "jac": ineq_jac},
        {"type":"eq", "fun": eq_fun, "jac": eq_jac},
    ]
    opts = {"maxiter": max_iter, "ftol":1e-9, "disp":False}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                            constraints=constraints, options=opts)
    if not res.success:
        # fallback: use h0
        h_opt = h0
    else:
        h_opt = res.x[:n]
    # Ensure feasible projection (in case of numeric violation)
    h_opt = simple_proj(h_opt, n/2.0)
    c5 = compute_c5(h_opt)
    return h_opt, c5

def hill_climb(h, steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # precompute random pairs maybe
    while steps>0 and time.time() < time_limit:
        steps -= 1
        # choose i,j randomly
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # compute max epsilon
        max_down = best_h[i]
        max_up = 1 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        # maybe adapt epsilon size
        eps = np.random.rand() * max_eps
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Projection onto the intersection of a hyperplane (sum = target) and [0,1] box
# ----------------------------------------------------------------------
def _project_to_feasible(v: np.ndarray, target: float, iters: int = 50) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(iters):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction
    res = target - proj.sum()
    if abs(res) > 1e-12:
        for i in range(len(proj)):
            if res > 0 and proj[i] < 1 - 1e-12:
                add = min(res, 1 - proj[i])
                proj[i] += add
                res -= add
            elif res < 0 and proj[i] > 1e-12:
                sub = min(-res, proj[i])
                proj[i] -= sub
                res += sub
            if abs(res) < 1e-12:
                break
    return proj


# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


# ----------------------------------------------------------------------
# Constraint definitions (with analytical Jacobians)
# ----------------------------------------------------------------------
def _ineq_fun(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _ineq_jac(x: np.ndarray) -> np.ndarray:
    """Jacobian of the inequality g_k(x)=t-C(k)  (shape (2n-1, n+1))."""
    n = x.size - 1
    h = x[:n]

    i = np.arange(n)[:, None]                 # (n,1)
    shifts = np.arange(-n + 1, n)[None, :]    # (1,2n-1)

    ip = i + shifts                           # (n,2n-1)
    im = i - shifts

    term1 = np.full((n, 2 * n - 1), 1.0, dtype=float)
    mask1 = (ip >= 0) & (ip < n)
    term1[mask1] = 1.0 - h[ip[mask1]]

    term2 = np.zeros((n, 2 * n - 1), dtype=float)
    mask2 = (im >= 0) & (im < n)
    term2[mask2] = h[im[mask2]]

    grad_h = (-term1 + term2).T               # (2n-1, n)
    grad = np.hstack([grad_h, np.ones((2 * n - 1, 1), dtype=float)])
    return grad


def _eq_fun(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - n / 2.0


def _eq_jac(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    g = np.zeros(n + 1, dtype=float)
    g[:n] = 1.0
    return g


# ----------------------------------------------------------------------
# Local optimiser using SLSQP with analytical Jacobians
# ----------------------------------------------------------------------
def _solve_h(start_h: np.ndarray, max_iter: int = 800) -> tuple:
    n = start_h.size
    h0 = _project_to_feasible(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [
        {"type": "ineq", "fun": _ineq_fun, "jac": _ineq_jac},
        {"type": "eq", "fun": _eq_fun, "jac": _eq_jac},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    while steps and time.time() < time_limit:
        steps -= 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        down = best_h[i]
        up = 1.0 - best_h[j]
        eps = np.random.rand() * min(down, up) * 0.5
        if eps <= 0:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Linear interpolation to change resolution
# ----------------------------------------------------------------------
def _upscale(h: np.ndarray, n_new: int) -> np.ndarray:
    n_old = h.size
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)
