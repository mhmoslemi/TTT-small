def fix_worst_shift(h, rng, amount=0.1):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # find index of max C
    idx = np.argmax(corr)
    k = idx - (n - 1)  # shift
    # compute contributions
    # create arrays of shifted h
    # We'll compute sum_shift = np.zeros(n)
    # we can compute sum_shift[i] = (h[i+k] if i+k<n else 0) + (h[i-k] if i-k>=0 else 0)
    # derivative of C w.r.t h_i: 1 - sum_shift[i] (dx factor omitted)
    # So direction to reduce C is opposite of gradient: if gradient positive (1 - sum>0) then decreasing h_i reduces C (since h_i large increases C), else increasing h_i reduces C.
    # So we compute desired sign: if 1 - sum_shift[i] > 0 => decrease h_i; else increase h_i.
    # Let's compute pos_decrease = indices where (1 - sum_shift) > 0 and h_i > 0 (can reduce)
    # compute pos_increase = indices where (1 - sum_shift) < 0 and h_i < 1.
    # Choose i_neg (to decrease) where gradient is most positive, i_pos (to increase) where gradient most negative.
    # Then move epsilon = amount * min( h[i_neg], 1 - h[i_pos]).
    #
    sum_shift = np.zeros(n)
    if k >= 0:
        sum_shift[: n - k] += h[k:]
        sum_shift[k:] += h[: n - k]
    else:
        kk = -k
        sum_shift[kk:] += h[: n - kk]
        sum_shift[: n - kk] += h[kk:]
    grad = 1.0 - sum_shift
    # find candidate indices
    dec_candidates = np.where((grad > 1e-12) & (h > 1e-12))[0]
    inc_candidates = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
    if dec_candidates.size == 0 or inc_candidates.size == 0:
        return h
    i_dec = dec_candidates[np.argmax(grad[dec_candidates])]  # highest positive gradient
    i_inc = inc_candidates[np.argmin(grad[inc_candidates])]  # most negative gradient
    max_eps = min(h[i_dec], 1 - h[i_inc])
    eps = amount * max_eps * (0.5 + 0.5 * rng.random())  # random fraction
    new_h = h.copy()
    new_h[i_dec] -= eps
    new_h[i_inc] += eps
    return new_h

import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    m = n // 2
    half = project_onto_box_sum(rng.random(m), n / 4.0)
    return np.concatenate([half, half[::-1]])


def random_feasible(n, rng):
    return project_onto_box_sum(rng.random(n), n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


# ---------- targeted worst‑shift improvement ----------
def fix_worst_shift(h, rng, amount=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # shift giving the current max overlap
    # compute sum of neighbours that appear in the gradient of C_k
    sum_shift = np.zeros(n)
    if k >= 0:
        sum_shift[: n - k] += h[k:]
        sum_shift[k:]      += h[: n - k]
    else:
        kk = -k
        sum_shift[kk:]    += h[: n - kk]
        sum_shift[: n - kk] += h[kk:]
    grad = 1.0 - sum_shift                   # dC/dh
    dec = np.where((grad > 1e-12) & (h > 1e-12))[0]
    inc = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
    if dec.size == 0 or inc.size == 0:
        return h
    i_dec = dec[np.argmax(grad[dec])]
    i_inc = inc[np.argmin(grad[inc])]
    max_eps = min(h[i_dec], 1 - h[i_inc])
    eps = amount * max_eps * (0.5 + 0.5 * rng.random())
    out = h.copy()
    out[i_dec] -= eps
    out[i_inc] += eps
    return out


# ---------- hill‑climb ----------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5, fix_prob=0.1):
    h = h_start.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        if rng.random() < fix_prob:
            new_h = fix_worst_shift(h, rng, amount=step)
        else:
            new_h = perturb_h(h, rng, scale=step)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr * (2.0 / x.size)
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- gradient descent on log‑sum‑exp ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)               # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c, best_h = c_new, new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():

        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)

        h0 = (h0 + h0[::-1]) * 0.5          # enforce symmetry

    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 1500.0]

        else:
            h_start = random_feasible(n, rng)

        if rng.random() < 0.12:

                pass

        # final symmetric projection (optional, keeps search space small)
        h_opt = (h_opt + h_opt[::-1]) * 0.5
        h_opt = project_onto_box_sum(h_opt, n / 2.0)

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- up‑sample if time permits ----
    if time.time() - start < budget_s * 0.90:
        n_up = n * 2
        # linear interpolation to finer grid
        xs = np.linspace(0, 2, n, endpoint=False)
        xs_up = np.linspace(0, 2, n_up, endpoint=False)
        up = np.interp(xs_up, xs, best_h)
        up = project_onto_box_sum(up, n_up / 2.0)
        up = (up + up[::-1]) * 0.5
        up = project_onto_box_sum(up, n_up / 2.0)

        # refine on up‑sampled version
        up, c_up = hill_climb(up, rng, max_iters=2200, start_step=0.4, fix_prob=0.2)
        up, c_up = grad_descent_optimize(up, beta_sched, steps_per_beta, rng)

        for beta in (200.0, 500.0):
            try:
                up = slsqp_optimize(up, beta=beta, max_iter=300)
            except Exception:
                pass
        up = (up + up[::-1]) * 0.5
        up = project_onto_box_sum(up, n_up / 2.0)
        c_up = compute_c5(up)
        if c_up < best_c:
            best_h, best_c, n = up, c_up, n_up

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
