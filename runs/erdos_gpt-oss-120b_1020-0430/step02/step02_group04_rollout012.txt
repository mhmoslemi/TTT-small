import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = 0.5 * (lo + hi)
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + 0.5 * (lo + hi), lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

def random_feasible_symmetric(n, rng):
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    half = np.concatenate([x, x[::-1]])
    # if n odd, add middle element
    if n % 2 == 1:
        mid = rng.random(1)
        mid = project_onto_box_sum(mid, target_half - x.sum())
        half = np.concatenate([half[:m], mid, half[m:]])
    return half

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)

    if i == j:
        return h

        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # smooth maximum scaled by beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] and enforce exact sum = target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = 0.5 * (lo + hi)
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + 0.5 * (lo + hi), lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Even‑length symmetric feasible vector (0≤h≤1, Σh=n/2)."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def perturb_h(h, rng, scale=0.2):
    """Swap a small amount of mass between two coordinates (keeps sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- smooth‑max gradient descent ----------
def grad_descent_optimize(h0, betas, steps_per_beta, rng):
    """Deterministic annealing using a smooth max surrogate."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta, steps in zip(betas, steps_per_beta):
        step = 0.5
        for _ in range(steps):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)               # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum

            g_raw = np.ones(n)
            nn = n
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (nn - 1)
                if k >= 0:
                    nk = nn - k
                    g_raw[:nk] -= wk * h[k:]
                    g_raw[k:]   -= wk * h[:nk]
                else:
                    kk = -k
                    nk = nn - kk
                    g_raw[kk:]   -= wk * h[:nk]
                    g_raw[:nk]   -= wk * h[kk:]

            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_onto_box_sum(new_h, target)

            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.001:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 180
        h0 = random_feasible_symmetric(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- annealing schedule ----
    betas = [0.5, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    steps_per_beta = [300, 250, 200, 170, 150, 130, 110, 90, 70, 50]

    # ---- stochastic restarts ----
    while time.time() - start < budget_s * 0.75:
        start_h = random_feasible_symmetric(best_h.size, rng)

        # occasional blending with current best
        if rng.random() < 0.12:
            start_h = (start_h + best_h) * 0.5
            start_h = project_onto_box_sum(start_h, start_h.size / 2.0)
            start_h = (start_h + start_h[::-1]) * 0.5
            start_h = project_onto_box_sum(start_h, start_h.size / 2.0)

        opt_h, opt_c = grad_descent_optimize(start_h, betas, steps_per_beta, rng)

    # ---- optional up‑sampling (if time permits) ----
    if time.time() - start < budget_s * 0.9:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
