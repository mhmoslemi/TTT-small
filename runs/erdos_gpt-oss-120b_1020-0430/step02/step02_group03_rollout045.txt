def compute_c5_fft(h):
    n = h.size
    # compute correlation full
    # using FFT of size L = 2**next_pow2(2*n-1)
    L = 1 << (2*n - 1).bit_length()
    H = np.fft.rfft(h, L)
    inv = np.fft.rfft(np.flip(1 - h), L)
    conv = np.fft.irfft(H * inv, L)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def hill_climb_sa(h, rng, max_iters=2000, start_step=0.5):
    best = h.copy()
    best_c = compute_c5(best)
    cur = h.copy()
    cur_c = best_c
    step = start_step
    T0 = best_c - 0.3 # start temperature maybe based on difference; but we can just set T0 = 0.01
    for it in range(max_iters):
        # generate perturbation
        i = rng.integers(0, cur.size)
        j = rng.integers(0, cur.size)
        if i == j:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() * 2 - 1) * max_eps * step
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        # enforce bounds via clipping? Already ensures within [0,1]
        # sum unchanged.
        new_c = compute_c5(new)
        # accept if better or with probability exp(-(new_c - cur_c)/temp)
        if new_c < cur_c:
            cur = new
            cur_c = new_c
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
        else:
            # temperature schedule
            t = max(1e-6, T0 * (1 - it / max_iters))
            prob = np.exp(-(new_c - cur_c) / t) if t > 0 else 0.0
            if rng.random() < prob:
                cur = new
                cur_c = new_c
        # schedule step decay
        if (it + 1) % (max_iters // 10) == 0:
            step = max(step * 0.7, 1e-4)
    return best, best_c

    n = h0.size
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    cons = {'type':'eq','fun': lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0,1)
    # correct sum if drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:

        for _ in range(2000):
            i = rng.integers(0, n)

            x[i] += add
            diff = target - x.sum()

    return x

import time
import numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng):
    """Project h onto [0,1] box and adjust to meet sum(target)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # simple iterative redistribution
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Create a random feasible h (0≤h≤1, sum=n/2)."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer ε from i to j, preserving the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def correlation_full(h):
    """Full correlation of h with (1‑h) using FFT (length 2·n‑1)."""
    n = h.shape[0]
    L = 1 << (2 * n - 1).bit_length()          # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(np.flip(1.0 - h), n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return conv.real

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    return np.max(correlation_full(h)) * (2.0 / n)

def softmax_obj(x, beta):
    """Smooth approximation of max using log‑sum‑exp."""
    corr = correlation_full(x)
    m = np.max(corr)
    # stable log‑sum‑exp
    exp_term = np.exp(beta * (corr - m))
    return m + np.log(np.sum(exp_term)) / beta

def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of h with smooth surrogate for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        lambda x: softmax_obj(x, beta),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=eq_cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def hill_climb_sa(h_start, rng, max_iters=2000, start_step=0.5):
    """Simulated‑annealing hill climb (keeps best solution)."""
    cur = h_start.copy()
    cur_c = compute_c5(cur)
    best_h = cur.copy()
    best_c = cur_c
    step = start_step
    T0 = max(1e-4, cur_c * 0.02)   # initial temperature
    for it in range(max_iters):
        new = perturb_h(cur, rng, scale=step)
        new_c = compute_c5(new)
        if new_c < cur_c:
            cur, cur_c = new, new_c
            if new_c < best_c:
                best_h, best_c = new.copy(), new_c
        else:
            # temperature schedule linear
            t = T0 * (1.0 - it / max_iters)
            if t > 0 and rng.random() < np.exp(-(new_c - cur_c) / t):
                cur, cur_c = new, new_c
        # decay step size
        if (it + 1) % (max_iters // 10) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def symmetrize(h, rng):
    """Replace h by (h + flip(1‑h))/2 and re‑enforce the sum."""
    h2 = (h + np.flip(1.0 - h)) / 2.0
    return enforce_sum(h2, h2.size / 2.0, rng)
