def _subgradient_optimize(h0, time_limit):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_from_h(h)
    step0 = 1.0  # initial step size maybe large; we'll tune.

    it = 0
    no_improve = 0
    max_no_improve = 5000
    while time.time() < time_limit:
        # compute correlation and max index
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        # compute gradient
        if k >= 0:
            forward = np.concatenate([h[k:], np.zeros(k, dtype=float)])
            backward = np.concatenate([np.zeros(k, dtype=float), h[:n - k]])
        else:
            k_neg = -k
            forward = np.concatenate([np.zeros(k_neg, dtype=float), h[:n - k_neg]])
            backward = np.concatenate([h[k_neg:], np.zeros(k_neg, dtype=float)])
        g = 1.0 - (forward + backward)
        # step size diminishing
        alpha = step0 / np.sqrt(it + 1)
        improved = False
        for factor in (1.0, 0.5, 0.25):
            cand = _project_to_feasible(h - alpha * factor * g, n)
            cand_c = _c5_from_h(cand)
            if cand_c < best_c - 1e-12:
                # Accept improvement over best, also update current h
                h = cand
                best_h = cand.copy()
                best_c = cand_c
                improved = True
                no_improve = 0
                break
            # Also accept if improvement over current c (maybe not best)
            cur_c = _c5_from_h(h)
            if cand_c < cur_c - 1e-12:

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _project_to_feasible(v: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip values to [0,1] and enforce sum(v)=n/2 by water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(v > 0)[0]
        else:
            free = np.where(v < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and v[i] < 1 - 1e-12:
                add = min(residual, 1 - v[i])
                v[i] += add
                residual -= add
            elif residual < 0 and v[i] > 1e-12:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(v, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a given feasible h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _objective_func(x):
    return x[-1]


def _ineq_constraints_func(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraints_func(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 200):
    """Local refinement using SLSQP."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraints_func},
        {"type": "eq", "fun": _eq_constraints_func},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective_func, x0, method="SLSQP",
                            bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float):
    """Small pairwise exchanges, greedy acceptance."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    steps = 0
    while steps < max_steps and time.time() < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        eps_max = min(max_down, max_up)
        if eps_max <= 0:

        eps = np.random.rand() * eps_max * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
        steps += 1
    return best_h, best_c


def _subgradient_optimize(h0: np.ndarray, time_limit: float):
    """Projected subgradient descent on the active shift."""
    n = h0.size
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    step0 = 0.8
    it = 0
    stagnation = 0
    while time.time() < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        cur_c = corr[max_idx] * (2.0 / n)

        if cur_c < best_c - 1e-12:
            best_h, best_c = h.copy(), cur_c
            stagnation = 0
        else:
            stagnation += 1

        # gradient of the active correlation
        if k >= 0:
            forward = np.concatenate([h[k:], np.zeros(k, dtype=float)])
            backward = np.concatenate([np.zeros(k, dtype=float), h[:n - k]])
        else:
            kn = -k
            forward = np.concatenate([np.zeros(kn, dtype=float), h[:n - kn]])
            backward = np.concatenate([h[kn:], np.zeros(kn, dtype=float)])
        g = 1.0 - (forward + backward)

        alpha = step0 / np.sqrt(it + 1)
        improved = False
        for factor in (1.0, 0.5, 0.25):
            cand = _project_to_feasible(h - alpha * factor * g, n)
            cand_c = _c5_from_h(cand)
            if cand_c < cur_c - 1e-12:
                h = cand
                improved = True
                break
        if not improved:
            if stagnation % 200 == 0:
                noise = np.random.randn(n) * 0.05
                h = _project_to_feasible(h + noise, n)
        it += 1
        if it > 200000:
            break
    return best_h, best_c
