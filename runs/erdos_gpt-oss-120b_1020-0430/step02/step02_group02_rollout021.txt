import time
import numpy as np
from scipy import optimize

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Small epsilon exchange between two coordinates, accepting only improvements."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Smooth‑max objective (gradient available)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(x: np.ndarray, alpha: float, lam: float):
    """Smooth approximation of max C5 with gradient."""
    n = x.size
    total = x.sum()
    conv = np.correlate(x, x, mode="full")          # h * h
    c = total - conv                               # vector of length 2n‑1
    a_c = alpha * c
    # stabilize exp
    max_ac = a_c.max()
    exp_shifted = np.exp(a_c - max_ac)
    sum_exp_shifted = exp_shifted.sum()
    p = exp_shifted / sum_exp_shifted               # softmax weights, sums to 1

    # gradient of the smooth term
    grad_base = np.ones_like(x)                     # derivative of total part
    half = n - 1
    for idx, pk in enumerate(p):
        if pk == 0.0:
            continue
        k = idx - half
        if k >= 0:
            grad_base[: n - k] -= pk * x[k:]
            grad_base[k:]      -= pk * x[: n - k]
        else:
            kk = -k
            grad_base[kk:]    -= pk * x[: n - kk]
            grad_base[: n - kk] -= pk * x[kk:]

    diff = total - (n / 2.0)                         # equality‑penalty term
    dx = 2.0 / n
    f = (max_ac + np.log(sum_exp_shifted)) / alpha * dx + lam * diff * diff
    grad = dx * grad_base + 2.0 * lam * diff
    return f, grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Load any provided initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # ------------------------------------------------------------------
    # Helper to evaluate C5 quickly (same as _c5_from_h)
    # ------------------------------------------------------------------
    def eval_c5(arr):
        return _c5_from_h(arr)

    # ------------------------------------------------------------------
    # Binary simulated annealing on the original grid (if present)
    # ------------------------------------------------------------------
    best_h, best_c5, best_n = None, np.inf, None
    if init_h is not None and init_h.size % 2 == 0:
        n0 = init_h.size
        # build binary start from init_h (threshold)
        binary = (init_h >= 0.5).astype(float)
        # adjust to exact half‑ones
        cur_ones = int(binary.sum())
        target_ones = n0 // 2
        while cur_ones > target_ones:
            i = np.random.choice(np.flatnonzero(binary == 1))
            binary[i] = 0
            cur_ones -= 1
        while cur_ones < target_ones:
            i = np.random.choice(np.flatnonzero(binary == 0))
            binary[i] = 1
            cur_ones += 1

        # annealing parameters
        max_iter = 20000
        temp0 = 0.05 * eval_c5(binary)
        final_temp = 1e-5
        cooling = (final_temp / max(temp0, 1e-12)) ** (1.0 / max_iter)

        cur = binary.copy()
        cur_c = eval_c5(cur)
        best_local_h, best_local_c = cur.copy(), cur_c
        temp = temp0
        for _ in range(max_iter):
            if time.time() - start > budget_s * 0.2:
                break
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            cur[i], cur[j] = 0.0, 1.0
            new_c = eval_c5(cur)
            delta = new_c - cur_c
            if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-12)):
                cur_c = new_c
                if cur_c < best_local_c:
                    best_local_c, best_local_h = cur_c, cur.copy()
            else:
                cur[i], cur[j] = 1.0, 0.0
            temp *= cooling

        best_h, best_c5, best_n = best_local_h, best_local_c, n0

    # ------------------------------------------------------------------
    # Candidate grids (including up‑scaled versions)
    # ------------------------------------------------------------------
    candidates = set()
    if init_h is not None:
        candidates.add(init_h.size)
    for mul in (1, 2, 3, 4):
        if init_h is not None:
            n = init_h.size * mul
        else:
            n = 80 * mul
        if n % 2 == 0 and n <= 512:
            candidates.add(n)
    # extra hand‑picked even sizes
    candidates.update([96, 128, 160, 192, 224, 256, 320, 384, 512])
    cand_list = sorted(candidates)

    # ------------------------------------------------------------------
    # Core continuous optimisation (smooth‑max + hill‑climb)
    # ------------------------------------------------------------------
    for n in cand_list:
        if time_left() < 5.0:
            break

        # build starting vector for this size
        if best_h is not None and best_n == n:
            start_h = best_h.copy()
        elif best_h is not None:
            start_h = _upscale_h(best_h, n)
        elif init_h is not None and init_h.size == n:
            start_h = init_h.copy()
        else:
            start_h = np.full(n, 0.5)

        start_h = _project_to_feasible(start_h, n)

        # smooth‑max optimisation with a few restarts / alphas
        alphas = [5.0, 15.0, 30.0]
        lam = 10.0
        cur_h, cur_c = start_h, eval_c5(start_h)

        for alpha in alphas:
            if time_left() < 5.0:
                break
            for _ in range(3):          # three restarts per alpha
                if time_left() < 5.0:
                    break
                pert = np.random.uniform(-0.2, 0.2, size=n)
                h0 = cur_h + pert
                h0 = _project_to_feasible(h0, n)

                def fun_and_grad(x):
                    return _smooth_obj_and_grad(x, alpha, lam)

                res = optimize.minimize(
                    fun_and_grad,
                    h0,
                    method="L-BFGS-B",
                    jac=True,
                    bounds=[(0.0, 1.0)] * n,
                    options={"maxiter": 300, "ftol": 1e-12},
                )
                h_opt = _project_to_feasible(res.x, n)
                c_opt = eval_c5(h_opt)
                if c_opt < cur_c:
                    cur_h, cur_c = h_opt, c_opt

        # final local refinement (small swaps)
        deadline = start + budget_s - 0.5
        cur_h, cur_c = _hill_climb(cur_h, max_steps=3000, time_limit=deadline)

        if cur_c < best_c5:
            best_h, best_c5, best_n = cur_h, cur_c, n

        # early exit if we beat the target record
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = eval_c5(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
