def _gradient_max_corr(h):
    corr = np.correlate(h, 1-h, mode='full')
    max_val = corr.max()
    # indices of max (with tolerance)
    eps = 1e-12
    idxs = np.where(np.abs(corr - max_val) <= eps)[0]
    shifts = idxs - (len(h)-1)
    # compute gradient as average over these shifts
    grad = np.zeros_like(h)
    for k in shifts:
        idx = np.arange(len(h))
        idx_minus = idx - k
        idx_plus = idx + k
        term1 = np.where((idx_minus >= 0) & (idx_minus < len(h)), 1 - h[idx_minus], 1.0)
        term2 = np.where((idx_plus >= 0) & (idx_plus < len(h)), h[idx_plus], 0.0)
        grad += term1 - term2
    grad /= len(shifts)
    return grad

import time
import numpy as np

# -------------------- Feasibility projection -------------------- #
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# -------------------- Evaluation -------------------- #
def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a discrete step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# -------------------- Upscaling -------------------- #
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    nc = coarse_h.size
    x_coarse = (np.arange(nc) + 0.5) * (2.0 / nc)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# -------------------- Gradient of max correlation -------------------- #
def _gradient_max_corr(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal correlation term."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = corr.max()
    eps = 1e-12
    idxs = np.where(np.abs(corr - max_val) <= eps)[0]
    shifts = idxs - (n - 1)               # lag = index - (n-1)
    idx = np.arange(n)
    grad = np.zeros(n)
    for k in shifts:
        im = idx - k
        ip = idx + k
        term1 = np.where((im >= 0) & (im < n), 1 - h[im], 1.0)
        term2 = np.where((ip >= 0) & (ip < n), h[ip], 0.0)
        grad += term1 - term2
    grad /= len(shifts)
    return grad


# -------------------- Local refinement -------------------- #
def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02):
    """Small stochastic hill‑climb around h."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_c5, best_h = c5, trial.copy()
    return best_h, best_c5


# -------------------- Projected gradient optimiser -------------------- #
def _optimize_h(initial_h: np.ndarray, time_budget: float):
    """Projected sub‑gradient descent with simple simulated‑annealing."""
    n = initial_h.size
    h = _project_to_feasible(initial_h.copy(), n)
    cur_c5 = _c5_from_h(h)
    best_h, best_c5 = h.copy(), cur_c5

    start = time.time()
    stop = start + time_budget

    step = 0.5
    decay = 0.9998
    min_step = 1e-5
    it = 0

    while time.time() < stop:
        # gradient based on current maximal shift(s)
        grad = _gradient_max_corr(h)
        norm = np.linalg.norm(grad)
        if norm == 0:
            # random restart
            h = _project_to_feasible(h + 0.1 * np.random.randn(n), n)
            cur_c5 = _c5_from_h(h)
            continue
        direction = grad / norm

        # propose move
        h_new = h - step * direction
        h_new = _project_to_feasible(h_new, n)
        new_c5 = _c5_from_h(h_new)

        # accept if better, else probabilistic (simulated annealing)
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (step + 1e-12)):
            h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, h_new.copy()

        step = max(min_step, step * decay)
        it += 1

        # occasional jitter to escape plateaux
        if it % 5000 == 0:
            h = _project_to_feasible(h + 0.05 * np.random.randn(n), n)
            cur_c5 = _c5_from_h(h)

    return best_h, best_c5
