def perturb(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:

        return h.copy()

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def enforce_sum(h, target, rng, max_iter=10000):
    """Clip to [0,1] and adjust to achieve exact sum=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb(h, rng, scale=0.2):
    """Transfer a small amount ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def anneal_h(h_start, rng, time_limit):
    """Simple simulated annealing on C5."""
    h = h_start.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    temp = 1.0
    scale = 0.3
    n = h.size
    while time.time() < time_limit:
        cand = perturb(h, rng, scale=scale)
        cand_c = compute_c5(cand)
        delta = c - cand_c  # positive if improvement
        if delta > 0 or rng.random() < np.exp(delta / temp):
            h, c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
        temp *= 0.9999
        if rng.random() < 0.001:
            scale = max(scale * 0.9, 1e-5)
        # occasional big restart
        if rng.random() < 0.001:

            c = compute_c5(h)
    return best_h, best_c

def targeted_improvement(h, rng, max_iter=5000, time_limit=None):
    """Move mass from low‑neighbor to high‑neighbor for the worst shift."""
    if time_limit is None:
        time_limit = time.time() + 1.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        ac = np.correlate(best_h, best_h, mode="full")
        min_idx = np.argmin(ac)
        shift = min_idx - (n - 1)
        # neighbor sum for this shift
        neigh = np.zeros(n)
        if shift > 0:
            neigh[: n - shift] += best_h[shift:]
            neigh[shift:] += best_h[: n - shift]
        elif shift < 0:
            s = -shift
            neigh[s:] += best_h[: n - s]

        else:

        a_idx = np.where(mask_a)[0][np.argmin(neigh[mask_a])]

# ---------- smooth surrogate & SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp in a stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP on smooth surrogate with equality sum constraint."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def slsqp_refine(h0, beta_seq, time_limit):
    """Apply a ladder of β values, respecting the remaining time."""
    h = h0.copy()
    best_c = compute_c5(h)
