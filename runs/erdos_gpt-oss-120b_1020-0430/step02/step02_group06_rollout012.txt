def _project_sum_box(x, target):
    # Ensure 0<=x<=1, sum==target
    x = np.clip(x,0,1)
    # Simple iterative scaling
    # While sum not approx target:
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Identify free positions:
        if diff > 0:
            free = (x < 1)
        else:
            free = (x > 0)
        if not free.any():
            break
        delta = diff / free.sum()
        x[free] += delta
        # Clip again
        x = np.clip(x,0,1)
    # final correction:
    residual = target - x.sum()
    if abs(residual) > 1e-9:
        # distribute residual to first free positions
        indices = np.where((residual>0) & (x < 1) | (residual<0) & (x > 0))[0]
        for i in indices:

                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
            else:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return x

def _c5_raw(h):
    return np.max(np.correlate(h, 1 - h, mode='full'))

def _c5_from_h(h):
    n = h.shape[0]
    return _c5_raw(h) * (2.0 / n)

def _project_to_feasible(x, n, target=None):
    if target is None:
        target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    # quick fix
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0
        else:
            free = x > 0.0
        if not np.any(free):
            break
        delta = diff / free.sum()
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction if needed
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # distribute greedily
        for i in range(n):
            if residual > 0 and x[i] < 1.0:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 0.0:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return x

import time
import numpy as np

# -------------------- Core utilities -------------------- #

def _c5_raw(h: np.ndarray) -> float:
    """Maximum (unscaled) overlap integral for a discretised h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def _c5_from_h(h: np.ndarray) -> float:
    """Scaled C5 value."""
    n = h.shape[0]
    return _c5_raw(h) * (2.0 / n)


def _project_to_feasible(x: np.ndarray, n: int, target: float = None) -> np.ndarray:
    """Project onto [0,1]^n with sum(x)=target (default n/2)."""
    if target is None:
        target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    # simple water‑filling, a few passes are enough
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0
        else:
            free = x > 0.0
        if not free.any():
            break
        delta = diff / free.sum()
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction (rare)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and x[i] < 1.0:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 0.0:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _gradient_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Subgradient of the correlation at integer shift s."""
    n = h.shape[0]
    if s >= 0:
        left = np.concatenate((np.zeros(s), h[: n - s]))
        right = np.concatenate((h[s:], np.zeros(s)))
    else:
        s = -s
        left = np.concatenate((h[s:], np.zeros(s)))
        right = np.concatenate((np.zeros(s), h[: n - s]))
    return (1.0 - left) - right


def _subgradient_opt(h0: np.ndarray, steps: int, step0: float) -> np.ndarray:
    """Projected subgradient descent on max_{k} Σ h_i(1−h_{i−k})."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    for t in range(1, steps + 1):
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        idxs = np.where(corr == max_val)[0]
        shifts = idxs - (n - 1)               # integer shift values
        g = np.zeros_like(h)
        for s in shifts:
            g += _gradient_for_shift(h, int(s))
        g /= len(shifts)
        alpha = step0 / np.sqrt(t)
        h -= alpha * g
        h = _project_to_feasible(h, n, target)
    return h


def _local_refine(h: np.ndarray, attempts: int = 15, delta: float = 0.015) -> np.ndarray:
    """Tiny stochastic hill‑climbing around h."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse)


def _binary_hill_climb(bin_h: np.ndarray, max_iter: int, target_bound: float) -> tuple:
    """Swap‑based local search on binary vectors."""
    n = bin_h.size
    best = bin_h.copy()
    best_c5 = _c5_from_h(best)
    for _ in range(max_iter):
        if best_c5 < target_bound:
            break
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        c5 = _c5_from_h(cand)
        if c5 < best_c5:
            best, best_c5 = cand, c5
    return best, best_c5

    start_time = time.time()

    # parent construction, if supplied
