def hill_climb_binary(h, rng, max_iters=5000):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        # choose a 1 and a 0
        ones = np.where(best_h > 0.5)[0] # boolean for 1's
        zeros = np.where(best_h < 0.5)[0] # for 0's
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the soft‑max surrogate for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with Σ = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def binary_from_continuous(h):
    """Turn a continuous vector into a binary one with the same L1‑mass."""
    n = h.size
    half = n // 2
    order = np.argsort(-h)  # descending
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[order[:half]] = 1.0
    return bin_vec


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000,
                        init_temp=0.02, decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c


def simulated_annealing_binary(h_start, rng, max_iters=20000,
                               init_temp=0.05, decay=0.9995):
    """Binary simulated annealing – swaps a 1 and a 0."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    n = h.size
    for it in range(max_iters):
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # fix any tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target = (n * factor) / 2.0
    return enforce_sum(h_up, target, rng)


def refine_continuous(h, rng):
    """Intensive local refinement for a continuous vector."""
    # a ladder of soft‑max β values
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
        try:
            h = slsqp_optimize(h, beta=beta, max_iter=300)
        except Exception:
            pass
    # deterministic hill climb
    h, _ = hill_climb(h, rng, max_iters=4000, start_step=0.5)
    # long simulated annealing
    h, _ = simulated_annealing(
        h,
        rng,
        max_iters=15000,
        init_temp=0.08,
        decay=0.9997,
        scale=0.3,
    )
    return h


def binary_refine(h_bin, rng, time_limit):
    """Binary optimisation with hill‑climb and annealing, respecting time limit."""
    best_h, best_c = h_bin.copy(), compute_c5(h_bin)
    # deterministic hill climb (swaps only)
    for _ in range(3):
        h, c = hill_climb_binary(best_h, rng, max_iters=5000)
        if c < best_c:
            best_h, best_c = h, c
        # binary simulated annealing
        h, c = simulated_annealing_binary(best_h, rng, max_iters=20000)
        if c < best_c:
            best_h, best_c = h, c
        if time.time() > time_limit:
            break
    return best_h, best_c


def hill_climb_binary(h_start, rng, max_iters=5000):
    """Hill climbing on binary vectors (swap a 1 with a 0)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
    return best_h, best_c
