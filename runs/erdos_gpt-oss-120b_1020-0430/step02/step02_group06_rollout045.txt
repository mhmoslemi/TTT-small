def _project_to_feasible(h,n, tol=1e-9, max_iter=100):
    h = np.clip(h,0.,1.)
    target = n/2.
    # compute sum, diff
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # compute free indices where we can adjust diff: if diff>0 need decrease values >0; if diff<0 need increase values <1
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        # compute delta adjustment per free variable
        delta = diff / free.size
        h[free] -= delta
        # clip again
        h = np.clip(h,0.,1.)
    # final residual adjust: adjust any variable not saturated
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # allocate to first free indices
        if residual > 0:
            # increase up to 1
            inc_idx = np.where(h < 1 - 1e-12)[0]
            for i in inc_idx:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
                if residual <= 1e-12:
                    break
        else:
            dec_idx = np.where(h > 1e-12)[0]
            for i in dec_idx:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if residual >= -1e-12:
                    break
    return np.clip(h,0.,1.)

import time
import numpy as np
import scipy.optimize as opt

# ---------- Feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            inc = np.where(h < 1 - 1e-12)[0]
            for i in inc:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
                if residual <= 1e-12:
                    break
        else:
            dec = np.where(h > 1e-12)[0]
            for i in dec:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if residual >= -1e-12:
                    break
    return np.clip(h, 0.0, 1.0)


# ---------- C5 evaluation ----------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- Local refinement ----------
def _local_refine(h: np.ndarray, attempts: int = 20, delta: float = 0.02) -> np.ndarray:
    n = h.size
    best = h.copy()
    best_val = _c5_from_h(best)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        val = _c5_from_h(trial)
        if val < best_val:
            best, best_val = trial, val
    return best


# ---------- Upscaling ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse_h)


# ---------- Genetic operators ----------
def _crossover(p1: np.ndarray, p2: np.ndarray, prob: float = 0.7) -> tuple:
    n = p1.size
    if np.random.rand() > prob:
        return p1.copy(), p2.copy()
    i, j = sorted(np.random.choice(range(1, n), size=2, replace=False))
    c1 = np.concatenate([p1[:i], p2[i:j], p1[j:]])
    c2 = np.concatenate([p2[:i], p1[i:j], p2[j:]])
    return c1, c2


def _mutate(child: np.ndarray, prob: float = 0.3, scale: float = 0.05) -> np.ndarray:
    n = child.size
    mask = np.random.rand(n) < prob
    child = child.copy()
    child[mask] += scale * np.random.randn(mask.sum())
    return child


# ---------- SLSQP refinement ----------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": _ineq_constraint},
            {"type": "eq", "fun": _eq_constraint}]
    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds,

    h_opt = res.x[:n] if res.success and res.x is not None else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

    n_candidates = []
    if parent_h is not None:
        n_candidates.append(parent_h.size)
    for mul in (1, 2, 3, 4):
        if parent_h is not None:
            n = int(parent_h.size * mul)
        else:
            n = 64 * mul
        if n % 2 == 0 and n <= 256:
            n_candidates.append(n)
    n_candidates.extend([80, 96, 112, 128, 144, 160, 192, 224, 256])
    n_candidates = sorted(set([n for n in n_candidates if n % 2 == 0 and n <= 256]))

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Base vector of size n
        if parent_h is not None:
            base_h = _upscale_h(parent_h, n) if parent_h.size != n else parent_h.copy()
        else:
            base_h = np.full(n, 0.5)

        # Population initialization
        pop_size = 20
        population = []
        population.append(_project_to_feasible(base_h, n))
        while len(population) < pop_size:
            trial = base_h + 0.2 * np.random.randn(n)
            population.append(_project_to_feasible(trial, n))

        fitness = np.array([_c5_from_h(ind) for ind in population])
        generation = 0
        no_improve = 0

        while time_left() > 2.0 and generation < 2000:
            generation += 1
            # Elitism: keep best 2
            elite_idx = np.argsort(fitness)[:2]
            new_pop = [population[i].copy() for i in elite_idx]

            # Fill the rest via crossover/mutation
            while len(new_pop) < pop_size:
                # tournament selection size 3
                def tournament():
                    cand = np.random.choice(pop_size, size=3, replace=False)
                    return population[cand[np.argmin(fitness[cand])]]

                p1 = tournament()
                p2 = tournament()
                c1, c2 = _crossover(p1, p2, prob=0.7)
                c1 = _mutate(c1, prob=0.3, scale=0.05)
                c2 = _mutate(c2, prob=0.3, scale=0.05)
                c1 = _project_to_feasible(c1, n)
                c2 = _project_to_feasible(c2, n)
                new_pop.append(c1)
                if len(new_pop) < pop_size:
                    new_pop.append(c2)

            new_fitness = np.array([_c5_from_h(ind) for ind in new_pop])

            # Update global best
            cur_best_idx = np.argmin(new_fitness)
            cur_best_c5 = new_fitness[cur_best_idx]
            if cur_best_c5 + 1e-12 < best_c5:
                best_c5 = cur_best_c5
                best_h = new_pop[cur_best_idx].copy()
                best_n = n
                no_improve = 0
                if best_c5 < target:
                    break
            else:
                no_improve += 1

            population = new_pop
            fitness = new_fitness

            # Periodic local refinement on the current global best
            if generation % 20 == 0:
                refined = _local_refine(best_h, attempts=30, delta=0.02)
                refined_c5 = _c5_from_h(refined)
                if refined_c5 < best_c5:
                    best_h, best_c5 = refined, refined_c5
                    best_n = n
                    if best_c5 < target:
                        break

            # Early stop if stagnation
            if no_improve >= 50:
                break

        # After GA phase try a short SLSQP refinement if time permits
        if best_h is not None and best_h.size == n and time_left() > 5.0:
            h_opt, c5_opt = _solve_h(best_h, max_iter=300)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
                best_n = n
                if best_c5 < target:
                    break

        if best_c5 < target:
            break

    # Fallback
    if best_h is None:
        n = 64
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
