def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
```

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing(h0, rng, max_iters=3000, init_temp=1e-3, final_temp=1e-6, scale=0.3):
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    T = init_temp
    for i in range(max_iters):
        new_h = perturb_h(h_cur, rng, scale=scale)
        c_new = compute_c5(new_h)
        delta = c_new - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h_cur = new_h
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
        # geometric cooling
        T = T * (final_temp / init_temp) ** (1.0 / max_iters)
    return best_h, best_c

def greedy_extremize(h, rng):
    h = h.copy()
    n = h.size
    target = n/2
    improved = True
    while improved:
        improved = False
        # random order of indices
        order = rng.permutation(n)
        for i in order:
            if h[i] <= 0 or h[i] >= 1:
                continue
            # try move to 0
            diff = -h[i]  # amount removed
            # need to add -diff to some other index j where h[j] < 1
            # find such j
            candidates = np.where(h < 1)[0]
            if len(candidates) == 0:
                continue
            # choose j that yields best improvement
            best_local_c = None
            best_j = None
            # To keep runtime low we can test a few random js
            for _ in range(5):
                j = rng.choice(candidates)
                if j == i:
                    continue
                delta = -diff # amount to add to j
                if h[j] + delta > 1 or h[j] + delta < 0:
                    continue
                h_new = h.copy()
                h_new[i] = 0.0
                h_new[j] += delta
                c_new = compute_c5(h_new)
                if best_local_c is None or c_new < best_local_c:
                    best_local_c = c_new
                    best_j = j
            if best_j is not None and best_local_c < compute_c5(h):
                h[i] = 0.0
                h[best_j] += -diff
                improved = True
                break
            # try move to 1 similarly
            diff = 1 - h[i] # amount added
            # need to remove diff from some other index j with h[j] > 0
            candidates = np.where(h > 0)[0]
            for _ in range(5):
                j = rng.choice(candidates)
                if j == i:
                    continue
                delta = diff
                if h[j] - delta < 0:
                    continue
                h_new = h.copy()
                h_new[i] = 1.0
                h_new[j] -= delta
                c_new = compute_c5(h_new)
                if best_local_c is None or c_new < best_local_c:
                    best_local_c = c_new
                    best_j = j
            if best_j is not None and best_local_c < compute_c5(h):

import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp in a numerically stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    """Adjust h so that Σ h = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    # start with uniform values then project to the simplex with bounds
    h = rng.random(n)
    # scale to desired sum
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    # clip and fix any residual using the repair loop
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def random_binary_vector(n, rng):
    """Random 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)[: n // 2]
    h[idx] = 1.0
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing(h0, rng, max_iters=3000,
                        init_temp=5e-4, final_temp=1e-7, scale=0.3):
    """Simple SA that keeps the sum constraint (by construction)."""
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    T = init_temp
    # geometric cooling factor
    factor = (final_temp / init_temp) ** (1.0 / max_iters)
    for _ in range(max_iters):
        new_h = perturb_h(h_cur, rng, scale=scale)
        c_new = compute_c5(new_h)
        delta = c_new - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h_cur = new_h
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
        T *= factor
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=500):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

    # linear ramps
    candidates.append(enforce_sum(np.clip(xs / 2.0, 0.0, 1.0), n / 2.0, rng))
