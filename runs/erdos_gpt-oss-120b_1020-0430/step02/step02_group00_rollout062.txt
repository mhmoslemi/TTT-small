def _simulated_annealing_binary(initial_h, max_iter, target_c5):
    n = initial_h.size
    h = initial_h.copy()
    N = n//2
    max_corr, c5 = _c5_from_binary(h)
    best_h = h.copy()
    best_c5 = c5
    best_corr = max_corr

    i1_positions = np.flatnonzero(h == 1)
    i0_positions = np.flatnonzero(h == 0)
    T0 = 1.0 * (max_corr - target_corr) if target_corr else 5.0
    if T0 <= 0: T0 = 1.0
    T = T0
    for it in range(max_iter):
        # temperature schedule
        T = T0 * (0.99995 ** it)
        # random swap
        idx1 = np.random.choice(i1_positions)
        idx0 = np.random.choice(i0_positions)
        # perform swap
        h[idx1] = 0
        h[idx0] = 1

        # update i1,i0 sets (swap indices)
        # after swap, idx1 becomes zero index, idx0 becomes one
        # we can update arrays easily: replace in i1_positions and i0_positions
        # but updating each iteration may be heavy; we can just recompute after each accepted move
        # let's just recompute after each iteration for simplicity using np.flatnonzero
        # that'll be O(n) each iteration, but n=256; 200k * 256 ~ 51M operations; may be okay.

        max_corr = np.max(np.correlate(h, 1 - h, mode='full'))
        c5 = max_corr / (n/2)

        delta = c5 - best_c5
        if c5 < best_c5:
            best_c5 = c5
            best_corr = max_corr
            best_h = h.copy()
            # check early stop
            if best_c5 < target_c5:
                break
        # Accept condition
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            # keep new h
            i1_positions = np.flatnonzero(h == 1)
            i0_positions = np.flatnonzero(h == 0)
            continue
        else:
            # revert swap
            h[idx1] = 1
            h[idx0] = 0
            # No need to update positions because they unchanged
            # i1_positions and i0_positions remain same
            continue

    return best_h, best_c5

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 via water-filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_h_slsqp(start_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Run SLSQP starting from start_h (may be fractional)."""
    n = start_h.size
    h0 = _project_to_feasible(start_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _max_corr_and_shift(h: np.ndarray) -> tuple:
    """Return (max_corr, shift) where shift ∈ [-(n-1), n-1] giving the max."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    max_corr = corr[idx]
    shift = idx - (n - 1)          # because full length = 2n-1
    return int(max_corr), shift


def _targeted_reduce(h: np.ndarray, target_corr: int, deadline: float) -> tuple:
    """Try to lower max correlation by swapping mismatched pairs on the worst shift."""
    n = h.size
    max_corr, shift = _max_corr_and_shift(h)

    while max_corr > target_corr and time.time() < deadline:
        k = shift
        # identify positions that contribute to the current worst shift
        if k >= 0:
            cond_i = (h[:n - k] == 1) & (h[k:] == 0)
            cand_i = np.where(cond_i)[0]
            cond_j = (h[:n - k] == 0) & (h[k:] == 1)
            cand_j = np.where(cond_j)[0]
        else:
            s = -k
            cond_i = (h[s:] == 1) & (h[:n - s] == 0)
            cand_i = np.where(cond_i)[0] + s
            cond_j = (h[s:] == 0) & (h[:n - s] == 1)
            cand_j = np.where(cond_j)[0] + s

        if cand_i.size == 0 or cand_j.size == 0:
            break

        improved = False
        # try a few random swaps before giving up
        for _ in range(10):
            i = np.random.choice(cand_i)
            j = np.random.choice(cand_j)
            h[i], h[j] = h[j], h[i]
            new_corr, _ = _max_corr_and_shift(h)
            if new_corr < max_corr:
                max_corr = new_corr
                shift = _max_corr_and_shift(h)[1]
                improved = True
                break
            h[i], h[j] = h[j], h[i]          # revert

        if not improved:
            break
    return h, max_corr


def _simulated_annealing_binary(start_h: np.ndarray, max_iter: int,
                                target_c5: float = None, deadline: float = None) -> tuple:
    """Simple SA on a binary vector (half ones). Returns best vector found."""
    n = start_h.size
    h = start_h.copy()
    max_corr, _ = _max_corr_and_shift(h)
    best_c5 = max_corr / (n / 2)
    best_h = h.copy()
    best_corr = max_corr

    # temperature schedule
    T0 = max_corr * 0.05 + 0.5

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # pick two opposite entries
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        cur_corr, _ = _max_corr_and_shift(h)
        cur_c5 = cur_corr / (n / 2)

        delta = cur_c5 - best_c5
        T = T0 * (0.99995 ** it)
        accept = False
        if cur_c5 < best_c5:
            accept = True
        else:
            if T > 1e-12 and np.random.rand() < np.exp(-delta / T):
                accept = True

        if accept:
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_corr = cur_corr
                if target_c5 is not None and best_c5 < target_c5:
                    break
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    return best_h, best_c5


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # parameters
    n = 256                     # base dimension (even)
    target_c5 = 0.3808
    target_corr = int(target_c5 * (n / 2))   # integer threshold for binary case

    # ------------------------------------------------------------------
    # use provided construction if any
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    try:
        init_h = np.array(initial_h_values, dtype=float)
        init_h = _project_to_feasible(init_h, n)
        init_c5 = _c5_from_h(init_h)
        best_h, best_c5 = init_h, init_c5
    except NameError:
        init_h = None

    # ------------------------------------------------------------------
    # deterministic “irrational rotation” candidates
    # ------------------------------------------------------------------
    alphas = [
        (np.sqrt(5) - 1) / 2,          # golden ratio conjugate
        np.sqrt(2) - 1,
        np.pi - 3,
        np.e - 2,
    ] + list(np.random.rand(6))
    for alpha in alphas:
        if time_left() < 2.0:
            break
        frac = (np.arange(n) * alpha) % 1.0
        h_bin = (frac < 0.5).astype(float)
        # enforce exact half ones
        cur = int(h_bin.sum())
        need = n // 2
        if cur > need:
            idx = np.where(h_bin == 1)[0]
            flip = np.random.choice(idx, cur - need, replace=False)
            h_bin[flip] = 0.0
        elif cur < need:
            idx = np.where(h_bin == 0)[0]
            flip = np.random.choice(idx, need - cur, replace=False)
            h_bin[flip] = 1.0
        c5 = _c5_from_h(h_bin)
        if c5 < best_c5:
            best_h, best_c5 = h_bin.copy(), c5

    # ------------------------------------------------------------------
    # main stochastic search (SA + targeted reduction + SLSQP polish)
    # ------------------------------------------------------------------
    while time_left() > 8.0:
        # random binary start
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

        # simulated annealing on the binary vector
        max_iter_sa = int(min(150000, max(5000, time_left() * 150)))
        h_sa, c5_sa = _simulated_annealing_binary(
            h0,
            max_iter=max_iter_sa,
            target_c5=target_c5,
            deadline=start_time + budget_s - 2.0,
        )
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa.copy(), c5_sa

        # if we already beat the target, stop early
        if best_c5 < target_c5:
            break

        # try a focused reduction on the current best binary vector
        deadline = start_time + budget_s - 2.0
        best_bin = (best_h > 0.5).astype(float)   # force binary while keeping the sum
        # ensure exact half ones
        cur = int(best_bin.sum())
        need = n // 2
        if cur != need:
            # tiny correction to meet the exact count
            diff = need - cur
            if diff > 0:
                zero_idx = np.where(best_bin == 0)[0]
                flip = np.random.choice(zero_idx, diff, replace=False)
                best_bin[flip] = 1.0
            else:
                one_idx = np.where(best_bin == 1)[0]
                flip = np.random.choice(one_idx, -diff, replace=False)
                best_bin[flip] = 0.0

        best_bin, final_corr = _targeted_reduce(
            best_bin, target_corr, deadline
        )
        c5_reduced = final_corr / (n / 2)
        if c5_reduced < best_c5:
            best_h, best_c5 = best_bin.copy(), c5_reduced

        # final SLSQP polish on the (now possibly fractional) best vector
        if time_left() > 4.0:
            h_opt, c5_opt = _solve_h_slsqp(best_h, max_iter=1200)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

        # early exit if we have succeeded
        if best_c5 < target_c5:
            break

    # ------------------------------------------------------------------
    # final safety polishing (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 3.0:
        h_opt, c5_opt = _solve_h_slsqp(best_h, max_iter=2000)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # ------------------------------------------------------------------
    # final feasibility check and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)

    best_h = _project_to_feasible(best_h, n)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(n)
