def gradient_softmax(h, alpha):
    n = h.size
    c = np.correlate(h, 1.0 - h, mode='full')
    # compute weights
    maxc = np.max(c)  # maybe for numerical stability
    expc = np.exp(alpha * (c - maxc))  # subtract maxc to avoid overflow
    Z = np.sum(expc)
    w = expc / Z   # shape 2n-1

    # compute gradient
    grad = np.zeros_like(h)
    # forward term: (1 - h[j+s])
    # iterate s from -(n-1) to (n-1)
    for idx_s, weight in enumerate(w):
        s = idx_s - (n - 1)
        if weight == 0:
            continue
        if s >= 0:
            # j from 0 to n-1-s
            j_end = n - s
            grad[:j_end] += weight * (1.0 - h[s:])  # h[j+s] = h[s:]; h slice starts at s
        else:
            # s < 0, j from -s to n-1
            start = -s
            grad[start:] += weight * (1.0 - h[:n + s])  # because j+s = (start + k) + s = k, k runs 0..n+s-1
    # backward term: - h[j-s]
    # we can reuse w similarly, sign negative
    # For each s, subtract weight * h[j - s] where valid.
    for idx_s, weight in enumerate(w):
        s = idx_s - (n - 1)
        if weight == 0:
            continue
        if s >= 0:
            # j from s to n-1, because j - s >=0
            start = s
            grad[start:] -= weight * h[:n - s]
        else:
            # s <0, j from 0 to n-1 + s
            end = n + s
            grad[:end] -= weight * h[-s:]  # since j - s = j + |s|, h index starts at -s
    # Also need derivative of scaling factor (dx) constant if included; ignore.
    return grad

def _grad_softmax(h: np.ndarray, alpha: float) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # compute weights w
    maxc = np.max(corr)
    expc = np.exp(alpha * (corr - maxc))
    Z = np.sum(expc)
    w = expc / Z
    grad = np.zeros_like(h)
    # forward term:
    # for s >= 0
    for s in range(0, n):
        weight = w[s + n - 1]
        if weight == 0:
            continue
        # j from 0 to n-1-s
        j_end = n - s
        if j_end > 0:
            grad[:j_end] += weight * (1.0 - h[s:])
    # s negative
    for s in range(-n + 1, 0):
        weight = w[s + n - 1]
        if weight == 0:
            continue
        start = -s
        grad[start:] += weight * (1.0 - h[:n + s])  # n + s = n - |s|
    # backward term:
    for s in range(0, n):
        weight = w[s + n - 1]
        if weight == 0:
            continue
        # j from s to n-1
        grad[s:] -= weight * h[:n - s]
    for s in range(-n + 1, 0):
        weight = w[s + n - 1]
        if weight == 0:
            continue
        end = n + s  # n - |s|
        grad[:end] -= weight * h[-s:]
    return grad

import time
import numpy as np
from scipy import optimize

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {x | 0<=x_i<=1, sum x = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """C5 for discrete h on [0,2] (dx = 2/n)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Refine with SLSQP (same as previous code, but max_iter configurable)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape h(x)=min(x,2-x) sampled on n points, projected to feasible set."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _grad_softmax(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of (1/α)·log Σ exp(α·corr) wrt h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    expc = np.exp(alpha * (corr - maxc))
    w = expc / np.sum(expc)                     # weights, sum to 1, length 2n‑1
    grad = np.zeros_like(h)

    # forward contribution: weight * (1 - h_{j+s})
    for s in range(0, n):
        weight = w[s + n - 1]
        if weight == 0.0:
            continue
        end = n - s
        if end > 0:
            grad[:end] += weight * (1.0 - h[s:])
    for s in range(-n + 1, 0):
        weight = w[s + n - 1]
        if weight == 0.0:
            continue
        start = -s
        grad[start:] += weight * (1.0 - h[:n + s])

    # backward contribution: -weight * h_{j-s}
    for s in range(0, n):
        weight = w[s + n - 1]
        if weight == 0.0:
            continue
        start = s
        grad[start:] -= weight * h[:n - s]
    for s in range(-n + 1, 0):
        weight = w[s + n - 1]
        if weight == 0.0:
            continue
        end = n + s
        grad[:end] -= weight * h[-s:]

    return grad


def _gradient_descent_h(start_h: np.ndarray, alphas, it_per_alpha, step0,
                        time_left):
    """Projected gradient descent on smooth max‑correlation."""
    n = start_h.size
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    for alpha in alphas:
        step = step0 / (alpha + 1.0)
        for _ in range(it_per_alpha):
            if time_left() < 0.5:
                return best_h, best_c5
            grad = _grad_softmax(h, alpha)
            h_new = h - step * grad
            h_new = _project_to_simplex_box(h_new, n / 2.0)
            c5_new = _c5_from_h(h_new)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            if c5_new <= _c5_from_h(h):
                h = h_new
                step *= 1.02
            else:
                step *= 0.6
                if step < 1e-6:
                    step = 1e-6
    return best_h, best_c5
