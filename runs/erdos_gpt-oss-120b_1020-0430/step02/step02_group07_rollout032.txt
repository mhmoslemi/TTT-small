def directed_perturb_weighted(h, rng, scale=0.2):
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (h.size - 1)
    # overlapping region
    if k >=0:
        i_start = 0
        i_end = h.size - k
    else:
        i_start = -k
        i_end = h.size
    i_range = np.arange(i_start, i_end)
    # compute weight w = 1 - h[i+k] for each i in i_range
    if k >=0:
        w = 1 - h[i_range + k]
    else:
        w = 1 - h[i_range + k]  # works because k negative
    # source candidates: high w and h > eps
    source_mask = (w > np.median(w)) & (h[i_range] > 1e-12)
    target_mask = (w < np.median(w)) & (h[i_range] < 1 - 1e-12)
    src_idx = np.where(source_mask)[0]
    tgt_idx = np.where(target_mask)[0]
    if src_idx.size > 0 and tgt_idx.size >0:
        s = rng.choice(src_idx)
        t = rng.choice(tgt_idx)
        max_eps = min(h[i_range[s]], 1 - h[i_range[t]])
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i_range[s]] -= eps
        new_h[i_range[t]] += eps
        return new_h
    else:
        return perturb_h(h, rng, scale)

def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0., 1., n, endpoint=False)
    x_new = np.linspace(0., 1., n*factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)

    target = (n*factor) / 2.0
    h_up = np.clip(h_up, 0., 1.)
    h_up = enforce_sum(h_up, target, rng)
    return h_up

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- local search ----------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < math.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def quick_refine(h, rng):
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=800, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def deep_refine(h, rng):
    n = h.size
    max_iter_slsqp = 200 if n > 200 else 300
    hill_iters = 800 if n > 200 else 1500
    anneal_iters = 2000 if n > 200 else 3000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter_slsqp)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    trial_sa, c_sa = simulated_annealing(
        best_h, rng, max_iters=anneal_iters, init_temp=0.03, decay=0.998, scale=0.2
    )
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


# ---------- directed perturbation ----------
def directed_perturb_weighted(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    if k >= 0:
        i_start = 0
        i_end = n - k
        i_range = np.arange(i_start, i_end)
        partner = i_range + k
    else:
        i_start = -k
        i_end = n
        i_range = np.arange(i_start, i_end)
        partner = i_range + k
    w = 1.0 - h[partner]
    median_w = np.median(w)
    src_mask = (w > median_w) & (h[i_range] > 1e-12)
    tgt_mask = (w < median_w) & (h[i_range] < 1.0 - 1e-12)
    src_idx = np.where(src_mask)[0]
    tgt_idx = np.where(tgt_mask)[0]
    if src_idx.size == 0 or tgt_idx.size == 0:
        return perturb_h(h, rng, scale)
    s = rng.choice(src_idx)
    t = rng.choice(tgt_idx)
    max_eps = min(h[i_range[s]], 1.0 - h[i_range[t]])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i_range[s]] -= eps
    new_h[i_range[t]] += eps
    return new_h


def directed_sa(h, rng, max_iters, init_scale=0.3, final_scale=0.02):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(max_iters):
        scale = init_scale * (1.0 - it / max_iters) + final_scale * (it / max_iters)
        new_h = directed_perturb_weighted(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        else:
            # occasional uphill acceptance
            prob = 0.01 * (1.0 - it / max_iters)
            if rng.random() < prob:
                cur_h, cur_c = new_h, new_c
    return best_h, best_c


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ---- initial solution ----
    if "initial_h_values" in globals():

        n = 200

    # ---- quick refinement (SLSQP + hill‑climb) ----
    if time.time() < deadline - 10:

    else:
        best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- first directed SA phase (still at original resolution) ----
    time_left = deadline - time.time()
    if time_left > budget_s * 0.15:
        max_iters = int(time_left * 250)  # approx 250 moves per second
        best_h, best_c = directed_sa(best_h, rng, max_iters=max_iters,
                                     init_scale=0.4, final_scale=0.02)

    # ---- up‑sample and deep refinement if time permits ----
    time_left = deadline - time.time()
    if time_left > budget_s * 0.20:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        up_h, up_c = deep_refine(h_up, rng)
        if up_c < best_c:

        # second directed SA at higher resolution
        time_left = deadline - time.time()
        if time_left > 5:

                                         init_scale=0.3, final_scale=0.01)

    # ---- final clean‑up ----
    n = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
