def gradient_descent_softmax(h, rng, betas, steps_list, lr_start=0.1):
    n = h.size
    target = n / 2.0
    for beta, steps in zip(betas, steps_list):
        # compute step size schedule maybe decaying with each iteration
        for it in range(steps):
            # corr
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            # compute softmax weights
            # For stability: subtract max_corr before exp
            exps = np.exp(beta * (corr - max_corr))
            p = exps / np.sum(exps)
            # compute gradient
            grad = np.zeros_like(h)
            # k=0 index:
            idx0 = n - 1
            w0 = p[idx0]
            grad += w0 * (1 - 2 * h)

def hill_climb(h_start, rng, max_iters=1000, start_step=0.5):
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = 0.95
    n = h.size
    for it in range(max_iters):
        # create new by perturbation similar to earlier (transfer epsilon)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # epsilon limited by bounds
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # ensure bounds already ok
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best = new_h.copy()
            h = new_h
        # decay step
        if (it+1) % (max_iters // 5 + 1) == 0:
            step = max(step * decay, 1e-5)
    return best, best_c

def gradient_descent_softmax(h, rng, betas, steps_list, lr_start=0.1):
    n = h.size
    target = n / 2.0
    ks_offset = n - 1
    for beta, steps in zip(betas, steps_list):
        for it in range(steps):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            exp_vals = np.exp(beta * (corr - max_corr))
            p = exp_vals / np.sum(exp_vals)
            grad = np.zeros_like(h)
            w0 = p[ks_offset]
            grad += w0 * (1 - 2 * h)
            w_pos = p[ks_offset + 1:]   # length n-1
            w_neg = p[:ks_offset][::-1] # reversed length n-1
            w_sum = w_pos + w_neg
            total_const = np.sum(w_sum)
            grad += total_const
            # subtract weighted shifts
            for k_idx, wk in enumerate(w_sum, start=1):
                d = k_idx
                grad[:n - d] -= wk * h[d:]
                grad[d:] -= wk * h[:n - d]
            # step size schedule
            step = lr_start / (1.0 + it * 0.001)
            h = h - step * grad
            # project onto feasible set
            np.clip(h, 0.0, 1.0, out=h)
            # sum correction
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                if diff > 0:
                    cap = np.sum(1.0 - h)
                    if cap > 0:
                        h += diff * (1.0 - h) / cap
                else:
                    cap = np.sum(h)
                    if cap > 0:
                        h += diff * h / cap

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C₅ for a step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_feasible(h):
    """Clip to [0,1] and enforce Σh = n/2 exactly (n = len(h))."""
    n = h.shape[0]
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the deficit proportionally to available slack
        if diff > 0:
            cap = np.sum(1.0 - h)
            if cap > 0:
                h += diff * (1.0 - h) / cap
        else:
            cap = np.sum(h)
            if cap > 0:
                h += diff * h / cap
        np.clip(h, 0.0, 1.0, out=h)
    return h


def perturb_h(h, rng, scale=0.5):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Simple hill‑climb that accepts only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 5)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def gradient_descent_softmax(h, rng, betas, steps_per_beta, lr_start=0.2):
    """Gradient descent on the log‑sum‑exp (soft‑max) surrogate."""
    n = h.size
    offset = n - 1                     # index of shift k=0 in the correlation vector
    target = n / 2.0
    for beta, steps in zip(betas, steps_per_beta):
        for it in range(steps):
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            exp_vals = np.exp(beta * (corr - m))
            p = exp_vals / np.sum(exp_vals)          # soft‑max weights
            grad = np.zeros_like(h)

            # contribution from k = 0
            w0 = p[offset]
            grad += w0 * (1.0 - 2.0 * h)

            # combine positive / negative shifts
            w_pos = p[offset + 1:]                     # k = 1 … n‑1
            w_neg = p[:offset][::-1]                   # k = –1 … –(n‑1)
            w = w_pos + w_neg                          # length n‑1
            const = np.sum(w)                          # each non‑zero shift adds 1 to all entries
            grad += const

            # subtract the shifted copies of h
            for k, wk in enumerate(w, start=1):
                grad[: n - k] -= wk * h[k:]
                grad[k:]     -= wk * h[: n - k]

            # step size (simple decay)
            step = lr_start / (1.0 + it * 0.001)
            h = h - step * grad

            # project back onto the box and the hyperplane Σh = n/2
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                if diff > 0:
                    cap = np.sum(1.0 - h)
                    if cap > 0:
                        h += diff * (1.0 - h) / cap
                else:
                    cap = np.sum(h)
                    if cap > 0:
                        h += diff * h / cap
                np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Initialise from a parent if available
    # ------------------------------------------------------------------
    best_h = None
    best_c = np.inf
    best_n = None
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = project_feasible(h0)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
        best_n = h0.size

    # ------------------------------------------------------------------
    # Candidate grid sizes (keep them modest)
    # ------------------------------------------------------------------
    cand_ns = [120, 180, 240]
    if best_n is not None:
        cand_ns = [best_n] + [n for n in cand_ns if n != best_n]

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for n in sorted(set(cand_ns)):
        if time_left() < 1.0:
            break
        # number of restarts depends on the size
        max_restarts = 8 if n <= 120 else 5
        for restart in range(max_restarts):
            if time_left() < 1.0:
                break

            # ---- initialise a feasible vector --------------------------------
            if best_n == n and restart == 0 and best_h is not None:
                h = best_h.copy()
            else:
                # random continuous feasible vector
                h = rng.random(n)
                h = project_feasible(h)

            # ---- quick hill‑climb -------------------------------------------
            h, c = hill_climb(h, rng, max_iters=1200, start_step=0.5)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ---- gradient‑descent on soft‑max surrogate ---------------------
            betas = [5.0, 20.0, 80.0, 200.0, 500.0]
            steps = [1200, 800, 500, 300, 200]
            h = gradient_descent_softmax(h, rng, betas, steps, lr_start=0.2)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ---- final short hill‑climb ------------------------------------
            h, c = hill_climb(h, rng, max_iters=800, start_step=0.3)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # Optional up‑sampling refinement (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 5.0 and best_n * 2 <= 240:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up = gradient_descent_softmax(h_up, rng,
                                        betas=[20.0, 80.0, 200.0, 500.0],
                                        steps_per_beta=[800, 500, 300, 200],
                                        lr_start=0.15)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final feasibility enforcement
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = project_feasible(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
