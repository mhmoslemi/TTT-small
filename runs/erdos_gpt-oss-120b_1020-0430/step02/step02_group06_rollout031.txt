def _project_to_feasible(x, n, tol=1e-9, max_iter=200):
    # water-filling to enforce sum = n/2 and 0<=x<=1
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(x > 0)[0]
        else:
            free = np.where(x < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    # final correction
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and x[i] < 1 - 1e-12:
                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 1e-12:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

                break
    return np.clip(x, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _local_refine(h: np.ndarray, attempts: int = 20, delta: float = 0.01) -> np.ndarray:
    """Small stochastic hill‑climb around h."""
    n = h.size
    best = h.copy()
    best_c = _c5_from_h(best)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = np.clip(trial, 0.0, 1.0)
        trial = _project_to_feasible(trial, n)
        c = _c5_from_h(trial)
        if c < best_c:
            best, best_c = trial, c
    return best


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _make_symmetric(h: np.ndarray) -> np.ndarray:
    """Force complement symmetry: h[i] = 1‑h[n‑1‑i]."""
    n = h.size
    half = n // 2
    a = (h[:half] + (1.0 - h[-1:-half - 1:-1])) / 2.0
    a = np.clip(a, 0.0, 1.0)
    h_sym = np.empty(n)
    h_sym[:half] = a
    h_sym[half:] = 1.0 - a[::-1]
    return h_sym


def _construct_sym(a: np.ndarray) -> np.ndarray:
    """Build the full symmetric vector from the half‑vector a."""
    half = a.size
    n = half * 2
    h = np.empty(n)
    h[:half] = a
    h[half:] = 1.0 - a[::-1]
    return h

def _improve_full(h_start: np.ndarray, time_limit: float):
    """Simulated‑annealing on the full vector (sum fixed)."""
    n = h_start.size
    cur = _project_to_feasible(h_start, n)
    cur_c = _c5_from_h(cur)
    best, best_c = cur.copy(), cur_c
    start = time.time()
    max_eps, min_eps = 0.08, 0.002
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        eps = max_eps - (max_eps - min_eps) * (elapsed / time_limit)
        src_candidates = np.where(cur > eps)[0]
        dst_candidates = np.where(cur < 1 - eps)[0]
        if src_candidates.size == 0 or dst_candidates.size == 0:
            break
        src = np.random.choice(src_candidates)
        dst = np.random.choice(dst_candidates)
        delta = eps * np.random.rand()
        new = cur.copy()
        new[src] -= delta
        new[dst] += delta
        # bounds are already respected, but clip for safety
        new = np.clip(new, 0.0, 1.0)
        new_c = _c5_from_h(new)
        T = 0.004 * (1 - elapsed / time_limit) + 1e-5
        if new_c < cur_c or np.random.rand() < np.exp((cur_c - new_c) / T):
            cur, cur_c = new, new_c
            if new_c < best_c:
                best, best_c = new.copy(), new_c
    best = _local_refine(best, attempts=12, delta=0.005)
    best_c = _c5_from_h(best)
    return best, best_c


def _improve_sym(h_sym_start: np.ndarray, time_limit: float):
    """Simulated‑annealing on a symmetric representation."""
    n = h_sym_start.size
    half = n // 2
    cur_a = h_sym_start[:half].copy()
    cur = _construct_sym(cur_a)
    cur_c = _c5_from_h(cur)
    best_a, best_c = cur_a.copy(), cur_c
    start = time.time()
    max_eps, min_eps = 0.08, 0.002
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        eps = max_eps - (max_eps - min_eps) * (elapsed / time_limit)
        i = np.random.randint(half)
        j = np.random.randint(half)
        if i == j:
            continue
        max_delta = min(eps, cur_a[i])
        max_delta = min(max_delta, 1.0 - cur_a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new = _construct_sym(new_a)
        new_c = _c5_from_h(new)
        T = 0.004 * (1 - elapsed / time_limit) + 1e-5
        if new_c < cur_c or np.random.rand() < np.exp((cur_c - new_c) / T):
            cur_a, cur_c = new_a, new_c
            if new_c < best_c:
                best_a, best_c = new_a.copy(), new_c
    best = _construct_sym(best_a)
    best = _local_refine(best, attempts=12, delta=0.005)
    best_c = _c5_from_h(best)
    return best, best_c

# Main driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # parent construction (if any)
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    target = 0.3808
    best_h = None
    best_c = float("inf")
    best_n = None

    # grid sizes to explore (even only)
    sizes = [96, 128, 144, 160, 192, 224, 256]
    for n in sizes:
        if time_left() < 6:
            break

        # base vector for this size
        if parent_h is not None:
            if parent_h.size != n:
                h0 = _upscale_h(parent_h, n)
            else:
                h0 = parent_h.copy()
        else:
            h0 = np.full(n, 0.5)

        # symmetric start
        h_sym0 = _make_symmetric(h0)

        # --- symmetric optimisation ---------------------------------
        t_sym = min(time_left() * 0.5, 180)
        h_sym_opt, c_sym = _improve_sym(h_sym0, t_sym)

        # update global best
        if c_sym < best_c:
            best_h, best_c, best_n = h_sym_opt.copy(), c_sym, n
        if best_c < target:
            break

        # --- full optimisation (starting from the symmetric result) ---
        t_full = min(time_left() * 0.4, 150)
        h_full_opt, c_full = _improve_full(h_sym_opt, t_full)

        if c_full < best_c:
            best_h, best_c, best_n = h_full_opt.copy(), c_full, n
        if best_c < target:
            break

        # quick refinement on the current global best for this size
        h_ref = _local_refine(best_h, attempts=15, delta=0.003)
        c_ref = _c5_from_h(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref.copy(), c_ref
            best_n = n
        if best_c < target:
            break

    # fallback if nothing was produced
    if best_h is None:
        n = 96
        best_h = np.full(n, 0.5)
        best_c = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c), int(best_n)
