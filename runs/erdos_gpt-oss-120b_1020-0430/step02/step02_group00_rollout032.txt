def _subgradient_descent(h, max_iter, time_left):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    eta = 0.5  # initial step size
    for it in range(max_iter):
        if time_left() < 0.01:
            break
        # compute correlation array
        corr = np.convolve(h, (1 - h)[::-1], mode='full')
        max_val = corr.max()
        # find index of max (choose one)
        k_idx = np.argmax(corr)
        # compute shift k
        k = k_idx - (n - 1)
        # compute gradient g
        g = np.zeros(n)
        if k >= 0:
            # 1 - h[j+k] term
            g[:n - k] += 1 - h[k:]
            # -h[j - k] term
            g[k:] -= h[:n - k]
        else:
            kp = -k
            # 1 - h[j+k] term for j >= kp
            g[kp:] += 1 - h[:n - kp]
            # -h[j - k] term for j < n - kp
            g[:n - kp] -= h[kp:]
        # step
        h_new = h - eta * g
        # project
        h_new = _project_to_feasible(h_new, n)
        # compute new c5
        new_c5 = _c5_from_h(h_new)
        # Accept if improvement
        if new_c5 < best_c5 - 1e-9:  # improvement
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new
            # maybe increase step size a little
            eta = min(eta * 1.05, 1.0)
        else:
            # not improving, reduce eta
            eta *= 0.5
            # optionally revert h (keep old)
            # If eta too small, random perturb to escape.
            if eta < 1e-5:
                # random perturb small
                delta = 0.1 * np.random.randn(n)
                h = _project_to_feasible(h + delta, n)
                eta = 0.5
        # optionally rotate best solution to avoid local minima also we keep best.
    return best_h, best_c5

def _project_to_feasible(h, n):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # compute diff to add or subtract
    diff = target - s
    # indices that are not at bounds the direction we need
    if diff > 0:
        # need to increase sum, choose indices where h < 1
        free = np.where(h < 1)[0]
        if free.size == 0:
            return h
        # distribute diff equally among free entries but respecting capacity
        # iterate
        while diff > 1e-12 and free.size > 0:
            add = diff / free.size
            # capacity per index
            cap = 1 - h[free]
            # if add <= min(cap) then we can distribute equally
            if add <= cap.min():
                h[free] += add
                diff = 0.0
                break
            # else fill those with smallest capacity
            # find indices where cap is minimal (or less than add)
            # fill up indices where cap <= add
            fill = free[cap <= add + 1e-12]
            if fill.size > 0:
                h[fill] = 1.0
                diff -= cap[cap <= add + 1e-12].sum()
                free = np.setdiff1d(free, fill, assume_unique=True)
            else:

                h[free] += add

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the feasible set {h∈[0,1]^n, sum(h)=n/2}
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

# ----------------------------------------------------------------------
def _time_left(start: float, budget: float) -> float:
    return budget - (time.time() - start)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the max‑correlation objective
# ----------------------------------------------------------------------
def _subgradient_descent(h0: np.ndarray, max_iter: int, start: float, budget: float):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    eta = 0.5  # initial step size

    for it in range(max_iter):
        if _time_left(start, budget) <= 0.0:
            break

        corr = np.convolve(h, (1.0 - h)[::-1], mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)  # shift corresponding to the maximal correlation

        # gradient of corr_k
        g = np.zeros(n)
        if k >= 0:
            if n - k > 0:
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
        else:
            kp = -k
            if n - kp > 0:
                g[kp:] += 1.0 - h[: n - kp]
                g[: n - kp] -= h[kp:]

        h_new = h - eta * g
        h_new = _project_to_feasible(h_new, n)
        new_c5 = _c5_from_h(h_new)

        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            eta = min(eta * 1.2, 2.0)   # try a larger step after improvement
        else:
            eta *= 0.5                    # shrink step on failure
            if eta < 1e-8:                # occasional random shake‑up
                h = _project_to_feasible(h + 0.01 * np.random.randn(n), n)
                eta = 0.5

    return best_h, best_c5


# ----------------------------------------------------------------------
# Upscaling (linear interpolation) and random binary starter
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _random_binary_feasible(n: int) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Starting point (use the parent construction if available)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is not None:
        h0 = init_h.copy()
    else:
        h0 = _random_binary_feasible(256)

    n0 = h0.size
    h0 = _project_to_feasible(h0, n0)

    best_h = h0.copy()
    best_c5 = _c5_from_h(best_h)
    best_n = n0

    # ------------------------------------------------------------------
    # Phase 1 – optimise at the current resolution
    # ------------------------------------------------------------------
    while _time_left(start_time, budget_s) > 20.0:
        remaining = _time_left(start_time, budget_s)
        # Rough budget: about 200 iterations per remaining second
        max_iter = int(min(remaining * 200, 20000))
        if max_iter < 10:
            max_iter = 10

        h_opt, c5_opt = _subgradient_descent(best_h, max_iter, start_time, budget_s)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            best_n = n0
            if best_c5 < 0.3808:
                break

        # occasional random restart
        if _time_left(start_time, budget_s) > 5.0:
            h_rand = _random_binary_feasible(n0)
            h_rand = _project_to_feasible(h_rand, n0)
            h_opt2, c5_opt2 = _subgradient_descent(h_rand, max_iter, start_time, budget_s)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2
                best_n = n0
                if best_c5 < 0.3808:
                    break
        # one pass is enough; further loops would just repeat work
        break

    # ------------------------------------------------------------------
    # Phase 2 – upscale and refine (try 512, then 1024 if time permits)
    # ------------------------------------------------------------------
    n_cur = best_n
    while _time_left(start_time, budget_s) > 5.0:
        n_next = n_cur * 2
        if n_next > 1024:
            break

        h_up = _upscale_h(best_h, n_next)
        h_up = _project_to_feasible(h_up, n_next)

        remaining = _time_left(start_time, budget_s)
        max_iter = int(min(remaining * 150, 15000))
        if max_iter < 10:
            max_iter = 10

        h_opt, c5_opt = _subgradient_descent(h_up, max_iter, start_time, budget_s)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            n_cur = n_next
            if best_c5 < 0.3808:
                break
        else:
            # try a random binary start at this resolution before giving up
            if _time_left(start_time, budget_s) > 10.0:
                h_rand = _random_binary_feasible(n_next)
                h_rand = _project_to_feasible(h_rand, n_next)
                h_opt2, c5_opt2 = _subgradient_descent(h_rand, max_iter, start_time, budget_s)
                if c5_opt2 < best_c5:
                    best_h, best_c5 = h_opt2, c5_opt2
                    n_cur = n_next
                    continue
            break

    # ------------------------------------------------------------------
    # Final polish (a last sub‑gradient run with a higher iteration cap)
    # ------------------------------------------------------------------
    if _time_left(start_time, budget_s) > 2.0:
        h_opt, c5_opt = _subgradient_descent(best_h, 50000, start_time, budget_s)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            best_n = best_h.size

    # ------------------------------------------------------------------
    # Ensure final feasibility and compute final bound
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)
    best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
