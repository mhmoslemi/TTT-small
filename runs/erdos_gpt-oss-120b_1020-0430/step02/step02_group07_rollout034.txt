Now design SA:

```
def simulated_annealing_binary(h0, rng, max_iters, temp_start, temp_end):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = temp_start
    decay = (temp_end / temp_start) ** (1.0 / max_iters)
    for it in range(max_iters):
        # propose swap
        ones = np.flatnonzero(h)  # indices of 1
        zeros = np.flatnonzero(1 - h)  # indices of 0
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        temp *= decay
    return best_h, best_c
```

Now hill climbing after SA: repeat swaps until no improvement:

```
def hill_climb_binary(h, rng, max_iters=10000):
    h_cur = h.copy()
    cur_c = compute_c5(h_cur)
    improved = True
    it = 0
    while improved and it < max_iters:
        improved = False
        # try random swap; if improvement, accept and set improved
        i = rng.choice(np.flatnonzero(h_cur))
        j = rng.choice(np.flatnonzero(1 - h_cur))
        # swap
        h_cur[i] = 0
        h_cur[j] = 1
        new_c = compute_c5(h_cur)
        if new_c < cur_c:
            cur_c = new_c
            improved = True
        else:
            # revert
            h_cur[i] = 1
            h_cur[j] = 0
        it += 1
    return h_cur, cur_c
```

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Fast C5 evaluation using FFT (full correlation of h with 1‑h)."""
    n = h.size
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    L = 2 * n - 1
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b_rev, n=L)
    corr = np.fft.irfft(fa * fb, n=L)
    return float(np.max(corr) * (2.0 / n))


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary vector of length n with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, half, replace=False)
    h[idx] = 1.0
    return h


def binary_from_continuous(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Round a continuous vector to a feasible binary one (sum = n/2)."""
    n = arr.size
    target = n // 2
    bin_arr = (arr >= 0.5).astype(float)
    cur = int(bin_arr.sum())
    if cur > target:

        # largest values among zeros are the most borderline to become 1
        order = zeros[np.argsort(-arr[zeros])]
        bin_arr[order[: target - cur]] = 1.0
    return bin_arr

    for _ in range(max_iters):
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break

        # try a few random swaps, keep the best improvement (if any)
        best_swap_c = cur_c
        best_i = best_j = None
        for _ in range(trials_per_iter):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            cand_c = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if cand_c < best_swap_c:
                best_swap_c = cand_c
                best_i, best_j = i, j

        if best_i is not None:
            # apply the improving swap
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_c = best_swap_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        else:
            # possibly accept a worse move
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h)
            if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
                cur_c = new_c
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0

        temp *= decay

    return best_h, best_c


def hill_climb_binary(
    h0: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 20000,
    no_improve_limit: int = 2000,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb on binary vectors (swap‑only moves)."""
    h = h0.copy()
    cur_c = compute_c5(h)
    no_improve = 0
    for _ in range(max_iters):
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
            if no_improve >= no_improve_limit:
                break
    return h, cur_c


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    TARGET = 0.3808  # we aim to get at most this value

    # ---- initialise base resolution ----
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n = base.size
        best_h = binary_from_continuous(base, rng)
    else:
        n = 360
        best_h = random_binary_vector(n, rng)
    best_c = compute_c5(best_h)

    # ---- phase 1 : many SA + HC restarts on size n ----
    SA_ITERS = 8000
    while time.time() - start < budget_s * 0.65:
        h0 = random_binary_vector(n, rng)
        h_sa, c_sa = simulated_annealing_binary(
            h0, rng, max_iters=SA_ITERS, temp_start=0.7, temp_end=0.001, trials_per_iter=3
        )
        h_hc, c_hc = hill_climb_binary(h_sa, rng, max_iters=20000, no_improve_limit=2000)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
            if best_c <= TARGET:
                break

    # ---- optional extra local refinement on the current best ----
    best_h, best_c = hill_climb_binary(best_h, rng, max_iters=40000, no_improve_limit=5000)

    # ---- phase 2 : up‑sample to 2 × n if time permits ----
    if time.time() - start < budget_s * 0.95:
        n_up = n * 2
        target_up = n_up // 2
        # build a mild bias (0.3 / 0.7) from the current binary pattern
        prob_coarse = 0.5 + 0.4 * (best_h - 0.5)  # yields 0.3 or 0.7
        prob_up = np.repeat(prob_coarse, 2)

        def biased_sample():
            rand = rng.random(n_up)
            h = (rand < prob_up).astype(float)
            cur = int(h.sum())
            diff = target_up - cur
            if diff > 0:
                zeros = np.flatnonzero(1.0 - h)
                if len(zeros) >= diff:
                    idx = zeros[np.argsort(-prob_up[zeros])[:diff]]
                    h[idx] = 1.0
                else:
                    h[:diff] = 1.0
            elif diff < 0:
                ones = np.flatnonzero(h)
                if len(ones) >= -diff:
                    idx = ones[np.argsort(prob_up[ones])[: -diff]]
                    h[idx] = 0.0
                else:
                    h[: -diff] = 0.0
            return h

        best_h_up, best_c_up = None, np.inf
        SA_ITERS_UP = 8000
        while time.time() - start < budget_s * 0.97:
            h_start = biased_sample()
            h_sa, c_sa = simulated_annealing_binary(
                h_start,
                rng,
                max_iters=SA_ITERS_UP,
                temp_start=0.7,
                temp_end=0.001,
                trials_per_iter=3,
            )
            h_hc, c_hc = hill_climb_binary(h_sa, rng, max_iters=20000, no_improve_limit=2000)
            if c_hc < best_c_up:
                best_c_up, best_h_up = c_hc, h_hc.copy()
                if best_c_up <= TARGET:
                    break

        if best_h_up is not None and best_c_up < best_c:
            best_h, best_c, n = best_h_up, best_c_up, n_up

    # ---- final feasibility clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n // 2
    cur = int(round(best_h.sum()))
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(1.0 - best_h)
        best_h[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(best_h)
        best_h[ones[: -diff]] = 0.0

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
