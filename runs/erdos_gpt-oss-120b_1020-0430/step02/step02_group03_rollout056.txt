import time
import numpy as np
from scipy.optimize import minimize

# ------------------- helper utilities ------------------- #
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that sum(h)=target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    target = n / 2.0
    # start from uniform random numbers
    h = rng.random(n)
    # scale to target sum
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    # final repair
    h = enforce_sum(h, target, rng)
    return h


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = n/2."""
    return np.sum(x) - (x.size / 2.0)


def smooth_objective(x, beta):
    """Log‑sum‑exp smooth approximation of max correlation."""
    s = np.correlate(x, 1.0 - x, mode="full")
    b = beta * s
    bmax = np.max(b)
    # stable log‑sum‑exp
    return (bmax + np.log(np.exp(b - bmax).sum())) / beta


def smooth_gradient(x, beta):
    """Gradient of smooth_objective."""
    n = x.size
    s = np.correlate(x, 1.0 - x, mode="full")
    b = beta * s
    bmax = np.max(b)
    exp_shift = np.exp(b - bmax)
    w = exp_shift / exp_shift.sum()          # normalized weights
    wcomp = 1.0 - x
    # convolution with zero padding (full mode)
    conv1 = np.convolve(w, wcomp, mode="full")
    conv2 = np.convolve(w[::-1], x, mode="full")
    start = n - 1
    grad = conv1[start:start + n] - conv2[start:start + n]
    return grad


def perturb_transfer(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out

    for f in (1, 2, 3):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(np.clip(xs / 2.0, 0.0, 1.0), n / 2.0, rng))
    candidates.append(enforce_sum(np.clip(1.0 - xs / 2.0, 0.0, 1.0), n / 2.0, rng))

    # few random vectors
    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    # ----- keep the best starter -----------------------------------------
    best_h = None
    best_c = float("inf")
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ----- continuation optimiser -----------------------------------------
    beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0]

    def refine(start_vec):
        """Run the β‑continuation from start_vec, update global best."""
        nonlocal best_h, best_c, n
        x = enforce_sum(start_vec, start_vec.size / 2.0, rng)
        for beta in beta_schedule:
            bounds = [(0.0, 1.0)] * x.size
            cons = {"type": "eq", "fun": sum_eq_constraint}
            try:
                res = minimize(
                    smooth_objective,
                    x,
                    args=(beta,),
                    jac=smooth_gradient,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 250, "ftol": 1e-12, "disp": False},
                )
            except Exception:
                break
            x = np.clip(res.x, 0.0, 1.0)
            # tiny exact sum correction
            diff = x.size / 2.0 - x.sum()
            if abs(diff) > 1e-12:
                x = enforce_sum(x, x.size / 2.0, rng)

        # optional local hill‑climb to polish the discrete C5
        x, c_val = hill_climb(x, rng, max_iters=2000, start_step=0.3)
        if c_val < best_c:
            best_c, best_h = c_val, x.copy()
        return x, c_val

    def hill_climb(h, rng, max_iters=1500, start_step=0.4):
        """Simple hill‑climb accepting only improvements on the exact C5."""
        cur = h.copy()
        cur_c = compute_c5(cur)
        step = start_step
        decay = max(1, max_iters // 10)
        for it in range(max_iters):
            cand = perturb_transfer(cur, rng, scale=step)
            cand_c = compute_c5(cand)
            if cand_c < cur_c:
                cur, cur_c = cand, cand_c
            if (it + 1) % decay == 0:
                step = max(step * 0.7, 5e-4)
        return cur, cur_c

    # ----- global search loop --------------------------------------------
    while time.time() - start_time < budget_s * 0.84:
        # decide a start point
        if rng.random() < 0.65 and best_h is not None:
            start = perturb_transfer(best_h, rng, scale=0.25)
        else:
            start = random_feasible_vector(n, rng)
        refine(start)
        # early break if we are already below the target record
        if best_c <= 0.3808:
            break

    # ----- optional up‑sampling (fine‑grid refinement) --------------------
    if time.time() - start_time < budget_s * 0.95 and n < 200:
        # linear interpolation to a larger grid (e.g., 200 points)
        new_n = 200
        xs_old = (np.arange(n) + 0.5) * (2.0 / n)
        xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        h_interp = np.interp(xs_new, xs_old, best_h)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        h_interp = enforce_sum(h_interp, new_n / 2.0, rng)
        # run continuation on the finer grid
        n = new_n
        refine(h_interp)

    # ----- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
