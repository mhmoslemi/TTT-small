    best_c5 = compute_c5(best)
    n = h.size

        frac = elapsed / budget
        eps_factor = eps_start * (eps_end / eps_start) ** frac

def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to decrease sum, move mass from >0 entries
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual greedily
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _hill_climb(h0, budget):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(h)
    start = time.time()
    # schedule parameters
    eps_start = 0.3
    eps_end = 0.005
    T_start = 0.015
    T_end = 1e-6
    while time.time() - start < budget:
        elapsed = time.time() - start
        frac = elapsed / budget
        eps_factor = eps_start * (eps_end / eps_start) ** frac
        T = T_start * (T_end / T_start) ** frac
        # random pair
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        hi = h[i]; hj = h[j]
        eps_max = min(hi, 1.0 - hj)
        if eps_max <= 0:
            continue
        eps = eps_max * np.random.rand() * eps_factor
        if eps <= 0:
            continue
        # propose
        h[i] = hi - eps
        h[j] = hj + eps
        val = _c5_from_h(h)
        delta = val - best_val
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
        # occasional re-projection to correct numeric drift
        if np.random.rand() < 0.001:
            h = _project_to_feasible(h, n)
    # final projection and refinement
    best_h = _project_to_feasible(best_h, n)
    best_val = _c5_from_h(best_h)
    return best_h, best_val

import time

# ---------- feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

        new_val = _c5_from_h(h)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

        if np.random.rand() < 0.001:
            h = _project_to_feasible(h, n)
            cur_val = _c5_from_h(h)

    best_h = _project_to_feasible(best_h, n)
    best_val = _c5_from_h(best_h)
    return best_h, best_val


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # parent construction, if any
    try:
        parent = np.array(initial_h_values, dtype=float)  # type: ignore
    except NameError:
        parent = None

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target = 0.3808

    # candidate grid sizes
    candidates = []
    if parent is not None:
        candidates.append(parent.size)
        base_n = parent.size
    else:
        base_n = 96
    for mul in (1, 2, 3, 4):
        n = base_n * mul
        if n % 2 == 0 and n <= 256:
            candidates.append(n)
    extra = [80, 96, 112, 128, 144, 160, 192, 224, 256]
    candidates.extend(extra)
    candidates = sorted(set([n for n in candidates if n % 2 == 0 and n <= 256]))

    # optimisation on each size
    for n in candidates:
        if time_left() < 5.0:
            break

        # initialise h on this grid
        if parent is not None:
            if parent.size == n:
                base_h = parent.copy()
            else:
                base_h = _upscale_h(parent, n)
        else:
            base_h = np.full(n, 0.5)
        base_h = _project_to_feasible(base_h, n)

        # time budget for this size (15 % of remaining, min 5 s)
        alloc = min(15.0, time_left() * 0.15)
        if alloc < 5.0:
            alloc = min(5.0, time_left() * 0.5)

        inner_start = time.time()
        while time.time() - inner_start < alloc and time_left() > 2.0:
            # perturb base
            h0 = base_h + (np.random.rand(n) - 0.5) * 0.10
            h0 = _project_to_feasible(h0, n)

            # hill‑climb (up to 5 s)
            sub_budget = min(5.0, alloc - (time.time() - inner_start))
            h_opt, c5_opt = _hill_climb(h0, sub_budget)

            # small local refinement
            h_opt = _local_refine(h_opt, attempts=12, delta=0.008)
            c5_opt = _c5_from_h(h_opt)

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                if best_c5 < target:
                    break
            if best_c5 < target:
                break

        if best_c5 < target:
            break

    # final refinement with any remaining time
    if best_h is not None and time_left() > 5.0:
        h_ref, c5_ref = _hill_climb(best_h, time_left())
        h_ref = _local_refine(h_ref, attempts=15, delta=0.005)
        c5_ref = _c5_from_h(h_ref)
        if c5_ref < best_c5:
            best_h, best_c5, best_n = h_ref, c5_ref, h_ref.size

    # fallback (should never happen)
    if best_h is None:
        n = 96 if parent is None else parent.size
        best_h = _project_to_feasible(np.full(n, 0.5), n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
