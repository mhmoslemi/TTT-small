def max_lag_and_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)  # shift
    # compute gradient
    g = np.zeros_like(h)
    if k >= 0:
        # indices i from 0 to n-k-1
        end = n - k
        g[:end] += 1.0 - h[k:]
        g[k:] += -h[:end]
    else:
        kk = -k
        end = n - kk
        g[kk:] += 1.0 - h[:end]
        g[:end] += -h[kk:]
    return k, corr[idx], g

def enforce_sum(h, target, rng, max_iter=200):
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # choose random index i where we can adjust
        i = rng.integers(0, h.size)
        if diff > 0:
            # need to increase sum: can increase h[i] up to 1
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])  # negative add
        h[i] += add
        diff = target - h.sum()
    return h

def subgradient_optimize(h, rng, target, max_iter=200, step0=0.5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):

        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # gradient compute
        g = np.zeros_like(h)
        if k >= 0:
            end = n - k
            g[:end] += 1.0 - h[k:]
            g[k:] += -h[:end]
        else:
            kk = -k
            end = n - kk
            g[kk:] += 1.0 - h[:end]
            g[:end] += -h[kk:]

        step = step0 / (it + 1)
        new_h = h - step * g

        new_h = np.clip(new_h, 0.0, 1.0)

                add = max(diff, -new_h[i])

            diff = target - new_h[i].sum()

        diff = target - new_h.sum()

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Adjust entries of h (in‑place) so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            delta = min(diff, 1.0 - h[i])
        else:
            delta = max(diff, -h[i])
        if abs(delta) < 1e-15:
            continue
        h[i] += delta
        diff = target - h.sum()
    return h


def random_binary_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random 0/1 vector with exactly n/2 ones (n even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Move a small ε from one coordinate to another, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 1500, start_step: float = 0.5) -> tuple[np.ndarray, float]:
    """Simple hill‑climb: accept only moves that lower C₅."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def subgradient_optimize(h: np.ndarray, rng: np.random.Generator,
                         target: float, max_iter: int = 200, step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the worst lag."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        # correlation and worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        # gradient of the worst lag
        g = np.zeros_like(h)
        if k >= 0:
            end = n - k
            g[:end] += 1.0 - h[k:]
            g[k:] += -h[:end]
        else:
            kk = -k
            end = n - kk
            g[kk:] += 1.0 - h[:end]
            g[:end] += -h[kk:]

        # step size decays
        step = step0 / (it + 1)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)

        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
