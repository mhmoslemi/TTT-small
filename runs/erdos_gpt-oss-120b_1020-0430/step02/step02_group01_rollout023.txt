def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def compute_min_cross(h):
    corr = np.correlate(h, h, mode='full')
    # cross for positive k's
    cross_pos = corr[h.size:]   # indices n ... 2n-2. length n-1
    return np.min(cross_pos)

def gradient_move(h, step):
    n = h.size
    corr = np.correlate(h, h, mode='full')
    cross_pos = corr[n:] 
    k_min = np.argmin(cross_pos) + 1
    # compute gradient g
    k = k_min
    left = h[:n - k]
    right = h[k:]
    g = np.zeros(n, dtype=float)
    g[:n - k] += right
    g[k:] += left
    # find i (available to decrease) with min g[i] among h[i] > 0
    # find j (available to increase) with max g[j] among h[j] < 1
    # ensure g[j] > g[i] else no improvement
    mask_decr = h > 1e-12
    mask_incr = h < 1 - 1e-12
    if not mask_decr.any() or not mask_incr.any():
        return h  # nothing to move
    i = np.argmin(g + (~mask_decr) * 1e6)  # penalize non-decr
    j = np.argmax(g + (~mask_incr) * -1e6) # penalize non-incr
    if g[j] <= g[i] + 1e-12:
        # no beneficial move
        return h
    # transfer epsilon
    max_eps = min(h[i], 1 - h[j])
    eps = step * max_eps
    # also could limit eps to small fraction of max or step param.
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def min_cross_gradient_ascent(h, rng, max_iter=2000, step_factor=0.5):
    n = h.size
    target_sum = n / 2.0
    step = step_factor
    best = h.copy()
    best_c = compute_c5(best)
    for it in range(max_iter):
        # identify worst shift
        corr = np.correlate(h, h, mode='full')
        cross_pos = corr[n:]  # length n-1
        k = np.argmin(cross_pos) + 1
        # compute gradient
        left = h[:n - k]
        right = h[k:]
        g = np.zeros(n)
        g[:n - k] = right
        g[k:] = left
        # find i with min g where h[i] > 1e-12
        # find j with max g where h[j] < 1-1e-12
        mask_i = h > 1e-12
        mask_j = h < 1 - 1e-12
        if not mask_i.any() or not mask_j.any():
            break
        # indices where mask_i
        i_candidates = np.where(mask_i)[0]
        j_candidates = np.where(mask_j)[0]
        i = i_candidates[np.argmin(g[i_candidates])]
        j = j_candidates[np.argmax(g[j_candidates])]
        if g[j] <= g[i] + 1e-12:
            break
        max_eps = min(h[i], 1 - h[j])
        eps = step * max_eps
        if eps < 1e-12:
            break
        h[i] -= eps
        h[j] += eps
        # enforce sum (by construction sum unchanged)
        # compute current C5
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c = cur_c
            best = h.copy()
        # decay step
        step *= 0.999 # or 0.99
        # optionally break if step too small
        if step < 1e-4:
            # maybe stop early
            # but continue to try?
            pass
    return best, best_c
```

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # after some adjustments may still have numeric error, just correct by scaling small.
    if abs(diff) > 1e-9:
        # final uniform distribution of remaining diff
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- problem utilities ----------
def compute_c5(h):
    """Exact C5 for discrete h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to hit the required sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Uniform random point in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n

    res = minimize(
        softmax_obj,

        method="SLSQP",
        bounds=bounds,

    x = np.clip(res.x, 0.0, 1.0)

def min_cross_gradient_ascent(h, rng, max_iter=1500, step_factor=0.5):
    """Increase the smallest h‑h cross‑correlation (equivalently lower C5)."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_factor
    for it in range(max_iter):
        corr = np.correlate(h, h, mode="full")
        cross_pos = corr[n:]          # shifts 1..n-1
        k = int(np.argmin(cross_pos)) + 1
        # gradient for the worst shift
        left = h[: n - k]
        right = h[k:]
        g = np.zeros(n)
        g[: n - k] = right
        g[k:] = left
        # viable indices
        mask_i = h > 1e-12
        mask_j = h < 1.0 - 1e-12
        if not np.any(mask_i) or not np.any(mask_j):
            break
        i_cand = np.where(mask_i)[0]
        j_cand = np.where(mask_j)[0]
        i = i_cand[np.argmin(g[i_cand])]
        j = j_cand[np.argmax(g[j_cand])]
        if g[j] <= g[i] + 1e-12:
            break
        max_eps = min(h[i], 1.0 - h[j])
        eps = step * max_eps
        if eps < 1e-12:
            break
        h[i] -= eps
        h[j] += eps
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        step *= 0.999
        if step < 1e-5:
            step = 1e-5
    return best_h, best_c
