def gradient_descent_h(initial_h, max_iter=2000, eta=0.1, alpha_start=1.0, alpha_inc=0.005):
    h = initial_h.copy()
    n = len(h)
    target = n/2
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_start
    for it in range(max_iter):
        c = np.correlate(h, 1 - h, mode='full')
        # compute smoothed objective approx:
        # we will compute p (softmax) where larger c get more weight
        max_c = np.max(c)  # for numerical stability subtract max
        # shift by max_c for exponent stability
        exp_c = np.exp(alpha * (c - max_c))
        p = exp_c / exp_c.sum()
        # compute gradient
        mid = n - 1
        w_pos = p[mid:]  # size n
        w_neg = p[:mid][::-1]  # size n
        # conv forward/backward
        conv_f = np.convolve(w_pos[::-1], h, mode='full')
        forward_term = conv_f[mid:mid+n]  # same as conv_f[n-1: n-1+n]
        conv_b = np.convolve(w_neg[::-1], h, mode='full')
        backward_term = conv_b[mid:mid+n]
        grad = 1.0 - (forward_term + backward_term)
        # gradient descent
        h = h - eta * grad
        # project
        h = project_to_feasible(h, target)
        # optional step size adaptation: if it % 200 == 0 increase alpha
        if (it+1) % 200 == 0:
            alpha = min(alpha * 1.2, 20.0)  # increase gradually
            # maybe also reduce eta
            eta *= 0.9
        # evaluate actual c5 periodically
        if (it+1) % 50 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                # early stopping if under target
                if best_c5 < target_bound:
                    break
    return best_h, best_c5

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to reduce sum: find indices >0
            free = h > 0
            if not np.any(free):
                break
            delta = diff / np.count_nonzero(free)
            h[free] -= delta
        else:
            free = h < 1
            if not np.any(free):
                break
            delta = diff / np.count_nonzero(free)
            h[free] -= delta  # diff negative -> add
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target_sum
    # final adjust
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual to any non-saturated entries
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            add = residual / free.size
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    return h
```

    h = h0.copy()
    n = len(h)
    target_sum = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    alpha = alpha_start
    for it in range(max_iter):
        # maybe time check
        # compute correlation
        c = np.correlate(h, 1 - h, mode='full')
        # smoothed max via softmax
        max_c = np.max(c)
        exp_c = np.exp(alpha * (c - max_c))
        p = exp_c / exp_c.sum()
        # compute gradient
        conv_fwd = np.convolve(p[::-1], h, mode='full')
        forward = conv_fwd[mid:mid + n]
        conv_bwd = np.convolve(p, h, mode='full')
        backward = conv_bwd[mid:mid + n]
        grad = 1.0 - (forward + backward)

def _project_to_feasible(v: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Clip to [0,1] and adjust by a common shift so that sum equals target."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= tol:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)

        else:

def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))

            h_opt, c5_opt = _local_refine(h_opt, attempts=15, delta=0.012)

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                if best_c5 < target_bound:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # final polishing on the best candidate found (if any)
    if best_h is not None:
        if time_left() > 2.0:
            extra_iter = int(min(3000, max(500, time_left() / 0.001)))
            best_h, best_c5 = _gradient_descent_h(
                best_h,
                max_iter=extra_iter,
                eta=0.15,
                alpha_start=1.0,
                target_sum=best_n / 2.0,
            )
            best_h, best_c5 = _local_refine(best_h, attempts=30, delta=0.008)

    # fallback to uniform construction if everything else failed
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
