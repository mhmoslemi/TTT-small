def simple_project(h, n):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        # Distribute diff evenly across variables not at bounds.
        # Identify indices where we can add/subtract.
        # We'll adjust by adding -diff/num_free to all free entries (where 0<h<1)
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            h[free] -= diff / free.size
        else:
            # if all at bounds, adjust smallest possible
            # simply scale or adjust at first element not at bound.
            idx = 0 if diff > 0 else -1
            h[idx] -= diff
    # Clip again
    h = np.clip(h, 0.0, 1.0)
    # Possibly small residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # Add to a random free variable
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            idx = free[0]
            h[idx] += residual
        else:
            # adjust any variable within bounds
            idx = np.argmin(h) if residual > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def generate_candidate_binary_smooth(n, rng):

def generate_candidate_fourier(n, rng):

from scipy import optimize

def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # determine indices that can be moved
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final small correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust at a free index if possible
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            idx = free[0]
            h[idx] += residual
        else:
            # adjust any index within bounds
            idx = 0 if residual > 0 else -1
            h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

def _eq_constraint(x):

    return np.sum(x[:n]) - (n / 2.0)

    return x[-1]
