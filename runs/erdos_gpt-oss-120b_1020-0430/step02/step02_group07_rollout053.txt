def compute_corr_fft(h):
   n=len(h)
   f=1-h
   # zero pad to length 2n-1 for convolution
   size = 1 << ((2*n-1).bit_length())
   H = np.fft.rfft(h, size)
   F_rev = np.fft.rfft(f[::-1], size)
   conv = np.fft.irfft(H * F_rev, size)
   conv = conv[:2*n-1]  # trim
   return conv

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.convolve(x, (1 - x)[::-1], mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    f = 1.0 - h
    size = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(H * F, size)[:2*n-1]
    max_corr = np.max(conv)
    return float(max_corr) * (2.0 / n)

def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # Clip just in case
    np.clip(h, 0.0, 1.0, out=h)
    return h

def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    # Transfer small epsilon from i to j
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h0, rng, max_iters=2000, start_step=0.5):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing(h0, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(temp, 1e-12)):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def quick_refine(h, rng):
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        trial = slsqp_optimize(best_h, beta=beta, max_iter=250)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=800, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c

def deep_refine(h, rng):
    n = h.size
    max_iter_slsqp = 200 if n > 200 else 300
    hill_iters = 800 if n > 200 else 1500
    anneal_iters = 2000 if n > 200 else 3000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter_slsqp)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    trial_sa, c_sa = simulated_annealing(best_h, rng, max_iters=anneal_iters,
                                         init_temp=0.03, decay=0.998, scale=0.2)
    if c_sa < best_c:

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the SLSQP surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.convolve(x, (1 - x)[::-1], mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2]."""
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    f = 1.0 - h
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(H * F, size)[: 2 * n - 1]
    max_corr = np.max(conv)
    return float(max_corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def random_feasible_vector(n, rng):
    """Generate a random real vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h0, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h0, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(temp, 1e-12)):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # Repair any tiny drift in the sum

                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
