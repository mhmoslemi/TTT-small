def refine_h(h, max_iter, beta_start=5.0, beta_end=50.0, step_start=0.5, step_end=0.02):
    n = h.size
    h = project_to_feasible(h, n)
    # compute initial L
    v = np.correlate(h, h, mode='full')
    L = - (1.0 / beta_start) * logsumexp(-beta_start * v)
    for i in range(max_iter):
        t = i / (max_iter - 1) if max_iter > 1 else 1.0
        beta = beta_start * ((beta_end / beta_start) ** t)
        step_size = step_start * ((step_end / step_start) ** t)
        # compute correlation v for current h
        v = np.correlate(h, h, mode='full')
        # compute p
        exp_vals = np.exp(-beta * v - np.max(-beta * v))  # to avoid overflow
        p = exp_vals / exp_vals.sum()
        # compute w
        mid = n - 1
        w = np.empty(n, dtype=float)
        w[0] = p[mid]
        for k in range(1, n):
            w[k] = p[mid + k] + p[mid - k]
        # compute gradient
        g = np.zeros_like(h)
        # k=0 contributions
        g += 2.0 * w[0] * h
        # other ks
        for k in range(1, n):
            wk = w[k]
            if wk == 0.0:
                continue
            # right shift
            g[:n - k] += wk * h[k:]
            # left shift
            g[k:] += wk * h[:n - k]
        # direction
        norm = np.linalg.norm(g)
        if norm == 0:
            continue
        direction = g / norm
        # propose step, line search
        best_h_candidate = h
        best_L = L
        step = step_size
        for ls in range(5):
            h_candidate = project_to_feasible(h + step * direction, n)
            v_cand = np.correlate(h_candidate, h_candidate, mode='full')
            L_cand = - (1.0 / beta) * logsumexp(-beta * v_cand)
            if L_cand > best_L + 1e-12:
                best_h_candidate = h_candidate
                best_L = L_cand
                break
            else:
                step *= 0.5
        h = best_h_candidate
        L = best_L
    return h

def _refine_h(h, max_iter, beta_start=5.0, beta_end=50.0, step_start=0.5, step_end=0.02):
    n = h.size
    h = _project_to_feasible(h, n)
    # compute initial L (for beta_start)
    v = np.correlate(h, h, mode='full')
    L = - (1.0 / beta_start) * logsumexp(-beta_start * v)
    for i in range(max_iter):
        # compute schedule fractions
        if max_iter > 1:
            t = i / (max_iter - 1)
        else:
            t = 1.0
        beta = beta_start * ((beta_end / beta_start) ** t)
        step = step_start * ((step_end / step_start) ** t)
        # compute correlation v for current h
        v = np.correlate(h, h, mode='full')
        # compute p
        # For numeric stability subtract max
        exparg = -beta * v
        maxarg = np.max(exparg)
        exp_vals = np.exp(exparg - maxarg)
        p = exp_vals / np.sum(exp_vals)
        # w computation
        mid = n - 1
        w = np.empty(n, dtype=float)
        w[0] = p[mid]
        for k in range(1, n):
            w[k] = p[mid + k] + p[mid - k]
        # gradient
        g = np.zeros_like(h)
        # k=0
        wk0 = w[0]
        if wk0 != 0.0:
            g += 2.0 * wk0 * h
        # other ks
        for k in range(1, n):
            wk = w[k]
            if wk == 0.0:
                continue
            # right shift
            g[:n - k] += wk * h[k:]
            # left shift
            g[k:] += wk * h[:n - k]
        norm = np.linalg.norm(g)
        if norm == 0.0:
            continue
        direction = g / norm
        # line search
        best_h_candidate = h
        best_L_candidate = L
        step_local = step

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

            start_h = 0.5 * (start_h + start_h[::-1])
            start_h = _project_to_feasible(start_h, nxt)
            refined = _refine_h(start_h, max_iter=1200)
            refined = 0.5 * (refined + refined[::-1])
            refined = _project_to_feasible(refined, nxt)
            _maybe_update(refined, nxt)
        cur_n = nxt

    # final clean‑up
    if best_h is not None:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_h = 0.5 * (best_h + best_h[::-1])
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _compute_c5(best_h)
        best_n = best_h.size
    else:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
