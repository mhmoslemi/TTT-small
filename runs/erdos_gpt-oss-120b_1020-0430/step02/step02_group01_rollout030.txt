```python
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def random_binary_vector(n, rng):

    h[ones_idx] = 1.0

def binary_anneal(n, rng, time_left, initial_h=None):
    # if initial provided, start from there, else random binary
    if initial_h is not None:
        h = initial_h.copy()
    else:
        h = random_binary_vector(n, rng)
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()

    # temperature schedule
    T0 = 0.001
    # precompute indices list for ones and zeros
    for it in range(max_iters):
        # time check
        if time.time() - start > time_left:
            break
        # Update temperature
        t = T0 * (1 - it / max_iters)
        # Choose random swap indices
        idx_one = np.where(h > 0.5)[0]
        idx_zero = np.where(h < 0.5)[0]

        h[i] = 0.0
        h[j] = 1.0

                best_h = h.copy()

            h[i] = 1.0
            h[j] = 0.0
```

```
def project_onto_sum_and_bounds(h, target):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Ensure sum = target via water-filling style:
    diff = target - h.sum()
    # Indices not at bounds
    free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    # If free empty, just distribute to all with slack? not needed
    # We'll loop distributing diff equally across free, saturating as needed
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if free.size == 0:
            # distribute to all; but update diff later.
            free = np.arange(h.size)
        adjust = diff / free.size
        # apply adjust
        h[free] += adjust
        # clip and recompute free where at bounds now
        # keep track of overshoot
        # For each over the bound, set to bound and fix diff
        over_upper = h > 1.0
        under_lower = h < 0.0
        if np.any(over_upper):
            excess = h[over_upper] - 1.0
            diff -= excess.sum()
            h[over_upper] = 1.0
        if np.any(under_lower):
            excess = h[under_lower]  # negative
            diff += excess.sum()
            h[under_lower] = 0.0
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    # final small correction via random distribution if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, h.size)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h
```

```
def gradient_descent_refine(h, target, rng, max_iters=2000, step0=0.5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        # compute gradient as described
        g = np.zeros_like(h)
        if k >= 0:
            # shift right
            g[:n - k] += 1 - h[k:]
            g[k:] -= h[:n - k]
        else:
            # negative shift -> left
            kk = -k
            g[kk:] += 1 - h[:n - kk]
            g[:n - kk] -= h[kk:]
        # update
        h = h - step * g
        # projection
        h = project_onto_sum_and_bounds(h, target)
        # compute current c
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # decay step
        step *= 0.9995
    return best_h, best_c
```

def slsqp_optimize(x0, beta, rng, max_iter=250):
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # stable log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=[cons],
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum precisely
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple random redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- utility functions ---------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def project_to_sum_and_bounds(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(h) == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple random redistribution; fast enough for n ≤ 200
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ------------------- binary annealing ---------------------------------

def binary_anneal(n: int,
                 rng: np.random.Generator,
                 time_budget: float,
                 init_h: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on binary vectors (0/1) with exactly n/2 ones.
    Returns the best vector found and its C5 value.
    """
    if init_h is not None:
        h = init_h.copy()
    else:
        h = random_binary_vector(n, rng)

    best_h = h.copy()

# ------------------- continuous refinement -----------------------------

def slsqp_optimize(x0: np.ndarray,
                   beta: float,
                   rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """SLSQP on the soft‑max surrogate for the max‑overlap objective."""
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction to satisfy the equality exactly
    x = project_to_sum_and_bounds(x, target, rng)
    return x


def gradient_descent_refine(
    h0: np.ndarray,
    target: float,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent targeting the worst shift."""
    n = h0.size
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h, best_c = h.copy(), c_cur
    step = step0

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)          # shift giving the max
        k = idx

        g = np.zeros_like(h)

        if k >= 0:
            m = n - k
            g[:m] += 1.0 - h[k : k + m]          # derivative w.r.t. h_i
            g[k : k + m] -= h[:m]               # derivative w.r.t. h_{i+k}
        else:
            kk = -k
            m = n - kk
            g[kk : kk + m] += 1.0 - h[:m]        # i is the larger index
            g[:m] -= h[kk : kk + m]             # j is the smaller index

        h_new = h - step * g
        h_new = project_to_sum_and_bounds(h_new, target, rng)
        c_new = compute_c5(h_new)

        if c_new < c_cur:
            h, c_cur = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()

        step *= 0.9995  # slow decay

    return best_h, best_c

        best_n = 120
        best_h = np.full(best_n, 0.5, dtype=np.float64)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    if abs(best_h.sum() - target) > 1e-12:
        best_h = project_to_sum_and_bounds(best_h, target, rng)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
