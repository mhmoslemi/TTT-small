import time
import numpy as np
from scipy import optimize

# ---------- Feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-10, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        for i in range(n):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- Objective ----------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- SLSQP helpers ----------
def _ineq_constraint(x: np.ndarray):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Local refinement via SLSQP."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    h_opt = _project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- Utility ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _random_binary(n: int) -> np.ndarray:
    h = np.zeros(n)
    h[np.random.choice(n, n // 2, replace=False)] = 1.0
    return h


def _crossover(p1: np.ndarray, p2: np.ndarray) -> np.ndarray:
    """Uniform crossover."""
    mask = np.random.rand(p1.size) < 0.5
    return np.where(mask, p1, p2)


def _tournament_select(fitness, k: int = 3):
    idx = np.random.choice(len(fitness), k, replace=False)
    return idx[int(np.argmin([fitness[i] for i in idx]))]


# ---------- Main optimisation ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left(): return budget_s - (time.time() - start)

    target = 0.3808

    # parent construction if available
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = parent_h.size
    except NameError:
        parent_h = None
        parent_n = None

    # set of grid sizes to explore
    candidates = []
    if parent_h is not None:
        candidates.append(parent_n)
    for mul in (1, 2, 3, 4):
        if parent_h is not None:
            n = int(parent_n * mul)
        else:
            n = 64 * mul
        if n <= 400:
            candidates.append(n)
    # handcrafted even sizes
    candidates.extend([80, 96, 112, 128, 144, 160, 192, 224, 256, 320])
    candidates = sorted(set([n for n in candidates if n % 2 == 0 and n <= 400]))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in candidates:
        if time_left() < 5:
            break

        # base vector for this size
        if parent_h is not None and parent_n != n:
            base = _upscale_h(parent_h, n)
        elif parent_h is not None and parent_n == n:
            base = parent_h.copy()
        else:
            base = np.full(n, 0.5)

        # ---------- Initialise population ----------
        pop_size = 30
        population = [base.copy()]
        fitness = [_c5_from_h(base)]
        for _ in range(pop_size - 1):
            h = _random_binary(n) if np.random.rand() < 0.5 else np.random.rand(n)
            h = _project_to_feasible(h, n)
            population.append(h)
            fitness.append(_c5_from_h(h))

        generation = 0
        no_improve = 0
        max_no_improve = 30
        best_idx = int(np.argmin(fitness))
        best_local = population[best_idx].copy()
        best_local_val = fitness[best_idx]

        while time_left() > 1.5 and generation < 300 and no_improve < max_no_improve:
            generation += 1
            sigma = max(0.25 * (0.97 ** generation), 0.005)

            # elitism
            order = np.argsort(fitness)
            new_pop = [population[i].copy() for i in order[:2]]
            new_fit = [fitness[i] for i in order[:2]]

            while len(new_pop) < pop_size:
                i1 = _tournament_select(fitness, k=3)
                i2 = _tournament_select(fitness, k=3)
                child = _crossover(population[i1], population[i2]) if np.random.rand() < 0.7 else population[i1].copy()
                if np.random.rand() < 0.4:
                    child += sigma * np.random.randn(n)
                child = _project_to_feasible(child, n)
                new_pop.append(child)
                new_fit.append(_c5_from_h(child))

            population, fitness = new_pop, new_fit

            cur_best = int(np.argmin(fitness))
            cur_val = fitness[cur_best]
            if cur_val < best_local_val - 1e-7:
                best_local_val = cur_val
                best_local = population[cur_best].copy()
                no_improve = 0
            else:
                no_improve += 1

            # occasional local SLSQP refinement of the current best
            if generation % 20 == 0:
                refined_h, refined_c = _solve_h(best_local, max_iter=300)
                if refined_c < best_local_val:
                    best_local_val = refined_c
                    best_local = refined_h.copy()
                    # replace a random individual
                    ridx = np.random.randint(pop_size)
                    population[ridx] = refined_h
                    fitness[ridx] = refined_c

            if best_local_val < target:
                break

        # final refinement with longer SLSQP runs
        final_h, final_c = _solve_h(best_local, max_iter=800)
        if final_c < best_local_val:
            best_local_val = final_c
            final_h = final_h

        # a few extra random restarts around the refined solution
        extra_attempts = 5
        for _ in range(extra_attempts):
            if time_left() < 1.0:
                break
            pert = final_h + 0.03 * np.random.randn(n)
            pert = _project_to_feasible(pert, n)
            h2, c2 = _solve_h(pert, max_iter=600)
            if c2 < best_local_val:
                best_local_val = c2
                final_h = h2

        if best_local_val < best_c5:
            best_c5 = best_local_val
            best_h = final_h.copy()
            best_n = n
            if best_c5 < target:
                break

    # fallback if nothing found
    if best_h is None:
        n = parent_n if parent_n is not None else 96
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
