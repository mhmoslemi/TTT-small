def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # Use simple loop
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        # choose random index that can accept diff
        if diff > 0:
            # need add mass: choose i where h[i] < 1
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
            diff -= add
        else:
            i = rng.integers(0, n)
            add = max(diff, -h[i])  # note diff negative
            if add >= 0:
                continue
            h[i] += add
            diff -= add
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

def perturb_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i: j += 1  # ensure j != i
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    # sometimes subtract; but we need random sign? We always moving from i to j (decrease i, increase j)
    # To allow both directions, we randomly choose direction.
    if rng.random() < 0.5:
        # move from i to j as above
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    else:
        # reverse direction: move from j to i
        max_eps_rev = min(h[j], 1.0 - h[i])
        if max_eps_rev <= 0:
            return h.copy()
        eps_rev = rng.random() * max_eps_rev * scale
        new_h = h.copy()
        new_h[j] -= eps_rev
        new_h[i] += eps_rev
    return new_h

    start_time = time.time()

    best_h = None
    best_c = float('inf')
    best_n = None

    # if initial_h_values present:
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)

import time
import numpy as np
import scipy.optimize

# ---------- helper functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and then adjust to hit the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- SLSQP on smooth surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp safely
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = scipy.optimize.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- perturbations ----------
def perturb_transfer(h, rng, scale=0.2):
    """Move a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    # random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[src] -= eps
    new_h[dst] += eps
    return new_h


# ---------- annealing search ----------
def anneal_search(n, rng, deadline):
    """Simple simulated‑annealing on the true C5."""
    # start from a random binary vector
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # occasional small noise
    if rng.random() < 0.3:
        h = np.clip(h + rng.random(n) * 0.05, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0, rng)

    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    T, decay = 0.5, 0.9997
    it = 0
    max_it = 20000
    while time.time() < deadline and it < max_it:
        cand = perturb_transfer(h, rng, scale=0.2)
        cand = enforce_sum(cand, n / 2.0, rng)
        cand_c = compute_c5(cand)
        if cand_c < c or rng.random() < np.exp(-(cand_c - c) / (T + 1e-12)):
            h, c = cand, cand_c
            if c < best_c:
                best_c, best_h = c, h.copy()
        T *= decay
        it += 1
    return best_h, best_c


def refine_with_slsqp(h, betas, deadline):
    """Run a short SLSQP ladder on the smooth surrogate."""
    for b in betas:
        if time.time() >= deadline:
            break
        try:
            h = slsqp_optimize(h, b, max_iter=300)
        except Exception:
            continue
    return h
