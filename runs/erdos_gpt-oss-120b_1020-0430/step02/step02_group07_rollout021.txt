def greedy_reduce(h, rng, max_swaps=5000):
    n = h.size
    best_c = compute_c5(h)
    swaps = 0
    while swaps < max_swaps:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        if max_val == 0:
            break
        shift = max_idx - (n - 1)
        # try to find swap for this shift
        swap = find_swap_for_shift(h, shift, rng)

            # cannot improve via simple swap; break
            break
        i, j = swap
        # execute swap
        h[i], h[j] = 0, 1
        swaps += 1
        # compute new C5 quickly
        new_c = compute_c5(h)
        if new_c < best_c:
            best_c = new_c
        # continue loop
    return h, best_c

def simulated_annealing_swap(h_start, rng, max_iters=20000, init_temp=0.02, decay=0.999):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    n = h.size
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    for it in range(max_iters):
        # pick random i from ones_idx, j from zeros_idx
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = 0, 1
        # update ones_idx, zeros_idx: we can just recompute after each swap maybe slower but okay given n small.
        new_c = compute_c5(h)
        # accept
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            cur_c = new_c
            # update indices
            # need to reflect swap: i becomes zero, j becomes one
            # Update ones_idx and zeros_idx by swapping i and j
            # Let's just recompute each iteration for simplicity (cost O(n) each, fine)
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
        else:
            # reject: revert swap
            h[i], h[j] = 1, 0
        temp *= decay
        # optional early break if temp low and no improvement for many steps
    return best_h, best_c

def perturb_h_cont(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for the surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max approximation of the overlap using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2]."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, ones, rng):
    """Generate a binary vector of length n with exactly `ones` ones."""
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1
    return h


def find_swap_for_shift(h, shift, rng):
    """
    For a given shift (may be negative) locate a beneficial 1↔0 swap.
    Returns (pos_i, pos_j) where h[pos_i] will become 0 and h[pos_j] will become 1.
    If no improving swap exists, returns None.
    """
    n = h.size
    if shift >= 0:
        a = h[: n - shift]            # h[i]
        b = h[shift:]                  # h[i+shift]
        bad_idxs = np.where((a == 1) & (b == 0))[0]
        good_idxs = np.where((a == 0) & (b == 1))[0]
        if bad_idxs.size == 0 or good_idxs.size == 0:
            return None
        i = rng.choice(bad_idxs)               # position in a
        j = rng.choice(good_idxs)               # position in a
        pos_i = i
        pos_j = j + shift
    else:
        s = -shift
        a = h[s:]                       # h[i+s]
        b = h[: n - s]                  # h[i]
        bad_idxs = np.where((a == 1) & (b == 0))[0]
        good_idxs = np.where((a == 0) & (b == 1))[0]
        if bad_idxs.size == 0 or good_idxs.size == 0:
            return None
        i = rng.choice(bad_idxs)
        j = rng.choice(good_idxs)
        pos_i = i + s
        pos_j = j
    return int(pos_i), int(pos_j)


def greedy_reduce(h, rng, max_swaps=2000):
    """
    Deterministic reduction: repeatedly swap a 1 and a 0 that lower the
    presently maximal overlap term. Stops when no such swap exists
    or after `max_swaps` exchanges.
    """
    n = h.size
    best_c = compute_c5(h)
    swaps = 0
    while swaps < max_swaps:
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        if max_val <= 0:
            break
        shift = max_idx - (n - 1)
        sw = find_swap_for_shift(h, shift, rng)
        if sw is None:
            break
        i, j = sw
        h[i], h[j] = 0, 1
        swaps += 1
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c = cur_c
    return h, best_c


def simulated_annealing_swap(h_start, rng, max_iters=3000,
                             init_temp=0.02, decay=0.9995):
    """
    Stochastic search using swaps of a 1 and a 0.
    Accepts uphill moves with probability exp((old‑new)/T).
    """
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    n = h.size

    for _ in range(max_iters):
        # pick random 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new_c = compute_c5(h)

        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()

        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            cur_c = new_c
        else:
            # revert
            h[i], h[j] = 1, 0
        temp *= decay
    return best_h, best_c


def perturb_h_cont(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another, preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_cont(h_start, rng, max_iters=1000, start_step=0.3):
    """Simple hill‑climb on the continuous surrogate (no surrogate, exact C5)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h_cont(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    # enforce exact sum after climbing
    target = h.size / 2.0
    best_h = enforce_sum(best_h, target, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """One SLSQP optimisation of the smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny repair of the linear equality
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def continuous_refine(h_float, rng, n, start_time, budget_s):
    """Intensive continuous refinement using SLSQP and a small hill‑climb."""
    best_h = h_float.copy()
    best_c = compute_c5(best_h)
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 2000.0, 5000.0)
    for beta in betas:
        if time.time() - start_time > budget_s * 0.96:
            break
        trial = slsqp_optimize(best_h, beta, rng, max_iter=300)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    # final local hill‑climb
    best_h, best_c = hill_climb_cont(best_h, rng, max_iters=1200, start_step=0.3)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Choose a modest resolution (fast to evaluate) and initialise best.
    # ----------------------------------------------------------------------
    n = 200                              # even number → integer target sum
    target_sum = n // 2                  # exact number of ones for binary vectors
    best_h = None
    best_c = np.inf

    # If a parent solution is provided, down‑sample it and use as a starter.
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_parent = np.interp(x_new, x_old, parent)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = enforce_sum(h_parent, target_sum, rng)
        c_parent = compute_c5(h_parent)
        best_h, best_c = h_parent, c_parent

    # ----------------------------------------------------------------------
    # Main binary search loop (random restarts + greedy + SA).
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.78:
        # random binary start
        h_bin = random_binary_vector(n, target_sum, rng)
        # deterministic greedy reduction
        h_g, c_g = greedy_reduce(h_bin.copy(), rng, max_swaps=1500)
        if c_g < best_c:
            best_h, best_c = h_g.copy(), c_g
        # short simulated annealing from the greedy result
        h_sa, c_sa = simulated_annealing_swap(
            h_g.copy(), rng, max_iters=2000,
            init_temp=0.025, decay=0.9992
        )
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

    # ----------------------------------------------------------------------
    # Late‑stage continuous refinement (uses the smooth surrogate).
    # ----------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.96:
        h_cont, c_cont = continuous_refine(
            best_h.astype(float), rng, n, start_time, budget_s
        )
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # ----------------------------------------------------------------------
    # Final feasibility enforcement and output.
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
