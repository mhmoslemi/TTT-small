def project_to_feasible(h, target, max_iter=1000):
    h = np.clip(h,0.0,1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # too much sum, reduce from those >0
            idx = np.where(h > 0)[0]
            if idx.size ==0: break
            delta = diff / idx.size
            h[idx] -= delta
        else:
            idx = np.where(h < 1)[0]
            if idx.size ==0: break
            delta = diff / idx.size
            h[idx] -= delta  # diff negative => delta negative => add
        h = np.clip(h,0.0,1.0)
    # final correction:
    s = h.sum()
    residual = target - s
    if abs(residual) > 1e-12:
        # distribute residual greedily
        for i in range(len(h)):
            if residual > 0 and h[i] < 1.0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0.0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h

def compute_c5(h):
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(conv)
    return max_corr * 2.0 / n

def slsqp_optimize(h0, max_iter=2000):
    n = len(h0)
    target = n/2.0

    # project initial
    h0 = project_to_feasible(h0.copy(), target)

    # compute initial t bound
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))  # not scaled
    # variable x = [h..., t]
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # constraints functions
    def ineq(x):
        h = x[:n]
        t = x[-1]
        # compute correlation unscaled
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

    def eq(x):
        h = x[:n]
        return h.sum() - target

    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]
    opt = {'maxiter': max_iter, 'ftol':1e-9, 'disp':False}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=cons, options=opt)
    if not res.success:
        # fallback: return projected h0
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = project_to_feasible(h_opt, target)
    return h_opt

import time
import numpy as np
from scipy import optimize

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _proj_feasible(vec: np.ndarray, target: float, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] then enforce sum(vec)=target by water‑filling."""
    vec = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = vec.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            free = np.where(vec > 0)[0]
        else:
            free = np.where(vec < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        vec[free] -= delta
        vec = np.clip(vec, 0.0, 1.0)
    # tiny final fix
    resid = target - vec.sum()
    if abs(resid) > 1e-12:
        for i in range(vec.shape[0]):
            if resid > 0 and vec[i] < 1.0:
                add = min(resid, 1.0 - vec[i])
                vec[i] += add
                resid -= add
            elif resid < 0 and vec[i] > 0.0:
                sub = min(-resid, vec[i])
                vec[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(vec, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """C5 = max_k  ∫ h(x)(1‑h(x+k))dx  (discrete approximation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _upscale(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step onto a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


# -------------------------------------------------------------
# Local deterministic optimisation (SLSQP)
# -------------------------------------------------------------
def _slsqp_opt(h0: np.ndarray, max_iter: int = 1500) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    h0 = _proj_feasible(h0.copy(), target)

    # initial bound on t (unscaled correlation)
    t0 = np.correlate(h0, 1.0 - h0, mode="full").max()
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return x[:n].sum() - target

    cons = [{"type": "ineq", "fun": _ineq},
            {"type": "eq",   "fun": _eq}]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0,
                            method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _proj_feasible(h_opt, target)
    return h_opt

# Simple hill‑climb (strictly improving moves)
# -------------------------------------------------------------
def _hill_climb(h: np.ndarray, steps: int, deadline: float) -> tuple:
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_value(best_h)
    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        eps = np.random.rand() * min(max_down, max_up) * 0.5
        if eps == 0:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_value(cand)
        if c < best_c:
            best_h, best_c = cand, c
    best_h = _proj_feasible(best_h, target)
    return best_h, best_c

# Simulated annealing with epsilon‑exchanges and occasional swaps
# -------------------------------------------------------------
def _simulated_anneal(start_h: np.ndarray, max_iter: int,
                      time_limit: float,
                      start_T: float = 0.01,
                      end_T: float = 1e-5) -> tuple:
    n = start_h.size
    target = n / 2.0
    h = start_h.copy()
    cur_c = _c5_value(h)
    best_h, best_c = h.copy(), cur_c

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # exponential cooling
        T = start_T * ((end_T / start_T) ** (it / max_iter))
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # 30 % chance to try a pure swap (preserves sum exactly)
        if np.random.rand() < 0.3:
            cand = h.copy()
            cand[i], cand[j] = cand[j], cand[i]
            c = _c5_value(cand)
            delta = c - cur_c
            if delta < 0 or np.exp(-delta / T) > np.random.rand():
                h, cur_c = cand, c
                if c < best_c:
                    best_h, best_c = cand, c
            continue

        # epsilon exchange (i loses, j gains)
        max_down = h[i]
        max_up = 1.0 - h[j]
        if max_down <= 0 or max_up <= 0:
            continue
        eps = (np.random.rand() - 0.5) * 2.0 * min(max_down, max_up)
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_value(cand)
        delta = c - cur_c
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h, cur_c = cand, c
            if c < best_c:
                best_h, best_c = cand, c
    best_h = _proj_feasible(best_h, target)
    return best_h, best_c

# Main entry point
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # obtain any supplied construction
    try:
        GLOBAL_INIT = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INIT = None

    # candidate grid sizes (favor multiples of 8, keep under 300 points)
    cand = [80, 96, 112, 128, 160, 192, 224, 256, 320]
    if GLOBAL_INIT is not None:
        cand.append(GLOBAL_INIT.size)
    cand = sorted(set(cand))

    best_h = None
    best_c = float("inf")
    best_n = None

    for n in cand:
        if time_left() < 5.0:
            break

        # build initial vector for this resolution
        if GLOBAL_INIT is not None:
            if GLOBAL_INIT.size == n:
                h0 = GLOBAL_INIT.copy()
            else:
                h0 = _upscale(GLOBAL_INIT, n)
        else:
            h0 = np.full(n, 0.5)

        h0 = _proj_feasible(h0, n / 2.0)

        h_opt, c_opt = _optimise_grid(n, h0, time_left)

        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, n

        if best_c < 0.3808:
            break

    # fallback (should never happen)
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c = _c5_value(best_h)

    # final safety projection
    best_h = _proj_feasible(np.array(best_h, copy=True), best_n / 2.0)
    best_c = _c5_value(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
