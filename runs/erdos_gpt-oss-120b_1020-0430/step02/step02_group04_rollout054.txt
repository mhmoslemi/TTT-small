def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta, steps in zip(beta_sched, steps_per_beta):
        step = 0.5
        for _ in range(steps):
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * (2.0 / n)  # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                break
            w /= w_sum

            # gradient of smooth max (raw)
            g_raw = np.ones(n)
            # iterate over shift indices
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    sl = n - k
                    if sl > 0:
                        g_raw[:sl] -= wk * h[k:]
                        g_raw[k:] -= wk * h[:sl]
                else:
                    kk = -k
                    sl = n - kk
                    if sl > 0:
                        g_raw[kk:] -= wk * h[:sl]
                        g_raw[:sl] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n/2.0)
            c_new = compute_c5(new_h)

            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
                step = 0.5  # maybe reset
            else:
                if rng.random() < 0.01:
                    h = new_h
                step *= 0.9995
    return best_h, best_c

def subgrad_refine(h0, max_iter, step0, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        # gradient
        g = np.ones(n)
        if k >= 0:
            sl = n - k
            if sl > 0:
                g[:sl] -= h[k:]
                g[k:] -= h[:sl]
        else:
            kk = -k
            sl = n - kk
            if sl > 0:
                g[kk:] -= h[:sl]
                g[:sl] -= h[kk:]
        new_h = h - step * g
        new_h = project_onto_box_sum(new_h, n/2.0)
        # enforce symmetry
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, n/2.0)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
            step = step0
        else:
            if rng.random() < 0.01:
                h = new_h

    return best_h, best_c

import time
import numpy as np

# ----- basic utilities -----
def compute_c5(h):
    """Maximum overlap integral for discrete step function h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Exact projection of v onto [lower,upper] with sum == target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible vector (0≤h≤1, sum = n/2)."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    full = np.concatenate([x, x[::-1]])
    return full


# ----- perturbation / hill climbing -----
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ----- smooth‑max gradient descent -----
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta, steps in zip(beta_sched, steps_per_beta):
        step = 0.5
        for _ in range(steps):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                break
            w /= w_sum

            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    sl = n - k
                    if sl > 0:
                        g_raw[:sl] -= wk * h[k:]
                        g_raw[k:] -= wk * h[:sl]
                else:
                    kk = -k
                    sl = n - kk
                    if sl > 0:
                        g_raw[kk:] -= wk * h[:sl]
                        g_raw[:sl] -= wk * h[kk:]

            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            # enforce symmetry each iteration
            new_h = (new_h + new_h[::-1]) * 0.5
            new_h = project_onto_box_sum(new_h, n / 2.0)

            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
                step = 0.5
            else:
                if rng.random() < 0.01:
                    h = new_h
                step *= 0.9995
    return best_h, best_c


# ----- hard‑max subgradient refinement -----
def subgrad_refine(h0, max_iter, step0, rng):
    """Projected subgradient descent on the true max objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)

        g = np.ones(n)
        if k >= 0:
            sl = n - k
            if sl > 0:
                g[:sl] -= h[k:]
                g[k:] -= h[:sl]
        else:
            kk = -k
            sl = n - kk
            if sl > 0:
                g[kk:] -= h[:sl]
                g[:sl] -= h[kk:]

        new_h = h - step * g
        new_h = project_onto_box_sum(new_h, n / 2.0)
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, n / 2.0)

        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
            step = step0
        else:
            if rng.random() < 0.01:
                h = new_h
            step *= 0.995
    return best_h, best_c


# ----- main driver -----
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- problem size ---
    n = 240                     # more points give finer control

    # --- initial feasible vector ---
    if "initial_h_values" in globals():
        old_h = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, old_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, old_h, left=0.0, right=0.0)
    else:
        h0 = random_feasible_symmetric(n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_sum(h0, n / 2.0)
    h0 = (h0 + h0[::-1]) * 0.5
    h0 = project_onto_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # schedule for smooth‑max descent
    beta_sched = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    steps_per_beta = [200, 200, 400, 800, 1200, 1600]   # total ≈ 4400 steps

    # main stochastic search
    while time.time() - start_time < budget_s * 0.95:
        # start from a blend of current best and a fresh random guess
        if rng.random() < 0.3:
            start_h = (best_h + random_feasible_symmetric(n, rng)) * 0.5
        else:
            start_h = random_feasible_symmetric(n, rng)

        start_h = (start_h + start_h[::-1]) * 0.5
        start_h = project_onto_box_sum(start_h, n / 2.0)

        # smooth‑max gradient descent
        h_opt, c_opt = grad_descent_optimize(start_h, beta_sched, steps_per_beta, rng)

        # hard‑max subgradient refinement
        h_opt, c_opt = subgrad_refine(h_opt, max_iter=2000, step0=0.4, rng=rng)

        # occasional short hill‑climb polishing
        if rng.random() < 0.2:
            h_tmp, c_tmp = hill_climb(h_opt, rng, max_iters=1000, start_step=0.3)
            if c_tmp < c_opt:
                h_opt, c_opt = h_tmp, c_tmp

        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # final polishing (if time permits)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10.0:
        best_h, best_c = hill_climb(best_h, rng, max_iters=3000, start_step=0.2)

    # clean up and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
