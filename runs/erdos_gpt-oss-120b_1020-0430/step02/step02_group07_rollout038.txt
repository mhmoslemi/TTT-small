def refine_solution(h, rng, start_time, budget_s):
    deadline = start_time + budget_s
    best_h, best_c = h.copy(), compute_c5(h)
    # define schedule of betas
    betas = [5,20,80,200,500,1000,2000]
    # loop through betas
    for beta in betas:
        if time.time() >= deadline:
            break
        try:
            h_slsqp = slsqp_optimize(best_h, beta=beta, max_iter=300)
        except Exception:
            continue
        c = compute_c5(h_slsqp)
        if c < best_c:
            best_h, best_c = h_slsqp, c
        # hill climb
        max_hc_iters = 2000 if best_h.size <= 200 else 3000
        hc_step = 0.4
        # we may do multiple hill climbs
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=max_hc_iters, start_step=hc_step, time_deadline=deadline)

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5, time_deadline=None):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        if time_deadline is not None and time.time() >= time_deadline:
            break
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # optionally adjust step size gradually
        if (it+1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3, time_deadline=None):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        if time_deadline is not None and time.time() >= time_deadline:
            break
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

CURRENT_BETA = None

    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    new_h[i] -= eps
    new_h[j] += eps
    return new_h

    h = h_start.copy()
    best_h = h.copy()

    step = start_step
    for it in range(max_iters):

            break

    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        if time_deadline is not None and time.time() >= time_deadline:
            break
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up

def refine_solution(h, rng, sub_budget):
    """
    Heavily refine a candidate, respecting a time budget (seconds).
    Returns the best vector found and its C5.
    """
    start_local = time.time()
    deadline = start_local + sub_budget
    best_h = h.copy()
    best_c = compute_c5(best_h)
    betas = [5, 20, 80, 200, 500, 1000, 2000]

    for beta in betas:
        if time.time() >= deadline:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=300)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        hc_iters = 2500 if best_h.size <= 200 else 3500
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hc_iters,
                                  start_step=0.4, time_deadline=deadline)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h

    if time.time() < deadline:
        sa_iters = 6000 if best_h.size <= 200 else 10000
        trial_sa, c_sa = simulated_annealing(
            best_h, rng, max_iters=sa_iters,
            init_temp=0.05, decay=0.9995, scale=0.35,
            time_deadline=deadline)
        if c_sa < best_c:
            best_h, best_c = trial_sa, c_sa
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 120
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ------------------------------------------------------------------
    # Phase A – explore with cheap hill climbs
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.45:
            break
        remaining = budget_s * 0.45 - elapsed
        if remaining < 1.0:
            break
        if rng.random() < 0.4:
            h_start = random_feasible_vector(best_h.size, rng)
        else:
            h_start = perturb_h(best_h, rng, scale=0.3)
        cand_h, cand_c = hill_climb(
            h_start, rng, max_iters=1200, start_step=0.4,
            time_deadline=time.time() + remaining)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ------------------------------------------------------------------
    # Phase B – deep refinement on the current best
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0:
        sub = min(remaining * 0.6, 40.0)
        cand_h, cand_c = refine_solution(best_h, rng, sub_budget=sub)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ------------------------------------------------------------------
    # Phase C – optional up‑sampling and refinement
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.90:
            break
        if best_h.size * 2 > 800:   # keep under 1 k points
            break
        remain = budget_s - elapsed
        if remain < 5.0:
            break
        h_up = upsample_h(best_h, factor=2, rng=rng)
        sub = min(remain * 0.5, 30.0)
        cand_h, cand_c = refine_solution(h_up, rng, sub_budget=sub)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        # occasionally try to down‑sample back to the original size
        if best_h.size > n0 and best_h.size % n0 == 0:
            factor = best_h.size // n0
            resh = best_h.reshape(n0, factor)
            h_down = resh.mean(axis=1)
            h_down = enforce_sum(h_down, n0 / 2.0, rng)
            sub2 = min(remain * 0.3, 20.0)
            cand_h2, cand_c2 = refine_solution(h_down, rng, sub_budget=sub2)
            if cand_c2 < best_c:
                best_h, best_c = cand_h2, cand_c2

    # ------------------------------------------------------------------
    # final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
