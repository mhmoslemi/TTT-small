def softmax_obj_and_grad(x, beta):
    n = len(x)
    corr = np.correlate(x, 1 - x, mode='full')
    max_corr = np.max(corr)
    exp_terms = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_terms)
    obj = (max_corr + np.log(sum_exp)) / beta
    w = exp_terms / sum_exp  # shape (2n-1)
    grad = np.zeros_like(x)
    # for each shift s
    for s in range(-(n-1), n):
        p = s + n - 1
        weight = w[p]
        # i range
        if s >= 0:
            i_start = 0
            i_end = n - s  # exclusive
            # i + s valid
            # compute contributions
            i = np.arange(i_start, i_end)
            grad[i] += weight * (1 - x[i + s])
            # negative part:
            grad[i + s] -= weight * x[i]
        else:
            # s < 0, i_start = -s
            i_start = -s
            i_end = n
            i = np.arange(i_start, i_end)
            grad[i] += weight * (1 - x[i + s])  # i+s is valid (since s negative)
            grad[i + s] -= weight * x[i]
    return obj, grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate for max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for a step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random h with 0≤h≤1 and Σh=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of the surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


# ---------- symmetric representation utilities ----------
def full_from_v(v):
    """Build the full length‑2m vector from half‑vector v (reflection symmetry)."""
    m = v.size
    n = m * 2
    full = np.empty(n, dtype=np.float64)
    full[:m] = v
    full[m:] = 1.0 - v[::-1]
    return full


def soft_obj_grad(v, beta):
    """
    Smooth surrogate and its gradient w.r.t. the half‑vector v.
    Uses the same log‑sum‑exp approximation as softmax_obj.
    """
    h = full_from_v(v)
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_terms = np.exp(beta * shifted)
    sum_exp = np.sum(exp_terms)
    obj = (max_corr + np.log(sum_exp)) / beta
    w = exp_terms / sum_exp                     # softmax weights
    grad_h = np.zeros_like(h)

    # vectorised accumulation over shifts
    for s in range(-(n - 1), n):
        weight = w[s + n - 1]
        if weight == 0.0:
            continue
        if s >= 0:
            i_end = n - s
            idx_i = slice(0, i_end)
            idx_j = slice(s, n)
            grad_h[idx_i] += weight * (1.0 - h[idx_j])
            grad_h[idx_j] -= weight * h[idx_i]
        else:
            i_start = -s
            idx_i = slice(i_start, n)
            idx_j = slice(0, n + s)
            grad_h[idx_i] += weight * (1.0 - h[idx_j])
            grad_h[idx_j] -= weight * h[idx_i]

    # chain rule: v appears in positions i and its mirror n‑1‑i (with opposite sign)
    grad_v = grad_h[:v.size] - grad_h[::-1][:v.size]
    return obj, grad_v


def sym_optimize(v0, beta_sched, rng, deadline):
    """Continuation with L‑BFGS‑B on the symmetric parametrisation."""
    v = np.clip(v0, 0.0, 1.0)
    for beta in beta_sched:
        if time.time() > deadline:
            break

        def fun(x):
            return soft_obj_grad(x, beta)

        res = minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            bounds=[(0.0, 1.0)] * v.size,
            options={"maxiter": 400, "ftol": 1e-12},
        )
        v = np.clip(res.x, 0.0, 1.0)

        # occasional tiny jitter to escape shallow plateaus
        if rng.random() < 0.1:
            v += rng.uniform(-0.02, 0.02, size=v.shape)
            v = np.clip(v, 0.0, 1.0)

    return v
