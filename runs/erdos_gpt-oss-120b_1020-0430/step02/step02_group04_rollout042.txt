import time
import numpy as np

# ---------- core utilities ----------
def compute_c5(h):
    """Discrete C5 for a step function (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto [lo,hi] while forcing sum(v)=target."""
    a, b = lo - np.max(v) - 1.0, hi - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (a + b) * 0.5
        proj = np.clip(v + lam, lo, hi)
        s = proj.sum()
        if s < target:
            a = lam
        else:
            b = lam
    proj = np.clip(v + (a + b) * 0.5, lo, hi)
    # tiny correction to hit the target exactly
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > lo + 1e-12) & (proj < hi - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lo, hi)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible vector (0≤h≤1, Σh=n/2)."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    h = np.concatenate([x, x[::-1]])
    return h


def perturb_h(h, rng, scale=0.3):
    """Move a random amount from one coordinate to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


def simple_hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb with decreasing step size."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)

        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # decay step occasionally
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.75, 1e-4)
    return best_h, best_c

        # heuristic: ~200 steps per second for N≈180
        long_steps = int(min(200000, max(20000, remaining * 180)))
        temp0 = max(best_c * 0.02, 0.008)      # a small but non‑zero temperature
        decay = 0.9995
        h_sa, c_sa = simulated_annealing(best_h, rng, long_steps, temp0, decay)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ---- occasional restarts with shorter SA -----------------------------
    while time.time() - start_time < budget_s * 0.95:
        # new start: either fresh random or blended with current best
        if rng.random() < 0.25:
            h_start = random_feasible_symmetric(N, rng)
        else:
            h_start = (best_h + random_feasible_symmetric(N, rng)) * 0.5
            h_start = (h_start + h_start[::-1]) * 0.5
            h_start = project_onto_box_sum(h_start, N / 2.0)

        short_steps = 12000
        temp0 = 0.01
        decay = 0.999
        h_sa, c_sa = simulated_annealing(h_start, rng, short_steps, temp0, decay)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ---- final hill‑climb polish ----------------------------------------
    h_final, c_final = simple_hill_climb(best_h, rng, max_iters=3000, start_step=0.4)
    if c_final < best_c:
        best_h, best_c = h_final, c_final

    # ---- clean up --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
