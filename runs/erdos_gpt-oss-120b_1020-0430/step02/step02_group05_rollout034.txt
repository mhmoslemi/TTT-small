import time
import numpy as np
from scipy import optimize

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} by binary search on λ."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

def _grad_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Subgradient of max‑correlation w.r.t. h for the shift k achieving the max."""
    n = h.size
    if k >= 0:
        left = np.concatenate((np.zeros(k, dtype=h.dtype), h[:n - k]))
        right = np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
    else:
        kp = -k
        left = np.concatenate((h[kp:], np.zeros(kp, dtype=h.dtype)))
        right = np.concatenate((np.zeros(kp, dtype=h.dtype), h[:n - kp]))
    return 1.0 - (left + right)

def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x: np.ndarray) -> float:
    return x[-1]

def _solve_h(start_h: np.ndarray, max_iter: int = 600):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)

def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # parent if available
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = np.inf
    best_n = None
    target = 0.3808

    # candidate sample counts
    cand = []
    if init_h is not None:
        cand.append(init_h.size)
    for n in (96, 128, 160, 192, 224, 256, 320, 384, 448):
        if n <= 500:
            cand.append(n)
    cand = sorted(set(cand))

    for n in cand:
        if time_left() < 8.0:
            break

        # base construction for this n
        if init_h is not None and init_h.size == n:
            base = init_h.copy()
        elif init_h is not None:
            base = _upscale_h(init_h, n)
        else:
            base = _tent_h(n)

        # evaluate base
        c5_base = _c5_from_h(base)
        if c5_base < best_c5:
            best_h, best_c5, best_n = base, c5_base, n
            if best_c5 < target:
                break

        # stochastic search
        attempts = 0
        while time_left() > 8.0 and attempts < 200:
            attempts += 1
            r = np.random.rand()
            if r < 0.4:
                start = np.random.rand(n)   # uniform
            elif r < 0.7:
                start = np.zeros(n)
                start[:n // 2] = 1.0
                np.random.shuffle(start)    # binary half‑ones
            else:
                start = base + 0.3 * np.random.randn(n)   # noisy

            start = _project_to_simplex_box(start, n / 2.0)

            # ----- subgradient descent -----
            h = start.copy()
            step0 = 0.6
            max_sgd = 250
            for it in range(max_sgd):
                corr = np.correlate(h, 1.0 - h, mode="full")
                idx = np.argmax(corr)
                k = idx - (n - 1)
                g = _grad_for_shift(h, k)
                step = step0 / np.sqrt(it + 1)
                h = _project_to_simplex_box(h - step * g, n / 2.0)

            # ----- refinement via SLSQP -----
            h_ref, c5_val = _solve_h(h, max_iter=800)

            if c5_val < best_c5:
                best_h, best_c5, best_n = h_ref, c5_val, n
                if best_c5 < target:
                    break
        if best_c5 < target:
            break

    # fallback
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
