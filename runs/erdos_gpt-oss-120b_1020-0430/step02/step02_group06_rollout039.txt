def _sa_search(initial_h, max_iter, time_func):

    h = initial_h.copy()

    T0 = 5e-4

            i = np.random.randint(-shift, n)

        if max_delta <= 0:
            continue

        # use step size fraction alpha decreasing
        alpha = 0.1  # can adapt based on iteration, optional

def _gradient_descent_opt(h, max_iter, time_func):
    n = h.size
    cur_h = h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    step_frac = 0.2
    for it in range(max_iter):
        if not time_func():
            break
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        # compute gradient
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n-shift] += 1 - cur_h[shift:]
            grad[n-shift:] += 1.0
            grad[shift:] -= cur_h[:n-shift]
        else:
            s = -shift
            grad[s:] += 1 - cur_h[:n-s]
            grad[:s] += 1.0
            grad[:n-s] -= cur_h[s:]
        # find indices
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        # only move if gradient difference positive
        if grad[i_max] - grad[i_min] > 1e-12:
            max_delta = min(cur_h[i_max], 1.0 - cur_h[i_min])
            if max_delta > 0:
                delta = max_delta * step_frac
                # move mass
                cur_h[i_max] -= delta
                cur_h[i_min] += delta
                # maybe small floating errors, ensure within bounds
                cur_h = np.clip(cur_h, 0.0, 1.0)
                # recalc c5
                new_c5 = _c5_from_h(cur_h)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = cur_h.copy()
                else:
                    # revert
                    cur_h[i_max] += delta
                    cur_h[i_min] -= delta
        # else optionally random jitter
        if it % 500 == 0:
            # small random perturbations and reproject
            trial = cur_h + np.random.randn(n) * 0.01
            trial = _project_to_feasible(trial, n)
            new_c5 = _c5_from_h(trial)
            if new_c5 < cur_c5:
                cur_h = trial
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur_h.copy()
        # decay step
        step_frac *= 0.9995
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# -----------------------------------------------------------
# Feasibility projection (clip to [0,1] and enforce sum = n/2)
# -----------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9,
                         max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# -----------------------------------------------------------
# C5 evaluation for a discretised step function
# -----------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# -----------------------------------------------------------
# Linear upscale / interpolation of a coarse grid
# -----------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# -----------------------------------------------------------
# SLSQP refinement (same formulation as the reference solution)
# -----------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds,
        constraints=constraints, options=opts,
    )
    if res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# -----------------------------------------------------------
# Small stochastic hill‑climb used after other phases
# -----------------------------------------------------------
def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02) -> np.ndarray:
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


# -----------------------------------------------------------
# Gradient‑guided deterministic descent on the worst shift
# -----------------------------------------------------------
def _gradient_descent_opt(initial_h: np.ndarray, max_iter: int,
                         start_time: float, budget_s: float) -> tuple:
    n = initial_h.size
    cur_h = initial_h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    step_frac = 0.2

    for it in range(max_iter):
        # time guard
        if budget_s - (time.time() - start_time) < 0.5:
            break

        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # gradient of the maximal correlation term
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n - shift] += 1.0 - cur_h[shift:]
            grad[n - shift:] += 1.0
            grad[shift:] -= cur_h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - cur_h[:n - s]
            grad[:s] += 1.0
            grad[:n - s] -= cur_h[s:]

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        if grad[i_max] - grad[i_min] > 1e-12:
            max_delta = min(cur_h[i_max], 1.0 - cur_h[i_min])
            if max_delta > 0:
                delta = max_delta * step_frac
                cur_h[i_max] -= delta
                cur_h[i_min] += delta
                cur_h = np.clip(cur_h, 0.0, 1.0)
                new_c5 = _c5_from_h(cur_h)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = cur_h.copy()
                else:
                    # revert move
                    cur_h[i_max] += delta
                    cur_h[i_min] -= delta

        # occasional random jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            trial = cur_h + np.random.randn(n) * 0.01
            trial = _project_to_feasible(trial, n)
            new_c5 = _c5_from_h(trial)
            if new_c5 < cur_c5:
                cur_h, cur_c5 = trial, new_c5
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, cur_h.copy()

        # slowly shrink step size
        step_frac *= 0.9995

    return best_h, best_c5
