def random_swap_move(h):
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    total = h[i] + h[j]
    # propose new value for i
    # sample uniform between max(0, total-1) and min(1,total)
    low = max(0.0, total - 1.0)
    high = min(1.0, total)
    new_i = np.random.uniform(low, high)
    new_j = total - new_i
    h_new = h.copy()
    h_new[i] = new_i
    h_new[j] = new_j
    return h_new, (i,j,new_i,new_j)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # simple iterative redistribution
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-13:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-13:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 bound for a feasible step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _anneal_optimize(start_h: np.ndarray, max_time: float) -> tuple:
    """
    Simulated‑annealing style optimisation.
    Moves preserve the integral exactly by swapping mass between two indices.
    """
    n = start_h.size
    cur_h = start_h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    # temperature schedule (linear decay)
    t0 = 2e-3    # initial temperature
    t1 = 1e-7    # final temperature

    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # linear temperature
        frac = elapsed / max_time
        temp = t0 * (1.0 - frac) + t1 * frac

        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        total = cur_h[i] + cur_h[j]
        lo = max(0.0, total - 1.0)
        hi = min(1.0, total)
        if hi - lo <= 1e-14:
            continue
        # propose new split of total mass
        new_i = np.random.uniform(lo, hi)
        new_j = total - new_i
        new_h = cur_h.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        new_c5 = _c5_from_h(new_h)
        delta = new_c5 - cur_c5
        if delta < 0.0 or np.random.rand() < math.exp(-delta / temp):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        # early exit if we hit the target bound
        if best_c5 < 0.3808:
            break
    return best_h, best_c5


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Starting construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n0 = init_h.size
    except NameError:
        init_h = None
        n0 = 256

    if init_h is None:
        init_h = np.random.rand(n0)
    init_h = _project_to_feasible(init_h, n0)

    best_h = init_h.copy()
    best_c5 = _c5_from_h(best_h)
    best_n = n0

    # ------------------------------------------------------------------
    # First optimisation phase on the original grid size
    # ------------------------------------------------------------------
    while time_left() > 30.0:
        phase_time = min(60.0, time_left() * 0.5)
        h_opt, c5_opt = _anneal_optimize(best_h, phase_time)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            best_n = best_h.size
        # if we have not improved for a while, try a fresh random start
        if c5_opt >= best_c5 * 0.9999:
            fresh = np.random.rand(best_n)
            fresh = _project_to_feasible(fresh, best_n)
            best_h, best_c5 = fresh, _c5_from_h(fresh)
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # If time permits, upscale to a finer grid and polish further
    # ------------------------------------------------------------------
    if time_left() > 20.0:
        fine_n = best_n * 2
        fine_h = _upscale_h(best_h, fine_n)
        fine_h = _project_to_feasible(fine_h, fine_n)
        h_fine, c5_fine = _anneal_optimize(fine_h, time_left())
        if c5_fine < best_c5:
            best_h, best_c5, best_n = h_fine, c5_fine, fine_n

    # ------------------------------------------------------------------
    # Final sanity projection and result
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)
    return best_h.tolist(), float(best_c5), int(best_h.size)
