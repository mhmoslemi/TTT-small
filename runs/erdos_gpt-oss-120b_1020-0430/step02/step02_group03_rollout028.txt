def h_from_a(a, n):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    L0 = (1 - a) / 2.0
    a_len = a
    b_len = 1 - a
    # using vectorized piecewise
    h = np.empty(n)
    # Use np.where
    # region1: xs < L0 -> 0
    # region2: xs < L0 + a_len -> ramp up
    # region3: xs < L0 + a_len + b_len -> 1
    # region4: xs < L0 + a_len + b_len + a_len -> ramp down
    # else 0
    # compute boundaries
    t1 = L0
    t2 = L0 + a_len
    t3 = L0 + a_len + b_len
    t4 = L0 + a_len + b_len + a_len
    # ramp up: (xs - t1) / a_len
    # plateau: 1
    # ramp down: (t4 - xs) / a_len
    h = np.where(xs < t1, 0.0,
            np.where(xs < t2, (xs - t1) / a_len,
                np.where(xs < t3, 1.0,
                    np.where(xs < t4, (t4 - xs) / a_len, 0.0))))
    # h may have NaNs if a_len==0; handle a_len=0 separately: shape reduces to block.
    if a_len == 0:
        # block shape: h = 1 on interval [L0, L0 + b_len]
        h = np.where((xs >= t1) & (xs < t1 + b_len), 1.0, 0.0)
    return h

    n = h.size

def enforce_sum(h, target, rng):
    # Simple iterative transfer to meet target
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            # need to increase sum: add to i (increase h[i]) limited by 1 - h[i]
            add = min(diff, 1.0 - h[i])
        else:
            # need to decrease sum: subtract from i limited by h[i]
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # After loop diff may be small; if still not zero due to rounding, adjust small
    if abs(diff) > 1e-10:
        # distribute remaining diff proportionally to non-bound elements
        mask = (diff > 0) & (h < 1) | (diff < 0) & (h > 0)
        idx = np.where(mask)[0]
        if idx.size > 0:
            per = diff / idx.size
            # Clip per to not exceed bounds
            for i in idx:
                limit = 1.0 - h[i] if diff > 0 else -h[i]
                adj = max(min(per, limit), -limit)
                h[i] += adj
    return h

CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # surrogate: (max + log_sum_exp)/beta
    # compute correlation via FFT for speed? But simpler use np.correlate.
    # Since n moderate, use np.correlate.
    corr = np.correlate(x, 1 - x, mode='full')

    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # simple random transfer to fix sum; each step changes one element
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # In rare case still off due to rounding, distribute uniformly
    if abs(diff) > 1e-12:
        # distribute sign to random indices not at bound
        sign = np.sign(diff)
        # find indices where we can move in required direction
        mask = (sign > 0) & (h < 1.0) | (sign < 0) & (h > 0.0)
        idx = np.where(mask)[0]
        if idx.size > 0:
            per = diff / idx.size
            for i in idx:
                limit = 1.0 - h[i] if sign > 0 else -h[i]
                adj = max(min(per, limit), -limit)
                h[i] += adj
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()

    return h

        new_h = perturb_h(h, rng, scale=step)
        new_c = compute_c5_fft(new_h)
        if new_c < curr_c:
            h, curr_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        else:
            elapsed = time.time() - start
            T = max(T0 * (1 - elapsed / time_limit), 1e-12)
            if rng.random() < np.exp(-(new_c - curr_c) / T):
                h, curr_c = new_h, new_c
        it += 1
        if it % 2000 == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------
    # choose discretisation size
    # -------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 1000                # finer grid when no parent solution is given
    target_sum = n / 2.0

    # -------------------------------------------------
    # initial candidates
    # -------------------------------------------------
    best_h = None
    best_c = float("inf")
    best_a = None

    # use parent construction if present
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
        c0 = compute_c5_fft(h0)
        if c0 < best_c:
            best_c, best_h = c0, h0.copy()

    # -------------------------------------------------
    # search over the one‑parameter family
    # -------------------------------------------------
    def try_shape(a):
        nonlocal best_h, best_c, best_a
        h = generate_h_from_a(a, n)
        h = enforce_sum(h, target_sum, rng)
        c = compute_c5_fft(h)
        if c < best_c:
            best_c, best_h, best_a = c, h.copy(), a

    # coarse sweep
    for a in np.linspace(0.0, 1.0, 41):
        try_shape(a)

    # two refinements around the current best a
    if best_a is not None:
        for width, steps in ((0.05, 41), (0.01, 41)):
            lo, hi = max(0.0, best_a - width), min(1.0, best_a + width)
            for a in np.linspace(lo, hi, steps):
                try_shape(a)

    # -------------------------------------------------
    # main stochastic hill‑climb (most of the remaining budget)
    # -------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0:
        hc_time = remaining * 0.7
        h_opt, c_opt = hill_climb(best_h, rng, hc_time, start_step=0.5)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # -------------------------------------------------
    # a quick second start if time permits
    # -------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 10.0:
        # fresh random start
        rnd = random_feasible_vector(n, rng)
        hc_time2 = remaining * 0.5
        h_opt2, c_opt2 = hill_climb(rnd, rng, hc_time2, start_step=0.5)
        if c_opt2 < best_c:
            best_h, best_c = h_opt2, c_opt2

    # -------------------------------------------------
    # optional SLSQP refinement (few short runs)
    # -------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0:
        betas = [20.0, 80.0, 200.0, 500.0]
        for beta in betas:
            if time.time() - start_time > budget_s * 0.98:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=300)
            except Exception:
                continue
            c_ref = compute_c5_fft(refined)
            if c_ref < best_c:
                best_h, best_c = refined.copy(), c_ref

    # -------------------------------------------------
    # final clean‑up
    # -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(n)
