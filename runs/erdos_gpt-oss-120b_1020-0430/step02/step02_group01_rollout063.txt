import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def project_to_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -5.0, 5.0
    for _ in range(45):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    v = rng.random(n)
    return project_to_sum_bounds(v, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 3000, start_step: float = 0.5):
    """Simple hill‑climb: accept only strictly improving moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-6)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 8):
    """Few sub‑gradient steps targeting the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum_bounds(h, target)
    return h


def softmax_surrogate(h: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max_k Σ h_i(1‑h_{i+k})."""
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    ex = np.exp(beta * (corr - m))
    return (np.log(np.sum(ex)) + m) / beta


def gradient_softmax(h: np.ndarray, beta: float):
    """
    Gradient of the softmax surrogate w.r.t. h.
    Returns a vector of same shape as h.
    """
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    expw = np.exp(beta * (corr - m))
    w = expw / expw.sum()                     # shape (2n‑1,)

    # term1 = Σ_s w_s * (1‑h_{j+s})   (j index)
    w_rev = w[::-1]
    term1_full = np.convolve(w_rev, 1 - h, mode="full")
    term1 = term1_full[(n - 1):(n - 1 + n)]

    # term2 = Σ_s w_s * h_{j‑s}
    term2_full = np.convolve(w, h, mode="full")
    term2 = term2_full[(n - 1):(n - 1 + n)]

    return term1 - term2


def gradient_descent_softmax(h0: np.ndarray, beta_seq, steps_seq,
                            step0: float, rng: np.random.Generator,
                            time_limit: float):
    """Projected gradient descent on the softmax surrogate (continuation on β)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    start = time.time()
    for beta, steps in zip(beta_seq, steps_seq):
        step = step0
        for _ in range(steps):
            if time.time() - start > time_limit:
                return h
            grad = gradient_softmax(h, beta)
            h_new = h - step * grad
            h_new = project_to_sum_bounds(h_new, target)
            # accept if surrogate improves
            cur = softmax_surrogate(h, beta)
            nxt = softmax_surrogate(h_new, beta)
            if nxt < cur:
                h = h_new
                step *= 1.02
            else:
                step *= 0.5
                if step < 1e-8:
                    break
    return h

# --------------------------------------------------------------
# Main optimisation entry point
# --------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf

    # ------------------------------------------------------------------
    # Base construction (use parent if provided)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_sum_bounds(h0, n0 / 2.0)
        best_h, best_c = h0, compute_c5(h0)
    else:
        n0 = 120
        h0 = random_feasible_vector(n0, rng)
        best_h, best_c = h0, compute_c5(h0)

    # ------------------------------------------------------------------
    # Try a few up‑sampling factors (1,2,3) – larger n gives more flexibility
    # ------------------------------------------------------------------
    for factor in (1, 2, 3):
        if time.time() - start_time > budget_s * 0.92:
            break
        n = n0 * factor
        # create initial vector for this resolution
        if factor == 1:
            h_start = best_h.copy()
        else:
            h_start = np.repeat(best_h, factor)
            # repeat keeps the integral correct automatically
        # break symmetry a little
        h_start = perturb_h(h_start, rng, scale=0.1)

        # --------------------------------------------------------------
        # Continuation on β with projected gradient descent
        # --------------------------------------------------------------
        beta_seq = [5.0, 20.0, 80.0, 200.0, 800.0]
        steps_seq = [400, 300, 200, 150, 120]
        time_limit = start_time + budget_s * 0.90
        h_opt = gradient_descent_softmax(
            h_start, beta_seq, steps_seq, step0=0.2, rng=rng, time_limit=time_limit
        )

        # --------------------------------------------------------------
        # Local refinement: hill‑climb + sub‑gradient
        # --------------------------------------------------------------
        h_opt, c_val = hill_climb(h_opt, rng, max_iters=2000, start_step=0.3)
        if c_val < best_c:
            best_h, best_c = h_opt.copy(), c_val

        h_opt = gradient_refine(h_opt, rng, step=0.02, max_iter=12)
        c_val = compute_c5(h_opt)
        if c_val < best_c:
            best_h, best_c = h_opt.copy(), c_val

        # --------------------------------------------------------------
        # Random restarts (few) in the remaining budget
        # --------------------------------------------------------------
        for _ in range(3):
            if time.time() - start_time > budget_s * 0.88:
                break
            h_rand = random_feasible_vector(n, rng)
            h_rand = gradient_descent_softmax(
                h_rand, beta_seq, [200] * len(beta_seq), step0=0.3, rng=rng,
                time_limit=start_time + budget_s * 0.85
            )
            h_rand, c_rand = hill_climb(h_rand, rng, max_iters=1500, start_step=0.3)
            if c_rand < best_c:
                best_h, best_c = h_rand.copy(), c_rand

    # ------------------------------------------------------------------
    # Final feasibility enforcement and exact evaluation
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(120, 0.5)
    best_h = project_to_sum_bounds(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
