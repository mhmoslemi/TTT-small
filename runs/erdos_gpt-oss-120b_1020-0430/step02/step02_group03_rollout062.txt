    # adjust to meet sum target with water filling

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def enforce_sum(h, target, rng):
    """Clip 0‑1 and adjust sum to target by moving mass on random free indices."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector with entries in [0,1] and sum = n/2."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (keeps sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

def compute_c5(h):
    """Exact C5 (max overlap) using linear correlation."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def linear_corr(h):
    """Linear auto‑correlation of h using FFT (length 2·n‑1)."""
    n = h.size
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    corr = np.fft.irfft(H * np.conj(H))[:2 * n - 1]
    return corr.real

def compute_c5_fast(h):
    """Fast C5 via min auto‑correlation (identical to compute_c5 up to round‑off)."""
    n = h.size
    min_corr = np.min(linear_corr(h))
    return 1.0 - min_corr * (2.0 / n)

# ---------- projection onto sum & box ----------
def project_to_feasible(y, target):
    """Project y onto {0≤x≤1, sum(x)=target} by a simple water‑filling loop."""
    x = np.clip(y, 0.0, 1.0)
    for _ in range(20):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if not np.any(mask):
            # all at a bound – distribute uniformly and clip again
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
            continue
        x[mask] += diff / mask.sum()
        x = np.clip(x, 0.0, 1.0)
    return x

# ---------- subgradient optimisation ----------
def subgradient_optimize(h0, target, rng, max_iters=3000, init_step=0.2):
    """Maximise the minimal auto‑correlation (equivalently minimise C5)."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5_fast(h)
    step = init_step
    n = h.size
    for it in range(max_iters):
        corr = linear_corr(h)
        min_corr = np.min(corr)
        # all shifts attaining the minimum
        idx = np.where(corr <= min_corr + 1e-12)[0] - (n - 1)
        g = np.zeros_like(h)
        for k in idx:
            if k >= 0:
                g[:n - k] += h[k:]
                g[k:]     += h[:n - k]
            else:
                kp = -k
                g[kp:]    += h[:n - kp]
                g[:n - kp] += h[kp:]
        norm = np.linalg.norm(g)
        if norm == 0.0:
            break
        g /= norm
        improved = False
        trial_step = step
        for _ in range(4):
            h_new = project_to_feasible(h + trial_step * g, target)
            c_new = compute_c5_fast(h_new)
            if c_new < best_c - 1e-12:
                best_c, best_h = c_new, h_new.copy()
                h = h_new
                improved = True

            trial_step *= 0.5
        if not improved:
            # occasional random kick to escape flat regions
            h = perturb_h(h, rng, scale=step * 5)
        else:
            step = min(step * 1.05, init_step)
    return best_h, best_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h, best_c = h.copy(), compute_c5(h)

# ---------- smooth surrogate + SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        seeds.append(enforce_sum(np.clip(c, 0, 1), target, rng))
    seeds.append(enforce_sum(np.linspace(0, 1, n), target, rng))
    seeds.append(enforce_sum(np.linspace(1, 0, n), target, rng))
    for _ in range(3):
        seeds.append(random_feasible_vector(n, rng))

    for h in seeds:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- main optimisation loop -----
    beta_seq = [5.0, 20.0, 80.0, 200.0, 500.0]
    while time.time() - start_time < budget_s * 0.85:
        # choose start point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        # subgradient ascent on the worst shift
        h_opt, c_opt = subgradient_optimize(start, target, rng,
                                            max_iters=2500, init_step=0.15)

        # optional local hill‑climb polishing
        if rng.random() < 0.4:
            h_opt, c_opt = hill_climb(h_opt, rng, max_iters=1500, start_step=0.4)

        # SLSQP refinement with increasing smoothness
        for beta in beta_seq:
            if time.time() - start_time > budget_s * 0.9:
                break
            try:
                h_opt = slsqp_optimize(h_opt, beta, max_iter=300)
            except Exception:
                continue
            c_opt = compute_c5(h_opt)

        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
