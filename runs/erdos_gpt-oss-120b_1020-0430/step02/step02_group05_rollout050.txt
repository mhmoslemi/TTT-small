def compute_c5(h):
    n = h.shape[0]
    L = 2*n - 1
    A = np.fft.rfft(h, L)
    B = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(A * B, n=L)

    # compute max times dx = 2/n
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]

import time
import numpy as np
import scipy.optimize as opt

# -------------------------------------------------
# Projection onto [0,1]^n ∩ {sum = target}
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

# -------------------------------------------------
# Fast C5 using FFT (used inside the optimiser)
def _c5_fast(h: np.ndarray) -> float:
    n = h.shape[0]
    L = 2 * n - 1
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(A * B, n=L)
    return np.max(corr) * (2.0 / n)

# -------------------------------------------------
# Accurate C5 using np.correlate (used for the final reported bound)
def _c5_exact(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

# -------------------------------------------------
# Constraints for SLSQP (global functions – no closures)
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

# -------------------------------------------------
# One SLSQP refinement step
def _solve_h(start_h: np.ndarray, max_iter: int = 200) -> tuple[np.ndarray, float]:
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = h0 if not res.success else res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_exact(h_opt)

# -------------------------------------------------
# Upscale / downscale a step function by linear interpolation
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# -------------------------------------------------
# Tent shape (simple fallback)
def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

# -------------------------------------------------
# Simulated annealing optimiser (pairwise mass exchange)
def _anneal_optimize(start_h: np.ndarray, max_iters: int, time_limit: float) -> tuple[np.ndarray, float]:
    n = start_h.size
    cur_h = start_h.copy()
    cur_val = _c5_fast(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    T = 0.002                     # initial temperature
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # max feasible change: increase i, decrease j
        max_inc = min(1.0 - cur_h[i], cur_h[j])
        # decrease i, increase j
        max_dec = min(cur_h[i], 1.0 - cur_h[j])
        if max_inc + max_dec == 0.0:
            continue
        delta = (np.random.rand() * (max_inc + max_dec)) - max_dec  # uniform in [-max_dec, max_inc]
        cur_h[i] += delta
        cur_h[j] -= delta
        # tiny numerical clean‑up
        cur_h[i] = 0.0 if cur_h[i] < 0.0 else (1.0 if cur_h[i] > 1.0 else cur_h[i])
        cur_h[j] = 0.0 if cur_h[j] < 0.0 else (1.0 if cur_h[j] > 1.0 else cur_h[j])
        new_val = _c5_fast(cur_h)
        dv = new_val - cur_val
        if dv < 0.0 or np.random.rand() < np.exp(-dv / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            cur_h[i] -= delta
            cur_h[j] += delta
        T *= 0.9997
        if T < 5e-7:
            T = 5e-7
    return best_h, best_val

# -------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # -----------------------------------------------------------------
    # parent construction, if supplied
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate grid sizes (favor the parent size)
    cand_sizes = []
    if init_h is not None:
        cand_sizes.append(init_h.size)
    for s in (128, 160, 192, 224, 256, 320, 384, 448):
        if s <= 500:
            cand_sizes.append(s)
    cand_sizes = sorted(set(cand_sizes))

    # -----------------------------------------------------------------
    # Main stochastic search loop
    while time_left() > 4.0:
        n = cand_sizes[np.random.randint(len(cand_sizes))]
        # base function at this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)
        base_h = _project_to_simplex_box(base_h, n / 2.0)

        # decide starting point
        if np.random.rand() < 0.5:
            start_h = base_h
        else:
            start_h = _project_to_simplex_box(np.random.rand(n), n / 2.0)

        # annealing – allocate a few seconds per trial
        trial_time = min(5.0, time_left() - 1.0)
        h_opt, _ = _anneal_optimize(start_h, max_iters=30000, time_limit=trial_time)

        # local polishing with SLSQP (quick)
        if time_left() > 2.0:
            h_opt, c5_val = _solve_h(h_opt, max_iter=150)
        else:
            c5_val = _c5_exact(h_opt)

        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_val, n
            if best_c5 < target_bound:
                break

    # -----------------------------------------------------------------
    # If nothing succeeded, fall back to a deterministic construction
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_exact(best_h)
        best_n = n

    # final sanity projection and exact evaluation
    best_h = _project_to_simplex_box(np.array(best_h, dtype=float), best_n / 2.0)
    best_c5 = _c5_exact(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
