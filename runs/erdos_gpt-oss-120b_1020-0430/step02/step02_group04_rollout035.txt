def subgrad_max(h):
    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # find index of max
    idx = np.argmax(corr)
    # offset = idx - (len(h) - 1)
    offset = idx - (len(h) - 1)
    n = len(h)
    # compute forward shifted: for positions where i+offset in [0,n)
    forward = np.zeros_like(h)
    if offset >= 0:
        forward[: n - offset] = h[offset:]  # h[i+offset] at i
    else:
        forward[-offset:] = h[: n + offset]
    # compute backward: h[i - offset]
    backward = np.zeros_like(h)
    if offset >= 0:
        backward[offset:] = h[: n - offset]
    else:
        backward[: n + offset] = h[-offset:]
    grad = 1.0 - forward - backward
    return grad, idx, offset

Now step: h_new = h - step * grad

def subgrad_max(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    offset = idx - (n - 1)
    # compute forward term: h[i+offset] where i+offset in bounds
    forward = np.zeros_like(h)
    if offset >= 0:
        forward[: n - offset] = h[offset:]
    else:
        forward[-offset:] = h[: n + offset]
    # compute backward term: h[i - offset]
    backward = np.zeros_like(h)
    if offset >= 0:
        backward[offset:] = h[: n - offset]
    else:
        backward[: n + offset] = h[-offset:]
    grad = 1.0 - forward - backward
    return grad, idx, offset

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while forcing sum(v)=target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random vector in [0,1] with exact sum n/2."""
    x = rng.random(n)
    return project_onto_box_sum(x, n / 2.0)


def make_symmetric(v):
    v2 = (v + v[::-1]) * 0.5
    return project_onto_box_sum(v2, v2.size / 2.0)


# ---------- subgradient for the max shift ----------
def subgrad_max(h):
    """Return subgradient of the active shift (max overlap) for h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    offset = idx - (n - 1)               # signed offset of the worst shift

    forward = np.zeros_like(h)
    if offset >= 0:
        forward[: n - offset] = h[offset:]
    else:
        forward[-offset:] = h[: n + offset]

    backward = np.zeros_like(h)
    if offset >= 0:
        backward[offset:] = h[: n - offset]
    else:
        backward[: n + offset] = h[-offset:]

    grad = 1.0 - forward - backward
    return grad, idx, offset


# ---------- subgradient descent with momentum ----------
def subgradient_descent(h0, max_iters, step0, momentum, rng):
    """Projected subgradient descent on max‑overlap objective."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    v = np.zeros_like(h)                     # momentum buffer

    for it in range(1, max_iters + 1):
        grad, _, _ = subgrad_max(h)
        v = momentum * v + grad
        step = step0 / np.sqrt(it)
        h_new = h - step * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target)

        # occasional random exchange to escape plateaus
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(h_new[i], 1.0 - h_new[j])
                if max_eps > 0:
                    eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
                    h_new[i] -= eps
                    h_new[j] += eps
                    h_new = project_onto_box_sum(h_new, target)

        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- main stochastic search (subgradient + restarts) -----
    # allocate roughly 80 % of the budget to this phase
    while time.time() - start_time < budget_s * 0.80:
        # generate a fresh feasible start (occasionally blend with current best)
        if rng.random() < 0.2:
            start_h = (best_h + random_feasible(n, rng)) * 0.5
        else:
            start_h = random_feasible(n, rng)

        # half of the restarts are forced symmetric – helps the landscape
        if rng.random() < 0.5:
            start_h = make_symmetric(start_h)

        # subgradient descent
        max_iters = 1800
        step0 = 0.25
        momentum = 0.85
        h_opt, c_opt = subgradient_descent(start_h, max_iters, step0, momentum, rng)

        # quick local hill‑climb refinement
        h_opt, c_opt = hill_climb(h_opt, rng, max_iters=800, start_step=0.3)

        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ----- up‑sampling stages (increase resolution) -----
    # remaining time is spent on finer grids, each time doubling the number of points
    up_factor = 2
    while (
        time.time() - start_time < budget_s * 0.97
        and best_h.size * up_factor <= 720
    ):
        h_up = np.repeat(best_h, up_factor)
        n_up = h_up.size
        h_up = project_onto_box_sum(h_up, n_up / 2.0)

        # subgradient descent on the finer grid
        max_iters = 2500
        step0 = 0.25 / up_factor
        h_up_opt, c_up_opt = subgradient_descent(
            h_up, max_iters, step0, momentum, rng
        )

        # final hill‑climb on the refined grid
        h_up_opt, c_up_opt = hill_climb(
            h_up_opt, rng, max_iters=1200, start_step=0.2 / up_factor
        )

        if c_up_opt < best_c:
            best_h, best_c = h_up_opt, c_up_opt

        up_factor *= 2  # try the next higher resolution if time permits

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
