    eps=1e-12
    for _ in range(20):
        # compute sum
        s = w.sum()
        if abs(s - target) <= eps:
            break
        # compute indices of free variables
        free = (w > eps) & (w < 1.0 - eps)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust, break
            break
        delta = (s - target) / n_free
        w[free] -= delta
        # after adjustment, clip again to bounds
        w = np.clip(w,0.0,1.0)
    # final minor correction: adjust residual evenly among non-boundary
    diff = w.sum() - target
    if abs(diff) > 1e-9:
        # distribute diff over variables not at bound
        free = (w > eps) & (w < 1.0 - eps)
        if free.any():
            w[free] -= diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _solve_h(start_h, max_iter=250):
    n = start_h.size
    h0 = _proj_simplex_box(start_h, n/2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}

    else:

def _proj_simplex_box(v, target):
    """Project onto {x in [0,1]^n : sum(x)=target} by water‑filling."""
    w = np.clip(v, 0.0, 1.0)
    eps = 1e-12
    for _ in range(20):
        s = w.sum()
        diff = s - target
        if abs(diff) <= eps:
            break
        free = (w > eps) & (w < 1.0 - eps)
        n_free = free.sum()
        if n_free == 0:
            break
        w[free] -= diff / n_free
        w = np.clip(w, 0.0, 1.0)
    diff = w.sum() - target
    if abs(diff) > 1e-9:
        free = (w > eps) & (w < 1.0 - eps)
        if free.any():
            w[free] -= diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def _c5_from_h(h):
    n = h.size
    f = np.fft.rfft(h, n=2 * n)
    g = np.fft.rfft(1.0 - h, n=2 * n)
    corr = np.fft.irfft(f * g, n=2 * n)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)

def _max_corr_and_shift(h):
    n = h.size
    f = np.fft.rfft(h, n=2 * n)
    g = np.fft.rfft(1.0 - h, n=2 * n)
    corr = np.fft.irfft(f * g, n=2 * n)
    corr = corr[: 2 * n - 1]
    idx = np.argmax(corr)
    return corr[idx], idx

def _gradient_worst_shift(h, shift):
    n = h.size
    if shift == 0:
        return 1.0 - 2.0 * h
    g = np.ones_like(h)
    if shift > 0:
        k = shift
        g[k:] -= h[:-k]
        g[: n - k] -= h[k:]
    else:
        p = -shift
        g[: n - p] -= h[p:]
        g[p:] -= h[: n - p]
    return g

def _sin_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.sin(np.pi * x / 2.0)
    scale = 1.0 / (h.sum() * (2.0 / n))
    h = h * scale
    return _proj_simplex_box(h, n / 2.0)

def _parabola_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 1.0 - (x - 1.0) * (x - 1.0)
    h = np.clip(h, 0.0, None)
    scale = 1.0 / (h.sum() * (2.0 / n))
    h = h * scale
    return _proj_simplex_box(h, n / 2.0)

def _upscale_h(base_h, n):
    if base_h.size == n:
        return base_h.copy()
    xb = (np.arange(base_h.size) + 0.5) * (2.0 / base_h.size)
    xt = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(xt, xb, base_h)

    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

    h = x[:n]
    return np.sum(h) - (n / 2.0)

    return x[-1]
