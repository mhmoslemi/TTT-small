import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip h to [0,1] and adjust sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _autocorr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear autocorrelation of h (no scaling)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    H = np.fft.fft(h, n=L)
    corr = np.fft.ifft(H * np.conj(H)).real
    return corr[: 2 * n - 1]


def _c5_from_h_fft(h: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx using FFT based autocorrelation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = _autocorr_fft(h)               # raw ∑ h_i h_{i‑k}
    min_autocorr = np.min(corr)           # the shift with smallest overlap
    return (n / 2.0 - min_autocorr) * dx   # = max_k C(k)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Core optimisation for a single start vector
# ----------------------------------------------------------------------
def _improve_h(start_h: np.ndarray, time_left_func) -> tuple:
    """Gradient ascent on the worst shift followed by simulated annealing."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n

    # -------- gradient ascent on the minimum autocorrelation --------
    step = 1.0
    for _ in range(2000):
        if time_left_func() < 0.5:
            break
        corr = _autocorr_fft(h)
        idx = int(np.argmin(corr))
        k = idx - (n - 1)                     # shift producing min autocorr

        # gradient of autocorr at shift k
        d = np.zeros_like(h)
        if k >= 0:
            d[k:] += h[: n - k]
            d[: n - k] += h[k:]
        else:
            kk = -k
            d[: n - kk] += h[kk:]
            d[kk:] += h[: n - kk]
        d *= dx

        h_new = _project_to_feasible(h + step * d, n)
        c5_new = _c5_from_h_fft(h_new)
        c5_cur = _c5_from_h_fft(h)

        if c5_new < c5_cur:
            h = h_new
            step *= 1.02
        else:
            step *= 0.70
        if step < 0.001:
            step = 0.001

    # -------- simulated annealing (pairwise mass transfers) --------
    c5_cur = _c5_from_h_fft(h)
    max_anneal = 400000                     # upper bound for cooling schedule
    T0 = 0.005
    T_min = 1e-7
    it = 0
    while time_left_func() > 0.1:
        # linear cooling
        T = T0 * max(0.0, 1.0 - it / max_anneal) + T_min

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random direction of transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = max_move * np.random.rand() * 0.5   # move at most half of admissible amount

        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta

        c5_new = _c5_from_h_fft(h_new)
        diff = c5_new - c5_cur
        if diff < 0 or np.random.rand() < np.exp(-diff / T):
            h = h_new
            c5_cur = c5_new

        it += 1
        if c5_cur < 0.3808:
            break
    return h, c5_cur

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Try to load a parent construction if the harness supplies one
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    # target: stop as soon as we beat 0.3808
    target = 0.3808

    # Choose a working grid size (even, ≤256)
    if parent_h is not None:
        n_target = max(128, parent_h.size)
    else:
        n_target = 128
    n_target = min(256, (n_target // 2) * 2)      # force even

    # Build a base feasible vector
    if parent_h is not None:
        base_h = _upscale_h(parent_h, n_target)
    else:
        base_h = np.full(n_target, 0.5)
    base_h = _project_to_feasible(base_h, n_target)

    best_h = base_h.copy()
    best_c5 = _c5_from_h_fft(best_h)
    best_n = n_target

    # --------------------------------------------------------------
    # First improve the base construction
    # --------------------------------------------------------------
    if time_left() > 1.0:
        h_opt, c5_opt = _improve_h(best_h, time_left)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if best_c5 < target:
            return best_h.tolist(), float(best_c5), int(best_n)

    # --------------------------------------------------------------
    # Random restarts while time permits
    # --------------------------------------------------------------
    while time_left() > 5.0:
        # fresh random feasible start
        rand_h = np.random.rand(n_target)
        rand_h = _project_to_feasible(rand_h, n_target)

        h_opt, c5_opt = _improve_h(rand_h, time_left)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if best_c5 < target:
            break

    # Final fallback: if nothing succeeded, return the uniform construction.
    if best_h is None:
        best_n = n_target
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h_fft(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
