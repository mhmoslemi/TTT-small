def compute_c5(h):
    n = h.shape[0]
    fft_len = 1 << (2 * n - 1).bit_length() # next power of two greater than 2n-1.
    H = np.fft.rfft(h, n=fft_len)
    G = np.fft.rfft(1 - h, n=fft_len)
    corr = np.fft.irfft(H * G, n=fft_len)
    # correlation 'full' is length 2n-1 centered at index n-1.
    # The element at index n-1 + k corresponds to lag k (positive shift).
    # We'll take max of corr[:2*n-1] (real part).
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

    """Clamp to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value (max correlation) for a feasible step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every integer shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce the integral constraint sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """We minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    """Run a single SLSQP optimisation from an initial guess."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    return np.interp(xf, xc, coarse_h)


def _local_search_h(start_h: np.ndarray, max_steps: int,
                    time_left_fn) -> tuple:
    """Simple greedy hill‑climbing with mass transfers."""
    h = start_h.copy()
    n = h.size
    best_c5 = _c5_from_h(h)
    step_scale = 0.2
    for step in range(max_steps):
        if time_left_fn() < 0.2:
            break
        # pick source (h>0) and target (h<1)
        src = np.random.randint(n)
        while h[src] <= 1e-12:
            src = np.random.randint(n)
        tgt = np.random.randint(n)
        while h[tgt] >= 1.0 - 1e-12:
            tgt = np.random.randint(n)
        # transfer amount
        delta = step_scale * np.random.rand()
        delta = min(delta, h[src], 1.0 - h[tgt])
        if delta < 1e-12:
            continue
        h_new = h.copy()
        h_new[src] -= delta
        h_new[tgt] += delta
        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            # slowly shrink step_scale when we improve
            step_scale *= 0.99
        else:
            # occasional increase to escape plateaux
            if np.random.rand() < 0.01:
                step_scale *= 1.02
    return h, best_c5


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Get optional parent construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # ------------------------------------------------------------------
    # Candidate grid sizes (even, up to 800)
    # ------------------------------------------------------------------
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    # start at 256 (known size) and go up in steps
    for n in (256, 320, 384, 448, 512, 640, 768):
        if n <= 800:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    def update(best_vec, val, n):
        nonlocal best_h, best_c5, best_n
        if val < best_c5:
            best_h = best_vec.copy()
            best_c5 = float(val)
            best_n = n

    # ------------------------------------------------------------------
    # If a parent solution exists, refine it first
    # ------------------------------------------------------------------
    if init_h is not None:
        n0 = init_h.size
        h_opt, c5_val = _solve_h(init_h, max_iter=800)
        update(h_opt, c5_val, n0)
        # a few perturbations around the parent
        for _ in range(2):
            if time_left() < 2.0:
                break
            pert = 0.1 * np.random.randn(n0)
            start = _project_to_feasible(init_h + pert, n0)
            h_opt, c5_val = _solve_h(start, max_iter=600)
            update(h_opt, c5_val, n0)

    # ------------------------------------------------------------------
    # Main multi‑resolution search
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 8.0:
            break

        # initialise base vector for this resolution
        if best_h is not None and best_n == n:
            base_h = best_h.copy()
        elif best_h is not None:
            base_h = _upscale_h(best_h, n)
        elif init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        else:
            # random binary with exact half ones
            half = n // 2
            base_h = np.zeros(n, dtype=float)
            base_h[:half] = 1.0
            np.random.shuffle(base_h)

        base_h = _project_to_feasible(base_h, n)

        # decide how many independent attempts we can afford
        remaining = time_left()
        attempts = max(3, int(min(30, remaining // (len(n_candidates) * 0.5))))

        sigma_start = 0.2

        for att in range(attempts):
            if time_left() < 4.0:
                break

            # choose start point for the local optimiser
            if att == 0:
                start_h = base_h.copy()
            else:
                sigma = sigma_start * (0.85 ** att)
                pert = sigma * np.random.randn(n)
                start_h = base_h + pert
                start_h = _project_to_feasible(start_h, n)

            # run SLSQP optimisation
            h_opt, c5_val = _solve_h(start_h, max_iter=800)
            update(h_opt, c5_val, n)

            # polish with cheap hill‑climbing
            if time_left() > 2.0:
                h_ref, c5_ref = _local_search_h(
                    h_opt, max_steps=2000, time_left_fn=time_left
                )
                update(h_ref, c5_ref, n)

            # if we found a markedly better vector, shift the base towards it
            if c5_val < best_c5 * 0.998:
                base_h = h_opt

            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final intensive polishing on the current best solution
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 5.0:
        h_opt, c5_val = _solve_h(best_h, max_iter=1200)
        if c5_val < best_c5:
            best_h, best_c5 = h_opt, c5_val
        # extra hill‑climbing
        if time_left() > 2.0:
            h_ref, c5_ref = _local_search_h(
                best_h, max_steps=5000, time_left_fn=time_left
            )
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref

    # ------------------------------------------------------------------
    # Fallback in the unlikely case no feasible vector was found
    # ------------------------------------------------------------------
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    else:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
