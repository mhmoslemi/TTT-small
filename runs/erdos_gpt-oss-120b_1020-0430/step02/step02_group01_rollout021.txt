import time, numpy as np, from scipy.optimize import minimize

# global variables for current target and beta
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def slsqp_optimize(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum exactly
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def simulated_annealing_binary(n, rng, time_deadline):
    """Return binary vector of length n with weight n/2 minimizing C5."""
    w = n // 2
    # initial random binary
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, w, replace=False)
    h[ones] = 1.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # indices list
    ones_idx = list(ones.tolist())
    zeros_idx = [i for i in range(n) if h[i] == 0.0]
    # SA parameters
    max_iters = 20000
    T0 = 5.0
    for it in range(max_iters):
        # time check every 1000
        if (it & 1023) == 0:
            if time.time() > time_deadline:
                break
        # pick random 1 and 0
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        T = T0 * (1.0 - it / max_iters)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            cur_score = new_score
            # update index lists
            ones_idx.remove(i)
            zeros_idx.append(i)
            zeros_idx.remove(j)
            ones_idx.append(j)

                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

def random_feasible_vector_continuous(n, rng):
    # generate random continuous feasible vector with sum = n/2
    x = rng.random(n)
    target = n / 2.0
    x *= target / x.sum()
    # clip
    x = np.clip(x, 0.0, 1.0)
    # adjust sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:

def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5):
    # same as previous perturb_h and hill_climb but with simple moves
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        # perturb: move epsilon from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        eps = max(-max_eps, min(eps, max_eps))
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

from scipy.optimize import minimize

# ----- globals used by the soft‑max surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA

    )
