def _soft_obj_and_grad(h, alpha):
    N = h.size
    G = np.convolve(h, h[::-1], mode='full')
    # compute soft max of -G
    # Use logsumexp from numpy? Not built-in; use scipy.special.logsumexp
    from scipy.special import logsumexp
    # compute w = np.exp(-alpha * G)
    # For numerical stability, subtract max before exponent.
    # Use log-sum-exp as: logsumexp(-alpha*G) = max_val + log(sum(exp(-alpha*G - max_val)))
    # But we need also weights w = exp(-alpha * G) / sum(...).
    max_val = np.max(-alpha * G)
    exp_shifted = np.exp(-alpha * G - max_val)
    Z = np.sum(exp_shifted)
    obj = max_val + np.log(Z)
    # Now weights
    coeffs = -alpha * exp_shifted / Z  # corresponds to -alpha * w / sum(w)
    # compute gradient
    grad = np.zeros(N)
    # use vectorized loops
    offset_center = N - 1
    for s in range(2*N - 1):
        coeff = coeffs[s]
        if coeff == 0:
            continue
        d = s - offset_center
        if d >= 0:
            # h[j+d] term
            grad[:N - d] += coeff * h[d:]
            # h[j-d] term
            grad[d:] += coeff * h[:N - d]
        else:
            dneg = -d
            grad[dneg:] += coeff * h[:N - dneg]
            grad[:N - dneg] += coeff * h[dneg:]
    return obj, grad
```

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h):
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _hill_climb(h, max_steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

def _soft_obj_and_grad(h, alpha):
    # auto-correlation G with zero padding
    N = h.size
    G = np.convolve(h, h[::-1], mode='full')
    # log-sum-exp of -alpha*G
    # Use shift for stability
    vals = -alpha * G
    max_val = np.max(vals)
    exp_shifted = np.exp(vals - max_val)
    Z = np.sum(exp_shifted)
    obj = max_val + np.log(Z)
    # coefficient for gradient: -alpha * exp(-alpha*G) / sum(exp(-alpha*G))
    coeffs = -alpha * exp_shifted / Z
    grad = np.zeros(N)
    offset_center = N - 1
    for s in range(2*N - 1):
        coeff = coeffs[s]
        if coeff == 0.0:
            continue
        d = s - offset_center
        if d >= 0:
            # h_{j+d}
            grad[:N - d] += coeff * h[d:]
            # h_{j-d}
            grad[d:] += coeff * h[:N - d]
        else:
            dneg = -d
            grad[dneg:] += coeff * h[:N - dneg]
            grad[:N - dneg] += coeff * h[dneg:]
    return obj, grad

            best_c = c_new
            best_h = h_new.copy()
        # adapt learning rate
        lr *= decay
        h = h_new
    return best_h, best_c

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the feasible simplex (0<=h<=1, sum(h)=n/2)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9,
                         max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation (max overlap integral)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Upscaling / interpolation of a coarse step function
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Simple hill‑climb using epsilon exchanges
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Smoothed max‑objective and its gradient
# ----------------------------------------------------------------------
def _soft_obj_and_grad(h: np.ndarray, alpha: float) -> tuple:
    N = h.size
    G = np.convolve(h, h[::-1], mode="full")               # auto‑correlation
    vals = -alpha * G
    max_val = np.max(vals)
    exp_shifted = np.exp(vals - max_val)                  # stable exp
    Z = np.sum(exp_shifted)
    obj = max_val + np.log(Z)                             # log‑sum‑exp
    coeffs = -alpha * exp_shifted / Z                     # -α·w_s
    grad = np.zeros(N)
    off = N - 1                                           # index of shift 0
    for s in range(2 * N - 1):
        coeff = coeffs[s]
        if coeff == 0.0:
            continue
        d = s - off
        if d >= 0:
            grad[: N - d] += coeff * h[d:]                # h_{j+d}
            grad[d:] += coeff * h[: N - d]                # h_{j-d}
        else:
            dneg = -d
            grad[dneg:] += coeff * h[: N - dneg]          # h_{j+d}
            grad[: N - dneg] += coeff * h[dneg:]          # h_{j-d}
    return obj, grad


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed objective
# ----------------------------------------------------------------------
def _gd_optimize(initial_h: np.ndarray, alpha: float, lr: float,
                 decay: float, max_iter: int, time_limit: float) -> tuple:
    N = initial_h.size
    h = _project_to_feasible(initial_h.copy(), N)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        _, grad = _soft_obj_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = _project_to_feasible(h_new, N)
        c_new = _c5_from_h(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        lr *= decay
        h = h_new
    return best_h, best_c


# ----------------------------------------------------------------------
# Sub‑gradient step that directly improves the worst shift
# ----------------------------------------------------------------------
def _subgradient_step(h: np.ndarray, lr: float) -> np.ndarray:
    N = h.size
    G = np.convolve(h, h[::-1], mode="full")
    k_idx = np.argmin(G)                  # shift with minimal auto‑correlation
    k = k_idx - (N - 1)                   # signed shift
    g = np.zeros(N)
    if k >= 0:
        g[: N - k] += h[k:]
        g[k:] += h[: N - k]
    else:
        d = -k
        g[d:] += h[: N - d]
        g[: N - d] += h[d:]
    delta = lr * g
    delta -= np.mean(delta)               # keep total sum unchanged
    h_new = h + delta
    return _project_to_feasible(h_new, N)
