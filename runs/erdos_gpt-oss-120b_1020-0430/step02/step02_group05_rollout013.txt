def propose_move(h, move_scale):
    i,j = np.random.choice(n,2,replace=False)
    delta_min = -min(h[i], 1 - h[j])
    delta_max = min(1 - h[i], h[j])
    if delta_max - delta_min <= 0:
        return None
    delta = move_scale * (delta_min + (delta_max - delta_min) * np.random.rand())
    if abs(delta) < 1e-12:
        return None
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # ensure bounds (should be safe)
    return h_new

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x : 0 ≤ x_i ≤ 1, sum x_i = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discrete step function on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points, then projected to feasible set."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 600):
    """Run SLSQP from a feasible start and return refined (h, C5)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds,
        constraints=constraints, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]

    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _propose_move(h: np.ndarray, move_scale: float):
    """Randomly transfer mass between two coordinates while respecting bounds."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)

    # feasible delta so that 0≤h_i+δ≤1 and 0≤h_j-δ≤1
    delta_min = -min(h[i], 1.0 - h[j])   # negative move (i loses, j gains)
    delta_max = min(1.0 - h[i], h[j])    # positive move (i gains, j loses)
    if delta_max <= delta_min:
        return None
    delta = delta_min + (delta_max - delta_min) * np.random.rand()
    delta *= move_scale
    if delta < delta_min:
        delta = delta_min
    elif delta > delta_max:
        delta = delta_max
    if abs(delta) < 1e-15:
        return None

    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    new_h[i] = min(1.0, max(0.0, new_h[i]))
    new_h[j] = min(1.0, max(0.0, new_h[j]))
    return new_h


def _optimize_at_n(start_h: np.ndarray, n: int, time_left):
    """Simulated‑annealing + occasional SLSQP refinement at a fixed resolution."""
    # match resolution
    h = _upscale_h(start_h, n) if start_h.size != n else start_h.copy()
    h = _project_to_simplex_box(h, n / 2.0)

    # one quick local refinement
    h_ref, c5_ref = _solve_h(h, max_iter=800)
    cur_c5 = min(c5_ref, _c5_from_h(h))
    best_h = h_ref if c5_ref < _c5_from_h(h) else h
    best_c5 = cur_c5

    cur_h = best_h.copy()

    # SA parameters
    T = 0.05
    temp_decay = 0.9999
    move_scale = 0.8
    move_decay = 0.99995
    min_temp = 1e-6
    iteration = 0

    while time_left() > 3.0:
        new_h = _propose_move(cur_h, move_scale)
        if new_h is None:
            continue
        new_c5 = _c5_from_h(new_h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        elif T > min_temp and np.random.rand() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5

        # schedule updates
        iteration += 1
        T = max(min_temp, T * temp_decay)
        move_scale = max(0.001, move_scale * move_decay)

        # periodic polishing with SLSQP
        if iteration % 5000 == 0:
            refined_h, refined_c5 = _solve_h(cur_h, max_iter=400)
            if refined_c5 < best_c5:
                best_h, best_c5 = refined_h.copy(), refined_c5
                cur_h, cur_c5 = refined_h.copy(), refined_c5

    # final intensive polishing
    final_h, final_c5 = _solve_h(best_h, max_iter=2000)
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5
    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    time_left = lambda: budget_s - elapsed()
    target_bound = 0.3808

    # Obtain initial construction if supplied
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # Stage 1 – work at the native resolution if reasonable
    if init_h is not None and init_h.size <= 256:
        n0 = init_h.size
        best_h, best_c5 = _optimize_at_n(init_h, n0, time_left)
    else:
        n0 = 256
        best_h = _tent_h(n0)
        best_h, best_c5 = _optimize_at_n(best_h, n0, time_left)

    if best_c5 < target_bound and time_left() > 1.0:
        return best_h.tolist(), float(best_c5), int(best_h.size)

    # Stage 2 – upscale to 256 points and continue refinement
    n2 = 256
    if best_h.size != n2:
        upscale_h = _upscale_h(best_h, n2)
    else:
        upscale_h = best_h.copy()
    best_h2, best_c5_2 = _optimize_at_n(upscale_h, n2, time_left)

    if best_c5_2 < best_c5:
        best_h, best_c5 = best_h2, best_c5_2

    # Final safety projection and evaluation
    best_h = _project_to_simplex_box(np.array(best_h, dtype=float), best_h.size / 2.0)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_h.size)
