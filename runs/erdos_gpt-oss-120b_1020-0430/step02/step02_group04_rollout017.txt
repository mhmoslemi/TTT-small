def smooth_max_descent(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    target_sum = n/2.0
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient: start from ones
            g = np.ones(n)
            # loop over lags
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g[: n - k] -= wk * h[k:]
                    g[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g[kk:]    -= wk * h[: n - kk]
                    g[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g
            # gradient descent step
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, target_sum)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
                # optionally increase step?
            else:
                # occasionally accept worse move to escape
                if rng.random() < 0.001:
                    h = new_h
            step *= 0.998
    return best_h, best_c

def max_subgrad_descent(h0, max_iters, rng, step0=0.5, decay=0.995):

    best_c = compute_c5(best_h)

def hill_climb_swap(h0, rng, max_iters=2000, step_scale=0.5):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

        i = rng.integers(0, n)

        # propose moving epsilon from i to j
        max_eps = min(h[i], 1.0 - h[j])

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam

                if rng.random() < 0.001:
                    h = new_h
            step *= 0.998
    return best_h, best_c

# ---------- sub‑gradient descent on the true max ----------
def max_subgrad_descent(h0, max_iters, rng, step0=0.5, decay=0.995):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    target = n / 2.0
    step = step0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        C = corr * (2.0 / n)
        max_c = np.max(C)
        mask = C >= max_c - 1e-9
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        g = np.ones(n)
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            k = idx - (n - 1)
            if k >= 0:
                g[: n - k] -= wk * h[k:]
                g[k:]      -= wk * h[: n - k]
            else:
                kk = -k
                g[kk:]    -= wk * h[: n - kk]
                g[: n - kk] -= wk * h[kk:]
        grad = (2.0 / n) * g
        new_h = h - step * grad
        new_h = project_onto_box_sum(new_h, target)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
        else:
            if rng.random() < 0.001:
                h = new_h
        step *= decay
    return best_h, best_c

# ---------- simple hill‑climb ----------
def hill_climb_swap(h0, rng, max_iters=2000, step_scale=0.5):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    step = step_scale
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 10) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# ---------- SLSQP refinement (soft‑max surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = (h0 + h0[::-1]) * 0.5                     # enforce symmetry
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 200
        dx = 2.0 / n
        xs = (np.arange(n) + 0.5) * dx
        h0 = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))   # triangular start
        h0 = project_onto_box_sum(h0, n / 2.0)
        h0 = (h0 + h0[::-1]) * 0.5                     # symmetry
        h0 = project_onto_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- a quick hill‑climb ----
    if time.time() - start < budget_s * 0.15:
        h_tmp, c_tmp = hill_climb_swap(best_h, rng, max_iters=2500, step_scale=0.6)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---- main search loop (smooth‑max + sub‑grad + hill‑climb) ----
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    steps_per_beta = 200
    while time.time() - start < budget_s * 0.75:
        # random start, sometimes blended with current best
        if rng.random() < 0.2:
            h_start = (rand_sym_feas(n, rng) + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)
            h_start = (h_start + h_start[::-1]) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)
        else:
            h_start = rand_sym_feas(n, rng)

        # smooth‑max gradient descent
        h_opt, c_opt = smooth_max_descent(h_start, beta_sched, steps_per_beta, rng)

        # sub‑gradient refinement on true max
        if time.time() - start < budget_s * 0.85:
            h_opt, c_opt = max_subgrad_descent(h_opt, max_iters=300, rng=rng,
                                               step0=0.4, decay=0.996)

        # short hill‑climb for fine tuning
        if time.time() - start < budget_s * 0.90:
            h_opt, c_opt = hill_climb_swap(h_opt, rng, max_iters=1200, step_scale=0.4)

        # SLSQP refinement (soft‑max surrogate)
        if time.time() - start < budget_s * 0.95:
            try:
                h_opt = slsqp_optimize(h_opt, beta=500.0, max_iter=250)
            except Exception:
                pass

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
