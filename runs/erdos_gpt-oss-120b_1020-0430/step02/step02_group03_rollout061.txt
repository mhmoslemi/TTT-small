def move_swap_binary(h, rng):
    # expects binary (0 or 1). Choose i where h[i]==1, j where h[j]==0, swap.
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones)==0 or len(zeros)==0: return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h

def move_transfer(h, rng, max_eps=0.5):
    n = len(h)
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i == j: return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0: return h.copy()
    eps = rng.random() * max_eps # uniform in [0, max_eps]
    # optionally random sign? For transfer from i to j, must reduce i, increase j.
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def move_transfer(h, rng, max_eps=1.0):

    i = rng.integers(0, n)
    j = rng.integers(0, n)

        return h.copy()

        return h.copy()

    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust to hit the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny numerical drift)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def random_binary(n, rng):
    """Generate a binary feasible vector (exactly n/2 ones)."""
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def move_swap_binary(h, rng):
    """Swap a 1 and a 0 – preserves the sum exactly."""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h


def move_transfer(h, rng, max_eps=1.0):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_possible = min(h[i], 1.0 - h[j])
    if max_possible <= 0.0:
        return h.copy()
    eps = rng.random() * max_possible
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_simple(h, rng, max_iters=2000, step_scale=0.5):
    """Deterministic hill‑climb using transfer moves."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for it in range(max_iters):
        new_h = move_transfer(cur_h, rng, max_eps=step_scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
        # gradual decay of the step scale
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return cur_h, cur_c


def anneal_optimize(start_h, rng, time_limit):
    """Simulated‑annealing style optimiser."""
    best_h = start_h.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    start = time.time()
    T0 = 0.02  # initial temperature
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * max(0.0, 1.0 - frac)  # linear cooling

        # decide move type
        if rng.random() < 0.7:
            if np.all(np.logical_or(cur_h < 1e-8, cur_h > 1 - 1e-8)):
                new_h = move_swap_binary(cur_h, rng)
            else:
                new_h = move_transfer(cur_h, rng, max_eps=1.0)
        else:
            new_h = move_transfer(cur_h, rng, max_eps=1.0)

        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / (T + 1e-12))):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # occasional restart from the best seen so far
        if rng.random() < 0.005:
            cur_h, cur_c = best_h.copy(), best_c
    return best_h, best_c
