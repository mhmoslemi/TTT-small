def _smooth_max_corr(h, alpha):
    # compute correlation conv via FFT
    n = h.size
    m = 2*n - 1
    f1 = np.fft.rfft(h, m)
    f2 = np.fft.rfft(np.flip(1 - h), m)
    corr = np.fft.irfft(f1 * f2, m)
    maxval = np.max(corr)
    # compute smoothed
    # stable: subtract max before exponent
    e = np.exp(alpha * (corr - maxval))
    smooth = maxval + np.log(np.sum(e)) / alpha
    return smooth * (2.0 / n)   # scaling

    m = 2*n - 1
    f1 = np.fft.rfft(h, m)
    f2 = np.fft.rfft(np.flip(1 - h), m)
    corr = np.fft.irfft(f1 * f2, m)

import time
import numpy as np
import scipy.optimize as opt
import functools

# ---------- feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int,
                         tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- correlation utilities ----------
def _c5_from_h_fft(h: np.ndarray) -> float:
    """Exact C5 using FFT – fast for many evaluations."""
    n = h.size
    m = 2 * n - 1
    f1 = np.fft.rfft(h, m)
    f2 = np.fft.rfft(np.flip(1.0 - h), m)
    corr = np.fft.irfft(f1 * f2, m)
    max_corr = corr.max()
    return max_corr * (2.0 / n)


def _smooth_corr_max(h: np.ndarray, n: int, alpha: float) -> float:
    """Smooth approximation of max correlation (sum, not yet scaled)."""
    m = 2 * n - 1
    f1 = np.fft.rfft(h, m)
    f2 = np.fft.rfft(np.flip(1.0 - h), m)
    corr = np.fft.irfft(f1 * f2, m)
    mx = corr.max()
    # log‑sum‑exp trick for stability
    shifted = corr - mx
    e = np.exp(alpha * shifted)
    return mx + np.log(e.sum()) / alpha


def _smooth_obj_vec(x: np.ndarray, n: int, alpha: float, penalty: float) -> float:
    """Objective for L‑BFGS‑B: smoothed C5 + quadratic sum penalty."""
    smooth = _smooth_corr_max(x, n, alpha) * (2.0 / n)
    diff = x.sum() - n / 2.0
    return smooth + penalty * diff * diff


# ---------- interpolation ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- optimisation core ----------
def _optimize_start(start_h: np.ndarray, n: int,
                    alphas, penalty: float,
                    maxiter_per_alpha: int = 150) -> np.ndarray:
    """Continuation L‑BFGS‑B over a list of smoothing parameters."""
    h = start_h.copy()
    for alp in alphas:
        fun = functools.partial(_smooth_obj_vec, n=n, alpha=alp, penalty=penalty)
        bounds = [(0.0, 1.0)] * n
        res = opt.minimize(fun, h, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter_per_alpha,
                                    'ftol': 1e-9,
                                    'disp': False})
        if res.x is not None:
            h = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum after optimisation
    h = _project_to_feasible(h, n)
    return h


def _hill_climb_refine(h: np.ndarray, n: int,
                       best_c5: float,
                       time_limit: float,
                       start_time: float) -> tuple:
    """Simple stochastic hill‑climbing while time permits."""
    while time.time() - start_time < time_limit:
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = h.copy()
        trial[idx] += 0.05 * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h_fft(trial)
        if c5 < best_c5:
            h, best_c5 = trial, c5
    return h, best_c5
