import time
import numpy as np
from scipy import optimize

# ---------------------- Utility functions ---------------------- #

def _project_to_feasible(v: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 300) -> np.ndarray:
    """Project vector v onto the box [0,1]^n intersect {sum = n/2}."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(v > 0)[0]
        else:
            free = np.where(v < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    # tiny correction (rare)
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and v[i] < 1 - 1e-12:
                add = min(residual, 1 - v[i])
                v[i] += add
                residual -= add
            elif residual < 0 and v[i] > 1e-12:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(v, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Run SLSQP from initial_h and return (h_opt, C5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse)


def _local_refine(h: np.ndarray, attempts: int = 30) -> np.ndarray:
    """Stochastic hill‑climbing with a decaying step size."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for k in range(attempts):
        delta = 0.05 * (0.7 ** k)  # decay
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random point in the feasible polytope."""
    v = np.random.rand(n)
    return _project_to_feasible(v, n)

        cand.append(parent.size)
    cand = sorted({n for n in cand if n % 2 == 0 and n <= 256})

    # allocate a modest number of attempts per size (more for larger n)
    attempts_for_n = {
        64: 120, 80: 120, 96: 140, 112: 150, 128: 180,
        144: 150, 160: 140, 176: 120, 192: 110,
        208: 100, 224:  90, 240: 80, 256: 70
    }

    for n in cand:
        if time_left() < 4.0:
            break

        # base (deterministic) start derived from parent when possible
        if parent is not None:
            if parent.size == n:
                base_h = parent.copy()
            else:
                base_h = _upscale_h(parent, n)
            base_h = _project_to_feasible(base_h, n)
        else:
            base_h = np.full(n, 0.5)

        # first deterministic optimisation on the base
        h_opt, c5_opt = _solve_h(base_h, max_iter=900)
        h_opt = _local_refine(h_opt, attempts=40)
        c5_opt = _c5_from_h(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < target:
                break

        # stochastic restarts
        max_attempts = attempts_for_n.get(n, 80)
        sigma = 0.45
        for att in range(max_attempts):
            if time_left() < 3.0:
                break
            # random feasible start
            start_h = _random_feasible(n)
            # occasionally inject a noisy version of the current best for diversification
            if np.random.rand() < 0.15 and best_h is not None and best_n == n:
                start_h = best_h + sigma * np.random.randn(n)
                start_h = _project_to_feasible(start_h, n)

            h_local, c5_local = _solve_h(start_h, max_iter=900)
            h_local = _local_refine(h_local, attempts=30)
            c5_local = _c5_from_h(h_local)

            if c5_local < best_c5:
                best_h, best_c5, best_n = h_local, c5_local, n
                if best_c5 < target:
                    break
            sigma *= 0.985  # very gradual cooling

        if best_c5 < target:
            break

    # final polish on the overall best (if time permits)
    if best_h is not None and time_left() > 2.0:
        best_h = _local_refine(best_h, attempts=80)
        best_c5 = _c5_from_h(best_h)

    # fallback safe construction
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
