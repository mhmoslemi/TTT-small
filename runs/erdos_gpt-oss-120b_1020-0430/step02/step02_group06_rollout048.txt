def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    # clip to bounds and adjust sum to n/2 using water-filling
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    # iterative approach: compute diff and distribute uniformly among free vars
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        # apply
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # add/subtract small amount from any non-bound entries
        for i in range(n):
            if residual > 0 and h[i] < 1-1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h):
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

def _upscale_h(coarse, fine_n):
    n_coarse = coarse.size
    # positions at cell centers
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(fine_n) + 0.5) * (2.0 / fine_n)
    fine = np.interp(x_fine, x_coarse, coarse)
    return fine

def _local_refine(h, attempts=10, delta=0.02):
    n = h.size
    best = h.copy()
    best_val = _c5_from_h(best)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        val = _c5_from_h(trial)
        if val < best_val:
            best = trial
            best_val = val
    return best

def _solve_h(initial_h, max_iter=500):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode='full')
    def eq(x):
        return np.sum(x[:n]) - n/2.0
    def obj(x):
        return x[-1]
    constraints = [{'type':'ineq','fun':ineq},
                   {'type':'eq','fun':eq}]
    res = opt.minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter':max_iter, 'ftol':1e-9, 'disp':False})
    if res.success and res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

import time
import numpy as np
import scipy.optimize as opt

# ---------- feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = int(free.sum())
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- objective ----------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- interpolation ----------
def _upscale_h(coarse: np.ndarray, fine_n: int) -> np.ndarray:
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(fine_n) + 0.5) * (2.0 / fine_n)
    return np.interp(x_fine, x_coarse, coarse)


# ---------- local hill climbing ----------
def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02) -> np.ndarray:
    n = h.size
    best = h.copy()
    best_val = _c5_from_h(best)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        val = _c5_from_h(trial)
        if val < best_val:
            best, best_val = trial, val
    return best


# ---------- SLSQP refinement ----------
def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n * 0.5

    def obj(x):
        return x[-1]

    constr = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    res = opt.minimize(
        obj, x0, method="SLSQP", bounds=bounds, constraints=constr,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    )
    h_opt = res.x[:n] if res.success and res.x is not None else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- simple Genetic Algorithm ----------
def _ga_optimize(
    n: int,
    init_pop: list,
    time_limit: float,
    target: float = 0.3808,
) -> tuple:
    start = time.time()
    pop = [np.copy(v) for v in init_pop]
    pop_c5 = [_c5_from_h(v) for v in pop]
    order = np.argsort(pop_c5)
    pop = [pop[i] for i in order]
    pop_c5 = [pop_c5[i] for i in order]

    pop_size = len(pop)
    best_h = pop[0]
    best_val = pop_c5[0]
    gen = 0

    crossover_p = 0.9
    mutation_p = 0.3

    while time.time() - start < time_limit:
        gen += 1
        new_pop = pop[:2]  # elitism

        while len(new_pop) < pop_size:
            # tournament selection (size 3)
            def tournament():
                idx = np.random.choice(pop_size, 3, replace=False)
                best_idx = idx[np.argmin([pop_c5[i] for i in idx])]
                return pop[best_idx]

            p1 = tournament()
            p2 = tournament()
            child = p1.copy()

            if np.random.rand() < crossover_p:
                mask = np.random.rand(n) < 0.5
                child[mask] = p2[mask]

            if np.random.rand() < mutation_p:
                subset = np.random.choice(n, max(1, n // 5), replace=False)
                sigma = 0.12 * np.exp(-gen / 2000) + 0.02
                child[subset] += sigma * np.random.randn(len(subset))

            child = _project_to_feasible(child, n)
            new_pop.append(child)

        pop = new_pop
        pop_c5 = [_c5_from_h(v) for v in pop]
        order = np.argsort(pop_c5)
        pop = [pop[i] for i in order]
        pop_c5 = [pop_c5[i] for i in order]

        if pop_c5[0] < best_val:
            best_val, best_h = pop_c5[0], pop[0]
            if best_val <= target:
                break

    return best_h, best_val


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    target_bound = 0.3808

    # ---- decide grid size (use 256, upscale parent if available) ----
    fine_n = 256
    try:
        parent = np.array(initial_h_values, dtype=float)
    except NameError:
        parent = None

    if parent is not None:
        if parent.size != fine_n:
            base = _upscale_h(parent, fine_n)
        else:
            base = parent.copy()
    else:
        base = np.full(fine_n, 0.5)

    # ---- initial population for GA ----
    pop = []
    pop.append(base)                               # parent / upscale
    pop.append(np.full(fine_n, 0.5))                # uniform
    # deterministic half‑step (feasible automatically)
    half = np.concatenate([np.zeros(fine_n // 2), np.ones(fine_n // 2)])
    pop.append(_project_to_feasible(half, fine_n))

    # fill with random feasible vectors
    while len(pop) < 30:
        rand_vec = np.random.rand(fine_n)
        pop.append(_project_to_feasible(rand_vec, fine_n))

    # ---- GA phase (major part of budget) ----
    time_for_ga = budget_s * 0.85
    best_h, best_val = _ga_optimize(
        fine_n, pop, time_limit=time_for_ga, target=target_bound
    )

    # ---- local refinement & SLSQP ----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        # a quick SLSQP pass
        best_h, best_val = _solve_h(best_h, max_iter=800)
        # stochastic hill climbing
        best_h = _local_refine(best_h, attempts=20, delta=0.015)
        best_val = _c5_from_h(best_h)

    # ---- final fallback (should never happen) ----
    if best_h is None:
        fine_n = 96
        best_h = np.full(fine_n, 0.5)
        best_val = _c5_from_h(best_h)

    return best_h.tolist(), float(best_val), int(fine_n)
