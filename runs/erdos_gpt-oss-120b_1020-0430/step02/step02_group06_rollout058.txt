import time
import numpy as np

# ---------- feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling style loop."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- up‑sample ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- C5 computation (FFT based) ----------
def _max_corr_fft(h: np.ndarray) -> float:
    """Maximum of full correlation between h and (1-h) using FFT."""
    n = h.shape[0]
    # length needed for full convolution
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(np.pad(h, (0, size - n)))
    rev_comp = (1.0 - h)[::-1]
    R = np.fft.rfft(np.pad(rev_comp, (0, size - n)))
    conv = np.fft.irfft(H * R)[: 2 * n - 1]
    return float(conv.max())


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value from a discretised step function."""
    return _max_corr_fft(h) * (2.0 / h.size)


# ---------- gradient descent on active shift ----------
def _local_gradient_descent(h: np.ndarray, steps: int = 200, lr: float = 0.02) -> np.ndarray:
    """Monotone gradient descent on the shift attaining the maximal correlation."""
    n = h.size
    for _ in range(steps):
        # active shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(corr))
        shift = k_idx - (n - 1)          # shift = k - (n-1)
        grad = np.zeros(n, dtype=np.float64)

        if shift >= 0:
            i = np.arange(n - shift)
            grad[i] += 1.0 - h[i + shift]
            i2 = np.arange(shift, n)
            grad[i2] -= h[i2 - shift]
        else:  # shift < 0
            s = -shift
            i = np.arange(s, n)
            grad[i] += 1.0 - h[i + shift]
            i2 = np.arange(0, n - s)
            grad[i2] -= h[i2 + s]

        new_h = h - lr * grad
        new_h = _project_to_feasible(new_h, n)
        new_c5 = _c5_from_h(new_h)
        cur_c5 = _c5_from_h(h)
        if new_c5 < cur_c5:
            h = new_h
    return h


# ---------- simulated annealing ----------
def _anneal_optimize(
    n: int, time_limit: float, init_h: np.ndarray, rng: np.random.Generator
) -> tuple:
    """Run a simple simulated‑annealing on a vector of length n."""
    h = init_h.copy()
    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()

    # temperature schedule (linear)
    init_T = max(0.01, cur_c5 * 0.04)
    final_T = 1e-6

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = init_T * (1.0 - frac) + final_T * frac
        max_delta = 0.12 * (1.0 - frac) + 0.006 * frac
        # occasional larger jumps
        if rng.random() < 0.015:
            max_delta = 0.35

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # direction of mass transfer
        if rng.random() < 0.5:
            cap = min(max_delta, 1.0 - h[i], h[j])
            if cap <= 0.0:
                continue
            delta = rng.random() * cap
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            cap = min(max_delta, h[i], 1.0 - h[j])
            if cap <= 0.0:
                continue
            delta = rng.random() * cap
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_c5 = _c5_from_h(new_h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                if best_c5 < 0.3808:
                    break

    # final polish with a few gradient steps
    refined = _local_gradient_descent(best_h, steps=300, lr=0.02)
    refined_c5 = _c5_from_h(refined)
    if refined_c5 < best_c5:
        best_h, best_c5 = refined, refined_c5
    return best_h, best_c5


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise the Erdős C5 bound and return (h_values, c5_bound, n_points)."""
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # parent construction if present
    try:
        parent = np.array(initial_h_values, dtype=float)
    except NameError:
        parent = None

    # choose a working resolution (larger gives more freedom)
    if parent is not None:
        base_n = max(256, (parent.size + 1) // 2 * 2)  # even, at least 256
    else:
        base_n = 256
    n = min(base_n, 512)  # keep it reasonable

    # initialise
    if parent is not None:
        h0 = _upscale_h(parent, n)
    else:
        h0 = rng.random(n)
    h0 = _project_to_feasible(h0, n)

    # split available time between two independent searches
    total_alloc = max(0.0, time_left() - 0.5)  # small reserve for final checks
    half = total_alloc / 2.0

    h_best, c5_best = _anneal_optimize(n, half, h0, rng)

    # a second run starting from a fresh random point (uses remaining time)
    if time_left() > 1.0:
        h_rand = rng.random(n)
        h_rand = _project_to_feasible(h_rand, n)
        h2, c5_2 = _anneal_optimize(n, time_left(), h_rand, rng)
        if c5_2 < c5_best:
            h_best, c5_best = h2, c5_2

    # final safeguard projection and exact C5 recomputation
    h_best = _project_to_feasible(h_best, n)
    c5_best = _c5_from_h(h_best)

    return h_best.tolist(), float(c5_best), int(n)
