def hill_climb(h_initial, time_limit):
    h = h_initial.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    start = time.time()
    while time.time() - start < time_limit:
        # compute correlation vector
        corr_vec = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr_vec)
        k = max_idx - (len(h) - 1)  # shift
        # compute b_shift
        b = np.ones_like(h)
        n = len(h)
        if k >= 0:
            b[:n - k] = 1 - h[k:]
        else:
            b[-k:] = 1 - h[:n + k]
        # compute dec and inc weights
        dec_weights = b * (h > 1e-12)
        inc_weights = (1 - b) * (h < 1 - 1e-12)
        # choose indices with positive weight
        if not np.any(dec_weights) or not np.any(inc_weights):
            # fallback random move
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            # ensure feasible delta
            max_delta = min(h[i], 1 - h[j])
            if max_delta < 1e-8:
                continue
            delta = max_delta * np.random.rand()
        else:
            i = np.argmax(dec_weights)
            j = np.argmax(inc_weights)

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    # adjust iteratively:
    # We can set all free variables (i.e., not at bound) equal update = diff / num_free.
    # But after updating may push some to bounds, need iterate.
    for _ in range(100):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        # clip and recompute diff
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:

    # if still residual, distribute among all not at bound
    if abs(h.sum() - target) > 1e-9:

        current = h.sum()
        if current != 0:
            h *= target / current
        h = np.clip(h, 0.0, 1.0)
    return h

def _solve_h(initial_h, max_iter=500):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

    def eq(x):
        h = x[:-1]
        return np.sum(h) - n/2.0

    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]

    opt = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    try:
        res = scipy.optimize.minimize(lambda x: x[-1], x0, method='SLSQP',
                                      bounds=bounds, constraints=cons, options=opt)
        if not res.success:
            h_opt = h0
        else:
            h_opt = res.x[:-1]
    except Exception:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

import time
import numpy as np
from scipy import optimize


def _project_to_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute proportionally
        scale = target / h.sum()
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    return h


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(initial_h: np.ndarray, max_iter: int = 500):
    """One SLSQP refinement starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:-1]) - n / 2.0

    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                                bounds=bounds, constraints=cons, options=opts)
        h_opt = res.x[:-1] if res.success else h0
    except Exception:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h_start: np.ndarray, time_limit: float):
    """Targeted greedy hill‑climbing for a fixed time budget."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit:
        # occasionally shake the system
        if np.random.rand() < 0.001:
            noise = 0.02 * np.random.randn(n)
            h += noise
            h = _project_to_feasible(h, n)

        corr = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(corr) - (n - 1)          # shift giving the max
        b = np.ones_like(h)
        if k >= 0:
            b[: n - k] = 1 - h[k:]
        else:
            b[-k:] = 1 - h[: n + k]

        dec_mask = h > 1e-12
        inc_mask = h < 1 - 1e-12
        dec_weights = b * dec_mask
        inc_weights = (1 - b) * inc_mask

        if dec_weights.max() > 0 and inc_weights.max() > 0:
            i = int(np.argmax(dec_weights))
            j = int(np.argmax(inc_weights))
            max_delta = min(h[i], 1 - h[j])
            if max_delta < 1e-12:
                i = np.random.randint(n)
                j = np.random.randint(n)
                while i == j:
                    j = np.random.randint(n)
                max_delta = min(h[i], 1 - h[j])
                if max_delta < 1e-12:
                    continue
            delta = max_delta * 0.5
        else:
            i = np.random.randint(n)
            j = np.random.randint(n)
            while i == j:
                j = np.random.randint(n)
            max_delta = min(h[i], 1 - h[j])
            if max_delta < 1e-12:
                continue
            delta = max_delta * np.random.rand()

        h[i] -= delta
        h[j] += delta
        new_c5 = _c5_from_h(h)

        if new_c5 + 1e-12 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] += delta
            h[j] -= delta
            no_improve += 1
            if no_improve > 2000:
                # random restart to escape possible stagnation
                h = best_h + 0.01 * np.random.randn(n)
                h = _project_to_feasible(h, n)
                no_improve = 0

    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Base construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is not None:
        n0 = init_h.size
        h0 = _project_to_feasible(init_h, n0)
    else:
        n0 = 256
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        np.random.shuffle(h0)

    # ------------------------------------------------------------------
    # First SLSQP refinement (coarse grid)
    # ------------------------------------------------------------------
    if time_left() > 5.0:
        h_opt, c5_val = _solve_h(h0, max_iter=800)
    else:
        h_opt, c5_val = h0, _c5_from_h(h0)

    best_h = h_opt
    best_c5 = c5_val
    best_n = n0

    # ------------------------------------------------------------------
    # Multi‑resolution improvement
    # ------------------------------------------------------------------
    for n_target in (512, 1024):
        if time_left() < 8.0:
            break
        # upscale current best to the finer grid
        h_up = _upscale_h(best_h, n_target)
        h_up = _project_to_feasible(h_up, n_target)

        # short SLSQP pass
        if time_left() > 5.0:
            h_up, c5_up = _solve_h(h_up, max_iter=400)
        else:
            c5_up = _c5_from_h(h_up)

        # hill climbing on the finer grid
        remaining = time_left()
        hill_time = remaining * 0.85
        if hill_time > 2.0:
            h_climb, c5_climb = _hill_climb(h_up, hill_time)
        else:
            h_climb, c5_climb = h_up, c5_up

        # keep the best so far
        if c5_climb < best_c5:
            best_h, best_c5, best_n = h_climb, c5_climb, n_target

        # final quick SLSQP polishing if time permits
        if time_left() > 2.0:
            h_pol, c5_pol = _solve_h(best_h, max_iter=300)
            if c5_pol < best_c5:
                best_h, best_c5, best_n = h_pol, c5_pol, best_h.size

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final feasibility and bound computation
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_h.size)
