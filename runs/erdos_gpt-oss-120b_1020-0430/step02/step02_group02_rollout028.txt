def project_to_feasible(h):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    # water-filling
    # use iterative approach until sum close enough
    # We'll use while loop with tolerance.
    sum_h = h.sum()
    diff = sum_h - target
    # We'll adjust by distributing diff equally among entries that are not at bounds.
    # Use while loop.
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) <= 1e-10:
            break
        if diff > 0:
            # need to subtract diff
            free_idx = np.where(h > 0)[0]
        else:
            # need to add
            free_idx = np.where(h < 1)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        # apply delta with clipping
        if diff > 0:
            # subtract
            h[free_idx] -= delta
        else:
            h[free_idx] -= delta  # delta negative => addition
        # Clip again
        h = np.clip(h, 0.0, 1.0)
        sum_h = h.sum()
        diff = sum_h - target
    # final correction if small residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first not full/empty entries
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h,0.0,1.0)

def hill_climb(h, time_limit, steps=2000):
    n = h.size
    best_h = h.copy()
    best_c = c5_from_h(best_h)
    while time.time() < time_limit and steps > 0:
        steps -= 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # define max epsilon
        max_down = best_h[i]
        max_up = 1 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        # no need to project smoothly because eps ensures within bounds; sum unchanged.
        c = c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

import time
import numpy as np
from scipy import optimize

# Basic utilities

def _project_to_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=len(h)/2 using water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a step function (integral approximated by samples)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _max_corr_and_shifts(h: np.ndarray):
    """Return maximum correlation value and the list of shifts achieving it."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    ks = np.where(corr == maxc)[0] - (h.size - 1)
    return maxc, ks


def _gradient_for_shift(k: int, h: np.ndarray) -> np.ndarray:
    """Gradient of correlation_{k} w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        nk = n - k
        g[:nk] += 1.0 - h[k:]
        g[k:] -= h[:nk]
    else:
        p = -k
        np_ = n - p
        g[:np_] -= h[p:]
        g[p:] += 1.0 - h[:np_]
    return g


def _gradient_step(h: np.ndarray, alpha: float) -> np.ndarray:
    """One gradient step aimed at reducing the maximal correlation."""
    _, ks = _max_corr_and_shifts(h)
    g = np.zeros_like(h)
    for k in ks:
        g += _gradient_for_shift(k, h)
    g /= len(ks)
    h_new = h - alpha * g
    return _project_to_feasible(h_new)


def _gradient_descent(h: np.ndarray, time_limit: float,
                     init_alpha: float = 0.5,
                     min_alpha: float = 1e-5,
                     max_iter: int = 5000):
    """Simple sub‑gradient descent with back‑tracking step size."""
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    alpha = init_alpha
    it = 0
    while time.time() < time_limit and alpha >= min_alpha and it < max_iter:
        it += 1
        cand = _gradient_step(best_h, alpha)
        cand_c = _c5_from_h(cand)
        if cand_c + 1e-12 < best_c:
            best_h, best_c = cand, cand_c
            alpha = min(alpha * 1.1, init_alpha)
        else:
            alpha *= 0.5
    return best_h, best_c


def _hill_climb(h: np.ndarray, time_limit: float, steps: int = 5000):
    """Random epsilon exchanges, accepting only improvements."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    while time.time() < time_limit and steps > 0:
        steps -= 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c + 1e-12 < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


def _resize_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function onto a different grid."""
    old_n = h.size
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return _project_to_feasible(h_new)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400):
    """SLSQP refinement of a feasible vector."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt)
    return h_opt, _c5_from_h(h_opt)
