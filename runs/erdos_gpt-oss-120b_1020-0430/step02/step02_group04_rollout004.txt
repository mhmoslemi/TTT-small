def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax(x):

        bc = beta * corr
        m = np.max(bc)

        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    # Implementation: binary search on lambda for projection onto box with sum
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + tol) & (proj < upper - tol)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj
```

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale

    new_h[i] -= eps

def smooth_max_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    C = corr * (2.0 / n)
    bc = beta * C
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / np.sum(exp_bc)
    g = np.ones(n)
    # loop over lags
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] -= wk * h[k:]
            g[k:] -= wk * h[: n - k]
        else:
            kk = -k
            g[kk:] -= wk * h[: n - kk]
            g[: n - kk] -= wk * h[kk:]
    grad = (2.0 / n) * g
    return grad
```

    """Discrete C5 for step function h with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + tol) & (proj < upper - tol)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random vector in [0,1] with prescribed sum = n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a small amount from one coordinate to another (keeps the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple stochastic hill‑climb."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth‑max gradient ----------
def smooth_max_grad(h, beta):
    """Gradient of (1/β)·log‑sum‑exp of the overlap integrals."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    C = corr * (2.0 / n)                 # scaled overlaps
    bc = beta * C
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= w.sum()
    g = np.ones(n)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] -= wk * h[k:]
            g[k:]      -= wk * h[: n - k]
        else:
            kk = -k
            g[kk:]    -= wk * h[: n - kk]
            g[: n - kk] -= wk * h[kk:]
    return (2.0 / n) * g


def grad_descent_smooth_max(h0, beta_sched, steps_sched, rng):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    v = np.zeros_like(h)          # momentum
    mu = 0.9
    for beta, steps in zip(beta_sched, steps_sched):
        step = 0.2
        for _ in range(steps):
            grad = smooth_max_grad(h, beta)
            v = mu * v - step * grad
            new_h = h + v
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
                v = np.zeros_like(h)
            else:
                if rng.random() < 0.001:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- SLSQP refinement ----------
def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the soft‑max surrogate."""
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, target)


# ---------- up‑sampling ----------
def upscale_h(h, target_n):
    """Linear interpolation to a larger grid (preserves the integral)."""
    n0 = h.size
    if n0 == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_box_sum(h_new, target_n / 2.0)


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    N = 200                                 # target resolution (≤200)

    # starting vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = upscale_h(h0, N) if h0.size != N else project_onto_box_sum(h0, N / 2.0)
    else:
        h0 = random_feasible(N, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # quick hill‑climb on the parent (if time permits)
    if time.time() - start < budget_s * 0.1:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2500, start_step=0.4)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # optimisation schedule
    beta_sched = [5, 20, 80, 300, 1000, 3000]
    steps_sched = [150, 150, 150, 250, 350, 500]   # total ≈ 1550 gradient steps

    # multi‑restart loop
    while time.time() - start < budget_s * 0.85:
        # generate a fresh start (mixing with current best occasionally)
        if rng.random() < 0.3:
            h_start = (best_h + random_feasible(N, rng)) * 0.5
            h_start = project_onto_box_sum(h_start, N / 2.0)
        else:
            h_start = random_feasible(N, rng)

        # gradient descent on smooth‑max
        h_opt, c_opt = grad_descent_smooth_max(h_start, beta_sched, steps_sched, rng)

        # SLSQP refinement (large β)
        for β in (1000, 3000):
            try:
                h_opt = slsqp_optimize(h_opt, beta=β, max_iter=300)
            except Exception:
                pass
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            if best_c <= 0.3808:                # early stop if we beat the target
                break

        # occasional extra hill‑climb for fine tuning
        if rng.random() < 0.2:
            h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=1500, start_step=0.3)
            if c_tmp < best_c:
                best_h, best_c = h_tmp, c_tmp
                if best_c <= 0.3808:
                    break

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
