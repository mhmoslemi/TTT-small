def gradient_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        # first term: 1 - h[i+s] for i in [0, n-1-s]
        grad[:n-s] += (1 - h[s:])
        # second term: -h[i] for i in [0, n-1-s] contributes to grad[i+s]
        grad[s:] += -h[:n-s]

        # Derivative w.r.t h[i] for i >= t: (1 - h[i - t])
        grad[t:] += (1 - h[:n-t])
        # Derivative w.r.t h[i - t] for i >= t: -h[i]
        grad[:n-t] += -h[t:]
    return grad

def _project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    # iterative water-filling
    # Compute free indices where 0<h<1
    for _ in range(100):
        diff = target_sum - s
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # no free vars to adjust, just clip extremes
            # distribute diff to all entries up/down and clip
            # but this case occurs only if all at bounds, but sum mismatch can't be fixed.
            # So break.
            break
        delta = diff / free.size
        h[free] += delta
        # Clip after adjustment
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
    # final correction if residual remains due to rounding
    residual = target_sum - s
    if abs(residual) > 1e-12:
        for i in range(len(h)):
            if residual > 1e-12 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < -1e-12 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def _project_sum_half(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to have sum = len(h)/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 value for a feasible step function."""
    n = h.size
    r = np.correlate(h, 1.0 - h, mode="full")
    return r.max() * (2.0 / n)


def _gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of correlation for a single shift s."""
    n = h.size
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1 - h[s:])
        g[s:] += -h[:n - s]
    else:
        t = -s
        g[t:] += (1 - h[:n - t])
        g[:n - t] += -h[t:]
    return g


def _worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient for the maximal correlation shift."""
    n = h.size
    r = np.correlate(h, 1.0 - h, mode="full")
    max_r = r.max()
    idxs = np.where(np.abs(r - max_r) < 1e-12)[0]
    if idxs.size == 0:
        return np.zeros_like(h)
    grad = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)
        grad += _gradient_shift(h, s)
    grad /= idxs.size
    return grad


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse grid to a finer grid."""
    n_coarse = coarse.size
    x_coarse = np.arange(n_coarse) / n_coarse
    x_fine = np.arange(n_fine) / n_fine
    return np.interp(x_fine, x_coarse, coarse)


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random feasible vector of length n."""
    h = np.random.rand(n)
    return _project_sum_half(h)


# --------------------------------------------------------------
# Projected Gradient Descent on the worst shift
# --------------------------------------------------------------
def _pgd_optimize(h0: np.ndarray, budget: float, target: float = 0.3808) -> tuple:
    """Run PGD for at most *budget* seconds, return best (h, C5)."""
    start = time.time()
    n = h0.size
    h = _project_sum_half(h0.copy())
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    step0 = 0.2
    decay = 0.9995
    it = 0
    stall = 0
    while time.time() - start < budget:
        grad = _worst_shift_gradient(h)
        step = step0 * (decay ** it)
        h_new = _project_sum_half(h - step * grad)
        val = _c5_from_h(h_new)
        if val + 1e-12 < best_val:
            best_h, best_val = h_new, val
            h = h_new
            stall = 0
        else:
            stall += 1
            # drift a little to escape flat regions
            h = h_new
        it += 1
        if stall >= 250:
            # random perturbation around current best
            h = _project_sum_half(best_h + 0.05 * np.random.randn(n))
            stall = 0
        if best_val < target:
            break
    return best_h, best_val

# Simple pairwise hill climbing (mass transfer)
# --------------------------------------------------------------
def _hill_climb(h0: np.ndarray, budget: float, target: float = 0.3808) -> tuple:
    """Transfer epsilon between random pairs, keep sum, improve C5."""
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_val = _c5_from_h(h)
    while time.time() - start < budget:
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # decide direction
        if np.random.rand() < 0.5:
            # move from j to i
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            # move from i to j
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        val = _c5_from_h(h_new)
        if val + 1e-12 < best_val:
            h, best_val = h_new, val
            if best_val < target:
                break
    return h, best_val


# --------------------------------------------------------------
# SLSQP refinement (same formulation as previous solver)
# --------------------------------------------------------------
def _solve_slsqp(h_init: np.ndarray, max_iter: int = 800) -> tuple:
    n = h_init.size
    h0 = _project_sum_half(h_init)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": ineq},
            {"type": "eq", "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_sum_half(h_opt)
    return h_opt, _c5_from_h(h_opt)
