import time
import numpy as np
from scipy import optimize

# ---------------------------------------------------------
# Projection onto the box [0,1]^n intersected with sum = target
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

# ---------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

# ---------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x: np.ndarray) -> float:
    return x[-1]

def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

# ---------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

# ---------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Load possible parent construction.
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Candidate grid sizes (prefer parent size, explore a few larger ones)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (192, 224, 256, 320, 384, 448):
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Build a base h for this resolution.
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        # Simple baseline evaluation.
        c5_val = _c5_from_h(base_h)
        if c5_val < best_c5:
            best_h, best_c5, best_n = base_h, c5_val, n
            if best_c5 < target_bound:
                break

        # Local refinement of the baseline.
        if time_left() > 8.0:
            h_opt, c5_opt = _solve_h(base_h, max_iter=500)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                if best_c5 < target_bound:
                    break

        # ------- Global search using a low‑dimensional Fourier basis ----------
        if time_left() < 30.0:
            continue

        m = 10  # number of cosine basis functions
        x = (np.arange(n) + 0.5) * (2.0 / n)
        basis = np.column_stack([np.cos(np.pi * (j + 1) * x / 2.0) for j in range(m)])

        def c5_from_params(a):
            h = 0.5 + basis @ a
            h = _project_to_simplex_box(h, n / 2.0)
            return _c5_from_h(h)

        # Differential Evolution (global optimisation)
        bounds = [(-0.5, 0.5)] * m
        de_opts = {
            "maxiter": 35,
            "popsize": 15,
            "seed": seed,
            "polish": False,
            "updating": "deferred",
        }
        result_de = optimize.differential_evolution(
            c5_from_params, bounds, **de_opts
        )
        a_best = result_de.x
        h_de = _project_to_simplex_box(0.5 + basis @ a_best, n / 2.0)
        c5_de = _c5_from_h(h_de)
        if c5_de < best_c5:
            best_h, best_c5, best_n = h_de, c5_de, n
            if best_c5 < target_bound:
                break

        # Local refinement of DE result.
        if time_left() > 8.0:
            h_opt2, c5_opt2 = _solve_h(h_de, max_iter=500)
            if c5_opt2 < best_c5:
                best_h, best_c5, best_n = h_opt2, c5_opt2, n
                if best_c5 < target_bound:
                    break

        # Slightly more stochastic exploration around the best found so far.
        attempts = 8
        sigma = 0.12
        for _ in range(attempts):
            if time_left() < 5.0:
                break
            rnd = np.random.rand(n) - 0.5  # centered uniform
            start = best_h + sigma * rnd
            start = _project_to_simplex_box(start, n / 2.0)
            h_opt3, c5_opt3 = _solve_h(start, max_iter=400)
            if c5_opt3 < best_c5:
                best_h, best_c5, best_n = h_opt3, c5_opt3, n
                if best_c5 < target_bound:
                    break
            sigma *= 0.88

        if best_c5 < target_bound:
            break

    # Fallback if nothing succeeded
    if best_h is None:
        n = 256
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
