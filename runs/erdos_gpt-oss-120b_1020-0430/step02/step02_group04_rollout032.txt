def binary_hill_climb(h_initial, rng, max_iters):
    h = h_initial.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for iter in range(max_iters):
        # compute overlap array to find worst shift
        O = np.correlate(h, 1 - h, mode='full')
        max_o = O.max()
        # if max_o < best_c: update best (should be <= current)
        if max_o < best_c - 1e-12:
            best_c = max_o
            best_h = h.copy()
        # Identify shift(s) giving max
        idxs = np.where(O == max_o)[0]
        # Choose one shift offset randomly
        idx = rng.choice(idxs)
        k = idx - (h.size - 1)
        # compute region length L = n - abs(k)
        L = h.size - abs(k)
        # Determine set of i in overlapping region that are 1 while partner is 0
        if k >= 0:
            h1 = h[:L]
            h2 = h[k:k+L]
        else:
            # shift left: i goes from -k .. n-1
            h1 = h[-k:h.size]
            h2 = h[:L]
        # mask for bad: h1=1, h2=0
        bad_mask = (h1 == 1) & (h2 == 0)
        # indices in original array for those positions
        if k >= 0:
            bad_i = np.nonzero(bad_mask)[0]
        else:
            bad_i = np.nonzero(bad_mask)[0] - k  # shift offset
        # if no bad, pick random 1 index
        if bad_i.size == 0:
            # fallback random swap
            i_candidates = np.where(h==1)[0]
            j_candidates = np.where(h==0)[0]
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            i = rng.choice(bad_i)
            # Choose j randomly among zeros not equal i
            zeros = np.where(h==0)[0]
            j = rng.choice(zeros)
        # propose swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        # compute new max overlap
        O_new = np.correlate(h_new, 1 - h_new, mode='full')
        max_o_new = O_new.max()
        # Accept if improved, else maybe accept with small probability (simulated annealing)
        if max_o_new < max_o:
            h = h_new
        else:
            # occasional uphill acceptance decreasing over time
            if rng.random() < 0.001:
                h = h_new
        # early exit if best_c already below target threshold e.g., 0.3808
        if best_c <= target_c:
            break
    return best_h, best_c

def binary_hill_climb(h_start, rng, max_iters=5000, target_c=None):
    h = h_start.copy()
    best_h = h.copy()
    best_raw = np.max(np.correlate(h, 1 - h, mode='full'))  # raw sum
    best_c = best_raw * (2.0 / h.size)

    for it in range(max_iters):
        # compute overlap
        O = np.correlate(h, 1 - h, mode='full')
        max_raw = O.max()
        max_idx = O.argmax()
        # optional early stop if max_raw * 2/n <= target_c
        if target_c is not None:
            if max_raw * (2.0 / h.size) <= target_c:
                best_c = max_raw * (2.0 / h.size)
                best_h = h.copy()
                break
        # determine shift
        n = h.size
        k = max_idx - (n - 1)

        # compute overlapping region size
        L = n - abs(k)
        if L <= 0:
            # can't happen
            continue
        # compute bad mask
        if k >= 0:
            # i in [0, L-1]

            h2 = h[k:k+L]
            bad_i = np.where((h1 == 1) & (h2 == 0))[0]

            # i in [-k, n-1], but we map to original indices
            h1 = h[-k:n]
            h2 = h[:L]
            bad_i_local = np.where((h1 == 1) & (h2 == 0))[0]
            # convert to original indices: i = -k + idx

        if bad_i.size == 0:
            # fallback random 1 and 0
            idx1 = rng.choice(np.where(h == 1)[0])
            idx0 = rng.choice(np.where(h == 0)[0])
        else:

import time
import numpy as np

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


# ------------------------------------------------------------
# random binary generator (exact weight)
# ------------------------------------------------------------
def random_binary(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h


# ------------------------------------------------------------
# binary hill‑climb focused on the worst shift
# ------------------------------------------------------------
def binary_hill_climb(h_start, rng, max_iters=3000, target_c=None):
    """
    h_start : binary (0/1) numpy array, weight = n/2
    Returns best binary array and its C5 (scaled).
    """
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_raw = np.max(np.correlate(h, 1 - h, mode="full"))
    best_c = best_raw * (2.0 / n)

    for it in range(max_iters):
        # current overlap profile
        O = np.correlate(h, 1 - h, mode="full")
        raw_max = O.max()
        if target_c is not None and raw_max * (2.0 / n) <= target_c:
            best_raw = raw_max
            best_c = raw_max * (2.0 / n)
            best_h = h.copy()
            break
        idx_max = O.argmax()
        k = idx_max - (n - 1)                # shift giving the current worst overlap
        L = n - abs(k)                       # length of overlapping region

        # locate a "bad" 1‑0 pair for that shift
        if k >= 0:
            h1 = h[:L]
            h2 = h[k:k + L]
            bad_local = np.where((h1 == 1) & (h2 == 0))[0]
        else:
            h1 = h[-k:n]
            h2 = h[:L]
            bad_local = np.where((h1 == 1) & (h2 == 0))[0]
        if bad_local.size > 0:
            i = rng.choice(bad_local)
            # original index of the chosen 1
            i_orig = i if k >= 0 else i - k
            # try to swap with its partner if possible
            j_candidate = i_orig + k
            if 0 <= j_candidate < n and h[j_candidate] == 0:
                j = j_candidate
            else:
                zeros = np.where(h == 0)[0]
                j = rng.choice(zeros)
        else:
            # no bad pair (unlikely), just random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # perform the swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        # evaluate new worst overlap
        O_new = np.correlate(h_new, 1 - h_new, mode="full")
        raw_new = O_new.max()

        # accept if improves the worst overlap
        if raw_new < raw_max:
            h = h_new
            if raw_new < best_raw:
                best_raw = raw_new
                best_c = raw_new * (2.0 / n)
                best_h = h_new.copy()
        else:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.001:
                h = h_new

    return best_h, best_c

# small continuous hill‑climb (used for final polishing)
# ------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_h = project_onto_box_sum(new_h, target)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
