def shape_h(a,b,n):
    c = 2 - 2*a - 2*b
    if c < 0: c=0 (should not happen).
    denom = 2 - 2*a - b
    if denom <= 0: denom = 1e-12
    h_max = 1.0 / denom
    x = (np.arange(n)+0.5)*(2.0/n)
    h = np.zeros_like(x)
    # region definitions
    left = a
    up_end = a + b
    plateau_end = up_end + c
    right = 2 - a  # start of right zero region

    # ramp up
    if b > 1e-12:
        mask_up = (x >= left) & (x < up_end)
        h[mask_up] = ((x[mask_up] - left) / b) * h_max
    # plateau
    if c > 1e-12:
        mask_plateau = (x >= up_end) & (x < plateau_end)
        h[mask_plateau] = h_max
    # ramp down
    if b > 1e-12:
        mask_down = (x >= plateau_end) & (x < right)
        h[mask_down] = ((right - x[mask_down]) / b) * h_max
    # else region (zero)
    # now project
    return _project_to_simplex_box(h, n/2.0)

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the intersection of [0,1]^n and sum(v)=target."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] (dx = 2/n)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Refine a feasible h using SLSQP (same as previous baseline)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _make_h_from_params(a: float, b: float, n: int) -> np.ndarray:
    """Symmetric shape with zero padding a, ramp length b, plateau c, ramp down b."""
    # ensure feasibility
    if a < 0.0:
        a = 0.0
    if a > 0.5:
        a = 0.5
    max_b = max(0.0, 1.0 - 2.0 * a)
    if b < 0.0:
        b = 0.0
    if b > max_b:
        b = max_b

    c = 2.0 - 2.0 * a - 2.0 * b
    if c < 0.0:
        c = 0.0

    denom = 2.0 - 2.0 * a - b
    if denom <= 0.0:
        denom = 1e-12
    h_max = 1.0 / denom

    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.zeros_like(x)

    right_start = 2.0 - a
    plateau_end = a + b + c

    if b > 1e-12:
        mask_up = (x >= a) & (x < a + b)
        h[mask_up] = ((x[mask_up] - a) / b) * h_max
