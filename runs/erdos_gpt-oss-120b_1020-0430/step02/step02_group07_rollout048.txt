def project_onto_box_simplex(v, target):
    lo = v.min() - 1 # lower bound for lambda; but we can start with -1
    hi = v.max()    # high bound.
    for _ in range(50):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s > 0:
        h *= target / s
    else:
        h[:] = 0.0
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def project_box_simplex(v: np.ndarray, target: float, max_iter: int = 80, tol: float = 1e-12) -> np.ndarray:
    """Projection onto {x | 0≤x_i≤1, Σx_i = target} (Euclidean)."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            adj = residual / free.sum()
            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
        else:  # rare case: all at bounds, just adjust a single entry safely
            i = np.argmin(np.abs(x - 0.5))
            x[i] = np.clip(x[i] + residual, 0.0, 1.0)
    return x


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        # try opposite direction
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0.0:
            return h
        eps = rng.random() * max_transfer * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h
    eps = rng.random() * max_transfer * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def upsample_and_project(h: np.ndarray, factor: int, rng: np.random.Generator) -> np.ndarray:
    """Linear up‑sampling by an integer factor, then project to feasibility."""
    n = h.size
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, n * factor)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target = (n * factor) / 2.0
    return project_box_simplex(h_up, target)


# ------------------------------------------------------------
# Core optimisation routine (targeted local search)
# ------------------------------------------------------------
def targeted_refine(h_start: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Greedy hill‑climb that always attacks the shift attaining the current maximum."""
    t0 = time.time()
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    cur_h = h_start.copy()
    cur_c = best_c
    step_scale = 0.5
    n = cur_h.size

    while time.time() - t0 < time_budget:
        # correlation & max shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(corr.argmax())
        k = max_idx - (n - 1)

        # indices involved in this shift
        if k >= 0:
            i_idx = np.arange(0, n - k)
            j_idx = i_idx + k
        else:
            i_idx = np.arange(-k, n)
            j_idx = i_idx + k

        # product h_i * (1 - h_j) – the contribution to the max term
        prod = cur_h[i_idx] * (1.0 - cur_h[j_idx])
        local_max = int(prod.argmax())
        i0 = i_idx[local_max]
        j0 = j_idx[local_max]

        # transfer amount (keeps feasibility)
        max_eps = min(cur_h[i0], 1.0 - cur_h[j0])
        if max_eps > 1e-12:
            eps = max_eps * step_scale
            new_h = cur_h.copy()
            new_h[i0] -= eps
            new_h[j0] += eps
        else:
            new_h = perturb_h(cur_h, rng, scale=0.2)

        new_c = compute_c5(new_h)

        # accept improvements (or rare uphill moves)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        elif rng.random() < 0.001:
            cur_h, cur_c = new_h, new_c

        # decay step size, occasional restart
        step_scale = max(step_scale * 0.9995, 0.001)
        if rng.random() < 0.001:
            cur_h, cur_c = best_h.copy(), best_c
            step_scale = 0.5

    return best_h, best_c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise base solution
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_simplex(h0, target)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts with rapid targeted refinement
    # ------------------------------------------------------------------
    phase1_end = start_time + budget_s * 0.30
    while time.time() < phase1_end:
        h_start = random_feasible_vector(n, rng)
        remain = max(0.0, phase1_end - time.time())
        # allocate at most 2 seconds per refinement
        refine_time = min(2.0, remain)
        h_ref, c_ref = targeted_refine(h_start, rng, refine_time)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample once and deep refinement
    # ------------------------------------------------------------------
    if time.time() < start_time + budget_s * 0.65:
        h_up = upsample_and_project(best_h, factor=2, rng=rng)
        n = h_up.size
        remaining = max(0.0, start_time + budget_s * 0.90 - time.time())
        # give up to ~30 seconds for this phase
        refine_time = min(30.0, remaining)
        h_ref, c_ref = targeted_refine(h_up, rng, refine_time)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 3 – optional second up‑sample if time permits
    # ------------------------------------------------------------------
    if time.time() < start_time + budget_s * 0.97 and n * 2 <= 800:
        h_up2 = upsample_and_project(best_h, factor=2, rng=rng)
        n = h_up2.size
        remaining = max(0.0, start_time + budget_s - time.time())
        refine_time = min(30.0, remaining)
        h_ref, c_ref = targeted_refine(h_up2, rng, refine_time)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    target = n / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_simplex(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
