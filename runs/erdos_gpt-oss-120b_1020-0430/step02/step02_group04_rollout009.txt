import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    n = h.size
    corr = np.convolve(h, (1 - h)[::-1], mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

def random_feasible_symmetric(n, rng):
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])

def enforce_symmetry(v):
    v = (v + v[::-1]) * 0.5
    v = project_onto_box_sum(v, v.size / 2.0)
    return v

def subgrad_optimize(h0, max_iters, init_step, rng):
    h = enforce_symmetry(h0.copy())
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        corr = np.convolve(h, (1 - h)[::-1], mode='full')
        max_corr = np.max(corr)
        mask = np.isclose(corr, max_corr, atol=1e-12)
        grads = np.zeros_like(h)
        cnt = 0
        for idx, m in enumerate(mask):
            if not m:
                continue
            cnt += 1
            k = idx - (n - 1)
            if k >= 0:
                grads[: n - k] += (1 - h[k:])
                grads[k:] -= h[: n - k]
            else:
                kk = -k
                grads[kk:] += (1 - h[: n - kk])
                grads[: n - kk] -= h[kk:]
        if cnt == 0:
            break
        g = grads / cnt
        new_h = h - step * g
        new_h = project_onto_box_sum(new_h, target)
        new_h = enforce_symmetry(new_h)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
        step = max(step * 0.9995, 1e-4)
        h = new_h
    return best_h, best_c

def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    h = enforce_symmetry(h0.copy())
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                break
            w /= w_sum
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, target)
            new_h = enforce_symmetry(new_h)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, best_h = c, new_h.copy()
            step = max(step * 0.9995, 1e-4)
            h = new_h
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=200):

def softmax_obj(x):

def compute_c5(h):

    corr = np.convolve(h, (1 - h)[::-1], mode="full")

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
