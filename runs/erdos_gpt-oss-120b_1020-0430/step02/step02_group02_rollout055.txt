def _pgd_optimize(initial_h, max_iter, step0, time_limit):
    n = len(initial_h)
    h = initial_h.copy()
    best_h = h.copy()
    best_c = _c5_from_h(h)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(c)
        c_val = c[max_idx] * (2.0 / n)
        # update best
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        # gradient
        lag = max_idx - (n - 1)
        grad = np.zeros_like(h)
        if lag >= 0:
            lc = n - lag
            grad[:lc] += 1.0 - h[lag:]
            grad[lag:] -= h[:lc]
        else:
            lc = n + lag  # where lag negative
            # lag = -l_neg
            l_neg = -lag
            grad[l_neg:] += 1.0 - h[:lc]
            grad[:lc] -= h[l_neg:]
        # step size schedule
        alpha = step0 / np.sqrt(it + 1)
        h = h - alpha * grad
        # project to feasible region
        h = _project_to_feasible(h, n)
    # final check after loop
    final_c = _c5_from_h(best_h)
    return best_h, final_c
```

def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to reduce sum
            # find indices where h > 0
            idx = h > 0
        else:
            idx = h < 1
        if not np.any(idx):
            break
        delta = diff / np.count_nonzero(idx)
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first element that can
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)
```

def _full_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n=L)
    G = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(H * np.conj(G)).real[:2 * n - 1]
    return corr
```

def _pgd_optimize(start_h, max_iter, step0, time_limit):
    n = len(start_h)
    h = start_h.copy()
    best_h = h.copy()
    best_c = _c5_from_h(h)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = _full_corr_fft(h)
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        if lag >= 0:
            lc = n - lag
            grad[:lc] += 1.0 - h[lag:]
            grad[lag:] -= h[:lc]
        else:
            lneg = -lag
            lc = n - lneg
            grad[lneg:] += 1.0 - h[:lc]
            grad[:lc] -= h[lneg:]
        # step size
        alpha = step0 / np.sqrt(it + 1)
        h = h - alpha * grad
        # projection
        h = _project_to_feasible(h, n)
        # evaluate
        c = _c5_from_h(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # optional early break if c is below target
        # but we continue
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = h > 0
        else:
            idx = h < 1
        cnt = np.count_nonzero(idx)
        if cnt == 0:
            break
        delta = diff / cnt
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _full_corr_fft(h):
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n=L)
    G = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return corr


def _c5_from_h(h):
    """Compute C₅ for a given step function."""
    n = h.shape[0]
    return np.max(_full_corr_fft(h)) * (2.0 / n)


def _pgd_optimize(start_h, max_iter, step0, time_limit):
    """Projected gradient descent on max‑correlation objective."""
    n = start_h.size
    h = start_h.copy()
    best_h, best_c = h.copy(), _c5_from_h(h)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = _full_corr_fft(h)
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        grad = np.zeros_like(h)
        if lag >= 0:
            lc = n - lag
            grad[:lc] += 1.0 - h[lag:]
            grad[lag:] -= h[:lc]
        else:
            lneg = -lag
            lc = n - lneg
            grad[lneg:] += 1.0 - h[:lc]
            grad[:lc] -= h[lneg:]
        alpha = step0 / np.sqrt(it + 1)
        h = h - alpha * grad
        h = _project_to_feasible(h, n)
        c = _c5_from_h(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if best_c < 0.3808:
            break
    return best_h, best_c


def _upscale_h(coarse_h, n_fine):
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _run_optimizer_for_n(n, init_h, deadline):
    """Perform PGD + binary hill‑climb on size‑n problem."""
    best_h, best_c = None, np.inf

    # 1) gradient descent from provided start
    if time.time() < deadline:
        h_pg, c_pg = _pgd_optimize(init_h, max_iter=6000,
                                   step0=0.5, time_limit=deadline)
        best_h, best_c = h_pg, c_pg

    # 2) binary hill‑climb for remaining time
    if time.time() < deadline - 0.5:
        h_bin = np.zeros(n, dtype=float)
        h_bin[np.random.choice(n, n // 2, replace=False)] = 1.0
        c_bin = _c5_from_h(h_bin)
        if c_bin < best_c:
            best_h, best_c = h_bin.copy(), c_bin
        while time.time() < deadline - 0.2:
            i, j = np.random.randint(n, size=2)
            if h_bin[i] == h_bin[j]:
                continue
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c_new = _c5_from_h(h_bin)
            if c_new < c_bin:
                c_bin = c_new
                if c_new < best_c:
                    best_h, best_c = h_bin.copy(), c_new
                if best_c < 0.3808:
                    break
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]

    # 3) final short PGD refinement if time permits
    if best_h is not None and time.time() < deadline - 0.1:
        sub_deadline = time.time() + min(5.0, deadline - time.time() - 0.05)
        h_ref, c_ref = _pgd_optimize(best_h, max_iter=4000,
                                      step0=0.5, time_limit=sub_deadline)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    return best_h, best_c
