    step_factor_up = 1.02

        corr = _corr_full_fft(h)
        maxc = np.max(corr)
        # compute smooth max
        # to improve numeric stability: subtract maxc
        exp_terms = np.exp(beta * (corr - maxc))
        sum_exp = np.sum(exp_terms)
        smooth = maxc + np.log(sum_exp) / beta
        # objective C5 value
        obj = smooth * (2.0 / n)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        # compute weights
        w = exp_terms / sum_exp
        # gradient
        grad = _gradient_fft_from_weights(h, w)  # already includes factor (2/n)
        # update
        h_new = h - step * grad
        h_new = _project_to_simplex_box(h_new, n/2.0)

        sum_exp_new = np.sum(exp_terms_new)
        smooth_new = maxc_new + np.log(sum_exp_new) / beta
        obj_new = smooth_new * (2.0 / n)

            h = h_new
            step = step * step_factor_up  # increase step a bit after improvement

            beta = 500.0

            step = 2.0
        if step < 1e-6:

    final_c5 = _c5_from_h(best_h)

    m = w.shape[0]  # 2n-1

import time
import numpy as np

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _corr_full_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    size = 2 * n - 1
    N = 1 << ((size - 1).bit_length())
    H = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    Hc = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(N - n)]))
    conv = np.fft.irfft(H * Hc, n=N)
    return conv[:size]

def _gradient_fft_from_weights(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    m = w.shape[0]          # = 2n-1
    L = 1 << ((n + m - 1 - 1).bit_length())
    a = np.fft.rfft(np.concatenate([1.0 - 2.0 * h, np.zeros(L - n)]))
    b = np.fft.rfft(np.concatenate([w[::-1], np.zeros(L - m)]))
    conv = np.fft.irfft(a * b, n=L)
    conv = conv[:n + m - 1]
    grad = conv[n - 1:n - 1 + n]
    return (2.0 / n) * grad

def _optimize_pgdh(start_h: np.ndarray, time_left) -> tuple:
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_obj = _c5_from_h(h)

    beta = 2.0
    beta_factor = 1.04
    step = 0.8
    step_up = 1.04
    step_down = 0.55
    max_iters = 2500
    stall = 0

    for it in range(max_iters):
        if time_left() < 0.8:
            break
        corr = _corr_full_fft(h)
        maxc = np.max(corr)
        exp_terms = np.exp(beta * (corr - maxc))
        sum_exp = np.exp(0.0) + np.sum(exp_terms) - 1.0  # ensure non‑zero
        smooth = maxc + np.log(sum_exp) / beta
        obj = smooth * (2.0 / n)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 400:
                break

        w = exp_terms / sum_exp
        grad = _gradient_fft_from_weights(h, w)

        h_new = h - step * grad
        h_new = _project_to_simplex_box(h_new, n / 2.0)

        # evaluate new point (smooth)
        corr_new = _corr_full_fft(h_new)
        maxc_new = np.max(corr_new)
        exp_new = np.exp(beta * (corr_new - maxc_new))
        sum_exp_new = np.exp(0.0) + np.sum(exp_new) - 1.0
        smooth_new = maxc_new + np.log(sum_exp_new) / beta
        obj_new = smooth_new * (2.0 / n)

        if obj_new < obj:

            step = min(step * step_up, 2.0)

        beta = min(beta * beta_factor, 500.0)

    # final exact evaluation of the best found
    final_c5 = _c5_from_h(best_h)
    return best_h, final_c5

def _hill_climb(h: np.ndarray, current_c5: float, time_left) -> tuple:
    n = h.size
    best_h = h.copy()
    best_val = current_c5
    while time_left() > 0.4:
        i, j = np.random.randint(n, size=2)
        if i == j:

        delta = (np.random.rand() - 0.5) * 0.02

        cand[j] = hj
