def slsqp_optimize(initial_h, max_iter=200):
    n = initial_h.size
    h0 = project_to_feasible(initial_h, n)
    # compute initial t = max correlation
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, np.array([t0])])
    bounds = [(0.0,1.0)]*n + [(0.0,None)]
    constraints = [{'type':'eq','fun':eq_constraint},
                   {'type':'ineq','fun':ineq_constraint}]
    options = {'maxiter': max_iter, 'ftol': 1e-9}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                            constraints=constraints, options=options)
    # accept result or fallback
    h_opt = res.x[:n] if res.success else h0
    h_opt = project_to_feasible(h_opt, n)
    c5 = c5_from_h(h_opt)
    return h_opt, c5
```

def simulated_annealing(h_start, time_limit):
    n = h_start.size
    h_cur = h_start.copy()
    cur_c = c5_from_h(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c
    start = time.time()
    total = time_limit
    T0 = best_c * 0.01 if best_c>0 else 0.001
    # ensure T0 > 0
    while time.time() - start < total:
        # compute temperature
        elapsed = time.time() - start
        T = T0 * (1 - elapsed/total)
        # if T too small, break?
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # decide direction: move from i to j
        max_eps = min(h_cur[i], 1.0 - h_cur[j])
        if max_eps <= 0:
            continue
        # random epsilon up to max_eps
        eps = np.random.rand() * max_eps
        if eps == 0:
            continue
        cand = h_cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = c5_from_h(cand)
        # acceptance
        if cand_c < cur_c:
            h_cur = cand
            cur_c = cand_c
            if cand_c < best_c:
                best_h = cand.copy()
                best_c = cand_c
        else:
            # probability
            if T > 0:
                prob = np.exp(-(cand_c - cur_c) / T)
                if np.random.rand() < prob:
                    h_cur = cand
                    cur_c = cand_c
    return best_h, best_c
```

def project_to_feasible(h, n, tol=1e-9, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual greedily
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def simulated_annealing(h_start, time_limit):
    n = h_start.size
    h_cur = h_start.copy()
    cur_c = c5_from_h(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c
    start = time.time()
    # initial temperature based on typical range
    T0 = best_c * 0.02 if best_c > 0 else 0.001
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # linear cooling
        T = T0 * (1.0 - elapsed / time_limit)
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # decide direction: move from i to j
        max_eps = min(h_cur[i], 1.0 - h_cur[j])
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps
        if eps == 0.0:
            continue
        cand = h_cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = c5_from_h(cand)
        # acceptance
        if cand_c < cur_c:
            h_cur = cand
            cur_c = cand_c
            if cand_c < best_c:
                best_h = cand.copy()
                best_c = cand_c
        else:
            if T > 0.0:
                prob = np.exp(-(cand_c - cur_c) / T)
                if np.random.rand() < prob:
                    h_cur = cand
                    cur_c = cand_c
    return best_h, best_c

            h_cur, cur_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
        else:
            if T > 0.0:
                prob = np.exp(-(cand_c - cur_c) / T)
                if np.random.rand() < prob:
                    h_cur, cur_c = cand, cand_c
    return best_h, best_c

# ----------------------------------------------------------------------
# Upscale a coarse step function to a finer grid (linear interpolation)
# ----------------------------------------------------------------------
def upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # try to obtain a parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # candidate grid sizes
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    base = init_h.size if init_h is not None else 80
    for mul in (1, 2, 3, 4):
        n = base * mul
        if n <= 256:
            n_candidates.append(n)
    n_candidates.extend([96, 112, 128, 160, 192, 224, 256])
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # iterate over resolutions
    for n in n_candidates:
        if time_left() < 8.0:
            break

        # build base h for this resolution
        if init_h is not None and init_h.size != n:
            base_h = upscale_h(init_h, n)
        elif init_h is not None:
            base_h = init_h.copy()
        else:
            base_h = np.full(n, 0.5)

        # deterministic local optimisation
        h_opt, c5_opt = slsqp_optimize(base_h, max_iter=250)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

        # simulated annealing on the local optimum
        if time_left() > 5.0:
            sa_time = min(time_left() * 0.3, 30.0)
            h_sa, c5_sa = simulated_annealing(h_opt, sa_time)
            if c5_sa < best_c5:
                best_h, best_c5, best_n = h_sa, c5_sa, n
                # quick refinement
                h_ref, c5_ref = slsqp_optimize(h_sa, max_iter=150)
                if c5_ref < best_c5:
                    best_h, best_c5, best_n = h_ref, c5_ref, n

        # stochastic restarts within remaining budget for this n
        attempts = 0
        while time_left() > 6.0 and attempts < 30:
            attempts += 1
            sigma = 0.3
            pert = sigma * np.random.randn(n)
            start_h = base_h + pert
            start_h = project_to_feasible(start_h)
            h_local, c5_local = slsqp_optimize(start_h, max_iter=150)
            if c5_local < best_c5:
                best_h, best_c5, best_n = h_local, c5_local, n
                if time_left() > 4.0:
                    sa_time = min(time_left() * 0.2, 20.0)
                    h_sa2, c5_sa2 = simulated_annealing(h_local, sa_time)
                    if c5_sa2 < best_c5:
                        best_h, best_c5, best_n = h_sa2, c5_sa2, n
            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # final polishing with any remaining time
    if best_h is not None and time_left() > 2.0:
        h_final, c5_final = simulated_annealing(best_h, time_left())
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
            best_n = best_h.size

    # fallback if nothing succeeded
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
