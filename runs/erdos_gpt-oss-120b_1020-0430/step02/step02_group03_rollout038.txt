def subgrad_optimize(h0, max_iter, alpha0, rng):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)  # shift can be negative
        # compute gradient
        g = np.zeros_like(h)
        if shift >= 0:
            for i in range(n - shift):
                g[i] += (1 - h[i + shift])
                g[i + shift] += -h[i]
        else:
            t = -shift
            for i in range(t, n):
                j = i - t
                g[i] += (1 - h[j])
                g[j] += -h[i]
        # step size schedule
        step = alpha0 / (1.0 + it * 0.01)
        # update
        h -= step * g
        # project onto [0,1]
        np.clip(h, 0.0, 1.0, out=h)
        # enforce sum exactly
        # Using same enforce_sum function
        h = enforce_sum(h, target, rng)
    return h, compute_c5(h)

def perturb_h(h, rng, scale=0.2):

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the SLSQP surrogate (used only in slsqp_optimize)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x, beta):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σ h = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)                     # uniform in [0,1)
    target = n / 2.0
    total = h.sum()
    if total > 0.0:
        h *= target / total               # scale to correct sum
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def round_to_binary(h):
    """Binary rounding: keep the largest n/2 entries as 1, rest 0."""
    n = h.size
    half = n // 2
    idx = np.argpartition(h, -half)[-half:]
    b = np.zeros_like(h)
    b[idx] = 1.0
    return b


def subgrad_optimize(h0, max_iter, alpha0, rng):
    """Sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)          # true lag (may be negative)

        # gradient of the active (max) term
        g = np.zeros_like(h)
        if shift >= 0:
            g[: n - shift] = 1.0 - h[shift:]
            g[shift:] = -h[: n - shift]
        else:
            t = -shift
            g[t:] = 1.0 - h[: n - t]
            g[: n - t] = -h[t:]

        # diminishing step size
        step = alpha0 / (1.0 + it * 0.01)
        h -= step * g

        # keep within bounds and fix the sum
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        for _ in range(10):               # quick correction (usually enough)
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h, compute_c5(h)


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h using SLSQP on the smooth log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction (numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Determine dimension and initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # binary rounding of the initial point – sometimes better
    b0 = round_to_binary(best_h)
    c_b0 = compute_c5(b0)
    if c_b0 < best_c:
        best_h, best_c = b0, c_b0

    # ------------------------------------------------------------------
    # 2. Main stochastic search (sub‑gradient + binary rounding)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        # start from a perturbed elite solution or a fresh random one
        if best_h is not None and rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.15)
        else:
            start = random_feasible_vector(n, rng)

        # sub‑gradient descent
        max_iter = 800
        # shrink iteration count if time is running out
        if time.time() - start_time > budget_s * 0.80:
            max_iter = max(200, int((budget_s - (time.time() - start_time)) * 10))

        h_opt, c_opt = subgrad_optimize(start, max_iter, alpha0=0.2, rng=rng)

        # try a binary version of the result
        h_bin = round_to_binary(h_opt)
        c_bin = compute_c5(h_bin)
        if c_bin < c_opt:
            h_opt, c_opt = h_bin, c_bin

        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # ------------------------------------------------------------------
    # 3. Final refinement with the smooth surrogate (SLSQP)
    # ------------------------------------------------------------------
    for beta in (5.0, 20.0, 80.0, 200.0):
        if time.time() - start_time > budget_s * 0.96:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 4. Clean‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
