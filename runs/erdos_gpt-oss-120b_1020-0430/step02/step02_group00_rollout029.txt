def binary_from_continuous(v):
    n = v.size

    return h

def simulate_annealing_binary(n, h0, time_limit):
    start = time.time()
    h = h0.copy()
    curr_max = np.max(np.correlate(h, 1 - h, mode='full'))
    best_h = h.copy()
    best_max = curr_max
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # maybe schedule cooling based on iterations log
        T = max(0.001, 0.5 * (1 - it / 200000.0))  # simple
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        hi, hj = h[i], h[j]
        h[i], h[j] = hj, hi  # swap
        new_max = np.max(np.correlate(h, 1 - h, mode='full'))
        delta = new_max - curr_max
        # if delta <=0 accept else probability
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            curr_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # reject revert
            h[i], h[j] = hi, hj
        # To avoid infinite loops we may break early if best_max reduces to target threshold.
        if best_max * (2.0 / n) <= 0.3808:
            # early stop
            break
    return best_h, best_max

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (all top level, no closures)
# ----------------------------------------------------------------------


def binary_from_continuous(v: np.ndarray) -> np.ndarray:
    """Return a binary vector with exactly half ones (floor for even length)."""
    n = v.size
    half = n // 2
    idx = np.argpartition(v, -half)[-half:]
    h = np.zeros(n, dtype=np.int8)
    h[idx] = 1
    return h


def random_binary(n: int) -> np.ndarray:
    """Uniform random binary vector with n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    half = n // 2
    h[:half] = 1
    np.random.shuffle(h)
    return h


def upscale_binary(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a coarse binary pattern to a finer grid and binarise."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False) + 0.5 / old_n
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False) + 0.5 / new_n
    interp = np.interp(x_new, x_old, old_h.astype(float))
    return binary_from_continuous(interp)


def max_corr_from_h(h: np.ndarray) -> int:
    """Maximum correlation Σ h[i]*(1‑h[i+k]) over all integer shifts."""
    # h is integer (0/1); 1-h also integer; np.correlate works in int domain.
    return int(np.max(np.correlate(h, 1 - h, mode="full")))


def simulate_annealing_binary(n: int, init_h: np.ndarray, time_budget: float) -> tuple:
    """
    Simple simulated‑annealing on binary vectors.
    Returns a pair (best_h, best_max_corr).
    """
    start = time.time()
    h = init_h.astype(np.int8).copy()
    curr_max = max_corr_from_h(h)
    best_h = h.copy()
    best_max = curr_max

    # annealing schedule parameters
    target_iters = 200_000
    T0 = 5.0
    it = 0

    while time.time() - start < time_budget:
        it += 1
        # linear cooling
        T = max(0.001, T0 * (1.0 - it / target_iters))
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue  # no effect on the sum constraint
        hi, hj = h[i], h[j]  # hi/ hj are 0/1 and different
        h[i], h[j] = hj, hi
        new_max = max_corr_from_h(h)
        delta = new_max - curr_max
        if delta <= 0 or np.random.rand() < math.exp(-delta / T):
            curr_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                # early stop if we already beat the target bound 0.3808
                if best_max * (2.0 / n) <= 0.3808:
                    break
        else:
            # revert the swap
            h[i], h[j] = hi, hj
        # safety break
        if time.time() - start > time_budget:
            break
    return best_h, best_max

def _project_to_feasible(x: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(x)=n/2 by water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(x > 0)[0]
        else:
            free = np.where(x < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    # tiny correction
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and x[i] < 1 - 1e-12:
                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 1e-12:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _c5_from_h_cont(x: np.ndarray) -> float:
    n = x.size
    max_corr = np.max(np.correlate(x, 1.0 - x, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h_continuous(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Run a single SLSQP optimisation starting from a feasible point."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h_cont(h_opt)
    return h_opt, c5_val

    # Final continuous refinement (optional, limited time)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if best_h is not None and remaining > 6.0:
        h_float = best_h.astype(float)
        refined_h, refined_c5 = _solve_h_continuous(h_float, max_iter=800)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
            best_n = refined_h.size

    # ------------------------------------------------------------------
    # Ensure feasibility and final bound
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h_cont(best_h)
    else:
        # enforce binary feasibility (half ones) if slight numerical drift occurs
        half = best_n // 2
        cur = int(np.rint(best_h).sum())
        best_h = np.rint(best_h).astype(np.int8)
        if cur > half:
            # turn excess ones to zero
            idx = np.flatnonzero(best_h)[: cur - half]
            best_h[idx] = 0
        elif cur < half:
            idx = np.flatnonzero(best_h == 0)[: half - cur]
            best_h[idx] = 1
        best_c5 = float(np.max(np.correlate(best_h, 1 - best_h, mode="full")) * (2.0 / best_n))

    return best_h.tolist(), best_c5, int(best_n)
