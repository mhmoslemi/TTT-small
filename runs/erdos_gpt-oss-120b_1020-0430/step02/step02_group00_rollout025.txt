import time, math
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Feasibility projection (clip to [0,1] and enforce sum = n/2)
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny clean‑up
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# --------------------------------------------------------------
# C5 evaluation
# --------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# --------------------------------------------------------------
# Upscaling (linear interpolation)
# --------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


# --------------------------------------------------------------
# SLSQP refinement (same structure as previous solution)
# --------------------------------------------------------------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# --------------------------------------------------------------
# Simple hill‑climbing that preserves the sum exactly
# --------------------------------------------------------------
def _hill_climb_optimize(h: np.ndarray, max_iter: int, time_left):
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    cur = h.copy()
    for _ in range(max_iter):
        if time_left() <= 0:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        eps = 0.05 * np.random.rand()
        if cur[i] + eps > 1.0 or cur[j] - eps < 0.0:
            continue
        new = cur.copy()
        new[i] += eps
        new[j] -= eps
        val = _c5_from_h(new)
        if val < best_val:
            best_val, best_h, cur = val, new.copy(), new
    return best_h, best_val


# --------------------------------------------------------------
# Sub‑gradient descent on the worst shift
# --------------------------------------------------------------
def _subgrad_optimize(h: np.ndarray, max_iter: int, step0: float, time_left):
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    cur = h.copy()
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)                     # shift that attains the max
        g = np.ones(n)
        if k >= 0:
            if k < n:
                g[: n - k] -= cur[k:]
                g[k:] -= cur[: n - k]
        else:
            kk = -k
            if kk < n:
                g[kk:] -= cur[: n - kk]
                g[: n - kk] -= cur[kk:]
        step = step0 / math.sqrt(it)
        nxt = cur - step * g
        nxt = _project_to_feasible(nxt, n)
        val = _c5_from_h(nxt)
        if val < best_val:
            best_val, best_h = val, nxt.copy()
        cur = nxt
    return best_h, best_val

# --------------------------------------------------------------
