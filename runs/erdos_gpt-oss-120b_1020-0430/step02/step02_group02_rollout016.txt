import time, numpy as np
from scipy import optimize

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    ...
def _c5_from_h(h):
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode='full')
def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n/2.0)
def _solve_h(initial_h, max_iter=800):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    constraints = [{'type':'ineq','fun':_ineq_constraint},
                   {'type':'eq','fun':_eq_constraint}]
    res = optimize.minimize(lambda x: x[-1], x0,
                            method='SLSQP',
                            bounds=bounds,
                            constraints=constraints,
                            options={'maxiter':max_iter,'ftol':1e-9,'disp':False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt,n)
    return h_opt, _c5_from_h(h_opt)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse)+0.5)*(2.0/n_coarse)
    x_fine = (np.arange(n_fine)+0.5)*(2.0/n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _simulated_annealing(h_start, time_deadline):
    n = h_start.size
    # compute initial C5
    cur_h = h_start.copy()
    cur_c = _c5_from_h(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    # temperature schedule parameters
    T = 0.005 # starting temperature
    Tf = 1e-7
    cooling_factor = 0.995
    step = 0
    while time.time() < time_deadline:
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # determine feasible epsilon magnitude
        # we move epsilon from j to i (positive epsilon)
        max_pos = min(1.0 - cur_h[i], cur_h[j])
        max_neg = min(cur_h[i], 1.0 - cur_h[j])
        if np.random.rand() < 0.5 and max_pos > 0:
            eps = np.random.rand() * max_pos * 0.5
        elif max_neg > 0:
            eps = -np.random.rand() * max_neg * 0.5
        else:
            continue
        cand = cur_h.copy()
        cand[i] += eps
        cand[j] -= eps
        cand_c = _c5_from_h(cand)
        delta = cand_c - cur_c
        if delta < 0 or np.random.rand() < np.exp(-delta / (T+1e-12)):
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # temperature update
        step += 1
        if step % 200 == 0:
            T = max(T * cooling_factor, Tf)
    return best_h, best_c

    n = h.size
    for _ in range(max_iters):

        max_idx = np.argmax(corr)

        # compute derivative array
        deriv = np.zeros(n)
        # for i where i+shift in range:
        if shift >= 0:
            i_range = np.arange(n-shift)
            deriv[i_range] = (1.0 - h[i_range + shift]) - 0.0  # missing h[i-shift] term? Actually i - shift <0, so only (1 - h[i+shift]) term?

            # So we need also for i where i-shift >=0.
            deriv[i_range + shift] += -h[i_range]  # for positions i+shift where i-shift is i_range
        else:
            shift_abs = -shift
            i_range = np.arange(shift_abs, n)
            deriv[i_range] = (1.0 - h[i_range + shift]) - 0.0  # i+shift in range
            deriv[i_range - shift_abs] += -h[i_range]  # similar

            if 0 <= j < n:
                d += (1.0 - h[j])
            k = i - shift
            if 0 <= k < n:
                d -= h[k]
            deriv[i] = d
        # Now we want to decrease h where deriv > 0 (since increasing h would increase C)
        pos_indices = np.where(deriv > 1e-12)[0]

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Feasibility utilities
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip h into [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a step function h (integral version)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# --------------------------------------------------------------
# SLSQP optimisation (epigraph formulation)
# --------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every integer shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Refine a candidate h with SLSQP (minimise the epigraph variable)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# --------------------------------------------------------------
# Helper for different resolutions
# --------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# --------------------------------------------------------------
# Gradient‑descent on the worst shift
# --------------------------------------------------------------
def _reduce_max_shift(h: np.ndarray, steps: int = 2000, max_step: float = 0.02) -> np.ndarray:
    """Iteratively move mass from positions that increase the worst correlation
    to positions that decrease it, preserving the total integral."""
    n = h.size
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)                     # integer offset
        idx = np.arange(n)
        plus = idx + shift
        minus = idx - shift
        deriv = np.zeros(n)
        valid_plus = (plus >= 0) & (plus < n)
        valid_minus = (minus >= 0) & (minus < n)
        deriv[valid_plus] += (1.0 - h[plus[valid_plus]])
        deriv[valid_minus] -= h[minus[valid_minus]]

        pos = np.where(deriv > 1e-12)[0]
        neg = np.where(deriv < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(deriv[pos])]          # decreasing this reduces C5
        j = neg[np.argmin(deriv[neg])]          # increasing this reduces C5
        transfer = min(h[i], 1.0 - h[j], max_step)
        if transfer <= 0:
            break
        h[i] -= transfer
        h[j] += transfer
    return h


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with the required integral."""
    v = np.random.rand(n)
    return _project_to_feasible(v, n)

# Main optimisation driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small C5 bound."""
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load optional initial construction (n = 80 by default)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c = float("inf")
    best_n = None

    # Candidate grid sizes (even, modest)
    candidates = [80, 96, 112, 128, 160, 192, 224, 256]

    for n in candidates:
        if time_left() < 10.0:
            break

        # ----- build a base h for this resolution --------------------
        if init_h is not None:
            if init_h.size != n:
                base_h = _upscale_h(init_h, n)
            else:
                base_h = init_h.copy()
        else:
            base_h = _random_feasible(n)

        # ----- deterministic SLSQP refinement -------------------------
        h_opt, c_opt = _solve_h(base_h, max_iter=800)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n

        # ----- allocate a slice of the remaining budget for stochastic search
        remaining = time_left()
        if remaining < 5.0:
            continue
        slice_time = min(remaining * 0.3, 30.0)
        slice_deadline = time.time() + slice_time

        sigma = 0.5
        # ----- stochastic restarts ------------------------------------
        while time.time() < slice_deadline:

            start_h = _project_to_feasible(base_h + perturb, n)
            h_local, c_local = _solve_h(start_h, max_iter=600)

            h_gd = _reduce_max_shift(h_local.copy(), steps=800, max_step=0.015)

            # final polish
            h_final, c_final = _solve_h(h_gd, max_iter=600)
