        grad = 1.0 - left - right

import time, numpy as np, scipy.optimize as opt

def _project_to_simplex_box(v, target):
    # same binary search method
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n/2.0)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _c5_from_h(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.fft(h, L)
    b_fft = np.fft.fft((1.0 - h)[::-1], L)
    conv = np.fft.ifft(a_fft * b_fft).real
    corr = conv[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def _max_corr_shift(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.fft(h, L)
    b_fft = np.fft.fft((1.0 - h)[::-1], L)

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x):
    return x[-1]

def _solve_h(start_h, max_iter=500):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{"type": "ineq", "fun": _ineq_constraint},
                   {"type": "eq", "fun": _eq_constraint}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                       constraints=constraints, options=opts)
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape on n points, projected to feasible set."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse_h)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C₅ from h using FFT based correlation."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.fft(h, L)
    b_fft = np.fft.fft((1.0 - h)[::-1], L)
    conv = np.fft.ifft(a_fft * b_fft).real
    corr = conv[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def _max_corr_shift(h: np.ndarray):
    """Return (max correlation value, corresponding shift integer)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.fft(h, L)
    b_fft = np.fft.fft((1.0 - h)[::-1], L)
    conv = np.fft.ifft(a_fft * b_fft).real
    corr = conv[: 2 * n - 1]
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    return corr[idx], shift


def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 600):
    """Refine h with SLSQP (auxiliary variable t)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                       constraints=cons, options=opts)
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _subgradient_descent(h0: np.ndarray, max_iter: int, time_budget: float, start_time: float):
    """Projected sub‑gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    step0 = 0.8  # initial step size
    for it in range(1, max_iter + 1):
        if time.time() - start_time > time_budget:
            break
        max_corr, shift = _max_corr_shift(h)
        cur_c5 = max_corr * (2.0 / n)

        # gradient of the active shift
        if shift >= 0:
            left = np.concatenate([np.zeros(shift), h[: n - shift]])
            right = np.concatenate([h[shift:], np.zeros(shift)])
        else:
            s = -shift
            left = np.concatenate([h[s:], np.zeros(s)])
            right = np.concatenate([np.zeros(s), h[: n - s]])
        grad = 1.0 - left - right

        step = step0 / np.sqrt(it)
        h_new = h - step * grad
        h_new = _project_to_simplex_box(h_new, n / 2.0)

        new_c5 = _c5_from_h(h_new)
        if new_c5 < cur_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # occasional uphill move (simulated‑annealing style)
            T = 0.01 / np.sqrt(it)
            if np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                h = h_new

        # small random mass exchange every few hundred steps
        if it % 300 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.02
            delta = max(-h[i], min(delta, 1 - h[i]))
            delta = min(delta, h[j])  # keep h[j]-delta >= 0
            h_tmp = h.copy()
            h_tmp[i] += delta
            h_tmp[j] -= delta
            h_tmp = _project_to_simplex_box(h_tmp, n / 2.0)
            tmp_c5 = _c5_from_h(h_tmp)
            if tmp_c5 < best_c5:
                h = h_tmp
                best_c5 = tmp_c5
                best_h = h.copy()
    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Parent construction, if supplied
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Grid sizes to explore
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (128, 160, 192, 224, 256, 320, 384):
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Build base function for this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        # Local refinement with SLSQP
        h_opt, c5_val = _solve_h(base_h, max_iter=600)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target:
                break

        # Sub‑gradient descent from the SLSQP result
        if time_left() > 2.0:
            h_sd, c5_sd = _subgradient_descent(
                h_opt, max_iter=3000, time_budget=budget_s, start_time=start_time
            )
            if c5_sd < best_c5:
                best_h, best_c5, best_n = h_sd, c5_sd, n
                if best_c5 < target:
                    break
            # a short SLSQP polish after sub‑gradient
            if time_left() > 1.0:
                h_ref, c5_ref = _solve_h(h_sd, max_iter=300)
                if c5_ref < best_c5:
                    best_h, best_c5, best_n = h_ref, c5_ref, n
                    if best_c5 < target:
                        break

        # A few random restarts with sub‑gradient
        for _ in range(2):
            if time_left() < 2.0:
                break
            rnd = np.random.rand(n)
            rnd = _project_to_simplex_box(rnd, n / 2.0)
            h_sd2, c5_sd2 = _subgradient_descent(
                rnd, max_iter=2000, time_budget=budget_s, start_time=start_time
            )
            if c5_sd2 < best_c5:
                best_h, best_c5, best_n = h_sd2, c5_sd2, n
                if best_c5 < target:
                    break
        if best_c5 < target:
            break

    # Fallback (should rarely happen)
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
