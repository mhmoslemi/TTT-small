import time
import numpy as np

def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n with sum == target via bisection."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return _project_to_feasible(v, n / 2.0)


def _compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _perturb_transfer(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Move a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


def _hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                max_iters: int = 1500, start_step: float = 0.5) -> tuple:
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = _compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = _perturb_transfer(h, rng, scale=step)
        c = _compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def _softmax_weights(h: np.ndarray, beta: float):
    """Return correlation array and soft‑max weights w_k = exp(βc_k)/∑."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / exp_bc.sum()
    return corr, w


def _gradient_step(h: np.ndarray, beta: float, step_size: float) -> np.ndarray:
    """One gradient descent step on the soft‑max surrogate, followed by projection."""
    n = h.size
    corr, w = _softmax_weights(h, beta)

    grad = np.zeros_like(h)

    w_center = w[n - 1]
    grad -= 2.0 * w_center * h

    # positive and negative lags
    for l in range(1, n):
        wpos = w[n - 1 + l]
        wneg = w[n - 1 - l]
        if wpos != 0.0:
            grad[: n - l] -= wpos * h[l:]
        if wneg != 0.0:
            grad[l:] -= wneg * h[: n - l]

    new_h = h - step_size * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = _project_to_feasible(new_h, n / 2.0)
    return new_h


def _refine_with_gradient(h: np.ndarray, rng: np.random.Generator,
                          betas=(5.0, 20.0, 80.0, 200.0, 500.0),
                          steps_per_beta=200, base_step=0.3,
                          time_deadline=None) -> np.ndarray:
    """Apply a sequence of gradient‑descent phases with increasing β."""
    step = base_step
    for beta in betas:
        for _ in range(steps_per_beta):
            h = _gradient_step(h, beta, step)
            # small decay of step size
            step *= 0.9995
            if time_deadline is not None and time.time() > time_deadline:
                return h
    return h


def _interpolate(h_src: np.ndarray, n_dest: int) -> np.ndarray:
    """Linear interpolation of a step function to a different number of points."""
    src_n = h_src.size
    xs_src = np.linspace(0.0, 2.0, src_n, endpoint=False) + 1.0 / src_n
    xs_dst = np.linspace(0.0, 2.0, n_dest, endpoint=False) + 1.0 / n_dest
    h_interp = np.interp(xs_dst, xs_src, h_src)
    return _project_to_feasible(h_interp, n_dest / 2.0)
