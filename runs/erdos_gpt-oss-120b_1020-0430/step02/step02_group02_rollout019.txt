def _gradient_worst_shift(h):

    n = len(h)

    corr_full = np.correlate(h, 1 - h, mode='full')  # length 2n-1

    max_val = corr_full.max()

    worst_shifts = np.where(corr_full == max_val)[0]  # indices

    # convert shift indexes to integer shift relative to zero: shift = index - (n - 1)

    shifts = worst_shifts - (n - 1)

    grad = np.zeros_like(h)

    for s in shifts:

        if s >= 0:

            # term1: j in [0, n-s-1] => 1 - h[j+s]

            grad[:n - s] += 1 - h[s:]

            # term2: j in [s, n-1] => -h[j - s]

            grad[s:] -= h[:n - s]

        else:

            s = -s

            # shift negative

            # term1: j in [s, n-1] => 1 - h[j - s]

            grad[s:] += 1 - h[:n - s]

            # term2: j in [0, n - s - 1] => -h[j + s]

            grad[:n - s] -= h[s:]

    # gradient of O(k) computed; we want gradient of C5 (max O), same.

    return grad

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
#   Core utilities
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given step function h (dx = 2/n)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _project_to_feasible(h: np.ndarray, n: int, target: float = None, tol: float = 1e-9) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=target (default n/2) by a fast water‑filling."""
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) <= tol:
        return h
    # simple iterative redistribution (very few passes are enough)
    for _ in range(10):
        if abs(diff) <= tol:
            break
        if diff > 0.0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            h[free] += delta
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free[0]] += residual
    return np.clip(h, 0.0, 1.0)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse)


# ----------------------------------------------------------------------
#   Gradient of the worst overlap term
# ----------------------------------------------------------------------
def _worst_overlap_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of max_i ∑ h[j] (1‑h[j+i]) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    shift_idxs = np.where(corr == max_val)[0] - (n - 1)
    grad = np.zeros_like(h)
    for s in shift_idxs:
        if s >= 0:
            # term (1‑h[j+s])
            grad[: n - s] += 1.0 - h[s:]
            # term –h[j‑s]
            grad[s:] -= h[: n - s]
        else:
            t = -s
            # constant 1 for j < t (out‑of‑range h[j+s])
            grad[:t] += 1.0
            # term (1‑h[j‑t]) for j ≥ t
            grad[t:] += 1.0 - h[: n - t]
            # term –h[j+t] where j+t < n
            grad[: n - t] -= h[t:]
    return grad


# ----------------------------------------------------------------------
#   Projected gradient descent on the max‑overlap objective
# ----------------------------------------------------------------------
def _pgd_max_overlap(h0: np.ndarray, iters: int, deadline: float) -> tuple:
    """Projected gradient descent on C5 (max overlap)."""
    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), _c5_from_h(h)
    alpha = 0.5
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = _worst_overlap_gradient(h)
        cand = _project_to_feasible(h - alpha * g, n)
        c = _c5_from_h(cand)
        if c < best_c - 1e-12:
            best_c, best_h = c, cand.copy()
            h = cand
            alpha = min(alpha * 1.2, 1.0)
        else:
            alpha = max(alpha * 0.7, 1e-6)
    return best_h, best_c


# ----------------------------------------------------------------------
#   Simple hill‑climb (pairwise epsilon moves)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, steps: int, deadline: float) -> tuple:
    n = h.size
    best_h, best_c = h.copy(), _c5_from_h(h)
    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
#   SLSQP refinement (upper‑bound guarantee)
# ----------------------------------------------------------------------
class _IneqConstr:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        h = x[: self.n]
        t = x[self.n]
        return t - np.correlate(h, 1.0 - h, mode="full")

class _EqConstr:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[: self.n]) - (self.n / 2.0)

def _slsqp_refine(h0: np.ndarray, max_iter: int = 300) -> tuple:
    n = h0.size
    h0 = _project_to_feasible(h0, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _IneqConstr(n)},
        {"type": "eq", "fun": _EqConstr(n)},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
#   Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    #   Load parent construction if present
    # ------------------------------------------------------------------
    try:
        GLOBAL_INITIAL_H = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL_H = None

    # ------------------------------------------------------------------
    #   Candidate grid sizes (favor multiples of the parent size)
    # ------------------------------------------------------------------
    n_candidates = []
    if GLOBAL_INITIAL_H is not None:
        n_candidates.append(GLOBAL_INITIAL_H.size)
    for mul in (1, 2, 3, 4):
        if GLOBAL_INITIAL_H is not None:
            n = int(GLOBAL_INITIAL_H.size * mul)
        else:
            n = 64 * mul
        if n <= 512:
            n_candidates.append(n)
    n_candidates.extend([80, 96, 112, 128, 160, 192, 224, 256, 320, 384, 512])
    n_candidates = sorted(set(n_candidates))

    target = 0.3808
    best_h, best_c, best_n = None, float("inf"), None

    # ------------------------------------------------------------------
    #   Optimisation loop
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # base construction for this resolution
        if GLOBAL_INITIAL_H is not None and GLOBAL_INITIAL_H.size != n:
            base_h = _upscale_h(GLOBAL_INITIAL_H, n)
        elif GLOBAL_INITIAL_H is not None:
            base_h = GLOBAL_INITIAL_H.copy()
        else:
            base_h = np.full(n, 0.5)

        # several restarts per grid size
        restarts = 0
        while time_left() > 5.0 and restarts < 8:
            restarts += 1
            # random perturbation or fresh random start
            if restarts == 1:
                h_start = base_h.copy()
            else:
                h_start = np.random.rand(n)
                h_start = _project_to_feasible(h_start, n)

            # 1) projected gradient descent
            pgd_dead = start + budget_s - 0.5
            h_gd, c_gd = _pgd_max_overlap(h_start, iters=2000, deadline=pgd_dead)

            # 2) hill‑climb refinement
            hc_dead = start + budget_s - 0.3
            h_hc, c_hc = _hill_climb(h_gd, steps=2000, deadline=hc_dead)

            # 3) final SLSQP polish
            h_opt, c_opt = _slsqp_refine(h_hc, max_iter=300)

            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n
                if best_c < target:
                    break
        if best_c < target:
            break

    # ------------------------------------------------------------------
    #   Fallback
    # ------------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c), int(best_n)
