def improve_h(h_init, rng, start_time, budget_s):
    h = h_init.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # simulated annealing schedule
    temp = 0.01
    decay = 0.9995
    while time.time() - start_time < budget_s * 0.02:  # small slice per call? Actually allow for many iterations per restart.
        # find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        if max_val * (2.0 / n) <= best_c:
            # reached best already maybe
            pass
        k = max_idx - (n - 1)
        # find source and target indices
        if k >= 0:
            i_range = np.arange(n - k)
            src_mask = (h[i_range] == 1) & (h[i_range + k] == 0)
            tgt_mask = (h[i_range] == 0) & (h[i_range + k] == 1)
            src_indices = i_range[src_mask]
            tgt_indices = i_range[tgt_mask]
        else:
            d = -k
            i_range = np.arange(d, n)
            src_mask = (h[i_range] == 1) & (h[i_range - d] == 0)
            tgt_mask = (h[i_range] == 0) & (h[i_range - d] == 1)
            src_indices = i_range[src_mask]
            tgt_indices = i_range[tgt_mask]
        if src_indices.size > 0 and tgt_indices.size > 0:
            i = int(rng.choice(src_indices))
            j = int(rng.choice(tgt_indices))
        else:
            # random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        # accept if improved or with temperature
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / (temp + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # temperature decay
        temp = max(temp * decay, 1e-6)
        # optionally break if time too low
        if time.time() - start_time > budget_s * 0.95:
            break
    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def binary_from_float(x, weight, rng):
    """Convert a real vector to a {0,1} vector with exactly `weight` ones."""
    # take the largest entries as ones
    idx = np.argsort(-x, kind="stable")
    b = np.zeros_like(x, dtype=int)
    b[idx[:weight]] = 1
    # if we need fractional weight (odd n) we fall back to a simple rounding
    if weight * 2 != x.size:
        # adjust by one element if necessary
        deficit = weight - b.sum()
        if deficit > 0:
            extra = rng.choice(np.where(b == 0)[0], deficit, replace=False)
            b[extra] = 1
        elif deficit < 0:
            drop = rng.choice(np.where(b == 1)[0], -deficit, replace=False)
            b[drop] = 0
    return b


def greedy_improve(h, rng, deadline):
    """Greedy/annealing improvement of binary h to lower C5."""
    n = h.size
    target = n // 2
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = 0.02
    decay = 0.9995

    while time.time() < deadline:
        # worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        # source: h[i]==1, partner 0
        # target: h[j]==0, partner 1
        if k >= 0:
            rng_range = np.arange(n - k)
            src_mask = (h[rng_range] == 1) & (h[rng_range + k] == 0)
            tgt_mask = (h[rng_range] == 0) & (h[rng_range + k] == 1)
            src = rng_range[src_mask]
            tgt = rng_range[tgt_mask]
        else:
            d = -k
            rng_range = np.arange(d, n)
            src_mask = (h[rng_range] == 1) & (h[rng_range - d] == 0)
            tgt_mask = (h[rng_range] == 0) & (h[rng_range - d] == 1)
            src = rng_range[src_mask]
            tgt = rng_range[tgt_mask]

        if src.size and tgt.size:
            i = int(rng.choice(src))
            j = int(rng.choice(tgt))
        else:
            # random feasible swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))

        # try the swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)

        # accept if better, else maybe accept thermally
        accept = False
        if new_c < cur_c:
            accept = True
        elif rng.random() < np.exp((cur_c - new_c) / (temp + 1e-12)):
            accept = True

        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]

        temp = max(temp * decay, 5e-7)

    return best_h, best_c


def hill_climb_binary(h, rng, max_iters, deadline):
    """Simple deterministic hill‑climb (swap moves only)."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        cand = best_h.copy()
        cand[i], cand[j] = 0, 1
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Initialise size and (optional) parent
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_float = np.array(initial_h_values, dtype=float)
        n = init_float.size
    else:
        n = 360                     # default size when no parent is given
        init_float = None

    weight = n // 2                # exact number of ones (n is even in practice)

    # ----------------------------------------------------------------------
    # Build initial binary candidate (parent if available)
    # ----------------------------------------------------------------------
    if init_float is not None:
        h0 = binary_from_float(init_float, weight, rng).astype(int)
    else:
        idx = rng.choice(n, weight, replace=False)
        h0 = np.zeros(n, dtype=int)
        h0[idx] = 1

    best_h, best_c = greedy_improve(h0.copy(), rng,
                                    start + budget_s * 0.55)  # cheap phase

    # ----------------------------------------------------------------------
    # Random restarts with greedy/annealing improvement
    # ----------------------------------------------------------------------
    while time.time() < start + budget_s * 0.85:
        # generate a fresh random binary vector
        idx = rng.choice(n, weight, replace=False)
        h_start = np.zeros(n, dtype=int)
        h_start[idx] = 1
        h_ref, c_ref = greedy_improve(h_start, rng,
                                      start + budget_s * 0.85)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

        # early exit if we already beat the target
        if best_c <= 0.3808:
            break

    # ----------------------------------------------------------------------
    # Final deterministic hill‑climb to polish the best candidate
    # ----------------------------------------------------------------------
    deadline = start + budget_s * 0.98
    best_h, best_c = hill_climb_binary(best_h, rng, max_iters=5000, deadline=deadline)

    # ----------------------------------------------------------------------
    # Return exact feasible solution
    # ----------------------------------------------------------------------
    final_h = best_h.astype(float)
    final_c = compute_c5(final_h)
    return final_h.tolist(), float(final_c), int(n)
