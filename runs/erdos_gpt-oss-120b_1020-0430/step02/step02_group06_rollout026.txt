def compute_gradient(h):
    N = h.shape[0]
    corr = np.correlate(h, 1-h, mode='full')
    k_max = np.argmax(corr) - (N-1)  # shift range - (N-1) .. (N-1)
    # shift positive means aligning (1-h) shifted right; So k_max index in correlation array:
    shift = k_max
    # compute gradient vector:
    grad = np.zeros(N)
    # term (1 - h[i+shift]) for i indexes where i+shift in [0,N-1]
    # term h[i-shift] for i-shift in [0,N-1]
    # use vectorized:
    if shift >= 0:
        # i from 0..N-1-shift:
        idx = np.arange(N-shift)
        grad[idx] += (1.0 - h[idx + shift])
        # i from shift..N-1 have i-shift >=0:
        grad[shift:] -= h[idx]  # where idx = 0..N-shift-1
    else:
        s = -shift
        # shift negative => i+shift = i - s
        # valid for i >= s
        idx = np.arange(s, N)
        grad[idx] += (1.0 - h[idx - s])  # i+shift in domain
        # i - shift = i + s range? For i where i - s >=0 => i >= s
        # second term: h[i+ s]? Actually i - shift = i + s, need h[i - shift] but we compute -h[i - shift] if i - shift in domain
        # When shift negative, i - shift = i + s, i + s < N? That requires i < N - s
        idx2 = np.arange(N - s)
        grad[idx2] -= h[idx2 + s]
    return grad

def _project_to_feasible(v, target, tol=1e-9, max_iter=200):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # free indices where moving can correct diff
        if diff > 0:
            free = np.where(v > 0)[0]
        else:
            free = np.where(v < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    # final correction if still off:
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        # adjust smallest number of entries
        # find indices not at bound for addition/subtraction
        if residual > 0:
            # can increase entries where v < 1
            inc_idx = np.where(v < 1 - 1e-12)[0]
            # distribute resid equally
            if inc_idx.size > 0:
                inc = min(residual, (1 - v[inc_idx]).sum())
                # for simplicity add to first idx
                v[inc_idx[0]] += inc
        else:
            dec_idx = np.where(v > 1e-12)[0]
            if dec_idx.size > 0:
                dec = min(-residual, v[dec_idx].sum())
                v[dec_idx[0]] -= dec
    return np.clip(v, 0.0, 1.0)

def _interpolate_h(coarse_h, n_fine):
    Nc = coarse_h.size
    x_coarse = (np.arange(Nc) + 0.5) * (2.0 / Nc)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    fine_h = np.interp(x_fine, x_coarse, coarse_h)
    return fine_h

# ---------- feasibility projection ----------
def _project_to_feasible(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(v > 0)[0]
        else:
            free = np.where(v < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    # Tiny final correction
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            inc = np.where(v < 1)[0]
            if inc.size:
                add = min(residual, (1 - v[inc]).sum())
                v[inc[0]] += add
        else:
            dec = np.where(v > 0)[0]
            if dec.size:
                sub = min(-residual, v[dec].sum())
                v[dec[0]] -= sub
    return np.clip(v, 0.0, 1.0)


# ---------- C5 computation ----------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    rev = (1.0 - h)[::-1]
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def _max_corr_and_shift(h: np.ndarray):
    n = h.size
    rev = (1.0 - h)[::-1]
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)[: 2 * n - 1]
    idx = np.argmax(conv)
    max_corr = conv[idx]
    shift = idx - (n - 1)          # shift in [-n+1, n-1]
    return max_corr, shift


# ---------- gradient for the maximal shift ----------
def _gradient_max_corr(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.size
    g = np.zeros(n)
    if shift >= 0:
        m = n - shift
        i = np.arange(m)
        g[i] += 1.0 - h[i + shift]
        g[shift:] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += 1.0 - h[i - s]
        j = np.arange(n - s)
        g[j] -= h[j + s]
    return g


# ---------- simple stochastic refinement ----------
def _local_refine(h: np.ndarray, attempts: int = 30, delta: float = 0.02) -> np.ndarray:
    n = h.size
    target = n / 2.0
    best = h.copy()
    best_val = _c5_from_h(best)
    for _ in range(attempts):
        k = max(1, int(n * 0.03))
        idx = np.random.choice(n, k, replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(k)
        trial = _project_to_feasible(trial, target)
        val = _c5_from_h(trial)
        if val < best_val:
            best, best_val = trial, val
    return best


# ---------- interpolation ----------
def _interpolate_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse)


# ---------- core optimiser for a given grid ----------
def _optimize_h(start_h: np.ndarray, time_budget: float) -> tuple:
    n = start_h.size
    target = n / 2.0
    cur = _project_to_feasible(start_h, target)
    max_corr, shift = _max_corr_and_shift(cur)
    cur_val = max_corr * (2.0 / n)
    best, best_val = cur.copy(), cur_val

    start = time.time()
    step = 0.03
    step_decay = 0.9995
    accept_prob = 0.001
    perturb_prob = 0.005

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        grad = _gradient_max_corr(cur, shift)
        trial = cur - step * grad
        trial = _project_to_feasible(trial, target)

        max_corr_t, shift_t = _max_corr_and_shift(trial)
        trial_val = max_corr_t * (2.0 / n)

        if trial_val < cur_val or np.random.rand() < accept_prob:
            cur, cur_val, shift = trial, trial_val, shift_t
            if trial_val < best_val:
                best, best_val = trial.copy(), trial_val

        if np.random.rand() < perturb_prob:
            pert = cur + 0.05 * np.random.randn(n)
            pert = _project_to_feasible(pert, target)
            cur = pert
            max_corr, shift = _max_corr_and_shift(cur)
            cur_val = max_corr * (2.0 / n)

        step *= step_decay
        if step < 1e-6:
            break

    best = _local_refine(best, attempts=50, delta=0.008)
    best_val = _c5_from_h(best)
    return best, best_val
