    best_val = c5_of_h(h)
    cur_h = h.copy()
    cur_val = best_val
    n = len(h)
    T0 = best_val * 0.1 or maybe 0.01 * best_val?
    T = T0
    for iter in range(max_iter):
        # compute step size dynamic
        # Choose i, j randomly
        i, j = np.random.randint(n, size=2)
        if i==j:
            continue
        # compute max delta that keeps within [0,1]; delta can be positive or negative.
        # Suppose we want to transfer delta from i to j: delta can be negative (i gets, j loses). We sample delta magnitude up to limit L = min(cur_h[i], 1 - cur_h[j]).
        # We choose direction random sign.
        max_up = min(cur_h[i], 1 - cur_h[j])
        max_down = min(cur_h[j], 1 - cur_h[i])
        # random choose direction:
        if np.random.rand() < 0.5:
            # move from i to j
            delta = np.random.rand() * max_up
        else:
            # move from j to i (i gains)
            delta = -np.random.rand() * max_down
        # apply
        cur_h[i] -= delta
        cur_h[j] += delta
        # compute new value
        val = c5_of_h(cur_h)
        # acceptance
        if val < cur_val or np.random.rand() < np.exp((cur_val - val) / T):
            cur_val = val
            # if improvement over best, update best
            if val < best_val:
                best_val = val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # compute maximum allowed delta in positive direction (transfer from i to j)
    max_transfer = min(h[i], 1 - h[j], max_delta)
    # if max_transfer <= 0, try reverse swap?
    if max_transfer <= 0:

        max_transfer = min(h[j], 1 - h[i], max_delta)
        if max_transfer <= 0:

        delta = np.random.rand() * max_transfer

    delta = np.random.rand() * max_transfer

def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] -= delta

                h[i] += add
                residual -= add

                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break

# Core utilities

def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a discretised step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            movable = h > 0
        else:
            movable = h < 1
        if not np.any(movable):
            break
        delta = diff / np.count_nonzero(movable)
        h[movable] -= delta
        h = np.clip(h, 0.0, 1.0)
    # small residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse_h)


def _triangle_h(n: int) -> np.ndarray:
    """Symmetric triangular shape (zero at edges, height 1 at centre)."""
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = 1.0 - np.abs(xs - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return _project_to_feasible(h, n)


def _random_feasible(n: int) -> np.ndarray:
    """Uniform random vector projected onto the feasible set."""
    h = np.random.rand(n)
    return _project_to_feasible(h, n)


def _local_refine(h: np.ndarray, attempts: int = 15, delta: float = 0.02) -> np.ndarray:
    """Simple stochastic hill‑climbing around h."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        val = _c5_from_h(trial)
        if val < best_c5:
            best_h, best_c5 = trial, val
    return best_h

def _anneal_optimize(start_h: np.ndarray, time_allow: float, target_bound: float = None):
    """Annealing with pairwise mass transfers."""
    start = time.time()
    end = start + time_allow
    h = start_h.copy()
    n = h.size
    cur_val = _c5_from_h(h)
    best_h = h.copy()
    best_val = cur_val
    T0 = max(1e-4, cur_val * 0.2)
    max_delta = 0.4

    while time.time() < end:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)

        # try moving mass from i to j
        allowed = min(h[i], 1.0 - h[j], max_delta)
        if allowed <= 0:
            # try opposite direction
            allowed = min(h[j], 1.0 - h[i], max_delta)
            if allowed <= 0:
                continue
            delta = np.random.rand() * allowed
            new_h = h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        else:
            delta = np.random.rand() * allowed
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_val = _c5_from_h(new_h)
        d = new_val - cur_val
        elapsed = time.time() - start
        frac = elapsed / time_allow
        T = max(1e-9, T0 * (1.0 - frac))
        if d < 0 or np.random.rand() < np.exp(-d / T):
            h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                if target_bound is not None and best_val < target_bound:
                    break
        # slowly reduce move size
        max_delta = max(0.02, max_delta * (1.0 - 0.005 * frac))
    return best_h, best_val

    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h, t = x[:n], x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def obj(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opt = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(obj, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opt)
    h_opt = res.x[:n] if res.success else h0
    return _project_to_feasible(h_opt, n)


# -------------------------------------------------------------
# Main driver
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdos C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    target = 0.3808

    # parent construction, if supplied
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    # candidate grid sizes
    n_candidates = []
    if parent_h is not None:
        n_candidates.append(parent_h.size)
    # a set of even sizes up to 256 (larger sizes give more flexibility)
    for n in (96, 128, 160, 192, 224, 256):
        if n not in n_candidates:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_val = float("inf")
    best_n = None

    # iterate over grid sizes
    for n in n_candidates:
        # time guard
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5.0:
            break
        time_left = budget_s - elapsed

        # build a pool of initial vectors for this size
        inits = []
        if parent_h is not None and parent_h.size == n:
            inits.append(parent_h.copy())
        elif parent_h is not None:
            inits.append(_upscale_h(parent_h, n))
        inits.append(_triangle_h(n))
        # sin^2 start
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        sin2 = np.sin(np.pi * xs / 2) ** 2
        inits.append(_project_to_feasible(sin2, n))
        inits.append(_random_feasible(n))

        # allocate a slice of the remaining budget to this size
        slice_time = min(0.25 * time_left, 45.0)

        for init_h in inits:
            if time.time() - start_time > budget_s - 3.0:
                break
            # annealing
            anneal_time = min(slice_time / len(inits), 20.0)
            h_opt, val = _anneal_optimize(init_h, anneal_time,
                                          target_bound=target if best_val > target else None)
            # local stochastic refinement
            h_opt = _local_refine(h_opt, attempts=12, delta=0.015)
            val = _c5_from_h(h_opt)

            if val < best_val:
                best_h, best_val, best_n = h_opt.copy(), val, n
                if best_val < target:
                    break

            # occasional SLSQP polishing if we still have time
            remaining = budget_s - (time.time() - start_time)
            if remaining > 6.0 and best_val > target:
                h_ref = _solve_h_slsqp(h_opt, max_iter=300)
                h_ref = _local_refine(h_ref, attempts=8, delta=0.01)
                val_ref = _c5_from_h(h_ref)
                if val_ref < best_val:
                    best_h, best_val, best_n = h_ref.copy(), val_ref, n
                    if best_val < target:
                        break

        if best_val < target:
            break

    # fallback if nothing found
    if best_h is None:
        n = 128
        best_h = _project_to_feasible(np.full(n, 0.5), n)
        best_val = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_val), int(best_n)
