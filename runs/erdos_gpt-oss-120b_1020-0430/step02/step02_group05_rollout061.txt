def _build_h_from_a(a):
    m = a.shape[0]
    n = m * 2
    h = np.empty(n, dtype=np.float64)
    h[:m] = a
    h[m:] = 1 - a[::-1]
    return h

def _objective_and_grad_a(a, beta=12.0):
    h = _build_h_from_a(a)
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    diff = corr - maxc
    exp_term = np.exp(beta * diff)  # values <=1
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    n = h.size

    conv1 = np.convolve(w, h, mode='full')[n-1:2*n-1]

    conv2 = np.convolve(w_rev, h, mode='full')[n-1:2*n-1]
    h_mir = h[::-1]

import time
import numpy as np
import scipy.optimize as opt

def _build_h_from_a(a):
    m = a.shape[0]
    n = m * 2
    h = np.empty(n, dtype=np.float64)
    h[:m] = a
    h[m:] = 1.0 - a[::-1]
    return h

def _c5_from_h(h):
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

def _objective_and_grad_a(a, beta=12.0):
    h = _build_h_from_a(a)
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = corr.max()
    diff = corr - maxc                     # <=0
    exp_term = np.exp(beta * diff)         # safe, max 1
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                 # softmax weights, sum=1

    n = h.size
    conv1 = np.convolve(w, h, mode="full")[n - 1 : 2 * n - 1]
    w_rev = w[::-1]
    conv2 = np.convolve(w_rev, h, mode="full")[n - 1 : 2 * n - 1]

    h_mir = h[::-1]
    conv1_mir = np.convolve(w, h_mir, mode="full")[n - 1 : 2 * n - 1]
    conv2_mir = np.convolve(w_rev, h_mir, mode="full")[n - 1 : 2 * n - 1]

    grad = -(conv2 + conv1) + (conv2_mir + conv1_mir)
    m = a.shape[0]
    grad_a = grad[:m]
    f = maxc + np.log(sum_exp) / beta
    return f, grad_a

def _obj(a):
    f, _ = _objective_and_grad_a(a)
    return f

def _grad(a):
    _, g = _objective_and_grad_a(a)
    return g

def _symmetrize_a_from_h(h):
    n = h.shape[0]
    if n % 2:
        n -= 1
        h = h[:n]
    m = n // 2
    a = (h[:m] + (1.0 - h[m:][::-1])) * 0.5
    return np.clip(a, 0.0, 1.0)

def _upscale_a(a_coarse, m_fine):
    m_coarse = a_coarse.shape[0]
    if m_coarse == m_fine:
        return a_coarse.copy()
    x_coarse = (np.arange(m_coarse) + 0.5) / m_coarse
    x_fine = (np.arange(m_fine) + 0.5) / m_fine
    return np.interp(x_fine, x_coarse, a_coarse)
