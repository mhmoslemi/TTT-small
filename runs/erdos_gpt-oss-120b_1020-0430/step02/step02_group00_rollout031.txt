import time
import numpy as np

# --------------------------------------------------------------
# Feasibility projection (water‑filling)
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# --------------------------------------------------------------
# Fast C5 evaluation via FFT (linear correlation)
# --------------------------------------------------------------
def _c5_fast(h: np.ndarray) -> float:
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    fa = np.fft.rfft(h, n=L)
    fb = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fa * np.conj(fb), n=L)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


# --------------------------------------------------------------
# Random feasible constructions
# --------------------------------------------------------------
def _random_binary_h(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def _random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    return _project_to_feasible(h, n)

            # pick a low‑product partner
            for _ in range(5):
                j = rng.integers(n - shift_abs)
                if j != i_max:
                    break
            else:
                j = (i_max + 1) % (n - shift_abs)

            i1, i2 = i_max, i_max + shift_abs
            j1, j2 = j, j + shift_abs
            max_delta = min(cur_h[i1], cur_h[i2], 1.0 - cur_h[j1], 1.0 - cur_h[j2])
            if max_delta > 0:
                delta = rng.random() * max_delta * 0.5
                new_h = cur_h.copy()
                new_h[i1] -= delta
                new_h[i2] -= delta
                new_h[j1] += delta
                new_h[j2] += delta

        # --- random mass transfer (preserves sum) ---
        if new_h is None:
            i = rng.integers(n)
            if rng.random() < 0.5:
                # i gains, donor j loses
                max_up = 1.0 - cur_h[i]
                j = None
                for _ in range(5):
                    jj = rng.integers(n)
                    if jj != i and cur_h[jj] > 1e-12:
                        j = jj
                        break
                if j is None:
                    continue
                max_delta = min(max_up, cur_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta * 0.5
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                # i loses, receiver j gains
                max_down = cur_h[i]
                j = None
                for _ in range(5):
                    jj = rng.integers(n)
                    if jj != i and cur_h[jj] < 1 - 1e-12:
                        j = jj
                        break
                if j is None:
                    continue
                max_delta = min(max_down, 1.0 - cur_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta * 0.5
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta

        if new_h is None:
            continue

        new_val = _c5_fast(new_h)
        d = new_val - cur_val
        # simple annealing schedule
        T = max(1e-6, 0.01 * (0.995 ** it))
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
                if best_val < target:
                    break

    return best_h, best_val


# --------------------------------------------------------------
# Main driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------- initial construction -------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        init_h = _project_to_feasible(init_h, init_h.size)
    except NameError:
        init_h = _random_feasible_h(256, rng)

    best_h = init_h.copy()
    best_val = _c5_fast(best_h)
    best_n = best_h.size

    # ------------------- resolution schedule -------------------
    resolutions = [256, 512, 1024, 2048]
    target = 0.3808

    for idx, n in enumerate(resolutions):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break

        # upscale current best to required size
        if best_h.size != n:
            factor = n // best_h.size
            if factor * best_h.size == n:
                cur_h = np.repeat(best_h, factor)
            else:
                # fallback interpolation
                x_src = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
                x_tgt = (np.arange(n) + 0.5) * (2.0 / n)
                cur_h = np.interp(x_tgt, x_src, best_h)
            cur_h = _project_to_feasible(cur_h, n)
        else:
            cur_h = best_h.copy()

        # ----- random binary sampling (quick) -----
        if remaining > 2.0:
            # allocate a tiny slice for sampling
            sample_time = min(2.0, remaining * 0.03)
            sample_start = time.time()
            # rough estimate of number of samples we can afford
            est_per = 0.0003   # seconds per evaluation (conservative)
            max_samp = int(sample_time / est_per)
            for _ in range(min(max_samp, 5000)):
                h_bin = _random_binary_h(n, rng)
                val = _c5_fast(h_bin)
                if val < best_val:
                    best_val, best_h, best_n = val, h_bin.copy(), n
                    if best_val < target:
                        break
                if time.time() - sample_start > sample_time:
                    break

        # ----- local optimisation (hill climbing) -----
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        # allocate roughly half of remaining time to this resolution
        rem_sizes = len(resolutions) - idx - 1
        alloc = max(5.0, remaining / (rem_sizes + 2))
        alloc = min(alloc, remaining - 1.0)
        h_opt, val_opt = _hill_climb(cur_h, alloc, rng)
        if val_opt < best_val:
            best_val, best_h, best_n = val_opt, h_opt.copy(), n
        if best_val < target:
            break

    # ----- final polishing with remaining budget -----
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 3.0:
        h_opt, val_opt = _hill_climb(best_h, remaining - 1.0, rng)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

    # final feasibility enforcement and exact C5 computation
    best_h = _project_to_feasible(best_h, best_h.size)
    best_val = _c5_fast(best_h)

    return best_h.tolist(), float(best_val), int(best_h.size)
