import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c_max(h):
    """Maximum raw overlap count (without the 2/n factor)."""
    return np.max(np.correlate(h, 1 - h, mode="full"))


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def greedy_binary_improve(h, rng, max_iter=3000):
    """Targeted swap improvement for the worst shift."""
    n = h.size
    best_c = compute_c5(h)
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        k_abs = int(np.argmax(corr) - (n - 1))
        k_abs = abs(k_abs)
        if k_abs == 0:
            break
        overlap = n - k_abs
        hs = h[:overlap]
        ht = h[k_abs:k_abs + overlap]

        # (1,0) mismatches
        mism = np.flatnonzero((hs == 1) & (ht == 0))
        if mism.size == 0:
            break
        i = mism[0]

        # (0,1) reverse mismatches
        rev = np.flatnonzero((hs == 0) & (ht == 1))
        if rev.size > 0:
            j = rev[0]
        else:
            zero_idxs = np.flatnonzero(h == 0)
            if zero_idxs.size == 0:
                break
            j = rng.choice(zero_idxs)

        if i == j:
            # ensure distinct indices
            zero_idxs = zero_idxs[zero_idxs != i]
            if zero_idxs.size == 0:
                break
            j = rng.choice(zero_idxs)

        # swap
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)
        if new_c < best_c - 1e-12:
            best_c = new_c
            continue
        # revert if no improvement
        h[i], h[j] = 1.0, 0.0
        break
    return h, best_c


def max_corr_constraints(x, n):
    """Inequality constraints: M >= sum_i h_i - sum_i h_i h_{i+k} for all k."""
    h = x[:n]
    M = x[-1]
    # prefix sums for sum of h over the overlapping region
    pref = np.concatenate([[0.0], np.cumsum(h)])
    # autocorrelation h with h
    corrhh = np.correlate(h, h, mode="full")
    sum_hh = corrhh[n - 1 : n - 1 + n]          # length n, k=0..n-1
    sum_h = pref[n - np.arange(n)]               # sum of first n-k entries
    # constraint must be >= 0 : M - (sum_h - sum_hh) >= 0
    return M - (sum_h - sum_hh)


def slsqp_minmax_optimize(h0, rng, max_iter=300):
    """
    Direct min–max optimisation:
      minimise M such that for every shift k,
      sum_i h_i (1 - h_{i+k}) <= M,
    with sum(h)=n/2 and 0<=h<=1.
    Returns refined h and the final M.
    """
    n = h0.size
    M0 = compute_c_max(h0)
    x0 = np.concatenate([h0, [M0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, n)]

    cons = [
        {"type": "eq", "fun": lambda x, n=n: np.sum(x[:n]) - n / 2.0},
        {"type": "ineq", "fun": lambda x, n=n: max_corr_constraints(x, n)},
    ]

    res = minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )

    x = np.clip(res.x, 0.0, 1.0)
    h = x[:n]
    M = x[-1]

    # enforce the sum exactly (tiny adjustments if needed)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h, M


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb (accept only strict improvements)."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            h = nh
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return h, best_c

                    best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # Optional up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.985:
        # double the resolution by simple repetition
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # small hill‑climb on the up‑sampled version
        h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.3)
        # final min‑max refinement
        try:
            h_up, _ = slsqp_minmax_optimize(h_up, rng, max_iter=300)
        except Exception:
            pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final feasibility enforcement
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
