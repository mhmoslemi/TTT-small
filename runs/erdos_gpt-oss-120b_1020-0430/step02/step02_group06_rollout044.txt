import time, numpy as np
from scipy import optimize

# ---------- feasibility utilities ----------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 300) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap for discretised h (scaled to real interval)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- deterministic refinement (SLSQP) ----------

def _ineq_constraint(x):
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    """sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """SLSQP refinement."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": _ineq_constraint},
            {"type": "eq", "fun": _eq_constraint}]
    opt = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opt)
    h_opt = _project_to_feasible(res.x[:n] if res.success else h0, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- stochastic local search ----------

def _local_refine(h: np.ndarray, time_budget: float, target: float) -> tuple:
    """Hill‑climbing with small random tweaks."""
    n = h.size
    start = time.time()
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    while time.time() - start < time_budget:
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += 0.02 * np.random.randn(len(idx))
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
            if best_c5 <= target:
                break
    return best_h, best_c5


# ---------- genetic algorithm core ----------

def _tournament(pop, scores, k=3):
    """Return a parent via tournament selection."""
    ids = np.random.choice(len(pop), size=k, replace=False)
    best = ids[np.argmin([scores[i] for i in ids])]
    return pop[best]


def _ga_optimize(n: int, init_h, time_limit: float, target: float):
    """GA for a fixed grid size."""
    pop_size, elite_sz = 30, 2
    mut_prob, cross_prob = 0.2, 0.7
    sigma = 0.15
    # initial population
    pop = []
    for i in range(pop_size):
        if i == 0 and init_h is not None:
            h0 = init_h.copy()
            if h0.size != n:
                h0 = _upscale_h(init_h, n)
        else:
            h0 = np.random.rand(n)
        pop.append(_project_to_feasible(h0, n))
    # evaluate
    scores = [_c5_from_h(ind) for ind in pop]
    best_idx = int(np.argmin(scores))
    best_h, best_c5 = pop[best_idx].copy(), scores[best_idx]

    gen = 0
    start = time.time()
    while time.time() - start < time_limit:
        gen += 1
        # elite preservation
        elite_idxs = np.argsort(scores)[:elite_sz]
        new_pop = [pop[i].copy() for i in elite_idxs]

        # offspring
        while len(new_pop) < pop_size:
            p1 = _tournament(pop, scores)
            p2 = _tournament(pop, scores)
            # crossover
            if np.random.rand() < cross_prob:
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, p1, p2)
            else:
                child = p1.copy()
            # mutation
            if np.random.rand() < mut_prob:
                mask = np.random.rand(n) < 0.3
                child[mask] += sigma * np.random.randn(mask.sum())
            child = np.clip(child, 0.0, 1.0)
            child = _project_to_feasible(child, n)
            new_pop.append(child)

        pop = new_pop
        scores = [_c5_from_h(ind) for ind in pop]
        cur_best = int(np.argmin(scores))
        if scores[cur_best] < best_c5:
            best_h, best_c5 = pop[cur_best].copy(), scores[cur_best]
            if best_c5 <= target:
                break
        # adapt sigma
        sigma *= 0.995
        # early stop if time low
        if time.time() - start > time_limit - 0.5:
            break
    return best_h, best_c5


# ---------- main driver ----------
