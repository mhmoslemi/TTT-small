def _smooth_max_gradient(h, beta):
    n = len(h)
    L = 2*n-1
    # compute f
    corr_h_one = np.convolve(h, np.ones(n), mode='full')   # size L
    corr_h_h = np.convolve(h, h, mode='full')
    f = corr_h_one - corr_h_h

    # softmax
    max_f = np.max(beta * f)
    exp_beta_f = np.exp(beta * f - max_f)
    w = exp_beta_f / exp_beta_f.sum()

    # compute A via prefix sums
    ps = np.concatenate([[0.0], np.cumsum(w)])  # length L+1
    idx = np.arange(n)
    start = (n-1) - idx
    end = (2*n-1) - idx
    A = ps[end] - ps[start]   # shape n

    # compute B
    B = np.zeros(n)
    # iterate k from -(n-1) to (n-1)
    for shift in range(-(n-1), n):
        wk = w[shift + (n-1)]
        if wk == 0.0:
            continue
        if shift >= 0:
            B[:n-shift] += wk * h[shift:]  # h[i+shift]
            B[shift:] += wk * h[:n-shift]  # h[i-shift] (i-shift = i - shift)
        else:
            # shift negative
            k = -shift
            B[k:] += wk * h[:n-k]          # h[i+(-k)] = h[i-k]
            B[:n-k] += wk * h[k:]          # h[i-(-k)] = h[i+k]
    # gradient of phi is A - B
    return A - B, f

def _smooth_max_and_grad(h, beta):
    n = h.size
    # compute correlations via np.convolve
    # ones array
    ones = np.ones(n, dtype=h.dtype)
    corr_one = np.convolve(h, ones, mode='full')  # length L = 2n-1
    corr_hh = np.convolve(h, h, mode='full')
    f = corr_one - corr_hh
    # Softmax weights
    # We want weights for each shift.
    # Use stable exponent
    scaled = beta * f
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    # Compute A via prefix sum of w
    ps = np.concatenate([[0.0], np.cumsum(w)])
    idx = np.arange(n)
    start = (n-1) - idx
    end = (2*n-1) - idx
    A = ps[end] - ps[start]   # shape (n,)
    # Compute B: sum_k w_k * (h_{i+k} + h_{i-k})
    B = np.zeros(n, dtype=h.dtype)
    # For each shift k, get wk
    for shift in range(-(n-1), n):
        wk = w[shift + (n-1)]
        if wk == 0.0:
            continue
        if shift >= 0:
            # positive shift
            if n - shift > 0:
                B[:n-shift] += wk * h[shift:]          # h_{i+shift}
                B[shift:] += wk * h[:n-shift]          # h_{i-shift}
        else:
            k = -shift
            if n - k > 0:
                B[k:] += wk * h[:n-k]                  # h_{i+shift} where shift negative
                B[:n-k] += wk * h[k:]                  # h_{i-shift}
    grad = A - B
    # Return grad, max_f (could be used)
    return grad, f

def _gradient_descent(h0, betas, steps_per_beta, lr=0.1):
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps=1e-8
    t = 0
    best_h = h.copy()
    # compute initial C5
    best_c5 = _c5_from_h(h)
    t0 = time.time()
    for beta in betas:
        for _ in range(steps_per_beta):
            t += 1
            grad, f = _smooth_max_and_grad(h, beta)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            # project onto box and simplex
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_simplex_box(h, n/2.0)
            # Periodically evaluate C5
            if t % 20 == 0:
                c5 = _c5_from_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            # optionally break if time left low
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Projection onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# ------------------------------------------------------------
# C5 evaluation for a discrete step function
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ------------------------------------------------------------
# Upscale a coarse step function by linear interpolation
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ------------------------------------------------------------
# Simple symmetric tent shape, projected to feasibility
def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


# ------------------------------------------------------------
# Smooth‑max gradient (softmax approximation of the max over shifts)
def _smooth_max_and_grad(h: np.ndarray, beta: float):
    n = h.shape[0]
    # full correlations: h * 1  and h * h
    ones = np.ones(n, dtype=h.dtype)
    corr_one = np.convolve(h, ones, mode="full")          # length 2n‑1
    corr_hh = np.convolve(h, h, mode="full")
    f = corr_one - corr_hh                               # true overlaps

    # softmax weights
    scaled = beta * f
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()                    # length L = 2n‑1

    # ----- part A : sum of weights whose mask covers each index -----
    ps = np.concatenate([[0.0], np.cumsum(w)])           # prefix sums, length L+1
    idx = np.arange(n)
    start = (n - 1) - idx
    end = (2 * n - 1) - idx
    A = ps[end] - ps[start]                               # shape (n,)

    # ----- part B : weighted shifted sum of h (both directions) -----
    conv1 = np.convolve(w, h, mode="full")               # length L+n‑1
    conv2 = np.convolve(w[::-1], h, mode="full")
    off = n - 1
    B = conv1[off:off + n] + conv2[off:off + n]           # shape (n,)

    grad = A - B
    return grad, f


# ------------------------------------------------------------
# Adam‑style gradient descent on the smooth‑max surrogate
def _gradient_descent(start_h, betas, steps_per_beta, lr, time_left):
    h = _project_to_simplex_box(start_h, start_h.size / 2.0)
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    for beta in betas:
        for _ in range(steps_per_beta):
            if time_left() < 5.0:
                return best_h, best_c5
            t += 1
            grad, _ = _smooth_max_and_grad(h, beta)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_simplex_box(h, n / 2.0)

            if t % 20 == 0:
                cur_c5 = _c5_from_h(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()

    return best_h, best_c5

# Linear‑programming refinement (majorisation–minimisation)
def _lp_refine(h_start, max_iter, time_left):
    n = h_start.size
    h_curr = _project_to_simplex_box(h_start, n / 2.0)
    best_h = h_curr.copy()
    best_c5 = _c5_from_h(h_curr)

    L = 2 * n - 1
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    for it in range(max_iter):
        if time_left() < 5.0:
            break

        # correlation of current h with itself
        corr_hh = np.convolve(h_curr, h_curr, mode="full")

        A_ub = np.empty((L, n + 1), dtype=np.float64)
        b_ub = np.empty(L, dtype=np.float64)

        for s in range(L):
            k = s - (n - 1)
            coeff = np.ones(n, dtype=np.float64)
            if k >= 0:
                coeff[k:] -= h_curr[: n - k]
                coeff[: n - k] -= h_curr[k:]
            else:
                kk = -k
                coeff[: n - kk] -= h_curr[kk:]
                coeff[kk:] -= h_curr[: n - kk]
            A_ub[s, :n] = coeff
            A_ub[s, n] = -1.0
            b_ub[s] = -corr_hh[s]

        res = optimize.linprog(c_vec, A_ub=A_ub, b_ub=b_ub,
                               A_eq=A_eq, b_eq=b_eq,
                               bounds=bounds, method='highs')
        if not res.success:
            break

        new_h = res.x[:n]
        new_c5 = _c5_from_h(new_h)
        if new_c5 < best_c5 - 1e-7:
            best_c5 = new_c5
            best_h = new_h
            h_curr = new_h
        else:
            break

    return best_h, best_c5
