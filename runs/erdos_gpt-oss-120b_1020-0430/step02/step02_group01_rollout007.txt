Implementation steps:

def project_box_sum(v, target, tol=1e-9, max_iter=100):
    lo = -np.max(v)  # such that v+lo <=0 -> all clipped to 0 sum=0
    hi = 1 - np.min(v)  # such that v+hi >=1 -> all clipped to 1 sum=n
    # We want sum = target
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_sum_box(v, target, tol=1e-12, max_iter=60):
    """Project v onto {0<=x<=1, sum(x)=target} by shifting and clipping."""
    lo = -np.max(v)               # makes all entries <=0 → sum 0
    hi = 1.0 - np.min(v)          # makes all entries >=1 → sum = n
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    h = project_onto_sum_box(h, target)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def hill_climb(h_start, rng, max_iters=2000, step=0.5):
    """Simple hill‑climb that accepts only improving moves."""
    h = h_start.copy()
    best = h.copy()
    bestc = compute_c5(best)
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < bestc:
            bestc, best, h = c, cand.copy(), cand
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best, bestc


def softmax_weights(q, beta):
    """Stable softmax over -beta*q."""
    a = -beta * q
    m = np.max(a)
    e = np.exp(a - m)
    return e / e.sum()


def gradient_step(x, beta, step, target):
    """One projected gradient step on the soft‑max surrogate."""
    n = x.size
    q = np.convolve(x, x[::-1], mode="full")
    w = softmax_weights(q, beta)
    # D = convolution of w with x, centred
    d = np.convolve(w, x, mode="full")[n - 1 : n - 1 + n]
    y = x + step * d
    return project_onto_sum_box(y, target)


def gradient_descent(x0, beta_sched, step0, iters_per, rng):
    """Run a multi‑beta projected gradient descent."""
    x = x0.copy()
    target = x.size * 0.5
    for beta, its in zip(beta_sched, iters_per):
        for i in range(its):
            step = step0 * (0.95 ** i)
            x = gradient_step(x, beta, step, target)
        # small random shake to escape flat regions
        x = perturb_h(x, rng, scale=0.05)
    return x


def upsample_vector(v, new_n):
    """Linear interpolation up‑sampling to new_n points."""
    old_n = v.size
    if old_n == new_n:
        return v.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, v)
