def perturb_h(h, rng, scale=0.2):
    # pick i where h[i]>0 and j where h[j]<1
    ones = np.where(h > 1e-12)[0]
    zeros = np.where(h < 1 - 1e-12)[0]
    if len(ones)==0 or len(zeros)==0:
        return h
    i = rng.choice(ones)
    j = rng.choice(zeros)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time, numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h[:] = 0.0
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount from a non‑zero entry to a non‑full entry."""
    ones = np.where(h > 1e-12)[0]
    zeros = np.where(h < 1 - 1e-12)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i = rng.choice(ones)
    j = rng.choice(zeros)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.4) -> tuple:
    """Deterministic hill‑climb."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing(h_start: np.ndarray, rng: np.random.Generator,
                        max_iters: int = 2000, init_temp: float = 0.02,
                        decay: float = 0.999, scale: float = 0.3) -> tuple:
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12: break
            i = np.random.default_rng().integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def quick_refine(h: np.ndarray, rng: np.random.Generator) -> tuple:
    """Fast refinement: limited SLSQP (for modest n) + hill‑climb."""
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    if n <= 200:
        for beta in (5.0, 20.0, 80.0):
            try:
                trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
            except Exception:
                continue
            c = compute_c5(trial)
            if c < best_c:
                best_h, best_c = trial, c
    trial_h, c_h = hill_climb(best_h, rng,
                              max_iters=800 if n <= 200 else 1500,
                              start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c

def deep_refine(h: np.ndarray, rng: np.random.Generator) -> tuple:
    """Intensive refinement for larger n."""
    n = h.size
    max_hill = 1500 if n <= 200 else 3000
    best_h, best_c = h.copy(), compute_c5(h)
    for _ in range(3):
        trial_h, c_h = hill_climb(best_h, rng, max_iters=max_hill, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    sa_iters = 2000 if n <= 200 else 4000
    trial_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iters=sa_iters,
                                         init_temp=0.03, decay=0.998, scale=0.2)
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c

def upsample_h(h: np.ndarray, factor: int, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target, rng)
    return h_up

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # initial vector (use provided parent if available)
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 180
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = quick_refine(h0, rng)

    # --------------------------------------------------------------
    # Phase 1 – many random restarts at base resolution
    # --------------------------------------------------------------
    while time.time() - start < budget_s * 0.55:
        h_start = random_feasible_vector(n0, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # --------------------------------------------------------------
    # Phase 2 – up‑sample and deep refinement
    # --------------------------------------------------------------
    if time.time() - start < budget_s * 0.85:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        h_up_ref, c_up_ref = deep_refine(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n0 = n_up  # keep track of current size

        # optional second up‑sample if time permits and size stays reasonable
        if time.time() - start < budget_s * 0.95 and n0 * 2 <= 1600:
            h_up2 = upsample_h(best_h, factor=2, rng=rng)
            n_up2 = h_up2.size
            h_up2_ref, c_up2_ref = deep_refine(h_up2, rng)
            if c_up2_ref < best_c:
                best_h, best_c = h_up2_ref, c_up2_ref
                n0 = n_up2

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n0 / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n0)
