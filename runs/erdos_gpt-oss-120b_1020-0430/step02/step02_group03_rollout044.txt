def project_onto_feasible(z, target, max_iter=100, tol=1e-8):
    # ensures 0 <= x_i <= 1, sum x_i = target.
    lo = np.min(z) - 1  # lower bound for lambda
    hi = np.max(z) + 1  # upper bound for lambda
    # binary search for lambda such that sum(clip(z - lambda,0,1)) = target
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        x = np.clip(z - lam, 0, 1)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    # final
    x = np.clip(z - (lo+hi)/2, 0, 1)
    # Due to rounding, adjust tiny diff by adding to a few components not at bound
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # Simple adjustment
        # Distribute diff to entries not at bound
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            # distribute equally
            inc = diff / free.size
            x[free] += inc
            # clip again
            x = np.clip(x, 0, 1)
    return x

def project_onto_feasible(z, target, tol=1e-9, max_iter=100):
    lo = -1e6  # Very low
    hi = 1e6   # Very high
    # Or compute lo/hi from min(z)-1, max(z)+1 for tighter.
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(z - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    # After loop, return x with lam = (lo+hi)/2
    x = np.clip(z - (lo+hi)/2.0, 0.0, 1.0)
    # Might still be off; adjust diff with simple method:
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non-bound variables
        free = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if free.size > 0:
            inc = diff / free.size
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: add to any that are not at 1 (if diff>0) and not at 0 if diff<0
            if diff > 0:
                idx = np.where(x < 1)[0]
                if idx.size > 0:
                    inc = diff / idx.size
                    x[idx] = np.clip(x[idx] + inc, 0.0, 1.0)
            else:
                idx = np.where(x > 0)[0]
                if idx.size > 0:
                    inc = diff / idx.size
                    x[idx] = np.clip(x[idx] + inc, 0.0, 1.0)
    return x

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum if needed
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # project onto feasible using project_onto_feasible
        x = project_onto_feasible(x, target)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----- helpers -----------------------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_feasible(z, target, tol=1e-9, max_iter=80):
    """Projection of z onto {0≤x≤1, Σx=target} by water‑filling."""
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(z - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            inc = diff / free.size
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n, rng):
    """Uniform random point in the feasible polytope."""
    return project_onto_feasible(rng.random(n), n / 2.0)


def hill_climb_gauss(h_start, rng, time_limit):
    """Gaussian perturbation hill‑climb with projection."""
    start = time.time()
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5(h)
    sigma = 0.30
    it = 0
    while time.time() - start < time_limit:
        it += 1
        if it % 500 == 0:
            sigma = max(sigma * 0.9, 0.02)
        z = h + sigma * rng.standard_normal(h.size)
        h_new = project_onto_feasible(z, h.size / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best = h_new.copy()
            h = h_new
    return best, best_c


def slsqp_optimize(h0, beta, max_iter=800):
    """Refine using SLSQP on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final feasible projection (tiny drift may remain)
    x = project_onto_feasible(x, CURRENT_TARGET)
    return x


# ----- main -------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # ---- initialise ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_feasible(h0, n / 2.0)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    # ---- build a diverse pool of seeds ---------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))
    for f in (1, 2, 3, 4):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(project_onto_feasible(h, n / 2.0))
    # linear ramps
    candidates.append(project_onto_feasible(xs / 2.0, n / 2.0))
    candidates.append(project_onto_feasible(1.0 - xs / 2.0, n / 2.0))

    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- main optimisation loop ------------------------------------
    beta_list = [10.0, 50.0, 200.0, 800.0, 3000.0]
    while elapsed() < budget_s * 0.85:
        if rng.random() < 0.7:
            start = best_h.copy()
        else:
            start = random_feasible_vector(n, rng)

        # hill‑climb (few seconds)
        hc_time = min(4.0, (budget_s - elapsed()) * 0.3)
        h_opt, c_opt = hill_climb_gauss(start, rng, hc_time)

        # SLSQP ladder
        for beta in beta_list:
            if elapsed() >= budget_s * 0.85:
                break
            h_opt = slsqp_optimize(h_opt, beta, max_iter=800)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()

        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ---- optional up‑sampling --------------------------------------
    if elapsed() < budget_s * 0.95 and n * 2 <= 400:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up = project_onto_feasible(h_up, n_up / 2.0)
        # coarse hill‑climb
        hc_time_up = min((budget_s - elapsed()) * 0.5, 30.0)
        h_up_opt, c_up_opt = hill_climb_gauss(h_up, rng, hc_time_up)
        # refinement ladder
        for beta in beta_list:
            if elapsed() >= budget_s:
                break
            h_up_opt = slsqp_optimize(h_up_opt, beta, max_iter=800)
            c_up_opt = compute_c5(h_up_opt)
            if c_up_opt < best_c:
                best_c, best_h, n = c_up_opt, h_up_opt.copy(), n_up
        # final fine hill‑climb (very small steps)
        if elapsed() < budget_s:
            h_up_opt, c_up_opt = hill_climb_gauss(best_h, rng, min(2.0, budget_s - elapsed()))
            if c_up_opt < best_c:
                best_c, best_h = c_up_opt, h_up_opt.copy()

    # ---- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_feasible(best_h, len(best_h) / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(len(best_h))
