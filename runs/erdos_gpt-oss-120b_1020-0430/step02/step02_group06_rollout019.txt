import time
import numpy as np

_TARGET_BOUND = 0.3808  # stop early if we beat this bound


def _max_corr(h: np.ndarray) -> int:
    """Maximum linear correlation of binary h with its complement (1‑h)."""
    # np.correlate works with integer arrays and returns an array of ints
    return int(np.max(np.correlate(h, 1 - h, mode="full")))


def _binary_from_parent(arr: np.ndarray) -> np.ndarray:
    """
    Convert a (possibly fractional) parent array to a binary vector
    with exactly n//2 ones, preserving values near 0.5 when possible.
    """
    n = arr.size
    # initial threshold
    bin_arr = (arr > 0.5).astype(np.int8)
    target = n // 2
    diff = int(bin_arr.sum() - target)
    if diff > 0:
        # too many ones – turn the ones with smallest original value to zero
        ones_idx = np.flatnonzero(bin_arr)
        vals = arr[ones_idx]
        order = np.argsort(vals)  # smallest first
        bin_arr[ones_idx[order[:diff]]] = 0
    elif diff < 0:
        # too few ones – turn the zeros with largest original value to one
        zeros_idx = np.flatnonzero(bin_arr == 0)
        vals = arr[zeros_idx]
        order = np.argsort(-vals)  # largest first
        bin_arr[zeros_idx[order[: -diff]]] = 1
    return bin_arr


def _solve_n_with_time(
    n: int,
    time_limit: float,
    parent_arr: np.ndarray | None,
    target_bound: float,
) -> tuple[np.ndarray | None, int]:
    """
    Search for a binary vector of length n with sum n//2 that minimises
    max correlation.  Returns (best_h, best_max_corr) found before time_limit.
    If a solution reaches the integer bound implied by target_bound, it is
    returned immediately.
    """
    start = time.time()
    best_h = None
    best_max = np.iinfo(np.int64).max
    max_allowed = int(np.floor(target_bound * n / 2.0))

    # try parent if it matches the size
    if parent_arr is not None and parent_arr.size == n:
        h0 = _binary_from_parent(parent_arr)
        cur_max = _max_corr(h0)
        if cur_max < best_max:
            best_h, best_max = h0.copy(), cur_max
        if cur_max <= max_allowed:
            return best_h, best_max

    # main stochastic search
    while time.time() - start < time_limit:
        # random balanced start
        h = np.zeros(n, dtype=np.int8)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1
        cur_max = _max_corr(h)
        if cur_max < best_max:
            best_h, best_max = h.copy(), cur_max
        if cur_max <= max_allowed:
            return best_h, best_max

        # annealing parameters
        max_iter = 250_000
        for it in range(max_iter):
            if time.time() - start >= time_limit:
                break
            # linear temperature schedule
            t_frac = it / max_iter
            T = 2.0 * (1.0 - t_frac) + 0.001 * t_frac

            # propose a swap of a 1 with a 0
            i = np.random.randint(n)
            while h[i] == 0:
                i = np.random.randint(n)
            j = np.random.randint(n)
            while h[j] == 1:
                j = np.random.randint(n)

            h[i], h[j] = 0, 1
            new_max = _max_corr(h)
            delta = new_max - cur_max
            if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                    if best_max <= max_allowed:
                        return best_h, best_max
                # keep swap

                h[i], h[j] = 1, 0

        # (only if time permits)
        if time.time() - start < time_limit - 0.5:
            for _ in range(2_000):

                    i = np.random.randint(n)

                    j = np.random.randint(n)

                    if cur_max < best_max:

                        best_h = h.copy()
                        if best_max <= max_allowed:
                            return best_h, best_max
                else:
                    h[i], h[j] = 1, 0

    return best_h, best_max
