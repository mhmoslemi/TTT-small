import time
import numpy as np
from scipy import optimize as _opt

# -------------------- Feasibility utilities -------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip values to [0,1] and adjust to satisfy sum(h)=n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap for a discretised step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# -------------------- SLSQP optimiser -------------------- #

def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h_slqp(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Run SLSQP from `initial_h`. Returns (h_opt, C5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = _opt.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                        constraints=constraints, options=opts)
    if res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# -------------------- Upscaling -------------------- #

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# -------------------- Simulated annealing (continuous) -------------------- #

def _sa_optimize(start_h: np.ndarray, time_left_func, max_iters: int = 200_000,
                 temp_start: float = 0.02, temp_end: float = 1e-6) -> tuple:
    """Mass‑transfer SA that preserves the sum exactly."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5
    eps = 1e-12

    for it in range(max_iters):
        if (it & 1023) == 0 and time_left_func() < 0.5:
            break
        # exponential cooling
        T = temp_start * ((temp_end / temp_start) ** (it / max_iters))

        # pick donor (h>0) and receiver (h<1)
        i = np.random.randint(n)
        while cur_h[i] <= eps:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while cur_h[j] >= 1.0 - eps:
            j = np.random.randint(n)

        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        if delta < eps:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = _c5_from_h(new_h)

        diff = new_c5 - cur_c5
        if diff < 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
    return best_h, best_c5


# -------------------- Simulated annealing (binary) -------------------- #

def _binary_sa_optimize(start_h: np.ndarray, time_left_func,
                        max_iters: int = 200_000,
                        temp_start: float = 0.02, temp_end: float = 1e-6) -> tuple:
    """Swap‑only SA for a 0/1 vector with exactly n/2 ones."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5

    for it in range(max_iters):
        if (it & 1023) == 0 and time_left_func() < 0.5:
            break
        T = temp_start * ((temp_end / temp_start) ** (it / max_iters))

        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = _c5_from_h(new_h)

        diff = new_c5 - cur_c5
        if diff < 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
    return best_h, best_c5
