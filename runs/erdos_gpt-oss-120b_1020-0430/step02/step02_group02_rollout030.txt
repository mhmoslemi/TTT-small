    # Use binary search.
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo+hi)/2
        proj = np.clip(h - lam, 0.0, 1.0)
        sproj = proj.sum()
        if sproj > target_sum:
            lo = lam
        else:
            hi = lam
    h = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Clamp to [0,1] and adjust sum to target via water‑filling (binary search)."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        proj = np.clip(h - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # final tiny fix
    diff = h.sum() - target
    if abs(diff) > tol:
        # distribute the residual greedily
        for i in range(h.size):
            if diff > 0 and h[i] > 0:
                d = min(diff, h[i])
                h[i] -= d
                diff -= d
            elif diff < 0 and h[i] < 1:
                d = min(-diff, 1 - h[i])
                h[i] += d
                diff += d
            if abs(diff) <= tol:
                break
    return h

def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a step function h (integral already enforced)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)

def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)

# SLSQP optimisation (used for a quick local improvement)
# --------------------------------------------------------------
def _solve_slsqp(h0: np.ndarray, max_iter: int = 200) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    h0 = _project_to_feasible(h0, target)

    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = corr0.max()
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c

    def _eq(x):
        return np.sum(x[:n]) - target

    cons = [{"type": "ineq", "fun": _ineq}, {"type": "eq", "fun": _eq}]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1], x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, target)
    return h_opt

        c5_val = max_corr * (2.0 / n)

        if c5_val < best_c5:
            best_c5, best_h, best_n = c5_val, h_opt.copy(), n

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Optional refinement on a finer grid if time permits
    # ------------------------------------------------------------------
    if best_h is not None and best_c5 > 0.3808 and best_n * 2 <= 512 and time_left() > 30.0:
        n2 = best_n * 2
        h2 = _upscale_h(best_h, n2)
        h2 = _project_to_feasible(h2, n2 / 2.0)
        if time_left() > 20.0:
            try:
                h2 = _solve_slsqp(h2, max_iter=150)
            except Exception:
                pass
        deadline = start + budget_s - 0.5
        h2_opt, max_corr2 = _hill_climb(h2, deadline)
        c5_val2 = max_corr2 * (2.0 / n2)
        if c5_val2 < best_c5:
            best_c5, best_h, best_n = c5_val2, h2_opt.copy(), n2

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    # final projection for safety and accurate C5 evaluation
    best_h = _project_to_feasible(best_h, best_n / 2.0)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
