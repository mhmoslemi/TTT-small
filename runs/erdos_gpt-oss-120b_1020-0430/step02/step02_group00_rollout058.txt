def _subgradient_descent(h, time_left_func, max_iters=5000):
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    # initial step
    step = 0.1
    decay = 0.995
    for it in range(max_iters):
        if time_left_func() < 0.5:
            break
        # compute correlation and max
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_val = np.max(corr)
        # find shifts achieving max
        max_idxs = np.where(corr == max_val)[0]
        # compute subgradient
        g = np.zeros_like(cur_h)
        for idx in max_idxs:
            shift = idx - (n - 1)
            # add contribution for this shift
            if shift >= 0:
                k = shift
                # first: g[:n-k] += 1 - cur_h[k:]
                g[: n - k] += (1.0 - cur_h[k:])
                # second: g[k:] -= cur_h[: n - k]
                g[k:] -= cur_h[: n - k]
            else:
                l = -shift
                # first: g[l:] += 1 - cur_h[: n - l]
                g[l:] += (1.0 - cur_h[: n - l])
                # second: g[: n - l] -= cur_h[l:]
                g[: n - l] -= cur_h[l:]
        # Normalize gradient maybe
        norm = np.linalg.norm(g)
        if norm == 0:
            break
        # step size adapt
        step = max(1e-4, step * decay)
        # compute tentative new h
        new_h = cur_h - step * g / norm
        new_h = _project_to_feasible(new_h, n)
        new_c5 = _c5_from_h(new_h)
        # accept if better
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5
        # else maybe occasional accept (skip)
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        for i in range(n):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a feasible step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h)=n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Run a single SLSQP optimisation from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = float(np.max(np.correlate(h0, 1.0 - h0, mode="full")))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse grid to fine grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _subgradient_descent(h: np.ndarray, time_left_func, max_iters: int = 8000) -> tuple:
    """Projected sub‑gradient descent on max‑correlation objective."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    step = 0.2
    decay = 0.9995

    for it in range(max_iters):
        if time_left_func() < 0.5:
            break

        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_val = np.max(corr)
        max_idxs = np.where(corr == max_val)[0]

        g = np.zeros_like(cur_h)
        for idx in max_idxs:
            shift = idx - (n - 1)
            if shift >= 0:
                k = shift
                g[: n - k] += (1.0 - cur_h[k:])
                g[k:] -= cur_h[: n - k]
            else:
                l = -shift
                g[l:] += (1.0 - cur_h[: n - l])
                g[: n - l] -= cur_h[l:]

        norm = np.linalg.norm(g)
        if norm == 0:
            break
        step = max(1e-5, step * decay)
        new_h = cur_h - (step / norm) * g
        new_h = _project_to_feasible(new_h, n)
        new_c5 = _c5_from_h(new_h)

        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5

    return best_h, best_c5

    start_time = time.time()

    # ------------------------------------------------------------------
