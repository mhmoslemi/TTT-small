def targeted_swap_local_search(h, rng, max_iters=5000, time_limit=budget_s, start_time):
    n = h.size
    best_c = compute_c5(h)
    for it in range(max_iters):
        # check time
        if time.time() - start_time > time_limit * 0.99:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)  # integer shift
        # compute valid mask
        idx_i = np.arange(n)
        idx_j = idx_i + shift
        valid = (idx_j >= 0) & (idx_j < n)
        # find bad and good positions
        bad_mask = valid & (h == 1) & ( (1 - h[idx_j]) == 1 )  # h[i]=1 && h[j]=0
        # But (1 - h[idx_j]) == 1 same as h[idx_j]==0
        bad_mask = valid & (h == 1) & (h[idx_j] == 0)
        good_mask = valid & (h == 0) & (h[idx_j] == 1)
        # find indices
        bad = np.where(bad_mask)[0]
        good = np.where(good_mask)[0]
        if bad.size > 0 and good.size > 0:
            i = rng.choice(bad)
            j = rng.choice(good)
            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h)
            if new_c < best_c:
                best_c = new_c
                # keep swap
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0
        else:

    return h, best_c

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def gradient_refine(h, rng, step=0.02, max_iter=8):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            # handle positive and negative shifts with same formula using wrap? Use generic
            # For each i, derivative of c_k w.r.t h_i:
            # c_k = sum_i h_i - sum_i h_i h_{i+k}
            # gradient = 1 - h_{i+k} - h_{i-k} (if both indices valid)
            # We'll compute contributions for both sides.
            # For simplicity we approximate using convolution:
            shift = k
            # Compute contributions from terms -h_i h_{i+shift}
            if shift > 0:
                g[:n-shift] -= h[shift:]
                g[shift:] -= h[:n-shift]
            else:
                s = -shift
                g[s:] -= h[:n-s]
                g[:n-s] -= h[s:]
            # add derivative of sum_i h_i: +1 for all i
            g += 1.0
        # step size decay
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def targeted_swap_local_search(h, rng, max_iters=2000, time_limit=0.0, start_time=0.0):
    n = h.size
    best_c = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        if time_limit and (time.time() - start_time) > time_limit * 0.99:
            break
        # compute correlation and worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)  # integer shift could be negative
        # build masks for bad/good positions
        i_arr = np.arange(n)
        j_arr = i_arr + shift
        valid = (j_arr >= 0) & (j_arr < n)
        # bad: h[i]=1 and h[j]=0
        bad = i_arr[valid & (h == 1) & (h[j_arr] == 0)]
        # good: h[i]=0 and h[j]=1
        good = i_arr[valid & (h == 0) & (h[j_arr] == 1)]
        improved = False
        if bad.size and good.size:
            i = rng.choice(bad)
            j = rng.choice(good)
            # swap
            h[i], h[j] = 0.0, 1.0
            new_c = compute_c5(h)
            if new_c < best_c - 1e-12:
                best_c = new_c
                improved = True
                no_improve = 0
            else:
                # revert
                h[i], h[j] = 1.0, 0.0
        if not improved:
            # random swap fallback
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size and zeros.size:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = 0.0, 1.0
                new_c = compute_c5(h)
                if new_c < best_c - 1e-12:
                    best_c = new_c
                    improved = True
                    no_improve = 0
                else:
                    # revert
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            no_improve += 1
            if no_improve >= 200:
                break
    return h, best_c

    start_time = time.time()
    best_h = None
    best_c = np.inf

    # handle initial parent if present

        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n0)
                add = min(diff, 1.0 - h0[i]) if diff > 0 else max(diff, -h0[i])
                h0[i] += add
                diff = target0 - h0.sum()
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, n0

    candidate_ns = [48, 60, 72, 84, 96, 108, 120]
    if best_n is not None and best_n not in candidate_ns:
        candidate_ns = [best_n] + candidate_ns

    # main stochastic search
    while time.time() - start_time < budget_s * 0.9:
        n = rng.choice(candidate_ns)
        # generate random binary feasible vector
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, n // 2, replace=False)
        h[ones_idx] = 1.0

        # local search with targeted swaps
        time_limit = budget_s - (time.time() - start_time)

        # small continuous refinement
        h = gradient_refine(h, rng, step=0.02, max_iter=10)
        c = compute_c5(h)
        if c < best_c:

        # occasional random perturb to escape if improvement stagnates
        if best_c < 0.381 and (time.time() - start_time) > budget_s * 0.8:
            # try a few more refinements around best
            for _ in range(3):
                h_tmp = best_h.copy()
                # random swap pair
                i = rng.integers(0, best_n)
                j = rng.integers(0, best_n)
                if i != j:
                    h_tmp[i], h_tmp[j] = h_tmp[j], h_tmp[i]
                h_tmp = gradient_refine(h_tmp, rng, step=0.01, max_iter=15)
                c_tmp = compute_c5(h_tmp)
                if c_tmp < best_c:
                    best_h, best_c = h_tmp.copy(), c_tmp
        # end while

    # final feasibility enforcement
    if best_h is None:
        best_n = 48
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
