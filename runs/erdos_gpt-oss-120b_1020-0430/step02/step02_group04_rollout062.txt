import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while fixing sum(v)=target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0≤h≤1, Σh=n/2)."""
    half = n // 2
    target = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target)
    h = np.concatenate([x, x[::-1]])
    if n % 2:
        mid = rng.random()
        mid = np.clip(mid, 0.0, 1.0)
        h = np.insert(h, half, mid)
        # re‑project sum (tiny adjustment)
        h = project_onto_box_sum(h, n / 2.0)
    return h


def perturb_h(h, rng, scale=0.2):

    max_eps = min(h[i], 1.0 - h[j])

    nh = h.copy()

    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        # enforce symmetry and feasibility
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c = compute_c5(new_h)
        if c < best_c:

        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Refinement via SLSQP on the smooth‑max surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def sum_eq(x):
        return np.sum(x) - target

    def soft_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        C = corr * (2.0 / n)
        bc = beta * C
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(
        soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints={"type": "eq", "fun": sum_eq},
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, target)
    x = (x + x[::-1]) * 0.5
    x = project_onto_box_sum(x, target)
    return x
