def _c5_from_h(h):
    n = h.shape[0]
    # compute linear correlation via FFT
    h = h.astype(np.float64)
    g = 1.0 - h
    size = 2 * n - 1
    # next power of two for speed
    L = 1 << (size - 1).bit_length()
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:size]
    # The linear correlation output length size, corr[k] corresponds to shift -(n-1) ... (n-1)
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures or lambdas)

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        for i in range(n):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _max_corr_unscaled(h: np.ndarray) -> float:
    """Maximum correlation without the 2/n scaling."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def _c5_from_h(h: np.ndarray) -> float:
    """Scaled C5 value for a step function h."""
    n = h.size
    return _max_corr_unscaled(h) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----- constraint objects for SLSQP (no closures) -----
class _IneqConstraint:
    def __init__(self, n: int):
        self.n = n

    def __call__(self, x: np.ndarray) -> np.ndarray:
        h = x[: self.n]
        t = x[-1]
        # t must dominate every correlation entry
        return t - np.correlate(h, 1.0 - h, mode="full")


class _EqConstraint:
    def __init__(self, n: int):
        self.n = n

    def __call__(self, x: np.ndarray) -> float:
        return np.sum(x[: self.n]) - (self.n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Refine h with SLSQP (minimising the auxiliary variable t)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = _max_corr_unscaled(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _IneqConstraint(n)},
        {"type": "eq", "fun": _EqConstraint(n)},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _binary_sa(h_bin: np.ndarray, time_left_func, target: float = 0.3808) -> tuple:
    """Simulated‑annealing on a binary (0/1) vector."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c5 = _c5_from_h(cur_h.astype(float))
    best_h = cur_h.copy()
    best_c5 = cur_c5

    max_iter = 20000
    T0, Tfinal = 0.02, 0.0005

    for it in range(max_iter):
        if time_left_func() < 0.5:
            break
        # pick random 1 and 0 positions
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # perform swap
        cur_h[i], cur_h[j] = 0, 1
        new_c5 = _c5_from_h(cur_h.astype(float))

        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            T = T0 * (1 - it / max_iter) + Tfinal * (it / max_iter)
            if np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                cur_c5 = new_c5
            else:
                # revert swap
                cur_h[i], cur_h[j] = 1, 0

        if best_c5 < target:
            break
    return best_h.astype(float), best_c5

            if time_left() < 4.0:
                break
            h_opt, c5_opt = _solve_h(cur_start, max_iter=500)
            update_candidate(h_opt, c5_opt)
            # small random perturbation for the next attempt
            pert = 0.08 * np.random.randn(n)
            cur_start = _project_to_feasible(h_opt + pert, n)

        if best_c5 < 0.3808:
            break

    # --------------------------------------------------------------
    # Final polishing on the best solution found so far
    # --------------------------------------------------------------
    if best_h is not None and time_left() > 5.0:
        h_opt, c5_opt = _solve_h(best_h, max_iter=1000)
        update_candidate(h_opt, c5_opt)

    # --------------------------------------------------------------
    # Fallback (should never happen)
    # --------------------------------------------------------------
    if best_h is None:
        n_fallback = 128
        best_h = np.full(n_fallback, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n_fallback
    else:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
