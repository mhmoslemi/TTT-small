def softmax_obj_and_grad(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    # compute exp shifted
    exp_shifted = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_shifted)
    obj = (max_bc + np.log(sum_exp)) / beta
    # weights w = exp(bc - max_bc) / sum_exp
    w = exp_shifted / sum_exp
    # Now compute gradient of obj wrt h = sum_s w_s * d corr_s / dh
    N = h.size
    g = np.zeros_like(h)
    # s index mapping: p = s + N - 1
    for p, wp in enumerate(w):
        if wp == 0.0:
            continue
        s = p - (N - 1)
        if s >= 0:
            # first term: (1 - h[j+s]) for j in 0..N-1-s
            # second term: -h[j-s] for j in s..N-1 (since j-s is >=0)
            # can update slices
            # first term:
            g[:N - s] += wp * (1 - h[s:])
            # second term:
            g[s:] -= wp * h[:N - s]
        else:
            # s negative => treat similarly using symmetry:
            # we can also compute using same formulas as above because j+s may be negative; we can just use s negative.
            # For s < 0, then first term (1 - h[j+s]) applies for j >= -s (so that j+s >=0) and j <= N-1.
            # That's same as slice: j from -s..N-1; length N + s (since s negative)
            # Let t = -s
            # first term: g[t:] += wp * (1 - h[:N - t])
            # second term: -h[j-s] = -h[j+|s|] for j in 0..N-1-|s| 
            # So g[:N + s] -= wp * h[-s:]
            t = -s
            # first term:
            g[t:] += wp * (1 - h[:N - t])
            # second term:
            g[:N + s] -= wp * h[t:]
    return obj, g

def slsqp_optimize_grad(h0, beta, rng, max_iter=200):
    n = h0.size
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # use log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        exp_shifted = np.exp(bc - m)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp
        N = n
        g = np.zeros_like(x)
        # loop over shifts
        # using for idx in range(len(w)):
        for p in range(len(w)):
            wp = w[p]
            if wp == 0.0:
                continue
            s = p - (N - 1)
            if s >= 0:
                g[:N - s] += wp * (1 - x[s:])
                g[s:] -= wp * x[:N - s]
            else:
                t = -s
                g[t:] += wp * (1 - x[:N - t])
                g[:N + s] -= wp * x[t:]
        return g
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    # Provide Jacobian for constraint maybe not needed.
    res = minimize(obj, np.clip(h0, 0.0, 1.0), method='SLSQP', jac=grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # Repair sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust with random indices
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------------------- helper utilities ---------------------- #
def compute_c5(h):
    """Exact C5 for a discrete step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def adjust_sum(x, target, rng, max_iter=2000):
    """Randomly adjust entries to hit the target sum (preserves bounds)."""
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x


def make_feasible(h, target, rng):
    """Clip and repair to satisfy bounds and sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    return adjust_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Move a small amount epsilon from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb: accept only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def softmax_obj_and_grad(h, beta):
    """Smooth max of the overlap integral and its gradient."""
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_shifted = np.exp(bc - m)
    sum_exp = np.sum(exp_shifted)
    obj = (m + np.log(sum_exp)) / beta

    w = exp_shifted / sum_exp  # softmax weights over shifts
    g = np.zeros_like(h)
    # loop over all shifts
    for p, wp in enumerate(w):
        if wp == 0.0:
            continue
        s = p - (N - 1)          # shift value
        if s >= 0:
            g[: N - s] += wp * (1 - h[s:])
            g[s:]      -= wp * h[: N - s]
        else:
            t = -s
            g[t:]      += wp * (1 - h[: N - t])
            g[: N + s] -= wp * h[t:]
    return obj, g


def slsqp_optimize_grad(h0, beta, rng, max_iter=250):
    """SLSQP on the soft‑max surrogate using analytical gradient."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def grad(x):
        N = n
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        exp_shifted = np.exp(bc - m)
        sum_exp = np.exp(-m) * np.sum(np.exp(bc))
        w = exp_shifted / sum_exp
        g = np.zeros_like(x)
        for p, wp in enumerate(w):
            if wp == 0.0:
                continue
            s = p - (N - 1)
            if s >= 0:
                g[: N - s] += wp * (1 - x[s:])
                g[s:]      -= wp * x[: N - s]
            else:
                t = -s
                g[t:]      += wp * (1 - x[: N - t])
                g[: N + s] -= wp * x[t:]
        return g

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        np.clip(h0, 0.0, 1.0),
        method="SLSQP",
        jac=grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return adjust_sum(x, target, rng)


def gradient_refine_worst_shift(h, rng, step=0.02, max_iter=6):
    """Local descent targeting the currently worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)      # shift with maximal overlap
        k = idx
        g = np.zeros_like(h)
        if k >= 0:
            upper = n - k
            g[:upper] += 1 - h[k:]
            g[k:]     -= h[:upper]
        else:
            kk = -k
            upper = n - kk
            g[kk:]    += 1 - h[:upper]
            g[:upper] -= h[kk:]
        sz = step * (0.5 ** it)
        h = h - sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = adjust_sum(h, target, rng)
    return h


def simulated_annealing(h, rng, max_iters=4000, init_temp=0.12, cooling=0.992):
    """Very light SA using the simple perturbation move."""
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    temp = init_temp
    for _ in range(max_iters):
        prop = perturb_h(cur, rng, scale=0.5)
        prop_c = compute_c5(prop)
        if prop_c < cur_c or rng.random() < np.exp((cur_c - prop_c) / temp):
            cur, cur_c = prop, prop_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
        temp *= cooling
        if temp < 1e-8:
            temp = 1e-8
    return best, best_c


def upsample(h, factor):
    """Nearest‑neighbour up‑sampling by integer factor."""
    return np.repeat(h, factor)


# --------------------------- main run --------------------------- #
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h, best_c, best_n = None, np.inf, None

    # ---- use supplied parent if any ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = make_feasible(h0, n0 / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # candidate grid sizes (focus on 120 and larger)
    candidate_ns = [120, 180, 240]
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    # time‑guard
    def time_left():
        return budget_s - (time.time() - start)

    # ---- optimisation over different discretisations ----
    for n in candidate_ns:
        if time_left() < 0.1 * budget_s:
            break
        max_attempts = 30 if n <= 120 else 15
        attempts = 0
        while attempts < max_attempts and time_left() > 0.05 * budget_s:
            attempts += 1
            # ---- initialise viable vector ----
            if best_n == n and best_h is not None and attempts == 1:
                h = best_h.copy()
            else:
                if rng.random() < 0.5:
                    h = np.zeros(n)
                    idx = rng.choice(n, n // 2, replace=False)
                    h[idx] = 1.0
                else:
                    h = rng.random(n)
                    h = make_feasible(h, n / 2.0, rng)

            # ---- cheap hill‑climb ----
            h, c = hill_climb(h, rng, max_iters=1500, start_step=0.5)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ---- optional SLSQP on promising candidates ----
            if c <= best_c + 0.005:
                for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                    if time_left() < 0.02 * budget_s:
                        break
                    h = slsqp_optimize_grad(h, beta, rng, max_iter=200)
                    c = compute_c5(h)
                    if c < best_c:
                        best_h, best_c, best_n = h.copy(), c, n

            # ---- refined gradient on worst shift ----
            if time_left() > 0.01 * budget_s:
                h = gradient_refine_worst_shift(h, rng, step=0.02, max_iter=8)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ---- short final hill‑climb ----
            if time_left() > 0.005 * budget_s:
                h, c = hill_climb(h, rng, max_iters=800, start_step=0.3)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # ---- up‑sample refinement (if time permits) ----
    if best_h is not None and time_left() > 0.1 * budget_s:
        if best_n * 2 <= 240:
            h_up = upsample(best_h, 2)
            n_up = h_up.size
            h_up = hill_climb(h_up, rng, max_iters=2500, start_step=0.5)[0]
            for beta in (5.0, 20.0, 80.0, 200.0):
                if time_left() < 0.01 * budget_s:
                    break
                h_up = slsqp_optimize_grad(h_up, beta, rng, max_iter=250)
            h_up = gradient_refine_worst_shift(h_up, rng, step=0.02, max_iter=10)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ---- final simulated‑annealing polish (if any time) ----
    if best_h is not None and time_left() > 0.02 * budget_s:
        best_h, best_c = simulated_annealing(best_h, rng, max_iters=2000)

    # ---- final feasibility enforcement ----
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = make_feasible(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
