def subgradient_descent(h0, max_iters, target, time_left):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    n = len(h)
    for k in range(1, max_iters+1):
        # compute correlation
        # Use FFT for speed
        n2 = 2*n
        H = np.fft.fft(h, n2)
        G = np.fft.fft(1.0 - h, n2)
        conv = np.fft.ifft(H*G).real[:2*n-1]
        maxc = conv.max()
        # Update best
        if maxc < best_c5:
            best_c5 = maxc
            best_h = h.copy()
        # compute gradient g
        # get lags indices where conv near maxc (within tolerance)
        # tolerance maybe 1e-12 or relative 1e-10
        # np.where(np.isclose(conv, maxc, atol=1e-12))[0]
        idxs = np.where(conv >= maxc - 1e-12)[0]  # maybe many duplicates but fine

        # compute gradient sums
        g = np.zeros_like(h)
        for idx in idxs:
            lag = idx - (n - 1)
            if lag >= 0:
                # i from 0 to n-1-lag
                i_end = n - lag
                # contributions
                # g[i] += 1 - h[i+lag]
                # g[i+lag] -= h[i]
                # vectorized
                i_range = slice(0, i_end)
                j_range = slice(lag, n)
                # To avoid creating new arrays large times, use slicing
                g[i_range] += 1.0 - h[j_range]
                g[j_range] -= h[i_range]
            else:
                # lag negative
                lag_pos = -lag
                i_end = n - lag_pos
                # i from lag_pos to n-1
                i_range = slice(lag_pos, n)
                j_range = slice(0, i_end)  # j = i - lag_pos
                g[i_range] += 1.0 - h[j_range]
                g[j_range] -= h[i_range]

        # step size: alpha = 1.0 / np.sqrt(k) maybe scaled by factor
        alpha = 1.0 / np.sqrt(k)
        # update
        h = h - alpha * g
        # clip to [0,1]
        np.clip(h, 0.0, 1.0, out=h)
        # project to sum = target
        h = _project_to_simplex_box(h, target)
        # time check
        if k % 100 == 0 and time_left() < 5:
            break
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n ∩ {sum = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a discrete step function using linear convolution via FFT."""
    n = h.size
    n2 = 2 * n
    H = np.fft.fft(h, n2)
    G = np.fft.fft(1.0 - h, n2)
    conv = np.fft.ifft(H * G).real[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape sampled on n points, then projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 400):
    """Refine h with SLSQP (same routine as the reference implementation)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _subgradient_descent(start_h: np.ndarray, n_iters: int, target: float, time_left):
    """Projected sub‑gradient descent on max_{k}∫h(1−h(·+k))."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_from_h(h)

    for it in range(1, n_iters + 1):
        n = h.size
        # correlation via FFT
        n2 = 2 * n
        H = np.fft.fft(h, n2)
        G = np.fft.fft(1.0 - h, n2)
        conv = np.fft.ifft(H * G).real[: 2 * n - 1]
        maxc = conv.max()
        if maxc < best_val:
            best_val = maxc
            best_h = h.copy()
        # lag attaining the maximum
        idx = np.argmax(conv)
        lag = idx - (n - 1)

        # gradient for this lag (direction of increase)
        g = np.zeros_like(h)
        if lag >= 0:
            i_end = n - lag
            i_slice = slice(0, i_end)
            j_slice = slice(lag, n)
            g[i_slice] += 1.0 - h[j_slice]
            g[j_slice] -= h[i_slice]
        else:
            lp = -lag
            i_end = n - lp
            i_slice = slice(lp, n)
            j_slice = slice(0, i_end)
            g[i_slice] += 1.0 - h[j_slice]
            g[j_slice] -= h[i_slice]

        # step size (diminishing)
        alpha = 0.5 / np.sqrt(it)
        h = h - alpha * g
        np.clip(h, 0.0, 1.0, out=h)
        h = _project_to_simplex_box(h, target)

        if it % 500 == 0 and time_left() < 5.0:
            break

    final_val = _c5_from_h(best_h)
    return best_h, final_val
