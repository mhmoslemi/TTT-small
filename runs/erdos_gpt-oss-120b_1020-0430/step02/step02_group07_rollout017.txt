def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialization
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)

        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:

        h0 = random_feasible_vector(n, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # generate seeds list
    seeds = [h0]
    # add few Sturmian seeds
    alphas = [ (np.sqrt(5)-1)/2, np.sqrt(2)-1, np.pi - np.floor(np.pi)]
    for a in alphas:
        h = generate_sturmian(n, a)
        h = enforce_sum(h.astype(float), n/2.0, rng)
        seeds.append(h)

    # random seeds added later
    # perform hill climb on each seed
    for h0seed in seeds:
        if time.time() - start > budget_s * 0.5:
            break
        # hill climb
        h_opt, c_opt = hill_climb(h0seed, rng, max_iters=3000, start_step=0.5)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # also random restarts
    while time.time() - start < budget_s * 0.6:
        h_rand = random_feasible_vector(n, rng)
        h_opt, c_opt = hill_climb(h_rand, rng, max_iters=2000, start_step=0.5)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # continuation: increase beta
    betas = [5,20,80,200,500,1000]
    for b in betas:
        if time.time() - start > budget_s * 0.75:
            break
        h_opt = slsqp_optimize(best_h, beta=b, max_iter=500)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        # hill climb after each beta
        h_opt2, c_opt2 = hill_climb(best_h, rng, max_iters=2000, start_step=0.3)
        if c_opt2 < best_c:
            best_h, best_c = h_opt2, c_opt2

    # upsample if time permits
    if time.time() - start < budget_s * 0.9:
        # factor 2 upsample
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        # deep refine: hill climb + maybe simulated annealing
        h_up_opt, c_up_opt = hill_climb(h_up, rng, max_iters=4000, start_step=0.3)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with sum = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def generate_sturmian(n, alpha):
    """Sturmian (balanced) binary word of length n with density 1/2."""
    i = np.arange(n, dtype=float)
    frac = (i * alpha) % 1.0
    h = (frac < 0.5).astype(float)
    return h


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i(1‑x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP minimisation of the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair tiny sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Deterministic hill‑climb accepting only improvements."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(max_iters // 8, 1)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=3000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by integer factor whilst preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target = (n * factor) / 2.0
    return enforce_sum(h_up, target, rng)

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

        if c_sa < c_up:

        # Accept if improvement

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

    best_h = np.clip(best_h, 0.0, 1.0)
