def greedy_improve(h, rng, max_swaps, time_limit):
    n = h.size
    best_c = compute_c5(h)
    best_h = h.copy()
    swaps = 0
    while swaps < max_swaps and time.time() < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (n - 1)
        # get shifted array depends on sign of k
        if k >= 0:
            shifted = np.empty_like(h)
            shifted[:n - k] = h[k:]
            shifted[n - k:] = 0.0
        else:
            kk = -k
            shifted = np.empty_like(h)
            shifted[kk:] = h[:n - kk]
            shifted[:kk] = 0.0
        # find candidates
        L = np.where((h == 1) & (shifted == 0))[0]  # contributes 1 to Ck
        R = np.where((h == 0) & (shifted == 1))[0]  # would contribute 0 after placing 1
        improved = False
        if L.size > 0 and R.size > 0:
            # try random pairs up to some attempts
            for _ in range(10):
                i = rng.choice(L)
                j = rng.choice(R)
                # perform swap
                h[i] = 0.0
                h[j] = 1.0
                new_c = compute_c5(h)
                if new_c < best_c - 1e-9:  # improvement
                    best_c = new_c
                    best_h = h.copy()
                    improved = True
                    swaps += 1
                    break
                else:
                    # revert
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            # try flip random pair (maybe reduces)
            # random swap any 1 and 0 with probability accept if improves
            # or just break if cannot improve
            # We'll attempt a few random swaps to escape.
            for _ in range(10):
                i = rng.integers(0, n)

                    zero_idxs = np.where(h == 0)[0]

                        break
                    j = rng.choice(zero_idxs)

                    if one_idxs.size == 0:

                # no improvement found, break out of while loop
                break

    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact C5 for a step function on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _fix_sum(h, rng):
    """Adjust h to satisfy Σh = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def _shift(arr, k):
    """Return arr shifted by k (positive = left shift), zero padding outside."""
    n = arr.size
    out = np.empty_like(arr)
    if k >= 0:
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
    else:
        kk = -k
        out[kk:] = arr[: n - kk]
        out[:kk] = 0.0
    return out


def greedy_improve(h, rng, max_swaps, time_limit):
    """Greedy swaps reducing the worst overlap; returns best vector seen."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    swaps = 0
    no_improve = 0
    while swaps < max_swaps and time.time() < time_limit and no_improve < 500:
        corr = np.correlate(h, 1 - h, mode="full")
        k = np.argmax(corr) - (n - 1)          # shift giving maximal overlap
        shifted = _shift(h, k)

        L = np.where((h == 1) & (shifted == 0))[0]   # 1‑positions hurting this shift
        R = np.where((h == 0) & (shifted == 1))[0]   # 0‑positions that would help

        improved = False
        if L.size and R.size:
            for _ in range(5):
                i = rng.choice(L)
                j = rng.choice(R)
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    swaps += 1
                    break
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            # fallback random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                improved = True
                swaps += 1

                h[i] = 1.0
                h[j] = 0.0

            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Small gradient steps targeting the worst shift, keeping feasibility."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # (1 - h[i+k])
            g[k:] -= h[:upper]              # -h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # repair sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def _softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP on the soft‑max surrogate."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        return _softmax_obj(x, beta)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum repair
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
