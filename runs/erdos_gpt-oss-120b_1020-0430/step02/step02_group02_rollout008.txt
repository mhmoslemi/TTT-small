def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_smooth_max(c, beta):
    # returns f approx, and weights w
    maxc = c.max()
    exp_vals = np.exp(beta * (c - maxc))
    sum_exp = exp_vals.sum()
    f = (np.log(sum_exp) + beta*maxc) / beta
    w = exp_vals / sum_exp
    return f, w

def compute_gradient_smooth(h, w):
    n = h.size
    grad = np.zeros_like(h)
    # w indexed same as correlation c of length 2n-1, with shift s = idx - (n-1)
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            # forward
            if n - s > 0:
                grad[:n - s] += -weight * h[s:]
                grad[s:] += -weight * h[:n - s]
        else:
            s2 = -s
            if n - s2 > 0:
                grad[s2:] += -weight * h[:n - s2]
                grad[:n - s2] += -weight * h[s2:]
    return grad

def _hill_climb(h, max_steps, time_limit):
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        # choose i, j
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # maximum possible transfer:
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        # also consider reverse direction probabilistically: perhaps also move from j to i, but that's symmetric via random picks
        if max_eps <= 0:
            continue
        # choose epsilon as random fraction up to max_eps
        eps = max_eps * np.random.rand() * 0.5  # 0-0.5*max_eps
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

import time
import numpy as np

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip h to [0,1] and adjust to satisfy sum(h)=n/2 using a water‑filling approach."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function h on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """
    Simple stochastic hill‑climbing: move epsilon between two coordinates
    or swap two values. Accept only improvements.
    """
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    start = time.time()
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # epsilon transfer
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps > 0:
            eps = np.random.rand() * max_eps * 0.5
            cand = best_h.copy()
            cand[i] -= eps
            cand[j] += eps
            c = _c5_from_h(cand)
            if c < best_c:
                best_h, best_c = cand, c
                continue
        # swap attempt
        if best_h[i] != best_h[j]:
            cand = best_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
            c = _c5_from_h(cand)
            if c < best_c:
                best_h, best_c = cand, c
    return best_h, best_c


def _smooth_optimize(h_init: np.ndarray, time_limit: float, target: float = 0.3808) -> tuple:
    """Gradient descent on a smooth approximation of the max‑overlap objective."""
    n = h_init.size
    h = _project_to_feasible(h_init, n)
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    start = time.time()
    beta = 5.0          # smoothing parameter
    step = 0.1          # initial step size
    it = 0
    while time.time() - start < time_limit:
        # correlation and current bound
        c = np.correlate(h, 1.0 - h, mode="full")
        cur_max = c.max()
        cur_c = cur_max * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
            if cur_c < target:
                break
        # smooth max via log‑sum‑exp
        exp_w = np.exp(beta * (c - cur_max))
        w = exp_w / exp_w.sum()
        # gradient of the smooth objective (before projection)
        grad = np.zeros_like(h)
        off = n - 1
        for shift in range(-n + 1, n):
            weight = w[shift + off]
            if weight == 0.0:
                continue
            if shift >= 0:
                s = shift
                if s < n:
                    grad[: n - s] -= weight * h[s:]
                    grad[s:]      -= weight * h[: n - s]
            else:
                s = -shift
                if s < n:
                    grad[s:]      -= weight * h[: n - s]
                    grad[: n - s] -= weight * h[s:]
        # line search over a few step sizes
        cand_h = None
        cand_c = float('inf')
        for a in (step, step * 0.5, step * 0.2, step * 0.05):

            c_tmp = _c5_from_h(h_tmp)

        if cand_h is None:

        # accept candidate

        h = cand_h

    # final local refinement with hill climbing

        best_h, best_c = _hill_climb(best_h, max_steps=4000, time_limit=limit)
    return best_h, best_c
