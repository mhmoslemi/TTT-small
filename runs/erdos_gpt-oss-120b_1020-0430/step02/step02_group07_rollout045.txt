def improve_h(h, rng, total_time, start_time):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # Phase: repeated cycles of SA then hill climb
    while time.time() - start_time < total_time:
        # SA
        cur_h = best_h.copy()
        cur_c = best_c
        temp = 1.0
        for i in range(5000):
            new_h = perturb_h(cur_h, rng, scale=0.1)
            new_c = compute_c5(new_h)
            if new_c < cur_c or rng.random() < math.exp((cur_c - new_c)/temp):
                cur_h, cur_c = new_h, new_c
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            temp *= 0.9995
            if time.time() - start_time > total_time:
                break
        # hill climb
        cur_h = best_h.copy()
        cur_c = best_c
        step = 0.3
        for i in range(2000):
            new_h = perturb_h(cur_h, rng, scale=step)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                cur_h, cur_c = new_h, new_c
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            if i % 200 ==0:
                step *= 0.8
            if time.time() - start_time > total_time:
                break
    return best_h, best_c

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = len(h)

        corr = np.correlate(h, 1 - h, mode='full')

        grad = np.zeros(n)

            grad[k:] -= h[:n-k]

            kk = -k
            grad[kk:] += (1 - h[:n-kk])
            grad[:n-kk] -= h[kk:]

        neg_candidates = np.where((grad < -1e-12) & (h < 1-1e-12))[0]

        h *= target / s
    else:
        h[:] = 0.0
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def simulated_annealing(h_start, rng, max_iters=2000,
                        init_temp=0.02, decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < math.exp((cur_c - new_c) / max(temp, 1e-12)):
            h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    return enforce_sum(h_up, target_up, rng)

def max_shift_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)           # shift relative to zero
    grad = np.zeros(n)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:]   -= h[:n - k]
    else:
        kk = -k
        grad[kk:]  += (1.0 - h[:n - kk])
        grad[:n - kk] -= h[kk:]
    return k, grad

def gradient_descent_max_shift(h, rng, max_iters=2000, eps_frac=0.5):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    for it in range(max_iters):
        k, grad = max_shift_gradient(h)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[rng.integers(0, pos.size)]
        j = neg[rng.integers(0, neg.size)]
        max_tr = min(h[i], 1.0 - h[j])
        if max_tr <= 0:
            continue
        eps = max_tr * eps_frac
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c <= best_c:
            h, best_c, best_h = new_h, c, new_h.copy()
        eps_frac = max(eps_frac * 0.995, 1e-4)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- first round of refinement (continuous) --------------------
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0)
    for beta in betas:
        if time.time() - start > budget_s * 0.55:
            break
        try:
            h_opt = slsqp_optimize(best_h, beta=beta, max_iter=500)
        except Exception:
            continue
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        # short hill climb
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.3)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ----- targeted gradient descent on the worst shift ---------------
    if time.time() - start < budget_s * 0.60:
        h_gd, c_gd = gradient_descent_max_shift(best_h, rng,
                                                max_iters=3000, eps_frac=0.5)
        if c_gd < best_c:
            best_h, best_c = h_gd, c_gd

    # ----- simulated annealing for diversification --------------------
    if time.time() - start < budget_s * 0.65:
        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iters=4000,
                                         init_temp=0.05,
                                         decay=0.9995,
                                         scale=0.25)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ----- up‑sample if time permits ----------------------------------
    if time.time() - start < budget_s * 0.90:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        # intensive refinement at higher resolution
        for beta in betas:
            if time.time() - start > budget_s * 0.95:
                break
            try:
                h_opt = slsqp_optimize(h_up, beta=beta, max_iter=500)
            except Exception:
                continue
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                h_up, best_c = h_opt, c_opt
            h_hc, c_hc = hill_climb(h_up, rng, max_iters=3000, start_step=0.3)
            if c_hc < best_c:
                h_up, best_c = h_hc, c_hc
        # final gradient descent on upsampled version
        if time.time() - start < budget_s * 0.98:
            h_gd, c_gd = gradient_descent_max_shift(h_up, rng,
                                                    max_iters=4000,
                                                    eps_frac=0.5)
            if c_gd < best_c:
                h_up, best_c = h_gd, c_gd
        # adopt if better
        if best_c < compute_c5(best_h):
            best_h, n = h_up, n_up

    # ----- final cleanup ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
