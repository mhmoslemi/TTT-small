def anneal_binary(h_start, rng, max_iters, init_temp=5.0, decay=0.99995, scale=1):
   # start with given vector, but ensure it's binary
   h = h_start.copy()
   cur_c = compute_c5(h)
   best_h = h.copy()
   best_c = cur_c
   # derive initial lists
   ones = list(np.where(h == 1)[0])
   zeros = list(np.where(h == 0)[0])
   temp = init_temp
   for it in range(max_iters):
       if not ones or not zeros:
           break
       i_idx = rng.integers(len(ones))
       j_idx = rng.integers(len(zeros))
       i = ones[i_idx]
       j = zeros[j_idx]
       # propose swap
       h_new = h.copy()
       h_new[i] = 0
       h_new[j] = 1
       new_c = compute_c5(h_new)
       accept = False
       if new_c < cur_c:
           accept = True
       else:
           # probability
           prob = np.exp(-(new_c - cur_c) / temp)
           if rng.random() < prob:
               accept = True
       if accept:
           # accept
           h = h_new
           cur_c = new_c
           # update ones/zeros lists
           ones[i_idx] = j
           zeros[j_idx] = i
           # update best
           if cur_c < best_c:
               best_c = cur_c
               best_h = h.copy()
       temp *= decay
   return best_h, best_c

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the SLSQP surrogate (used by softmax_obj)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for a step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary(n, rng):
    """Generate a random 0‑1 vector with exactly n/2 ones."""
    half = n // 2
    inds = rng.choice(n, half, replace=False)
    h = np.zeros(n, dtype=float)
    h[inds] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb on the exact C₅ (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the sum exactly (numeric drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def continuous_refine(h, rng):
    """Apply a short ladder of SLSQP + hill‑climb on a real‑valued h."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
        trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    # final hill–climb
    trial_h, c_h = hill_climb(best_h, rng, max_iters=1500, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up

# Binary‑specific utilities
# ----------------------------------------------------------------------
def anneal_binary(h_start, rng, max_iters, init_temp=5.0, decay=0.99995):
    """Simulated annealing on pure 0‑1 vectors (swap one 1↔0 each step)."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    temp = init_temp
    for _ in range(max_iters):
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_c = compute_c5(h_new)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            prob = math.exp(-(new_c - cur_c) / temp) if temp > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            h = h_new
            cur_c = new_c
            # update the index lists in place
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        temp *= decay
    return best_h, best_c


def binary_local_opt(h, rng, max_iters=4000):
    """Deterministic improvement: swap a (1,0) pair with a (0,1) pair for the worst shift."""
    h = h.copy()
    N = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        max_idx = corr.argmax()
        shift = max_idx - (N - 1)
        if max_corr == 0:
            break
        K = abs(shift)
        if K == 0:
            break
        # find indices that participate in a (1,0) pair for this shift
        if shift >= 0:
            rng_len = N - K
            i_mask = (h[:rng_len] == 1) & (h[K:] == 0)
            j_mask = (h[:rng_len] == 0) & (h[K:] == 1)
            i_cand = np.where(i_mask)[0]
            j_cand = np.where(j_mask)[0]
        else:  # shift < 0
            rng_len = N - K
            i_mask = (h[K:] == 1) & (h[:rng_len] == 0)
            j_mask = (h[K:] == 0) & (h[:rng_len] == 1)
            i_cand = np.where(i_mask)[0] + K
            j_cand = np.where(j_mask)[0] + K
        if i_cand.size == 0 or j_cand.size == 0:
            break
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        if new_c < best_c:
            best_c, best_h = new_c, h.copy()
        else:
            # revert – no improvement
            h[i], h[j] = h[j], h[i]
        # early exit if already very good
        if best_c <= (68 * 2.0 / N) + 1e-12:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise – use parent if available, otherwise a random binary start
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():

        # keep the provided vector as a continuous candidate
        cont_best_h = init_h.copy()
        cont_best_c = compute_c5(cont_best_h)

        # also create a binary version (top half entries set to 1)
        half = n // 2
        idx = np.argsort(-init_h)[:half]
        bin_h = np.zeros(n, dtype=float)
        bin_h[idx] = 1.0
        best_h = bin_h
        best_c = compute_c5(best_h)
        # keep the best of the two starters
        if cont_best_c < best_c:
            best_h, best_c = cont_best_h, cont_best_c
    else:
        n = 360
        best_h = random_binary(n, rng)
        best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1 – binary simulated annealing + deterministic local search
    # ------------------------------------------------------------------
    TIME_BINARY = budget_s * 0.60
    while time.time() - start_time < TIME_BINARY:
        remaining = budget_s - (time.time() - start_time)
        # rough estimate: each iteration ~0.00045 s (correlation + bookkeeping)
        max_iters = int(min(200_000, remaining / 0.00045))
        if max_iters <= 0:
            break

        # occasional fresh random restart
        if rng.random() < 0.15:
            start_vec = random_binary(n, rng)
        else:
            start_vec = best_h.copy()

        # annealing on the binary space
        h_sa, c_sa = anneal_binary(start_vec, rng, max_iters)

        # deterministic refinement targeting the worst shift
        h_loc, c_loc = binary_local_opt(h_sa, rng, max_iters=3000)

        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

        # Early exit if we have reached the integer target (≈68 pairs)
        if best_c <= (68 * 2.0 / n) + 1e-12:
            break

    # ------------------------------------------------------------------
    # Phase 2 – continuous refinement (SLSQP + hill‑climb)
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.90:
        best_h, best_c = continuous_refine(best_h, rng)

    # ------------------------------------------------------------------
    # Phase 3 – optional up‑sampling if still some time left
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.97 and n * 2 <= 800:
        # one upsample step
        h_up = upsample_h(best_h, factor=2, rng=rng)
        up_h, up_c = continuous_refine(h_up, rng)
        if up_c < best_c:

            n = best_h.size

    # ------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
