def local_search(h, rng, max_moves, step_scale):
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_moves):
        i = rng.integers(0, n)
        j = rng.integers(0, n)

        max_eps = min(best_h[i], 1 - best_h[j])

    return best_h, best_c

            # start with -1 for all entries
            dir_vec = np.full_like(h, -1.0)
            if offset >= 0:
                # h_i+offset
                dir_vec[:n - offset] += h[offset:]
                # h_i - offset
                dir_vec[offset:] += h[:n - offset]
            else:
                off = -offset
                dir_vec[off:] += h[:n - off]
                dir_vec[:n - off] += h[off:]
            d_sum += dir_vec
        # average if multiple offsets
        d = d_sum / len(idxs)
        # step size schedule
        step = alpha0 / np.sqrt(t + 1.0)
        h = h + step * d
        # project
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# basic utilities
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project onto [lower,upper] with exact sum = target (box‑sum projection)."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def subgrad_optimize(h0, rng, max_iters=1500, alpha0=0.07):
    """Projected sub‑gradient descent on f(h)=max_k O(k)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(max_iters):
        Ovec = np.correlate(h, 1.0 - h, mode="full")
        max_val = Ovec.max()
        idxs = np.where(Ovec >= max_val - 1e-12)[0]
        d_sum = np.zeros_like(h)
        for idx in idxs:
            offset = idx - (n - 1)
            dir_vec = np.full_like(h, -1.0)
            if offset >= 0:
                dir_vec[: n - offset] += h[offset:]
                dir_vec[offset:] += h[: n - offset]
            else:
                off = -offset
                dir_vec[off:] += h[: n - off]
                dir_vec[: n - off] += h[off:]
            d_sum += dir_vec
        d = d_sum / len(idxs)
        step = alpha0 / np.sqrt(t + 1.0)
        h = h + step * d
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

# smooth‑max SLSQP refinement (kept from previous code)
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
