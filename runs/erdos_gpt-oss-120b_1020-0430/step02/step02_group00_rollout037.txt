def project_to_feasible(h, n, tol=1e-9, max_iter=50):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        # distribute diff equally among all free variables (not at bounds)
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds but sum not correct, adjust smallest distance from bound
            # just adjust any element
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] += diff
        else:
            delta = diff / cnt
            h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correct small residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust some entries
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)
```

import time
import numpy as np

def shift(v, k):
    n = v.shape[0]
    out = np.zeros_like(v)
    if k >= 0:
        out[:n - k] = v[k:]
    else:
        out[-k:] = v[:n + k]
    return out

def project_to_feasible(h, n, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt == 0:
            # adjust any element
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] += diff
        else:
            delta = diff / cnt
            h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    try:
        return float(evaluate_erdos_solution(h))
    except Exception:
        n = len(h)
        dx = 2.0 / n
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def to_binary(h):
    n = h.shape[0]
    k = n // 2
    idx = np.argsort(-h)
    bh = np.zeros(n, dtype=float)
    bh[idx[:k]] = 1.0
    return bh

def subgradient_descent(start_h, deadline, start_step=0.3):
    n = len(start_h)
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best_h = cur.copy()
    best_val = cur_val
    it = 0
    while time.time() < deadline:
        # compute full correlation (unscaled)
        corr_full = np.correlate(cur, 1.0 - cur, mode='full')
        max_corr = corr_full.max()
        active = np.where(corr_full >= max_corr - 1e-12)[0]
        ks = active - (n - 1)
        # subgradient
        g = np.full(n, float(len(ks)), dtype=float)
        for k in ks:
            g -= shift(cur, k) + shift(cur, -k)
        g /= len(ks)
        step = start_step / np.sqrt(it + 1)
        new = cur - step * g
        new = project_to_feasible(new, n)
        new_val = compute_c5(new)
        if new_val <= cur_val + 1e-12:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_h, best_val = new.copy(), new_val
        it += 1
        # safety break if it grows huge
        if it > 200000:
            break
    return best_h, best_val

def simulated_annealing_binary(start_h, deadline):
    n = len(start_h)
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best_h = cur.copy()
    best_val = cur_val
    iteration = 0
    # initial temperature roughly scale of C5 differences
    T0 = 0.02
    while time.time() < deadline:
        # pick two distinct indices with opposite values
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:
            continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = compute_c5(cur)
        delta = new_val - cur_val
        T = T0 * (0.9995 ** iteration)
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val, best_h = new_val, cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]
        iteration += 1
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is not None:
        h0 = init_h.copy()
        n = h0.size
    else:
        n = 256
        h0 = np.random.rand(n)

    h0 = project_to_feasible(h0, n)

    best_h = h0.copy()

import time
import numpy as np

# -------------------------------------------------------------
# Utility functions (all top‑level, no closures)
# -------------------------------------------------------------
def shift(v: np.ndarray, k: int) -> np.ndarray:
    n = v.shape[0]
    out = np.zeros_like(v)
    if k >= 0:
        out[: n - k] = v[k:]
    else:
        out[-k :] = v[: n + k]
    return out


def project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt == 0:  # all at bounds – adjust the most central entry
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] += diff
        else:
            delta = diff / cnt
            h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function on [0,2] using the same discretisation as the evaluator."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def to_binary(h: np.ndarray) -> np.ndarray:
    """Convert a real vector to a binary one with exactly n/2 ones."""
    n = h.shape[0]
    k = n // 2
    idx = np.argsort(-h)  # descending
    bh = np.zeros(n, dtype=float)
    bh[idx[:k]] = 1.0
    return bh


def subgradient_descent(start_h: np.ndarray, deadline: float, start_step: float = 0.3) -> tuple:
    """Sub‑gradient descent on the max‑correlation objective."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best_h, best_val = cur.copy(), cur_val
    it = 0
    while time.time() < deadline:
        # unscaled correlation array (zero‑padding)
        corr_full = np.correlate(cur, 1.0 - cur, mode="full")
        max_corr = corr_full.max()
        active = np.where(corr_full >= max_corr - 1e-12)[0]
        ks = active - (n - 1)

        # sub‑gradient (average over all active shifts)
        g = np.full(n, float(len(ks)), dtype=float)
        for k in ks:
            g -= shift(cur, k) + shift(cur, -k)
        g /= len(ks)

        step = start_step / np.sqrt(it + 1)
        new = cur - step * g
        new = project_to_feasible(new, n)
        new_val = compute_c5(new)

        # monotone acceptance
        if new_val <= cur_val + 1e-12:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_h, best_val = new.copy(), new_val
        it += 1
        if it > 200000:  # safety guard
            break
    return best_h, best_val


def simulated_annealing_binary(start_h: np.ndarray, deadline: float) -> tuple:
    """Swap‑based SA on binary vectors."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best_h, best_val = cur.copy(), cur_val
    iteration = 0
    T0 = 0.02
    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_val = compute_c5(cur)
        delta = new_val - cur_val
        T = T0 * (0.9995 ** iteration)
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_h, best_val = cur.copy(), new_val
        else:
            cur[i], cur[j] = cur[j], cur[i]  # revert
        iteration += 1
    return best_h, best_val
