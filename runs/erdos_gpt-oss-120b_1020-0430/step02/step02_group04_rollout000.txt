def compute_grad(h, w):
    n = h.size
    g_raw = np.ones(n)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g_raw[:n - k] -= wk * h[k:]
            g_raw[k:] -= wk * h[:n - k]
        else:
            kk = -k
            g_raw[kk:] -= wk * h[:n - kk]
            g_raw[:n - kk] -= wk * h[kk:]
    return (2.0 / n) * g_raw

def subgradient_optimize(h0, max_iters, init_step, rng):

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 value for step function h on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project onto [lower,upper] with exact sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0≤h≤1, Σh=n/2, n even)."""
    assert n % 2 == 0
    half = n // 2
    target_half = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_transfer * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 10) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ---------- soft‑max SLSQP utilities ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- subgradient optimizer ----------
def compute_grad(h, w):
    """Gradient of weighted max (weights w over shifts)."""
    n = h.size
    g_raw = np.ones(n, dtype=float)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g_raw[: n - k] -= wk * h[k:]
            g_raw[k:] -= wk * h[: n - k]
        else:
            kk = -k
            g_raw[kk:] -= wk * h[: n - kk]
            g_raw[: n - kk] -= wk * h[kk:]
    return (2.0 / n) * g_raw


def subgradient_optimize(h0, max_iters, init_step, rng):
    """Projected subgradient descent on the exact max objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        active = np.where(corr == max_val)[0]
        w = np.zeros_like(corr, dtype=float)
        if active.size:
            w[active] = 1.0 / active.size
        grad = compute_grad(h, w)
        new_h = h - step * grad
        new_h = project_onto_box_sum(new_h, n / 2.0)
        # enforce symmetry (helps stability)
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            h = new_h
        else:
            if rng.random() < 0.001:
                h = new_h
        step *= 0.9995
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 180
        dx = 2.0 / n
        xs = (np.arange(n) + 0.5) * dx
        h0 = 0.5 * (1 + np.cos(np.pi * (xs - 1.0)))  # cosine bump, exact integral 0.5
        h0 = project_onto_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- quick hill‑climb (optional) -------------------------------
    if time.time() - start < budget_s * 0.2:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=1500, start_step=0.4)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ----- main stochastic optimisation loop -------------------------
    while time.time() - start < budget_s * 0.85:
        # build a fresh start
        if rng.random() < 0.3:
            h_start = (best_h + random_feasible_symmetric(n, rng)) * 0.5
        else:
            if rng.random() < 0.2:
                dx = 2.0 / n
                xs = (np.arange(n) + 0.5) * dx
                h_start = 0.5 * (1 + np.cos(np.pi * (xs - 1.0)))
            else:
                h_start = random_feasible_symmetric(n, rng)
        h_start = (h_start + h_start[::-1]) * 0.5
        h_start = project_onto_box_sum(h_start, n / 2.0)

        # subgradient phase
        max_iters = 3000
        # shorten near the deadline
        if time.time() - start > budget_s * 0.75:
            max_iters = 1200
        h_opt, c_opt = subgradient_optimize(h_start, max_iters=max_iters,
                                            init_step=1.0, rng=rng)

        # optional fine hill‑climb
        if time.time() - start < budget_s * 0.80:
            h_opt2, c_opt2 = hill_climb(h_opt, rng, max_iters=800, start_step=0.2)
            if c_opt2 < c_opt:
                h_opt, c_opt = h_opt2, c_opt2

        # short SLSQP refinement
        try:
            for beta in (80.0, 200.0, 500.0):
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=200)
        except Exception:
            pass
        c_opt = compute_c5(h_opt)

        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ----- optional up‑sampling ---------------------------------------
    if time.time() - start < budget_s * 0.95 and best_h.size * 2 <= 500:
        h_up = np.repeat(best_h, 2)
        h_up, c_up = subgradient_optimize(h_up, max_iters=2000,
                                          init_step=0.8, rng=rng)
        h_up, c_up = hill_climb(h_up, rng, max_iters=1200, start_step=0.2)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # ----- final clean‑up ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
