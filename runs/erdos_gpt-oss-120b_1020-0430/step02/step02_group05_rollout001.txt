def _grad_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxs = np.where(corr == max_corr)[0]  # potentially multiple
    grad = np.zeros(n)
    for idx in idxs:
        k = idx - (n - 1)   # shift offset: positive => shift forward
        if k >= 0:
            # i from 0 to n - 1 - k
            # grad[i] += (1 - h[i + k])
            grad[:n - k] += (1.0 - h[k:])
            # grad[i + k] += -h[i]
            grad[k:] += -h[:n - k]
        else:
            # k negative => shift backward
            k_abs = -k
            # For shift -k: correlation term sum_i h[i] * (1 - h[i - k_abs]), where i from k_abs to n-1
            # Equivalent to using same as positive with roles swapped? Use i range k_abs..n-1.
            # Derivative wrt h[i]: (1 - h[i - k_abs]) for i >= k_abs
            grad[k_abs:] += (1.0 - h[:n - k_abs])
            # derivative -h[i + k_abs] for i in 0..n - k_abs -1
            grad[:n - k_abs] += -h[k_abs:]
    # average if multiple indices
    if idxs.size > 1:
        grad /= idxs.size
    return grad

def _solve_h(start_h, max_iter=400):
    n = start_h.size
    # enforce feasibility
    h0 = _project_to_simplex_box(start_h, n/2.0)
    # initial t
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    # constraint functions need closure to n? Use lambda with default n.
    # But can define inner functions referencing n.
    # For speed, we can compute corr using np.correlate each call.

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n/2.0

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n/2.0)
    return h_opt, _c5_from_h(h_opt)

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {x in [0,1]^n : sum(x)=target} using binary search for the Lagrange multiplier."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] using correlation."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _grad_max_corr(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal correlation w.r.t. h (ignoring scaling factor)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    idxs = np.where(corr == maxc)[0]
    grad = np.zeros(n)
    for idx in idxs:
        k = idx - (n - 1)  # shift offset: positive => forward
        if k >= 0:
            # i in [0, n-k-1]
            grad[: n - k] += 1.0 - h[k:]
            grad[k:] += -h[: n - k]
        else:
            ka = -k
            grad[ka:] += 1.0 - h[: n - ka]
            grad[: n - ka] += -h[ka:]
    if idxs.size > 1:
        grad /= idxs.size
    return grad


def _gradient_descent(h: np.ndarray, time_limit: float):
    """Projected gradient descent on the maximal correlation."""
    start = time.time()
    n = h.size
    cur_h = _project_to_simplex_box(h, n / 2.0)
    cur_c5 = _c5_from_h(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5
    step = 0.2
    while time.time() - start < time_limit:
        grad = _grad_max_corr(cur_h)
        new_h = cur_h - step * grad
        new_h = _project_to_simplex_box(new_h, n / 2.0)
        new_c5 = _c5_from_h(new_h)
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-6)
    return best_h, best_c5


def _pairwise_search(h: np.ndarray, time_limit: float):
    """Simulated‑annealing style pairwise mass transfers."""
    cur_h = h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5
    n = cur_h.size
    start = time.time()
    # temperature schedule parameters
    T0, T_end = 5e-3, 1e-7
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_limit) + T_end
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand()
        # propose move
        cur_h[i] -= delta
        cur_h[j] += delta
        new_c5 = _c5_from_h(cur_h)
        d = new_c5 - cur_c5
        if d < 0.0 or np.random.rand() < np.exp(-d / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_c5


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(start_h: np.ndarray, max_iter: int = 500):
    """Polish a feasible h with SLSQP (auxiliary variable t)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    def ineq(x, n=n):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x, n=n):
        h = x[:n]
        return np.sum(h) - n / 2.0

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)
