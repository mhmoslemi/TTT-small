def _simulated_annealing(h, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_c = best_c

    start = time.time()
    # schedule parameters
    init_T = 0.02 # maybe larger initially? Given objective values around 0.38, difference of 0.001; set T ~ 0.001 initially? For acceptance of small increase
    # Let's make init_T relative to best_c: maybe 0.001.
    init_T = 0.001
    final_T = 1e-6

    # At each iteration compute T = init_T * (1 - frac) + final_T * frac
    while True:
        now = time.time()
        if now >= time_limit:
            break
        elapsed = now - start
        total = time_limit - start
        frac = elapsed / total
        T = init_T * (1 - frac) + final_T * frac

        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_down = cur_h[src]
        max_up = 1 - cur_h[dst]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.5
        new_h = cur_h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        new_c = _c5_from_h(new_h)
        d = new_c - cur_c
        if d < 0 or np.random.rand() < np.exp(-d / T):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_h = new_h
                best_c = new_c
    return best_h, best_c
```

def _hill_climb(h, max_steps, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    start = time.time()
    steps = 0
    while steps < max_steps and time.time() < time_limit:
        i, j = np.random.randint(n, size=2)
        # small epsilon
        max_down = best_h[i]
        max_up = 1 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.1
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = _c5_from_h(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c
        steps += 1
    return best_h, best_c
```

def _simulated_annealing(h0, time_limit):
    n = h0.size
    cur = h0.copy()
    cur_c = _c5_from_h(cur)
    best_h = cur.copy()
    best_c = cur_c
    start = time.time()
    while time.time() < time_limit:
        # compute fraction of elapsed time relative to limit
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)
        T = 0.001 * (1 - frac) + 1e-7 * frac
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # randomly decide direction
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_down = cur[src]
        max_up = 1.0 - cur[dst]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.5
        new = cur.copy()
        new[src] -= eps
        new[dst] += eps
        new_c = _c5_from_h(new)
        d = new_c - cur_c
        if d < 0 or np.random.rand() < np.exp(-d / T):
            cur = new
            cur_c = new_c
            if new_c < best_c:
                best_h = new.copy()
                best_c = new_c
    return best_h, best_c

import time
import numpy as np

# -------------------------------------------------
# Helper utilities
# -------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction for tiny residuals
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


# -------------------------------------------------
# Greedy reduction targeting the active shift
# -------------------------------------------------
def _greedy_reduce(h0: np.ndarray, max_steps: int, time_limit: float):
    n = h0.size
    h = h0.copy()
    best_c = _c5_from_h(h)
    no_improve = 0
    step = 0
    while step < max_steps and time.time() < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        max_idxs = np.where(corr == max_corr)[0]
        k = np.random.choice(max_idxs) - (n - 1)  # shift value
        # gradient = 1 - (h[i+k] + h[i-k]) (missing terms treated as 0)
        term = np.zeros(n)
        if k >= 0:
            term[: n - k] += h[k:]
            term[k:] += h[: n - k]
        else:
            kp = -k
            term[kp:] += h[: n - kp]
            term[: n - kp] += h[kp:]
        grad = 1.0 - term

        i_up = int(np.argmin(grad))    # most negative → increase
        i_down = int(np.argmax(grad))  # most positive → decrease
        if i_up == i_down:
            break
        # only move if genuine descent direction exists
        if grad[i_up] >= -1e-12 or grad[i_down] <= 1e-12:
            break

        max_up = 1.0 - h[i_up]
        max_down = h[i_down]
        max_eps = min(max_up, max_down)
        if max_eps <= 0:
            break
        eps = max_eps * 0.5
        h[i_up] += eps
        h[i_down] -= eps

        new_c = _c5_from_h(h)
        if new_c < best_c - 1e-12:
            best_c = new_c
            no_improve = 0
        else:
            # revert – move was not improving
            h[i_up] -= eps
            h[i_down] += eps
            no_improve += 1
            if no_improve > 500:
                break
        step += 1
    return h, best_c


# -------------------------------------------------
# Simulated annealing (pairwise mass exchange)
# -------------------------------------------------
def _simulated_annealing(h_start: np.ndarray, time_limit: float):
    n = h_start.size
    cur = h_start.copy()
    cur_c = _c5_from_h(cur)
    best_h = cur.copy()
    best_c = cur_c
    start = time.time()
    while time.time() < time_limit:
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)
        T = 0.001 * (1 - frac) + 1e-7 * frac
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random direction
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_down = cur[src]
        max_up = 1.0 - cur[dst]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.5
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand_c = _c5_from_h(cand)
        d = cand_c - cur_c
        if d < 0 or np.random.rand() < np.exp(-d / T):
            cur, cur_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
    return best_h, best_c

# Simple hill climbing (small moves)
# -------------------------------------------------
def _hill_climb(h_start: np.ndarray, max_steps: int, time_limit: float):
    n = h_start.size
    best = h_start.copy()
    best_c = _c5_from_h(best)
    step = 0
    while step < max_steps and time.time() < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best[i]
        max_up = 1.0 - best[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.1
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = _c5_from_h(cand)
        if cand_c < best_c:
            best, best_c = cand, cand_c
        step += 1
    return best, best_c
