def binary_hill_climb(h0, time_limit):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    max_corr = max_corr_fft(h)
    best_corr = max_corr
    start_time = time.time()
    # parameters for annealing
    T0 = 1.0  # temperature
    # For each iteration:
    it = 0
    while time.time() - start_time < time_limit:
        # sample two indices with different values
        i = np.random.randint(n)
        # ensure opposite value: maybe loop until h[i] != h[j]
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_corr = max_corr_fft(h)
        # Accept if improvement or with prob
        if new_corr < max_corr:
            max_corr = new_corr
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = h.copy()
        else:
            # acceptance probability based on temperature schedule
            # degrade temperature linearly with iteration count
            # Let's compute temperature as:
            T = max(0.001, T0 * (1 - it / max_iters))
            # Accept with probability exp(-(new_corr - max_corr)/T)
            if np.random.rand() < np.exp(-(new_corr - max_corr) / T):
                max_corr = new_corr
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
        it += 1
        # optional change of T0 gradually
    return best_h, best_corr

def max_corr_fft(h):
    n = h.size
    L = 2*n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real
    # The valid part is first 2*n-1 entries (0..2n-2). The max may be at these.
    return np.max(corr[:2*n-1])

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Basic feasibility helpers
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling loop."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible (binary or fractional) step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# --------------------------------------------------------------
# SLSQP refinement (same structure as the previous solver)
# --------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """Run a single SLSQP optimisation starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


# --------------------------------------------------------------
# FFT based correlation (fast max‑correlation for binary vectors)
# --------------------------------------------------------------
def _max_corr_fft(h: np.ndarray) -> float:
    """Maximum of the linear correlation h * (1-h) using FFT."""
    n = h.size
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real
    return float(np.max(corr[: 2 * n - 1]))

# Helpers for binary vectors
# --------------------------------------------------------------
def _binarize_and_balance(v: np.ndarray) -> np.ndarray:
    """Threshold at 0.5 and adjust to exactly n/2 ones."""
    n = v.shape[0]
    h = (v >= 0.5).astype(float)
    cur = int(h.sum())
    target = n // 2
    if cur > target:
        idx = np.where(h == 1)[0]
        flip = np.random.choice(idx, size=cur - target, replace=False)
        h[flip] = 0.0
    elif cur < target:
        idx = np.where(h == 0)[0]
        flip = np.random.choice(idx, size=target - cur, replace=False)
        h[flip] = 1.0
    return h


def _binary_hill_climb_with_restarts(h_start: np.ndarray, time_budget: float) -> tuple:
    """Greedy swap‑based hill climbing with random restarts."""
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_corr = _max_corr_fft(best_h)
    t0 = time.time()
    while time.time() - t0 < time_budget:
        # start from the current best but perturb a few bits
        cur_h = best_h.copy()
        # small random perturbation
        swaps = max(1, n // 50)
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if ones.size and zeros.size:
            for _ in range(swaps):
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                cur_h[i], cur_h[j] = 0.0, 1.0
                # keep index sets updated
                ones = np.where(cur_h == 1)[0]
                zeros = np.where(cur_h == 0)[0]

        cur_corr = _max_corr_fft(cur_h)
        stagn = 0
        max_stagn = 2000
        # greedy local optimisation
        while stagn < max_stagn and time.time() - t0 < time_budget:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if cur_h[i] == cur_h[j]:
                continue
            # perform swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_corr = _max_corr_fft(cur_h)
            if new_corr < cur_corr:
                cur_corr = new_corr
                stagn = 0
                if cur_corr < best_corr:
                    best_corr = cur_corr
                    best_h = cur_h.copy()
                    # early exit if target already reached

            else:
                # revert
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
                stagn += 1
    return best_h, best_corr

    start_total = time.time()
    time_left = lambda: budget_s - (time.time() - start_total)

    try:

    # Candidate resolutions (we focus on 256 and 512 – still small enough)
    # ------------------------------------------------------------------
    candidate_ns = [256, 512]

    for n in candidate_ns:

        # Build an initial binary vector for this resolution
        # --------------------------------------------------------------
        if init_h is not None and init_h.size == n:

            x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
            x_fine = (np.arange(n) + 0.5) * (2.0 / n)
            interp = np.interp(x_fine, x_coarse, init_h)
            h0 = _binarize_and_balance(interp)
        else:

            h0[: n // 2] = 1.0
            np.random.shuffle(h0)

        # Binary hill climbing (major share of the budget)
        # --------------------------------------------------------------
        alloc = max(5.0, 0.45 * time_left())
        h_bin, max_corr = _binary_hill_climb_with_restarts(h0, alloc)

        c5_val = max_corr * (2.0 / n)

        if c5_val < best_c5:
            best_h, best_c5, best_n = h_bin.copy(), c5_val, n

        # --------------------------------------------------------------

        if best_c5 < 0.3808:

    # Final safety check – ensure feasibility and recompute exact bound
    # ------------------------------------------------------------------

        best_h[: n // 2] = 1.0
        np.random.shuffle(best_h)
        best_n = n

    best_c5 = _c5_from_h(best_h)
