def build_h_from_intervals(a, b, n):
    # a,b arrays of lengths m each, sum to 1 each
    m = a.size
    seg_lengths = np.empty(2*m)
    seg_lengths[0::2] = a
    seg_lengths[1::2] = b
    # Convert lengths to counts of points (n points total)
    raw_counts = seg_lengths * n / 2.0  # because total interval length 2
    floor_counts = np.floor(raw_counts).astype(int)
    remainder = n - floor_counts.sum()
    # fractional parts
    fractional = raw_counts - floor_counts
    if remainder > 0:
        idx = np.argsort(-fractional)[:remainder]
        floor_counts[idx] += 1
    elif remainder < 0:
        idx = np.argsort(fractional)[: -remainder]
        floor_counts[idx] -= 1
    # Now assign values
    h = np.empty(n, dtype=float)
    pos = 0
    for i, cnt in enumerate(floor_counts):
        if cnt > 0:
            val = 1.0 if (i % 2 == 0) else 0.0  # even idx: a (1), odd: b (0)
            h[pos:pos+cnt] = val
            pos += cnt
    # If any leftover (due to rounding errors), fill zero
    if pos < n:
        h[pos:] = 0.0
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global parameters for the smooth surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp in a numerically stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- feasibility utilities ----------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Return a random h with 0≤h≤1 and Σh = n/2."""
    h = np.full(n, 0.5)
    # a simple mixing loop
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Simple hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the log‑sum‑exp surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction of the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- objective ----------
def compute_c5(h):
    """Exact discrete C₅ for step function h on a uniform grid."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ---------- block‑model utilities ----------
def build_h_from_intervals(a, b, n):
    """Construct binary h from lengths a (value 1) and b (value 0)."""
    m = a.size
    seg = np.empty(2 * m)
    seg[0::2] = a
    seg[1::2] = b
    # expected number of samples for each segment (total length 2)
    raw = seg * n / 2.0
    cnt = np.floor(raw).astype(int)
    rem = n - cnt.sum()
    frac = raw - cnt
    if rem > 0:
        idx = np.argsort(-frac)[:rem]
        cnt[idx] += 1
    elif rem < 0:
        idx = np.argsort(frac)[: -rem]
        cnt[idx] -= 1
    # fill h
    h = np.empty(n, dtype=float)
    pos = 0
    for i, c in enumerate(cnt):
        if c <= 0:
            continue
        val = 1.0 if (i % 2 == 0) else 0.0
        h[pos : pos + c] = val
        pos += c
    if pos < n:
        h[pos:] = 0.0
    return h


def random_block_solution(m, n, rng, iterations=1200):
    """Search over alternating 0/1 blocks (m blocks of each type)."""
    best_c = np.inf
    best_h = None
    for _ in range(iterations):
        a_raw = rng.random(m)
        a = a_raw / a_raw.sum()
        b_raw = rng.random(m)
        b = b_raw / b_raw.sum()
        h = build_h_from_intervals(a, b, n)
        h = enforce_sum(h, n / 2.0, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


# ---------- global perturbation ----------
def perturb_global(h, rng, sigma=0.12):
    """Gaussian perturbation with zero‑sum, followed by clipping + sum repair."""
    n = h.size
    delta = rng.normal(scale=sigma, size=n)
    delta -= delta.mean()
    h_new = h + delta
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, n / 2.0, rng)
    return h_new


# ---------- simulated annealing ----------
def simulated_annealing(h0, rng, time_limit, max_iter=2500):
    start = time.time()
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h, best_c = h.copy(), c_cur
    T0 = 0.018
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        T = T0 * (1 - it / max_iter)
        h_new = perturb_global(h, rng, sigma=0.14)
        c_new = compute_c5(h_new)
        diff = c_new - c_cur
        if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            h, c_cur = h_new, c_new
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    return best_h, best_c


# ---------- main optimisation driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    for m in (2, 3, 4, 5):

    # ----- refine the current best with hill‑climb + SLSQP -----
    def refine(candidate):
        h_opt, c_opt = hill_climb(candidate, rng, max_iters=1500, start_step=0.5)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0):

                # immediate update of the outer best
                nonlocal global_best_h, global_best_c
                global_best_h, global_best_c = h_opt.copy(), c_opt
        return h_opt, c_opt

    # first refinement of the best block candidate
    _, _ = refine(global_best_h)

    while time.time() - start_time < budget_s * 0.85:
        # pick a starter
