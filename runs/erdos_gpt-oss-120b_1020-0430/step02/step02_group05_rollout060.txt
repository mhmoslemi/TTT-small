def _c5_and_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()

    # shift offset: shift_idx - (len(h) - 1)
    shift = shift_idx - (len(h) - 1)  # can be negative
    return max_val * (2.0 / len(h)), shift, corr

def _solve_h(start_h, max_iter_slsqp=800, pgd_iters=400):
    n = start_h.size
    h = _project_to_simplex_box(start_h, n/2.0)
    # SLSQP
    corr = np.correlate(h, 1 - h, mode='full')
    t0 = np.max(corr)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    constraints = [{'type': 'ineq', 'fun': _ineq_constraint},
                   {'type': 'eq', 'fun': _eq_constraint}]
    opts = {'maxiter': max_iter_slsqp, 'ftol': 1e-9, 'disp': False}
    res = optimize.minimize(_objective, x0, method='SLSQP', bounds=bounds,
                            constraints=constraints, options=opts)
    if res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h
    h_opt = _project_to_simplex_box(h_opt, n/2.0)
    # PGD refinement
    h_refined, c5_val = _pgd_refine(h_opt, max_iter=pgd_iters, step0=0.5, decay=0.98)
    return h_refined, c5_val

import time
import numpy as np
from scipy import optimize

# ---------- projection onto simplex + box ----------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# ---------- correlation / C5 ----------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _c5_and_shift(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (n - 1)          # integer lag
    return max_val * (2.0 / n), shift, corr


# ---------- gradient of the active shift ----------
def _gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    n = h.size
    g = np.zeros_like(h)
    if s >= 0:
        g[: n - s] += 1.0 - h[s:]         # ∂/∂h_i of h_i part
        g[s:] -= h[: n - s]               # ∂/∂h_i of -h_i·h_{i-s}
    else:
        t = -s
        g[t:] += 1.0                      # +1 term for i >= t
        g[: n - t] -= h[t:]               # -h_{i+t}
        g[t:] -= h[: n - t]               # -h_{i-t}
    return g


# ---------- projected gradient descent for fine tuning ----------
def _pgd_refine(h: np.ndarray,
                max_iter: int = 300,
                step0: float = 0.5,
                decay: float = 0.98):
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    step = step0
    for _ in range(max_iter):
        cval, shift, _ = _c5_and_shift(h)
        g = _gradient_shift(h, shift)
        # backtracking line search
        cur_step = step
        improved = False
        while cur_step > 1e-6:
            h_new = _project_to_simplex_box(h - cur_step * g, n / 2.0)
            new_c5 = _c5_from_h(h_new)
            if new_c5 < best_c5 - 1e-12:
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
                improved = True
                break
            cur_step *= 0.5
        step *= decay
        if not improved:
            # if we couldn't improve, keep stepping down a bit
            step = max(step, 1e-4)
    return best_h, best_c5


# ---------- SLSQP wrapper ----------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(start_h: np.ndarray,
             max_iter_slsqp: int = 600,
             pgd_iters: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter_slsqp, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(_objective, x0, method="SLSQP",
                                bounds=bounds, constraints=cons, options=opts)
    except Exception:
        res = None
    if res is not None and res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    h_opt, c5_val = _pgd_refine(h_opt, max_iter=pgd_iters, step0=0.5, decay=0.98)
    return h_opt, c5_val


# ---------- auxiliary constructors ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _random_feasible(n: int) -> np.ndarray:
    v = np.random.rand(n)
    return _project_to_simplex_box(v, n / 2.0)
