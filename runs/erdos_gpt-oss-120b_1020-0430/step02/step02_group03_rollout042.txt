def compute_c5(v):
    h = np.concatenate([v, 1 - v[::-1]])
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)
```

def slsqp_optimize_v(v0, beta, max_iter=200):
    n = v0.shape[0]
    def obj(v):
        h = np.concatenate([v, 1 - v[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0,1.0)] * n
    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    return np.clip(res.x, 0.0, 1.0)
```

import time, numpy as np, scipy.optimize as opt

def v_to_h(v):
    return np.concatenate([v, 1 - v[::-1]])

def compute_c5(v):
    h = v_to_h(v)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def softmax_obj(v, beta):
    h = v_to_h(v)
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize_v(v0, beta, max_iter=300):
    n = v0.size
    bounds = [(0.0, 1.0)] * n
    def obj(v):
        h = np.concatenate([v, 1 - v[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = opt.minimize(obj, v0, method='SLSQP', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    return np.clip(res.x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def _v_to_h(v: np.ndarray) -> np.ndarray:
    """Map half‑vector to full symmetric step function."""
    return np.concatenate([v, 1.0 - v[::-1]])


def _c5_from_v(v: np.ndarray) -> float:
    """Exact discrete C₅ for a half‑vector v."""
    h = _v_to_h(v)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _softmax_obj(v: np.ndarray, beta: float) -> float:
    """Smooth log‑sum‑exp surrogate of maxₖ Σᵢ v_i(1‑v_{i+k})."""
    h = _v_to_h(v)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def _slsqp_optimize(v0: np.ndarray, beta: float, max_iter: int = 400) -> np.ndarray:
    """Refine v with SLSQP on the smooth surrogate."""
    n = v0.size
    bounds = [(0.0, 1.0)] * n

    def obj(v):
        h = np.concatenate([v, 1.0 - v[::-1]])
        corr = np.correlate(h, 1.0 - h, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(
        obj,
        v0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    start_time = time.time()
