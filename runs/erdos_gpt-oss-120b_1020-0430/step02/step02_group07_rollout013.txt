def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=10000):
    # adjust to sum target, keep 0<=h<=1
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add to zeros
            zero_idxs = np.where(h < 1 - 1e-12)[0]
            i = rng.choice(zero_idxs)

def simulated_annealing(start_h, rng, time_limit):

    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # temperature schedule
    temp = 0.02
    decay = 0.9995
    start_time = time.time()
    # maintain indices of ones and zeros for fast pick
    while time.time() - start_time < time_limit:
        # choose random positions of opposite values
        i = rng.integers(0, n)
        # find opposite
        # loop until distinct value found
        if cur_h[i] > 0.5:
            # need j with 0
            zeros = np.where(cur_h < 0.5)[0]
            if zeros.size == 0:
                continue
            j = rng.choice(zeros)
        else:
            ones = np.where(cur_h > 0.5)[0]
            if ones.size == 0:
                continue
            j = rng.choice(ones)
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_h, best_c = new_h.copy(), new_c
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            cur_h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c

    elif cur_ones < target_int:

        turn_on = rng.choice(idx0, size=target_int - cur_ones, replace=False)
        binary_best_h[turn_on] = 1.0

    # allocate time
    sa_budget = budget_s * 0.55
    refine_budget = budget_s * 0.35
    final_budget = budget_s * 0.10

    # simulated annealing with several restarts
    while time.time() - start_time < sa_budget:
        # 50 % chance to start from current best, else random
        if rng.random() < 0.5:
            h0 = binary_best_h.copy()
        else:
            h0 = binary_random_feasible(n, rng)
        # run a short anneal (max 5 s or remaining budget)
        remaining = start_time + sa_budget - time.time()
        run_time = min(5.0, remaining)
        if run_time <= 0:
            break
        cand_h, cand_c = simulated_annealing_binary(h0, rng, run_time)
        if cand_c < cont_best_c:
            cont_best_h, cont_best_c = cand_h.copy(), cand_c
        if cand_c < compute_c5(binary_best_h):
            binary_best_h = cand_h.copy()

    # ------------------------------------------------------------------
    # Continuous refinement of the best binary solution
    # ------------------------------------------------------------------
    refine_start = time.time()
    # treat binary solution as float and polish
    refined_h, refined_c = continuous_refine(binary_best_h, rng)
    if refined_c < cont_best_c:
        cont_best_h, cont_best_c = refined_h, refined_c
    # extra deterministic hill‑climb on the refined continuous vector
    if time.time() - start_time < start_time + refine_budget:
        hc_h, hc_c = hill_climb_continuous(cont_best_h, rng, max_iters=3000, start_step=0.3)
        if hc_c < cont_best_c:
            cont_best_h, cont_best_c = hc_h, hc_c

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    cont_best_h = np.clip(cont_best_h, 0.0, 1.0)
    enforce_sum(cont_best_h, n / 2.0, rng)
    final_c = compute_c5(cont_best_h)

    return cont_best_h.tolist(), float(final_c), int(n)
