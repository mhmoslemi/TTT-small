def _gd_optimize(h, max_iters=2000, time_budget=...):
    start = time.time()
    N = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        cmax = corr[idx] * (2.0 / N)
        if cmax < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = cmax
        lag = idx - (N - 1)
        # compute gradient
        grad = np.empty(N)
        if lag >= 0:
            # term1
            grad[:N - lag] = (1 - h[lag:])
            grad[N - lag:] = 0.0  # beyond range
            # term2
            grad[lag:] -= h[:N - lag]
        else:
            lag_abs = -lag
            grad[lag_abs:] = (1 - h[:N - lag_abs])
            grad[:lag_abs] = 0.0
            grad[:N - lag_abs] -= h[lag_abs:]

        # Determine optimal step size via simple backtracking line search
        alpha = 0.5
        # line search decreasing alpha if not improving
        for _ in range(10):
            h_new = _project_to_feasible(h - alpha * grad, N)
            new_cmax = _c5_from_h(h_new)
            if new_cmax <= cmax + 1e-12:  # not increase
                break
            alpha *= 0.5
        else:
            # no improvement found, break
            break
        h = h_new
    return best_h, best_c5

def _gradient_max_corr(h):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (N - 1)
    # compute gradient as described.
    grad = np.zeros(N)
    if lag >= 0:
        # term1: (1 - h[i+lag]) for i=0..N-1-lag
        grad[:N - lag] = (1 - h[lag:])

# ----- feasibility projection -------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # small correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h: np.ndarray) -> float:

    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
