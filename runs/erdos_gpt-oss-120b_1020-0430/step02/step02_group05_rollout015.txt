def _grad_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    # find shifts achieving max (tolerance)
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    grads = np.zeros(n)
    for idx in idxs:
        offset = idx - (n - 1)  # shift s
        # compute gradient for this offset
        # vectorized: compute contributions for each j
        # g_j = 1 - (h_{j+offset} if in range else 0) - (h_{j-offset} if in range else 0)
        # if offset >= 0:
        #   h_{j+offset} defined for j <= n-1-offset
        #   So we can compute a shifted array padded appropriately.
        # To avoid loops, we can compute using slices.
        # Let's implement using array of ones then subtract contributions.
        g = np.ones(n)
        if offset >=0:
            # j+offset within range for j in [0, n-1-offset]
            g[:n-offset] -= h[offset:]
            # j-offset within range for j in [offset, n-1]
            g[offset:] -= h[:n-offset]
        else:
            # offset negative: offset = -d
            d = -offset
            # j+offset = j-d, defined for j>=d
            g[d:] -= h[:n-d]
            # j-offset = j+d, defined for j< n-d
            g[:n-d] -= h[d:]
        grads += g
    grads /= len(idxs)  # average
    return grads

When offset<0 i.e., d>0: offset = -d, j+offset = j-d defined for j>=d, subtract h[0:n-d] from g[d:]. j-offset = j+d defined for j <= n-1-d (i.e., j< n-d), we subtract h[d:] from g[0:n-d].

Define projection:

def _project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    # t - corr >= 0
    return t - corr

def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x):
    return x[-1]

def _solve_h(start_h, max_iter=300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

def _tent_h(n):

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    # positions of centers
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    h_fine = np.interp(x_fine, x_coarse, coarse_h)
    # project to satisfy constraints
    return _project_to_simplex_box(h_fine, n_fine / 2.0)

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _grad_max_corr(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    eps = 1e-12
    idxs = np.where(corr >= max_val - eps)[0]
    grads = np.zeros(n)
    for idx in idxs:
        offset = idx - (n - 1)
        g = np.ones(n)
        if offset >= 0:
            if offset < n:
                g[: n - offset] -= h[offset:]
                g[offset:] -= h[: n - offset]
        else:
            d = -offset
            if d < n:
                g[d:] -= h[: n - d]
                g[: n - d] -= h[d:]
        grads += g
    grads /= len(idxs)
    return grads, max_val


def _pgd_optimize(
    h0: np.ndarray, max_iter: int = 2000, step0: float = 0.5, time_deadline: float = None
):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    cur_c5 = _c5_from_h(h)
    step = step0
    for _ in range(max_iter):
        if time_deadline is not None and time.time() > time_deadline:
            break
        grad, _ = _grad_max_corr(h)
        improved = False
        for _ in range(5):
            h_new = _project_to_simplex_box(h - step * grad, target)
            new_c5 = _c5_from_h(h_new)
            if new_c5 < cur_c5 - 1e-12:
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
            continue
        h = h_new
        cur_c5 = new_c5
        step *= 1.05
    return h, cur_c5


def _ineq_constraint(x: np.ndarray):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray):
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=constraints, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    h_fine = np.interp(x_fine, x_coarse, coarse_h)
    return _project_to_simplex_box(h_fine, n_fine / 2.0)


def _pair_perturb_search(
    h: np.ndarray, cur_c5: float, steps: int = 500, time_deadline: float = None
):
    n = h.size
    h_cur = h.copy()
    for _ in range(steps):
        if time_deadline is not None and time.time() > time_deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_delta = min(1.0 - h_cur[i], h_cur[j])
        min_delta = -min(h_cur[i], 1.0 - h_cur[j])
        if max_delta <= min_delta:

        delta = (np.random.rand() * 2 - 1) * 0.05
        delta = np.clip(delta, min_delta, max_delta)
        if abs(delta) < 1e-12:

        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = _c5_from_h(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h_cur = h_new
            cur_c5 = new_c5
    return h_cur, cur_c5
