def simulated_annealing(n, half, budget_s, start_time, rng):
    # n is even
    full = expand(half)
    best_full = full.copy()
    best_c = compute_c5(full)
    cur_half = half.copy()
    cur_full = full
    cur_c = best_c
    # schedule parameters
    max_iter = int(1e6) maybe limited by time
    T0 = 0.005
    Tf = 1e-6
    total_time_allowed = budget_s * 0.8 (stop earlier)
    next_report maybe not needed.

    while time.time() - start_time < total_time_allowed:
        # compute fraction of total iterations based on elapsed time
        elapsed = time.time() - start_time
        # compute temperature schedule linear in time
        T = T0 * (1 - elapsed / total_time_allowed) + Tf
        # propose random index
        i = rng.integers(0, half.shape[0])
        # propose new value
        # epsilon from normal scale maybe 0.2 * T? Could use scale based on T.
        # Use proposal: new = cur_half[i] + rng.normal(scale=0.1) (or based on T)
        # Clip to [0,1]
        new_val = cur_half[i] + rng.normal(scale=0.02) # stable
        new_val = min(1, max(0, new_val))
        if new_val == cur_half[i]:
            continue
        # update half copy for candidate
        cand_half = cur_half.copy()
        cand_half[i] = new_val
        cand_full = expand(cand_half)
        cand_c = compute_c5(cand_full)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            cur_half = cand_half
            cur_full = cand_full
            cur_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_full = cand_full.copy()
        # occasional larger move: with small prob, perturb random index by larger amount, or swap two half values.
        if rng.random() < 0.001:
            i2 = rng.integers(0, half.shape[0])
            # swap values
            cand_half = cur_half.copy()
            cand_half[i], cand_half[i2] = cand_half[i2], cand_half[i]
            cand_full = expand(cand_half)
            cand_c = compute_c5(cand_full)
            delta = cand_c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_half = cand_half
                cur_full = cand_full
                cur_c = cand_c
                if cand_c < best_c:
                    best_c = cand_c
                    best_full = cand_full.copy()
    return best_full, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global helpers ----------
CURRENT_BETA = None  # used by the softmax objective (half‑vector version)


def expand_half(half):
    """Build the full symmetric step function from its first half."""
    m = half.shape[0]
    h = np.empty(m * 2, dtype=np.float64)
    h[:m] = half
    h[m:] = 1.0 - half[::-1]
    return h


def compute_c5(h):
    """Exact discrete C5 for a full vector h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def softmax_obj_half(half):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i h_i(1‑h_{i+k}) for a half‑vector."""
    beta = CURRENT_BETA
    h = expand_half(half)
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize_half(start_half, beta, max_iter=200):
    """Run SLSQP on the smooth surrogate for a given β (full symmetry enforced)."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    n = start_half.size
    bounds = [(0.0, 1.0)] * n
    # No equality constraints – symmetry guarantees the integral condition.
    res = minimize(
        softmax_obj_half,
        start_half,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return x


def hill_climb_half(start_half, rng, max_iters=4000, step_scale=0.05):
    """Deterministic hill‑climb (accept only improvements) on the half‑vector."""
    cur = start_half.copy()
    cur_c = compute_c5(expand_half(cur))
    best = cur.copy()
    best_c = cur_c
    n = cur.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        eps = (rng.random() - 0.5) * 2.0 * step_scale
        new_val = cur[i] + eps
        if new_val < 0.0:
            new_val = 0.0
        elif new_val > 1.0:
            new_val = 1.0
        if new_val == cur[i]:
            continue
        cand = cur.copy()
        cand[i] = new_val
        cand_c = compute_c5(expand_half(cand))
        if cand_c < cur_c:
            cur = cand
            cur_c = cand_c
            if cand_c < best_c:
                best = cand.copy()
                best_c = cand_c
    return best, best_c


def simulated_annealing(start_half, time_budget, rng):
    """Simple SA on the half‑vector; respects bounds and symmetry automatically."""
    sa_start = time.time()
    deadline = sa_start + time_budget
    cur = start_half.copy()
    cur_c = compute_c5(expand_half(cur))
    best = cur.copy()
    best_c = cur_c
    n = cur.size

    sigma_start = 0.07
    sigma_end = 0.0005
    T0 = 0.02
    Tf = 1e-6

    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - sa_start
        frac = elapsed / time_budget
        # temperature (linear decay)
        T = T0 * (1.0 - frac) + Tf * frac
        # proposal scale (linear decay)
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac

        # propose change of a single coordinate
        i = rng.integers(0, n)
        new_val = cur[i] + rng.normal(scale=sigma)
        if new_val < 0.0:
            new_val = 0.0
        elif new_val > 1.0:
            new_val = 1.0
        if new_val == cur[i]:
            continue
        cand = cur.copy()
        cand[i] = new_val
        cand_c = compute_c5(expand_half(cand))
        d = cand_c - cur_c
        if d < 0.0 or (T > 0.0 and rng.random() < np.exp(-d / T)):
            cur = cand
            cur_c = cand_c
            if cand_c < best_c:
                best = cand.copy()
                best_c = cand_c

        # occasional swap of two entries (helps escape plateaus)
        if rng.random() < 0.001:
            i2 = rng.integers(0, n)
            cand = cur.copy()
            cand[i], cand[i2] = cand[i2], cand[i]
            cand_c = compute_c5(expand_half(cand))
            d = cand_c - cur_c
            if d < 0.0 or (T > 0.0 and rng.random() < np.exp(-d / T)):
                cur = cand
                cur_c = cand_c
                if cand_c < best_c:
                    best = cand.copy()
                    best_c = cand_c
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a symmetric step function for the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- 1. Determine discretisation size and initialise half‑vector -----
    if "initial_h_values" in globals():
        full0 = np.array(initial_h_values, dtype=np.float64)
        n = full0.size
        # force even length (the constant prefers an even number)
        if n % 2 == 1:
            n -= 1
            full0 = full0[:n]
        half_n = n // 2
        # symmetrise: average the pair (h_i, 1‑h_{n‑1‑i})
        half0 = (full0[:half_n] + (1.0 - full0[::-1][:half_n])) * 0.5
    else:
        n = 200                     # a modest even size if no starter is given
        half_n = n // 2
        half0 = rng.random(half_n)

    half0 = np.clip(half0, 0.0, 1.0)

    # ----- 2. Keep the best encountered solution -----
    best_half = half0.copy()
    best_c = compute_c5(expand_half(best_half))

    # time budgeting
    slsqp_time = budget_s * 0.30
    sa_time = budget_s * 0.60
    final_hc_time = budget_s * 0.09   # the remaining margin is kept as safety

    # ----- 3. Multi‑start SLSQP (coarse → fine) -----
    betas = [5.0, 50.0, 200.0, 800.0]
    while time.time() - start_time < slsqp_time:
        # random start blended occasionally with the current best
        start_half = rng.random(half_n)
        if rng.random() < 0.3:
            mask = rng.random(half_n) < 0.5
            start_half[mask] = best_half[mask]

        cur = start_half
        for beta in betas:
            if time.time() - start_time >= slsqp_time:
                break
            cur = slsqp_optimize_half(cur, beta, max_iter=250)
            cur_c = compute_c5(expand_half(cur))
            if cur_c < best_c:
                best_c = cur_c
                best_half = cur.copy()
        # allow the loop to generate a few more starts
        if time.time() - start_time >= slsqp_time:
            break

    # ----- 4. Simulated annealing on the half‑vector -----
    remaining_for_sa = max(0.0, start_time + sa_time - time.time() - final_hc_time)
    if remaining_for_sa > 5.0:                     # ensure a minimal budget
        sa_half, sa_c = simulated_annealing(best_half, remaining_for_sa, rng)
        if sa_c < best_c:
            best_c = sa_c
            best_half = sa_half.copy()

    # ----- 5. Final deterministic hill‑climb (fine tuning) -----
    hc_iters = 5000
    hc_half, hc_c = hill_climb_half(best_half, rng, max_iters=hc_iters, step_scale=0.04)
    if hc_c < best_c:
        best_c = hc_c
        best_half = hc_half.copy()

    # ----- 6. One last high‑β SLSQP refinement (sharp max) -----
    if time.time() - start_time < budget_s * 0.99:
        try:
            best_half = slsqp_optimize_half(best_half, beta=2000.0, max_iter=400)
            best_c = compute_c5(expand_half(best_half))
        except Exception:
            # fallback to current best if something goes wrong
            best_c = compute_c5(expand_half(best_half))

    # ----- 7. Assemble final full vector, enforce exact constraints and return -----
    final_h = expand_half(best_half)
    final_h = np.clip(final_h, 0.0, 1.0)

    # exact integral check (symmetry guarantees the sum, but a safety pass)
    target_sum = n / 2.0
    sum_h = final_h.sum()
    if abs(sum_h - target_sum) > 1e-12:
        diff = target_sum - sum_h
        for i in range(n):
            adj = min(diff, 1.0 - final_h[i]) if diff > 0 else max(diff, -final_h[i])
            final_h[i] += adj
            diff = target_sum - final_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(final_h)
    return final_h.tolist(), float(final_c), int(n)
