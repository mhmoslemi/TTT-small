def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final small correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _optimize_h(h0, time_budget):
    import time, numpy as np
    n = h0.size
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    cur_val = max_corr * (2.0 / n)
    best_h = h.copy()
    best_val = cur_val
    t_start = time.time()
    t_range = np.arange(-(n-1), n, dtype=int)  # shape 2n-1
    no_improve = 0
    while time.time() - t_start < time_budget:
        elapsed = time.time() - t_start
        # adaptive step scale
        step_scale = 0.02 * max(0.0, 1.0 - elapsed / time_budget) + 1e-4
        # try guided move
        t_center = np.argmax(corr) - (n - 1)
        idx = np.arange(n)
        idx_plus = idx + t_center
        idx_minus = idx - t_center
        term = np.zeros(n, dtype=float)
        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        term[mask_plus] = 1.0 - h[idx_plus[mask_plus]]
        mask_minus = (idx_minus >= 0) & (idx_minus < n)
        term[mask_minus] -= h[idx_minus[mask_minus]]
        pos = np.where(term > 1e-9)[0]
        neg = np.where(term < -1e-9)[0]
        moved = False
        if pos.size and neg.size:
            i = pos[np.argmax(term[pos])]
            j = neg[np.argmin(term[neg])]
            max_delta = min(step_scale, 1.0 - h[i], h[j])
            if max_delta > 1e-12:
                delta = max_delta
                delta_corr_i = _delta_corr_for_index(h, i, delta, t_range)
                delta_corr_j = _delta_corr_for_index(h, j, -delta, t_range)
                corr = corr + delta_corr_i + delta_corr_j
                h[i] += delta
                h[j] -= delta
                max_corr = corr.max()
                cur_val = max_corr * (2.0 / n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                moved = True
                no_improve = 0
        if moved:
            continue
        # random move
        i, j = np.random.choice(n, 2, replace=False)
        if np.random.rand() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_delta = min(step_scale, h[donor], 1.0 - h[receiver])
        if max_delta <= 1e-12:
            continue
        delta = np.random.rand() * max_delta
        delta_corr_donor = _delta_corr_for_index(h, donor, -delta, t_range)
        delta_corr_receiver = _delta_corr_for_index(h, receiver, delta, t_range)
        new_corr = corr + delta_corr_donor + delta_corr_receiver
        new_max = new_corr.max()
        new_val = new_max * (2.0 / n)
        # acceptance criteria
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # temperature schedule
            T = 0.01 * max(0.0, 1.0 - elapsed / time_budget)
            if T > 0 and np.exp(-(new_val - cur_val) / T) > np.random.rand():
                accept = True
        if accept:
            corr = new_corr
            max_corr = new_max
            cur_val = new_val
            h[donor] -= delta
            h[receiver] += delta
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # occasional random perturb when stuck
        if no_improve >= 2000:
            # perturb around best
            h = best_h + 0.02 * np.random.randn(n)
            h = _project_to_feasible(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = corr.max()
            cur_val = max_corr * (2.0 / n)
            no_improve = 0
    return best_h, best_val

import time

# ----------------------------------------------------------------------
# Feasibility projection (bounds [0,1] and sum = n/2)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int,
                         tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

    np.random.seed(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Initialise from provided construction if any
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    n = 256
    if init_h is not None:
        if init_h.size == n:
            h0 = init_h.copy()
        elif init_h.size < n:
            h0 = _upscale_h(init_h, n)
        else:
            h0 = _downscale_h(init_h, n)
    else:
        h0 = np.zeros(n, dtype=float)
        k = n // 2
        h0[:k] = 1.0
        np.random.shuffle(h0)

    h0 = _project_to_feasible(h0, n)

    # ------------------------------------------------------------------
    # Primary hill‑climbing phase (≈70% of the budget)
    # ------------------------------------------------------------------
    hc_budget = time_left() * 0.7
    best_h, best_val = _optimize_h(h0, hc_budget)

    # ------------------------------------------------------------------
    # Optional random restarts while time permits
    # ------------------------------------------------------------------
    while time_left() > 5.0:
        h_rand = np.random.rand(n)
        h_rand = _project_to_feasible(h_rand, n)
        sub_budget = min(time_left() * 0.2, 30.0)
        cand_h, cand_val = _optimize_h(h_rand, sub_budget)
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val

    # ------------------------------------------------------------------
    # Final polishing with SLSQP if we still have a few seconds left
    # ------------------------------------------------------------------
    if time_left() > 20.0:
        h_ref, val_ref = _solve_h(best_h, max_iter=800)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    # ------------------------------------------------------------------
    # Ensure feasibility and return result
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_val = _c5_from_h(best_h)
    return best_h.tolist(), float(best_val), int(best_h.size)
