from scipy import optimize

def _project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1.0 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _corr_full(h):
    n = h.shape[0]
    L = 2 * n
    a = np.fft.rfft(h, L)
    b = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(a * np.conj(b), L)
    return corr[:2 * n - 1]

def _c5_from_h(h):
    n = h.shape[0]
    max_corr = np.max(_corr_full(h))
    return max_corr * (2.0 / n)

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[n]
    corr = _corr_full(h)
    return t - corr

def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x):
    return x[-1]

def _solve_h(start_h, max_iter=600):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = _corr_full(h0)
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]

    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _tent_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

def _gradient_worst_shift(h):
    n = h.shape[0]
    corr = _corr_full(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)                 # shift value
    grad = np.zeros_like(h)

    if k >= 0:
        L = n - k
        i = np.arange(L)
        j = np.arange(k, k + L)
    else:
        L = n + k          # since k negative
        i = np.arange(-k, -k + L)
        j = np.arange(L)
    # contributions
    grad[i] += 1.0 - h[j]
    grad[j] -= h[i]
    return grad, k

def _refine_gradient(h, max_iter=200, step0=0.2):
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    step = step0
    for _ in range(max_iter):
        grad, _ = _gradient_worst_shift(best_h)
        if np.allclose(grad, 0):
            break
        new_h = best_h - step * grad
        new_h = _project_to_simplex_box(new_h, n / 2.0)
        new_val = _c5_from_h(new_h)
        if new_val < best_val:
            best_val = new_val
            best_h = new_h
            step *= 1.05
        else:
            step *= 0.7
        if step < 1e-6:
            break
    return best_h, best_val

        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
