def greedy_optimize(h, max_steps=5000, tol=1e-6):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    last_max = np.inf
    for step in range(max_steps):
        # compute corr (non-averaged)
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        O_max = corr[idx_max]
        if O_max < last_max - tol:
            last_max = O_max
        else:
            # no improvement after many attempts; break
            pass
        # if improvement small break
        if O_max <= 0:
            break
        k = idx_max - (n-1)
        # compute range based on k
        if k >= 0:
            i_start = 0
            i_end = n - k
        else:
            i_start = -k
            i_end = n
        idxs = np.arange(i_start, i_end)
        hi = h[idxs]
        hj = h[idxs + k]
        diff = hi - hj
        # consider pairs with diff > 0
        mask = (diff > 1e-12) & (hi > 1e-12) & (hj < 1 - 1e-12)
        if not np.any(mask):
            # can't improve this shift
            # break? continue to next iteration with random perturb.
            # We'll break out and do random perturb.
            break
        possible = np.minimum(hi[mask], 1 - hj[mask])
        impacts = diff[mask] * possible
        # choose best
        best_idx = np.argmax(impacts)
        i_sel = idxs[mask][best_idx]
        delta = possible[best_idx]  # full possible
        # apply delta
        h[i_sel] -= delta
        h[i_sel + k] += delta
        # continue
    return h

def gradient_transfer_step(h, grad, step_scale=1.0):
    # find indices with max and min grad
    i_pos = np.argmax(grad)   # largest gradient (positive)
    i_neg = np.argmin(grad)   # smallest gradient (most negative)
    g_pos = grad[i_pos]
    g_neg = grad[i_neg]
    if g_pos <= 0 or g_neg >= 0:
        # not possible to improve O_k by moving mass (grad all nonpositive or nonnegative)
        return False, h
    # transfer amount
    delta = min(h[i_pos], 1 - h[i_neg])
    if delta <= 0:
        # cannot transfer
        return False, h
    # optionally scale delta by factor
    delta = delta * step_scale
    # apply
    h_new = h.copy()
    h_new[i_pos] -= delta
    h_new[i_neg] += delta
    return True, h_new

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- helper utilities ---------------------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final clip (numerical safety)
    h[:] = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ------------------- SLSQP surrogate ---------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000,
                        init_temp=0.02, decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def quick_refine(h, rng):
    """Fast refinement: few β values and a short hill‑climb."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=800, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


# ------------------- gradient based local optimizer -------------------
def worst_shift_and_gradient(h):
    """Return (idx, O_max, k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    O = corr[idx]
    k = idx - (n - 1)
    grad = np.zeros_like(h)
    if k >= 0:
        # (1 - h[i+k]) for i=0..n-1-k
        grad[: n - k] += (1 - h[k:])
        # -h[i-k] for i=k..n-1
        grad[k:] -= h[: n - k]
    else:
        kp = -k
        # (1 - h[i-kp]) for i=kp..n-1
        grad[kp:] += (1 - h[: n - kp])
        # -h[i+kp] for i=0..n-1-kp
        grad[: n - kp] -= h[kp:]
    return idx, O, k, grad


def gradient_local_optimize(h, rng, time_deadline,
                            max_iters=5000, init_step=0.5):
    """Iteratively transfer mass from positive‑gradient to negative‑gradient sites."""
    step = init_step
    for _ in range(max_iters):
        if time.time() >= time_deadline:
            break
        idx, O, k, grad = worst_shift_and_gradient(h)
        g_max = grad.max()
        g_min = grad.min()
        if g_max <= 0 or g_min >= 0:
            break
        i_pos = int(np.argmax(grad))
        i_neg = int(np.argmin(grad))
        # feasible amount to move
        delta = min(h[i_pos], 1 - h[i_neg])
        if delta <= 0:
            break
        delta *= step
        if delta < 1e-12:
            break
        h[i_pos] -= delta
        h[i_neg] += delta
        # slowly shrink step size
        step *= 0.998
    return h


# ------------------- main entry point ---------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Initialise from the supplied parent solution (if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = quick_refine(h0, rng)

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts with quick refinement
    # ------------------------------------------------------------------
    while time.time() < start_time + budget_s * 0.45:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑based local optimisation on the current best
    # ------------------------------------------------------------------
    grad_deadline = start_time + budget_s * 0.80
    best_h = gradient_local_optimize(best_h.copy(), rng, grad_deadline,
                                    max_iters=8000, init_step=0.5)
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample and repeat gradient optimisation if time permits
    # ------------------------------------------------------------------
    if time.time() < start_time + budget_s * 0.95:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        deadline_up = start_time + budget_s * 0.98
        h_up = gradient_local_optimize(h_up, rng, deadline_up,
                                       max_iters=12000, init_step=0.5)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up, c_up
            n = h_up.size

    # ------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
