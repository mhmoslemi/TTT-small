def _simulated_annealing_binary(initial_h, n, time_budget):
    # ensure binary and half ones
    h = _to_binary_half(initial_h if initial_h is not None else _random_binary(n))
    # compute current C5
    def compute_c5(h_array):
        # compute min overlap via autocorrelation
        auto = np.correlate(h_array, h_array, mode='full')
        min_overlap = auto[n-1:].min()
        return 1.0 - (2.0 / n) * min_overlap

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # maintain ones and zeros lists
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]

    start_time = time.time()
    end_time = start_time + time_budget

    # Temperature schedule: linear cooling from T0 to T_end
    T0 = max(0.5, (best_c5 - 0.35))  # maybe scaling
    T_end = 1e-4

    iteration = 0
    while time.time() < end_time:
        # compute temperature
        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac  # exponential decay

        # choose random swap
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new C5
        new_c5 = compute_c5(h)

        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with probability exp(-delta/T)
            if T > 0 and np.random.rand() < np.exp(-delta / T):
                accept = True

        if accept:
            # accept: update indices
            cur_c5 = new_c5
            # update ones/zeros lists: replace i with j in ones_idx; replace j with i in zeros_idx
            # To modify arrays efficiently we can use list representation.
            # Convert to list for easier update
            # but we can implement as python list.
            # For simplicity, we store as python list.
            # But we currently have numpy arrays; we can convert to list once.

            pass

            best_h = h.copy()

def _to_binary_half(arr):
    """Given array of floats, return binary array with exactly half ones, picking largest values."""
    n = arr.size
    half = n // 2
    # get indices sorted descending
    idx = np.argpartition(-arr, half-1)[:half]  # faster than argsort
    bin_arr = np.zeros(n, dtype=float)
    bin_arr[idx] = 1.0
    return bin_arr

# Feasibility projection (water‑filling)

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation for a (feasible) step function
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Quadratic programming (SLSQP) to polish a continuous vector
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Interpolation utilities (coarse ↔ fine grids)
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Binary handling
# ----------------------------------------------------------------------
def _to_binary_half(arr: np.ndarray) -> np.ndarray:
    """Return a 0/1 array with exactly half ones (choose largest values)."""
    n = arr.size
    half = n // 2
    # argpartition is O(n), faster than argsort
    idx = np.argpartition(-arr, half - 1)[:half]
    bin_arr = np.zeros(n, dtype=float)
    bin_arr[idx] = 1.0
    return bin_arr


# ----------------------------------------------------------------------
# Simulated annealing on the binary feasible set
# ----------------------------------------------------------------------
def _simulated_annealing_binary(start_vec: np.ndarray, n: int, budget_s: float) -> tuple:
    """Binary SA respecting the half‑ones constraint."""
    h = _to_binary_half(start_vec)
    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5

    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    t0 = time.time()
    while time.time() - t0 < budget_s:
        elapsed = time.time() - t0
        frac = elapsed / budget_s
        # simple linear temperature schedule
        T = max(1e-6, 0.6 * (1.0 - frac) + 0.001)

        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _c5_from_h(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if best_c5 < 0.3808:
                    break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function for the Erdos C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Optional provided construction
    # ------------------------------------------------------------------
    try:
        init_h_vals = np.array(initial_h_values, dtype=float)
        init_h_vals = _project_to_feasible(init_h_vals, init_h_vals.size)
        best_h = init_h_vals.copy()
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size
    except NameError:
        init_h_vals = None
        best_h = None
        best_c5 = float("inf")
        best_n = None

    # ------------------------------------------------------------------
    # Grid sizes to explore (all even, < 800)
    # ------------------------------------------------------------------
    candidate_ns = [128, 160, 192, 224, 256, 320, 384, 512]
    candidate_ns = [n for n in candidate_ns if n <= 800]
    candidate_ns = sorted(set(candidate_ns))

    # ------------------------------------------------------------------
    # Search loop
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time_left() < 6.0:
            break

        # Prepare a start vector for this resolution
        if best_h is not None and best_n == n:
            start_vec = best_h.copy()
        elif best_h is not None:
            start_vec = _upscale_h(best_h, n)
        elif init_h_vals is not None and init_h_vals.size == n:
            start_vec = init_h_vals.copy()
        else:
            start_vec = np.random.rand(n)

        # Allocate time for simulated annealing; give extra budget to n=256
        remaining_candidates = len(candidate_ns) - candidate_ns.index(n)
        if n == 256:
            sa_budget = min(time_left() * 0.5 / remaining_candidates, 500.0)
        else:
            sa_budget = min(time_left() * 0.35 / remaining_candidates, 180.0)

        sa_h, sa_c5 = _simulated_annealing_binary(start_vec, n, sa_budget)
        if sa_c5 < best_c5:
            best_h, best_c5, best_n = sa_h.copy(), sa_c5, n

        # --------------------------------------------------------------
        # Local continuous polishing (SLSQP)
        # --------------------------------------------------------------
        if time_left() > 8.0:
            h_float = best_h.astype(float)
            h_opt, c5_opt = _solve_h(h_float, max_iter=800)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final polish if time permits
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 6.0:
        h_opt, c5_opt = _solve_h(best_h, max_iter=1000)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # ------------------------------------------------------------------
    # Fallback safety net
    # ------------------------------------------------------------------
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    else:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
