def local_search(h0, time_budget):
    n = len(h0)
    best_h = np.copy(h0)
    best_c5 = _c5_from_h(best_h)
    cur_h = np.copy(best_h)
    cur_c5 = best_c5
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # pick indices with some probability based on gradient? But random okay.
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # pick direction ensuring donor > receiver? Actually donor i should have higher value? Not strict.
        # We'll pick donor = i, receiver = j, then delta positive if h_i > h_j? Not required.
        # To make delta positive and keep within bounds:
        # Compute max delta that can be transferred: max_transfer = min(cur_h[i], 1 - cur_h[j])
        # If max_transfer ==0 continue.
        max_transfer = min(cur_h[i], 1 - cur_h[j])
        if max_transfer <= 1e-12:
            continue
        # Choose delta: random fraction of max_transfer
        delta = max_transfer * np.random.rand() * 0.5  # step factor <=0.5
        # Apply move
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # compute C5
        new_c5 = _c5_from_h(new_h)
        if new_c5 < cur_c5 - 1e-9:
            cur_h, cur_c5 = new_h, new_c5
            # record best
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
        else:
            # occasional acceptance with probability
            # compute temperature schedule
            total_time = time_budget
            elapsed = time.time() - start
            T = 0.01 * (1 - elapsed / total_time)  # linearly decay
            # acceptance probability
            if np.random.rand() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                cur_h, cur_c5 = new_h, new_c5
        it += 1
    return best_h, best_c5

def _project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    # adjust sum using waterfilling
    # if sum > target: lower values
    # else: increase low values
    # Use iterative scheme with at most max_iter.
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            # Need to reduce sum
            # Reduce proportionally from values >0
            mask = h > 0
            if not np.any(mask):
                break
            # reduce equally from these entries
            delta = diff / mask.sum()
            h[mask] -= delta
        else:
            mask = h < 1
            if not np.any(mask):
                break
            delta = -diff / mask.sum()
            h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction: adjust small residual by adding to first element that can.
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # find index where adjusting stays within bounds
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h,0.0,1.0)

        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12,
                         max_iter: int = 200) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = h > 0.0
        else:
            mask = h < 1.0
        if not np.any(mask):
            break
        delta = diff / mask.sum()
        h[mask] -= delta
        h = np.clip(h, 0.0, 1.0)
    # Tiny residual fix
    residual = target - h.sum()
    if abs(residual) > tol:
        for i in range(n):
            if residual > 0 and h[i] < 1.0 - tol:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > tol:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) <= tol:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a feasible step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """SLSQP refinement of a feasible h."""
    n = initial_h.size
    t0 = np.max(np.correlate(initial_h, 1.0 - initial_h, mode="full"))
    x0 = np.concatenate([initial_h, [t0]])

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": ineq},
            {"type": "eq",   "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0,
                            method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    if res.success:
        h_opt = res.x[:n]
    else:
        h_opt = initial_h.copy()
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _local_search(h_start: np.ndarray, time_budget: float) -> tuple:
    """Simple simulated‑annealing style search that preserves the sum."""
    n = h_start.size
    best_h = h_start.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer from i to j
        if cur_h[i] <= 1e-14 or cur_h[j] >= 1.0 - 1e-14:
            continue
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 1e-14:
            continue
        # fraction of the available transfer (scaled down for stability)
        delta = max_transfer * np.random.rand() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = _c5_from_h(new_h)

        if new_c5 < cur_c5 - 1e-12:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h, best_c5 = new_h.copy(), new_c5
                no_improve = 0
        else:
            # simulated annealing acceptance
            elapsed = time.time() - start
            T0 = 0.001
            T = T0 * (1.0 - elapsed / time_budget)
            if T > 0 and np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                cur_h, cur_c5 = new_h, new_c5
            no_improve += 1

        # occasional random perturbation to escape stagnation
        if no_improve > 5000 and (time.time() - start) < time_budget * 0.9:
            pert = 0.05 * np.random.randn(n)
            cur_h = _project_to_feasible(best_h + pert, n)
            cur_c5 = _c5_from_h(cur_h)
            no_improve = 0

    return best_h, best_c5

# ------------------------------------------------------------

    # ----------------------------------------------------------------
