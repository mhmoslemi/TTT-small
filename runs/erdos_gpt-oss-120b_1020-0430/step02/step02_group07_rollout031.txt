def compute_c5_fft(h):
    n = h.size
    f = h.astype(np.float64)
    g = 1.0 - f
    # reverse g
    g_rev = g[::-1]
    L = 1 << ((2*n - 1).bit_length())
    # compute FFTs
    F = np.fft.rfft(f, n=L)
    G = np.fft.rfft(g_rev, n=L)
    conv = np.fft.irfft(F * G, n=L)
    # conv length L, but relevant part first (2n-1)
    conv = conv[:2*n-1]
    # This conv is correlation (like np.correlate). Actually conv = correlate(f,g)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def compute_c5_fft(h):
    n = h.size
    f = h.astype(np.float64)
    g_rev = (1.0 - f)[::-1]
    L = 1 << ((2*n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(f, L) * np.fft.rfft(g_rev, L), n=L)[:2*n-1]
    max_corr = np.rint(conv).max()
    return max_corr * (2.0 / n)

def simulated_annealing_time(h_start, rng, time_limit, init_temp=0.05, decay=0.9999, scale=0.3):
    cur = h_start.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # propose perturb
        new_h = perturb_h(cur, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best = new_h.copy()
        # accept based on SA
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / (init_temp * (decay ** it) + 1e-12)):
            cur = new_h
            cur_c = new_c
        it += 1
    return best, best_c

def random_feasible_vector(n, rng):

        h = h * (target / s)

        h = np.zeros(n)
    # clip
    h = np.clip(h, 0.0, 1.0)
    # enforce sum precisely
    h = enforce_sum(h, target, rng)
    return h

def enforce_sum(h, target, rng, max_iter=5000):
    # target is exact sum target (float). h array values 0..1.
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    # fallback clip
    np.clip(h, 0.0, 1.0, out=h)
    return h

import time

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (float) with Σh = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s > 0:
        h *= target / s
    else:
        h[:] = 0.0
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)

def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP minimisation of the smooth max‑approximation."""
    n = h0.size
    target = n / 2.0

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny numerical correction to respect the sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
