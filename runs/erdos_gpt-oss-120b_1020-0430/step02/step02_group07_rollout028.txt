def refine(h0, rng, max_iter_slsqp=400, hill_iters=4000, anneal_iters=5000):
    # run series of beta values, gradually increasing
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in [2.0, 5.0, 10.0, 30.0, 80.0, 200.0, 500.0, 1000.0]:
        try:
            h_candidate = slsqp_optimize(h, beta, max_iter=max_iter_slsqp)
        except Exception:
            continue
        c_candidate = compute_c5(h_candidate)
        if c_candidate < best_c:
            best_c = c_candidate
            best_h = h_candidate
            h = h_candidate   # continue from best
        else:
            # maybe still set h = h_candidate for next beta
            h = h_candidate

    # hill climb
    h_hc, c_hc = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
    if c_hc < best_c:
        best_c = c_hc
        best_h = h_hc

    # Simulated annealing
    h_sa, c_sa = simulated_annealing(best_h, rng, max_iters=anneal_iters, init_temp=0.05, decay=0.999, scale=0.2)
    if c_sa < best_c:
        best_h, best_c = h_sa, c_sa

    return best_h, best_c

def compute_c5(h):
    n = h.size
    # Use FFT for n > 200 maybe
    if n <= 200:
        corr = np.correlate(h, 1 - h, mode='full')

import time, numpy as np, scipy.optimize

# global variables for SLSQP constraints
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def correlation_full(x):
    # use fast method
    n = x.size
    if n <= 250:
        return np.correlate(x, 1 - x, mode='full')
    else:
        # FFT method
        L = 2 * n
        H = np.fft.fft(x, L)
        G = np.fft.fft(1 - x, L)
        corr = np.fft.ifft(np.conj(H) * G).real
        return np.concatenate((corr[-(n-1):], corr[:n]))

def compute_c5(h):
    n = h.size
    corr = correlation_full(h)
    return np.max(corr) * (2.0 / n)

def enforce_sum(h, target, rng, max_iter=2000):
    # adjust using random transfers as earlier
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # ensure clipping
    np.clip(h, 0.0, 1.0, out=h)
    # final correction if still off due to rounding
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # monotonic adjustment across many indices
        # compute indices needing increase or decrease
        if diff > 0:
            # need to increase sum
            idxs = np.where(h < 1.0)[0]
            # if insufficient capacity, set to 1 for all
            for i in idxs:
                inc = min(1.0 - h[i], diff)
                h[i] += inc
                diff -= inc
                if abs(diff) < 1e-12:
                    break
        else:
            idxs = np.where(h > 0.0)[0]
            for i in idxs:
                dec = min(h[i], -diff)
                h[i] -= dec
                diff += dec
                if abs(diff) < 1e-12:
                    break
    return h

def random_feasible_vector(n, rng):
    target = n / 2.0

    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # ensure clip
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)

            best_c = c
            best_h = new_h.copy()
            h = new_h

        if (it + 1) % (max_iters // 10 + 1) == 0:

def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):

        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(temp, 1e-12)):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

def softmax_obj(x):
    # global variables used
    corr = correlation_full(x)  # full correlation values
    # scaling by beta
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

        h0,

import time
import numpy as np
from scipy.optimize import minimize

# globals used by SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def correlation_full(x):
    """Full correlation of x with (1-x). Uses FFT for speed when n is moderate."""
    n = x.size
    if n <= 250:
        return np.correlate(x, 1 - x, mode="full")
    L = 2 * n
    H = np.fft.fft(x, L)
    G = np.fft.fft(1 - x, L)
    corr = np.fft.ifft(np.conj(H) * G).real
    # reorder to match np.correlate's full mode output
    return np.concatenate((corr[-(n - 1):], corr[:n]))


def compute_c5(h):
    """Discrete C5 bound for step function represented by array h."""
    n = h.size
    corr = correlation_full(h)
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

    diff = target - h.sum()
    if abs(diff) > 1e-12:

    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa

    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise base solution
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ------------------------------------------------------------------
    # First intensive refinement of the provided / random start
    # ------------------------------------------------------------------
    best_h, best_c = deep_refine(h0, rng)

    # ------------------------------------------------------------------
    # Light quick refinement on many random restarts (exploration)
    # ------------------------------------------------------------------
    def quick_refine(h):
        cur_h, cur_c = h, compute_c5(h)
        for beta in (5.0, 20.0, 80.0):
            try:
                trial = slsqp_optimize(cur_h, beta=beta, max_iter=150)
            except Exception:
                continue
            c_trial = compute_c5(trial)
            if c_trial < cur_c:
                cur_h, cur_c = trial, c_trial
        cur_h, cur_c = hill_climb(cur_h, rng, max_iters=500, start_step=0.5)
        return cur_h, cur_c

    while time.time() - start_time < budget_s * 0.75:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # One more deep refinement on the current best if time permits
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.90:
        best_h, best_c = deep_refine(best_h, rng)

    # ------------------------------------------------------------------
    # Upsample to higher resolution (factor 2) and refine if budget remains
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.98:
        if best_h.size * 2 > 1200:
            break
        h_up = upsample_h(best_h, factor=2, rng=rng)
        h_up_ref, c_up_ref = deep_refine(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
        else:
            # if no improvement, stop further up‑sampling
            break

    # ------------------------------------------------------------------
    # Optional binary projection (might improve the bound)
    # ------------------------------------------------------------------
    h_bin = (best_h >= 0.5).astype(float)
    h_bin = enforce_sum(h_bin, h_bin.size / 2.0, rng)
    c_bin = compute_c5(h_bin)
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin

    # ------------------------------------------------------------------
    # Final feasible enforcement and evaluation
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
