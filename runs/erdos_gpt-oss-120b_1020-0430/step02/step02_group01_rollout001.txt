def random_feasible_vector(n, rng):
    h = rng.random(n)
    # scale to target sum
    target = n/2
    h = h * target / h.sum()
    # clip to [0,1] and then fix sum
    np.clip(h, 0, 1, out=h)
    diff = target - h.sum()
    # adjust using random indices until sum corrected
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0: continue
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n/2.0
    # Bounds
    bounds = [(0.0, 1.0)] * n
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    # objective wrapper
    m = 2*n -1
    def obj(x):
        H = np.fft.fft(x, m)
        G = np.fft.fft(1 - x, m)
        corr = np.fft.ifft(H * np.conj(G)).real
        bc = beta * corr
        mmax = np.max(bc)
        # compute log-sum-exp
        return (mmax + np.log(np.sum(np.exp(bc - mmax)))) / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': max_iter, 'ftol':1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust similarly to previous
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def anneal(h, rng, max_iters=2000, start_temp=0.02, end_temp=0.001):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    # temperature schedule geometric
    temp = start_temp
    decay = (end_temp / start_temp) ** (1.0 / max_iters)
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j += 1
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        # epsilon scaled by step size maybe similar to start_temp? We'll use random fraction of max_eps
        eps = (rng.random() * 2 - 1) * max_eps * 0.5 # limit to +-0.5*max_eps
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            cur_h, cur_c = new_h, new_c
        temp *= decay
    return cur_h

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log-sum-exp on the correlation of x with (1-x)."""
    n = x.size
    m = 2 * n - 1
    H = np.fft.fft(x, m)
    G = np.fft.fft(1 - x, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    bc = CURRENT_BETA * corr
    mx = np.max(bc)
    return (mx + np.log(np.sum(np.exp(bc - mx)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    m = 2 * n - 1
    H = np.fft.fft(h, m)
    G = np.fft.fft(1 - h, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    return np.max(corr) * (2.0 / n)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-13:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.5):
    """Transfer a random amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() * 2 - 1) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h0, rng, max_iters=4000, start_step=0.6):
    """Deterministic hill‑climb that only accepts improvements."""
    h = h0.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            h = cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.9, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Run SLSQP on the smooth surrogate; repairs bounds and sum afterwards."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def anneal(h0, rng, max_iters=3000, start_temp=0.02, end_temp=0.001):
    """Simple simulated‑annealing using pairwise transfers."""
    h = h0.copy()
    cur_c = compute_c5(h)
    temp = start_temp
    decay = (end_temp / start_temp) ** (1.0 / max_iters)
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            temp *= decay
            continue
        eps = (rng.random() * 2 - 1) * max_eps * 0.5
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            h, cur_c = new_h, new_c
        temp *= decay
    return h


def upsample_to_n(parent, target_n):
    """Upsample a parent vector to target_n points (simple repeat if divisible)."""
    n0 = parent.size
    if target_n % n0 == 0:
        factor = target_n // n0
        h = np.repeat(parent, factor)
    else:
        # linear interpolation on the interval [0,2)
        x_parent = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_target = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_target, x_parent, parent)
    # enforce feasibility
    target = target_n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    rng = np.random.default_rng(0)
    for _ in range(2000):
        if abs(diff) < 1e-13:
            break
        i = rng.integers(0, target_n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose a modest resolution that offers enough freedom
    n = 240

    if "initial_h_values" in globals():

    best_h = h0.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop (time‑aware)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.94:

        if c < best_c:

        for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
            if time.time() - start > budget_s * 0.96:

            c = compute_c5(cand)

        # occasional random restart to escape possible stagnation
