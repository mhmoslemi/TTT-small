import time
import numpy as np

# -------------------------------------------------
# basic utilities
# -------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target (box‑sum projection)."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

def random_binary_half(n, rng):
    """Return a random 0/1 half‑vector of length n/2 with exactly n/4 ones."""
    half = n // 2
    need = half // 2
    half_vec = np.zeros(half, dtype=int)
    half_vec[:need] = 1
    rng.shuffle(half_vec)
    return half_vec

    """Construct the full symmetric vector from its half."""
    return np.concatenate([half, half[::-1]]).astype(float)

        cand_full = full_from_half(cand)
        c_new = compute_c5(cand_full)
        if c_new < best_c:
            best_half, best_c = cand, c_new
            no_imp = 0
        else:
            no_imp += 1
    return best_half, best_c


# -------------------------------------------------
# projected sub‑gradient descent on the max‑overlap
# -------------------------------------------------
def subgrad_descent(h0, steps, step_size, rng, deadline):
    """Projected sub‑gradient descent on max_k C(k)."""
    n = h0.size
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    it = 0
    while time.time() < deadline and it < steps:
        corr = np.correlate(h, 1 - h, mode="full")
        C = corr * (2.0 / n)
        max_val = C.max()
        idxs = np.where(np.abs(C - max_val) < 1e-12)[0]
        g = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                g[: n - k] += 1 - h[k:]
                g[k:] -= h[: n - k]
            else:
                kk = -k
                g[kk:] += 1 - h[: n - kk]
                g[: n - kk] -= h[kk:]
        h_new = h - step_size * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        h_new = (h_new + h_new[::-1]) * 0.5  # enforce symmetry
        h_new = project_onto_box_sum(h_new, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        # accept downhill or occasional uphill
        if c_new < c_curr or rng.random() < 0.005:
            h, c_curr = h_new, c_new
        step_size *= 0.9995
        it += 1
    return best_h, best_c


# -------------------------------------------------
# main entry point
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # -----------------------------------------------------------------
    # initialise a feasible symmetric h (continuous)
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, n / 2.0)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 180
        half = n // 2
        half_target = n / 4.0
        x = rng.random(half)
        x = project_onto_box_sum(x, half_target)
        h0 = np.concatenate([x, x[::-1]])

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # -----------------------------------------------------------------
    # Phase 1: continuous sub‑gradient descent (≈30% of budget)
    # -----------------------------------------------------------------
    t_end_phase1 = start + budget_s * 0.30
    best_h, best_c = subgrad_descent(best_h, steps=4000, step_size=0.5,
                                     rng=rng, deadline=t_end_phase1)

    # -----------------------------------------------------------------
    # Phase 2: binary hill‑climbing restarts (≈45% of budget)
    # -----------------------------------------------------------------
    t_end_phase2 = start + budget_s * 0.75
    while time.time() < t_end_phase2:
        half_vec = random_binary_half(n, rng)
        half_opt, c_bin = binary_hill_climb(half_vec, rng, max_no_improve=250)
        if c_bin < best_c:
            best_c = c_bin
            best_h = full_from_half(half_opt)
        # early exit if already very good
        if best_c < 0.3807:
            break

    # -----------------------------------------------------------------
    # Phase 3: final continuous refinement (≈25% of budget)
    # -----------------------------------------------------------------
    t_end_phase3 = start + budget_s
    best_h, best_c = subgrad_descent(best_h, steps=5000, step_size=0.05,
                                     rng=rng, deadline=t_end_phase3)

    # -----------------------------------------------------------------
    # final clean‑up and return
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
