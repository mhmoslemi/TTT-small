def simulated_annealing(h, rng, max_iter, init_temp, final_temp):
    best = h.copy()
    best_c = compute_c5(h)
    cur = h.copy()
    cur_c = best_c
    n = len(h)
    for t in np.linspace(init_temp, final_temp, max_iter):
        # propose swap
        # pick i where cur[i]==1 and j where cur[j]==0
        # get indices from arrays of ones and zeros for fast selection
        # if no such positions (unlikely) continue
        i = rng.choice(np.flatnonzero(cur == 1))
        j = rng.choice(np.flatnonzero(cur == 0))
        # swap
        cur[i], cur[j] = 0, 1
        new_c = compute_c5(cur)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / t):
            # accept
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert
            cur[i], cur[j] = 1, 0
    return best, best_c

def binary_swap_opt(h, rng, max_swaps, temp_start, temp_end):
    """Perform simulated annealing on binary vector h (0/1) with fixed half ones."""
    n = len(h)
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # maintain indices
    for step in range(max_swaps):
        # compute temperature:
        T = temp_start + (temp_end - temp_start) * (step / max_swaps)
        # choose a 1 and 0
        ones = np.flatnonzero(cur_h == 1.0)
        zeros = np.flatnonzero(cur_h == 0.0)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            if T > 0:
                accept = (rng.random() < np.exp(-delta / T))
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Global variables for the soft‑max surrogate used in SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k]) for i< n-k
            g[k:] -= h[:upper]                # - h[i] for i>=k
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def repair_binary(arr):
    """Return a binary vector with exactly half of the entries =1 (rounding up)."""
    n = arr.size
    target = n // 2
    idx = np.argpartition(arr, -target)[-target:]
    bin_arr = np.zeros(n, dtype=float)
    bin_arr[idx] = 1.0
    return bin_arr

def binary_swap_opt(h_start, rng, max_swaps, temp_start, temp_end):
    """
    Simulated‑annealing style optimisation on binary vectors.
    h_start must already be binary with half ones.
    """
    cur = h_start.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    n = cur.size
    for s in range(max_swaps):
        # linear temperature schedule
        T = temp_start + (temp_end - temp_start) * (s / max_swaps)
        ones = np.flatnonzero(cur == 1.0)
        zeros = np.flatnonzero(cur == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_c = compute_c5(cur)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = (rng.random() < np.exp(-delta / T))
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_c
