def shift_with_zeros(arr, k):
    n = arr.size
    if k > 0:
        out = np.empty_like(arr)
        out[:n-k] = arr[k:]
        out[n-k:] = 0
    elif k < 0:
        k = -k
        out = np.empty_like(arr)
        out[k:] = arr[:n-k]
        out[:k] = 0
    else:
        out = arr.copy()
    return out

def subgradient_optimize(initial_h, max_iter, lr0, decay, tol=1e-12):
    best_h = initial_h.copy()
    best_c5 = _c5_from_h(best_h)
    h = initial_h.copy()
    n = len(h)
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        c5 = max_val * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            # maybe break if sufficiently low.
        # Determine shift(s) where corr close to max
        # Use tolerance relative
        idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
        k = idxs[np.random.randint(len(idxs))] - (n-1)
        # compute gradient
        grad = np.ones_like(h) - shift_with_zeros(h, k) - shift_with_zeros(h, -k)
        # step
        lr = lr0 * (decay ** it)
        h = h - lr * grad
        # project

        # maybe if lr becomes too small break
    return best_h, best_c5

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h) = n/2 by a water‑filling scheme.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for the discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _shift_with_zeros(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift arr by k positions, filling vacated entries with zero."""
    n = arr.size
    if k > 0:
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
    elif k < 0:
        k = -k
        out = np.empty_like(arr)
        out[k:] = arr[: n - k]
        out[:k] = 0.0
    else:
        out = arr.copy()
    return out


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (local refinement)
# ----------------------------------------------------------------------
def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02) -> np.ndarray:
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


# ----------------------------------------------------------------------
# Subgradient optimisation on the worst shift
# ----------------------------------------------------------------------
def _subgradient_optimize(
    init_h: np.ndarray,
    max_iter: int,
    lr0: float,
    decay: float,
    start_time: float,
    budget_s: float,
) -> tuple:
    """Projected subgradient descent on the shift attaining max overlap."""
    n = init_h.size
    h = init_h.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s - 0.5:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        c5 = max_val * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # choose a worst‑shift at random (if several tie)
        worst_idxs = np.where(corr == max_val)[0]
        k = worst_idxs[np.random.randint(len(worst_idxs))] - (n - 1)

        # gradient of r_k = sum_i h_i (1 - h_{i+k})
        grad = np.ones_like(h) - _shift_with_zeros(h, k) - _shift_with_zeros(h, -k)

        lr = lr0 * (decay ** it)
        h = h - lr * grad
        h = _project_to_feasible(h, n)

    return best_h, best_c5
