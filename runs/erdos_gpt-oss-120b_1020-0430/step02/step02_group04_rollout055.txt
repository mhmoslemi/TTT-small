import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(full_h):
    """Discrete C5 for step function (dx = 2/len(h))."""
    n = full_h.size
    corr = np.correlate(full_h, 1.0 - full_h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while fixing sum(v)=target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def build_full_from_half(half):
    return np.concatenate([half, half[::-1]])


def random_half_feasible(m, rng):
    v = rng.random(m)
    return project_onto_box_sum(v, m / 2.0)


def perturb_half(half, rng, scale=0.2):
    m = half.size
    i = rng.integers(0, m)
    j = rng.integers(0, m)
    if i == j:
        return half
    max_eps = min(half[i], 1.0 - half[j])
    if max_eps <= 0.0:
        return half
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = half.copy()
    new[i] -= eps
    new[j] += eps
    return new


def hill_climb_half(start_half, rng, max_iters=2000, start_scale=0.5):
    half = start_half.copy()
    best_half = half.copy()
    cur_c = compute_c5(build_full_from_half(half))
    best_c = cur_c
    scale = start_scale
    reduce_every = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_half(half, rng, scale)
        cand_c = compute_c5(build_full_from_half(cand))
        if cand_c < best_c:
            best_c, best_half = cand_c, cand.copy()
        if cand_c < cur_c:
            half, cur_c = cand, cand_c
        # occasional uphill acceptance
        elif rng.random() < 0.001:
            half, cur_c = cand, cand_c
        if (it + 1) % reduce_every == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_half, best_c


def upsample_half(half, factor=2):
    m = half.size
    new_m = m * factor
    x_old = np.arange(m)
    x_new = np.linspace(0, m - 1, new_m)
    up = np.interp(x_new, x_old, half)
    return project_onto_box_sum(up, new_m / 2.0)


# ---------- smooth‑max gradient descent ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """One SLSQP pass on the log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, n / 2.0)


def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    dx = 2.0 / n
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            C = corr * dx
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()
            # gradient via explicit double‑loop (numpy slices)
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:] -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:] -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c, best_h, h = c_new, new_h.copy(), new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialize full vector ----
    if "initial_h_values" in globals():
        init_full = np.array(initial_h_values, dtype=float)
    else:
        init_full = None

    # target length (prefer 200 points for flexibility)
    if init_full is not None and init_full.size <= 200:
        n = 200
        if n % 2:
            n += 1
        old_n = init_full.size
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, n)
        full = np.interp(x_new, x_old, init_full)
    elif init_full is not None:
        n = init_full.size
        if n % 2:
            n += 1
            full = np.append(init_full, init_full[-1])
        else:
            full = init_full.copy()
    else:
        n = 200
        if n % 2:
            n += 1
        full = np.concatenate(
            [random_half_feasible(n // 2, rng), random_half_feasible(n // 2, rng)[::-1]]
        )

    # enforce feasibility and symmetry
    full = np.clip(full, 0.0, 1.0)
    full = (full + full[::-1]) * 0.5
    target_sum = n / 2.0
    full = project_onto_box_sum(full, target_sum)
    full = (full + full[::-1]) * 0.5
    full = project_onto_box_sum(full, target_sum)

    half_n = n // 2
    half = full[:half_n].copy()
    best_half, best_c = half.copy(), compute_c5(build_full_from_half(half))

    # ---- Phase 1: stochastic hill‑climbing restarts ----
    while time.time() - start_time < budget_s * 0.55:
        # start from a random point, occasionally blended with current best
        if rng.random() < 0.2:
            start_half = (random_half_feasible(half_n, rng) + best_half) * 0.5
        else:
            start_half = random_half_feasible(half_n, rng)
        start_half = project_onto_box_sum(start_half, n / 4.0)

        # adapt iteration budget to remaining time
        remaining = budget_s - (time.time() - start_time)
        iters = int(min(6000, max(1500, remaining * 5)))  # generous early budget
        local_half, local_c = hill_climb_half(start_half, rng, max_iters=iters, start_scale=0.5)

        if local_c < best_c:
            best_half, best_c = local_half, local_c

        # occasional fine‑tuning of the incumbent
        if rng.random() < 0.1:
            fine_half, fine_c = hill_climb_half(best_half, rng, max_iters=2000, start_scale=0.2)
            if fine_c < best_c:
                best_half, best_c = fine_half, fine_c

        if best_c <= 0.3808:
            break

    # ---- Phase 2: smooth‑max gradient descent + optional SLSQP ----
    best_full = build_full_from_half(best_half)
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 2000.0]
    steps_per_beta = 300
    while time.time() - start_time < budget_s * 0.90:
        refined = grad_descent_optimize(best_full, beta_sched, steps_per_beta, rng)
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_full, best_c = refined, refined_c
            best_half = best_full[:half_n]

        # a quick SLSQP polish
        try:
            best_full = slsqp_optimize(best_full, beta=200.0, max_iter=200)
            best_c = compute_c5(best_full)
            best_half = best_full[:half_n]
        except Exception:
            pass

        if best_c <= 0.3808:
            break

    # ---- Phase 3: up‑sample if size permits and time remains ----
    if time.time() - start_time < budget_s * 0.97 and n <= 200:
        up_half = upsample_half(best_half, factor=2)
        up_best, up_c = hill_climb_half(up_half, rng, max_iters=6000, start_scale=0.5)
        up_full = build_full_from_half(up_best)
        up_c = compute_c5(up_full)
        if up_c < best_c:
            best_full, best_c = up_full, up_c
            n = best_full.size
            half_n = n // 2

    # ---- final clean‑up ----
    best_full = (best_full + best_full[::-1]) * 0.5
    best_full = np.clip(best_full, 0.0, 1.0)
    best_full = project_onto_box_sum(best_full, best_full.size / 2.0)
    best_full = (best_full + best_full[::-1]) * 0.5
    best_full = project_onto_box_sum(best_full, best_full.size / 2.0)
    final_c = compute_c5(best_full)

    return best_full.tolist(), float(final_c), int(best_full.size)
