def c5_of_h(h):
    n = h.shape[0]
    # compute convolution of h with (1-h) via FFT
    f = np.fft.fft(h, n*2)  # zero-pad to length 2n for linear convolution.
    g = np.fft.fft(1 - h, n*2)
    conv = np.fft.ifft(f * g).real
    # conv length 2n, but actual linear convolution length 2n-1 (due to zero padding)
    max_corr = np.max(conv[:2*n-1])
    return max_corr * (2.0 / n)

def search_best_h(n, time_limit):
    best_h = None
    best_c5 = float('inf')
    start = time.time()
    while time.time() - start < time_limit:
        # init random binary half
        h = np.zeros(n, dtype=np.float64)
        h[:n//2] = 1.0
        np.random.shuffle(h)
        c5 = compute_c5(h)
        # hill climb
        no_improve = 0
        # Precompute ones and zeros indices
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)

            # pick random swap
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # perform trial swap
            h[i] = 0.0
            h[j] = 1.0
            # compute C5

            if trial_c5 + 1e-12 < c5:

import time

# -------------------------------------------------------------
#  Helper utilities

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h)=n/2 by a fast water‑filling.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # Free indices that can move in the needed direction
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (rare)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx  (discretised)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# -------------------------------------------------------------
#  Gradient‑descent optimiser for the max‑overlap objective
# -------------------------------------------------------------
def _gd_optimize(h: np.ndarray, max_iter: int = 200, alpha: float = 0.2, decay: float = 0.96) -> tuple:
    """Projected gradient descent on the max‑overlap objective."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    for _ in range(max_iter):
        # correlation and dominant shift
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr_full))
        k = abs(idx - (n - 1))

        if k == 0:
            fwd = h
            bwd = h
        else:
            fwd = np.concatenate([h[k:], np.zeros(k)])
            bwd = np.concatenate([np.zeros(k), h[: n - k]])

        grad = 1.0 - (fwd + bwd)          # sub‑gradient of the active max‑term
        h = h - alpha * grad
        h = _project_to_feasible(h, n)

        alpha *= decay
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def _local_refine(h: np.ndarray, attempts: int = 20, delta: float = 0.02) -> tuple:
    """Small stochastic perturbations around a feasible point."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    for _ in range(attempts):
        trial = best_h.copy()
        trial += delta * np.random.randn(n)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h, best_c5

        n = 96
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        np.random.shuffle(best_h)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
