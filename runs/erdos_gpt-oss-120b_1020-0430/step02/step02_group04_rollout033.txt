def adam_optimize(h0, beta, n_iters, lr, rng):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1=0.9; beta2=0.999; eps=1e-8
    for t in range(1, n_iters+1):
        g = smooth_max_gradient(h, beta)
        m = beta1 * m + (1 - beta1) * g

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target (box‑sum projection)."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Random symmetric feasible vector (0≤h≤1, Σh=n/2)."""
    half = n // 2
    target_half = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


# ---------- perturbation ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth‑max gradient ----------
def smooth_max_gradient(h, beta):
    """Gradient of (1/β)·log‑sum‑exp(β·O_k) where O_k = overlap for shift k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * (2.0 / n)                     # overlap values for all shifts
    bc = beta * O
    m = np.max(bc)
    w = np.exp(bc - m)
    w_sum = w.sum()
    w = w / w_sum                            # soft‑max weights (sum to 1)

    # convolutions to compute Σ w_k·h_{i±k}
    conv_plus = np.convolve(w[::-1], h, mode="full")
    conv_minus = np.convolve(w, h, mode="full")
    start = n - 1
    plus_slice = conv_plus[start:start + n]
    minus_slice = conv_minus[start:start + n]

    grad = (2.0 / n) * (1.0 - plus_slice - minus_slice)
    return grad


def gradient_descent_softmax(h0, beta_sched, steps_per_beta, lr0, rng):
    """Projected gradient descent on smooth‑max surrogate."""
    h = h0.copy()
    for beta in beta_sched:
        lr = lr0
        for _ in range(steps_per_beta):
            grad = smooth_max_gradient(h, beta)
            h = h - lr * grad
            h = np.clip(h, 0.0, 1.0)
            # keep exact sum
            h = project_onto_box_sum(h, h.size / 2.0)
            # enforce symmetry
            h = (h + h[::-1]) * 0.5
            h = project_onto_box_sum(h, h.size / 2.0)
            lr *= 0.9995
    return h
