def gradient_descent(h, rng, max_iters, time_remaining):
    best_h = h.copy()
    best_c = compute_c5(h)
    for iter in range(max_iters):
        # Stop if time exceeded
        if time.time() - start_time > time_remaining: break
        # Compute corr, worst shifts
        corr = compute_corr_raw(best_h)  # returns raw sum (no dx)
        C_max = corr.max()
        worst_idx = np.where(corr == C_max)[0]
        # Map to shifts
        n = best_h.size
        shifts = worst_idx - (n - 1)
        # Compute subgradient
        g = np.zeros(n)
        # For each shift
        for k in shifts:
            # Build term_plus: vector of length n
            idx_plus = np.arange(n) + k
            mask_plus = (idx_plus >= 0) & (idx_plus < n)
            term_plus = np.empty(n)
            term_plus[:] = 1.0
            term_plus[mask_plus] = 1.0 - best_h[idx_plus[mask_plus]]
            # term_minus
            idx_minus = np.arange(n) - k
            mask_minus = (idx_minus >= 0) & (idx_minus < n)
            term_minus = np.zeros(n)
            term_minus[mask_minus] = best_h[idx_minus[mask_minus]]
            g += term_plus - term_minus
        # find i_from (max g) and i_to (min g)
        i_from = np.argmax(g)
        i_to = np.argmin(g)
        # Check if moving reduces max correlation
        # epsilon candidate
        max_eps = min(best_h[i_from], 1 - best_h[i_to])
        if max_eps <= 1e-12:
            # can't move, break
            break
        # try epsilon = max_eps * factor, start with factor=0.5, then shrink if needed.
        factor = 0.5
        improved = False
        while factor > 1e-4:
            eps = max_eps * factor
            h_new = best_h.copy()
            h_new[i_from] -= eps
            h_new[i_to] += eps
            # compute new C
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_h = h_new
                best_c = c_new
                improved = True
                break
            factor *= 0.5
        if not improved:
            # try random small perturbation maybe
            # If can't improve via gradient, break
            break
    return best_h, best_c

def corr_raw(h):
    g = 1 - h
    # correlation: sum_i h_i * (1 - h_{i+k}) for shift k (k = -n+1,...,n-1).
    # Using convolution: c[k] = sum_i h[i] * g[i + k] where g index out-of-range yields 0.
    # This is same as convolution h * rev(g) (since np.convolve uses sum_i a[i] * v[k-i]).
    return np.convolve(h, g[::-1], mode='full')

def gradient_improve(h, rng, max_iters, time_left):

    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_left:
            break
        corr = compute_corr_raw(best_h)
        max_corr = corr.max()
        # stops if max_corr <= best_c*some tolerance?
        # but we need to compute new max_corr messages after update; but we will.
        # Get worst shifts
        worst_idx = np.where(corr == max_corr)[0]
        # Map to shifts
        shifts = worst_idx - (n - 1)
        # Compute subgradient
        g = np.zeros(n)
        ar = np.arange(n)
        for k in shifts:
            idx_plus = ar + k
            mask_plus = (idx_plus >= 0) & (idx_plus < n)
            # term_plus default 1
            term_plus = np.ones(n)
            term_plus[mask_plus] = 1.0 - best_h[idx_plus[mask_plus]]
            idx_minus = ar - k
            mask_minus = (idx_minus >= 0) & (idx_minus < n)
            term_minus = np.zeros(n)
            term_minus[mask_minus] = best_h[idx_minus[mask_minus]]
            g += term_plus - term_minus
        # Choose indices
        i_from = np.argmax(g)
        i_to = np.argmin(g)
        # Ensure beneficial direction: g[i_from] > 0 > g[i_to] ideally
        if g[i_from] <= 0 or g[i_to] >= 0:
            # no direction for improvement
            break
        max_eps = min(best_h[i_from], 1.0 - best_h[i_to])
        if max_eps < 1e-12:
            break
        # try factor decreasing
        improved = False
        # use decreasing factors: 0.5, 0.25, 0.125, ...
        for factor in (0.5, 0.25, 0.125, 0.0625):
            eps = max_eps * factor
            if eps < 1e-12:
                continue
            h_new = best_h.copy()
            h_new[i_from] -= eps
            h_new[i_to] += eps
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            # maybe try a random perturb move
            # choose random i,j and small epsilon
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and best_h[i] > 1e-12 and best_h[j] < 1.0 - 1e-12:
                max_eps2 = min(best_h[i], 1.0 - best_h[j])
                eps2 = max_eps2 * 0.1
                h_new = best_h.copy()
                h_new[i] -= eps2
                h_new[j] += eps2
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_h = h_new
                    best_c = c_new
        # stop early if improvement minimal
        if not improved:
            # cannot improve further this iteration
            # maybe continue but break after few no-improve loops
            break
    return best_h, best_c

        break

def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    target = n / 2.0
    # generate random numbers and scale to target
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
    else:
        h = h * (target / s)
    # clip and enforce sum
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def upsample_h(h, factor, rng):
    n = h.size
    # treat h as over [0, 1) (since we ignore actual domain length)
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    target_up = (n * factor) / 2.0
    h_up = np.clip(h_up, 0.0, 1.0)
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up

def compute_corr_raw(h):
    g = 1 - h
    return np.convolve(h, g[::-1], mode='full')

def compute_c5(h):
    corr = compute_corr_raw(h)
    return np.max(corr) * (2.0 / h.size)

import time
import numpy as np

def compute_corr_raw(h):
    """Linear correlation of h with (1‑h). Length = 2*len(h)-1."""
    g = 1.0 - h
    return np.convolve(h, g[::-1], mode="full")


def compute_c5(h):
    """Exact discrete C₅ for a step function on [0,2]."""
    return np.max(compute_corr_raw(h)) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector with values in [0,1] and Σ = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0.0:
        h = np.zeros(n)
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    target_up = (n * factor) / 2.0
    h_up = np.clip(h_up, 0.0, 1.0)
    return enforce_sum(h_up, target_up, rng)


def simple_hill_climb(h, rng, max_iters=2000):
    """Transfer small amounts between coordinates, accepting only improvements."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * (rng.random() * 0.5 + 0.01)  # 1%‑50%
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = compute_c5(new_h)
        if c_new < best_c:
            best_h, best_c = new_h, c_new
    return best_h, best_c


def gradient_improve(h, rng, max_iters=3000, time_limit=5.0):
    """Sub‑gradient descent on the max‑correlation objective."""
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = best_h.size
    ar = np.arange(n)

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr_raw(best_h)
        max_corr = corr.max()
        worst_idx = np.where(corr == max_corr)[0]
        shifts = worst_idx - (n - 1)

        # sub‑gradient for the active shifts
        g = np.zeros(n)
        for k in shifts:
            idx_plus = ar + k
            mask_plus = (idx_plus >= 0) & (idx_plus < n)
            term_plus = np.ones(n)
            term_plus[mask_plus] = 1.0 - best_h[idx_plus[mask_plus]]

            idx_minus = ar - k
            mask_minus = (idx_minus >= 0) & (idx_minus < n)
            term_minus = np.zeros(n)
            term_minus[mask_minus] = best_h[idx_minus[mask_minus]]

            g += term_plus - term_minus

        i_from = int(np.argmax(g))
        i_to = int(np.argmin(g))

        if g[i_from] <= 0.0 or g[i_to] >= 0.0:
            break            # no descent direction for the active set

        max_eps = min(best_h[i_from], 1.0 - best_h[i_to])
        if max_eps < 1e-12:
            break

        improved = False
        factor = 0.5
        while factor > 1e-4:
            eps = max_eps * factor
            h_new = best_h.copy()
            h_new[i_from] -= eps
            h_new[i_to] += eps
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_h, best_c = h_new, c_new
                improved = True
                break
            factor *= 0.5

        if not improved:
            # a small random transfer as a fallback
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps2 = min(best_h[i], 1.0 - best_h[j])
                if max_eps2 > 0.0:
                    eps2 = max_eps2 * 0.1
                    h_new = best_h.copy()
                    h_new[i] -= eps2
                    h_new[j] += eps2
                    c_new = compute_c5(h_new)
                    if c_new < best_c:
                        best_h, best_c = h_new, c_new
    return best_h, best_c


def refine_h(h, rng, time_budget):
    """Combine gradient descent and hill climbing under a time budget."""
    start = time.time()
    best_h, best_c = h.copy(), compute_c5(h)
    while time.time() - start < time_budget:
        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # gradient phase (up to half of remaining time)
        sub_time = min(remaining * 0.6, 5.0)
        h_g, c_g = gradient_improve(best_h, rng,
                                     max_iters=2000,
                                     time_limit=sub_time)
        if c_g < best_c - 1e-12:
            best_h, best_c = h_g, c_g

        # short hill‑climb to polish
        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        max_hc_iters = int(1500 * (remaining / time_budget) + 1)
        max_hc_iters = min(max_hc_iters, 2000)
        h_hc, c_hc = simple_hill_climb(best_h, rng, max_iters=max_hc_iters)
        if c_hc < best_c - 1e-12:
            best_h, best_c = h_hc, c_hc

        # if no progress in this round, exit
        if abs(c_g - best_c) < 1e-12 and abs(c_hc - best_c) < 1e-12:
            break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Starting vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ----------------------------------------------------------------------
    # Phase 0 – refine the supplied or random start (≈15 % budget)
    # ----------------------------------------------------------------------
    phase0 = budget_s * 0.15
    best_h, best_c = refine_h(h0, rng, phase0)

    # ----------------------------------------------------------------------
    # Phase 1 – many random restarts at the same resolution (≈30 % budget)
    # ----------------------------------------------------------------------
    phase1 = budget_s * 0.30
    while time.time() - start_time < phase0 + phase1:
        remaining = phase0 + phase1 - (time.time() - start_time)
        if remaining < 0.5:
            break
        sub = min(1.0, remaining * 0.5)
        h_rand = random_feasible_vector(n, rng)
        h_ref, c_ref = refine_h(h_rand, rng, sub)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----------------------------------------------------------------------
    # Phase 2 – up‑sample and deep refine (use the rest of the budget)
    # ----------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    factor = 2
    while remaining > 0.4 and n * factor <= 1200:
        h_up = upsample_h(best_h, factor, rng)
        n_up = h_up.size
        # allocate the majority of the remaining time to this level
        allocated = remaining * 0.8
        h_up_ref, c_up_ref = refine_h(h_up, rng, allocated)
        if c_up_ref < best_c:
            best_h, best_c, n = h_up_ref, c_up_ref, n_up
        # prepare for possible second up‑sample
        remaining = budget_s - (time.time() - start_time)
        factor *= 2

    # ----------------------------------------------------------------------
    # Final clean‑up
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
