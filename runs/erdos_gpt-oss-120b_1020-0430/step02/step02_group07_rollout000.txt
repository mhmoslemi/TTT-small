import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the soft‑max surrogate (used by SLSQP)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    # correlation of x with its complement (linear, not circular)
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h[:] = 0.0
    else:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # gradually shrink step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=3000,
                        init_temp=0.03, decay=0.998, scale=0.2):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Repair possible drift in the sum constraint
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def quick_refine(h, rng):
    """Fast refinement: a few β values + short hill‑climb."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=800, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def deep_refine(h, rng):
    """Intensive refinement for larger n."""
    n = h.size
    max_iter_slsqp = 250 if n > 300 else 300
    hill_iters = 1200 if n > 300 else 1500
    anneal_iters = 2500 if n > 300 else 3000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1200.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter_slsqp)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        # short hill‑climb after each β
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters // 2, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    # final annealing
    trial_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iters=anneal_iters,
                                         init_temp=0.02,
                                         decay=0.999,
                                         scale=0.2)
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise from supplied parent (if any) or generate a fresh start
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = quick_refine(h0, rng)

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts with quick refinement
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.48:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to higher resolution and deep refinement
    # ------------------------------------------------------------------
    # First up‑sample (factor 2) if time permits
    if time.time() - start_time < budget_s * 0.70:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        h_up_ref, c_up_ref = deep_refine(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = n_up

    # Optional second up‑sample (factor 2 again) if budget remains
    if time.time() - start_time < budget_s * 0.90 and n * 2 <= 1440:
        h_up2 = upsample_h(best_h, factor=2, rng=rng)
        n_up2 = h_up2.size
        h_up2_ref, c_up2_ref = deep_refine(h_up2, rng)
        if c_up2_ref < best_c:
            best_h, best_c = h_up2_ref, c_up2_ref
            n = n_up2

    # ------------------------------------------------------------------
    # Phase 3 – final polishing with a few high‑β SLSQP passes
    # ------------------------------------------------------------------
    for beta in (800.0, 2000.0, 5000.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=300)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            continue

    # ------------------------------------------------------------------
    # Final exact feasibility enforcement
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
