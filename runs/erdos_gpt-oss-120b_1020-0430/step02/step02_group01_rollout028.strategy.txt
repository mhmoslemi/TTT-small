def softmax_surrogate(h, beta=50):
    corr = np.correlate(h, 1-h, mode='full')
    m = np.max(corr)
    if m < 1e-10:
        return 0
    return (np.log(sum(np.exp((corr - m) / beta))) + beta * m) / beta
