def _project_to_feasible(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum(x)=len(x)/2}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    target = n / 2.0

    # bisection on λ such that sum clamp(v-λ,0,1) = target
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for round‑off
    resid = target - x.sum()
    if abs(resid) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            x[free] += resid / free.size
            x = np.clip(x, 0.0, 1.0)
    return x


def _full_corr(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) using FFT (mode='full')."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft((1.0 - h)[::-1], N)
    corr = np.fft.irfft(H * G, N)[: 2 * n - 1]
    return corr


def _c5_of_h(h: np.ndarray) -> float:
    """C₅ value for a feasible step function h."""
    n = h.shape[0]
    max_corr = np.max(_full_corr(h))
    return max_corr * (2.0 / n)


def _max_corr_and_shift(h: np.ndarray):
    """Return (max_corr, shift) where shift∈[−(n‑1), n‑1]."""
    n = h.shape[0]
    corr = _full_corr(h)
    idx = np.argmax(corr)
    return corr[idx], idx - (n - 1)


def _grad_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation at the given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        if k < n:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
    else:
        k = -shift
        if k < n:
            g[k:] = 1.0 - h[: n - k]
            g[: n - k] -= h[k:]
    return g

    h0: np.ndarray,

def _local_search(
    h0: np.ndarray,
    steps: int,
    delta_max: float = 0.1,
    improve_every: int = 10,
):
    """Transfer small mass between random coordinates, keeping the sum fixed."""
    h = h0.copy()
    best_val = _c5_of_h(h)
    n = h.size
    for step in range(steps):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # max transferable amount without violating bounds
        max_up = 1.0 - h[i]
        max_down = h[j]
        max_delta = min(max_up, max_down)
        if max_delta <= 0.0:
            continue
        delta = (np.random.rand() * 2.0 - 1.0) * delta_max
        delta = np.clip(delta, -max_delta, max_delta)
        if abs(delta) < 1e-12:
            continue
        h[i] += delta
        h[j] -= delta
        val = _c5_of_h(h)
        if val < best_val:
            best_val = val
        else:
            # revert if no improvement
            h[i] -= delta
            h[j] += delta

        # occasional re‑projection to fix tiny drift
        if (step + 1) % improve_every == 0:
            h = _project_to_feasible(h)
    return h, best_val

            starts.append(np.random.rand(n))

        for base in starts:
            if time_left() < 5.0:
                break
            # several passes with decreasing step size
            for alpha in (0.6, 0.4, 0.25, 0.15):
                if time_left() < 2.0:
                    break
                h_opt, val = _improve_h(base, max_iters=3000,
                                         step_alpha=alpha,
                                         patience=250)
                if val < best_val:
                    best_h, best_val, best_n = h_opt, val, n
                if best_val < 0.3808:
                    break
            if best_val < 0.3808:
                break

        if best_val < 0.3808:
            break

    # -----------------------------------------------------------------
    #  Final polishing: a longer refinement + local stochastic search
    # -----------------------------------------------------------------
    if best_h is not None and time_left() > 10.0:
        h_ref, val_ref = _improve_h(best_h, max_iters=6000,
                                    step_alpha=0.05,
                                    patience=500)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref
        # stochastic local search
        steps = int(min(50000, max(2000, time_left() * 300)))   # rough scaling
        h_final, val_final = _local_search(best_h, steps=steps,
                                           delta_max=0.08)
        if val_final < best_val:
            best_h, best_val = h_final, val_final

    # -----------------------------------------------------------------
    #  Fallback (should never be needed)
    # -----------------------------------------------------------------
    if best_h is None:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_val = _c5_of_h(best_h)

    # final feasibility enforcement and exact bound computation
    best_h = _project_to_feasible(best_h)
    best_val = _c5_of_h(best_h)

    return best_h.tolist(), float(best_val), int(best_n)
