import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target (box‑sum projection)."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0≤h≤1, Σh=n/2)."""
    half = n // 2
    target_half = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target_half)
    h = np.concatenate([x, x[::-1]])
    return h


def thue_morse_sequence(n):
    """Return binary Thue‑Morse sequence of length n (requires n a power of two)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count("1") & 1
    return seq


# ---------- hill‑climb ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- SLSQP (soft‑max) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr * (2.0 / x.size)
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                     # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()                              # soft‑max weights

            # gradient of the smooth max (raw, before scaling)
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            # enforce symmetry to keep search space small
            new_h = (new_h + new_h[::-1]) * 0.5
            new_h = project_onto_box_sum(new_h, n / 2.0)

            c_new = compute_c5(new_h)

                best_c = c_new
                best_h = new_h.copy()

            else:

                    h = new_h
            step *= 0.9995
    # final clean‑up
    best_h = (best_h + best_h[::-1]) * 0.5
    best_h = project_onto_box_sum(best_h, n / 2.0)
    return best_h, best_c

        h_start = project_onto_box_sum(h_start, n / 2.0)

        # smooth‑max gradient descent
        h_opt, c_opt = grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng)

        # hill‑climb refinement
        h_opt, c_opt = hill_climb(h_opt, rng, max_iters=2000, start_step=0.4)

        # a few SLSQP refinement passes at high beta
        for beta in (500, 1000, 2000):
            try:
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=250)
            except Exception:
                pass

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # ---- up‑sampling loop (if time permits) ----
    while best_h is not None and time.time() - start_time < budget_s * 0.98:
        n = best_h.size
        if n * 2 > 500:      # keep the problem small enough
            break
        # double resolution
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up = (h_up + h_up[::-1]) * 0.5
        h_up = project_onto_box_sum(h_up, n_up / 2.0)

        # optimise on the finer grid
        h_up, c_up = grad_descent_optimize(h_up, beta_sched, steps_per_beta, rng)
        h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.3)
        for beta in (500, 1000, 2000):
            try:
                h_up = slsqp_optimize(h_up, beta=beta, max_iter=250)
            except Exception:
                pass
        c_up = compute_c5(h_up)

        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()
        else:
            break    # no further gain from finer grid

    # ---- final tidy‑up ----
    if best_h is None:
        # very unlikely, but fallback to a random start
        n = 180
        best_h = random_feasible_symmetric(n, rng)
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = (best_h + best_h[::-1]) * 0.5
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_h.size)
