import time
import numpy as np
from scipy import optimize

# ---------- basic projection -------------------------------------------------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto 0<=x<=1 and sum(x)=target."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(30):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# ---------- fast correlation -------------------------------------------------
def _full_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (full mode)."""
    n = h.size
    g = 1.0 - h
    h_rev = h[::-1]
    n_fft = 1 << ((2 * n - 2).bit_length())  # next power of two >= 2n-1
    H = np.fft.rfft(h_rev, n_fft)
    G = np.fft.rfft(g, n_fft)
    conv = np.fft.irfft(H * G, n_fft)
    return conv[: 2 * n - 1]


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a discrete h."""
    return np.max(_full_corr_fft(h)) * (2.0 / h.size)


# ---------- constraints for SLSQP -------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = _full_corr_fft(h)
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 400):
    """Local refinement with SLSQP (starts from a feasible h)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(_full_corr_fft(h0))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


# ---------- helpers for building starting points -----------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


# ---------- main optimisation routine ----------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # load possible parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate grid sizes

    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (64, 96, 128, 160, 192, 224, 256, 300, 384, 448):
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 8.0:
            break

        # base construction
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:

        # local refinement from deterministic start
        h_opt, c5_val = _solve_h(base_h, max_iter=600)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target_bound:
                break

        # ------- global search in a low‑dimensional Fourier basis -------------
        # choose number of cosine modes
        m = max(3, min(8, n // 80))
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        cos_basis = np.cos(np.outer(xs, np.arange(1, m + 1) * np.pi))

        def de_obj(a):
            raw = 0.5 + 0.5 * (cos_basis @ a)          # shape (n,)
            h = np.clip(raw, 0.0, 1.0)
            h = _project_to_simplex_box(h, n / 2.0)
            return _c5_from_h(h)

        bounds = [(-1.0, 1.0)] * m
        # DE with modest budget; polish=False because we handle polishing later
        de_res = optimize.differential_evolution(

            updating='deferred',

        )
        a_best = de_res.x
        raw = 0.5 + 0.5 * (cos_basis @ a_best)
        h_de = np.clip(raw, 0.0, 1.0)
        h_de = _project_to_simplex_box(h_de, n / 2.0)

        # polish with SLSQP
        h_opt, c5_val = _solve_h(h_de, max_iter=800)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target_bound:
                break

        # ------- stochastic local restarts ---------------------------------
        attempts = 8
        sigma = 0.12
        for _ in range(attempts):
            if time_left() < 5.0:
                break
            if np.random.rand() < 0.5:
                start = best_h + sigma * np.random.randn(n)
            else:
                start = np.random.rand(n)
            start = _project_to_simplex_box(start, n / 2.0)
            h_opt, c5_val = _solve_h(start, max_iter=500)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 < target_bound:
                    break
            sigma *= 0.88
        if best_c5 < target_bound:
            break

    # fallback
    if best_h is None:
        n = 128
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
