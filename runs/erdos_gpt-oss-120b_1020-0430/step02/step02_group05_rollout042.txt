def _shape_h(r, p, n):
    if r <= 0.0:
        h0 = 0.5
        h = np.full(n, h0)
        return _project_to_simplex_box(h, n/2.0)  # though already sum = n*0.5 = n/2
    # compute h0 = 1 / (2 - 2r + 2r/(p+1))
    denom = 2.0 - 2.0 * r + 2.0 * r / (p + 1.0)
    h0 = 1.0 / denom
    # ensure h0 <= 1; if >1, clip and might break area; just clip
    if h0 > 1.0:
        h0 = 1.0
    # compute positions
    x = (np.arange(n) + 0.5) * (2.0 / n)
    # compute distance from nearer edge
    dist = np.minimum(x, 2.0 - x)
    # shape
    h = np.empty_like(x)
    mask = dist < r
    if p == 1.0:
        # faster compute linear ramp
        h[mask] = h0 * (dist[mask] / r)
    else:
        h[mask] = h0 * (dist[mask] / r) ** p
    h[~mask] = h0
    # Clip just in case
    h = np.clip(h, 0.0, 1.0)
    # Project onto simplex sum = n/2
    h = _project_to_simplex_box(h, n/2.0)
    # symmetrize (should already symmetric due to construction but numerical errors)
    h = (h + h[::-1]) * 0.5
    return h

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n ∩ {sum = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _soft_min_gradient(h: np.ndarray, beta: float):
    """Gradient of soft‑min of pairwise overlaps (to be maximised)."""
    conv = np.correlate(h, h, mode="full")
    m = conv.min()
    w = np.exp(-beta * (conv - m))
    w /= w.sum()
    grad = 2.0 * np.convolve(h, w, mode="same")
    return grad, conv


def _symmetrize_and_project(h: np.ndarray, target: float) -> np.ndarray:
    h = (h + h[::-1]) * 0.5
    return _project_to_simplex_box(h, target)


def _gradient_ascent_h(h0: np.ndarray, target: float, time_left):
    h = h0.copy()
    schedule = [
        (0.5, 0.08, 30),
        (1.0, 0.06, 30),
        (2.0, 0.04, 40),
        (5.0, 0.03, 50),
        (10.0, 0.02, 60),
        (20.0, 0.015, 80),
        (40.0, 0.012, 80),
        (80.0, 0.01, 100),
    ]
    for beta, lr, it in schedule:
        for _ in range(it):
            if time_left() < 0.2:
                return h
            grad, _ = _soft_min_gradient(h, beta)
            h = h + lr * grad
            h = _symmetrize_and_project(h, target)
    return h


def _shape_h(r: float, p: float, n: int) -> np.ndarray:
    """Return a symmetric piecewise‑power shape with ramp length r and power p."""
    if r <= 0.0:
        h = np.full(n, 0.5)
        return _project_to_simplex_box(h, n / 2.0)
    denom = 2.0 - 2.0 * r + 2.0 * r / (p + 1.0)
    h0 = 1.0 / denom
    if h0 > 1.0:
        return None
    x = (np.arange(n) + 0.5) * (2.0 / n)
    d = np.minimum(x, 2.0 - x)
    h = np.empty_like(x)
    mask = d < r
    if p == 1.0:
        h[mask] = h0 * (d[mask] / r)
    else:
        h[mask] = h0 * (d[mask] / r) ** p
    h[~mask] = h0
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_simplex_box(h, n / 2.0)
    h = (h + h[::-1]) * 0.5
    return h


def _search_shape_for_n(n: int, time_left):
    target_sum = n / 2.0
    best_h = None
    best_c5 = float("inf")
    p_vals = [0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.4, 1.6]
    for p in p_vals:
        if time_left() < 0.5:
            break
        r_max = min(1.0, (p + 1.0) / (2.0 * p))
        grid = np.linspace(0.0, r_max, 25)
        # coarse sweep
        for r in grid:

            h = _shape_h(r, p, n)
            if h is None:

            c5 = _c5_from_h(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        # refine around best r for this p
        vals = [(r, _c5_from_h(_shape_h(r, p, n))) for r in grid if _shape_h(r, p, n) is not None]
        if not vals:
            continue
        r_best, _ = min(vals, key=lambda x: x[1])
        delta = max(r_max / 10.0, 0.02)
        a, b = max(0.0, r_best - delta), min(r_max, r_best + delta)
        phi = (np.sqrt(5) - 1) / 2
        c = b - phi * (b - a)
        d = a + phi * (b - a)
        c_val = _c5_from_h(_shape_h(c, p, n))
        d_val = _c5_from_h(_shape_h(d, p, n))
        for _ in range(12):

            if c_val <= d_val:
                b, d, d_val = d, c, c_val
                c = b - phi * (b - a)
                c_val = _c5_from_h(_shape_h(c, p, n))
            else:
                a, c, c_val = c, d, d_val
                d = a + phi * (b - a)
                d_val = _c5_from_h(_shape_h(d, p, n))
        for r_candidate in (a, b, c, d):
