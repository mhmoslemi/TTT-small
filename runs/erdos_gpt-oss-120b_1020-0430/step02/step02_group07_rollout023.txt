import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility helpers
# --------------------------------------------------------------
def compute_c5(h):
    """Exact C5 for a discrete step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def binary_swap_move(h, rng):
    """Swap a random 1 with a random 0 (preserves feasibility)."""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if ones.size == 0 or zeros.size == 0:
        return h
    i = rng.choice(ones)
    j = rng.choice(zeros)
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# --------------------------------------------------------------
# Soft‑max surrogate and SLSQP wrapper
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Repair any tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def refine_with_slsqp_and_hill(h, rng, time_left):
    """Apply a ladder of β‑values followed by a short hill‑climb."""
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1200.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        # Reduce max_iter for larger n to keep within time budget
        max_iter = 300 if best_h.size <= 200 else 200
        trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        # quick hill‑climb after each β
        trial_h, c_h = hill_climb(best_h, rng, max_iters=600, start_step=0.25)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
        if time.time() - start_time >= time_left:
            break
    return best_h, best_c


def upsample_h(h, target_n, rng):
    """Upsample by integer factor if possible, otherwise linear interpolation."""
    factor = target_n // h.size
    if factor * h.size == target_n:
        h_up = np.repeat(h, factor)
    else:
        x_old = np.linspace(0.0, 1.0, h.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    enforce_sum(h_up, target_n / 2.0, rng)
    return h_up
