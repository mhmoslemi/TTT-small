def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h,0.0,1.0)

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a given feasible h."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _neighbor_sum_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sum of neighbours needed to increase autocorrelation for a given shift."""
    n = h.shape[0]
    ns = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            ns[: n - shift] += h[shift:]
            ns[shift:] += h[: n - shift]
    else:
        s = -shift
        if s < n:
            ns[s:] += h[: n - s]
            ns[: n - s] += h[s:]
    return ns

def _improve_h(h0: np.ndarray, time_limit: float) -> tuple:
    """Greedy neighbour‑exchange descent guided by the worst shift."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_c = _c5_from_h(h)
    step = 0.05
    it = 0
    max_it = 400000

    while time.time() < time_limit and it < max_it:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        cur_c = max_corr * dx
        if cur_c < best_c - 1e-12:
            best_c = cur_c

        # worst shift
        k = int(np.argmax(corr))
        shift = k - (n - 1)

        ns = _neighbor_sum_for_shift(h, shift)

        mask_up = h < 1 - 1e-12
        mask_down = h > 1e-12
        if not np.any(mask_up) or not np.any(mask_down):
            break

        ns_up = ns.copy()
        ns_up[~mask_up] = -np.inf
        i_up = int(np.argmax(ns_up))

        ns_down = ns.copy()
        ns_down[~mask_down] = np.inf
        i_down = int(np.argmin(ns_down))

        if i_up == i_down:
            ns_up[i_up] = -np.inf
            if np.any(ns_up != -np.inf):
                i_up = int(np.argmax(ns_up))
            ns_down[i_down] = np.inf
            if np.any(ns_down != np.inf):
                i_down = int(np.argmin(ns_down))
            else:
                break

        eps = min(step, h[i_down], 1.0 - h[i_up])
        if eps <= 0:
            step *= 0.5
            continue

        cand = h.copy()
        cand[i_down] -= eps
        cand[i_up] += eps

        cand_c = np.max(np.correlate(cand, 1.0 - cand, mode="full")) * dx
        if cand_c < cur_c - 1e-12:
            h = cand
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.9
            if step < 1e-4:
                step = 1e-4

        # occasional random perturbation that improves the global best
        if np.random.rand() < 0.01:
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 1e-12:
                    eps2 = np.random.rand() * max_eps * 0.5
                    cand2 = h.copy()
                    cand2[i] -= eps2
                    cand2[j] += eps2
                    cand2_c = np.max(np.correlate(cand2, 1.0 - cand2, mode="full")) * dx
                    if cand2_c < best_c - 1e-12:
                        h = cand2
                        best_c = cand2_c
    return h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Starting vector
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: uniform half‑density on 80 points
        n0 = 80
        init_h = np.full(n0, 0.5)
    n0 = init_h.size
    init_h = _project_to_feasible(init_h, n0)

    # ------------------------------------------------------------------
    # First intensive improvement
    # ------------------------------------------------------------------
    best_h, best_c = _improve_h(init_h, deadline)
    best_n = n0

    # ------------------------------------------------------------------
    # Random restarts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5:
        sigma = 0.02 + 0.08 * np.random.rand()
        start_h = best_h + sigma * np.random.randn(best_n)
        start_h = _project_to_feasible(start_h, best_n)
        cand_h, cand_c = _improve_h(start_h, deadline)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ------------------------------------------------------------------
    # Upscale attempts (2×,3×,4×) with further refinement
    # ------------------------------------------------------------------
    for factor in (2, 3, 4):
        if time.time() >= deadline - 5:
            break
        n_up = best_n * factor
        if n_up > 512:
            continue
        # linear interpolation to the finer grid
        old_x = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        new_x = (np.arange(n_up) + 0.5) * (2.0 / n_up)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = _project_to_feasible(up_h, n_up)

        up_h_opt, up_c = _improve_h(up_h, deadline)
        if up_c < best_c:
            best_h, best_c, best_n = up_h_opt, up_c, n_up

    # ------------------------------------------------------------------
    # Return the (list, bound, n) triple
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_c), int(best_n)
