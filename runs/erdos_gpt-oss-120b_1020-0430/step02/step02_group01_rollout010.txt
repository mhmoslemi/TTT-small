def propose_move(h, rng, step_scale=0.1):
    n = h.size
    # attempt up to 10 times to find a feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if i == j:
            continue
        # choose direction
        if rng.random() < 0.5:
            # move from i to j
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * step_scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            return new_h
        else:
            max_eps = min(h[j], 1 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * step_scale
            new_h = h.copy()
            new_h[j] -= eps
            new_h[i] += eps
            return new_h
    return h  # no move

def refine_gradient(h, rng, steps=50, init_step=0.02):
    n = h.size
    target = n/2
    step = init_step
    for it in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')
        # find shift index where corr is max
        idx = np.argmax(corr) - (n - 1)  # shift k (negative possible)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1 - 2*h  # gradient for each i (since corr for shift 0 is zero always but for completeness)
        else:
            # g_i = (1 - h_{i+k}) for i in [0, n-k-1], minus h_{i-k} for i in [k, n-1]?
            # Actually for shift k>0: sum_i h[i] * (1 - h[i+k]) includes i from 0 to n-k-1.
            # Derivative w.r.t h[i] for i< n-k is (1 - h[i+k]), for i>=k there is negative term -h[i-k] because h[i] appears as (1 - h[i]) factor for shift -k? Need to careful.
            # However correlation is symmetric; using full correlation, the gradient for each index can be derived as in previous gradient_refine code: 
            # g[:upper] += 1 - h[k:] 
            # g[k:] -= h[:upper]
            upper = n - k
            g[:upper] += 1 - h[k:]
            g[k:] -= h[:upper]
        # gradient descent step (negative direction)
        h = h - step * g
        # project to [0,1]
        np.clip(h, 0, 1, out=h)
        # repair sum
        diff = target - h.sum()
        # adjust randomly as before
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            if diff > 0:
                add = min(diff, 1 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
        # optionally reduce step
        step *= 0.95
    return h

def compute_c5(h):
    """Exact C5 for a discretized step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def rand_feasible(n, rng):
    """Random vector in [0,1]^n with sum == n/2."""
    target = n / 2.0
    h = rng.random(n)
    # scale to hit the target on average
    scale = target / h.sum()
    h *= scale
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def propose_move(h, rng, step_scale=0.5):
    """Transfer a random amount from i to j (preserves sum)."""
    n = h.size
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if i == j:
            continue
        # direction i -> j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0 and rng.random() < 0.5:
            eps = rng.random() * max_eps * step_scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            return new_h
        # direction j -> i
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps > 0:
            eps = rng.random() * max_eps * step_scale
            new_h = h.copy()
            new_h[j] -= eps
            new_h[i] += eps
            return new_h
    return h  # fallback (no feasible move)


def perturb_small(h, rng, scale=0.2, steps=5):

def hill_climb_local(h_start, rng, max_iters=3000, step_scale=0.5):

def simulated_annealing(start_h, rng, max_iters=120000, time_limit=None):
    """Simple SA with linear temperature schedule."""
    h = start_h.copy()
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    n = h.size
    target = n / 2.0
    init_T, final_T = 0.06, 0.0004
    start_time = time.time()
    for it in range(max_iters):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        t = init_T + (final_T - init_T) * (it / max_iters)
        # step scale proportional to temperature (smaller later)
        step_scale = max(0.01, t / init_T)
        new_h = propose_move(h, rng, step_scale=step_scale)
        new_c = compute_c5(new_h)
        delta = new_c - curr_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            h = new_h
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c


def refine_gradient(h, rng, steps=60, init_step=0.02):
    """Subgradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    step = init_step
    for it in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * g
        np.clip(h, 0.0, 1.0, out=h)
        # repair sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= 0.95
    return h

    if abs(diff) > 1e-12:
        for _ in range(5000):
            i = rng.integers(best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
