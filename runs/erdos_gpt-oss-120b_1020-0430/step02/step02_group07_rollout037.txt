def simulated_annealing_binary(h, rng, max_iters, init_temp=0.1, decay=0.999):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()

    for it in range(max_iters):

        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new < cur_c or rng.random() < np.exp((cur_c - c_new) / temp):
            cur_h = h_new
            cur_c = c_new
        temp *= decay
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    # stable logsumexp
    max_bc = np.max(bc)
    lse = max_bc + np.log(np.sum(np.exp(bc - max_bc)))
    return lse / CURRENT_BETA
```

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly (drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # adjust by adding diff to random positions respecting bounds
        n = len(x)
        # we can distribute diff equally maybe.
        for _ in range(2000):
            i = np.random.randint(n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n, rng):
    """Return a binary (0/1) vector of length n with exactly n/2 ones (n should be even)."""
    assert n % 2 == 0, "n must be even for a binary construction"
    ones = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)[:ones]
    h[idx] = 1.0
    return h


def repair_binary_to_target(h, target, rng):
    """Flip bits to obtain exactly target ones (target integer)."""
    target_int = int(round(target))
    cur = int(h.sum())
    diff = cur - target_int
    if diff > 0:
        ones = np.where(h > 0.5)[0]
        if len(ones) >= diff:
            sel = rng.choice(ones, size=diff, replace=False)
            h[sel] = 0.0
    elif diff < 0:
        zeros = np.where(h <= 0.5)[0]
        if len(zeros) >= -diff:
            sel = rng.choice(zeros, size=-diff, replace=False)
            h[sel] = 1.0
    return h


def swap_move(h, rng):
    """Swap a 1 with a 0, preserving the sum."""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    nh = h.copy()
    nh[i] = 0.0
    nh[j] = 1.0
    return nh


def binary_hill_climb(h, rng, max_iter=5000, patience=1000):
    """Simple hill‑climb on binary vectors via random swaps."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    no_imp = 0
    for _ in range(max_iter):
        cand = swap_move(best_h, rng)
        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best_h, best_c = cand, c
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= patience:
            break
    return best_h, best_c


def simulated_annealing_binary(h, rng, max_iters, init_temp=0.05, decay=0.999):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    temp = init_temp
    for _ in range(max_iters):
        cand = swap_move(cur_h, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c = cand.copy(), c
        if c < cur_c or rng.random() < np.exp((cur_c - c) / temp):
            cur_h, cur_c = cand, c
        temp *= decay
    return best_h, best_c


# ------------------------------------------------------------
# Continuous refinement (soft‑max surrogate + SLSQP)
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300, rng=None):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Fix any tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if rng is None:
        rng = np.random.default_rng()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def continuous_refine(h, rng, time_budget):
    """Run a small ladder of SLSQP refinements respecting the time budget."""
    betas = (5.0, 20.0, 80.0, 200.0, 500.0)
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    for b in betas:
        if time.time() - start > time_budget:
            break
        try:
            trial = slsqp_optimize(best_h, beta=b, max_iter=300, rng=rng)
        except Exception:
            continue
        c = compute_c5(trial)
        if c < best_c:
            best_h, best_c = trial, c
        # a brief local SA on the continuous vector
        if time.time() - start < time_budget:
            trial2, c2 = simulated_annealing_binary(
                trial, rng, max_iters=800, init_temp=0.02, decay=0.998
            )
            if c2 < best_c:
                best_h, best_c = trial2, c2
    return best_h, best_c


def upsample_h_linear(h, factor, rng):
    """Linear interpolation up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.arange(n) / n
    x_new = np.arange(n * factor) / (n * factor)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up
