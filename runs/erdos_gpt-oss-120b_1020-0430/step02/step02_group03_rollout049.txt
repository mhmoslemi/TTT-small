def perturb(h, step, rng):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # choose delta uniformly from [-max_delta, max_delta] where max_delta limited by bound.
    # Let max_down = h[i]  # can't go below 0
    # Let max_up = 1 - h[j] # can't exceed 1
    # max delta positive = min(step, max_up)
    # max delta negative = -min(step, max_down)
    max_pos = min(step, 1.0 - h[j])
    max_neg = -min(step, h[i])
    if max_pos <= -max_neg:
        # only negative possible?
        delta = rng.uniform(max_neg, max_pos) # maybe both
    else:
        delta = rng.uniform(max_neg, max_pos)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
```

import time
import numpy as np
from scipy import optimize as opt

# ---------- helper utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while keeping 0≤h_i≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb(h: np.ndarray, step: float, rng: np.random.Generator) -> np.ndarray:
    """Transfer a bounded amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    # allowed range for delta
    max_pos = min(step, 1.0 - h[j])
    max_neg = -min(step, h[i])
    delta = rng.uniform(max_neg, max_pos)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


# ---------- smooth surrogate for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _smooth_obj(x):
    # log‑sum‑exp approximation of max_k Σ_i x_i(1‑x_{i+k})
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 400) -> np.ndarray:
    """Refine h using SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = opt.minimize(
        _smooth_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, CURRENT_TARGET, rng)


# ---------- main optimisation ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = rng.integers(120, 200)          # choose a modest resolution
        h0 = np.full(n, 0.5, dtype=float)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n / 2.0, rng)

    # ----- simulated annealing -----
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c

    max_iters = 400_000
    max_step = 0.30
    T0 = 0.005
    restart_interval = 50_000

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.78:
            break

        # annealing schedule
        step = max_step * (1.0 - it / max_iters) ** 2
        T = T0 * (1.0 - it / max_iters)

        cand = perturb(cur_h, step, rng)
        cand_c = compute_c5(cand)

        if cand_c < cur_c:
            cur_h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
        else:
            if T > 0 and rng.random() < np.exp(-(cand_c - cur_c) / T):
                cur_h, cur_c = cand, cand_c

        # occasional random restart to diversify
        if it % restart_interval == 0 and it > 0:
            if rng.random() < 0.2:
                cur_h = random_feasible_vector(n, rng)
                cur_c = compute_c5(cur_h)

    # ----- local SLSQP refinement -----
    beta_seq = [5.0, 20.0, 80.0, 200.0, 800.0]
    for beta in beta_seq:
        if time.time() - start_time > budget_s * 0.95:

        try:
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=800)

            continue
        best_c = compute_c5(best_h)

    # ----- optional up‑sampling -----
    if (time.time() - start_time < budget_s * 0.96) and (n * 2 <= 300):

        up_n = up_h.size

        cur_c = compute_c5(cur_h)
        best_up_h = cur_h.copy()
        best_up_c = cur_c
        max_iters_up = 200_000
        for it in range(max_iters_up):
            if time.time() - start_time > budget_s * 0.96:
                break
            step = 0.15 * (1.0 - it / max_iters_up) ** 2
            T = T0 * (1.0 - it / max_iters_up)
            cand = perturb(cur_h, step, rng)
            cand_c = compute_c5(cand)
            if cand_c < cur_c:
                cur_h, cur_c = cand, cand_c
                if cand_c < best_up_c:
                    best_up_h, best_up_c = cand.copy(), cand_c
            else:
                if T > 0 and rng.random() < np.exp(-(cand_c - cur_c) / T):
                    cur_h, cur_c = cand, cand_c

        # refine with SLSQP on the up‑sampled candidate
        for beta in beta_seq:
            if time.time() - start_time > budget_s:
                break
            try:
                best_up_h = slsqp_optimize(best_up_h, beta, rng, max_iter=600)
            except Exception:
                continue
            best_up_c = compute_c5(best_up_h)

        # keep whichever is better
        if best_up_c < best_c:
            best_h, best_c, n = best_up_h, best_up_c, up_n

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
