def c5_of_h(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    fft_h = np.fft.rfft(h, n=L)
    fft_1mh = np.fft.rfft(1-h, n=L)
    conv = np.fft.irfft(fft_h * fft_1mh, n=L)
    # conv length L, but we want first 2n-1 terms.
    conv = conv[:2*n-1]
    max_corr = conv.max()
    return max_corr * (2.0/n)

def _project_to_feasible(h, n):
    # enforce [0,1] and sum = n/2
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    current = h.sum()
    if abs(current - target) < 1e-12:
        return h
    # Compute needed adjustments
    diff = current - target
    # If diff>0 need to reduce some non-zero entries, else increase some non-1 entries
    # We'll adjust uniformly across eligible positions.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        count = mask.sum()
        if count == 0:
            break
        delta = diff / count
        # apply delta on masked entries
        h[mask] -= delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
    # final residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first eligible entries
        if residual > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            for i in idx:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if residual <= 1e-12:

        s = h.sum()

        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
