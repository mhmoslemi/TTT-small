import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Feasibility projection (box + sum constraint)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # Tiny correction
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        for i in range(n):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# SLSQP local refinement (same as previous solution)
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=constraints, options=opts
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


# ----------------------------------------------------------------------
# Upscale via linear interpolation
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Gradient of smooth max objective
# ----------------------------------------------------------------------
def _grad_smooth_max(h: np.ndarray, alpha: float) -> np.ndarray:
    n = h.size
    C = np.correlate(h, 1.0 - h, mode="full")
    Cmax = C.max()
    # stable softmax
    exparg = alpha * (C - Cmax)
    w = np.exp(exparg)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # convolution to get sums of weighted neighbours
    conv1 = np.convolve(w, h, mode="full")
    conv2 = np.convolve(w[::-1], h, mode="full")
    start = n - 1
    s1 = conv1[start:start + n]      # sum_s w_s * h_{i-s}
    s2 = conv2[start:start + n]      # sum_s w_s * h_{i+s}
    grad = 1.0 - s1 - s2
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent on smooth approximation
# ----------------------------------------------------------------------
def _pgd_smooth(h0: np.ndarray, alphas, step0: float, iters_per_alpha, time_left) -> np.ndarray:
    h = h0.copy()
    n = h.size
    for alpha in alphas:
        if not time_left():
            break
        step = step0 / np.sqrt(alpha + 1.0)
        for _ in range(iters_per_alpha):
            if not time_left():
                break
            grad = _grad_smooth_max(h, alpha)
            h = h - step * grad
            h = _project_to_feasible(h, n)
    return h

        # update index lists
        ones[i_idx] = j
        zeros[j_idx] = i
        new_c5 = _c5_from_h(cur)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and np.random.rand() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1.0
            cur[j] = 0.0
            ones[i_idx] = i
            zeros[j_idx] = j
    return best


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left(min_needed=0.5):
        return (time.time() - start_time) < (budget_s - min_needed)

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # provided by harness if any
    except NameError:
        init_h = None

    # default size if nothing provided
    base_n = init_h.size if init_h is not None else 256
    if base_n % 2 == 1:
        base_n += 1  # ensure even
    # ------------------------------------------------------------------
    # Helper to keep the best solution found
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")
    best_n = None

    def record_candidate(h):
        nonlocal best_h, best_c5, best_n
        c = _c5_from_h(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = float(c)
            best_n = h.size

    # ------------------------------------------------------------------
    # Start from the given or a random binary vector
    # ------------------------------------------------------------------
    if init_h is not None and init_h.size == base_n:
        h0 = _project_to_feasible(init_h, base_n)
    else:
        # random binary half 1's
        h0 = np.zeros(base_n, dtype=float)
        h0[: base_n // 2] = 1.0
        np.random.shuffle(h0)

    h0 = _project_to_feasible(h0, base_n)
    record_candidate(h0)

    # ------------------------------------------------------------------
    # Continuous refinement (smooth max + PGD)
    # ------------------------------------------------------------------
    alphas = [0.5, 2.0, 5.0, 12.0, 25.0, 50.0, 80.0]
    step0 = 0.07
    iters_per_alpha = 180
    h_cont = _pgd_smooth(h0, alphas, step0, iters_per_alpha, time_left)
    record_candidate(h_cont)

    # local SLSQP polish on the continuous solution
    if time_left(2.0):
        h_opt, c_val = _solve_h(h_cont, max_iter=500)
        record_candidate(h_opt)

    # ------------------------------------------------------------------
    # Binary hill‑climbing via simulated annealing
    # ------------------------------------------------------------------
    # Threshold continuous solution to a binary one (exact half‑ones)
    thresh = np.partition(h_cont, -base_n // 2)[-base_n // 2]
    h_bin = (h_cont >= thresh).astype(float)
    # enforce exact count (rare tie case)
    cur_ones = int(h_bin.sum())
    if cur_ones > base_n // 2:
        # turn some ones to zero
        ones_idx = np.flatnonzero(h_bin == 1)
        np.random.shuffle(ones_idx)
        for idx in ones_idx[: cur_ones - base_n // 2]:
            h_bin[idx] = 0.0
    elif cur_ones < base_n // 2:
        zeros_idx = np.flatnonzero(h_bin == 0)
        np.random.shuffle(zeros_idx)
        for idx in zeros_idx[: base_n // 2 - cur_ones]:
            h_bin[idx] = 1.0
    h_bin = _project_to_feasible(h_bin, base_n)  # just to be safe
    record_candidate(h_bin)

    if time_left(5.0):
        h_bin_best = _simulated_annealing_binary(h_bin, max_iter=50000, time_left=time_left)
        record_candidate(h_bin_best)

    # ------------------------------------------------------------------
    # If time permits, try a finer grid (512 points)
    # ------------------------------------------------------------------
    if time_left(30.0):
        fine_n = max(512, base_n * 2)
        if fine_n % 2 == 1:
            fine_n += 1
        h_fine = _upscale_h(best_h, fine_n)
        h_fine = _project_to_feasible(h_fine, fine_n)
        # Continuous refinement on fine grid
        alphas_f = [1.0, 3.0, 8.0, 20.0, 40.0, 80.0]
        h_fine = _pgd_smooth(h_fine, alphas_f, step0, 150, time_left)
        record_candidate(h_fine)
        if time_left(3.0):
            h_opt_f, _ = _solve_h(h_fine, max_iter=400)
            record_candidate(h_opt_f)
        # binary SA on fine grid (derived from continuous)
        thresh = np.partition(h_fine, -fine_n // 2)[-fine_n // 2]
        h_bin_f = (h_fine >= thresh).astype(float)
        cur_ones = int(h_bin_f.sum())
        if cur_ones > fine_n // 2:
            ones_idx = np.flatnonzero(h_bin_f == 1)
            np.random.shuffle(ones_idx)
            for idx in ones_idx[: cur_ones - fine_n // 2]:
                h_bin_f[idx] = 0.0
        elif cur_ones < fine_n // 2:
            zeros_idx = np.flatnonzero(h_bin_f == 0)
            np.random.shuffle(zeros_idx)
            for idx in zeros_idx[: fine_n // 2 - cur_ones]:
                h_bin_f[idx] = 1.0
        h_bin_f = _project_to_feasible(h_bin_f, fine_n)
        record_candidate(h_bin_f)
        if time_left(10.0):
            h_bin_f_best = _simulated_annealing_binary(h_bin_f, max_iter=80000, time_left=time_left)
            record_candidate(h_bin_f_best)

    # ------------------------------------------------------------------
    # Final safety check
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback uniform vector
        n = 256
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    else:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
