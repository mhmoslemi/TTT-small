```
def _c5_from_h(h):
    n = h.size
    # Zero pad to next power of two
    size = 1 << ((2*n-1).bit_length())
    # compute convolution of h and (1-h)[::-1]
    g_rev = (1.0 - h)[::-1]
    fft_h = np.fft.rfft(h, size)
    fft_g = np.fft.rfft(g_rev, size)
    conv = np.fft.irfft(fft_h * fft_g, size)
    # take first 2n-1 entries
    max_corr = np.max(conv[:2*n-1])
    return max_corr * (2.0 / n)
```

import time
import numpy as np
from scipy import optimize

# ---------- projection onto box∩simplex ----------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# ---------- C5 evaluation (FFT version) ----------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    g_rev = (1.0 - h)[::-1]
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft(g_rev, size)
    conv = np.fft.irfft(fh * fg, size)
    max_corr = np.max(conv[: 2 * n - 1])
    return max_corr * (2.0 / n)


# ---------- constraints for SLSQP ----------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


# ---------- local optimisation (SLSQP) ----------
def _solve_h(start_h: np.ndarray, max_iter: int = 400):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


# ---------- simple tent shape ----------
def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


# ---------- plateau family (linear ramps + flat top) ----------
def _plateau_h(n: int, a: float) -> np.ndarray:
    # a in (0,1]; width of each ramp
    if a <= 0.0:
        a = 1e-6
    H = 1.0 / (2.0 - a)  # height of plateau
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.empty(n)
    left = xs <= a
    right = xs >= 2.0 - a
    mid = (~left) & (~right)
    h[left] = (H / a) * xs[left]
    h[mid] = H
    h[right] = H * (2.0 - xs[right]) / a
    h = np.clip(h, 0.0, 1.0)
    # enforce exact integral on the discrete grid
    return _project_to_simplex_box(h, n / 2.0)


def _search_plateau(n: int):
    """coarse‑to‑fine search of the best a in the plateau family."""
    best_c5 = float("inf")
    best_a = 0.7
    best_h = None
    # coarse sweep
    for a in np.linspace(0.5, 0.99, 50):
        h = _plateau_h(n, a)
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5, best_a, best_h = c5, a, h
    # ternary refinement around best_a
    lo = max(0.5, best_a - 0.05)
    hi = min(0.99, best_a + 0.05)
    for _ in range(30):
        a1 = lo + (hi - lo) / 3.0
        a2 = hi - (hi - lo) / 3.0
        c5_1 = _c5_from_h(_plateau_h(n, a1))
        c5_2 = _c5_from_h(_plateau_h(n, a2))
        if c5_1 < c5_2:
            hi, best_c5, best_a, best_h = a2, c5_1, a1, _plateau_h(n, a1)
        else:
            lo, best_c5, best_a, best_h = a1, c5_2, a2, _plateau_h(n, a2)
    a_opt = (lo + hi) / 2.0
    h_opt = _plateau_h(n, a_opt)
    final_c5 = _c5_from_h(h_opt)
    return h_opt, final_c5, a_opt


# ---------- simulated annealing on full vector ----------
def _simulated_annealing(start_h: np.ndarray, time_budget: float):
    """simple two‑point SA that preserves the sum exactly."""
    n = start_h.size
    best_h = start_h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = start_h.copy()
    cur_c5 = best_c5
    start = time.time()
    T0 = 1e-3
    Tmin = 1e-8
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / time_budget) + Tmin
        # pick i with room to increase, j with room to decrease
        i = np.random.randint(n)
        if cur_h[i] >= 1.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if cur_h[j] <= 0.0:
            continue
        max_delta = min(1.0 - cur_h[i], cur_h[j])
        if max_delta <= 0.0:
            continue
        # step size shrinks with temperature
        delta = max_delta * np.random.rand() * (0.5 + 0.5 * T / T0)
        if delta <= 0.0:
            continue
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c5 = _c5_from_h(new_h)
        if new_c5 < cur_c5 or np.random.rand() < np.exp((cur_c5 - new_c5) / T):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
    return best_h, best_c5
