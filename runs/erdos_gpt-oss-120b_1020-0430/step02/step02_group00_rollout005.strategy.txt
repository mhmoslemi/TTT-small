import numpy as np
from scipy.optimize import minimize
import time

def project_to_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Project to [0,1] and adjust for sum=n/2"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n/2.0
    if np.abs(s - target) < 1e-6:
        return h
    # Waterfilling-like adjustment
    diff = target - s
    if abs(diff) < 1:
        # Simple reduction/increase if small
        if diff > 0:
            idx = np.where(h > 0.5)[0]
            if len(idx) > 0:
                add = diff / len(idx)
                h[idx] += add
            else:
                # Cannot add, scale only active part
                usable = n - np.count_nonzero(h == 0)
                if usable > 0:
                    add = diff / usable
                    h += add
        else:
            idx = np.where(h < 0.5)[0]
            if len(idx) > 0:
                sub = -diff / len(idx)
                h[idx] -= sub
    else:
        # Larger adjustment needed
        if diff > 0:
            # Reduce high values
            idx = np.where(h > 0.5)[0]
            if len(idx) > 0:
                indices = np.vstack([idx]*int(diff*2+1))
                selected = np.sort(np.column_stack(indices)[np.random.random(len(indices)) < 0.4])
                n_sel = min(50, len(selected))
                if n_sel > 0:
                    selected = selected[:n_sel]
                    addup = 0.5 - h[selected]
                    max_add = np.max(addup)
                    norm_add = diff / np.sum(addup)
                    h[selected] += addup * norm_add
                    h = project_to_feasible(h, n)
        else:
            # Increase low values
            idx = np.where(h < 0.5)[0]
            if len(idx) > 0:
                indices = np.vstack([idx]*int(-diff*2+1))
                selected = np.sort(np.column_stack(indices)[np.random.random(len(indices)) < 0.4])
                n_sel = min(50, len(selected))
                if n_sel > 0:
                    selected = selected[:n_sel]
                    subdown = 0.5 - h[selected]
                    max_sub = np.max(subdown)
                    norm_sub = -diff / np.sum(subdown)
                    h[selected] -= subdown * norm_sub
                    h = project_to_feasible(h, n)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if np.abs(s - target) < 1e-8:
        return h
    # Last resort: scale
    if s > 0:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

                h0 = np.concatenate([np.ones(128), np.zeros(128)])[:n]
                np.random.shuffle(h0)
                h0 = project_to_feasible(h0, n)
            # Give SLSQP options to escape local minima
            cons = ({'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n/2})
            bounds = [(0, 1)]*n
            res = minimize(
                lambda h_vec: c5_from_h(np.clip(np.array(h_vec), 0, 1), n),
                h0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'disp': False, 'ftol': 1e-8})
            if res.success:
                h_opt = res.x
                best_c5 = c5_from_h(h_opt, n) if best_c5 is None else min(best_c5, c5_from_h(h_opt, n))
            else:
                # Use coordinate solver
                h_opt = coordinate_optimize(h0)
                best_c5 = c5_from_h(h_opt, n) if best_c5 is None else min(best_c5, c5_from_h(h_opt, n))
            best_h = h_opt.copy()
            best_n = n
        except:
            # Fallback
            h_rand = np.concatenate([np.ones(128), np.zeros(128)])[:n]
            np.random.shuffle(h_rand)
            h_rand = project_to_feasible(h_rand, n)
            best_h = h_rand
            best_c5 = c5_from_h(best_h, n)
        if time_left() < 30:
            # Early termination if time runs out
            break
    return best_h.tolist(), float(best_c5), int(best_n)
