def projected_gradient_descent(h0, max_iters, step0, tol, beta_schedule):
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_from_h(h)
    for it in range(max_iters):
        # compute corr
        c = np.correlate(h, 1.0 - h, mode='full')
        maxc = c.max()
        # active indices:
        active_idx = np.where(c >= maxc - 1e-12)[0] # tolerance maybe 1e-12?

        # compute gradient as average of grads for each active shift
        grad = np.zeros_like(h)
        for idx in active_idx:
            shift = idx - (len(h) - 1)
            # compute gradient for this shift:
            # compute term1 and term2 as described
            if shift >= 0:
                s = shift
                # term1: (1 - h[j+s]) for j < n-s
                grad[:len(h)-s] += (1 - h[s:])
                # term2: -h[j-s] for j >= s
                grad[s:] += -h[:len(h)-s]
            else:
                s = -shift
                # shift negative
                # gradient: -h[j+s] for j <= n-1-s (j < n-s)
                grad[:len(h)-s] += -h[s:]
                # + (1 - h[j-s]) for j >= s
                grad[s:] += (1 - h[:len(h)-s])
        grad /= len(active_idx)  # average
        # step size schedule: maybe decreasing step0 / sqrt(it+1) or something
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * grad
        # project onto feasible region
        h_new = _project_to_simplex_box(h_new, len(h)/2.0)
        # evaluate bound
        new_val = _c5_from_h(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        # early stop if no improvement
        if np.linalg.norm(h_new - h) < tol:
            break
        h = h_new
    return best_h, best_val

import time
import numpy as np

# -------------------------------------------------------------
# Projection onto [0,1]^n intersect sum(x)=target
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# -------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

    best_val = _c5_from_h(h)
    val = best_val

    step = 0.5
    min_step = 1e-7
    patience = 30
    no_improve = 0

    for it in range(max_iters):
        if time_left() < 0.8:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)

        grad = np.zeros_like(h)
        if shift >= 0:
            s = shift
            grad[: n - s] = 1.0 - h[s:]
            grad[s:] += -h[: n - s]
        else:
            s = -shift
            grad[: n - s] = -h[s:]
            grad[s:] += 1.0 - h[: n - s]

        h_new = h - step * grad
        h_new = _project_to_simplex_box(h_new, n / 2.0)
        new_val = _c5_from_h(h_new)

        if new_val < val:
            h = h_new
            val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            step = min(step * 1.07, 1.0)
        else:
            step *= 0.5
            no_improve += 1

        if step < min_step or no_improve >= patience:
            break

    # ---------------------------------------------------------
    # Simple pairwise swap neighbourhood search (preserves sum)
    swaps = 400
    for _ in range(swaps):
        if time_left() < 0.4:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        max_delta = min(1.0 - hi, hj)
        min_delta = -min(hi, 1.0 - hj)
        if max_delta <= min_delta:
            continue
        delta = (np.random.rand() * (max_delta - min_delta) + min_delta)
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand_val = _c5_from_h(cand)
        if cand_val < best_val:
            best_h = cand
            best_val = cand_val

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # initial construction if supplied
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate resolutions (even numbers, ≤ 500)
    candidates = []
    if init_h is not None:
        candidates.append(init_h.size)
    for n in (96, 128, 160, 192, 224, 256, 320, 384, 448):
        if n % 2 == 0:
            candidates.append(n)
    candidates = sorted(set(candidates))

    for n in candidates:
        if time_left() < 6.0:
            break

        # base shape for this resolution
        if init_h is not None and init_h.size == n:
            base = init_h.copy()
        elif best_h is not None:
            base = _upscale_h(best_h, n)
        else:
            base = _tent_h(n)

        base = _project_to_simplex_box(base, n / 2.0)

        # deterministic refinement
        max_iters = 2000
        refined_h, refined_c5 = _refine_h(base, max_iters, time_left)

        if refined_c5 < best_c5:
            best_h, best_c5, best_n = refined_h, refined_c5, n
            if best_c5 < target:
                break

        # stochastic restarts
        sigma = 0.22
        restarts = 0
        while time_left() > 6.0 and restarts < 8:
            start = base + sigma * np.random.randn(n)
            start = _project_to_simplex_box(start, n / 2.0)
            h_r, c5_r = _refine_h(start, max_iters, time_left)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r, c5_r, n
                if best_c5 < target:
                    break
            sigma *= 0.84
            restarts += 1

        if best_c5 < target:
            break

    # final polish if time permits
    if best_h is not None and time_left() > 5.0:
        best_h, best_c5 = _refine_h(best_h, 3000, time_left)

    # fallback
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
