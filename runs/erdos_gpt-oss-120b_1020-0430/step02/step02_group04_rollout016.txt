import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the box [0,1] with exact sum = target
def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


# ----------------------------------------------------------------------
# Random feasible vector (0<=h_i<=1, sum = n/2)
def random_feasible(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


# ----------------------------------------------------------------------
# Simple pairwise perturbation (conserves sum)
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ----------------------------------------------------------------------
# Hill‑climb local search
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
# Linear (full) correlation via FFT (real inputs)
def linear_correlation(a, b):
    n, m = a.size, b.size
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


def raw_corr(h):
    return linear_correlation(h, 1.0 - h)


def compute_c5(h):
    n = h.size
    corr = raw_corr(h)
    return np.max(corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Soft‑max surrogate for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = raw_corr(x)
    bc = CURRENT_BETA * corr * (2.0 / x.size)
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ----------------------------------------------------------------------
# Gradient descent on the smooth max surrogate
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = raw_corr(h)
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()
            # convolution for weighted shifted sums
            conv_plus = np.convolve(w, h, mode="full")
            conv_rev = np.convolve(w[::-1], h, mode="full")
            off = n - 1
            sum_w_h_plus = conv_rev[off : off + n]      # Σ w_k h_{j+k}
            sum_w_h_minus = conv_plus[off : off + n]    # Σ w_k h_{j-k}
            g_raw = 1.0 - (sum_w_h_plus + sum_w_h_minus)
            grad = (2.0 / n) * g_raw
            h_new = h - step * grad
            h_new = project_onto_box_sum(h_new, n / 2.0)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            step *= 0.9995
    return best_h, best_c


# ----------------------------------------------------------------------
# Gradient descent focusing on the worst (max) shift
def worst_shift_descent(h0, steps=2000, step0=0.5, rng=None):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for _ in range(steps):
        corr = raw_corr(h)
        idx = np.argmax(corr)
        k = idx - (n - 1)
        if k >= 0:
            h_plus = np.concatenate([h[k:], np.zeros(k, dtype=h.dtype)])
            h_minus = np.concatenate([np.zeros(k, dtype=h.dtype), h[: n - k]])
        else:
            p = -k
            h_plus = np.concatenate([np.zeros(p, dtype=h.dtype), h[: n - p]])
            h_minus = np.concatenate([h[p:], np.zeros(p, dtype=h.dtype)])
        grad = (2.0 / n) * (1.0 - (h_plus + h_minus))
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        step *= 0.9995
        if rng is not None and rng.random() < 0.001:
            step = step0 * rng.random()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    if time.time() - start < budget_s * 0.2:

    while time.time() - start < budget_s * 0.70:

        if rng.random() < 0.2:

        # smooth‑max gradient descent
        h_opt, c_opt = grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng)

        h_opt, c_opt = worst_shift_descent(h_opt, steps=2500, step0=0.5, rng=rng)

                pass
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- up‑sampling and final polishing ----
    if time.time() - start < budget_s * 0.95:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size

            best_h, best_c, n = h_opt, c_opt, n_up

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
