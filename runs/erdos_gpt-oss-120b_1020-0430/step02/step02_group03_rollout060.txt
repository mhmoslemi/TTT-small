def propose_move(h, rng, sigma=0.1):

import time, numpy as np, math
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # iterative balancing
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    # perform a number of random transfers to introduce randomness
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction to sum
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time
import math
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # ------------------------------------------------------------------
    # Choose dimension and initial candidates
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 200
        h0 = random_feasible_vector(n, rng)

    candidates = [h0]
    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # Quick SLSQP warm‑up (small budget)
    # ------------------------------------------------------------------
    if time.time() - start_time < total_budget * 0.10:
        for beta in (5, 20, 80):
            try:
                best_h = slsqp_optimize(best_h, beta, max_iter=200)
                best_c = compute_c5(best_h)
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Simulated‑annealing phase (major part of the budget)
    # ------------------------------------------------------------------
    sa_end = start_time + total_budget * 0.80
    sigma = 0.4
    start_T = 0.02
    final_T = 1e-6

    while time.time() < sa_end:
        # start point for this SA block
        if rng.random() < 0.6:
            cur_h = best_h.copy()
        else:
            cur_h = random_feasible_vector(n, rng)

        # evaluate current state
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        worst_idx = int(np.argmax(corr))
        cur_c = corr[worst_idx] * (2.0 / n)

        remaining = sa_end - time.time()
        if remaining <= 0:
            break
        # not more than 100 000 inner steps per block
        max_iters = int(min(100_000, max(1, remaining * 1500)))

        for it in range(max_iters):
            # temperature schedule
            T = start_T * (final_T / start_T) ** (it / max_iters)

            # Decide move type
            if rng.random() < 0.7:
                # ---- targeted move using the worst shift ----
                shift = worst_idx - (n - 1)
                weight = np.zeros(n)
                if shift >= 0:
                    weight[: n - shift] = 1.0 - cur_h[shift:]
                else:
                    weight[-shift:] = 1.0 - cur_h[: n + shift]

                i_cand = np.where(cur_h > 1e-12)[0]
                j_cand = np.where(cur_h < 1.0 - 1e-12)[0]
                if i_cand.size and j_cand.size:
                    wi = weight[i_cand]
                    wj = weight[j_cand]
                    i = i_cand[int(np.argmax(wi))]
                    j = j_cand[int(np.argmin(wj))]
                    max_eps = min(cur_h[i], 1.0 - cur_h[j])
                    if max_eps > 0.0:
                        eps = rng.uniform(0.0, max_eps) * 0.5
                    else:
                        continue
                else:
                    # fall back to random move if no candidates
                    i = rng.integers(n)
                    j = rng.integers(n)
                    while j == i:
                        j = rng.integers(n)
                    max_eps = min(cur_h[i], 1.0 - cur_h[j])
                    if max_eps <= 0.0:
                        continue
                    eps = rng.normal(0.0, sigma * max_eps)
                    eps = max(-max_eps, min(eps, max_eps))
            else:
                # ---- pure random transfer move ----
                i = rng.integers(n)
                j = rng.integers(n)
                while j == i:
                    j = rng.integers(n)
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps <= 0.0:
                    continue
                eps = rng.normal(0.0, sigma * max_eps)
                eps = max(-max_eps, min(eps, max_eps))

            # apply move (preserves the sum)
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps

            # evaluate new state
            new_c = compute_c5(new_h)
            delta = new_c - cur_c
            if delta < 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
                cur_h, cur_c = new_h, new_c
                if new_c < best_c:
                    best_c, best_h = new_c, new_h.copy()

            # occasionally refresh worst shift information
            if (it + 1) % 5000 == 0:
                corr = np.correlate(cur_h, 1 - cur_h, mode="full")
                worst_idx = int(np.argmax(corr))

        # cheap SLSQP refinement after each SA block (if time permits)
        if time.time() < sa_end - 5:
            try:
                best_h = slsqp_optimize(best_h, 100, max_iter=250)
                best_c = compute_c5(best_h)
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Final high‑β SLSQP refinement
    # ------------------------------------------------------------------
    for beta in (200, 800, 2000):
        if time.time() - start_time > total_budget * 0.97:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=300)
            refined_c = compute_c5(refined)
            if refined_c < best_c:
                best_c, best_h = refined_c, refined.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
