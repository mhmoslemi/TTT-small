def delta_f_swap(i, j, H):
    n = len(H)
    size = 2*n - 1
    # create zeros array
    delta = np.zeros(size, dtype=int)
    # compute contributions for j
    # s index will be offset by (n-1)
    # For convenience, we let offsets = np.arange(-(n-1), n)
    offsets = np.arange(-(n-1), n, dtype=int)
    # For j contributions +:
    idx_plus = j + offsets
    mask_plus = (0 <= idx_plus) & (idx_plus < n)
    delta[mask_plus] += H[idx_plus[mask_plus]]
    idx_minus = j - offsets
    mask_minus = (0 <= idx_minus) & (idx_minus < n)
    delta[mask_minus] += H[idx_minus[mask_minus]]
    # For i contributions -:
    idx_plus_i = i + offsets
    mask_plus_i = (0 <= idx_plus_i) & (idx_plus_i < n)
    delta[mask_plus_i] -= H[idx_plus_i[mask_plus_i]]
    idx_minus_i = i - offsets
    mask_minus_i = (0 <= idx_minus_i) & (idx_minus_i < n)
    delta[mask_minus_i] -= H[idx_minus_i[mask_minus_i]]
    return delta

import time, numpy as np

def compute_c5(h):
    """Discrete C5 for binary or float step function."""
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / h.size))

def random_binary_vector(n, rng):
    """Uniform random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1
    return h

def random_swap(H, rng):
    """Return indices i (H[i]==1) and j (H[j]==0) for a swap."""
    n = H.shape[0]
    i = rng.integers(n)
    while H[i] != 1:
        i = rng.integers(n)
    j = rng.integers(n)
    while H[j] != 0:
        j = rng.integers(n)
    return i, j

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------- initialise ------------------------------------------------
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n = h_init.size
        # binary projection
        H = (h_init >= 0.5).astype(np.int8)
        target = n // 2
        diff = int(H.sum() - target)
        if diff > 0:
            ones = np.where(H == 1)[0]
            to_flip = rng.choice(ones, diff, replace=False)
            H[to_flip] = 0
        elif diff < 0:
            zeros = np.where(H == 0)[0]
            to_flip = rng.choice(zeros, -diff, replace=False)
            H[to_flip] = 1
    else:
        n = 360
        H = random_binary_vector(n, rng)

    cur_H = H.copy()
    cur_c = compute_c5(cur_H)
    best_H = cur_H.copy()
    best_c = cur_c

    # ------------------- simulated annealing ------------------------------------
    temp = 0.05
    decay = 0.99995
    no_improve = 0
    # keep some slack for final check
    limit = start + budget_s * 0.95

    while time.time() < limit:
        i, j = random_swap(cur_H, rng)
        # perform tentative swap
        cur_H[i] = 0
        cur_H[j] = 1
        new_c = compute_c5(cur_H)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-delta / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_H = cur_H.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert swap
            cur_H[i] = 1
            cur_H[j] = 0
            no_improve += 1

        temp = max(temp * decay, 1e-7)

        # occasional diversification
        if no_improve > 20000:
            # restart from current best with a few random swaps
            cur_H = best_H.copy()
            cur_c = best_c
            for _ in range(15):
                ii, jj = random_swap(cur_H, rng)
                cur_H[ii] = 0
                cur_H[jj] = 1
            cur_c = compute_c5(cur_H)
            no_improve = 0

    final_c = compute_c5(best_H)
    return best_H.tolist(), float(final_c), int(n)

def targeted_swap(H, s, rng):
    n = H.shape[0]
    if s >= 0:
        i_range = np.arange(0, n - s)
        # bad = positions where H[i]==1 and H[i+s]==0

        good = i_range[mask_good]

def upsample_and_refine(h_int, factor, rng, time_deadline):
    n = len(h_int)
    n_up = n * factor
    x_old = np.linspace(0., 2., n, endpoint=False)
    x_new = np.linspace(0., 2., n_up, endpoint=False)
    # linear interpolation
    h_up = np.interp(x_new, x_old, h_int.astype(float))
    # enforce sum exactly (n_up/2)
    target = n_up / 2.0
    # adjust
    diff = target - h_up.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n_up)
        if diff > 0:
            add = min(diff, 1.0 - h_up[i])
        else:
            add = max(diff, -h_up[i])
        h_up[i] += add
        diff = target - h_up.sum()
    # simple hill climb using epsilon transfers
    cur = h_up.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    it = 0
    while time.time() < time_deadline and it < 2000:
        i = rng.integers(0, n_up)
        j = rng.integers(0, n_up)
        if i == j: continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0: continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.3
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_c = cand_c
            best = cand.copy()
        if cand_c < cur_c:
            cur = cand
            cur_c = cand_c
        it += 1
    return best, best_c

import time
import numpy as np

def compute_c5(arr):
    """Exact C5 for a step function (binary or float) on [0,2]."""
    corr = np.correlate(arr, 1 - arr, mode="full")
    return float(np.max(corr) * (2.0 / arr.size))

def random_binary_vector(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1
    return h

def random_swap(H, rng):
    """Pick i with H[i]==1 and j with H[j]==0 for a swap."""
    n = H.shape[0]
    i = rng.integers(n)
    while H[i] != 1:
        i = rng.integers(n)
    j = rng.integers(n)
    while H[j] != 0:
        j = rng.integers(n)
    return i, j

def targeted_swap(H, shift, rng):
    """Swap a 1‑0 pair that hurts the worst shift, if possible."""
    n = H.shape[0]
    if shift >= 0:
        i0 = np.arange(0, n - shift)
        i1 = i0 + shift
        bad = i0[(H[i0] == 1) & (H[i1] == 0)]
        good = i0[(H[i0] == 0) & (H[i1] == 1)]
    else:
        d = -shift
        i0 = np.arange(d, n)
        i1 = i0 - d
        bad = i0[(H[i0] == 1) & (H[i1] == 0)]
        good = i0[(H[i0] == 0) & (H[i1] == 1)]
    if bad.size == 0 or good.size == 0:
        return None, None
    i = rng.choice(bad)
    j = rng.choice(good)
    if i == j:
        return None, None
    return i, j
