def optimize_h(initial_h, n, time_budget):
    start = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    # Force h feasible
    h = _project_to_feasible(h, n)
    it = 0
    step0 = 0.4 # maybe.
    last_improve_iter = 0
    while time.time() - start < time_budget:
        it += 1
        # compute gradient from max shift(s)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # Determine positions with max (allow small tolerance to include near ties)
        # Use tolerance = 1e-12 * maybe; just pick indices where equal.
        # Indices: idxs = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        # We'll just pick one index randomly among max.
        max_idx = np.flatnonzero(corr == max_corr)[0] # chooses first occurrence.
        shift = max_idx - (n - 1)  # shift value

        # compute gradient
        grad = np.full(n, 1.0)
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]

        # step size
        step = step0 / np.sqrt(it)
        # update
        h_new = h - step * grad
        # project
        h_new = _project_to_feasible(h_new, n)
        h = h_new

        # evaluate
        c5 = _c5_from_h(h)
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h.copy()
            last_improve_iter = it

        # random restart condition

            h = best_h + sigma * np.random.randn(n)
            h = _project_to_feasible(h, n)
            last_improve_iter = it
    return best_h, best_c5

def _project_to_feasible(h, n, target=None):
    if target is None: target = n/2

    # Clip initially.
    h = np.clip(h, 0.0, 1.0)
    # If sum already target within tolerance, done.
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # define function f(lambda) = sum(clip(h - lambda, 0, 1)) - target.
    # Find lambda (could be positive or negative). Use bisection on range [-1, 1] maybe broader.
    lo, hi = -1.0, 1.0
    # Expand bounds until f(lo) and f(hi) have opposite signs.
    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target

    # Ensure signs:
    f_lo = f(lo)
    f_hi = f(hi)
    # Adjust bounds if needed
    while f_lo > 0:
        lo -= 1.0
        f_lo = f(lo)
    while f_hi < 0:
        hi += 1.0
        f_hi = f(hi)
    # Bisection
    for _ in range(50):
        mid = (lo + hi) / 2.0
        fm = f(mid)
        if fm > 0:
            lo = mid
        else:
            hi = mid
        if abs(fm) < 1e-12:
            break
    h_proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    return h_proj

def _optimize_subgradient(initial_h, n, time_budget):
    start = time.time()
    h = _project_to_feasible(initial_h, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    # set parameters
    step0 = 0.6
    it = 0
    last_improve_it = 0
    # Determine tolerance for multiple max shifts
    max_shift_tol = 1e-12
    while time.time() - start < time_budget:
        it += 1
        # correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # Indices within tolerance
        idxs = np.where(corr >= max_corr - max_shift_tol)[0]
        # compute average gradient
        grad = np.zeros(n)
        for idx in idxs:
            shift = idx - (n - 1)
            # accumulate gradient as per one shift
            g_shift = np.full(n, 1.0)
            if shift >= 0:
                g_shift[:n - shift] -= h[shift:]
                g_shift[shift:] -= h[:n - shift]
            else:
                s = -shift
                g_shift[s:] -= h[:n - s]
                g_shift[:n - s] -= h[s:]
            grad += g_shift
        # average
        grad /= len(idxs)

        step = step0 / np.sqrt(it)
        # update
        h_new = h - step * grad
        # project
        h_new = _project_to_feasible(h_new, n)

        c5 = _c5_from_h(h)

import time
import numpy as np

# ---------------------------------------------------------
# Utility functions
# ---------------------------------------------------------

def _c5_from_h(h: np.ndarray) -> float:
    """Return the discretised C5 value for a step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _project_to_feasible(v: np.ndarray, n: int, target: float = None) -> np.ndarray:
    """
    Project a vector onto the set {0≤h_i≤1, sum h_i = target}.
    Uses a simple bisection on the Lagrange multiplier λ for the equality.
    """
    if target is None:
        target = n / 2.0
    # Clip to box first
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v

    # monotone decreasing function f(λ) = sum clip(v-λ) - target
    def f(lam):
        return (np.clip(v - lam, 0.0, 1.0).sum() - target)

    lo, hi = -1.0, 1.0
    # expand bounds until signs differ
    while f(lo) > 0:
        lo -= 1.0
    while f(hi) < 0:
        hi += 1.0

    for _ in range(60):
        mid = (lo + hi) / 2.0
        fm = f(mid)
        if fm > 0:
            lo = mid
        else:
            hi = mid
        if abs(fm) < 1e-12:
            break

    lam = (lo + hi) / 2.0
    proj = np.clip(v - lam, 0.0, 1.0)

    # tiny correction to hit the target exactly
    residual = target - proj.sum()
    if abs(residual) > 1e-12:
        free = np.where((proj > 0.0) & (proj < 1.0))[0]
        if free.size > 0:
            proj[free] += residual / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _local_refine(h: np.ndarray, attempts: int = 15, delta: float = 0.015) -> np.ndarray:
    """Simple stochastic hill‑climbing around h."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


# ---------------------------------------------------------
# Core optimisation routine (sub‑gradient descent)
# ---------------------------------------------------------

def _optimize_subgradient(init_h: np.ndarray, n: int, time_budget: float):
    """
    Sub‑gradient descent on the max‑overlap objective.
    Returns (best_h, best_c5) found within the supplied time budget.
    """
    start = time.time()
    h = _project_to_feasible(init_h, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    step0 = 0.6          # initial step size
    momentum = 0.9       # heavy‑ball momentum
    direction = np.zeros(n)

    it = 0
    last_improve = 0
    tol_shift = 1e-12    # tolerance for treating equal shifts

    while time.time() - start < time_budget:
        it += 1

        # correlation for all shifts
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        # all shifts attaining (or almost attaining) the maximum
        idxs = np.where(corr >= max_corr - tol_shift)[0]

        grad = np.zeros(n)
        for idx in idxs:
            shift = idx - (n - 1)
            g = np.full(n, 1.0)
            if shift >= 0:
                g[:n - shift] -= h[shift:]
                g[shift:] -= h[:n - shift]
            else:
                s = -shift
                g[s:] -= h[:n - s]
                g[:n - s] -= h[s:]
            grad += g
        grad /= len(idxs)

        # heavy‑ball update
        direction = momentum * direction + grad
        step = step0 / np.sqrt(it)
        h_new = h - step * direction
        h_new = _project_to_feasible(h_new, n)

        h = h_new
        c5 = _c5_from_h(h)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            last_improve = it

        # restart if stuck for a while
        if it - last_improve > 5000:
            sigma = 0.02
            h = best_h + sigma * np.random.randn(n)
            h = _project_to_feasible(h, n)
            direction[:] = 0.0
            last_improve = it

    return best_h, best_c5

    """

    # -----------------------------------------------------------------
