def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
    # If still off, maybe minimal numeric error; ignore
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions (feasibility, objective, local search)
# ------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Shift mass among coordinates to make sum(h) == target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # simple random walk adjustment
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible step function of length n."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h, h = c, cand.copy(), cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# ------------------------------------------------------------
# SLSQP surrogate (smooth approximation of max)
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    rng = np.random.default_rng()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ------------------------------------------------------------
# Genetic algorithm core
# ------------------------------------------------------------
def tournament(pop, scores, rng, k=2):
    idx = rng.integers(0, len(pop), size=k)
    best_i = idx[0]
    best_s = scores[best_i]
    for i in idx[1:]:
        if scores[i] < best_s:
            best_i, best_s = i, scores[i]
    return pop[best_i]

def crossover(p1, p2, rng):
    mask = rng.random(p1.size) < 0.5
    return np.where(mask, p1, p2)

def mutate(h, rng, scale=0.3):
    return perturb_h(h, rng, scale=scale)

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- Determine problem size ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = enforce_sum(np.full(n, 0.5), n / 2.0, rng)

    target_sum = n / 2.0
    dx = 2.0 / n

    # ---- Build initial population ----
    pop_size = 80
    population = []
    scores = []

    # 1) include the supplied construction if any
    if "initial_h_values" in globals():
        population.append(h0)
        scores.append(compute_c5(h0))

    # 2) deterministic sinusoidal / linear patterns
    xs = (np.arange(n) + 0.5) * dx
    for f in range(1, 5):
        for kind in ("sin", "cos"):
            if kind == "sin":
                h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
            else:
                h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target_sum, rng)
            population.append(h)
            scores.append(compute_c5(h))

    # linear ramps
    for h in (xs / 2.0, 1.0 - xs / 2.0):
        h = enforce_sum(np.clip(h, 0.0, 1.0), target_sum, rng)
        population.append(h)
        scores.append(compute_c5(h))

    # random members to fill the rest
    while len(population) < pop_size:
        h = random_feasible_vector(n, rng)
        population.append(h)
        scores.append(compute_c5(h))

    # keep only the best `pop_size` individuals
    idx = np.argsort(scores)[:pop_size]
    population = [population[i] for i in idx]
    scores = [scores[i] for i in idx]

    best_h = population[0]
    best_c = scores[0]

    # ------------------------------------------------------------
    # Genetic algorithm loop (time‑controlled)
    # ------------------------------------------------------------
    mutation_rate = 0.15
    mutation_scale = 0.3
    max_ga_time = budget_s * 0.65
    while time.time() - start < max_ga_time:
        # generate offspring
        offspring = []
        for _ in range(pop_size):
            p1 = tournament(population, scores, rng, k=2)
            p2 = tournament(population, scores, rng, k=2)
            child = crossover(p1, p2, rng)
            if rng.random() < mutation_rate:
                child = mutate(child, rng, scale=mutation_scale)
            child = enforce_sum(child, target_sum, rng)
            offspring.append(child)

        # evaluate offspring
        off_scores = [compute_c5(h) for h in offspring]

        # elitist (μ+λ) selection
        combined = population + offspring
        combined_scores = scores + off_scores
        idx = np.argsort(combined_scores)[:pop_size]
        population = [combined[i] for i in idx]
        scores = [combined_scores[i] for i in idx]

        # update global best
        if scores[0] < best_c:
            best_c = scores[0]
            best_h = population[0].copy()

    # ------------------------------------------------------------
    # Local refinement (hill climbing)
    # ------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        # allocate most of the remaining budget to hill climbing
        hc_iters = min(8000, int(remaining * 1000))  # very rough scaling
        best_h, best_c = hill_climb(best_h, rng, max_iters=hc_iters, start_step=0.5)

    # ------------------------------------------------------------
    # Optional surrogate SLSQP refinement (if time permits)
    # ------------------------------------------------------------
    beta_list = [5.0, 20.0, 80.0, 200.0]
    for beta in beta_list:
        if time.time() - start > budget_s * 0.96:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=400)
        except Exception:
            continue
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_c = c_ref
            best_h = refined

    # ------------------------------------------------------------
    # Final cleanup and return
    # ------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
