def max_overlap_and_subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    overlaps = corr * (2.0 / n)
    max_val = overlaps.max()
    # find indices with tolerance
    mask = overlaps >= max_val - 1e-12
    # indices offset: s = idx - (n-1)
    idxs = np.where(mask)[0]
    # Initialize grad vector zeros.
    grad = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)
        # compute contributions vectorized:
        # For each j: grad[j] += 1 - (h[j-s] if valid) - (h[j+s] if valid)
        # Use slicing approach
        grad[:] += 1
        # subtract h[j-s]
        if s >= 0:
            grad[s:] -= h[:n-s]
        else:
            grad[:n+s] -= h[-s:]
        # subtract h[j+s]
        if s <= 0:
            grad[:n+s] -= h[-s:]
        else:
            grad[s:] -= h[:n-s]
    # If multiple max shifts, average gradient:
    grad /= len(idxs)
    # Because we want subgradient of max, which is convex combination, average is okay.
    # Multiply by dx factor? The objective max overlaps already includes dx scaling; we want gradient of max
    # of overlaps (including scaling). So gradient should be scaled by dx factor (2/n). Let's compute:
    scaling = 2.0 / n
    grad *= scaling
    return max_val, grad

def smooth_max_descent(h0, beta_seq, steps_per_beta, step0, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    scaling = 2.0 / n
    for beta in beta_seq:
        step = step0
        for _ in range(steps_per_beta):
            # compute overlaps
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * scaling
            # compute softmax
            bc = beta * C
            m = bc.max()
            w = np.exp(bc - m)
            w_sum = w.sum()
            w /= w_sum
            # compute gradient using convolution
            conv_left = np.convolve(w, h, mode='full')[n-1:n-1+n]
            conv_right = np.convolve(w[::-1], h, mode='full')[n-1:n-1+n]
            grad = (1.0 - conv_left - conv_right) * scaling
            # step
            h_new = h - step * grad
            # projection
            h_new = project_onto_box_sum(h_new, n/2.0)
            # enforce symmetry
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = project_onto_box_sum(h_new, n/2.0)
            # compute c5
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                # optionally increase step?
            else:
                # reduce step if no improvement
                step *= 0.99
                # random acceptance occasionally
                if rng.random() < 0.005:
                    h = h_new
    return best_h, best_c

def subgradient_descent(h0, max_iters, step0, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    scaling = 2.0 / n
    for it in range(max_iters):
        # compute corr and overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        C = corr * scaling
        max_val = C.max()
        # compute mask for max within tolerance
        tol = 1e-12
        mask = C >= max_val - tol
        # We need gradient for these shifts combined equally
        # Build w = mask.astype(float) / mask.sum()
        w = mask.astype(float) / float(mask.sum())

        conv_right = np.convolve(w[::-1], h, mode='full')[n-1:n-1+n]

            h = h_new

            step0 *= 0.999
            if rng.random() < 0.005:
                h = h_new
    return best_h, best_c

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project onto [lower,upper] with exact sum = target (box‑sum projection)."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0≤h≤1, Σh=n/2). n must be even."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def perturb_h(h, rng, scale=0.2):
    """Swap a small amount of mass between two positions (keeps sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_local(h_start, rng, max_iters=300, scale=0.2):
    """Simple hill‑climb using pairwise perturbations."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        # keep feasibility
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, n / 2.0)
        new_h = (new_h + new_h[::-1]) * 0.5
        new_h = project_onto_box_sum(new_h, n / 2.0)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
    return best_h, best_c


# ---------- smooth‑max gradient descent ----------
def smooth_max_descent(h0, beta_seq, steps_per_beta, step0, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    scaling = 2.0 / n
    for beta in beta_seq:
        step = step0
        for _ in range(steps_per_beta):
            # overlaps
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * scaling
            # smooth max weights
            bc = beta * C
            m = bc.max()
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # gradient via convolution
            conv_left = np.convolve(w, h, mode="full")[n - 1 : n - 1 + n]
            conv_right = np.convolve(w[::-1], h, mode="full")[n - 1 : n - 1 + n]
            grad = (1.0 - conv_left - conv_right) * scaling
            # step and project
            h_new = h - step * grad
            h_new = project_onto_box_sum(h_new, n / 2.0)
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = project_onto_box_sum(h_new, n / 2.0)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                # keep step size after improvement
            else:
                step *= 0.995
                if rng.random() < 0.004:
                    h = h_new
    return best_h, best_c


# ---------- subgradient descent on the exact max ----------
def subgradient_descent(h0, max_iters, step0, rng):
    """Projected subgradient descent on the true max (averaged over active shifts)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    scaling = 2.0 / n
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        C = corr * scaling
        max_val = C.max()
        # active shifts (within tiny tolerance)
        active = C >= max_val - 1e-12
        w = active.astype(float)
        cnt = w.sum()
        if cnt == 0:
            w = np.ones_like(C) / C.size
        else:
            w /= cnt
        conv_left = np.convolve(w, h, mode="full")[n - 1 : n - 1 + n]
        conv_right = np.convolve(w[::-1], h, mode="full")[n - 1 : n - 1 + n]
        grad = (1.0 - conv_left - conv_right) * scaling
        h_new = h - step0 * grad
        h_new = project_onto_box_sum(h_new, n / 2.0)
        h_new = (h_new + h_new[::-1]) * 0.5
        h_new = project_onto_box_sum(h_new, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step0 *= 0.999
            if rng.random() < 0.003:
                h = h_new
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = budget_s

    # ---- initialise best solution ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        # enforce symmetry
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, target)
        best_h = h0.copy()
    else:
        n = 180  # fallback size
        best_h = random_feasible_symmetric(n, rng)

    best_c = compute_c5(best_h)

    # ---- optimisation loop ----
    # parameters (tuned for speed vs quality)
    beta_seq = [5.0, 20.0, 80.0, 200.0, 500.0]
    steps_per_beta = 220
    subgrad_iters = 250
    hill_iters = 250
    step0_smooth = 0.3
    step0_subgrad = 0.07

    while time.time() - start < time_limit * 0.92:
        # random symmetric start; occasionally blend with current best
        h_start = random_feasible_symmetric(n, rng)
        if rng.random() < 0.15:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)
            h_start = (h_start + h_start[::-1]) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)

        # smooth‑max descent
        h_s, c_s = smooth_max_descent(h_start, beta_seq, steps_per_beta, step0_smooth, rng)

        # subgradient refinement
        h_sub, c_sub = subgradient_descent(h_s, subgrad_iters, step0_subgrad, rng)

        # final local hill‑climb
        h_fin, c_fin = hill_climb_local(h_sub, rng, max_iters=hill_iters, scale=0.15)

        if c_fin + 1e-12 < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

        # occasional up‑sampling attempt if time permits
        if time.time() - start > time_limit * 0.80 and n * 2 <= 300:
            # double resolution
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            # quick refinement at higher resolution
            h_up, c_up = smooth_max_descent(
                h_up,
                beta_seq,
                steps_per_beta // 2,
                step0_smooth * 0.5,
                rng,
            )
            h_up, c_up = subgradient_descent(h_up, subgrad_iters // 2, step0_subgrad * 0.5, rng)
            h_up, c_up = hill_climb_local(h_up, rng, max_iters=hill_iters // 2, scale=0.12)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up.copy()
                n = n_up

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
