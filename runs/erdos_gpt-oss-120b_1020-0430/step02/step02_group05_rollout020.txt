import time, numpy as np, scipy.optimize as opt

def _project_to_simplex_box(v, target):
    # Use sorting method for projection onto simplex with box [0,1]
    # Equivalent to clamp v with lambda such that sum clipped = target.
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # Small residual fix
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_from_h(h):
    n = h.size
    # correlation full length 2n-1
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

def _solve_local(start_h, max_iter=500):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n/2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr  # must be >=0
    def eq(x):
        h = x[:n]
        return np.sum(h) - n/2.0

    cons = [{'type':'ineq','fun':ineq}, {'type':'eq','fun':eq}]

    opt_res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds,
                           constraints=cons,
                           options={'maxiter':max_iter,'ftol':1e-9,'disp':False})
    if not opt_res.success:
        h_opt = h0
    else:
        h_opt = opt_res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n/2.0)
    return h_opt, _c5_from_h(h_opt)
```

import time, math
import numpy as np
from scipy import optimize

def _project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _full_corr(h):
    n = h.size
    # Use direct correlate for small n, FFT for large n
    if n <= 256:
        return np.correlate(h, 1.0 - h, mode='full')
    # FFT based convolution
    N = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(H * G, N)
    return conv[:2*n-1]

def _c5_from_h(h):
    n = h.size
    max_corr = np.max(_full_corr(h))
    return max_corr * (2.0 / n)

def _solve_local(start_h, max_iter=300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n/2.0)
    corr0 = _full_corr(h0)
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - _full_corr(h)

    def eq(x):
        h = x[:n]
        return np.sum(h) - n/2.0

    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]

    res = optimize.minimize(lambda x: x[-1], x0,
                            method='SLSQP',
                            bounds=bounds,
                            constraints=cons,
                            options={'maxiter':max_iter,'ftol':1e-9,'disp':False})
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n/2.0)
    return h_opt, _c5_from_h(h_opt)

def _gradient_descent(h, steps=80, lr=0.15):
    n = h.size
    for _ in range(steps):
        corr = _full_corr(h)
        max_val = np.max(corr)
        # Choose shifts attaining max (within tolerance)
        idxs = np.where(corr >= max_val - 1e-12)[0]
        shifts = idxs - (n-1)
        g = np.zeros(n)
        for k in shifts:
            if k >= 0:
                # first term: + (1 - h_{j+k})
                g[:n-k] += 1.0 - h[k:]
                # second term: - h_{j-k}
                g[k:]   -= h[:n-k]
            else:
                kp = -k
                g[kp:] += 1.0 - h[:n-kp]
                g[:n-kp] -= h[kp:]
        if len(shifts) > 1:
            g /= len(shifts)
        h = h - lr * g
        h = _project_to_simplex_box(h, n/2.0)
        lr *= 0.99
    return h, _c5_from_h(h)

    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _full_corr(h: np.ndarray) -> np.ndarray:
    n = h.size
    if n <= 256:
        return np.correlate(h, 1.0 - h, mode="full")
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(H * G, N)
    return conv[: 2 * n - 1]


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(_full_corr(h)) * (2.0 / n)


def _solve_local(start_h: np.ndarray, max_iter: int = 300):

    h0 = _project_to_simplex_box(start_h, n / 2.0)
    t0 = np.max(_full_corr(h0))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        h = x[:n]
        t = x[-1]
        return t - _full_corr(h)

        h = x[:n]
        return np.sum(h) - n / 2.0

        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
