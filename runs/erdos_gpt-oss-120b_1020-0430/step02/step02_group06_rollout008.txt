def crossover(p1, p2, prob=0.7):
    if np.random.rand() > prob:
        return p1.copy()
    n = len(p1)
    a,b = sorted(np.random.choice(n, 2, replace=False))
    child = np.empty_like(p1)
    child[:a] = p1[:a]
    child[a:b] = p2[a:b]
    child[b:] = p1[b:]
    return child
```

def project_feasible(h, n, tol=1e-9, max_iter=100):
    h = np.clip(h, 0, 1)
    target = n/2
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # indices that can move: if diff>0 we need to reduce sum: can reduce values >0; if diff<0 we need increase values <1.
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0, 1)
    # final residual correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first element not at bound
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1-h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) <= 1e-12:
                break
    return h

import time
import numpy as np

# ----- Basic utilities -------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)

def _tournament(pop, fitness, k=3):
    idx = np.random.choice(len(pop), k, replace=False)
    best = min(idx, key=lambda i: fitness[i])
    return pop[best]


def _crossover(p1: np.ndarray, p2: np.ndarray, prob: float = 0.7):
    if np.random.rand() > prob:
        return p1.copy()
    n = p1.size
    a, b = sorted(np.random.choice(n, 2, replace=False))
    child = np.empty_like(p1)
    child[:a] = p1[:a]
    child[a:b] = p2[a:b]
    child[b:] = p1[b:]
    return child


def _mutate(h: np.ndarray, prob: float = 0.3, scale: float = 0.05):
    if np.random.rand() > prob:
        return h
    n = h.size
    m = max(1, int(0.1 * n))
    idx = np.random.choice(n, m, replace=False)
    h[idx] += scale * np.random.randn(m)
    return h


# ----- Local refinement -------------------------------------------------

def _local_refine(h: np.ndarray, attempts: int = 30, delta: float = 0.01) -> np.ndarray:
    n = h.size
    best = h.copy()
    best_c5 = _c5_from_h(best)
    for _ in range(attempts):
        idx = np.random.choice(n, min(5, n), replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c = _c5_from_h(trial)
        if c < best_c5:
            best, best_c5 = trial, c
    return best


def _make_binary_feasible(h: np.ndarray) -> np.ndarray:
    """Round to 0/1 while keeping exactly n/2 ones."""
    n = h.size
    target = n // 2
    bin_h = (h >= 0.5).astype(float)
    cur = int(bin_h.sum())
    if cur > target:
        ones = np.where(bin_h == 1)[0]
        to_flip = cur - target
        flip_idx = np.random.choice(ones, to_flip, replace=False)
        bin_h[flip_idx] = 0
    elif cur < target:
        zeros = np.where(bin_h == 0)[0]
        to_flip = target - cur
        flip_idx = np.random.choice(zeros, to_flip, replace=False)
        bin_h[flip_idx] = 1
    return bin_h


def _binary_hill_climb(h: np.ndarray, attempts: int = 2000) -> tuple:
    """Swap a 1 with a 0 if it improves C5."""
    n = h.size
    best = h.copy()
    best_c5 = _c5_from_h(best)
    for _ in range(attempts):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        trial = best.copy()
        trial[i] = 0
        trial[j] = 1
        c = _c5_from_h(trial)
        if c < best_c5:
            best, best_c5 = trial, c
    return best, best_c5


# ----- Main optimisation driver -----------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise the Erdős C5 bound. Returns (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # parent construction if provided
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # candidate grid sizes (prefer parent size, then a modest range)
    n_candidates = []
    if parent_h is not None:
        n_candidates.append(parent_h.size)
    for n in (80, 96, 112, 128, 144, 160, 192, 224, 256):
        n_candidates.append(n)
    n_candidates = sorted(set([n for n in n_candidates if n % 2 == 0]))

    for n in n_candidates:
        if time_left() < 8.0:
            break

        # base h for this size
        if parent_h is not None:
            base_h = _upscale_h(parent_h, n) if parent_h.size != n else parent_h.copy()
        else:
            base_h = np.full(n, 0.5)
        base_h = _project_to_feasible(base_h, n)

        # initialise population
        pop = [base_h]
        # a few noisy copies of base
        for _ in range(4):
            h = base_h + 0.05 * np.random.randn(n)
            pop.append(_project_to_feasible(h, n))
        # some random binary individuals
        for _ in range(5):
            bin_h = np.zeros(n)
            bin_h[: n // 2] = 1
            np.random.shuffle(bin_h)
            pop.append(bin_h.astype(float))
        # fill rest randomly
        while len(pop) < 30:
            h = np.random.rand(n)
            pop.append(_project_to_feasible(h, n))

        fitness = [_c5_from_h(h) for h in pop]

        generation = 0
        # run GA while time permits
        while time_left() > 5.0 and generation < 5000:
            # elite preservation
            elite_idx = np.argsort(fitness)[:2]
            new_pop = [pop[i].copy() for i in elite_idx]

            # create offspring
            while len(new_pop) < 30:
                p1 = _tournament(pop, fitness, k=3)
                p2 = _tournament(pop, fitness, k=3)
                child = _crossover(p1, p2, prob=0.7)
                # mutation scale linearly decays with generations
                scale = 0.08 * (1.0 - generation / 5000.0) + 0.02
                child = _mutate(child, prob=0.3, scale=scale)
                child = _project_to_feasible(child, n)
                new_pop.append(child)

            pop = new_pop
            fitness = [_c5_from_h(h) for h in pop]

            # update global best
            local_best_idx = int(np.argmin(fitness))
            if fitness[local_best_idx] < best_c5:
                best_c5 = fitness[local_best_idx]
                best_h = pop[local_best_idx].copy()
                best_n = n
                if best_c5 < target_bound:
                    break

            generation += 1

        if best_c5 < target_bound:
            break

    # final refinement on the globally best solution
    if best_h is not None:
        # try binary hill‑climb if close to binary
        binary_like = np.mean((best_h < 0.1) | (best_h > 0.9)) > 0.8
        if binary_like:
            hb = _make_binary_feasible(best_h)
            hb, c_bin = _binary_hill_climb(hb, attempts=2000)
            if c_bin < best_c5:
                best_h, best_c5 = hb, c_bin

        # generic local search
        best_h = _local_refine(best_h, attempts=200, delta=0.008)
        best_c5 = _c5_from_h(best_h)

    else:
        # fallback: uniform half‑half
        n = 96
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
