def _project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Use iterative water-filling
    for _ in range(100):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-9:
            break
        # Determine which entries can be adjusted
        if diff > 0:
            # need to decrease some positive entries
            idx = h > 0
        else:
            idx = h < 1
        if not np.any(idx):
            break
        delta = diff / np.count_nonzero(idx)
        h[idx] -= delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # final correction:
    s = h.sum()
    diff = s - target
    if abs(diff) > 1e-12:
        # adjust first element not at bound
        for i in range(n):
            if diff > 0 and h[i] > 0:
                adj = min(diff, h[i])
                h[i] -= adj
                diff -= adj
            elif diff < 0 and h[i] < 1:
                adj = min(-diff, 1 - h[i])
                h[i] += adj
                diff += adj
            if abs(diff) < 1e-12:
                break
    return h

def _project_to_feasible(h, n, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= 1e-9:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h):
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _linear_corr_fft(h):
    """Linear correlation of h with (1-h) using FFT, ordered like np.correlate."""
    n = h.size
    L = 2 * n - 1
    # FFTs with zero padding to length L
    fft_h = np.fft.fft(h, n=L)
    fft_b = np.fft.fft(1.0 - h, n=L)
    corr = np.real(np.fft.ifft(fft_h * np.conj(fft_b)))
    # reorder to shift -n+1 .. n-1 (centered)
    corr = np.fft.fftshift(corr)
    return corr

def _gradient_from_weights(h, w):
    """Gradient of smooth max w.r.t. h given weights w (same ordering as _linear_corr_fft)."""
    n = h.size
    L = 2 * n - 1
    # linear convolution w * h (both length L, h zero‑padded)
    fft_w = np.fft.fft(w, n=L)
    fft_h = np.fft.fft(h, n=L)
    conv = np.real(np.fft.ifft(fft_w * fft_h))
    # central N values correspond to sum_k w_k h_{m-k}
    conv_center = conv[n - 1 : n - 1 + n]
    return 1.0 - 2.0 * conv_center

def _smooth_max_weights(corr, beta):
    """Return smooth max value and normalized soft‑max weights."""
    m = np.max(corr)
    # subtract max for numerical stability
    shifted = corr - m
    expb = np.exp(beta * shifted)
    s = np.sum(expb)
    smooth = m + np.log(s) / beta
    w = expb / s
    return smooth, w

    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C₅ bound, and the sample count."""
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # optional initial construction from the parent harness
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (powers of two, up to 1024)
    sizes = [256, 512, 1024]
    if init_h is not None:
        sizes = sorted(set(sizes + [init_h.size]))

    for n in sizes:
        if time_left() < 5.0:
            break

        # build a starting vector for this resolution
        if init_h is not None and init_h.size == n:
            base = init_h.copy()
        elif init_h is not None:
            base = _upscale_h(init_h, n)
        else:
            base = np.zeros(n)
            base[: n // 2] = 1.0
            np.random.shuffle(base)
        base = _project_to_feasible(base, n)

        # several random restarts of the gradient‑descent phase
        restarts = 8
        for r in range(restarts):
            if time_left() < 2.0:
                break
            if r == 0:
                start_vec = base.copy()
            else:
                start_vec = np.random.rand(n)
                start_vec = _project_to_feasible(start_vec, n)

            h_opt, c5_val = _local_optimize(start_vec,
                                            max_iters=2000,
                                            lr_start=0.6, lr_end=0.04,
                                            beta_start=0.4, beta_end=15.0,
                                            time_left=time_left)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 < 0.3808:
                    break

        # final polish with SLSQP if time permits
        if best_h is not None and time_left() > 4.0:
            h_ref, c5_ref = _refine_with_slsqp(best_h, max_iter=600)
            if c5_ref < best_c5:
                best_h, best_c5, best_n = h_ref, c5_ref, n
                if best_c5 < 0.3808:
                    break

        if best_c5 < 0.3808:
            break

    # fallback to a trivial feasible vector
    if best_h is None:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    # final sanity check
    best_h = _project_to_feasible(best_h, best_n)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
