def crossover(p1, p2, target, rng):
    n = p1.size
    mask = rng.random(n) < 0.5
    child = np.where(mask, p1, p2)
    # adjust sum
    cur = child.sum()
    diff = cur - target
    if diff > 0:
        # need to turn diff ones to zero
        ones = np.where(child == 1)[0]
        if len(ones) > 0:
            flip = rng.choice(ones, diff, replace=False)
            child[flip] = 0
    elif diff < 0:
        zeros = np.where(child == 0)[0]
        if len(zeros) > 0:
            flip = rng.choice(zeros, -diff, replace=False)
            child[flip] = 1
    return child
```

Mutation:

def mutate(child, target, rng, swap_prob=0.2):
    if rng.random() < swap_prob:
        # perform a few swaps
        num_swaps = rng.integers(1, 4)  # 1 to 3
        for _ in range(num_swaps):
            ones = np.where(child == 1)[0]
            zeros = np.where(child == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            child[i], child[j] = 0, 1
    return child
```

def tournament(pop, scores, rng, k=2):
    idx = rng.choice(len(pop), k, replace=False)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best]
```

def evolve_binary(n, target, rng, time_limit):
    # Initialize population
    popsize = 30
    pop = [binary_random(n, target, rng) for _ in range(popsize)]
    scores = np.array([compute_c5(p) for p in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # Keep elite count 2
    elite_cnt = 2
    start = time.time()
    while time.time() - start < time_limit:
        # Selection
        p1 = tournament(pop, scores, rng)
        p2 = tournament(pop, scores, rng)
        child = crossover(p1, p2, target, rng)
        child = mutate(child, target, rng)

            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()
        # Optionally preserve elites explicitly
        # (pop already contains some elites)
    return best_h, best_score

def perturb_continuous(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum to exactly target
    diff = CURRENT_TARGET - np.sum(x)
    # adjust by adding diff to random coordinates within bounds
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - np.sum(x)
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def perturb_continuous(h, rng, scale=0.2):
    """Transfer a small amount ε between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def binary_random(n, target, rng):
    """Binary vector with exactly target ones."""
    h = np.zeros(n, dtype=np.uint8)
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1
    return h


def crossover(p1, p2, target, rng):
    """Uniform crossover then repair to exact target number of ones."""
    n = p1.size
    mask = rng.random(n) < 0.5
    child = np.where(mask, p1, p2).astype(np.uint8)
    cur = int(child.sum())
    diff = cur - target
    if diff > 0:
        ones = np.where(child == 1)[0]
        flip = rng.choice(ones, diff, replace=False)
        child[flip] = 0
    elif diff < 0:
        zeros = np.where(child == 0)[0]
        flip = rng.choice(zeros, -diff, replace=False)
        child[flip] = 1
    return child


def mutate(child, target, rng, swap_prob=0.2):
    """Swap a few (1,0) pairs with probability swap_prob."""
    if rng.random() < swap_prob:
        num_swaps = rng.integers(1, 4)
        for _ in range(num_swaps):
            ones = np.where(child == 1)[0]
            zeros = np.where(child == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            child[i], child[j] = 0, 1
    return child


def tournament(pop, scores, rng, k=2):
    """Select the best of k randomly chosen individuals."""
    idx = rng.choice(len(pop), k, replace=False)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best]


def evolve_binary(n, target, rng, time_limit):
    """Simple steady‑state GA for binary vectors."""
    popsize = 30
    pop = [binary_random(n, target, rng) for _ in range(popsize)]
    scores = np.array([compute_c5(p) for p in pop])

    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    start = time.time()
    while time.time() - start < time_limit:
        p1 = tournament(pop, scores, rng)
        p2 = tournament(pop, scores, rng)
        child = crossover(p1, p2, target, rng)
        child = mutate(child, target, rng)

            scores[worst] = c
            if c < best_score:
                best_score = c
                best_h = child.copy()
        # optional early stop if perfect bound reached

    # ---------- 5. Final feasibility enforcement ----------
    if best_h is None:
        best_n = n
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(best_n)
        add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
