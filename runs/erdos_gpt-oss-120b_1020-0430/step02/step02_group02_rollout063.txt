def _refine_h(h, time_left_func, max_iter=None):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    # loops until time left or max_iter reached
    it = 0
    while time_left_func() > 0 and (max_iter is None or it < max_iter):
        # compute correlation and its max index
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_val = corr.max()
        shift = np.argmax(corr) - (n - 1)
        # compute shift_h
        shift_h = np.zeros(n)
        if shift >= 0:
            shift_h[:n - shift] = best_h[shift:]
        else:
            shift_h[-shift:] = best_h[:n + shift]  # shift negative
        coeff = 1.0 - shift_h
        # donors
        donors = np.where(best_h > 1e-12)[0]
        receivers = np.where(best_h < 1 - 1e-12)[0]
        if donors.size > 0 and receivers.size > 0:
            donor_idx = donors[np.argmax(coeff[donors])]
            receiver_idx = receivers[np.argmin(coeff[receivers])]
            max_eps = min(best_h[donor_idx], 1 - best_h[receiver_idx])
            if max_eps > 1e-12:
                eps = max_eps * np.random.rand() * 0.5
                cand = best_h.copy()
                cand[donor_idx] -= eps
                cand[receiver_idx] += eps
                cand_c = _c5_from_h(cand)
                if cand_c < best_c:
                    best_h, best_c = cand, cand_c
        # random move fallback
        if np.random.rand() < 0.2:
            i, j = np.random.randint(n, size=2)
            if i != j and best_h[i] > 0 and best_h[j] < 1:

                cand_c = _c5_from_h(cand)

    n = len(h)
    best = h.copy()
    best_c = _c5_from_h(best)
    while time_left_func() > 0:
        # compute worst shift
        corr = np.correlate(best, 1 - best, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # compute shift_h as before
        shift_h = np.zeros(n)
        if shift >= 0:
            shift_h[:n-shift] = best[shift:]
        else:
            shift_h[-shift:] = best[:n+shift]
        # For each position i in overlapping region, contribution = best[i] * (1 - shift_h[i]).

def _project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # if sum too high, reduce; else increase.
    # Use while loop up to 1000:
    for _ in range(1000):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to reduce
            mask = h > 0
            # uniform reduction across mask
            delta = diff / np.count_nonzero(mask)
            h[mask] -= delta
        else:
            mask = h < 1
            delta = -diff / np.count_nonzero(mask)
            h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    s = h.sum()
    residual = target - s
    # distribute residual across any slot with space
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

# -------------------------------------------------------------
# Basic utilities
# -------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given step function (linearised by correlation)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme.
    The routine keeps the sum exact (up to numerical tolerance).
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (should be rare)
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-13:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-13:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

# -------------------------------------------------------------
def _gd_optimize(start_h: np.ndarray, time_func) -> tuple:
    """
    Sub‑gradient descent that always acts on the shift realizing the
    current maximum correlation.  The step size is adapted with a simple
    back‑tracking line search; occasional random perturbations avoid
    stagnation.
    """
    h = start_h.copy()
    n = h.size
    best_c = _c5_from_h(h)
    alpha = 0.1
    it = 0
    while time_func() > 0 and it < 5000:
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        t = int(np.argmax(corr))
        shift = t - (n - 1)

        # shifted version of h (zero‑padded)
        if shift >= 0:
            sh = np.empty_like(h)
            sh[: n - shift] = h[shift:]
            sh[n - shift :] = 0.0
        else:
            sh = np.empty_like(h)
            sh[-shift :] = h[: n + shift]
            sh[: -shift] = 0.0

        coeff = 1.0 - sh                     # linear coefficient for the worst shift
        improved = False
        step = alpha
        for _ in range(5):                  # back‑tracking
            h_new = h - step * coeff
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project_to_feasible(h_new, n)
            c_new = _c5_from_h(h_new)
            if c_new < best_c - 1e-12:
                h, best_c = h_new, c_new
                alpha = min(alpha * 1.2, 0.5)
                improved = True
                break
            step *= 0.5

            # small random shake to escape flat regions
            noise = 0.02 * np.random.randn(n)
            h = _project_to_feasible(np.clip(h + noise, 0.0, 1.0), n)
            best_c = _c5_from_h(h)
            alpha = 0.1
        it += 1
    return h, best_c

            c_cand = _c5_from_h(cand)
            if c_cand < best_c:
                h, best_c = cand, c_cand

        # occasional pure random swap to add diversity
        if np.random.rand() < 0.05:
            a, b = np.random.randint(n, size=2)
            if a != b:
                max_eps = min(h[a], 1.0 - h[b])
                if max_eps > 1e-12:
                    eps = max_eps * np.random.rand() * 0.5
                    cand = h.copy()
                    cand[a] -= eps
                    cand[b] += eps
                    c_cand = _c5_from_h(cand)
                    if c_cand < best_c:
                        h, best_c = cand, c_cand
        it += 1
    return h, best_c


# -------------------------------------------------------------
# Interpolation helper (coarse → fine grid)
# -------------------------------------------------------------
def _interp_to_fine(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # -----------------------------------------------------------------
    # Grab any provided initial construction (size may be arbitrary)
    # -----------------------------------------------------------------
    try:
        init_global = np.array(initial_h_values, dtype=float)
    except NameError:
        init_global = None

    target = 0.3808
    best_h = None
    best_c = float("inf")
    best_n = None

    # -----------------------------------------------------------------
    # Grid sizes to explore (small → medium → fine)
    # -----------------------------------------------------------------
    n_grid = [80, 128, 256, 384, 512]
    # ensure the supplied size (if any) is tried first
    if init_global is not None and init_global.size not in n_grid:
        n_grid = [init_global.size] + n_grid

    for n in n_grid:
        if time_left() < 5.0:
            break

        # -------------------------------------------------------------
        # Build a starting vector for this grid size
        # -------------------------------------------------------------
        if init_global is not None and init_global.size == n:
            base = init_global.copy()
        elif init_global is not None:
            base = _interp_to_fine(init_global, n)
        else:
            # random binary start (exactly n/2 ones)
            base = np.zeros(n)
            base[np.random.choice(n, n // 2, replace=False)] = 1.0

        base = _project_to_feasible(base, n)

        # -------------------------------------------------------------
        # Allocate a time slice for this grid size
        # -------------------------------------------------------------
        remaining_grids = len(n_grid) - n_grid.index(n)
        slice_time = min(time_left() * 0.5 / remaining_grids, time_left())
        slice_end = time.time() + slice_time

        # -------------------------------------------------------------
        # Local optimisation loop (restart → gradient descent → refine)
        # -------------------------------------------------------------
        local_best_h = base
        local_best_c = _c5_from_h(base)

        # occasional SLSQP call (lightweight, helps with very smooth starts)
        try:
            from scipy import optimize

            def _proj_vec(x):
                return _project_to_feasible(x, n)

            def _ineq(x):
                t = x[-1]
                h = x[:n]
                return t - np.correlate(h, 1 - h, mode="full")

            def _eq(x):
                return np.sum(x[:n]) - n / 2.0

            x0 = np.concatenate([local_best_h, [_c5_from_h(local_best_h)]])
            bounds = [(0.0, 1.0)] * n + [(0.0, None)]
            cons = [{"type": "ineq", "fun": _ineq},
                    {"type": "eq", "fun": _eq}]
            res = optimize.minimize(lambda x: x[-1], x0,
                                    method="SLSQP", bounds=bounds,
                                    constraints=cons,
                                    options={"maxiter": 200, "ftol": 1e-9, "disp": False})
            if res.success:
                cand = _project_to_feasible(res.x[:n], n)
                cand_c = _c5_from_h(cand)
                if cand_c < local_best_c:
                    local_best_h, local_best_c = cand, cand_c
        except Exception:
            pass  # scipy not available or optimisation failed

        # main restart loop
        while time.time() < slice_end:
            # random binary restart (keeps exact mass)
            h0 = np.zeros(n)
            h0[np.random.choice(n, n // 2, replace=False)] = 1.0
            h0 = _project_to_feasible(h0, n)

            # gradient‑descent phase
            h_opt, c_opt = _gd_optimize(h0,
                                        lambda: max(0.0, slice_end - time.time()))
            if c_opt < local_best_c:
                local_best_h, local_best_c = h_opt, c_opt

            # short refinement phase
            h_ref, c_ref = _refine_h(local_best_h,
                                      lambda: max(0.0, slice_end - time.time()))
            if c_ref < local_best_c:
                local_best_h, local_best_c = h_ref, c_ref

            # keep global best
            if local_best_c < best_c:
                best_h, best_c, best_n = local_best_h.copy(), local_best_c, n
                if best_c <= target:
                    break

        if best_c <= target:
            break

        # -------------------------------------------------------------
        # Upscale the current best to the next grid (if any) before the loop continues
        # -------------------------------------------------------------
        if best_h is not None and best_n == n:
            # prepare for potential next finer grid
            try:
                next_n = n_grid[n_grid.index(n) + 1]
                init_global = _interp_to_fine(best_h, next_n)
            except Exception:
                pass

    # -----------------------------------------------------------------
    # Fallback (unlikely)
    # -----------------------------------------------------------------
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
