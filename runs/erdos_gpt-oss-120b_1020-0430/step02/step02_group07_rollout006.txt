def binary_sa(h0, rng, time_limit):
    h = h0.copy()
    cur_c = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    iter = 0
    while time.time() < time_limit:
        # propose swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1-h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h.astype(float))
        # accept?
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()

            cur_c = new_c
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # update temp
        temp *= decay
        iter += 1
        if iter % 1000 == 0:
            # maybe re-evaluate temp schedule to avoid too low

    return best_c, best_h

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- Utility functions -------------------
def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ------------------- Continuous optimisation -------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing(h_start, rng, max_iters=2000,
                        init_temp=0.02, decay=0.999, scale=0.3):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

def quick_refine(h, rng):
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial

    if c_h < best_c:

    return best_h, best_c

    n = h.size
    max_iter_slsqp = 200 if n > 200 else 300
    hill_iters = 800 if n > 200 else 1500
    anneal_iters = 2000 if n > 200 else 3000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0)
    best_h, best_c = h.copy(), compute_c5(h)

    return best_h, best_c

def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up

# ------------------- Binary optimisation -------------------
def to_binary(h):
    """Convert a float vector to binary keeping exactly half ones."""
    n = h.size
    half = n // 2
    idx = np.argpartition(h, -half)[-half:]
    b = np.zeros(n, dtype=int)
    b[idx] = 1
    return b

def binary_sa(h_start, rng, time_limit):
    """Simulated annealing on binary vectors (swap 1↔0)."""
    n = h_start.size
    h = h_start.copy()
    cur_c = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c = cur_c
    init_temp = 0.02
    decay = 0.9995
    temp = init_temp
    # maintain lists of indices for O(1) swaps
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    while time.time() < time_limit:
        i_one_idx = rng.integers(len(ones))
        i_zero_idx = rng.integers(len(zeros))
        i_one = ones[i_one_idx]
        i_zero = zeros[i_zero_idx]
        # perform swap
        h[i_one] = 0
        h[i_zero] = 1
        # update index lists
        ones[i_one_idx] = i_zero
        zeros[i_zero_idx] = i_one
        new_c = compute_c5(h.astype(float))
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            cur_c = new_c
        else:
            # revert swap
            h[i_one] = 1
            h[i_zero] = 0
            ones[i_one_idx] = i_one
            zeros[i_zero_idx] = i_zero
        temp *= decay
        # occasional re‑heat: random restart of bits
        if rng.random() < 0.001:
            # random binary with exact half ones
            h[:] = 0
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1
            ones = list(idx)

            cur_c = compute_c5(h.astype(float))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            temp = init_temp
    return best_c, best_h
