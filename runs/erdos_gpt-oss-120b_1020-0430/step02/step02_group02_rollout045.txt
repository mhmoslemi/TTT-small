def max_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())  # next power of 2.
    f = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    g = np.fft.rfft(rev, L)
    conv = np.fft.irfft(f * g, L)
    # conv length L, first 2n-1 entries correspond to full convolution.
    # we want max of conv[:2*n-1] (non-negative due to real).
    max_val = conv[:2*n-1].max()
    return max_val * (2.0 / n)  # dx = 2/n

def compute_corr(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]

    conv = np.fft.irfft(f * g, L)

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction in case of rounding errors
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 value using full correlation."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _corr_full_fft(h: np.ndarray) -> np.ndarray:
    """Full correlation h * (1-h) via FFT (no scaling)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    f = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    g = np.fft.rfft(rev, L)
    conv = np.fft.irfft(f * g, L)
    return conv[: 2 * n - 1]  # length 2n-1


def _grad_active(h: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑correlation w.r.t. h for active shift indices."""
    n = h.shape[0]
    a = h
    b = 1.0 - h
    g = np.zeros_like(h)
    for idx in active_idxs:
        k = idx - (n - 1)  # shift amount (negative allowed)
        if k >= 0:
            # term1: b[i+k] for i=0..n-k-1
            g[: n - k] += b[k:]
            # term2: -a[i-k] for i=k..n-1
            g[k:] -= a[: n - k]
        else:
            m = -k
            # term1: b[i-m] for i=m..n-1
            g[m:] += b[: n - m]
            # term2: -a[i+m] for i=0..n-m-1
            g[: n - m] -= a[m:]
    if active_idxs.size > 0:
        g /= float(active_idxs.size)
    return g


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Simple pairwise mass‑exchange hill climb (only improvements)."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]          # can take at most this from i
        max_up = 1.0 - best_h[j]      # can give at most this to j
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c
