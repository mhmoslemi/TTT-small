def compute_c5_and_grad(h):
    # compute O_k vector as correlation of h and (1-h) * dx factor removed
    O = np.correlate(h, 1 - h, mode='full')
    max_val = O.max()
    idxs = np.where(O == max_val)[0]
    # compute subgradient as sum over idxs
    grad = np.zeros_like(h)
    n = len(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            # part1: positions 0..n-1-k
            i = np.arange(n - k)
            grad[i] += 1 - h[i + k]
            # part2: positions k..n-1
            i2 = np.arange(k, n)
            grad[i2] -= h[i2 - k]
        else:
            p = -k
            i = np.arange(p, n)
            grad[i] += 1 - h[i - p]
            i2 = np.arange(n - p)
            grad[i2] -= h[i2 + p]
    # final scaling because correlation scaling? Actually O has no factor dx; C5 = O.max() * dx. The gradient of O_k (without dx) is as computed. scaling not needed.
    return max_val, grad
```

def subgrad_optimize(h0, max_iters, step0, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c, _ = compute_c5_and_grad(h)
    for it in range(1, max_iters+1):
        cv, grad = compute_c5_and_grad(h)
        # step size diminishing
        step = step0 / np.sqrt(it)
        # gradient direction is to reduce cv: subtract gradient
        h = h - step * grad
        # clip and project onto sum=target
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, h.size/2.0)
        cv_new = compute_c5(h)
        if cv_new < best_c:
            best_c = cv_new
            best_h = h.copy()
        # occasionally increase step (restart)
        if rng.random() < 0.01:
            # random perturb
            h = perturb_h(best_h, rng, scale=0.5)
    return best_h, best_c

def subgrad_optimize(h0, max_iters, step0, rng):
    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(1, max_iters+1):
        # compute raw correlation (no dx)
        corr = np.correlate(h, 1-h, mode='full')
        max_raw = corr.max()
        # gradient
        idxs = np.where(corr == max_raw)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                i = np.arange(n - k)
                grad[i] += 1 - h[i + k]
                i2 = np.arange(k, n)
                grad[i2] -= h[i2 - k]
            else:
                p = -k
                i = np.arange(p, n)
                grad[i] += 1 - h[i - p]
                i2 = np.arange(n - p)
                grad[i2] -= h[i2 + p]
        # step size
        step = step0 / np.sqrt(it)
        h = h - step * grad
        # enforce constraints
        h = np.clip(h, 0.0, 1.0)
        # enforce symmetry (optional)
        h = (h + h[::-1]) * 0.5
        h = project_onto_box_sum(h, target)
        # compute new C5
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # occasional random restart (if stuck)
        if rng.random() < 0.001:
            # perturb current best a little
            h = best_h + rng.normal(scale=0.01, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_sum(h, target)
            h = (h + h[::-1]) * 0.5
    return best_h, best_c

def hill_climb_pairs(h, rng, max_iters=2000, start_temp=5.0, decay=0.9995):
    n = h.size
    half = n//2
    cur_h = h.copy()

    best_h = cur_h.copy()
    best_c = cur_c

        zero_idx = rng.choice(np.where(cur_h[:half]==0)[0]) if np.any(cur_h[:half]==0) else None
        one_idx = rng.choice(np.where(cur_h[:half]==1)[0]) if np.any(cur_h[:half]==1) else None
        if zero_idx is None or one_idx is None:
            break
        cand = cur_h.copy()
        cand[zero_idx] = 1.0
        cand[n-1-zero_idx] = 1.0
        cand[one_idx] = 0.0
        cand[n-1-one_idx] = 0.0

            cur_c = cand_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        T *= decay
    return best_h, best_c

def subgrad_optimize(h0, max_iters, step0, rng):
    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)

        corr = np.correlate(h, 1-h, mode='full')
        max_raw = corr.max()

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=50):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    # adjust residual
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1-h(x+k))dx."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Random symmetric vector satisfying 0≤h≤1 and Σh=n/2."""
    m = n // 2
    half_target = n / 4.0
    v = rng.random(m)
    v = project_onto_box_sum(v, half_target)
    return np.concatenate([v, v[::-1]])

    p = np.zeros(half, dtype=float)
    p[idx] = 1.0
    return np.concatenate([p, p[::-1]])


def binary_round_symmetric(h):
    """Round a symmetric vector to a symmetric binary vector (preserving sum)."""
    n = h.size
    half = n // 2
    ones_needed = half // 2
    scores = h[:half]
    idx = np.argpartition(-scores, ones_needed)[:ones_needed]
    p = np.zeros(half, dtype=float)
    p[idx] = 1.0
    return np.concatenate([p, p[::-1]])


# ---------- sub‑gradient descent ----------
def subgrad_optimize(h0, max_iters, step0, rng):
    """Projected sub‑gradient descent on raw max overlap (no dx scaling)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(1, max_iters + 1):
        corr = np.correlate(h, 1 - h, mode="full")
        max_raw = corr.max()
        idxs = np.where(corr == max_raw)[0]

        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                i = np.arange(n - k)
                grad[i] += 1 - h[i + k]
                i2 = np.arange(k, n)
                grad[i2] -= h[i2 - k]
            else:
                p = -k
                i = np.arange(p, n)
                grad[i] += 1 - h[i - p]
                i2 = np.arange(n - p)
                grad[i2] -= h[i2 + p]

        step = step0 / np.sqrt(it)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = (h + h[::-1]) * 0.5          # enforce symmetry
        h = project_onto_box_sum(h, target)

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


# ---------- simulated‑annealing hill climb on binary pairs ----------
def hill_climb_pairs_simanneal(h, rng, max_iters=3000, start_temp=5.0, decay=0.9995):
    """Swap symmetric 0/1 pairs, accepting occasional uphill moves."""
    n = h.size
    half = n // 2
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = start_temp

    zeros = np.where(cur_h[:half] == 0)[0]
    ones = np.where(cur_h[:half] == 1)[0]
    for _ in range(max_iters):
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)

        cand = cur_h.copy()
        cand[i] = 1.0
        cand[n - 1 - i] = 1.0
        cand[j] = 0.0
        cand[n - 1 - j] = 0.0
        cand_c = compute_c5(cand)

        accept = False
        if cand_c < cur_c:
            accept = True
        else:
            prob = np.exp(-(cand_c - cur_c) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = cand
            cur_c = cand_c
            zeros = np.where(cur_h[:half] == 0)[0]
            ones = np.where(cur_h[:half] == 1)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        T *= decay
    return best_h, best_c


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, n / 2.0)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 180
        h0 = random_feasible_symmetric(n, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # ---- sub‑gradient refinement (first 35% of budget) -------------
    if time.time() - start < budget_s * 0.35:
        iters = 15000
        h_opt, c_opt = subgrad_optimize(best_h, iters, step0=0.6, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- binary rounding + annealing (next 25% of budget) ----------
    if time.time() - start < budget_s * 0.60:
        h_bin = binary_round_symmetric(best_h)
        h_bin_opt, c_bin_opt = hill_climb_pairs_simanneal(
            h_bin, rng, max_iters=5000, start_temp=5.0
        )
        if c_bin_opt < best_c:
            best_h, best_c = h_bin_opt, c_bin_opt

    # ---- many random binary restarts (remaining budget) ------------
    while time.time() - start < budget_s * 0.95:
        # occasional perturbation of current best
        if rng.random() < 0.2:
            pert = best_h + rng.normal(scale=0.02, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_onto_box_sum(pert, n / 2.0)
            pert = (pert + pert[::-1]) * 0.5
            pert = project_onto_box_sum(pert, n / 2.0)
            h_start = binary_round_symmetric(pert)
        else:
            h_start = random_binary_symmetric(n, rng)

        h_opt, c_opt = hill_climb_pairs_simanneal(
            h_start, rng, max_iters=4000, start_temp=5.0
        )
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
