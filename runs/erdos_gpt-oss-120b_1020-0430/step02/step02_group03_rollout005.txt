def enforce_sum(arr, target, rng, max_iter=2000):
    diff = target - arr.sum()
    n = arr.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1 - arr[i])
        else:
            add = max(diff, -arr[i])
        arr[i] += add
        diff = target - arr.sum()
    return arr
```

def random_feasible_vector(n, rng):
    # generate uniform random numbers then scale to sum target while clipping
    # naive approach: draw uniform [0,1], then clip? better to use Dirichlet?
    # Dirichlet with parameters 1 yields uniform on simplex, but need bound [0,1].
    # we can generate random binary? Instead, use approach similar earlier: start with uniform 0.5 and random transfers.
    h = np.full(n, 0.5)
    # random perturbations.
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    # clip
    h = np.clip(h, 0, 1)
    # enforce exact sum
    enforce_sum(h, n/2, rng)
    return h
```

def genetic_algorithm(n, rng, time_limit):
    population_size = 150
    # generate initial pop
    population = []
    # add initial_h if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0, 1)
        enforce_sum(h0, n/2, rng)
        population.append(h0)
    # add random
    while len(population) < population_size:
        candidate = random_feasible_vector(n, rng)
        population.append(candidate)
    # evaluate
    scores = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(scores)
    best = population[best_idx].copy()
    best_score = scores[best_idx]
    # GA loop
    start = time.time()
    gen = 0
    while time.time() - start < time_limit:
        gen += 1
        # selection: tournament
        new_pop = []
        while len(new_pop) < population_size:
            # select two parents via tournament
            i1 = tournament_select(scores, rng, tournament_size=3)
            i2 = tournament_select(scores, rng, tournament_size=3)
            p1, p2 = population[i1], population[i2]
            # crossover
            child = blend_crossover(p1, p2, rng)
            # mutation
            child = mutate(child, rng, sigma=0.1)
            new_pop.append(child)
        # evaluate new pop
        new_scores = np.array([compute_c5(ind) for ind in new_pop])
        # combine old and new (elitism: keep best of both)
        combined_pop = population + new_pop
        combined_scores = np.concatenate([scores, new_scores])
        # select top population_size individuals
        idx = np.argsort(combined_scores)[:population_size]
        population = [combined_pop[i] for i in idx]
        scores = combined_scores[idx]
        # update best
        if scores[0] < best_score:
            best_score = scores[0]
            best = population[0].copy()
    return best, best_score
```

def tournament_select(scores, rng, tournament_size=3):

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for the surrogate objective
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(arr, target, rng, max_iter=2000):
    """Adjust arr in-place so that sum(arr) == target while keeping 0≤arr≤1."""
    diff = target - arr.sum()
    if abs(diff) < 1e-12:
        return arr
    n = arr.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - arr[i]
            if cap <= 0:
                continue
            add = min(diff, cap)
        else:
            cap = arr[i]
            if cap <= 0:
                continue
            add = max(diff, -cap)
        arr[i] += add
        diff = target - arr.sum()
    return arr


def compute_c5(h):
    """Exact discrete C5 for a step vector h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    enforce_sum(h, n / 2.0, rng)
    return h


def tournament_select(scores, rng, tournament_size=3):
    """Return index of the best individual from a random tournament."""
    indices = rng.integers(0, len(scores), size=tournament_size)
    return indices[np.argmin(scores[indices])]


def blend_crossover(p1, p2, rng, target):
    """Linear blend of two parents, followed by clipping and sum repair."""
    alpha = rng.random()
    child = alpha * p1 + (1.0 - alpha) * p2
    child = np.clip(child, 0.0, 1.0)
    enforce_sum(child, target, rng)
    return child


def mutate(ind, rng, sigma, target):
    """Gaussian mutation with clipping and sum repair."""
    child = ind + rng.normal(0.0, sigma, size=ind.shape)
    child = np.clip(child, 0.0, 1.0)
    enforce_sum(child, target, rng)
    return child


def hill_climb(h_start, rng, max_iters=4000, start_step=0.3):
    """Simple deterministic hill‑climb (accept improvements only)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        # transfer ε from i to j
        n = h.size
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0.0:
            eps = (rng.random() - 0.5) * 2.0 * max_eps * step
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                best_h = new_h.copy()
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """Refine h using SLSQP on the softmax surrogate (higher β → tighter)."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        enforce_sum(x, CURRENT_TARGET, rng)
    return x


def genetic_algorithm(n, rng, init_h, time_limit):
    """Population‑based search with blend crossover and Gaussian mutation."""
    pop_size = 150
    target = n / 2.0
    population = []
    if init_h is not None:
        population.append(init_h.copy())
    while len(population) < pop_size:
        population.append(random_feasible_vector(n, rng))
    scores = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(scores)
    best = population[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    while time.time() - start < time_limit:
        children = []
        while len(children) < pop_size:
            i1 = tournament_select(scores, rng, tournament_size=3)
            i2 = tournament_select(scores, rng, tournament_size=3)
            child = blend_crossover(population[i1], population[i2], rng, target)
            child = mutate(child, rng, sigma=0.05, target=target)
            children.append(child)
        child_scores = np.array([compute_c5(ind) for ind in children])
        # elitist selection
        combined = population + children
        combined_scores = np.concatenate([scores, child_scores])
        idx = np.argsort(combined_scores)[:pop_size]
        population = [combined[i] for i in idx]
        scores = combined_scores[idx]
        if scores[0] < best_score:
            best_score = scores[0]
            best = population[0].copy()
    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -----------------------------------------------------------------
    # Determine problem size and starting vector
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
        init_h = np.clip(init_h, 0.0, 1.0)
        enforce_sum(init_h, n / 2.0, rng)
    else:
        n = 150  # fallback size
        init_h = None

    # -----------------------------------------------------------------
    # Global optimisation: GA → hill‑climb → SLSQP refinement
    # -----------------------------------------------------------------
    # allocate ~75% of the budget to GA
    ga_time = budget_s * 0.75
    best, best_score = genetic_algorithm(n, rng, init_h, ga_time)

    # hill‑climb refinement (if time permits)
    if time.time() - start_time < budget_s * 0.90:
        best, best_score = hill_climb(best, rng, max_iters=8000, start_step=0.3)

    # SLSQP surrogate refinement (tightening β progressively)
    if time.time() - start_time < budget_s * 0.98:
        for beta in [5.0, 20.0, 80.0, 200.0, 500.0]:
            if time.time() - start_time >= budget_s:
                break
            try:
                best = slsqp_optimize(best, beta, max_iter=800)
            except Exception:
                continue
            best_score = compute_c5(best)

    # final feasibility enforcement
    best = np.clip(best, 0.0, 1.0)
    enforce_sum(best, n / 2.0, rng)
    final_c = compute_c5(best)

    return best.tolist(), float(final_c), int(n)
