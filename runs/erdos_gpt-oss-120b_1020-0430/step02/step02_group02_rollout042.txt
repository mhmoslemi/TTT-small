    g[j] = 1.0 - h[j]

def _project_to_feasible(h, target_sum):
    # clip to [0,1]
    np.clip(h, 0, 1, out=h)
    total = h.sum()
    # If total > target, need to decrease
    while abs(total - target_sum) > 1e-12:
        if total > target_sum:
            # reduce from entries > 0
            above = h > 0
            if not np.any(above):
                break
            reduction = (total - target_sum) / np.count_nonzero(above)
            # reduce but not below 0
            np.subtract(h, reduction, where=above, out=h)
            np.clip(h, 0, 1, out=h)
        else:
            # increase where < 1
            below = h < 1
            if not np.any(below):
                break
            addition = (target_sum - total) / np.count_nonzero(below)
            np.add(h, addition, where=below, out=h)
            np.clip(h, 0, 1, out=h)
        total = h.sum()
        if abs(total - target_sum) < 1e-12:
            break
    return h

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 200) -> np.ndarray:
    """Clip h to [0,1] and adjust its sum to the target by a water‑filling scheme."""
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:                     # need to decrease
            free = h > 0
            if not np.any(free):
                break
            delta = diff / free.sum()
            h[free] -= delta
        else:                            # need to increase
            free = h < 1
            if not np.any(free):
                break
            delta = diff / free.sum()
            h[free] -= delta                # diff is negative, so this adds
        np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-14:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-14:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h


def _apply_swap(corr: np.ndarray, h: np.ndarray, g: np.ndarray,
                i: int, j: int, eps: float, offset: int) -> None:
    """
    Transfer ``eps`` from h[i] to h[j] (i decreases, j increases).
    Updates ``corr`` (full‑mode correlation) and the vectors ``h`` and ``g`` exactly.
    """
    n = h.size
    h_rev = h[::-1]                # view, no copy
    # i  : decrease
    corr[offset - i: offset + n - i] -= eps * g
    corr[offset + i - (n - 1): offset + i + 1] += eps * h_rev
    # j : increase
    corr[offset - j: offset + n - j] += eps * g
    corr[offset + j - (n - 1): offset + j + 1] -= eps * h_rev
    # quadratic correction (exact)
    corr[offset] -= 2.0 * eps * eps
    corr[offset - j + i] += eps * eps
    corr[offset - i + j] += eps * eps
    # update the step function itself
    h[i] -= eps
    h[j] += eps
    g[i] = 1.0 - h[i]
    g[j] = 1.0 - h[j]


def _greedy_optimize(h0: np.ndarray, time_left, target: float = 0.3808,
                    stagnation: int = 15000) -> tuple:
    """
    Greedy pairwise exchange with exact correlation updates.
    Returns the best (h, c5) found before time runs out.
    """
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    h = _project_to_feasible(h, n / 2.0)
    g = 1.0 - h
    offset = n - 1
    corr = np.correlate(h, g, mode='full')
    best_h = h.copy()
    best_c5 = corr.max() * dx
    best_corr = corr.max()

    it = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time_left() > 0.05:
        # focus on the worst shift
        max_idx = int(np.argmax(corr))
        d = max_idx - offset
        i_start = max(0, -d)
        i_end = min(n - 1, n - 1 - d)
        if i_end >= i_start:
            i = rng.integers(i_start, i_end + 1)
        else:
            i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)

        improved = False
        # try moving mass from i→j and also j→i
        for src, dst in ((i, j), (j, i)):
            max_eps = min(h[src], 1.0 - h[dst])
            if max_eps <= 0.0:
                continue
            eps = max_eps * rng.random() * 0.7  # up to 70% of the feasible amount
            if eps < 1e-14:
                continue
            _apply_swap(corr, h, g, src, dst, eps, offset)
            cur_c5 = corr.max() * dx
            if cur_c5 + 1e-12 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_corr = corr.max()
                improved = True
                no_improve = 0
                break
            # revert the move
            _apply_swap(corr, h, g, dst, src, eps, offset)

        it += 1
        if not improved:
            no_improve += 1

        # occasional clean‑up
        if it % 4000 == 0:
            np.clip(h, 0.0, 1.0, out=h)
            g[:] = 1.0 - h

        # if we are stuck, shake the solution a little
        if no_improve >= stagnation:
            h = best_h.copy()
            noise = rng.normal(scale=0.04, size=n)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_feasible(h, n / 2.0)
            g = 1.0 - h
            corr = np.correlate(h, g, mode='full')
            no_improve = 0

        # early exit if we already beat the target
        if best_c5 < target:
            break

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound using a fast greedy exchange algorithm.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Load any existing construction

    try:
        init_h = np.array(initial_h_values, dtype=float)

    best_h, best_c5, best_n = None, float('inf'), None

    candidates = [n0]
    for mul in (2, 3, 4):
        n = n0 * mul
        if n <= 512:

    candidates = sorted(set(candidates))

    for n in candidates:
        if time_left() < 5.0:

        # ----- build starting vector -----
        if n == n0:

        else:
