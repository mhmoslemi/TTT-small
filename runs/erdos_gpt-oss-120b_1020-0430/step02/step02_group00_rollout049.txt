import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Projection onto the feasible set: 0 <= h <= 1 and sum(h) = n/2
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-14:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-14:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

# C5 evaluation (max overlap integral)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Helpers for the SLSQP refinement (same as previous program)
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Gradient‑descent optimiser using a smooth max (soft‑max) approximation
# ----------------------------------------------------------------------
def _softmax_optimize(h0: np.ndarray,
                      max_iter: int = 2000,
                      alpha0: float = 0.5,
                      beta0: float = 5.0) -> np.ndarray:
    """Projected gradient descent on the soft‑max approximation of max Ck."""
    h = _project_to_feasible(h0.copy(), h0.size)
    n = h.size
    alpha = alpha0
    beta = beta0
    for it in range(max_iter):
        # correlation for all shifts
        C = np.correlate(h, 1.0 - h, mode="full")
        maxC = C.max()
        # soft‑max weights (numerically stable)
        w = np.exp(beta * (C - maxC))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # convolutions needed for the gradient
        conv = np.convolve(w, h, mode="full")
        conv2 = conv[n - 1:n - 1 + n]          # Σ_k w_k h_{j-k}
        w_rev = w[::-1]
        conv_rev_full = np.convolve(w_rev, h, mode="full")
        conv_rev = conv_rev_full[n - 1:n - 1 + n]  # Σ_k w_k h_{j+k}
        grad = 1.0 - (conv2 + conv_rev)        # ∇ of smoothed max
        # gradient step + projection
        h = h - alpha * grad
        h = _project_to_feasible(h, n)
        # decay / anneal schedules
        alpha *= 0.9995
        beta *= 1.0005
    return h


# ----------------------------------------------------------------------
# Simple random feasible starter (binary then projected)
# ----------------------------------------------------------------------
def _random_feasible_h(n: int) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    # add tiny noise to escape flat regions
    h += 0.02 * np.random.randn(n)
    return _project_to_feasible(h, n)


# ----------------------------------------------------------------------
# Upscaling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Retrieve optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # ------------------------------------------------------------------
    # Candidate resolutions (we start with the size of the initial, then try larger)
    # ------------------------------------------------------------------
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    # add a couple of larger powers of two (still cheap)
    for n in (256, 512, 1024):
        if n not in n_candidates and n <= 1024:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper to record a new incumbent
    # ------------------------------------------------------------------
    def _record(h_vec, val, n):
        nonlocal best_h, best_c5, best_n
        if val < best_c5:
            best_h = h_vec.copy()
            best_c5 = float(val)
            best_n = n

    # ------------------------------------------------------------------
    # Main loop over resolutions
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Create a baseline vector for this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif best_h is not None and best_n == n:
            base_h = best_h.copy()
        elif best_h is not None:
            base_h = _upscale_h(best_h, n)
        else:
            # purely random start
            base_h = _random_feasible_h(n)

        base_h = _project_to_feasible(base_h, n)

        # number of independent attempts (budget‑aware)
        remaining = time_left()
        attempts = max(2, int(remaining // (len(n_candidates) * 1.5)))
        attempts = min(attempts, 30)

        for att in range(attempts):
            if time_left() < 3.0:
                break

            # pick a starting point
            if att % 5 == 0:
                start_h = _random_feasible_h(n)
            else:
                sigma = 0.1 * (0.85 ** (att // 5))
                pert = sigma * np.random.randn(n)
                start_h = base_h + pert
                start_h = _project_to_feasible(start_h, n)

            # gradient‑descent phase
            h_opt = _softmax_optimize(start_h,
                                      max_iter=1800,
                                      alpha0=0.5,
                                      beta0=5.0)

            # short SLSQP polish
            h_opt, c5_val = _solve_h(h_opt, max_iter=300)

            _record(h_opt, c5_val, n)

            # if we improve substantially, adopt as new base
            if c5_val < best_c5 * 0.999:
                base_h = h_opt

            # early exit if we already beat the target
            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final intensive polish on the best solution found
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 4.0:
        best_h, best_c5 = _solve_h(best_h, max_iter=800)

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    else:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
