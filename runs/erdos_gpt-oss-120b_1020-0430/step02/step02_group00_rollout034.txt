def smooth_binary(bin_vec, r):
    n = bin_vec.size
    kernel = np.zeros(n)
    idx = np.arange(-r, r+1)
    k = (r+1) - np.abs(idx)
    kernel[idx % n] = k
    kernel /= kernel.sum()
    # circular convolution via FFT:
    h = np.real(np.fft.ifft(np.fft.fft(bin_vec) * np.fft.fft(kernel)))
    # Clip to [0,1]
    np.clip(h,0,1,inplace=True)
    return h
```

```
def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to decrease sum, reduce from entries > 0
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final small correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first non-saturated entries
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)
```

def _solve_h(initial_h, max_iter=500):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'ineq', 'fun': _ineq_constraint},
                   {'type': 'eq', 'fun': _eq_constraint}]

    res = optimize.minimize(_objective, x0, method='SLSQP', bounds=bounds,

    h_opt = _project_to_feasible(h_opt, n)

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a feasible step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation_k >= 0 for all k."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Refine a vector with SLSQP."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _smooth_binary(bin_vec: np.ndarray, r: int) -> np.ndarray:
    """Circular convolution of a binary vector with a triangular kernel of radius r."""
    n = bin_vec.size
    if r <= 0:
        return bin_vec.astype(float)
    idx = np.arange(-r, r + 1)
    w = (r + 1) - np.abs(idx)
    kernel = np.zeros(n)
    kernel[idx % n] = w
    kernel /= kernel.sum()
    h = np.real(np.fft.ifft(np.fft.fft(bin_vec) * np.fft.fft(kernel)))
    np.clip(h, 0.0, 1.0, out=h)
    return h


def _gd_optimize(h0: np.ndarray, max_iters: int = 200, alpha0: float = 0.2) -> tuple:
    """Simple projected gradient descent on max_{k} Corr_k."""
    n = h0.size
    h = h0.copy()
    val = _c5_from_h(h)
    best_h, best_val = h.copy(), val
    alpha = alpha0
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = np.argmax(corr) - (n - 1)
        grad = 1.0 - (np.roll(h, shift) + np.roll(h, -shift))
        new_h = h - alpha * grad
        new_h = _project_to_feasible(new_h, n)
        new_val = _c5_from_h(new_h)
        if new_val < best_val:
            best_val, best_h = new_val, new_h.copy()
        if new_val <= val:
            h, val = new_h, new_val
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break
    return best_h, best_val
