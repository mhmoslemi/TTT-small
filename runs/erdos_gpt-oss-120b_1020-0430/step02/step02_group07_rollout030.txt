import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global placeholders for the SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.4):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2500,
                        init_temp=0.02, decay=0.998, scale=0.2):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce the sum exactly (numeric drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def improve_worst_shift(h, rng, iters=2000, factor=0.15):
    """Targeted deterministic improvement on the shift attaining max overlap."""
    n = h.size
    for _ in range(iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        if shift == 0:
            break
        # neighbor sums for each index
        left = np.zeros(n)
        right = np.zeros(n)
        if shift > 0:
            left[shift:] = h[:-shift]
            right[:n - shift] = h[shift:]
        else:
            s = -shift
            left[: n - s] = h[s:]
            right[s:] = h[: n - s]
        neigh = left + right
        pos = np.where(h > 0)[0]
        if pos.size == 0:
            break
        i = pos[np.argmax(neigh[pos])]
        neg = np.where(h < 1)[0]
        if neg.size == 0:
            break
        j = neg[np.argmin(neigh[neg])]
        eps = factor * min(h[i], 1 - h[j])
        if eps <= 0:
            break
        h[i] -= eps
        h[j] += eps
    return h


def quick_refine_custom(h, rng):
    """Fast refinement: a few SLSQP passes, targeted gradient, and hill‑climb."""
    best_h, best_c = h.copy(), compute_c5(h)

    for beta in (5.0, 20.0, 80.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial

    # deterministic gradient improvement on the worst shift
    trial = improve_worst_shift(best_h.copy(), rng, iters=1500, factor=0.2)
    c_trial = compute_c5(trial)
    if c_trial < best_c:
        best_h, best_c = trial, c_trial

    # a short hill‑climb to polish
    trial_h, c_h = hill_climb(best_h, rng, max_iters=800, start_step=0.3)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h

    return best_h, best_c


def deep_refine_custom(h, rng):
    """Intensive refinement: many SLSQP β values, deeper gradient steps, SA."""
    best_h, best_c = h.copy(), compute_c5(h)
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 2000.0)

    for beta in betas:
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=300)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial

        # deterministic gradient descent
        trial = improve_worst_shift(best_h.copy(), rng, iters=3000, factor=0.12)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial

        # moderate hill‑climb
        trial_h, c_h = hill_climb(best_h, rng, max_iters=1000, start_step=0.25)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h

    # final simulated‑annealing sweep
    trial_sa, c_sa = simulated_annealing(best_h, rng, max_iters=2500,
                                         init_temp=0.025, decay=0.997, scale=0.18)
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa

    return best_h, best_c

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- primary intensive refinement on the current size ----------
    best_h, best_c = deep_refine_custom(best_h, rng)

    # ---- stochastic restarts within remaining budget ---------------
    while time.time() - start < budget_s * 0.85:
        # small random perturbation of the best solution to escape possible plateaux
        h_start = perturb_h(best_h, rng, scale=0.15)
        h_ref, c_ref = quick_refine_custom(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ---- optional up‑sampling if time permits ----------------------
    if time.time() - start < budget_s * 0.95 and best_h.size * 2 <= 720:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        h_up, c_up = deep_refine_custom(h_up, rng)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # ---- final feasibility clean‑up -------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
