def hill_climb_incremental(h, max_steps, time_limit):
    n = len(h)
    # Compute initial correlation
    C = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()

    cur_val = best_val
    start = time.time()
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        # choose i != j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # compute max possible epsilon
        max_down = h[i]
        max_up = 1.0 - h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        # epsilon schedule: random fraction * max_eps * 0.5 maybe
        eps = max_eps * np.random.rand() * 0.5
        # apply
        # store positions to revert
        transfer_mass(h, C, i, j, eps)
        new_val = np.max(C) * (2.0 / n)
        if new_val < cur_val - 1e-12:
            cur_val = new_val

            if new_val < best_val:

        # periodically recompute full C to correct drift (e.g., every 5000 steps)

# Helper utilities

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 using a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction (tiny)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all lags."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum constraint."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 200) -> tuple:
    """Single SLSQP run; returns feasible h and its C5."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0) * 1.0
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)

def _apply_delta_corr(C: np.ndarray, h: np.ndarray, b: np.ndarray, idx: int, delta: float) -> None:
    """Update full correlation C when h[idx] changes by delta (b=1-h)."""
    n = h.size
    off = n - 1
    # part 1: delta * b[idx + l] for l in [-idx, n-1-idx]
    start1 = -idx + off
    end1 = (n - 1 - idx) + off
    C[start1 : end1 + 1] += delta * b
    # part 2: -delta * h[idx - l] for l in [idx-(n-1), idx]
    start2 = idx - (n - 1) + off
    end2 = idx + off
    C[start2 : end2 + 1] += -delta * h[::-1]


def _transfer_mass(h: np.ndarray, C: np.ndarray, i: int, j: int, eps: float) -> None:
    """Move eps from h[i] to h[j] (i.e. h[i]-=eps, h[j]+=eps) and update C."""
    # b = 1 - h (snapshot before any modification)
    b = 1.0 - h
    _apply_delta_corr(C, h, b, i, -eps)
    h[i] -= eps
    b[i] = 1.0 - h[i]
    _apply_delta_corr(C, h, b, j, eps)
    h[j] += eps
    b[j] = 1.0 - h[j]


def _revert_transfer(h: np.ndarray, C: np.ndarray, i: int, j: int, eps: float) -> None:
    """Undo a previous transfer (move eps back from j to i)."""
    b = 1.0 - h
    _apply_delta_corr(C, h, b, j, -eps)
    h[j] -= eps
    b[j] = 1.0 - h[j]
    _apply_delta_corr(C, h, b, i, eps)
    h[i] += eps
    b[i] = 1.0 - h[i]

    """

    """

    best_h = h.copy()
    best_val = np.max(C) * (2.0 / n)
    cur_val = best_val
