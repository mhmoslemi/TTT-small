def perturb_h_sum_preserving(h, rng, scale):
    n = h.size
    # choose source index with positive h
    src_candidates = np.where(h > 1e-12)[0]
    if len(src_candidates) == 0:
        return h
    src = rng.choice(src_candidates)
    # choose target where h < 1
    tgt_candidates = np.where(h < 1 - 1e-12)[0]
    if len(tgt_candidates) == 0:
        return h
    tgt = rng.choice(tgt_candidates)
    if src == tgt:
        return h
    max_eps = min(h[src], 1.0 - h[tgt])
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[src] -= eps
    new_h[tgt] += eps
    return new_h

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0≤h≤1, Σh=n/2)."""
    half = n // 2
    target_half = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


# ----------------------------------------------------------------------
# hill climbing (deterministic) – used for a quick warm‑up
# ----------------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                 # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()

            # gradient of smooth max (using same formula as original code)
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)

            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # occasional uphill move to escape shallow traps
                if rng.random() < 0.005:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

# SLSQP refinement using soft‑max surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr * (2.0 / x.size)
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """One SLSQP pass using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ----------------------------------------------------------------------
# simple simulated‑annealing on the original objective
# ----------------------------------------------------------------------
def simulated_annealing(h_start, rng, time_allowed, eval_func):
    """Metropolis annealing on the hard max objective."""
    start = time.time()
    n = h_start.size
    target = n / 2.0
    cur_h = h_start.copy()
    cur_c = eval_func(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T0, T1 = 0.04, 1e-5

    while time.time() - start < time_allowed:
        frac = (time.time() - start) / time_allowed
        T = T0 * (T1 / T0) ** frac
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.5
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c_new = eval_func(new_h)
        if c_new < cur_c or rng.random() < math.exp((cur_c - c_new) / max(T, 1e-12)):
            cur_h, cur_c = new_h, c_new
            if c_new < best_c:
                best_h, best_c = new_h.copy(), c_new
        # re‑project occasionally to keep the sum exact
        if (int((time.time() - start) * 10) % 47) == 0:
            diff = target - cur_h.sum()
            if abs(diff) > 1e-12:
                cur_h = project_onto_box_sum(cur_h, target)
                cur_c = eval_func(cur_h)
                if cur_c < best_c:
                    best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


# ----------------------------------------------------------------------
# main driver
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initial vector ----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        # enforce a mild symmetry (helps the search)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 180
        h0 = random_feasible_symmetric(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # quick deterministic local search ---------------------------------
    if time.time() - start < budget_s * 0.1:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2500, start_step=0.5)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # gradient‑descent restarts -----------------------------------------
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    steps_per_beta = 500
    while time.time() - start < budget_s * 0.78:
        # start point: either random or a blend with current best
        h_start = random_feasible_symmetric(n, rng)
        if rng.random() < 0.2:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)

        h_opt, c_opt = grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng)

        # high‑beta SLSQP polishing
        for beta in (200.0, 500.0, 1000.0):
            try:
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=150)
            except Exception:
                pass
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            if best_c <= 0.3808:      # reached target – can stop early
                break

    # simulated annealing fine‑tuning ------------------------------------
    elapsed = time.time() - start
    time_left = budget_s - elapsed
    if time_left > 12.0:
        best_h, best_c = simulated_annealing(best_h, rng, time_left, compute_c5)

    # final short hill‑climb for polishing --------------------------------
    elapsed = time.time() - start
    if time.time() - start < budget_s * 0.95:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2000, start_step=0.4)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # enforce constraints exactly -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
