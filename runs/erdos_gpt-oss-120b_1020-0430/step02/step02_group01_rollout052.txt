def genetic_optimize(n, rng, time_limit, init_h=None):
    start = time.time()
    pop_size = 30
    # Ensure half sum target
    target = n / 2.0

    # Initialize population
    population = []
    fitness = []
    # Add init_h if provided
    if init_h is not None:
        # If different size: interpolate
        if init_h.size != n:
            x_old = np.linspace(0, 1, init_h.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            h_resized = np.interp(x_new, x_old, init_h)
        else:
            h_resized = init_h.copy()
        # clip and fix sum
        h_resized = np.clip(h_resized, 0.0, 1.0)
        diff = target - h_resized.sum()
        if abs(diff) > 1e-12:
            h_resized = fix_sum(h_resized, target, rng)
        population.append(h_resized)
        fitness.append(compute_c5(h_resized))

    # Fill rest
    while len(population) < pop_size:
        # randomly binary or continuous
        if rng.random() < 0.5:
            # binary
            h = np.zeros(n)
            idx = rng.choice(n, int(np.floor(target)), replace=False)
            h[idx] = 1.0
            # if target not integer, add fractional value
            if not np.isclose(target, int(target)):
                # assign remaining fraction to random index
                rem = target - int(np.floor(target))
                i = rng.integers(n)
                h[i] = rem
        else:
            h = random_feasible_vector(n, rng)
        population.append(h)
        fitness.append(compute_c5(h))

    # Sort population by fitness
    pop_fitness = np.array(fitness)
    order = np.argsort(pop_fitness)
    population = [population[i] for i in order]
    fitness = list(pop_fitness[order])

    # Genetic loop
    while time.time() < time_limit:
        # produce offspring
        for _ in range(pop_size):
            # tournament selection
            i1 = rng.choice(pop_size, 3, replace=False)
            p1_idx = i1[np.argmin([fitness[idx] for idx in i1])]  # min fitness (lower C)

            parent2 = population[p2_idx]
            # Crossover
            alpha = rng.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            # Mutation (epsilon transfer)
            if rng.random() < 0.9:

            # Clip
            child = np.clip(child, 0.0, 1.0)
            # Fix sum
            s = child.sum()
            diff = target - s
            if abs(diff) > 1e-12:

            # Evaluate
            child_f = compute_c5(child)
            # Insert if better than worst
            if child_f < fitness[-1]:

                fitness[-1] = child_f

def fix_sum(x, target, rng, max_iter=2000):
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
            if add <= 0:
                continue
            x[i] += add
            diff -= add
        else:
            sub = min(-diff, x[i])
            if sub <= 0:
                continue
            x[i] -= sub
            diff += sub
    # If still diff not zero (unlikely), distribute uniformly and clip
    if abs(diff) > 1e-12:
        # adjust uniformly
        adj = diff / n
        x += adj
        # clip and iterate again maybe
    return x

def perturb_h(h, rng, scale=0.3):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure within bounds
    eps = max(-max_eps, min(eps, max_eps))
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # possibly progressive step reduction every some iterations
        if (it+1) % (max_iters // 5 + 1) == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=6):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # gradient g = derivative of correlation at worst shift w.r.t h
        g = np.zeros_like(h)
        if k == 0:
            # corr[center] = sum h*(1 - h) => derivative = 1 - 2h
            g = 1.0 - 2.0 * h
        else:
            # For shift k>0, corr = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
            # derivative w.r.t h[i] (i< n-k): 1 - h[i+k]
            # derivative w.r.t h[i+k] (i< n-k): -h[i]
            upper = n - k
            g[:upper] += 1.0 - h[k:]    # w.r.t h[i]
            g[k:] -= h[:upper]          # w.r.t h[i+k]
        # take step
        h = h - step * (0.5 ** it) * g
        # clip and fix sum
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = fix_sum(h, target, rng)
    return h

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq", "fun":sum_eq_constraint}
    # initial point
    x0 = h0.copy()
    # minimize
    res = minimize(softmax_obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = fix_sum(x, CURRENT_TARGET, rng)
    return x

Now sum_eq_constraint and softmax_obj:

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def fix_sum(x, target, rng, max_iter=2000):
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
            if add <= 0:
                continue
            x[i] += add
            diff -= add
        else:
            sub = min(-diff, x[i])
            if sub <= 0:
                continue
            x[i] -= sub
            diff += sub
    return np.clip(x, 0.0, 1.0)


def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h = h / h.sum() * target
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-12:
        h = fix_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.3):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = fix_sum(h, target, rng)
    return h


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = fix_sum(x, CURRENT_TARGET, rng)
    return x


def genetic_optimize(n, rng, time_limit, init_h=None):
    target = n / 2.0
    pop_size = 30
    population = []
    fitness = []

    # add provided prototype (interpolated if needed)
    if init_h is not None:
        if init_h.size != n:
            old_x = np.linspace(0, 1, init_h.size, endpoint=False)
            new_x = np.linspace(0, 1, n, endpoint=False)
            h = np.interp(new_x, old_x, init_h)
        else:
            h = init_h.copy()
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) > 1e-12:
            h = fix_sum(h, target, rng)
        population.append(h)
        fitness.append(compute_c5(h))

    # fill the rest
    while len(population) < pop_size:
        if rng.random() < 0.5:
            # binary seed
            h = np.zeros(n)
            ones = int(np.floor(target))
            idx = rng.choice(n, ones, replace=False)
            h[idx] = 1.0
            if not np.isclose(target, ones):
                i = rng.integers(n)
                h[i] = target - ones
        else:
            h = random_feasible_vector(n, rng)
        population.append(h)
        fitness.append(compute_c5(h))

    # sort
    order = np.argsort(fitness)
    population = [population[i] for i in order]
    fitness = [fitness[i] for i in order]

    # main GA loop
    while time.time() < time_limit:
        for _ in range(pop_size):
            # tournament selection for two parents
            t1 = rng.choice(pop_size, 3, replace=False)
            p1 = t1[np.argmin([fitness[i] for i in t1])]
            t2 = rng.choice(pop_size, 3, replace=False)
            p2 = t2[np.argmin([fitness[i] for i in t2])]
            parent1, parent2 = population[p1], population[p2]

            # crossover
            alpha = rng.random()
            child = alpha * parent1 + (1 - alpha) * parent2

            # mutation
            if rng.random() < 0.9:
                child = perturb_h(child, rng, scale=0.3)

            child = np.clip(child, 0.0, 1.0)
            if abs(child.sum() - target) > 1e-12:
                child = fix_sum(child, target, rng)

            child_f = compute_c5(child)

            # replace worst if better
            if child_f < fitness[-1]:
                population[-1] = child
                fitness[-1] = child_f
                order = np.argsort(fitness)
                population = [population[i] for i in order]
                fitness = [fitness[i] for i in order]

        # occasional improvement on the elite
        if time.time() < time_limit - 0.1:
            elite = population[0].copy()
            elite, elite_f = hill_climb(elite, rng, max_iters=400, start_step=0.4)
            if elite_f < fitness[0]:
                population[0] = elite
                fitness[0] = elite_f
                order = np.argsort(fitness)
                population = [population[i] for i in order]
                fitness = [fitness[i] for i in order]

    return population[0]
