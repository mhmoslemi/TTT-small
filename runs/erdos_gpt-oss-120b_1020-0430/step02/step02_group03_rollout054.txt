import time
import numpy as np
from scipy.optimize import minimize

# globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb accepting only improvements."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

            if max_transfer <= 0:
                continue
            eps = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[j] -= eps
            new_h[i] += eps

        # occasional extra perturbation for larger moves
        if rng.random() < 0.1:
            a, b = rng.integers(0, n, size=2)
            if a != b:
                if rng.random() < 0.5:
                    mt = min(new_h[a], 1.0 - new_h[b])
                    if mt > 0:
                        e2 = rng.random() * mt
                        new_h[a] -= e2
                        new_h[b] += e2
                else:
                    mt = min(new_h[b], 1.0 - new_h[a])
                    if mt > 0:
                        e2 = rng.random() * mt
                        new_h[b] -= e2
                        new_h[a] += e2

        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_h, cur_c = new_h, new_c
            if new_c < global_best_c:
                global_best_c = new_c
                global_best_h = new_h.copy()

        # occasional restart around the best known point
        if rng.random() < 0.004:
            cur_h = perturb_h(global_best_h, rng, scale=0.25)
            cur_c = compute_c5(cur_h)

    # -------------------------------------------------
    # 5. Final SLSQP refinement with increasing β
    # -------------------------------------------------
    beta_list = [5.0, 20.0, 80.0, 200.0, 500.0]
    h_opt = global_best_h.copy()
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.94:
            break
        try:
            h_opt = slsqp_optimize(h_opt, beta, max_iter=500)
        except Exception:
            continue
        c_opt = compute_c5(h_opt)
        if c_opt < global_best_c:
            global_best_c = c_opt
            global_best_h = h_opt.copy()

    # -------------------------------------------------
    # 6. Optional up‑sampling and refinement if time allows
    # -------------------------------------------------
    if time.time() - start_time < budget_s * 0.99 and n * 2 <= 300:
        h_up = np.repeat(global_best_h, 2)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # simple hill‑climb on the finer grid
        cur_up, cur_c_up = h_up.copy(), compute_c5(h_up)
        best_up, best_c_up = cur_up.copy(), cur_c_up
        for _ in range(4000):
            i, j = rng.integers(0, n_up, size=2)
            if i == j:
                continue
            if rng.random() < 0.5:
                mt = min(cur_up[i], 1.0 - cur_up[j])
                if mt <= 0:
                    continue
                eps = rng.random() * mt
                new_up = cur_up.copy()
                new_up[i] -= eps
                new_up[j] += eps
            else:
                mt = min(cur_up[j], 1.0 - cur_up[i])
                if mt <= 0:
                    continue
                eps = rng.random() * mt
                new_up = cur_up.copy()
                new_up[j] -= eps
                new_up[i] += eps
            c_new = compute_c5(new_up)
            if c_new < cur_c_up:
                cur_up, cur_c_up = new_up, c_new
                if c_new < best_c_up:
                    best_up, best_c_up = new_up.copy(), c_new

        # final SLSQP on up‑sampled candidate
        for beta in beta_list:
            if time.time() - start_time > budget_s:
                break
            try:
                best_up = slsqp_optimize(best_up, beta, max_iter=400)
            except Exception:
                pass
            best_c_up = compute_c5(best_up)
            if best_c_up < global_best_c:
                global_best_c = best_c_up
                global_best_h = best_up.copy()
                n = n_up

    # -------------------------------------------------
    # 7. One more brief hill‑climb if time remains
    # -------------------------------------------------
    if time.time() - start_time < budget_s * 0.998:
        global_best_h, _ = hill_climb(global_best_h, rng, max_iters=2500, start_step=0.1)

    # -------------------------------------------------
    # 8. Final clean‑up and return
    # -------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
