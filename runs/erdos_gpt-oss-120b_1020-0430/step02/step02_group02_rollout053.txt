def _upscale_h(coarse, n):

def _project_to_feasible(h, n, tol=1e-9):
    # project onto [0,1] and sum = n/2.
    # Use algorithm: clip to bounds, then adjust by water-filling.
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    # compute current sum
    s = h.sum()
    if abs(s - target) <= tol:
        return h
    # If sum > target, need to reduce some values.
    # We'll use iterative approach:
    while True:
        if s > target:
            # compute excess = s - target
            excess = s - target
            # indices that can be reduced (h > 0)
            free = np.where(h > 0)[0]
            if free.size == 0:
                break
            delta = excess / free.size
            # reduce

            h = np.clip(h, 0.0, 1.0)
        else:

            free = np.where(h < 1)[0]

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):

        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:

        h[free] -= delta

def _upscale_h(coarse: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    m = coarse.size
    x_coarse = (np.arange(m) + 0.5) * (2.0 / m)
    x_fine = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x_fine, x_coarse, coarse)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a given discretised h using FFT based correlation."""
    n = h.size
    m = 2 * n
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

            # no overlapping indices, fall back to random move
            i, j = np.random.randint(n, size=2)
            if i != j and best_h[i] > 0 and best_h[j] < 1:
                eps = min(best_h[i], 1 - best_h[j]) * np.random.rand() * 0.5
                cand = best_h.copy()
                cand[i] -= eps
                cand[j] += eps
                cand_c5 = _c5_from_h(cand)
                if cand_c5 < best_c5:
                    best_h, best_c5 = cand, cand_c5
            continue

        i_arr = np.arange(i_start, i_end, dtype=int)
        j_arr = i_arr + offset

        feasible = (best_h[i_arr] > 1e-12) & (best_h[j_arr] < 1 - 1e-12)
        if not np.any(feasible):
            # no feasible directed move, try a random exchange
            i, j = np.random.randint(n, size=2)
            if i != j and best_h[i] > 0 and best_h[j] < 1:
                eps = min(best_h[i], 1 - best_h[j]) * np.random.rand() * 0.5
                cand = best_h.copy()
                cand[i] -= eps
                cand[j] += eps
                cand_c5 = _c5_from_h(cand)
                if cand_c5 < best_c5:
                    best_h, best_c5 = cand, cand_c5
            continue

        # a heuristic score favouring moving mass from large h_i to small h_j
        scores = 1.0 + best_h[i_arr[feasible]] - best_h[j_arr[feasible]]
        idx_sel = i_arr[feasible][int(np.argmax(scores))]
        j_sel = idx_sel + offset

        eps_max = min(best_h[idx_sel], 1.0 - best_h[j_sel])
        if eps_max <= 0:
            continue
        # choose a modest fraction of the allowed amount
        eps = eps_max * (0.05 + 0.35 * np.random.rand())

        cand = best_h.copy()
        cand[idx_sel] -= eps
        cand[j_sel] += eps
        cand_c5 = _c5_from_h(cand)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
        else:
            # tiny probability to accept a worse move and escape plateaus
            if np.random.rand() < 0.001:
                best_h, best_c5 = cand, cand_c5

    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise the Erdős C5 bound and return (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load any provided initial construction
    # ------------------------------------------------------------------
    try:
        coarse_init = np.array(initial_h_values, dtype=float)  # type: ignore
        has_init = True
    except NameError:
        coarse_init = None
        has_init = False

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Grid sizes to explore (prefer multiples of any existing size)
    grid_sizes = []
    if has_init:
        grid_sizes.append(coarse_init.size)
    for mul in (2, 3, 4):
        if has_init:
            grid_sizes.append(int(coarse_init.size * mul))
    grid_sizes.extend([128, 160, 192, 224, 256, 320, 384, 512])
    grid_sizes = sorted(set(grid_sizes))

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for n in grid_sizes:
        if time_left() < 5.0:
            break

        # ------------------------------------------------------------------
        # Build a feasible starting vector for this grid size
        # ------------------------------------------------------------------
        if has_init:
            if coarse_init.size == n:
                h0 = coarse_init.copy()
            else:
                h0 = _upscale_h(coarse_init, n)
        else:
            h0 = np.full(n, 0.5)

        h0 = _project_to_feasible(h0, n)

        # ------------------------------------------------------------------
        # Deterministic directed hill climb
        # ------------------------------------------------------------------
        limit = start_time + budget_s * 0.85
        h_opt, c5_val = _directed_hill_climb(h0, time_limit=limit)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n

        if best_c5 < 0.3808:
            break

        # ------------------------------------------------------------------
        # Stochastic restarts – random perturbations + hill climb
        # ------------------------------------------------------------------
        attempts = 0
        while time_left() > 4.0 and attempts < 30:
            attempts += 1
            sigma = 0.3 * np.exp(-0.1 * attempts)  # slowly shrink perturbation
            pert = best_h + sigma * np.random.randn(n)
            pert = _project_to_feasible(pert, n)
            limit = start_time + budget_s * 0.95
            h2, c5_2 = _directed_hill_climb(pert, time_limit=limit)
            if c5_2 < best_c5:
                best_h, best_c5, best_n = h2, c5_2, n
                if best_c5 < 0.3808:
                    break

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Fallback – simple uniform function (should never happen)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
