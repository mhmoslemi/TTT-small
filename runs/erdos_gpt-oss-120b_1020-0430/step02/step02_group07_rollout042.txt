def anneal_full(h0, rng, time_budget):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    # ensure sums exactly: we trust h0 sum equals target
    dx = 2.0 / n
    T = target   # sum of h
    # pre compute arrays
    shift_range = np.arange(-(n-1), n, dtype=int)
    hp = np.concatenate((np.zeros(n-1), h, np.zeros(n-1)))
    S = np.correlate(h, h, mode='full')
    cur_min = S.min()
    cur_C = (T - cur_min) * dx
    best_h = h.copy()
    best_C = cur_C
    # annealing parameters
    temp = 0.02
    decay = 0.9995
    step_scale = 0.5
    start = time.time()
    iter_count = 0
    while True:
        if time.time() - start >= time_budget:
            break
        # sample i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # determine max epsilon that respects bounds
        max_eps = min(1.0 - h[i], h[j])
        if max_eps <= 1e-12:
            continue
        eps = rng.random() * max_eps * step_scale
        if eps <= 0:
            continue
        idx_i = n - 1 + i
        idx_j = n - 1 + j
        # compute A_i and A_j
        Ai = hp[idx_i + shift_range] + hp[idx_i - shift_range]
        Aj = hp[idx_j + shift_range] + hp[idx_j - shift_range]
        delta = eps * (Ai - Aj)
        eps2 = eps * eps
        delta[n - 1] += 2.0 * eps2
        delta[i - j + n - 1] -= eps2
        delta[j - i + n - 1] -= eps2
        new_S = S + delta
        new_min = new_S.min()
        new_C = (T - new_min) * dx
        accept = False
        if new_C < cur_C - 1e-12:
            accept = True
        else:
            # probabilistic uphill
            if temp > 1e-9:
                prob = np.exp(-(new_C - cur_C) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            # update state
            h[i] += eps
            h[j] -= eps
            hp[idx_i] = h[i]
            hp[idx_j] = h[j]
            S = new_S
            cur_C = new_C
            cur_min = new_min
            if new_C < best_C - 1e-12:
                best_C = new_C
                best_h = h.copy()
        # update parameters
        temp *= decay
        step_scale *= decay
        iter_count += 1
    return best_h, best_C

import time

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:   # final safety
        h[:] = np.clip(h, 0.0, 1.0)
        h *= target / h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    return enforce_sum(h_up, target_up, rng)

    time_used = time.time() - start_time
    remaining = budget_s - time_used
    phase1 = min(remaining * 0.35, 400.0)          # ≈ 350 s max for first phase
    best_h, best_c = anneal_full(h0, rng, phase1)

    # ---------------------------------------------------------
    # progressive up‑sampling and deeper annealing while time permits
    # ---------------------------------------------------------
    while True:
        now = time.time()
        left = budget_s - (now - start_time)
        if left < 30.0:
            break
        n = best_h.size
        if n >= 1080:                # stop growing resolution
            break
        # up‑sample by factor 2
        up_h = upsample_h(best_h, factor=2, rng=rng)
        # allocate most of the remaining time to this level
        alloc = left * 0.55
        best_h, best_c = anneal_full(up_h, rng, alloc)

    # ---------------------------------------------------------
    # final clean‑up (enforce exact feasibility and evaluate)
    # ---------------------------------------------------------
    n_final = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n_final / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_final)
