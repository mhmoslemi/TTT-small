def simulated_annealing(n, rng, max_iters, start_T, end_T, binary=False):
    if binary:
        h = random_binary_vector(n, rng)
    else:
        h = random_feasible_vector(n, rng) # using previous function but we can implement own.
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    for it in range(max_iters):
        t = start_T + (end_T - start_T) * (it / max_iters)  # linear schedule; start_T > end_T
        # propose move
        if binary:
            # swap
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            new_h = h.copy()
            new_h[i] = 0
            new_h[j] = 1
        else:
            # epsilon transfer
            i = rng.integers(0,n)
            j = rng.integers(0,n)
            while i==j:
                j = rng.integers(0,n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            # step scale factor: use some scale factor (maybe 1.0) * t / start_T
            scale = t / start_T
            eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        new_val = compute_c5(new_h)
        delta = new_val - cur
        if delta < 0 or rng.random() < np.exp(-delta / (t+1e-12)):
            h = new_h
            cur = new_val
            if cur < best:
                best = cur
                best_h = h.copy()
    return best_h, best

def random_feasible_vector(n, rng):

    h = rng.random(n)
    # scale to sum target
    h = h * (target / h.sum())

    if abs(diff) > 1e-12:

            total = cap.sum()
            if total > 0:

def simulated_annealing_continuous(h0, rng, max_iters, start_T, end_T):
    h = h0.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    n = h.size
    for it in range(max_iters):
        # temperature schedule
        t = start_T * ((end_T/start_T) ** (it / max_iters))
        # propose random epsilon transfer
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        while j == i:
            j = rng.integers(0,n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        scale = t / start_T
        eps = (rng.random() * 2 - 1) * max_eps * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # No clipping needed due to eps bound.

            cur = new_c
            if cur < best:
                best = cur
                best_h = h.copy()
    return best_h, best

import time
import numpy as np

# ----- basic utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    # proportional redistribution to fix the sum
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h
            tot = cap.sum()
            if tot > 0:
                h += cap * (diff / tot)
        else:
            sur = h
            tot = sur.sum()
            if tot > 0:
                h -= sur * (abs(diff) / tot)
        diff = target - h.sum()
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # re‑project to the feasible set
    np.clip(new_h, 0.0, 1.0, out=new_h)
    target = n / 2.0
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - new_h
            tot = cap.sum()
            if tot > 0:
                new_h += cap * (diff / tot)
        else:
            sur = new_h
            tot = sur.sum()
            if tot > 0:
                new_h -= sur * (abs(diff) / tot)
    return new_h


def hill_climb(
    h_start: np.ndarray, rng: np.random.Generator, max_iters: int = 3000, start_step: float = 0.5
):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_descent_optim(
    h: np.ndarray, rng: np.random.Generator, max_iter: int = 20, step: float = 0.05
) -> np.ndarray:
    """Iterative gradient descent on the currently worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.9 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                cap = 1.0 - h
                tot = cap.sum()
                if tot > 0:
                    h += cap * (diff / tot)
            else:
                sur = h
                tot = sur.sum()
                if tot > 0:
                    h -= sur * (abs(diff) / tot)
    return h


def simulated_annealing_continuous(
    h0: np.ndarray,
    rng: np.random.Generator,
    max_iters: int,
    start_T: float,
    end_T: float,
    start_time: float,
    budget_s: float,
):
    """SA on continuous h, respecting the sum and box constraints."""
    h = h0.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    n = h.size
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.97:
            break
        t = start_T * ((end_T / start_T) ** (it / max_iters))
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        scale = t / start_T
        eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # sum is unchanged by construction, bounds are respected
        new_c = compute_c5(new_h)
        delta = new_c - cur
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-16)):
            h = new_h
            cur = new_c
            if cur < best:
                best = cur
                best_h = h.copy()
    return best_h, best
