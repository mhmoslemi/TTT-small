import time
import numpy as np

# ----------------------------------------------------------------------
# Feasibility projection: clip to [0,1] and enforce sum(h)=n/2
# ----------------------------------------------------------------------
def project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-10) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0 and h[i] < 1 - 1e-12:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 1e-12:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


# ----------------------------------------------------------------------
# Compute C5 for (already feasible) h
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    return float(c_raw.max() * (2.0 / n))


# ----------------------------------------------------------------------
# Gradient of smoothed max objective
# ----------------------------------------------------------------------
def gradient_and_weights(h: np.ndarray, alpha: float):
    n = h.size
    c = np.correlate(h, 1.0 - h, mode="full")          # length 2n-1
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    w /= w.sum()                                      # softmax weights
    # convolutions to obtain Σ_k w_k h_{j‑k} and Σ_k w_k h_{j+k}
    conv = np.convolve(w, h, mode="full")
    conv_rev = np.convolve(w[::-1], h, mode="full")
    off = n - 1
    part = conv[off : off + n]
    part_rev = conv_rev[off : off + n]
    grad = 1.0 - part - part_rev                       # ∂f/∂h_j
    return grad, c


# ----------------------------------------------------------------------
# Adam optimiser with time deadline
# ----------------------------------------------------------------------
def adam_optimize(h0: np.ndarray, deadline: float) -> tuple:
    n = h0.size
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    start_alpha, end_alpha = 5.0, 40.0
    lr = 0.05
    best_h = h.copy()
    best_c5 = compute_c5(h)
    max_iters = 20000

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # schedule α linearly
        alpha = start_alpha + (end_alpha - start_alpha) * (it / max_iters)
        grad, _ = gradient_and_weights(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h, n)

        if (it % 200) == 0:
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        # occasional noise to escape tiny plateaus
        if (it % 1500) == 0 and it > 0:
            noise = 0.02 * np.random.randn(n)
            h = project_to_feasible(h + noise, n)

    # final validation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

# Main driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Load optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # ------------------------------------------------------------------
    # Determine problem size (must be even)
    # ------------------------------------------------------------------

        n = init_h.size
    else:
        n = 256  # default even size

    # ------------------------------------------------------------------
    # Prepare starting vectors
    # ------------------------------------------------------------------
    starts = []
    if init_h is not None:
        starts.append(project_to_feasible(init_h, n))
    # a few random feasible seeds
    for _ in range(4):
        rnd = np.random.rand(n)
        starts.append(project_to_feasible(rnd, n))

    best_h = None
    best_c5 = float("inf")

    # ------------------------------------------------------------------
    # Optimise each start (global search)
    # ------------------------------------------------------------------
    for s in starts:
        if time.time() > deadline - 2.0:
            break
        h_opt, c5_val = adam_optimize(s, deadline)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
        # early stop if we beat target comfortably
        if best_c5 < 0.3805:

    # ------------------------------------------------------------------
    # If nothing found, fall back to uniform half‑ones
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Final projection & bound recomputation (safety)
    # ------------------------------------------------------------------
    best_h = project_to_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h.tolist(), float(best_c5), int(n)
