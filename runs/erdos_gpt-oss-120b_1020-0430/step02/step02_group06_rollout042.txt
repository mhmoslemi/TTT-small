def _gradient_corr(h, d):
    n = h.shape[0]
    idx = np.arange(n)
    idxp = idx + d
    # first term
    g = np.empty(n, dtype=h.dtype)
    mask = (idxp >= 0) & (idxp < n)
    # if idxp inside, (1 - h[idxp])
    g[mask] = 1.0 - h[idxp[mask]]
    # out of bounds: 1.0
    g[~mask] = 1.0
    # second term: subtract h[idx-d] if idx-d in bounds
    idxm = idx - d
    mask2 = (idxm >= 0) & (idxm < n)
    g[mask2] -= h[idxm[mask2]]
    return g
```

import time
import numpy as np
from scipy import optimize

# -------------------- Core utilities -------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1−h(x+k))dx  (discrete version)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """SLSQP refinement: minimise the auxiliary variable t >= correlation."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": _ineq},
            {"type": "eq", "fun": _eq}]
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons,
                            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    h_opt = _project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    return h_opt, _c5_from_h(h_opt)


def _gradient_corr(h: np.ndarray, d: int) -> np.ndarray:
    """Sub‑gradient of the worst correlation term w.r.t. h."""
    n = h.size
    idx = np.arange(n)
    idxp = idx + d
    maskp = (idxp >= 0) & (idxp < n)
    g = np.empty(n, dtype=h.dtype)
    g[maskp] = 1.0 - h[idxp[maskp]]
    g[~maskp] = 1.0
    idxm = idx - d
    maskm = (idxm >= 0) & (idxm < n)
    g[maskm] -= h[idxm[maskm]]
    return g


def _gd_optimize(h0: np.ndarray, max_iters: int, time_left) -> tuple:
    """Projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    best_h, best_c5 = h.copy(), _c5_from_h(h)
    for it in range(max_iters):
        if time_left() < 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        d = np.argmax(corr) - (n - 1)          # signed lag of worst shift
        g = _gradient_corr(h, d)
        # simple linear schedule for step size
        alpha = 0.4 * (1.0 - it / max_iters) + 0.01
        h_new = h - alpha * g
        h_new = _project_to_feasible(h_new, n)
        c5 = _c5_from_h(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        h = h_new
    return best_h, best_c5


def _anneal_refine(h0: np.ndarray, max_iters: int, time_left) -> tuple:
    """Greedy stochastic refinement (mass transfer between two points)."""
    n = h0.size
    cur_h = h0.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5
    step = 0.05
    for it in range(max_iters):
        if time_left() < 0.5:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if cur_h[i] <= 0.0 or cur_h[j] >= 1.0:
            continue
        eps = (np.random.rand() - 0.5) * 2.0 * step
        eps = max(-cur_h[i], eps)
        eps = min(1.0 - cur_h[j], eps)
        if abs(eps) < 1e-12:
            continue
        trial = cur_h.copy()
        trial[i] -= eps
        trial[j] += eps
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_c5, best_h = c5, trial.copy()
        if c5 < cur_c5:
            cur_h, cur_c5 = trial, c5
        step *= 0.9999   # slowly cool the proposal size
    return best_h, best_c5


def _random_feasible(n: int) -> np.ndarray:
    """Create a random feasible vector (0‑1 bounds, correct sum)."""
    r = np.random.rand(n)
    return _project_to_feasible(r, n)


# -------------------- Main optimisation driver -------------------- #

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # parent construction if supplied
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # grid sizes – favour the parent size, then explore a broader range
    n_grid = [96, 128, 176, 224, 256, 320, 384, 448, 512]
    n_grid = [n for n in n_grid if n % 2 == 0 and n <= 512]

    for n in n_grid:
        if time_left() < 5.0:
            break

        # prepare a list of initial candidates for this n
        candidates = []

        # parent (up‑scaled / down‑scaled)
        if parent_h is not None:
            if parent_h.size != n:
                cand = _upscale_h(parent_h, n)
            else:
                cand = parent_h.copy()
            candidates.append(_project_to_feasible(cand, n))

        # uniform start
        candidates.append(np.full(n, 0.5))

        # a few random starts
        for _ in range(3):
            candidates.append(_random_feasible(n))

        # parametric polynomial shapes (h = max(0,1−|u|^p))
        for p in np.linspace(1.0, 8.0, 8):
            L = (p + 1) / (2.0 * p)               # half‑width giving exact integral 1
            x = (np.arange(n) + 0.5) * (2.0 / n)   # mid‑points in [0,2]
            u = (x - 1.0) / L
            hshape = np.maximum(0.0, 1.0 - np.abs(u) ** p)
            candidates.append(_project_to_feasible(hshape, n))

        # cosine bump (full support, height 1)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        hcos = (1.0 + np.cos(np.pi * (x - 1.0))) / 2.0
        candidates.append(_project_to_feasible(hcos, n))

        # main optimisation loop for the current grid size
        for init_h in candidates:
            if time_left() < 4.0:
                break

            # gradient‑descent phase
            max_iters_gd = max(1200, n * 3)
            h_gd, c5_gd = _gd_optimize(init_h, max_iters_gd, time_left)

            # stochastic refinement
            if time_left() > 2.0:
                h_ref, c5_ref = _anneal_refine(h_gd, 1500, time_left)
                if c5_ref < c5_gd:
                    h_gd, c5_gd = h_ref, c5_ref

            # final SLSQP polishing (shortened)
            if time_left() > 5.0:
                h_opt, c5_opt = _solve_h(h_gd, max_iter=250)
                if c5_opt < c5_gd:
                    h_gd, c5_gd = h_opt, c5_opt

            # keep the best overall
            if c5_gd < best_c5:
                best_h, best_c5, best_n = h_gd, c5_gd, n
                if best_c5 < target_bound:
                    break
        if best_c5 < target_bound:
            break

    # fallback (should never be needed)
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
