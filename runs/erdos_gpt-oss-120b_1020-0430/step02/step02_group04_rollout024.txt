def projected_gradient_descent(h0, beta_sched, steps_per_beta, init_step, rng):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for beta in beta_sched:
        for it in range(steps_per_beta):
            # compute gradient
            grad = gradient_smooth_max(h, beta)
            # line search: try step, if not improved, halve step, up to certain attempts
            for attempt in range(10):
                new_h = h - step * grad
                new_h = project_onto_box_sum(new_h, target)
                new_c = compute_c5(new_h)
                if new_c < best_c:
                    best_h = new_h.copy()
                    best_c = new_c
                    h = new_h
                    break
                else:
                    step *= 0.5
            else:
                # if all attempts fail, keep h unchanged and maybe increase step a bit
                step = init_step
            # optional step decay
            if (it+1) % (steps_per_beta//5) == 0:
                step = max(step * 0.9, 1e-4)
    return best_h, best_c

def gradient_smooth_max(h, beta):
    n = h.size
    sum_h = h.sum()
    # corr_hh = np.correlate(h, h, mode='full')
    corr = np.correlate(h, h, mode='full')
    C = 1.0 - (2.0 / n) * corr   # shape 2n-1
    # compute softmax weights
    betaC = beta * C
    max_bc = np.max(betaC)
    w = np.exp(betaC - max_bc)
    w_sum = w.sum()
    w = w / w_sum
    # compute conv terms
    one_minus_h = 1.0 - h
    # correlation of w with (1-h)
    conv1_full = np.correlate(w, one_minus_h, mode='full')
    start = n - 1
    conv1 = conv1_full[start : start + n]
    # correlation of reversed w with h
    w_rev = w[::-1]
    conv2_full = np.correlate(w_rev, h, mode='full')
    conv2 = conv2_full[start : start + n]
    grad = (2.0 / n) * (conv1 - conv2)
    return grad

def hill_climb(h_start, rng, max_iters=1000, start_step=0.3):
    h = h_start.copy()

    best_h = h.copy()
    best_c = compute_c5(h)

            h = new_h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project onto box [lower,upper] and enforce sum == target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random vector in [0,1] with exact sum = n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


# ---------- gradient of smooth max ----------
def gradient_smooth_max(h, beta):
    """Gradient of (1/beta) * logsumexp(beta*C_k) w.r.t. h."""
    n = h.size
    # C_k = 1 - (2/n) * corr_hh(k)
    corr = np.correlate(h, h, mode="full")          # length 2n-1
    C = 1.0 - (2.0 / n) * corr
    bc = beta * C
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= w.sum()                                    # softmax weights
    # conv1_i = sum_k w_k * (1 - h_{i+k})
    conv1_full = np.correlate(w, 1.0 - h, mode="full")
    start = n - 1
    conv1 = conv1_full[start:start + n]
    # conv2_i = sum_k w_k * h_{i-k}
    w_rev = w[::-1]
    conv2_full = np.correlate(w_rev, h, mode="full")
    conv2 = conv2_full[start:start + n]
    grad = (2.0 / n) * (conv1 - conv2)
    return grad


# ---------- projected gradient descent ----------
def projected_gradient_descent(h0, beta_sched, steps_per_beta, init_step, rng):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    step = init_step
    for beta in beta_sched:
        for _ in range(steps_per_beta):
            grad = gradient_smooth_max(h, beta)
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, target)
            h = new_h
            step *= 0.9998                     # gentle decay
        step = init_step                         # reset after each beta
    return h


# ---------- simple hill‑climb ----------
def hill_climb(h_start, rng, max_iters=800, start_step=0.4):
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, target)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 6) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- SLSQP refinement (soft‑max surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=150):
    """Single SLSQP pass on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

            h_start = random_feasible(n, rng)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = project_onto_box_sum(h_start, n / 2.0)

        # gradient descent with smooth‑max continuation
        h_gd = projected_gradient_descent(
            h_start, beta_sched, steps_per_beta, init_step, rng
        )

        # short hill‑climb refinement
        h_hc, c_hc = hill_climb(h_gd, rng, max_iters=600, start_step=0.3)

        # optional SLSQP polish
        try:
            h_opt = slsqp_optimize(h_hc, beta=200.0, max_iter=120)
        except Exception:
            h_opt = h_hc

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- final polishing -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
