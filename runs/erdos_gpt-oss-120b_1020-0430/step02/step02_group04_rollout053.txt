def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0

def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

    """Project v onto [lower,upper] with exact sum = target."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:

            best_c = new_c
            best_h = new_h.copy()

        curr_c = compute_c5(h)
        # accept if improving, otherwise occasional uphill move
        if new_c <= curr_c or (rng is not None and rng.random() < 0.01):
            h = new_h

        if step * np.linalg.norm(g) < 1e-8:
            break

    return best_h, best_c


# ---------------------------------------------------------
# SLSQP refinement of a smooth max surrogate
# ---------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP pass using the log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------------------------------------------------------
# Main entry point
# ---------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # -----------------------------------------------------
    # Initialise from parent if present
    # -----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 180
        h0 = random_feasible(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # -----------------------------------------------------
    # Quick local descent to get a decent starting point
    # -----------------------------------------------------
    if time.time() - start < budget_s * 0.1:
        h_tmp, c_tmp = gd_optimize(best_h, max_iters=200, init_step=5.0, rng=rng)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # -----------------------------------------------------
    # Stochastic restarts – gradient descent + occasional SLSQP
    # -----------------------------------------------------
    while time.time() - start < budget_s * 0.75:
        # random start (mix with current best sometimes)
        h_start = random_feasible(n, rng)
        if rng.random() < 0.2:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)
            h_start = (h_start + h_start[::-1]) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)

        h_opt, c_opt = gd_optimize(h_start, max_iters=300, init_step=5.0, rng=rng)

        # refine only if we have a new record
        if c_opt < best_c:
            for beta in (200.0, 500.0):
                try:
                    h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=250)
                except Exception:
                    pass
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # -----------------------------------------------------
    # Optional up‑sampling (double resolution) if time permits
    # -----------------------------------------------------
    if time.time() - start < budget_s * 0.95 and best_h.size * 2 <= 400:
        h_up = np.repeat(best_h, 2)               # sum is automatically correct
        n_up = h_up.size
        # local optimisation at higher resolution
        h_opt, c_opt = gd_optimize(h_up, max_iters=500, init_step=3.0, rng=rng)
        for beta in (200.0, 500.0):
            try:
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=300)
            except Exception:
                pass
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # -----------------------------------------------------
    # Final clean‑up and return
    # -----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
