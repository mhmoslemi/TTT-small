def slsqp_optimize_half(h0, beta, max_iter=200):
    n_half = h0.size
    bounds = [(0.0, 1.0)] * n_half
    # no equality constraint
    # define softmax surrogate on full vector build inside
    def obj(x):
        full = np.concatenate([x, 1 - x[::-1]])
        corr = np.correlate(full, 1 - full, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, options={'maxiter' : max_iter})
    x = np.clip(res.x, 0, 1)
    return x
```

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:

    diff = target - h.sum()

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

    if i == j:
        return h.copy()
    low = max(-h[j], h[i] - 1.0)
    high = min(h[i], 1.0 - h[j])
    if low >= high:
        return h.copy()
    delta = (rng.random() - 0.5) * 2.0 * (high - low) * scale
    delta = np.clip(delta, low, high)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_int = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_int == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine using SLSQP on a smooth log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # tiny exact correction
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- simulated annealing ----------------------------------------
    sigma = 0.05                # mutation scale
    T = 1e-3                    # initial temperature
    window = 200                # adaptation window
    accept_cnt = 0
    iters = 0
    h = best_h.copy()
    cur_c = best_c
    time_limit_sa = start_time + budget_s * 0.55

    while time.time() < time_limit_sa:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        low = max(-h[j], h[i] - 1.0)
        high = min(h[i], 1.0 - h[j])
        if low >= high:
            continue
        delta = rng.normal(0.0, sigma)
        delta = np.clip(delta, low, high)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c = compute_c5(new_h)
        dC = new_c - cur_c
        accept = False
        if dC < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-dC / T):
            accept = True
        if accept:
            h = new_h
            cur_c = new_c
            accept_cnt += 1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        iters += 1
        if iters % window == 0:
            rate = accept_cnt / window
            if rate > 0.2:
                sigma *= 1.1
            else:
                sigma *= 0.9
            sigma = min(max(sigma, 5e-5), 0.2)
            accept_cnt = 0
            T *= 0.95
            if T < 1e-12:
                T = 0.0

    # ---- deterministic hill‑climb -----------------------------------
    h_opt, c_opt = hill_climb(best_h, rng, max_iters=3000, start_step=0.3)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # ---- SLSQP refinement -------------------------------------------
    beta_seq = [5.0, 20.0, 80.0, 200.0]
    for beta in beta_seq:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_h, best_c = refined, refined_c

    # ---- final clean‑up ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
