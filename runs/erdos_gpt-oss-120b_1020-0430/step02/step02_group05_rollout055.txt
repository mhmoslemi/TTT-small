    target = n/4
    dim = n//2
    # initialize population list of half vectors
    pop = []
    fitness = []
    # first individual = half_init (projected)
    init = _project_to_simplex_box(half_init, target)
    pop.append(init)
    fitness.append(_c5_from_half(init, n))

    while len(pop) < popsize:
        # random vector from Dirichlet
        rand = np.random.rand(dim)
        # project onto simplex box: first scale to target sum via Dirichlet? Let's directly sample from Dirichlet: sample gamma, then norm to sum = target, then clip?
        # This may produce values >1; after clipping we need to adjust sum => projection.
        raw = np.random.rand(dim)
        # Actually better: sample from Dirichlet shape 1: yields nonnegative sum 1, then scale by target.
        y = np.random.gamma(1.0, 1.0, size=dim)
        raw = y / y.sum() * target
        # Then projection to box (clip 0-1 and then adjust)
        feas = _project_to_simplex_box(raw, target)
        pop.append(feas)
        fitness.append(_c5_from_half(feas, n))

    best_idx = np.argmin(fitness)
    best = pop[best_idx].copy()
    best_f = fitness[best_idx]

    for gen in range(max_gen):
        for i in range(popsize):
            # mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2 = np.random.choice(idxs, 2, replace=False)
            # Use best for better convergence
            F = 0.8
            mutant = best + F * (pop[r1] - pop[r2])
            # crossover
            CR = 0.9
            trial = np.empty(dim)
            jrand = np.random.randint(dim)
            for j in range(dim):
                if np.random.rand() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            # ensure feasible: clip and project
            trial = _project_to_simplex_box(trial, target)
            trial_f = _c5_from_half(trial, n)
            if trial_f < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_f:
                    best_f = trial_f
                    best = trial.copy()
        # optional early stop if best_f < target bound
        if best_f < 0.3808:
            break
    return best, best_f

def _project_to_simplex_box(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _full_from_half(half):
    # half length m, full length n = 2*m
    return np.concatenate([half, half[::-1]])

def _c5_from_half(half):
    n = half.size * 2
    full = _full_from_half(half)
    max_corr = np.max(np.correlate(full, 1.0 - full, mode='full'))
    return max_corr * (2.0 / n)

def _ineq_constraint_sym(x):
    n = x.size - 1
    half = x[:n]
    t = x[-1]
    full = _full_from_half(half)
    corr = np.correlate(full, 1.0 - full, mode='full')
    return t - corr

def _eq_constraint_sym(x):
    n = x.size - 1
    half = x[:n]
    target = n / 4.0
    return np.sum(half) - target

def _solve_sym(half_start, max_iter=400):
    n = half_start.size
    target = n/4.0
    half0 = _project_to_simplex_box(half_start, target)
    corr0 = np.max(np.correlate(_full_from_half(half0), 1.0 - _full_from_half(half0), mode='full'))
    t0 = corr0
    x0 = np.concatenate([half0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type":"ineq","fun":_ineq_constraint_sym},
        {"type":"eq","fun":_eq_constraint_sym}
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp":False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)
    if not res.success:
        half_opt = half0
    else:
        half_opt = res.x[:n]
    half_opt = _project_to_simplex_box(half_opt, target)
    c5 = _c5_from_half(half_opt)
    return half_opt, c5

def _de_optimize_half(n, half_init, max_gen=30, popsize=12):
    dim = n // 2
    target = n / 4.0
    # initialize population
    pop = []
    fitness = []
    # first from given init
    h0 = _project_to_simplex_box(half_init, target)
    pop.append(h0)
    fitness.append(_c5_from_half(h0))
    rng = np.random.default_rng()
    while len(pop) < popsize:
        # Dirichlet(1) sample scaled to target
        y = rng.gamma(1.0, 1.0, size=dim)
        raw = y / y.sum() * target
        # projection to box
        cand = _project_to_simplex_box(raw, target)
        pop.append(cand)
        fitness.append(_c5_from_half(cand))
    best_idx = int(np.argmin(fitness))
    best = pop[best_idx].copy()
    best_f = fitness[best_idx]
    F = 0.8
    CR = 0.9
    for gen in range(max_gen):
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2 = rng.choice(idxs, 2, replace=False)
            mutant = best + F * (pop[r1] - pop[r2])
            # Crossover
            trial = np.empty(dim)
            jrand = rng.integers(dim)
            for j in range(dim):
                if rng.random() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            trial = _project_to_simplex_box(trial, target)
            trial_f = _c5_from_half(trial)
            if trial_f < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_f:
                    best_f = trial_f
                    best = trial.copy()
        if best_f < 0.3808:
            break
    return best, best_f

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x | 0≤x≤1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _full_from_half(half: np.ndarray) -> np.ndarray:
    """Reconstruct a symmetric full vector from its left half."""
    return np.concatenate([half, half[::-1]])


def _c5_from_half(half: np.ndarray) -> float:
    """C5 value for a symmetric step function defined by half."""
    full = _full_from_half(half)
    max_corr = np.max(np.correlate(full, 1.0 - full, mode="full"))
    # dx = 2 / N, N = len(full) = 2*len(half)
    return max_corr * (2.0 / (full.size))


def _ineq_constraint_sym(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift (symmetrised)."""
    n = x.size - 1                     # half length
    half = x[:n]
    t = x[-1]
    full = _full_from_half(half)
    corr = np.correlate(full, 1.0 - full, mode="full")
    return t - corr


def _eq_constraint_sym(x: np.ndarray) -> float:
    """Enforce sum(half) = (len(half))/2."""
    n = x.size - 1
    half = x[:n]
    return np.sum(half) - (n / 2.0)


def _solve_sym(half_start: np.ndarray, max_iter: int = 300):
    """Local SLSQP refinement for a symmetric half vector."""
    n = half_start.size
    target = n / 2.0
    half0 = _project_to_simplex_box(half_start, target)
    full0 = _full_from_half(half0)
    t0 = np.max(np.correlate(full0, 1.0 - full0, mode="full"))
    x0 = np.concatenate([half0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint_sym},
        {"type": "eq", "fun": _eq_constraint_sym},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    if not res.success:
        half_opt = half0
    else:
        half_opt = res.x[:n]
    half_opt = _project_to_simplex_box(half_opt, target)
    return half_opt, _c5_from_half(half_opt)

                pop[i] = trial
                fit[i] = trial_f
                if trial_f < best_f:
                    best_f = trial_f
                    best = trial.copy()
        # optional early stop handled by caller
    return best, best_f


def _local_improve_half(half: np.ndarray, max_steps: int,
                        step_frac: float = 0.02,
                        bound: float = 0.3808):
    """Simple pairwise exchange hill‑climb preserving the sum."""
    dim = half.size
    target = dim / 2.0
    cur = half.copy()
    cur_c5 = _c5_from_half(cur)
    rng = np.random.default_rng()
    for _ in range(max_steps):
        i, j = rng.integers(dim, size=2)
        if i == j:
            continue
        # maximal admissible exchange
        delta_max = min(step_frac, cur[i], 1.0 - cur[j])
        if delta_max <= 0:
            continue
        delta = rng.random() * delta_max
        trial = cur.copy()
        trial[i] -= delta
        trial[j] += delta
        # small numeric drift repair
        trial = np.clip(trial, 0.0, 1.0)
        trial = _project_to_simplex_box(trial, target)
        trial_c5 = _c5_from_half(trial)
        if trial_c5 < cur_c5:
            cur, cur_c5 = trial, trial_c5
            if cur_c5 < bound:
                break
    return cur, cur_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a symmetric step function improving the Erdős C5 bound."""
    np.random.seed(seed)
    start_time = time.time()
    def time_left() -> float:
        return budget_s - (time.time() - start_time)

    # possibly provided mother construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # candidate grid sizes (must be even)
    cand = []
    if init_h is not None:
        if init_h.size % 2 == 1:
            init_h = init_h[:-1]
        cand.append(init_h.size)
    for N in (128, 160, 192, 224, 256, 320, 384, 448):
        if N % 2 == 0:
            cand.append(N)
    cand = sorted(set(cand))

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target_bound = 0.3808

    for N in cand:
        if time_left() < 10.0:
            break
        dim = N // 2
        # ---- starting half vector --------------------------------
        if init_h is not None and init_h.size == N:
            half0 = (init_h[:dim] + init_h[-dim:][::-1]) * 0.5
        else:
            half0 = (np.arange(dim) + 0.5) * (2.0 / N)          # linear ramp (tent)
        # ---- global DE -------------------------------------------
        half_best, _ = _de_optimize_half(N, half0,
                                         max_gen=30, popsize=12)
        # ---- local SLSQP refinement -------------------------------
        half_opt, c5_opt = _solve_sym(half_best, max_iter=400)
        # ---- cheap hill‑climb if time permits --------------------
        if time_left() > 5.0 and c5_opt > target_bound:
            steps = 2000
            # adapt steps to remaining time (roughly 1e-4 s per step)
            max_allowed = int(time_left() * 5000)
            steps = min(steps, max_allowed)
            half_opt, c5_opt = _local_improve_half(half_opt,
                                                   max_steps=steps,
                                                   step_frac=0.02,
                                                   bound=target_bound)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = _full_from_half(half_opt)
            best_n = N
        if best_c5 < target_bound:
            break

    # fallback
    if best_h is None:
        N = 128
        dim = N // 2
        best_h = _full_from_half((np.arange(dim) + 0.5) * (2.0 / N))
        best_c5 = _c5_from_half(best_h[:dim])
        best_n = N

    return best_h.tolist(), float(best_c5), int(best_n)
