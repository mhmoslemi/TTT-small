def binary_hill_climb(b, rng, time_limit, overall_start):
    n = b.shape[0]
    best = b.copy()
    best_corr = np.correlate(best, 1 - best, mode='full')
    best_max = best_corr.max()
    no_improve = 0
    max_no_improve = 500
    while time.time() - overall_start < time_limit and no_improve < max_no_improve:
        # compute worst shift
        cur_corr = np.correlate(best, 1 - best, mode='full')
        cur_max = cur_corr.max()
        idx_max = np.argmax(cur_corr)
        shift = idx_max - (n - 1)
        # find bad and good positions
        if shift >= 0:
            i_vals = np.arange(0, n - shift)
            partner = i_vals + shift
        else:
            i_vals = np.arange(-shift, n)
            partner = i_vals + shift
        hi = best[i_vals]
        hp = best[partner]
        bad = i_vals[(hi == 1) & (hp == 0)]
        good = i_vals[(hi == 0) & (hp == 1)]
        improved = False
        # try a few random bad-good swaps
        attempts = 0
        max_attempts = 10
        while attempts < max_attempts and len(bad) > 0 and len(good) > 0:
            i = rng.choice(bad)
            j = rng.choice(good)
            if i == j:
                attempts += 1
                continue
            cand = best.copy()
            cand[i] = 0
            cand[j] = 1
            cand_corr = np.correlate(cand, 1 - cand, mode='full')
            cand_max = cand_corr.max()
            if cand_max < cur_max - 1e-12:
                best = cand
                best_max = cand_max
                improved = True
                no_improve = 0
                break
            attempts += 1
        if not improved:
            # random swap or small perturb
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = best.copy()
            cand[i] = 0
            cand[j] = 1
            cand_corr = np.correlate(cand, 1 - cand, mode='full')

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact C5 for a discretised step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_binary_vector(n, rng):
    """Binary vector of length n with exactly n/2 ones (n must be even)."""
    half = n // 2
    vec = np.zeros(n, dtype=int)
    vec[rng.choice(n, size=half, replace=False)] = 1
    return vec


def get_bad_good(b, shift):
    """Positions i where b[i]=1 and b[i+shift]=0 (bad) and
       b[i]=0 and b[i+shift]=1 (good) for the given shift."""
    n = b.size
    if shift >= 0:
        i_vals = np.arange(0, n - shift)
        partner = i_vals + shift
    else:
        i_vals = np.arange(-shift, n)
        partner = i_vals + shift
    hi = b[i_vals]
    hp = b[partner]
    bad = i_vals[(hi == 1) & (hp == 0)]
    good = i_vals[(hi == 0) & (hp == 1)]
    return bad, good


def binary_hill_climb(b_start, rng, time_limit, global_start):
    """Improve a binary vector by swapping bits to reduce max correlation."""
    n = b_start.size
    best = b_start.copy()
    corr = np.correlate(best, 1 - best, mode="full")
    best_max = corr.max()
    no_improve = 0
    max_no_improve = 300  # generous but bounded

    while time.time() - global_start < time_limit and no_improve < max_no_improve:
        # identify worst shift
        corr = np.correlate(best, 1 - best, mode="full")
        cur_max = corr.max()
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)

        bad, good = get_bad_good(best, shift)

        improved = False
        # try swaps that directly address the worst shift
        if bad.size and good.size:
            for _ in range(8):
                i = rng.choice(bad)
                j = rng.choice(good)
                if i == j:
                    continue
                cand = best.copy()
                cand[i] = 0
                cand[j] = 1
                cand_max = np.correlate(cand, 1 - cand, mode="full").max()
                if cand_max < cur_max - 1e-12:
                    best = cand
                    best_max = cand_max
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            # random swap (potentially escaping local minima)
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = best.copy()
            cand[i] = 0
            cand[j] = 1
            cand_max = np.correlate(cand, 1 - cand, mode="full").max()
            if cand_max < cur_max - 1e-12:
                best = cand
                best_max = cand_max
                no_improve = 0
                continue
            else:
                no_improve += 1
        else:
            continue

    final_c5 = best_max * (2.0 / n)
    return best, final_c5


# ----------------------------------------------------------------------
# SLSQP refinement on the smooth surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the sum exactly (numeric drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng_loc = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng_loc.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    TARGET_C5 = 0.3808

    # ------------------------------------------------------------------
    # Initialise solution (binary) – use provided parent if available.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        if n > 400:                     # safeguard – down‑sample if too large
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, 400, endpoint=False)
            init = np.interp(x_new, x_old, init)
            n = 400
        # binary version: set the n/2 largest entries to 1
        binary = np.zeros(n, dtype=int)
        top = np.argpartition(-init, n // 2)[: n // 2]
        binary[top] = 1
        best_h = binary
        best_c = compute_c5(best_h)
    else:
        # No parent – start directly at the highest allowed resolution.
        n = 400
        best_h = random_binary_vector(n, rng)
        best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Primary binary hill‑climbing loop
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5.0:
            break
        # diversify start points
        if rng.random() < 0.3:
            start_vec = best_h.copy()
            # perform a few random swaps for perturbation
            for _ in range(3):
                ones = np.where(start_vec == 1)[0]
                zeros = np.where(start_vec == 0)[0]
                i = rng.choice(ones)
                j = rng.choice(zeros)
                start_vec[i] = 0
                start_vec[j] = 1
        else:
            start_vec = random_binary_vector(n, rng)

        # allocate a modest slice of remaining time to this climb
        sub_time = min(12.0, time_left * 0.5)
        cand_h, cand_c = binary_hill_climb(start_vec, rng, sub_time, start_time)
        if cand_c < best_c - 1e-12:
            best_h, best_c = cand_h, cand_c
            if best_c < TARGET_C5:
                break

    # ------------------------------------------------------------------
    # Optional continuous refinement (SLSQP) if time permits.
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.95:
        remaining = budget_s - (time.time() - start_time)
        if remaining > 20.0:
            # start from current best (cast to float)
            h_float = best_h.astype(float)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                try:
                    h_float = slsqp_optimize(h_float, beta=beta, max_iter=250)
                except Exception:
                    continue
                c_val = compute_c5(h_float)
                if c_val < best_c:
                    best_h, best_c = h_float, c_val
                    if best_c < TARGET_C5:
                        break
            # final feasibility enforcement
            best_h = enforce_sum(best_h.astype(float), n / 2.0, rng)
            best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    final_h = best_h.astype(float).tolist()
    final_c = float(best_c)
    return final_h, final_c, int(n)
