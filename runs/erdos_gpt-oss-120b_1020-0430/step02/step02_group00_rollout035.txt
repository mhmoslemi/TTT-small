def binarize_by_top_half(arr):

    idx = np.argpartition(-arr, k-1)[:k]  # indices of k largest
    binary = np.zeros(n, dtype=int)

def max_corr_and_shift_fft(h):
    n = h.size
    size = 2 * n
    H = np.fft.fft(h, n=size)
    B = np.fft.fft(1 - h, n=size)
    corr = np.fft.ifft(H * np.conj(B)).real
    corr = corr[:2*n - 1]
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)
    return max_val, shift
```

import time, numpy as np, scipy.optimize as opt

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0., 1.)
    target = n / 2.
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0., 1.)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0., 1.)

def _c5_from_h(h):
    n = h.size
    max_corr = np.max(np.correlate(h, 1. - h, mode='full'))
    return max_corr * (2. / n)

def _objective(x):
    return x[-1]

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1. - h, mode='full')

def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.)

def _solve_h(initial_h, max_iter=300):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1. - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0., 1.)] * n + [(0., None)]
    cons = [{'type': 'ineq', 'fun': _ineq_constraint},
            {'type': 'eq', 'fun': _eq_constraint}]
    opt_res = opt.minimize(_objective, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    h_opt = opt_res.x[:n] if opt_res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

def _max_corr_and_shift_fft(h):
    n = h.size
    size = 2 * n
    H = np.fft.fft(h, n=size)
    B = np.fft.fft(1. - h, n=size)
    corr = np.fft.ifft(H * np.conj(B)).real
    corr = corr[:2 * n - 1]
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)

def _binarize_top_half(arr):
    n = arr.size
    k = n // 2
    idx = np.argpartition(-arr, k-1)[:k]
    binv = np.zeros(n, dtype=int)
    binv[idx] = 1
    return binv

def _upscale_binary(coarse, n_fine):
    n_coarse = coarse.size
    # linear interpolation on indices
    x_coarse = np.linspace(0., 1., n_coarse, endpoint=False)
    x_fine = np.linspace(0., 1., n_fine, endpoint=False)
    interp = np.interp(x_fine, x_coarse, coarse.astype(float))
    return _binarize_top_half(interp)

def _hill_climb_binary(start_h, max_seconds):
    n = start_h.size
    h = start_h.astype(float).copy()
    start = time.time()
    best_corr, _ = _max_corr_and_shift_fft(h)
    best_corr = best_corr
    attempts_without_improve = 0
    while time.time() - start < max_seconds:
        cur_corr, shift = _max_corr_and_shift_fft(h)
        # recompute only once per loop, but cur_corr may equal best_corr.
        # identify mismatched indices for this shift
        if shift >= 0:
            k = shift
            rng = np.arange(n - k)
            mis1 = rng[(h[rng] == 1) & (h[rng + k] == 0)]
            mis0 = rng[(h[rng] == 0) & (h[rng + k] == 1)]
        else:
            k = -shift
            rng = np.arange(k, n)
            mis1 = rng[(h[rng] == 1) & (h[rng - k] == 0)]
            mis0 = rng[(h[rng] == 0) & (h[rng - k] == 1)]
        if mis1.size and mis0.size:
            i = mis1[np.random.randint(mis1.size)]
            j = mis0[np.random.randint(mis0.size)]
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            new_corr, _ = _max_corr_and_shift_fft(h_new)
            if new_corr + 1e-12 < best_corr:
                h = h_new
                best_corr = new_corr
                attempts_without_improve = 0
                continue
        # fallback random swap
        idx_one = np.flatnonzero(h == 1)
        idx_zero = np.flatnonzero(h == 0)
        if idx_one.size == 0 or idx_zero.size == 0:
            break
        i = idx_one[np.random.randint(idx_one.size)]
        j = idx_zero[np.random.randint(idx_zero.size)]
        h_new = h.copy()
        h_new[i], h_new[j] = 0., 1.
        new_corr, _ = _max_corr_and_shift_fft(h_new)
        if new_corr + 1e-12 < best_corr:
            h = h_new
            best_corr = new_corr
            attempts_without_improve = 0
        else:
            attempts_without_improve += 1
        if attempts_without_improve > 500:
            break
    return h, best_corr * (2. / n)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # candidate sizes

    n_vals = [n for n in n_vals if n % 2 == 0]
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # initialise from parent if present
    try:
        parent = np.array(initial_h_values, dtype=float)
        parent_bin = _binarize_top_half(parent)
    except NameError:
        parent = None
        parent_bin = None

    # main loop over sizes
    for n in n_vals:
        if time_left() < 8.0:
            break
        # build list of seeds
        seeds = []
        if parent_bin is not None:
            if parent_bin.size == n:
                seeds.append(parent_bin.copy())
            elif parent_bin.size < n:
                seeds.append(_upscale_binary(parent_bin, n))
            else:
                # downscale by taking top half after interpolation
                seeds.append(_binarize_top_half(np.interp(
                    np.linspace(0., 1., n, endpoint=False),
                    np.linspace(0., 1., parent_bin.size, endpoint=False),
                    parent_bin.astype(float))))
        # some random binaries
        for _ in range(3):
            b = np.zeros(n, dtype=int)
            b[:n//2] = 1
            np.random.shuffle(b)
            seeds.append(b)

        # allocate time per seed roughly

        for seed_vec in seeds:
            if time_left() < 4.0:
                break
            h_opt, c5_val = _hill_climb_binary(seed_vec.astype(float), per_seed)
            if c5_val < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_val
                best_n = n
            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # final polishing with SLSQP if time permits
    if best_h is not None and time_left() > 6.0:
        h_refined, c5_ref = _solve_h(best_h.astype(float), max_iter=800)
        if c5_ref < best_c5:
            best_h, best_c5 = h_refined, c5_ref
            best_n = best_h.size

    # fallback
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    else:
        best_h = _project_to_feasible(best_h, best_h.size)
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)

# ----------------------------------------------------------------------
# Feasibility projection (box + sum constraint)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

# ----------------------------------------------------------------------
def _objective(x: np.ndarray) -> float:
    return x[-1]


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300):

    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "ineq", "fun": _ineq_constraint},
            {"type": "eq", "fun": _eq_constraint}]
    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                       constraints=cons, options={"maxiter": max_iter,
                                                  "ftol": 1e-9,
                                                  "disp": False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)

def _max_corr_and_shift_fft(h: np.ndarray):

    size = 2 * n
    H = np.fft.fft(h, n=size)
    B = np.fft.fft(1.0 - h, n=size)
    corr = np.fft.ifft(H * np.conj(B)).real
    corr = corr[: 2 * n - 1]
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)
