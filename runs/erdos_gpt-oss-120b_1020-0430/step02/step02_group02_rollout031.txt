def solve_binary_ilp(n):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n // 2]
    # list for shift constraints
    pairs_i = []
    pairs_j = []
    shift_pair_idxs = [[] for _ in range(2*n-1)]
    shift_i_lists = [[] for _ in range(2*n-1)]
    for s in range(-(n-1), n):
        idx_shift = s + n - 1
        if s >= 0:
            i_vals = list(range(0, n-s))
            j_vals = list(range(s, n))
        else:
            i_vals = list(range(-s, n))
            j_vals = list(range(0, n+s))
        # for each overlapping i
        for i_val, j_val in zip(i_vals, j_vals):
            pair_idx = len(pairs_i)
            pairs_i.append(i_val)
            pairs_j.append(j_val)
            shift_pair_idxs[idx_shift].append(pair_idx)
        # store i index for O_s sum of h_i
        shift_i_lists[idx_shift].extend(i_vals)
    # After loops, define w variables
    m = len(pairs_i)
    w = cp.Variable(m, boolean=True)
    # Add product constraints
    for p in range(m):
        i = pairs_i[p]
        j = pairs_j[p]
        constraints.append(w[p] <= h[i])
        constraints.append(w[p] <= h[j])
        constraints.append(w[p] >= h[i] + h[j] - 1)
    # O_s <= t for each shift
    for idx_shift in range(2*n-1):
        i_list = shift_i_lists[idx_shift]
        w_idx_list = shift_pair_idxs[idx_shift]
        if len(i_list) > 0:
            expr = cp.sum(cp.vstack([h[i] for i in i_list])) - cp.sum(cp.vstack([w[p] for p in w_idx_list]))
        else:
            expr = 0
        constraints.append(expr <= t)
    # t >= 0
    constraints.append(t >= 0)
    # objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, warm_start=True, max_iters=100000)
    except cp.error.SolverError:
        # fallback - return None
        return None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback
        return None
    h_val = np.array(h.value).astype(float)
    t_val = int(round(t.value))
    return h_val, t_val

def _solve_h(initial_h, max_iter=400):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    # compute initial t as max correlation of h0
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": lambda x: x[-1] - np.correlate(x[:n], 1.0 - x[:n], mode="full")},
        {"type": "eq", "fun": lambda x: np.sum(x[:n]) - n/2.0},
    ]
    res = opt.minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds,
                       constraints=constraints,
                       options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

import time
import numpy as np
import scipy.optimize as opt
import cvxpy as cp

# Helper functions

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a step function already respecting the integral constraint."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """SLSQP refinement of a feasible h to minimise the maximal overlap."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    # initial bound for t
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        return x[-1] - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = opt.minimize(lambda x: x[-1], x0, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})

    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Simple pairwise epsilon exchange hill climbing."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        eps_max = min(max_down, max_up)
        if eps_max <= 0:
            continue
        eps = np.random.rand() * eps_max
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c + 1e-12 < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _binary_ilp(n: int):
    """Solve a MILP for a binary step function minimizing the maximal overlap."""
    if n % 2 != 0:
        return None
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)

    constraints = [cp.sum(h) == n // 2]

    # build pair list and shift mappings
    pairs_i = []
    pairs_j = []
    shift_pairs = [[] for _ in range(2 * n - 1)]
    shift_is = [[] for _ in range(2 * n - 1)]

    for s in range(-(n - 1), n):
        idx = s + n - 1
        if s >= 0:
            i_vals = np.arange(0, n - s)
            j_vals = i_vals + s
        else:
            i_vals = np.arange(-s, n)
            j_vals = i_vals + s
        shift_is[idx].extend(i_vals.tolist())
        for i_val, j_val in zip(i_vals, j_vals):
            pair_idx = len(pairs_i)
            pairs_i.append(int(i_val))
            pairs_j.append(int(j_val))
            shift_pairs[idx].append(pair_idx)

    m = len(pairs_i)
    w = cp.Variable(m, boolean=True)

    # product linearisation
    for p in range(m):
        i = pairs_i[p]
        j = pairs_j[p]
        constraints += [
            w[p] <= h[i],
            w[p] <= h[j],
            w[p] >= h[i] + h[j] - 1,
        ]

    # O_s <= t for each shift
    for idx in range(2 * n - 1):
        sum_h = sum([h[i] for i in shift_is[idx]]) if shift_is[idx] else 0
        sum_w = sum([w[p] for p in shift_pairs[idx]]) if shift_pairs[idx] else 0
        constraints.append(sum_h - sum_w <= t)

    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, warm_start=True)
    except Exception:
        return None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    h_val = np.array(h.value).astype(float)
    t_val = int(round(t.value))
    return h_val, t_val
