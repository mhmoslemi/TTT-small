import time
import numpy as np

# ----------------------------------------------------------------------
# Feasibility projection (water‑filling)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# Evaluation helpers
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Simple hill‑climbing (exchange epsilon between two coordinates)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]          # we can take at most this
        max_up = 1.0 - best_h[j]      # we can give at most this
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on   max_i corr_i
# ----------------------------------------------------------------------
def _subgradient_opt(h0: np.ndarray, max_iters: int, time_limit: float, step0: float = 0.5) -> tuple:
    h = _project_to_feasible(h0.copy(), h0.size)
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_from_h(h)
    offset = n - 1
    for it in range(1, max_iters + 1):
        if time.time() > time_limit:
            break
        corrs = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corrs.max()
        c_val = max_corr * dx
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        # active shifts (indices achieving the max)
        eps = 1e-12
        active_idx = np.where(corrs >= max_corr - eps)[0]
        if active_idx.size == 0:
            continue
        grad = np.zeros_like(h)
        for idx in active_idx:
            i = idx - offset
            i_abs = abs(i)
            if i_abs >= n:
                continue
            # forward shift (h[k+i_abs])
            fwd = np.zeros_like(h)
            if i_abs < n:
                fwd[: n - i_abs] = h[i_abs:]
            # backward shift (h[k-i_abs])
            back = np.zeros_like(h)
            if i_abs > 0:
                back[i_abs:] = h[: n - i_abs]
            else:  # i_abs == 0 -> both terms are h itself
                back = h.copy()
            grad -= (fwd + back)
        grad /= active_idx.size
        step = step0 / np.sqrt(it)
        h = h - step * grad  # move opposite to sub‑gradient of the max
        h = _project_to_feasible(h, n)
    return best_h, best_c
