def upsample_h(h, n_target):
    n_src = h.size
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)

def repair_sum_and_bounds(h, target, rng):
    # clip
    h = np.clip(h, 0., 1.)
    diff = target - h.sum()
    # while diff not near 0
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # choose random index where we can adjust in needed direction
        if diff > 0:
            # we need to increase some entries not at 1
            idx = rng.choice(np.where(h < 1)[0])
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.choice(np.where(h > 0)[0])
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h
```

import time, numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with len(h) points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def repair_sum_and_bounds(h, target, rng):
    """Clip to [0,1] and adjust to meet the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple iterative repair; diff will be small after most constructions
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass; choose an index where h<1
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            # need to remove mass; choose an index where h>0
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def upsample_h(h, n_target, rng):
    """Linear interpolation of h from its length to n_target points, then repair."""
    n_src = h.size
    if n_target == n_src:
        return h.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)
    h_up = np.interp(x_tgt, x_src, h)
    return repair_sum_and_bounds(h_up, n_target / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2500, start_step=0.5):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=6):
    """Few gradient steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = repair_sum_and_bounds(h, n / 2.0, rng)
    return h

# ---------- SLSQP surrogate optimisation ----------

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum repair (in case of tiny violation)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = repair_sum_and_bounds(x, CURRENT_TARGET, rng)
    return x

# ---------- main optimisation loop ----------
