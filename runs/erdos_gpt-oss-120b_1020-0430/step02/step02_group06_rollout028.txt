def subgradient_descent(h_init, max_iter, initial_eta, eta_decay):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = c5_from_h(h)
    eta = initial_eta
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        shift = max_idx - (len(h) - 1)
        # compute gradient:
        grad = np.zeros_like(h)
        n = len(h)
        if shift >= 0:
            grad[shift:] += 1 - h[:n-shift]
            grad[:n-shift] -= h[shift:]
        else:
            t = -shift
            grad[:n-t] += 1 - h[t:]
            grad[t:] -= h[:n-t]
        # step
        h = h - eta * grad
        # project to feasible
        h = _project_to_feasible(h, n)
        # evaluate
        c5 = _c5_from_h(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        # optionally reduce eta if not improving
        # simple schedule: eta *= eta_decay every iteration
        eta *= eta_decay
    return best_h, best_c5

def _subgradient_optimize(h_init, max_iter, eta0=0.5, decay=0.99):
    h = h_init.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    eta = eta0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        # gradient
        grad = np.zeros_like(h)
        if max_shift >= 0:
            # first term for i>=shift
            grad[max_shift:] += 1 - h[:n - max_shift]
            # second term for i < n - shift
            grad[:n - max_shift] -= h[max_shift:]
        else:
            t = -max_shift
            grad[:n - t] += 1 - h[t:]
            grad[t:] -= h[:n - t]
        # step
        h_new = h - eta * grad
        # project
        h = _project_to_feasible(h_new, n)
        # evaluate
        c5 = _c5_from_h(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        eta *= decay
    return best_h, best_c5

def _subgradient_optimize(h_start, max_iter, eta0=0.5, decay=0.999):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    eta = eta0
    for it in range(max_iter):
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        if max_shift >= 0:
            # first term for i>=shift
            grad[max_shift:] += 1.0 - h[:n - max_shift]
            # second term for i < n - shift
            grad[:n - max_shift] -= h[max_shift:]
        else:
            t = -max_shift
            grad[:n - t] += 1.0 - h[t:]

import time
import numpy as np

# -------------------- Helper utilities -------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02) -> np.ndarray:
    """Stochastic hill‑climbing around h to squeeze a few extra digits."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


def _subgradient_optimize(
    h_start: np.ndarray,
    max_iter: int,
    eta0: float = 0.5,
    decay: float = 0.999,
) -> tuple:
    """Subgradient descent on max ∫ h·(1‑hₖ) dx."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    eta = eta0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)  # integer shift corresponding to max overlap
        # gradient of the worst‑shift term
        grad = np.zeros_like(h)
        if shift >= 0:
            grad[shift:] += 1.0 - h[: n - shift]
            grad[: n - shift] -= h[shift:]
        else:
            t = -shift
            grad[: n - t] += 1.0 - h[t:]
            grad[t:] -= h[: n - t]
        # descent step and projection
        h = _project_to_feasible(h - eta * grad, n)
        c5 = _c5_from_h(h)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h = c5, h.copy()
        eta *= decay
    return best_h, best_c5
