def hill_climb(h, max_iter, step):
    best = h.copy()
    best_val = C5_from_h(best)
    for it in range(max_iter):
        # choose random i!=j
        i, j = np.random.randint(n, size=2)
        if i==j: continue
        # compute max possible delta respecting bounds
        delta_max = min(step, best[i], 1 - best[j])
        if delta_max <= 0: continue
        delta = delta_max * np.random.rand()
        # propose move
        new_h = best.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # evaluate
        val = C5_from_h(new_h)
        if val < best_val - 1e-12:
            best = new_h
            best_val = val
    return best, best_val

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute to any free indices
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            delta = residual / free.size
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback scaling
            h = h * (target / h.sum())
    return np.clip(h, 0.0, 1.0)

def _solve_slsqp(initial_h, max_iter=500):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0-h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)]*n + [(0.0, None)]
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr
    def eq(x):
        return np.sum(x[:n]) - n/2.0
    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    try:
        res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                                constraints=cons, options=opts)
    except Exception:
        res = None
    if res is not None and res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5

import time
import numpy as np
import scipy.optimize as optimize

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            h[free] += residual / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            h = h * (target / h.sum())
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a feasible step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_slsqp(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Run SLSQP on h with auxiliary variable t."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [{"type": "ineq", "fun": ineq},
            {"type": "eq", "fun": eq}]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                                bounds=bounds, constraints=cons, options=opts)
    except Exception:
        res = None

    if res is not None and res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

    # Load optional parent construction
    # ------------------------------------------------------------------

    except NameError:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Candidate grid sizes (even, modest)
    # ------------------------------------------------------------------
    n_candidates = [128, 256, 512]
    n_candidates = [n for n in n_candidates if n <= 800]  # safety cap
    n_candidates.sort()

    # ------------------------------------------------------------------
    # Stage 1 – many SLSQP restarts on each size
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 12.0:
            break

        # build a sensible starting vector for this resolution
        if init_h is not None and init_h.size == n:
            base = init_h.copy()
        elif init_h is not None:

        best_h, best_c5 = h, cur_val
        best_n = h.size

    # ------------------------------------------------------------------
    # Optional final SLSQP polish (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 5.0:

    best_n = int(best_h.size)
