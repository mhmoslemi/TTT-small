def _c5_from_h(h):
    n = h.shape[0]
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

def _upscale_h(coarse_h, n_fine):
    # linear interpolation using np.interp on midpoints
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _project_h(v, target):
    # binary search for lambda to satisfy sum clip(v - lam, 0,1) = target
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # adjust tiny residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # distribute residual to non-bound elements
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            add = residual / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
            # final small correction
            residual2 = target - x.sum()
            if abs(residual2) > 1e-12:
                # adjust one element
                idx = free[0]
                x[idx] += residual2
                x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy import interpolate

# ---------- Core utilities ----------

def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse discretisation onto a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


def _project_h(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x in [0,1]^n, sum(x)=target_sum}.
    Uses a monotonous binary search for the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    # binary search for lambda
    for _ in range(40):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lo, 0.0, 1.0)
    # final tiny correction (rare)
    resid = target_sum - x.sum()
    if abs(resid) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            add = resid / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_correlation(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Sub‑gradient of the correlation term c_shift = sum_i h[i]*(1-h[i+shift])
    with respect to h. Handles both positive and negative shifts.
    """
    n = h.size
    start = max(0, -shift)
    end = min(n - 1, n - 1 - shift)
    i = np.arange(start, end + 1)
    j = i + shift
    grad = np.zeros_like(h)
    grad[i] += 1.0 - h[j]     # derivative w.r.t. the first factor
    grad[j] += -h[i]          # derivative w.r.t. the second factor
    return grad


def _descend_h(
    h0: np.ndarray,
    target_sum: float,
    max_iters: int,
    time_left,
    init_step: float = 0.25,
) -> tuple:
    """
    Projected sub‑gradient descent on the max‑correlation objective.
    Returns (h_opt, c5_opt).
    """
    h = h0.copy()
    cur_c5 = _c5_from_h(h)
    eta = init_step
    n = h.size

    for it in range(max_iters):
        if time_left() < 0.5:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        idxs = np.where(corr == max_val)[0]

        # average gradient over all maximising shifts (helps when several tie)
        grad = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n - 1)
            grad += _grad_correlation(h, shift)
        grad /= len(idxs)

        # trial step
        h_trial = _project_h(h - eta * grad, target_sum)
        new_c5 = _c5_from_h(h_trial)

        if new_c5 < cur_c5 - 1e-12:
            h = h_trial
            cur_c5 = new_c5
            eta *= 1.02
        else:
            eta *= 0.5
            if eta < 1e-6:
                eta = init_step

    return h, cur_c5

    """Create a random feasible vector (uniform, then projected)."""
    v = np.random.rand(n)
    return _project_h(v, target_sum)


# ---------- Main optimisation driver ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # parent construction if supplied by the harness
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # candidate grid sizes (even only, up to 256)
    base_sizes = [80, 96, 112, 128, 144, 160, 176, 192, 224, 256]
    n_candidates = sorted(set(base_sizes + ([parent_h.size] if parent_h is not None else [])))

    for n in n_candidates:
        if time_left() < 4.0:
            break
        target_sum = n / 2.0

        # base vector (parent up‑scaled if possible)
        if parent_h is not None:
            if parent_h.size != n:
                base = _upscale_h(parent_h, n)
            else:
                base = parent_h.copy()
        else:
            base = np.full(n, 0.5)

        base = _project_h(base, target_sum)

        # evaluate the base candidate
        base_c5 = _c5_from_h(base)
        if base_c5 < best_c5:
            best_h, best_c5, best_n = base.copy(), base_c5, n
            if best_c5 < target_bound:
                break

        # stochastic restarts + gradient refinement
        restarts = max(3, int(min(15, time_left() // 6)))
        sigma = 0.35
        for _ in range(restarts):
            if time_left() < 3.0:
                break
            h0 = base + sigma * np.random.randn(n)
            h0 = _project_h(h0, target_sum)

            # number of descent iterations adaptively limited by remaining time
            max_it = int(min(5000, max(500, time_left() * 30)))
            h_opt, c5_opt = _descend_h(h0, target_sum, max_it, time_left)

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                if best_c5 < target_bound:
                    break
            sigma *= 0.92

        if best_c5 < target_bound:
            break

    # Fallback (should never happen)
    if best_h is None:
        n = 96
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
