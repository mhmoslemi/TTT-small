def _gradient_descent_h(start_h, n, max_iter, step0, step_decay, beta0, beta_factor, target_bound, time_left):
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    step = step0
    beta = beta0
    for it in range(max_iter):
        if time_left() < 1.0:
            break
        # compute a
        a_full = np.correlate(h, h, mode='full')
        a_pos = a_full[n-1:]
        a_min = a_pos.min()
        # weights
        w = np.exp(-beta * (a_pos - a_min))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum
        # gradient via correlation
        corr_f = np.correlate(w, h, mode='full')
        g_f = corr_f[n-1:n-1+n]
        h_rev = h[::-1]
        corr_rev = np.correlate(w, h_rev, mode='full')
        g_b = corr_rev[:n][::-1]
        grad = g_f + g_b
        # step
        h_new = h + step * grad
        # projection
        h_new = _project_to_simplex_box(h_new, n/2.0)
        # symmetrize
        h_new = (h_new + h_new[::-1]) * 0.5
        # project again to ensure sum (optional)
        h_new = _project_to_simplex_box(h_new, n/2.0)
        h = h_new
        # evaluate
        c5_val = _c5_from_h(h)
        if c5_val < best_c5:
            best_c5 = c5_val; best_h = h.copy()
            if best_c5 < target_bound:
                break
        # schedule
        step *= step_decay
        beta = min(beta * beta_factor, 50.0)  # cap at 50
    return best_h, best_c5

def _project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target

        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:

    beta = beta0
    for it in range(max_iter):
        if time_left() < 1.0:
            break
        # compute autocorrelation
        a_full = np.correlate(h, h, mode='full')
        a_pos = a_full[n-1:]   # length n, lag 0..n-1
        a_min = a_pos.min()
        # softmin weights
        w = np.exp(-beta * (a_pos - a_min))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient via correlations
        corr_f = np.correlate(w, h, mode='full')
        g_f = corr_f[n-1:n-1+n]
        h_rev = h[::-1]
        corr_rev = np.correlate(w, h_rev, mode='full')
        g_b = corr_rev[:n][::-1]
        grad = g_f + g_b
        # step and project
        h_new = h + step * grad
        h_new = _project_to_simplex_box(h_new, n/2.0)
        # symmetrize
        h_new = (h_new + h_new[::-1]) * 0.5
        h_new = _project_to_simplex_box(h_new, n/2.0)

            best_h = h.copy()
            if best_c5 < target_bound:

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto 0≤x≤1 and sum(x)=target."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)

        if s > target:

        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)

        if free.size:
