def _h_plateau(p, n):
    # p in [0, 0.5]
    h0 = 1.0 / (1.0 + p)  # peak height
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    # left ramp: x <= 1 - p
    # plateau: (1-p) < x < (1 + p)
    # right ramp: x >= 1 + p
    left_mask = x <= (1.0 - p)
    right_mask = x >= (1.0 + p)
    mid_mask = ~(left_mask | right_mask)
    # left ramp: h = h0 * (x / (1 - p))
    # careful: at x=0, h=0.
    h = np.empty_like(x)
    if 1.0 - p > 0:
        h[left_mask] = h0 * (x[left_mask] / (1.0 - p))
    else:
        h[left_mask] = 0.0
    h[mid_mask] = h0
    # right ramp: h = h0 * ((2 - x) / (1 - p))
    if 1.0 - p > 0:
        h[right_mask] = h0 * ((2.0 - x[right_mask]) / (1.0 - p))
    else:
        h[right_mask] = 0.0
    # maybe due to floating rounding near boundaries, small epsilon; fine.
    # projection onto simplex & box
    return _project_to_simplex_box(h, n / 2.0)

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr

def _eq_constraint(x):
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)

def _objective(x):
    return x[-1]

def _solve_h(start_h, max_iter=300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

def _make_h_plateau(p, n):
    # p in [0,0.5]
    h0 = 1.0 / (1.0 + p)
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    # left ramp condition
    left_end = 1.0 - p
    right_start = 1.0 + p
    # avoid division by zero
    denom = max(1.0 - p, 1e-12)
    h = np.empty_like(x)
    left = x <= left_end
    right = x >= right_start
    mid = ~(left | right)
    h[left] = h0 * (x[left] / denom)
    h[mid] = h0
    h[right] = h0 * ((2.0 - x[right]) / denom)
    # project onto feasible simplex box
    return _project_to_simplex_box(h, n / 2.0)
```

import time
import numpy as np
from scipy import optimize

# ---------- Helper functions (top level) ----------

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x)=target}.
    Uses a simple binary search on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny correction if needed
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """Inequality: t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Equality: sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Objective: minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Run a single SLSQP refinement starting from start_h."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape (min(x,2-x)) sampled on n points."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _make_h_plateau(p: float, n: int) -> np.ndarray:
    """
    Symmetric plateau shape.
    p ∈ [0,0.5] is half the plateau width.
    Height h0 = 1/(1+p) guarantees total integral = 1.
    """
    h0 = 1.0 / (1.0 + p)
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    left_end = 1.0 - p
    right_start = 1.0 + p
    denom = max(1.0 - p, 1e-12)  # avoid div‑by‑zero when p≈1
    h = np.empty_like(x)
    left = x <= left_end
    right = x >= right_start
    middle = ~(left | right)
    h[left] = h0 * (x[left] / denom)
    h[middle] = h0
    h[right] = h0 * ((2.0 - x[right]) / denom)
    return _project_to_simplex_box(h, n / 2.0)


class _PlateauC5:
    """Callable object for minimise_scalar that evaluates C5 for a given p."""
    def __init__(self, n: int):
        self.n = n

    def __call__(self, p: float) -> float:
        h = _make_h_plateau(p, self.n)
        return _c5_from_h(h)
