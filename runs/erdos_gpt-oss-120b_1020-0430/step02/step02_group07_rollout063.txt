def hill_climb_swap(h, rng, max_iters=2000):
    n = len(h)
    # Precompute list of positions of ones and zeros
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    current_c = compute_c5(h)
    for _ in range(max_iters):
        # pick random one and zero
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        # compute c
        c_new = compute_c5(h_new)
        if c_new < current_c - 1e-12:  # improvement
            h = h_new
            current_c = c_new
            # update positions
            # i became zero, j became one.
            # Update arrays
            # For efficiency, we could just recompute ones and zeros each iteration - cost moderate.
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return h, current_c

def simulated_annealing_swap(h, rng, max_iters=20000, init_temp=0.05, decay=0.9995):
    n = len(h)
    curr_h = h.copy()
    curr_c = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_c = curr_c
    temp = init_temp
    # Precompute ones and zeros each iteration may be updated.
    for it in range(max_iters):
        # propose swap
        # choose randomly if ones or zeros have length zero (should not).
        ones_idx = np.where(curr_h > 0.5)[0]
        zeros_idx = np.where(curr_h < 0.5)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        new_h = curr_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        # acceptance
        if new_c < curr_c or rng.random() < np.exp((curr_c - new_c) / temp):
            curr_h, curr_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        temp *= decay
        # optionally stop early if temp low
    return best_h, best_c

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary(n, rng):
    half = n // 2
    h = np.zeros(n, dtype=float)
    indices = rng.choice(n, size=half, replace=False)
    h[indices] = 1.0
    return h

def try_improve_worst_shift(h, rng):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    # argmax (first)
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    # Determine overlapping index ranges
    if k >= 0:
        i_range = np.arange(0, n - k)
        j_range = i_range + k
    else:
        k_abs = -k
        i_range = np.arange(k_abs, n)
        j_range = i_range - k_abs
    hi = h[i_range]
    hj = h[j_range]
    # candidate sets
    mask_i = (hi == 1) & (hj == 0)
    mask_j = (hi == 0) & (hj == 1)
    I = i_range[mask_i]
    J = i_range[mask_j]  # these are i positions where hi=0 & hj=1, so they are zeros with counterpart one
    if I.size == 0 or J.size == 0:
        return h, False
    # pick random
    i_sel = rng.choice(I)
    # j_sel is index from J where it's zero currently; we need to swap values at i_sel (1) with j_sel (0)
    j_sel = rng.choice(J)
    new_h = h.copy()
    new_h[i_sel] = 0.0
    new_h[j_sel] = 1.0
    return new_h, True

def local_search(h, rng, time_limit, start_time):
    curr_h = h.copy()
    curr_c = compute_c5(curr_h)
    best_h, best_c = curr_h.copy(), curr_c
    # prepare quick arrays of zeros and ones
    while time.time() - start_time < time_limit:
        # try targeted improvement
        new_h, improved = try_improve_worst_shift(curr_h, rng)
        if not improved:
            # random swap
            ones = np.where(curr_h > 0.5)[0]
            zeros = np.where(curr_h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            new_h = curr_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0
        new_c = compute_c5(new_h)
        # accept if improvement
        if new_c < curr_c - 1e-12:
            curr_h, curr_c = new_h, new_c
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h.copy(), new_c
        else:
            # occasional uphill acceptance with small probability decreasing over time
            # compute probability based on difference and remaining time
            # simple: accept with 0.001 prob
            if rng.random() < 0.001:
                curr_h, curr_c = new_h, new_c
        # optional early break if already reach target 0.3808
        if best_c <= 0.3808:
            break
    return best_h, best_c

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    # sum must be n/2
    target = n // 2
    cur_sum = int(round(best_h.sum()))
    diff = target - cur_sum
    if diff != 0:
        # correct by flipping
        if diff > 0:
            zeros = np.where(best_h < 0.5)[0]
            idx = rng.choice(zeros, diff, replace=False)

            idx = rng.choice(ones, -diff, replace=False)
            best_h[idx] = 0.0
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

def subgradient_max_overlap(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]  # active indices
    g = np.zeros(n)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            i = np.arange(0, n - k)
            j = i + k
            g[i] += (1 - h[j])
            g[j] -= h[i]
        else:
            k_abs = -k
            i = np.arange(k_abs, n)
            j = i - k_abs
            g[i] += (1 - h[j])
            g[j] -= h[i]
    # average over active shifts
    g = g / len(idxs)
    return g

def enforce_sum(h, target, rng, max_iter=1000):
    diff = target - h.sum()
    n = len(h)
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # adjust at i within bounds
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    if h.sum() == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(target), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


# --------------------------------------------------------------
# Sub‑gradient optimisation for the max‑overlap objective
# --------------------------------------------------------------
def active_shifts(corr, n, tol=1e-12):
    """Return the list of integer shifts achieving the maximal correlation."""
    max_val = np.max(corr)
    idxs = np.where(np.abs(corr - max_val) <= tol)[0]
    return idxs - (n - 1)   # translate to lag ∈ [-(n-1), …, n-1]


def subgradient(h, shifts):
    """Sub‑gradient of max_k Σ h_i(1‑h_{i+k}) for the active shifts."""
    n = h.size
    g = np.zeros(n)
    for k in shifts:
        if k >= 0:
            i = np.arange(0, n - k)
            j = i + k
        else:
            kk = -k
            i = np.arange(kk, n)
            j = i - kk
        hi = h[i]
        hj = h[j]
        g[i] += (1.0 - hj)
        g[j] -= hi
    return g / max(1, len(shifts))


def projected_subgradient_optimize(h0, rng, time_limit, start_time):
    """Run a projected sub‑gradient method until the given wall‑time expires."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = 0.5

    while time.time() - start_time < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        shifts = active_shifts(corr, n)
        g = subgradient(h, shifts)

        # direction that respects Σh = const: project to zero‑sum subspace
        d = -g + np.mean(g)

        # maximal feasible step respecting bounds
        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            step_pos = np.min((1.0 - h)[pos] / d[pos])
        else:
            step_pos = np.inf
        if np.any(neg):
            step_neg = np.min((-h)[neg] / d[neg])
        else:
            step_neg = np.inf
        max_step = min(step_pos, step_neg)

        if max_step <= 0:
            step *= 0.5
            continue

        eta = min(step, 0.9 * max_step)
        h_new = h + eta * d
        h_new = np.clip(h_new, 0.0, 1.0)

        # small correction to keep the exact sum
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            free = np.where((h_new > 0.0) & (h_new < 1.0))[0]
            if free.size:
                i = rng.choice(free)
                h_new[i] += diff
                h_new = np.clip(h_new, 0.0, 1.0)
            else:
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)

        c_new = compute_c5(h_new)

        if c_new < best_c - 1e-12:
            best_h, best_c = h_new.copy(), c_new

        # accept if we did not worsen the objective (sub‑gradient descent)
        if c_new <= compute_c5(h) + 1e-12:
            h = h_new
        else:
            step *= 0.7

        step *= 0.99
        if best_c <= 0.3808:
            break

    return best_h, best_c

def local_refine_transfer(h, rng, iters=2000):
    """Simple transfer moves that keep the sum exactly constant."""
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)

    for _ in range(iters):
        pos = np.where(best_h > 1e-12)[0]
        neg = np.where(best_h < 1.0 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        eps = min(best_h[i], 1.0 - best_h[j]) * rng.random() * 0.5
        if eps <= 0:
            continue
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c

    # final exact feasibility (tiny numerical drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    return best_h, best_c
