def binary_hill_climb(h, max_steps, time_limit):
    n = h.size
    # compute auto and C5
    auto = np.correlate(h, h, mode='full')
    cur_min = auto.min()
    cur_c5 = (n/2 - cur_min) * (2.0/n)
    # build ones and zeros lists
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    best_h = h.copy()
    best_c5 = cur_c5
    best_auto = auto.copy()
    steps_without_improve = 0
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        # pick random swap
        pos_i = np.random.randint(len(ones_idx))
        i = int(ones_idx[pos_i])
        pos_j = np.random.randint(len(zeros_idx))
        j = int(zeros_idx[pos_j])

        # compute other ones
        if len(ones_idx) <= 1:  # edge case not happen
            continue
        # p_other array
        if pos_i == 0:
            p_other = ones_idx[1:]
        elif pos_i == len(ones_idx)-1:
            p_other = ones_idx[:-1]
        else:
            p_other = np.concatenate((ones_idx[:pos_i], ones_idx[pos_i+1:]))

        # compute idx arrays
        n_offset = n-1
        idx_i1 = p_other - i + n_offset
        idx_i2 = i - p_other + n_offset
        idx_j1 = p_other - j + n_offset
        idx_j2 = j - p_other + n_offset

        # update auto copy
        new_auto = auto.copy()
        new_auto[idx_i1] -= 1
        new_auto[idx_i2] -= 1
        new_auto[idx_j1] += 1
        new_auto[idx_j2] += 1

        new_min = new_auto.min()
        new_c5 = (n/2 - new_min) * (2.0/n)

        # accept if improvement
        if new_c5 < cur_c5 - 1e-14:
            # accept
            auto = new_auto
            cur_c5 = new_c5
            cur_min = new_min
            # update h, ones_idx, zeros_idx
            h[i] = 0
            h[j] = 1
            ones_idx[pos_i] = j
            zeros_idx[pos_j] = i
            steps_without_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_auto = auto.copy()
        else:
            # maybe accept uphill with low prob (optional)
            steps_without_improve += 1
            # reject: no changes

                # stagnated: break early
                break
    return best_h, best_c5

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for the binary / continuous optimisation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a (possibly fractional) h using the original definition."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _c5_from_auto(auto: np.ndarray, n: int) -> float:
    """C5 when auto = np.correlate(h, h, mode='full') for a binary h."""
    min_auto = auto.min()
    return (n / 2.0 - min_auto) * (2.0 / n)


def _upscale_to_binary(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse function and binarisation."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    interp = np.interp(x_fine, x_coarse, coarse_h)
    idx = np.argsort(-interp)                     # descending
    bin_h = np.zeros(n_fine, dtype=np.uint8)
    bin_h[idx[: n_fine // 2]] = 1
    return bin_h


def _binary_hill_climb(h_start: np.ndarray, max_steps: int, time_limit: float):
    """
    Simple hill‑climbing on binary vectors with swaps of a 1 and a 0.
    Returns best (h, C5) seen.
    """
    n = h_start.size
    h = h_start.copy()
    auto = np.correlate(h, h, mode="full")
    cur_c5 = _c5_from_auto(auto, n)

    best_h = h.copy()
    best_c5 = cur_c5

    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    n_off = n - 1
    stagnation = 0

    for _ in range(max_steps):
        if time.time() > time_limit:
            break

        # random 1 ↔ 0 swap
        pos_i = np.random.randint(len(ones_idx))
        i = int(ones_idx[pos_i])
        pos_j = np.random.randint(len(zeros_idx))
        j = int(zeros_idx[pos_j])

        # all other 1‑positions (exclude i)
        if len(ones_idx) <= 1:
            continue
        if pos_i == 0:
            other = ones_idx[1:]
        elif pos_i == len(ones_idx) - 1:
            other = ones_idx[:-1]
        else:
            other = np.concatenate((ones_idx[:pos_i], ones_idx[pos_i + 1 :]))

        # indices of the affected correlation entries
        idx_i1 = other - i + n_off
        idx_i2 = i - other + n_off
        idx_j1 = other - j + n_off
        idx_j2 = j - other + n_off

        # propose new correlation
        new_auto = auto.copy()
        new_auto[idx_i1] -= 1
        new_auto[idx_i2] -= 1
        new_auto[idx_j1] += 1
        new_auto[idx_j2] += 1

        new_c5 = _c5_from_auto(new_auto, n)

        if new_c5 < cur_c5 - 1e-15:               # strictly improving
            # accept
            auto = new_auto
            cur_c5 = new_c5
            h[i] = 0
            h[j] = 1
            ones_idx[pos_i] = j
            zeros_idx[pos_j] = i
            stagnation = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stagnation += 1
            if stagnation > 2000:                 # give up on this run
                break

    return best_h, best_c5


def _continuous_fine_tune(h_start: np.ndarray, max_steps: int, time_limit: float):
    """
    Small epsilon exchanges on a (possibly fractional) h.
    Accepts only improvements.
    """
    n = h_start.size
    h = h_start.copy()
    best_c5 = _c5_from_h(h)
    best_h = h.copy()

    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = h[i]
        max_up = 1.0 - h[j]
        if max_down <= 0 or max_up <= 0:
            continue
        eps = np.random.rand() * min(max_down, max_up) * 0.5
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        c5 = _c5_from_h(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()
            h = cand
    return best_h, best_c5
