def perturb_h_full(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global helpers for SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- problem specific utilities ----------
def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h_full(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_full(start_h, rng, max_iters, start_step,
                    start_time, budget_s, time_frac=0.9):
    """Simple hill‑climb on the full‑resolution vector."""
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() - start_time > budget_s * time_frac:
            break
        new_h = perturb_h_full(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def make_h_from_params(params, m, block_sz):
    """
    Build a step function with symmetry h[i] + h[n-1-i] = 1.
    params: length m//2, values in [0,1].
    """
    half = len(params)
    b = np.empty(m)
    for i in range(half):
        b[i] = params[i]
        b[m - 1 - i] = 1.0 - params[i]
    if m % 2 == 1:
        b[half] = 0.5
    h = np.repeat(b, block_sz)
    return h


def slsqp_optimize_full(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

    best_c = compute_c5(best_h)

    # ----- early hill‑climb on the initial candidate -----
    best_h, best_c = hill_climb_full(
        best_h,
        rng,
        max_iters=4000,
        start_step=0.5,
        start_time=start_time,
        budget_s=budget_s,
        time_frac=0.45,
    )

    # ----- block‑wise search (symmetry + coarse parametrisation) -----
    # Choose a block size that divides n (6 → 30 blocks → 15 free variables)
    block_sz = 6
    m = n // block_sz          # number of symmetric blocks (even)
    var_dim = m // 2           # independent parameters

    while time.time() - start_time < budget_s * 0.85:
        # Randomly generate a symmetric parameter vector
        if rng.random() < 0.2:
            params = rng.integers(0, 2, size=var_dim).astype(float)  # binary seed
        else:
            params = rng.random(var_dim)                          # uniform seed
        h_candidate = make_h_from_params(params, m, block_sz)
        c_candidate = compute_c5(h_candidate)

        if c_candidate < best_c:
            # refine the promising candidate on the full grid
            refined_h, refined_c = hill_climb_full(
                h_candidate,
                rng,
                max_iters=2000,
                start_step=0.5,
                start_time=start_time,
                budget_s=budget_s,
                time_frac=0.85,
            )
            if refined_c < best_c:
                best_h, best_c = refined_h, refined_c

    # ----- final SLSQP refinement with increasing β -----
    if time.time() - start_time < budget_s * 0.97:
        for beta in [5.0, 20.0, 80.0, 200.0]:
            if time.time() - start_time > budget_s * 0.99:
                break
            try:
                best_h = slsqp_optimize_full(best_h, beta, max_iter=300)
            except Exception:
                continue
            best_c = compute_c5(best_h)

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
