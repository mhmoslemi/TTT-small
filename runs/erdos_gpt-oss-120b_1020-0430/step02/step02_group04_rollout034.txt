def grad_descent_optimize2(h0, beta_sched, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.6
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = np.sum(w)
            if w_sum == 0:
                break
            w /= w_sum
            # gradient accumulation
            g = np.ones(n)
            # Vectorized loops: we can iterate k indices
            # w index idx corresponds to shift k = idx - (n - 1)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    # for i in 0..n-k-1: subtract wk * h[i+k]
                    g[: n - k] -= wk * h[k:]
                    # for i in k..n-1: subtract wk * h[i-k]
                    g[k:] -= wk * h[: n - k]
                else:
                    kk = -k
                    g[kk:] -= wk * h[: n - kk]
                    g[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g
            # step update
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # maybe accept occasional uphill
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=150):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * corr * (2.0 / n)
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
```

def hill_climb(h_start, rng, max_iters=800, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        # pick random i != j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # ensure sum unchanged automatically; due to pairwise transfer
        # project onto [0,1] may not be needed because eps limited
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        # step decay intermittently
        if (it + 1) % (max_iters // 8) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Projection onto [lower,upper]^n with exact sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Uniform random vector with n/2 ones (in expectation) and exact half-sum, symmetric."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


def symmetrize(v):
    """Force mirror symmetry and re‑project onto the feasible box‑sum set."""
    v2 = (v + v[::-1]) * 0.5
    v2 = project_onto_box_sum(v2, v2.size / 2.0)
    return v2


# ---------- local search ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=800, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ---------- smooth‑max gradient descent ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the soft‑max surrogate of the max overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.6
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)        # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w /= w_sum                 # soft‑max weights over shifts

            # gradient of the smooth‑max (unscaled)
            g = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g[: n - k] -= wk * h[k:]
                    g[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g[kk:]     -= wk * h[: n - kk]
                    g[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g

            nh = h - step * grad
            nh = project_onto_box_sum(nh, n / 2.0)
            cnew = compute_c5(nh)

            if cnew < best_c:
                best_c = cnew
                best_h = nh.copy()
                h = nh
            else:
                if rng.random() < 0.01:   # occasional uphill move
                    h = nh
            step *= 0.9995
    return best_h, best_c


# ---------- SLSQP refinement (soft‑max surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr * (2.0 / n)
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=150):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
