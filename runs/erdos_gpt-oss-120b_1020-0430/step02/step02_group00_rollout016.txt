    def project_feasible(h):
        # Clip and adjust sum using water-filling
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        # Use simple iterative: adjust remainder by adding to those below 1 equally, subtract from those above 0 equally.
        # but simplest: we can compute deficit/excess and correct.
        # Use method: while sum != target: compute diff = current_sum - target.
        # If diff>0: need to reduce sum; identify indices with h>0; subtract diff/len.
        # If diff<0: need to increase sum; identify indices with h<1; add -diff/len.
        # iterate few times.
        for _ in range(10):
            s = h.sum()
            diff = s - target
            if abs(diff) < 1e-9:
                break
            if diff > 0:
                mask = h > 0
                if not np.any(mask):
                    break
                delta = diff / mask.sum()
                h[mask] -= delta
            else:
                mask = h < 1
                if not np.any(mask):
                    break
                delta = -diff / mask.sum()
                h[mask] += delta
            h = np.clip(h, 0.0, 1.0)
        # final fix:
        residual = target - h.sum()
        if abs(residual) > 1e-9:
            # adjust first element not at bound
            for i in range(n):
                if residual > 0 and h[i] < 1 - 1e-12:
                    add = min(residual, 1 - h[i])
                    h[i] += add
                    residual -= add
                elif residual < 0 and h[i] > 1e-12:
                    sub = min(-residual, h[i])
                    h[i] -= sub
                    residual += sub
                if abs(residual) < 1e-12:
                    break
        return np.clip(h, 0.0, 1.0)

        start_h = project_feasible(start_h)
        n = start_h.size
        t0 = c5_val(start_h)
        x0 = np.concatenate([start_h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        def ineq(x):
            h = x[:n]
            t = x[n]
            # t - correlation >= 0 for each shift
            # we compute correlation for all shifts quickly using FFT? but conformance may easier using np.correlate.
            # Use np.correlate
            corr = np.correlate(h, 1.0 - h, mode='full')
            return t - corr
        def eq(x):
            return x[:n].sum() - n/2.0
        def obj(x):
            return x[n]
        res = opt.minimize(obj, x0, method='SLSQP', bounds=bounds,
                           constraints=[{'type':'ineq','fun':ineq},
                                        {'type':'eq','fun':eq}],

        if res.success:

            return start_h, c5_val(start_h)

def sa_optimize(h_init, n_iter, temp_start, temp_end):
    h = h_init.copy()
    c5_cur = c5_val(h)
    best_h_local = h.copy()
    best_c5_local = c5_cur
    n = h.size
    for it in range(1, n_iter+1):
        T = temp_start * ((temp_end/temp_start) ** (it / n_iter))
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        delta_max = min(h[i], 1.0 - h[j])
        delta_min = -min(1.0 - h[i], h[j])
        if delta_max <= delta_min:
            continue
        # sample delta, sometimes full, sometimes scaled
        if rng.random() < 0.2:
            delta = delta_max if rng.random() < 0.5 else delta_min
        else:
            delta = rng.uniform(delta_min, delta_max) * 0.5
        # apply
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = c5_val(h_new)
        dE = c5_new - c5_cur
        if dE < -1e-12 or rng.random() < np.exp(-dE / (T + 1e-12)):

            if c5_new < best_c5_local:
                best_h_local = h_new.copy()
                best_c5_local = c5_new
        # early exit if time low
        if time_left() < 0.5:
            break
    return best_h_local, best_c5_local

import time

# Helper utilities (top‑level, no closures)

def _project_to_feasible(h: np.ndarray) -> np.ndarray:
    """Clip h∈[0,1] and enforce Σh = len(h)/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0.0
        else:
            mask = h < 1.0
        if not np.any(mask):
            break
        delta = diff / mask.sum()
        h[mask] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1.0 - 1e-12:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _corr_full_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation h ⋆ (1‑h) using FFT, returns length 2n‑1 array."""
    n = h.size
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_b = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_b), n=L)
    return corr[: 2 * n - 1]


def _c5_from_h(h: np.ndarray) -> float:
    """C5 bound for a (feasible) step function."""
    n = h.size
    max_corr = _corr_full_fft(h).max()
    return max_corr * (2.0 / n)


class _IneqConstr:
    def __init__(self, n: int):
        self.n = n

    def __call__(self, x: np.ndarray) -> np.ndarray:
        h = x[: self.n]
        t = x[-1]
        return t - _corr_full_fft(h)


class _EqConstr:
    def __init__(self, n: int):
        self.n = n

    def __call__(self, x: np.ndarray) -> float:
        return x[: self.n].sum() - (self.n / 2.0)


def _slsqp_optimize(start_h: np.ndarray, max_iter: int = 300) -> tuple:
    """Run a single SLSQP optimisation, return (h_opt, c5)."""
    h0 = _project_to_feasible(start_h)
    n = h0.size
    t0 = _c5_from_h(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constr = [
        {"type": "ineq", "fun": _IneqConstr(n)},
        {"type": "eq", "fun": _EqConstr(n)},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = _opt.minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds,
                        constraints=constr, options=opts)
    if res.success:
        h_opt = _project_to_feasible(res.x[:n])
        return h_opt, _c5_from_h(h_opt)
    else:
        return h0, _c5_from_h(h0)


def _upscale_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n = h.size
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)
