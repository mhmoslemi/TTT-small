def build_binary_symmetric(n, rng):
    assert n % 2 == 0
    m = n // 2
    target_pairs = n // 4   # number of ones in half
    p = np.zeros(m, dtype=int)
    for _ in range(target_pairs):
        best_i = None
        best_c = None
        # iterate candidates
        candidates = np.flatnonzero(p == 0)
        # maybe shuffle candidates for randomness tie-breaking
        rng.shuffle(candidates)
        for i in candidates:
            p[i] = 1
            h = np.concatenate([p, p[::-1]])
            corr = np.correlate(h, 1 - h, mode='full')
            c = corr.max()
            p[i] = 0
            if best_c is None or c < best_c:
                best_c = c
                best_i = i
        p[best_i] = 1
    h = np.concatenate([p, p[::-1]])
    return h.astype(float), best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] while fixing the sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def perturb_h(h, rng, scale=0.2):
    """Swap a small amount between two coordinates while preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- symmetric binary construction ----------
def greedy_binary_symmetric(n, rng):
    """Greedy construction of a symmetric binary vector (weight n/2)."""
    m = n // 2
    target_pairs = n // 4
    p = np.zeros(m, dtype=int)
    for _ in range(target_pairs):
        best_i = -1
        best_c = None
        cand = np.where(p == 0)[0]
        rng.shuffle(cand)
        for i in cand:
            p[i] = 1
            h = np.concatenate([p, p[::-1]])
            c = np.correlate(h, 1 - h, mode="full").max()
            p[i] = 0
            if best_c is None or c < best_c:
                best_c = c
                best_i = i
        p[best_i] = 1
    h = np.concatenate([p, p[::-1]]).astype(float)
    return h


def binary_hill_climb(p, rng, max_iters=2000, time_limit=None, start_time=None):
    """Hill‑climb on the half‑vector p (binary, fixed weight)."""
    m = p.size
    target = m // 2
    best_p = p.copy()
    best_c = compute_c5(np.concatenate([p, p[::-1]]))
    cur_p = p.copy()
    cur_c = best_c
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        ones = np.flatnonzero(cur_p == 1)
        zeros = np.flatnonzero(cur_p == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_p[i], cur_p[j] = 0, 1
        hc = compute_c5(np.concatenate([cur_p, cur_p[::-1]]))
        if hc <= cur_c:
            cur_c = hc
            if hc < best_c:
                best_c = hc
                best_p = cur_p.copy()
        else:
            # occasional uphill move (simulated annealing style)
            temp = 1.0 - it / max_iters
            if rng.random() < np.exp(-(hc - cur_c) / (0.05 + 0.5 * temp)):
                cur_c = hc
            else:
                cur_p[i], cur_p[j] = 1, 0
    return best_p, best_c


# ---------- SLSQP utilities ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- Gradient‑descent on smooth max ----------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                     # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()
            g_raw = np.ones(n)
            # vectorised contribution of each shift weight
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 200  # use a modest size that still offers flexibility

    # ---- multiple quick binary attempts ----
    best_h = None
    best_c = 1e9
    attempts = 0
    while time.time() - start < budget_s * 0.45 and attempts < 6:
        h_bin = greedy_binary_symmetric(n, rng)
        # quick hill‑climb on the binary representation
        p = (h_bin[: n // 2] > 0.5).astype(int)
        p_opt, c_opt = binary_hill_climb(
            p, rng, max_iters=2500,
            time_limit=budget_s * 0.05,
            start_time=time.time(),
        )
        h_opt = np.concatenate([p_opt, p_opt[::-1]]).astype(float)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        attempts += 1

    if best_h is None:
        # fallback: uniform half‑filled vector
        best_h = np.concatenate(
            [np.ones(n // 4), np.zeros(n // 2 - n // 4), np.ones(n // 4)]
        ).astype(float)
        best_c = compute_c5(best_h)

    # ---- continuous refinement (smooth‑max) ----
    # adapt schedule based on remaining time
    remain = budget_s - (time.time() - start)
    beta_sched = [5, 20, 80, 200, 500, 1000, 2000]
    steps_per_beta = 300
    # shrink schedule if time is tight
    if remain < budget_s * 0.3:
        steps_per_beta = max(50, steps_per_beta // 4)
    h_cont = best_h.copy()
    for beta in beta_sched:
        if time.time() - start > budget_s * 0.85:
            break
        h_cont, _ = grad_descent_optimize(h_cont, [beta], steps_per_beta, rng)

    # ---- short SLSQP polishing ----
    if time.time() - start < budget_s * 0.95:
        for beta in (500, 1000, 2000):
            if time.time() - start > budget_s:
                break
            try:
                h_cont = slsqp_optimize(h_cont, beta=beta, max_iter=250)
            except Exception:
                pass

    # ---- final local perturbations (continuous) ----
    if time.time() - start < budget_s * 0.98:
        for _ in range(500):
            cand = perturb_h(h_cont, rng, scale=0.07)
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_box_sum(cand, n / 2.0)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_c = c_cand
                h_cont = cand

    # ---- final clean‑up ----
    h_cont = np.clip(h_cont, 0.0, 1.0)
    h_cont = project_onto_box_sum(h_cont, n / 2.0)
    final_c = compute_c5(h_cont)

    return h_cont.tolist(), float(final_c), int(n)
