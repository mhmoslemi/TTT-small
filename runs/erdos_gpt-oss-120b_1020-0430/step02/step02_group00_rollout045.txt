def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Use iterative water-filling: naive O(N) approach.
    # While sum != target within tolerance:
    # Compute diff = sum - target. If diff > 0, we need to reduce sum: subtract equally from entries > 0 (non-zero).
    # But if an entry becomes negative, clip and repeat.
    # Similarly for diff < 0.
    # Use loop with max_iter like previous, but implement efficiently.
    max_iter = 1000
    tol = 1e-9
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to reduce
            idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            h[idx] -= delta
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = np.where(h < 1)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            h[idx] -= delta
            h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual to any indices that can be adjusted
        idx = np.where((residual > 0) & (h < 1 - 1e-12))[0]
        if idx.size > 0:
            add = min(residual / idx.size, 1 - h[idx])
            h[idx] += residual / idx.size
        else:
            idx = np.where((residual < 0) & (h > 1e-12))[0]
            if idx.size > 0:
                h[idx] += residual / idx.size
    return np.clip(h, 0.0, 1.0)

def propose_move(h, step):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # random delta in [-step, step]
    delta = (np.random.rand() * 2 - 1) * step
    # adjust to stay within bounds
    # possible increase on j (h_j + delta) and decrease on i (h_i - delta)
    if delta > 0:
        max_delta = min(h[i], 1.0 - h[j])
        delta = min(delta, max_delta)
    else:
        max_delta = min(h[j], 1.0 - h[i])
        delta = max(delta, -max_delta)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    # Should preserve sum and bounds.
    return h_new

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size:
            h[idx] += residual / idx.size
    return np.clip(h, 0.0, 1.0)

    n = h.size
    conv = np.fft.ifft(np.fft.fft(h, 2 * n) * np.fft.fft(1 - h, 2 * n)).real
    return float(conv.max())


def _c5_from_h(h: np.ndarray) -> float:
    """Scaled C₅ value as required by the evaluator."""
    n = h.size
    return _max_corr_unscaled(h) * (2.0 / n)


def _propose_move(h: np.ndarray, step: float) -> np.ndarray:
    """Return a new feasible h after moving mass δ from i to j."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    delta = (np.random.rand() * 2.0 - 1.0) * step
    if delta > 0:
        max_delta = min(h[i], 1.0 - h[j])
        delta = min(delta, max_delta)
    else:
        max_delta = min(h[j], 1.0 - h[i])
        delta = max(delta, -max_delta)
    if abs(delta) < 1e-14:
        return h
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2


def _simulated_annealing(h0: np.ndarray, max_time: float,
                         start_temp: float = 0.02, end_temp: float = 1e-5,
                         base_step: float = 0.04) -> tuple:
    """Simple SA that respects the sum constraint via paired moves."""
    start = time.time()
    h_cur = h0.copy()
    cur_val = _c5_from_h(h_cur)
    best_h = h_cur.copy()
    best_val = cur_val
    n = h0.size

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= max_time:
            break
        progress = elapsed / max_time
        temp = start_temp * (1.0 - progress) + end_temp * progress
        step = base_step * (temp / start_temp) if start_temp > 0 else base_step

        h_new = _propose_move(h_cur, step)
        new_val = _c5_from_h(h_new)

        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
            h_cur = h_new
            cur_val = new_val
            if new_val < best_val:
                best_h = h_new.copy()
                best_val = new_val
    return best_h, best_val


def _slsqp_refine(h0: np.ndarray, max_iter: int = 600) -> tuple:
    """One SLSQP refinement starting from a feasible vector."""
    n = h0.size
    h0 = _project_to_feasible(h0, n)

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.fft.ifft(np.fft.fft(h, 2 * n) * np.fft.fft(1 - h, 2 * n)).real
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    def obj(x):
        return x[-1]

    t0 = _max_corr_unscaled(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(obj, x0, method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)

    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # --------------------------------------------------------------
    # 1. Obtain initial vector (either provided or random binary)
    # --------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # may be defined by the harness
    except NameError:
        init_h = None

    if init_h is None:
        n0 = 256
        h0 = np.zeros(n0)
        h0[: n0 // 2] = 1.0
        np.random.shuffle(h0)
        init_h = _project_to_feasible(h0, n0)
    else:
        init_h = _project_to_feasible(init_h, init_h.size)

    best_h = init_h.copy()
    best_val = _c5_from_h(best_h)
    best_n = best_h.size

    # --------------------------------------------------------------
    # 2. Stage 1 – SA on the original resolution
    # --------------------------------------------------------------
    if time_left() > 15.0:
        t_stage = min(time_left() * 0.3, max(5.0, time_left() - 30.0))
        h_opt, val_opt = _simulated_annealing(best_h, t_stage,
                                              start_temp=0.025, end_temp=1e-5,
                                              base_step=0.05)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # --------------------------------------------------------------
    # 3. Incremental up‑scaling and refinement
    # --------------------------------------------------------------
    while time_left() > 25.0 and best_n < 1024:
        # double the resolution by repetition (preserves feasibility)
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # small random jitter to break symmetry
        jitter = (np.random.rand(n_up) - 0.5) * 0.02
        h_up = _project_to_feasible(h_up + jitter, n_up)

        # allocate time for this level
        t_stage = min(time_left() * 0.3, max(5.0, time_left() - 20.0))
        h_opt, val_opt = _simulated_annealing(h_up, t_stage,
                                              start_temp=0.018, end_temp=1e-5,
                                              base_step=0.04 / math.sqrt(best_n))
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
        best_n = best_h.size

        # optional early break if already excellent
        if best_val < 0.3807:
            break

    # --------------------------------------------------------------
    # 4. Final SLSQP polish if time permits
    # --------------------------------------------------------------
    if time_left() > 12.0:
        h_ref, val_ref = _slsqp_refine(best_h, max_iter=800)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    # --------------------------------------------------------------
    # 5. Return feasible solution
    # --------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    final_val = _c5_from_h(best_h)
    return best_h.tolist(), float(final_val), int(best_h.size)
