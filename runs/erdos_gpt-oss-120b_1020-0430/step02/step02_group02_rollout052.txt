def run(seed=42,budget_s=1100,**kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left(): return budget_s - (time.time() - start)

    # parameters
    n_candidates = [80, 96, 112, 128, 160, 192, 224, 256, 320, 384, 512]  # may include more

    best_h, best_c5, best_n = None, float('inf'), None

    for n in n_candidates:
        if time_left() < 5:
            break
        # Build initial guess
        if 'initial_h_values' in globals():
            init_h_global = np.array(initial_h_values, dtype=float)
            if init_h_global.size != n:
                base = _upscale_h(init_h_global, n)
            else:
                base = init_h_global.copy()
        else:
            base = np.full(n, 0.5)
        # best in this n
        h = _project_to_feasible(base, n)
        # maybe random start some times
        # optimization loop
        best_h_n = h.copy()
        best_c5_n = _c5_from_h(h)

        beta = 1.0
        step = 0.1
        for iter in range(5000):
            if time_left() < 0.5:
                break
            # compute correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = corr.max()
            c5 = max_corr * (2.0 / n)
            if c5 < best_c5_n:
                best_c5_n, best_h_n = c5, h.copy()
                # global best?
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h.copy(), n
            # compute gradient of softmax
            # weighting
            weights = np.exp(beta * corr - np.max(beta * corr))  # for numeric stability
            weights_sum = weights.sum()
            weights = weights / weights_sum
            # gradient vector
            g = np.zeros(n)
            half = n - 1
            for s in range(-half, half+1):
                w_s = weights[s + half]
                if w_s == 0.0:
                    continue
                if s >= 0:
                    i_start = 0
                    i_end = n - s
                    g[i_start:i_end] += w_s * (1 - h[i_start + s:i_end + s])
                    g[i_start + s:i_end + s] -= w_s * h[i_start:i_end]
                else:
                    i_start = -s
                    i_end = n
                    g[i_start:i_end] += w_s * (1 - h[i_start + s:i_end + s])
                    g[i_start + s:i_end + s] -= w_s * h[i_start:i_end]
            # gradient descent step
            step_factor = step / (1.0 + 0.01 * iter)  # decreasing step size
            h_new = h - step_factor * g
            # project onto feasible set
            h_new = _project_to_feasible(h_new, n)
            h = h_new
            # anneal beta gradually
            if (iter+1) % 200 == 0:
                beta = min(beta * 1.5, 50.0)
                step = max(step * 0.9, 0.001)

        # If we achieved target <0.3808, break
        if best_c5 < 0.3808:

    # fallback if nothing
    if best_h is None:

        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n
    return best_h.tolist(), float(best_c5), int(best_n)

import time, numpy as np

def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    # clip
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # find indices that can be adjusted in the right direction
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual greedily
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

    max_corr = corr.max()

                best_h, best_c, best_n = h_opt.copy(), c_opt, best_n
        if best_c < 0.3808:
            break

    # final hill‑climb refinement if time permits
    if best_h is not None and time_left() > 5.0:
        deadline = start + budget_s
        best_h, best_c = _hill_climb(best_h, steps=5000, deadline=deadline)

    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c), int(best_n)
