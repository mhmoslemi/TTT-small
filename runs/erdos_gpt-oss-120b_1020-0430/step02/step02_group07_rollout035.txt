   h = h0.copy()

   best_h = h.copy()
   best_c = cur_c
   temp = init_temp

   # To speed up, we might compute all possible swaps? That's huge.

   for t in range(max_iters):
      # pick random i where h[i]=1, random j where h[j]=0
      ones = np.flatnonzero(h)
      zeros = np.flatnonzero(1 - h) # h is int 0/1
      # sample from arrays; to reduce cost, pick random i = rng.choice(ones), etc.

      i = rng.choice(ones)

      # swap
      h[i], h[j] = 0, 1
      new_c = compute_c5_bin(h)
      # acceptance
      if new_c < best_c:
          best_c = new_c
          best_h = h.copy()
      if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):

      else:
          # revert swap
          h[i], h[j] = 1, 0

      temp *= decay
   return best_h, best_c

def compute_c5_bin(h):
    N = h.size
    fft_h = np.fft.rfft(h, n=N)
    # autocorrelation via inverse real FFT:
    # autop = np.fft.irfft(fft_h * np.conj(fft_h), n=N).real
    # However rfft returns half-spectrum; product conj yields real sequence.
    # Use np.fft.ifft with complex: simpler.
    fft_full = np.fft.fft(h)
    autocorr = np.fft.ifft(fft_full * np.conj(fft_full)).real
    # round
    autocorr = np.rint(autocorr)
    min_corr = int(autocorr.min())
    c5 = 1.0 - 2.0 * min_corr / N
    return float(c5)

def compute_c5_bin(h):
    N = h.size
    # Convert to float (0.,1.) for FFT
    h_float = h.astype(np.float64)
    # autocorrelation via FFT
    # Use real FFT: r = np.fft.rfft(h_float)
    # autop = np.fft.irfft(r * np.conj(r), n=N)
    # Using real version maybe faster.
    r = np.fft.rfft(h_float, n=N)
    autop = np.fft.irfft(r * np.conj(r), n=N)
    # autop should be real but due to numeric errors may be small imag; irfft returns real.
    # Round small errors
    autop = np.rint(autop).astype(np.int64)
    min_corr = int(autop.min())
    if min_corr < 0:
        min_corr = 0
    c5 = 1.0 - (2.0 * min_corr) / N
    return float(c5)

def simulated_annealing_binary(h0, rng, max_iters, init_temp, decay):

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_simplex_with_bounds(v, target, iters=60):
    """
    Euclidean projection of v onto the set {x | sum(x)=target, 0<=x<=1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(iters):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        # sum decreases when lam increases
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    v = rng.random(n)
    return project_onto_simplex_with_bounds(v, n / 2.0)


def gradient_of_overlap(h, shift):
    """Sub‑gradient of the maximal overlap term for a given shift."""
    n = h.size
    g = np.zeros(n, dtype=np.float64)
    if shift >= 0:
        # i from 0 .. n-1-shift
        g[: n - shift] += 1.0 - h[shift:]
        g[shift:] -= h[: n - shift]
    else:
        k = -shift
        g[k:] += 1.0 - h[: n - k]
        g[: n - k] -= h[k:]
    return g


def subgradient_descent(h0, max_iters, init_step, rng, time_limit):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    start = time.time()
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break

        # step size schedule
        step = init_step / math.sqrt(t + 1)

        # correlation & worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)

        # gradient step + projection
        grad = gradient_of_overlap(h, shift)
        h_new = h - step * grad
        h_new = project_onto_simplex_with_bounds(h_new, target)

        c = np.max(corr) * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    target_up = (n * factor) / 2.0
    h_up = project_onto_simplex_with_bounds(h_up, target_up)
    return h_up


def simulated_annealing_binary(h0, rng, max_iters, init_temp, decay):
    """SA on binary vectors (exactly n/2 ones)."""
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp

    ones = np.flatnonzero(h > 0.5).tolist()
    zeros = np.flatnonzero(h <= 0.5).tolist()

    for it in range(max_iters):
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # swap
        h[i] = 0.0
        h[j] = 1.0
        ones[i_idx] = j
        zeros[j_idx] = i

        new_c = compute_c5(h)
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            prob = math.exp((cur_c - new_c) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_c = new_c
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            ones[i_idx] = i
            zeros[j_idx] = j

        temp *= decay
    return np.array(best_h, dtype=np.float64), best_c
