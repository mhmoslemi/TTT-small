import time
import numpy as np
from scipy import optimize

# ---------- Feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clamp h to [0,1] and enforce sum(h)=n/2 using a water‑filling style loop."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- Objective ----------
def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


# ---------- Local optimisation (SLSQP) ----------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return float(np.sum(x[:n]) - (n / 2.0))


def _objective(x: np.ndarray) -> float:
    return float(x[-1])


def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    """Run SLSQP from initial_h, return feasible (h, C5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = float(np.max(np.correlate(h0, 1.0 - h0, mode="full")))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)
    h_opt = _project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- Genetic Algorithm ----------
def _tournament_select(pop, fit, k=3):
    idx = np.random.choice(len(pop), k, replace=False)
    best = idx[np.argmin(np.take(fit, idx))]
    return pop[best]


def _uniform_crossover(p1, p2):
    mask = np.random.rand(p1.size) < 0.5
    return np.where(mask, p1, p2)


def _mutate(child, sigma):
    return child + sigma * np.random.randn(child.size)


def _ga_optimize(n, init_h, time_budget, target, parent_h=None):
    """Return best (h, C5) found for size n within time_budget seconds."""
    start = time.time()
    # population parameters
    pop_size = 30
    elite_cnt = 2
    sigma = 0.25
    min_sigma = 0.001
    sigma_decay = 0.99

    # ---- initialise population ----
    pop = []
    if init_h is not None:
        pop.append(_project_to_feasible(init_h, n))
    if parent_h is not None and parent_h.size == n:
        pop.append(_project_to_feasible(parent_h + 0.02 * np.random.randn(n), n))
    while len(pop) < pop_size:
        cand = np.random.rand(n)
        pop.append(_project_to_feasible(cand, n))
    pop = np.array(pop)

    # evaluate
    fit = np.array([_c5_from_h(ind) for ind in pop])
    best_idx = int(np.argmin(fit))
    best_h = pop[best_idx].copy()
    best_c5 = fit[best_idx]

    # early exit
    if best_c5 < target:
        return best_h, best_c5

    # ---- generational loop ----
    while time.time() - start < time_budget:
        # elite preservation
        elite_idx = np.argsort(fit)[:elite_cnt]
        new_pop = pop[elite_idx].copy()
        # generate children
        while len(new_pop) < pop_size:
            p1 = _tournament_select(pop, fit)
            p2 = _tournament_select(pop, fit)
            child = _uniform_crossover(p1, p2)
            child = _mutate(child, sigma)
            child = _project_to_feasible(child, n)
            new_pop = np.vstack([new_pop, child])
        # evaluate new population
        fit = np.array([_c5_from_h(ind) for ind in new_pop])
        pop = new_pop
        # update global best
        cur_best = int(np.argmin(fit))
        if fit[cur_best] < best_c5:
            best_c5 = fit[cur_best]
            best_h = pop[cur_best].copy()
            if best_c5 < target:
                break
        # sigma schedule
        sigma = max(min_sigma, sigma * sigma_decay)
    return best_h, best_c5

    start_time = time.time()

    # try to load a parent construction

    for n in n_candidates:
