class ExchangeStep:
    def __init__(self, step_scale, rng):
        self.step_scale = step_scale
        self.rng = rng
    def __call__(self, x):
        n = len(x)
        # copy
        y = x.copy()
        i = self.rng.integers(0, n)
        j = self.rng.integers(0, n)
        if i == j:
            return y
        # max epsilon that can move from i to j without violating bounds
        max_eps = min(y[i], 1.0 - y[j])
        if max_eps <= 0:
            return y
        eps = (self.rng.random() - 0.5) * 2.0 * max_eps * self.step_scale
        eps = max(-max_eps, min(eps, max_eps))
        y[i] -= eps
        y[j] += eps
        # after move sum unchanged, still within bounds
        return y

def max_subgradient(h):
    n = len(h)
    corr_full = np.correlate(h, 1 - h, mode='full')
    # find index of max
    idx = np.argmax(corr_full)
    # compute lag: lag = idx - (n - 1)  (since center offset)
    lag = idx - (n - 1)
    # compute g array:
    g = np.zeros_like(h)
    if lag >= 0:
        # for i in range(0, n - lag)
        g[:n - lag] += (1 - h[lag:])  # (1 - h_{i+lag})
        g[lag:] -= h[:n - lag]        # -h_{i}
    else:
        d = -lag
        g[d:] += (1 - h[:n - d])      # (1 - h_{i-d})
        g[:n - d] -= h[d:]            # -h_{i+d}
    return g, corr_full[idx] * (2.0 / n) # return gradient and objective C5

def gradient_move(h, rng, scale=0.5):
    n = len(h)
    # compute max lag and gradient
    corr_full = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr_full)
    lag = idx - (n-1)
    g = np.zeros_like(h)
    if lag >=0:
        g[:n-lag] += (1 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        d = -lag
        g[d:] += (1 - h[:n-d])
        g[:n-d] -= h[d:]
    # Flatten P and N
    pos = np.where(g > 1e-12)[0]
    neg = np.where(g < -1e-12)[0]
    if pos.size ==0 or neg.size ==0:
        # fallback random move
        return perturb_h(h, rng, scale=scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    # maximum epsilon to move from i to j
    max_eps = min(h[i], 1.0 - h[j])

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate (log‑sum‑exp) objective
# ----------------------------------------------------------------------
CURRENT_TARGET = None   # target sum = n/2
CURRENT_BETA = None     # smoothing parameter for the surrogate

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# ----------------------------------------------------------------------
# Helpers for feasibility‑preserving transformations
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def random_feasible_vector(n, rng):
    """Produce a random vector with entries in [0,1] and sum = n/2."""
    h = np.full(n, 0.5)
    # a few shuffled exchanges to spread values
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ----------------------------------------------------------------------
# Gradient‑driven move that targets the lag attaining the current maximum
# ----------------------------------------------------------------------
def gradient_move(h, rng, scale=0.5):
    """Move mass from a positive‑gradient coordinate to a negative‑gradient one."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)          # signed lag
    g = np.zeros_like(h)

    if lag >= 0:
        # first part: i in [0, n‑lag‑1]
        i_end = n - lag
        g[:i_end] += (1 - h[lag:])      # ∂/∂h_i of term h_i (1‑h_{i+lag})
        g[lag:]    -= h[:i_end]         # ∂/∂h_{i+lag} of the same term
    else:
        d = -lag
        i_end = n - d
        g[d:]     += (1 - h[:i_end])
        g[:i_end] -= h[d:]

    pos = np.where(g > 1e-12)[0]
    neg = np.where(g < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_h(h, rng, scale=scale)

    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale=scale)

    eps = min(abs(g[i]), abs(g[j]), max_eps) * scale
    if eps <= 1e-12:
        eps = max_eps * scale * 0.5

    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ----------------------------------------------------------------------
# Local hill‑climb that mixes gradient‑guided and random moves
# ----------------------------------------------------------------------
def hill_climb_max(h_start, rng, max_iters=8000, init_step=0.5, decay=0.9):
    """Deterministic / stochastic descent on the exact C₅ (max‑overlap)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        # Occasionally inject a pure random exchange for diversity
        if rng.random() < 0.2:
            new_h = perturb_h(h, rng, scale=step)
        else:
            new_h = gradient_move(h, rng, scale=step)

        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * decay, 1e-4)
    return best_h, best_c

# ----------------------------------------------------------------------
# Refinement via SLSQP on the smooth surrogate for a given β
# ----------------------------------------------------------------------
def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny drift correction for the linear equality
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -----------------------------------------------------------

    if "initial_h_values" in globals():

    global_best_h = best_h

    # -----------------------------------------------------------
    # Main optimisation loop (gradient hill‑climb + SLSQP ladder)
    # -----------------------------------------------------------
    beta_sequence = [10.0, 30.0, 100.0, 300.0, 1000.0, 2000.0]
