import time
import numpy as np

def _c5_from_h(h: np.ndarray) -> float:
    """Return the max correlation * (2/n) for a discretised step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

def _make_linear_h(n: int) -> np.ndarray:
    """Construct the monotone linear step function h(x)=x/2 on [0,2] (discrete)."""
    # Points are at the centre of each sub‑interval.
    # x_i = (i+0.5)*(2/n)  ->  h_i = x_i/2 = (i+0.5)/n .
    h = (np.arange(n, dtype=float) + 0.5) / n
    # Numerical safety: enforce bounds and exact integral.
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) > 1e-12:                       # adjust if rounding caused tiny drift
        h -= diff / n
        h = np.clip(h, 0.0, 1.0)
        # final tiny correction (water‑filling style)
        remaining = target - h.sum()
        if abs(remaining) > 1e-12:
            for i in range(n):
                if remaining > 0 and h[i] < 1.0:
                    add = min(remaining, 1.0 - h[i])
                    h[i] += add
                    remaining -= add
                elif remaining < 0 and h[i] > 0.0:
                    sub = min(-remaining, h[i])
                    h[i] -= sub
                    remaining += sub
                if abs(remaining) < 1e-12:
                    break
        h = np.clip(h, 0.0, 1.0)
    return h
