def project_onto_box_sum(v, target, lower=0.0, upper=1.0, max_iter=50):
    # Clip and adjust via lamda method.
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    # fix small residual error
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj
```

def adam_optimize(h0, beta_seq, steps_per_beta, rng, symmetric=False):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05 # base learning rate
    for beta in beta_seq:
        for _ in range(steps_per_beta):
            t += 1
            # compute gradient of smooth max
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * (2.0 / n)
            bc = beta * C
            m_bc = np.max(bc)
            w = np.exp(bc - m_bc)
            w_sum = w.sum()
            w /= w_sum
            # compute convolution via np.convolve
            w_vec = w
            t_vec = np.convolve(w_vec, np.ones(n), mode='valid')
            B = np.convolve(w_vec, h, mode='valid')
            Cterm = np.convolve(w_vec[::-1], h, mode='valid')
            grad = (2.0 / n) * (t_vec - B - Cterm)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, n/2.0)
            if symmetric:
                h_new = (h_new + h_new[::-1]) * 0.5
                h_new = project_onto_box_sum(h_new, n/2.0)
            # Evaluate
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                # occasionally accept worse to escape (prob decays)
                if rng.random() < 0.001:
                    h = h_new
    return best_h, best_c
```

def smooth_max_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    C = corr * (2.0 / n)
    bc = beta * C
    m = np.max(bc)
    w = np.exp(bc - m)
    w_sum = w.sum()
    w /= w_sum
    # w shape (2n-1)
    t_vec = np.convolve(w, np.ones(n), mode='valid')
    B = np.convolve(w, h, mode='valid')
    Cterm = np.convolve(w[::-1], h, mode='valid')
    grad = (2.0 / n) * (t_vec - B - Cterm)
    # also return smoothed max value:
    val = (np.log(w_sum) + m) / beta
    return val, grad
```

def adam_optimize(h0, beta_seq, steps_per_beta, rng, symmetric=False):
    h = h0.copy()

    best_h = h.copy()

    v = np.zeros_like(h)
    t_iter = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02
    for beta in beta_seq:
        for _ in range(steps_per_beta):
            t_iter += 1
            _, grad = smooth_max_grad(h, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)

import time
import numpy as np

# ---------- helper utilities ----------
def compute_c5(h):
    """Discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, max_iter=60, tol=1e-12):
    """Project v onto [lower,upper] with exact sum = target (box‑sum projection)."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng, symmetric=False):
    """Generate a random feasible h (0≤h≤1, Σh=n/2)."""
    x = rng.exponential(1.0, size=n)
    x = project_onto_box_sum(x, n / 2.0)
    if symmetric:
        x = (x + x[::-1]) * 0.5
        x = project_onto_box_sum(x, n / 2.0)
    return x


def smooth_max_grad(h, beta):
    """Smooth max (log‑sum‑exp) of the overlap integrals and its gradient."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    C = corr * (2.0 / n)                # scaled overlaps
    bc = beta * C
    m = np.max(bc)
    w = np.exp(bc - m)
    w_sum = w.sum()
    w /= w_sum
    # w corresponds to lags -(n-1)…(n-1)
    t_vec = np.convolve(w, np.ones(n), mode="valid")
    B = np.convolve(w, h, mode="valid")
    Cterm = np.convolve(w[::-1], h, mode="valid")
    grad = (2.0 / n) * (t_vec - B - Cterm)
    val = (np.log(w_sum) + m) / beta    # smoothed max value
    return val, grad


def adam_optimize(h0, beta_seq, steps_per_beta, rng, symmetric=False):
    """Projected Adam optimizer on the smooth‑max surrogate."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t_iter = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02
    for beta in beta_seq:
        for _ in range(steps_per_beta):
            t_iter += 1
            _, grad = smooth_max_grad(h, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, n / 2.0)
            if symmetric:
                h_new = (h_new + h_new[::-1]) * 0.5
                h_new = project_onto_box_sum(h_new, n / 2.0)
            c_new = compute_c5(h_new)
            if c_new < best_c:

                best_h = h_new.copy()
                h = h_new
            else:

                    h = h_new
    return best_h, best_c


def hill_climb(h_start, rng, max_iters=2000, scale=0.2):
    """Simple coordinate swap hill‑climb."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:

        max_eps = min(h[i], 1.0 - h[j])

        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale

        cand[j] += eps

        cand = project_onto_box_sum(cand, n / 2.0)
        c_new = compute_c5(cand)
        if c_new < best_c:
            best_c = c_new
            best_h = cand.copy()
            h = cand
    return best_h, best_c
