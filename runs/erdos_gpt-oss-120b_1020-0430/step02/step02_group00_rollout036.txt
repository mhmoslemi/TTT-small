import time, numpy as np, math
from scipy import optimize

def _project_to_feasible(h, n, target=None, tol=1e-9, max_iter=1000):
    if target is None: target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # small residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

def _upscale_h(coarse_h, n_fine):

    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

    n = h.size
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)
    # avoid overflow
    ex = np.exp(alpha * (c - max_c))

    conv_rev_full = np.correlate(w_rev, h, mode="full")

    grad = 1.0 - (conv1 + conv2)

            best_val = val
            best_h = h.copy()
        # decay learning rate
        lr *= lr_decay
        alpha = min(alpha * alpha_inc, 80.0)
        # optional early stop if improvement tiny
    return best_h, best_val

def _solve_h(initial_h, max_iter=300):
    # same as in previous solution using SLSQP
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")
    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)
    cons = [{"type":"ineq","fun":ineq},{"type":"eq","fun":eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Feasibility projection (water‑filling)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, target: float = None,
                         tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# Evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Gradient of a smooth approximation of max correlation
# ----------------------------------------------------------------------
def _gradient_softmax_max(h: np.ndarray, alpha: float) -> np.ndarray:
    n = h.size
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)
    w = np.exp(alpha * (c - max_c))
    w /= w.sum()
    L = w.size
    # correlation w with h
    conv_full = np.correlate(w, h, mode="full")
    start = L - 1
    conv1 = conv_full[start:start + n]
    # correlation of reversed w with h
    conv_rev = np.correlate(w[::-1], h, mode="full")
    conv2 = conv_rev[start:start + n]
    grad = 1.0 - (conv1 + conv2)
    return grad


# ----------------------------------------------------------------------
# Simple projected gradient descent on the smoothed objective
# ----------------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, n_iters: int, lr0: float, alpha0: float,
                 lr_decay: float, alpha_inc: float, time_left) -> tuple:
    n = h0.size
    h = h0.copy()
    lr = lr0
    alpha = alpha0
    best_h = h.copy()
    best_val = _c5_from_h(h)
    for it in range(n_iters):
        if time_left() <= 0:
            break
        grad = _gradient_softmax_max(h, alpha)
        h = h - lr * grad
        h = _project_to_feasible(h, n)
        val = _c5_from_h(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        lr *= lr_decay
        alpha = min(alpha * alpha_inc, 80.0)
    return best_h, best_val


# ----------------------------------------------------------------------
# SLSQP polish (same as in the reference implementation)
# ----------------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": _ineq},
            {"type": "eq", "fun": _eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0,
                            method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)
