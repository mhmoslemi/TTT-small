import time
import numpy as np
from scipy import optimize

# ------------------- helper functions ------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h)=n/2 via water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _max_corr(h: np.ndarray) -> int:
    """Maximum of the discrete correlation h * (1‑h) over all shifts."""
    return int(np.max(np.correlate(h, 1.0 - h, mode="full")))


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a discretised step function."""
    n = h.size
    return _max_corr(h) * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t – correlation ≥ 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 2000) -> tuple:
    """Run SLSQP from an initial vector and return (h_opt, C5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = float(_max_corr(h0))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=constraints, options=opts)

    if res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


def random_binary(n: int) -> np.ndarray:
    """Generate a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def anneal_with_time(h_start: np.ndarray, allowed_seconds: float) -> tuple:
    """
    Simulated‑annealing style hill‑climbing on a binary vector.
    Returns the best (h, max_corr) found within the time budget.
    """
    n = h_start.size
    h = h_start.copy()
    cur_max = _max_corr(h)
    best_h = h.copy()
    best_max = cur_max

    start = time.time()
    # temperature schedule
    T0 = max(1.0, cur_max / 2.0)
    T_end = 0.01

    while True:
        if time.time() - start >= allowed_seconds:
            break
        elapsed = time.time() - start
        frac = elapsed / allowed_seconds
        T = T0 * (1.0 - frac) + T_end * frac

        i = np.random.randint(n)
        j = np.random.randint(n)
        while h[i] == h[j]:
            j = np.random.randint(n)

        # propose swap
        h[i], h[j] = h[j], h[i]
        new_max = _max_corr(h)
        delta = new_max - cur_max
        if delta <= 0 or np.exp(-delta / max(T, 1e-12)) > np.random.rand():
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    return best_h, best_max

            sigma = 0.15 * (1.0 - att / attempts)
            trial_h = base_h + sigma * np.random.randn(n)
            trial_h = _project_to_feasible(trial_h, n)
            try:
                h_opt, c5_opt = _solve_h(trial_h, max_iter=2500)
            except Exception:
                h_opt, c5_opt = base_h, best_c5
            # tiny stochastic refinement (similar to earlier local search)
            refined_h = h_opt.copy()
            refined_c5 = c5_opt
            for _ in range(7):
                idx = np.random.choice(n, size=min(5, n), replace=False)
                trial = refined_h.copy()
                trial[idx] += 0.02 * np.random.randn(idx.size)
                trial = _project_to_feasible(trial, n)
                c5_trial = _c5_from_h(trial)
                if c5_trial < refined_c5:
                    refined_h, refined_c5 = trial, c5_trial
            if refined_c5 < best_c5:
                best_h, best_c5, best_max = refined_h, refined_c5, _max_corr(refined_h)
                if best_c5 < target_c5:
                    break

    # fallback to uniform if nothing found
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(n)
