        autocorr = np.correlate(h, h, mode='full')
        # min product
        min_val = autocorr.min()
        # early exit if we already below target
        if best_c5 <= target:
            break
        # find shift(s)
        idxs = np.where(autocorr == min_val)[0]
        # choose the shift furthest from center? Actually we may want worst shift.

        # Let's just pick first.
        k = idxs[0] - (n - 1)
        # Compute gradient
        g = np.zeros_like(h)
        if k >= 0:
            if k != 0:
                g[:n - k] += h[k:]
                g[k:] += h[:n - k]
            else:
                # k == 0 case, gradient is 2*h
                g = 2.0 * h
        else:
            k_abs = -k

            if cand_c5 < best_c5 - 1e-12:

def _c5_from_h(h):
    n = h.shape[0]
    # sum_h via prefix sums
    cs = np.concatenate([[0.0], np.cumsum(h)])
    total = cs[-1]
    # right shifts s=0..n-1
    sum_right = cs[n - np.arange(n)] - cs[0]  # cs[n - k] - 0
    # left shifts s=-(n-1)..-1
    sum_left = total - cs[np.arange(1, n)]
    sum_h = np.concatenate([sum_left[::-1], sum_right])
    # compute autocorrelation via FFT
    L = 2**int(np.ceil(np.log2(2*n-1)))
    H = np.fft.rfft(h, n=L)
    auto = np.fft.irfft(H * np.conj(H), n=L)[:2*n-1]
    # C_s = sum_h - auto
    C = sum_h - auto
    max_C = C.max()
    return (2.0 / n) * max_C

def _improve_h_gradient(h_init, max_iter, time_left, target):
    h = _project_to_simplex_box(h_init, h_init.size/2.0)
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    alpha = 0.5
    for it in range(max_iter):
        # compute sum_h and auto
        cs = np.concatenate([[0.0], np.cumsum(h)])
        total = cs[-1]
        sum_right = cs[n - np.arange(n)] - cs[0]  # cs[n - k] - 0
        sum_left = total - cs[np.arange(1, n)]
        sum_h = np.concatenate([sum_left[::-1], sum_right])
        L = 2**int(np.ceil(np.log2(2*n-1)))
        H = np.fft.rfft(h, n=L)
        auto = np.fft.irfft(H * np.conj(H), n=L)[:2*n-1]
        C = sum_h - auto
        maxC = C.max()
        # compute shift for which maxC occurs
        idx = np.argmax(C)
        shift = idx - (n - 1)
        # compute gradient for this shift
        if shift >= 0:
            mask = np.concatenate([np.ones(n - shift), np.zeros(shift)])
            if shift == 0:
                g = 2.0 * h
            else:
                g = np.zeros_like(h)
                g[:n - shift] += h[shift:]
                g[shift:] += h[:n - shift]
        else:
            k = -shift
            mask = np.concatenate([np.zeros(k), np.ones(n - k)])
            g = np.zeros_like(h)
            g[k:] += h[:n - k]
            g[:n - k] += h[k:]
        grad = mask - g  # gradient of C_s
        # descent step
        improved = False
        for factor in (1.0, 0.5, 0.25):
            cand = h - alpha * factor * grad
            cand = _project_to_simplex_box(cand, n/2.0)
            cand_c5 = _c5_from_h(cand)
            if cand_c5 < best_c5 - 1e-12:
                h = cand
                best_h = cand
                best_c5 = cand_c5
                alpha = min(alpha * 1.2, 1.0)
                improved = True
                break
        if not improved:
            alpha *= 0.7
        # occasional random jitter
        if it % 500 == 0 and it > 0:
            h = _project_to_simplex_box(best_h + 0.01 * np.random.randn(n), n/2.0)
            alpha = max(alpha, 0.05)
        # early exit
        if best_c5 <= target:
            break
        if time_left() < 0.5:
            break
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n ∩ {sum = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Fast C5 evaluation using FFT for autocorrelation and prefix sums."""
    n = h.size
    # ----- sum_h for each shift -----
    cs = np.concatenate([[0.0], np.cumsum(h)])
    total = cs[-1]                     # should equal n/2
    # shifts >= 0
    sum_right = cs[n - np.arange(n)] - cs[0]          # length n, shift 0..n-1
    # shifts < 0
    sum_left = total - cs[np.arange(1, n)]            # length n-1, shifts -(n-1)..-1
    sum_h = np.concatenate([sum_left[::-1], sum_right])  # length 2n-1

    # ----- autocorrelation via FFT -----
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=L)
    auto = np.fft.irfft(H * np.conj(H), n=L)[: 2 * n - 1]

    # C_s = sum_h - auto, take maximum, scale by dx = 2/n
    max_C = (sum_h - auto).max()
    return (2.0 / n) * max_C


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _tent_h(n: int) -> np.ndarray:
    """Symmetric tent shape, then projected to the feasible set."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """SLSQP refinement from a feasible start."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = corr0.max()
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    h_opt = h0 if not res.success else res.x[:n]
    return _project_to_simplex_box(h_opt, n / 2.0), _c5_from_h(h_opt)


def _improve_h_gradient(h0: np.ndarray, max_iter: int, time_left, target):
    """Projected gradient descent on the max‑overlap objective."""
    n = h0.size
    h = _project_to_simplex_box(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    alpha = 0.5
    L = 1 << ((2 * n - 1).bit_length())  # FFT length (power of two)

    for it in range(max_iter):
        # prefix sums for sum_h
        cs = np.concatenate([[0.0], np.cumsum(h)])
        total = cs[-1]
        sum_right = cs[n - np.arange(n)] - cs[0]          # shift >=0
        sum_left = total - cs[np.arange(1, n)]            # shift <0
        sum_h = np.concatenate([sum_left[::-1], sum_right])

        # autocorrelation via FFT (linear, not circular)
        H = np.fft.rfft(h, n=L)
        auto = np.fft.irfft(H * np.conj(H), n=L)[: 2 * n - 1]

        C = sum_h - auto
        idx = np.argmax(C)
        shift = idx - (n - 1)          # actual integer shift producing max overlap

        # --- gradient of C for this shift ---
        if shift >= 0:
            mask = np.concatenate([np.ones(n - shift), np.zeros(shift)])
            if shift == 0:
                g = 2.0 * h
            else:
                g = np.zeros_like(h)
                g[: n - shift] += h[shift:]
                g[shift:] += h[: n - shift]
        else:
            k = -shift
            mask = np.concatenate([np.zeros(k), np.ones(n - k)])
            g = np.zeros_like(h)
            g[k:] += h[: n - k]
            g[: n - k] += h[k:]

        grad = mask - g                     # ∇C_s
        improved = False

        # try a few step scales
        for factor in (1.0, 0.5, 0.25):
            cand = h - alpha * factor * grad
            cand = _project_to_simplex_box(cand, n / 2.0)
            cand_c5 = _c5_from_h(cand)
            if cand_c5 < best_c5 - 1e-12:
                h = cand
                best_h = cand
                best_c5 = cand_c5
                alpha = min(alpha * 1.2, 1.0)
                improved = True
                break
        if not improved:
            alpha *= 0.7

        # occasional jitter to escape plateaus
        if (it + 1) % 500 == 0:
            h = _project_to_simplex_box(best_h + 0.01 * np.random.randn(n), n / 2.0)
            alpha = max(alpha, 0.05)

        if best_c5 <= target:
            break
        if time_left() < 0.5:
            break
    return best_h, best_c5
