def binary_anneal(h0, rng, max_iters, init_temp, decay):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    temp = init_temp
    n = len(h)
    for it in range(max_iters):
        # perform swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        c = compute_c5(h)
        delta = c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / temp):
                accept = True
        if accept:
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
        temp *= decay
    return best_h, best_c

def compute_c5(h):

def solve_mip_binary(n, time_limit):
    import cvxpy as cp, numpy as np, time
    # Ensure n even
    assert n % 2 == 0
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n // 2, T >= 0]
    # Build constraints for k = 1..n-1
    for k in range(1, n):
        i_range = n - k
        # w_k vector of length i_range
        w = cp.Variable(i_range, boolean=True)
        # source constraint: w <= h[:i_range]
        constraints.append(w <= h[:i_range])
        # w <= 1 - h[k:]
        constraints.append(w <= 1 - h[k:])
        # w >= h[:i_range] - h[k:]
        constraints.append(w >= h[:i_range] - h[k:])
        # out_sum = cp.sum(h[n-k:])  # trailing region
        if k > 0:
            out_sum = cp.sum(h[n - k:])
        else:
            out_sum = 0
        constraints.append(cp.sum(w) + out_sum <= T)
    prob = cp.Problem(cp.Minimize(T), constraints)
    # Solve with time limit
    prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)
    if h.value is None:
        return None, None
    h_sol = np.round(h.value).astype(int)
    T_val = int(round(T.value))
    return h_sol, T_val

import time
import math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic helpers
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random 0/1‑vector with Σh = n/2."""
    target = n / 2.0
    h = np.zeros(n)
    idx = rng.choice(n, int(target), replace=False)
    h[idx] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < math.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


# --------------------------------------------------------------
# SLSQP soft‑max surrogate
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up

        h_ref, c_ref = quick_refine_ext(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to higher resolution and deep refinement
    # ------------------------------------------------------------------
    # Upsample by factor 4 (90 → 360)
    if time.time() - start_time < budget_s * 0.80:
        h_up = upsample_h(best_h, factor=4, rng=rng)
        h_up_ref, c_up_ref = deep_refine_ext(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = h_up_ref.size

    # Optional second up‑sample (720 points) if time remains
    if time.time() - start_time < budget_s * 0.95 and n * 2 <= 800:
        h_up2 = upsample_h(best_h, factor=2, rng=rng)
        h_up2_ref, c_up2_ref = deep_refine_ext(h_up2, rng)
        if c_up2_ref < best_c:
            best_h, best_c = h_up2_ref, c_up2_ref
            n = h_up2_ref.size

    # ------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
```
