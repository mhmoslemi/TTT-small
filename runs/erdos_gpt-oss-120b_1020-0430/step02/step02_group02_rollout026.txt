def grad_overlap(h, lag):
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        # first part
        g[:n-lag] += h[lag:]   # h_{i+lag}
        # second part
        g[lag:] += h[:n-lag]   # h_{i-lag}
    else:
        lag = -lag
        # corresponds to shift -lag, same as positive but reversed
        # but easier: simply compute for lag<0 same as for positive using symmetry:
        # Overlap sum_i h_i h_{i-lag} = sum_i h_{i+lag} h_i
        # So gradient same as for positive lag.
        g[lag:] += h[:n-lag]    # h_{i-lag}
        g[:n-lag] += h[lag:]    # h_{i+lag}
    return g
```

        max_idx = np.argmax(corr)

        c_current = max_corr * (2.0 / n)

            best_h = h.copy()
        lag = max_idx - (n - 1)

        if idx_low.size == 0 or idx_high.size == 0:

        j = idx_high[np.argmax(g[idx_high])]

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a feasible h (integral constraint already satisfied)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _grad_overlap(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the overlap sum_i h[i] h[i+lag] w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            g[: n - lag] += h[lag:]
            g[lag:] += h[: n - lag]
    else:
        lag = -lag
        if lag < n:
            g[lag:] += h[: n - lag]
            g[: n - lag] += h[lag:]
    return g


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# SLSQP refinement (kept from previous solution but with a higher iteration cap)
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Simple pairwise epsilon exchange hill‑climb (accepts only improvements)."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    rng = np.random.default_rng()
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        down = best_h[i]
        up = 1.0 - best_h[j]
        eps = min(down, up) * rng.random() * 0.5
        if eps <= 0:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Gradient‑guided local search
# ----------------------------------------------------------------------
def _gradient_search(h0: np.ndarray, deadline: float) -> tuple:
    """Iteratively move mass from low‑gradient to high‑gradient positions."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    step = 0.05
    rng = np.random.default_rng()
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        c_cur = corr[idx] * (2.0 / n)
        if c_cur < best_c:
            best_c, best_h = c_cur, h.copy()
        lag = idx - (n - 1)
        g = _grad_overlap(h, lag)

        # candidate indices
        low_mask = h > 1e-9
        high_mask = h < 1.0 - 1e-9
        low_idx = np.where(low_mask)[0]
        high_idx = np.where(high_mask)[0]
        if low_idx.size == 0 or high_idx.size == 0:
            break
        i = low_idx[np.argmin(g[low_idx])]
        j = high_idx[np.argmax(g[high_idx])]
        if g[j] <= g[i] + 1e-12:
            # no clear improvement – pick a random admissible pair
            i = low_idx[rng.integers(low_idx.size)]
            j = high_idx[rng.integers(high_idx.size)]
            if i == j:
                continue

        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            step *= 0.9
            continue
        eps = min(step, max_move)
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        c_new = _c5_from_h(cand)

        if c_new < c_cur - 1e-12:
            h = cand
            step = min(step * 1.05, 0.2)
        else:
            step *= 0.9

        # occasional tiny noise injection to escape plateaus
        if rng.random() < 0.004:
            noise = rng.normal(scale=step * 0.5, size=n)
            h_noise = np.clip(h + noise, 0.0, 1.0)
            h_noise = _project_to_feasible(h_noise, n)
            c_noise = _c5_from_h(h_noise)
            if c_noise < best_c:
                h, best_c, best_h = h_noise, c_noise, h_noise.copy()
    return best_h, best_c
