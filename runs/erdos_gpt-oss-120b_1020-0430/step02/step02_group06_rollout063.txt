def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Use water-filling simple: compute difference and adjust proportionally among free entries.
    # We'll adapt algorithm: if s > target, reduce values that are >0 uniformly; else increase values<1.
    # iterate few steps (max 10) for convergence.
    for _ in range(10):
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    # final adjustment:
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first element that can accommodate
        for i in range(len(h)):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h
```

    best_h = h.copy()
    best_c5 = compute_c5(h)
    # initial temperature set based on best_c5
    T0 = best_c5 * 0.2  # maybe 0.1?
    if T0 < 1e-4:
        T0 = 0.001
    T = T0
    iteration = 0
    # schedule: decrease T geometrically
    while time.time() - start_time < time_budget:
        # choose two indices and amount
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0 and h[j] >= 1:
            # cannot transfer
            continue
        # max transfer
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            continue
        # scale step by temperature and max_transfer
        delta = rng.random() * max_transfer * min(1.0, T / T0 * 0.5 + 0.01)
        # apply
        h[i] -= delta
        h[j] += delta
        # compute
        c5 = compute_c5(h)
        diff = c5 - best_c5
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            # accept
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        # cooling
        iteration += 1
        if iteration % 1000 == 0:
            # exponential cooling
            T *= 0.995
            if T < 1e-6:
                T = 1e-6
    return best_h, best_c5

def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # water-filling iterative
    for _ in range(10):
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    # final small correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(len(h)):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def solve_slsqp(initial_h, n, rng, max_iter=1000):
    # Use SLSQP via scipy.optimize
    from scipy import optimize
    # initial solution variables
    h0 = project_to_feasible(initial_h, n/2)
    # compute initial t = max correlation
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - n/2.0

    def obj(x):
        return x[-1]

    cons = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq},
    ]
    res = optimize.minimize(
        obj,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    )
    if not res.success:
        # fallback to x0
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = project_to_feasible(h_opt, n/2)
    c5_val = compute_c5(h_opt)
    return h_opt, c5_val

import time
import numpy as np

# -------------------- Helper functions -------------------- #

def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to the target by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(12):
        diff = s - target
        if abs(diff) < 1e-12:
            break
        free = h > 0 if diff > 0 else h < 1
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    # final tiny correction
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        for i in range(len(h)):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """C₅ for a discrete step function h on [0,2] (dx = 2/n)."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * np.conj(G), n=m)
    corr = conv[: 2 * n - 1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector in [0,1]^n with sum = n/2."""
    raw = rng.random(n)
    return project_to_feasible(raw, n / 2.0)


def anneal_h(h0: np.ndarray, n: int, max_seconds: float,
             rng: np.random.Generator) -> tuple:
    """Simple simulated‑annealing optimisation of C₅."""
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = cur_c5 * 0.05 + 0.001          # initial temperature
    step_scale = 0.5                    # initial move magnitude
    start = time.time()
    it = 0
    while time.time() - start < max_seconds:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # try to move mass from i to j
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            # opposite direction
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer <= 0:
                continue
            i, j = j, i
        delta = max_transfer * rng.random() * step_scale
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta
        it += 1
        if it % 2000 == 0:
            T = max(1e-6, T * 0.9)
            step_scale = max(0.01, step_scale * 0.9)
    return best_h, best_c5


def local_refine(h: np.ndarray, attempts: int, delta: float,
                 rng: np.random.Generator) -> tuple:
    """Stochastic hill‑climbing around h."""
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(attempts):
        idx = rng.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * rng.standard_normal(idx.size)
        trial = project_to_feasible(trial, n / 2.0)
        c5 = compute_c5(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h, best_c5


def solve_slsqp(initial_h: np.ndarray, n: int,
                max_iter: int = 800) -> tuple:
    """Refine a solution with SciPy's SLSQP."""
    try:
        from scipy import optimize
    except Exception:  # pragma: no cover
        return initial_h, compute_c5(initial_h)

    h0 = project_to_feasible(initial_h, n / 2.0)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    def obj(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = optimize.minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)
