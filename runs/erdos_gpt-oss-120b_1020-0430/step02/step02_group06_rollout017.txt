def project_feasible(h, target, tol=1e-9, max_iter=100):
    # clip
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        h[free_idx] -= delta
        # after adjustment clip again
        h = np.clip(h, 0.0, 1.0)
    # final correction: adjust tiny residual by adding to first non-bound.
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # try to distribute residual
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
import scipy.optimize as opt

# -------------------------- feasibility -------------------------- #
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Clip to [0,1] and adjust sum to target by water-filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# -------------------------- evaluation -------------------------- #
def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap for discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse to fine grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _random_binary(n: int) -> np.ndarray:
    """Uniform random binary vector with exactly n/2 ones."""
    h = np.zeros(n)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def _random_continuous(n: int) -> np.ndarray:
    """Uniform random values projected to feasibility."""
    h = np.random.rand(n)
    return _project_to_feasible(h, n / 2.0)


# -------------------------- local refinement (SLSQP) -------------------------- #
def _ineq_constraints(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _refine_h(h_initial: np.ndarray, max_iter: int = 400) -> tuple:
    """SLSQP step to tighten the max‑overlap."""
    n = h_initial.size
    target = n / 2.0
    h0 = _project_to_feasible(h_initial.copy(), target)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "eq", "fun": _eq_constraint},
        {"type": "ineq", "fun": _ineq_constraints},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    if res.success and res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, target)
    return h_opt, _c5_from_h(h_opt)


# -------------------------- genetic algorithm -------------------------- #
def _ga_optimize(base_h: np.ndarray, n: int, target_bound: float, time_budget: float, start_time: float) -> tuple:
    popsize = 30
    elite_num = 2
    target = n / 2.0
    # initialise population
    pop = [_project_to_feasible(base_h.copy(), target)]
    pop.append(_random_binary(n))
    while len(pop) < popsize:
        pop.append(_random_binary(n) if np.random.rand() < 0.5 else _random_continuous(n))

    scores = np.array([_c5_from_h(h) for h in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    sigma = 0.3
    generation = 0

    while True:
        if time.time() - start_time > time_budget - 4.0:
            break
        if best_score <= target_bound:
            break
        generation += 1

        # elite preservation
        elite_idx = np.argsort(scores)[:elite_num]
        new_pop = [pop[i].copy() for i in elite_idx]

        # generate offspring
        while len(new_pop) < popsize:
            # tournament selection for parents
            cand = np.random.choice(popsize, 3, replace=False)
            p1 = pop[cand[np.argmin(scores[cand])]]
            cand = np.random.choice(popsize, 3, replace=False)
            p2 = pop[cand[np.argmin(scores[cand])]]

            # uniform crossover
            mask = np.random.rand(n) < 0.5
            child = np.where(mask, p1, p2).copy()

            # Gaussian mutation
            if np.random.rand() < 0.9:
                child += sigma * np.random.randn(n)

            # occasional binary rounding
            if np.random.rand() < 0.25:
                k = int(target)
                idx = np.argpartition(child, -k)[-k:]
                bin_child = np.zeros(n)
                bin_child[idx] = 1.0
                child = bin_child
                # already feasible (sum = target)
            else:
                child = _project_to_feasible(child, target)

            new_pop.append(child)

        pop = new_pop
        scores = np.array([_c5_from_h(h) for h in pop])

        # update best
        cur_idx = scores.argmin()
        if scores[cur_idx] < best_score:
            best_score = scores[cur_idx]
            best_h = pop[cur_idx].copy()

        sigma *= 0.99  # anneal mutation size

        # periodic local refinement on elites
        if generation % 30 == 0:
            refined_any = False
            for i in elite_idx:
                h_ref, s_ref = _refine_h(pop[i], max_iter=300)
                if s_ref + 1e-9 < scores[i]:
                    pop[i] = h_ref
                    scores[i] = s_ref
                    refined_any = True
            if refined_any:
                cur_idx = scores.argmin()
                if scores[cur_idx] < best_score:
                    best_score = scores[cur_idx]
                    best_h = pop[cur_idx].copy()

    # final refinement of overall best
    h_ref, s_ref = _refine_h(best_h, max_iter=600)
    if s_ref < best_score:
        best_h, best_score = h_ref, s_ref
    return best_h, best_score
