def project_to_feasible(h, target):
    # similar to earlier
    h = np.clip(h, 0, 1)
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # water-filling: adjust up/down uniformly among free entries
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] -= delta
        h = np.clip(h, 0, 1)
        diff = h.sum() - target
    # final correction
    residual = target - h.sum()
    for i in range(len(h)):
        if residual > 0 and h[i] < 1:
            add = min(residual, 1 - h[i])
            h[i] += add
            residual -= add
        elif residual < 0 and h[i] > 0:
            sub = min(-residual, h[i])
            h[i] -= sub
            residual += sub
        if abs(residual) < 1e-12:
            break
    return np.clip(h, 0, 1)

import time, math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and adjust to achieve the desired sum (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) <= tol:
        return h
    # iterative water‑filling
    for _ in range(200):
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
    # tiny final tweak
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Integral value max_k ∫ h(x)(1‑h(x+k)) dx approximated on a uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _slsqp_optimize(h0: np.ndarray, max_iter: int = 800) -> tuple:
    """Refine a feasible vector with SLSQP (minimise the maximal overlap)."""
    n = h0.size
    h0 = _project_to_feasible(h0, n / 2.0)
    dx = 2.0 / n
    # initial bound for t
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full")) * dx

    def obj(x):
        return x[-1]  # minimise t

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr * dx  # t >= each correlation*dx

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        obj, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _simulated_annealing(start_h: np.ndarray, time_limit: float) -> tuple:
    """Annealing that preserves the sum exactly by epsilon swaps."""
    n = start_h.size
    h = start_h.copy()
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    cur_c = best_c

    # temperature schedule
    T0, Tf = 0.02, 1e-6
    it = 0
    while time.time() < time_limit:
        it += 1
        T = T0 * ((Tf / T0) ** (it / 500000.0))  # very slow exponential cooling

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.5  # up‑to‑half of the feasible range
        h[i] -= eps
        h[j] += eps

        new_c = _c5_from_h(h)
        delta = new_c - cur_c
        if delta <= 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert move
            h[i] += eps
            h[j] -= eps

        # occasional polishing with a few SLSQP steps
        if it % 60000 == 0:
            refined, refined_c = _slsqp_optimize(best_h, max_iter=150)
            if refined_c < best_c:
                best_h, best_c = refined, refined_c
                h = refined.copy()
                cur_c = refined_c
    return best_h, best_c


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Deterministic improvement: epsilon exchange between two coordinates."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    steps = 0
    while steps < max_steps and time.time() < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * np.random.rand() * 0.3
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
        steps += 1
    return best_h, best_c
