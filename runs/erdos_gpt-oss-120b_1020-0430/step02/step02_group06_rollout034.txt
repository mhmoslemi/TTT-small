            break
        elapsed = time.time() - start_time
        if elapsed > total_alloc:
            break
        # temperature geometric
        T = T_start * (T_end / T_start) ** (elapsed / total_alloc) if total_alloc > 0 else T_end

        j_idx = np.random.randint(len(zeros))

        j = zeros[j_idx]

        h[i] = 0.0
        h[j] = 1.0

import time, math
import numpy as np
from scipy import optimize

# ---------- basic utilities ----------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- SLSQP local optimisation ----------

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    h_opt = _project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    return h_opt, _c5_from_h(h_opt)


def _local_refine(h: np.ndarray, attempts: int = 8, delta: float = 0.015) -> np.ndarray:
    n = h.size
    best = h.copy()
    best_c5 = _c5_from_h(best)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best, best_c5 = trial, c5
    return best

def _interpolate_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _make_binary_feasible(h: np.ndarray) -> np.ndarray:
    n = h.size
    target = n // 2
    b = (h > 0.5).astype(float)
    cur = int(b.sum())

        idx = np.flatnonzero(b == 0)
        on = np.random.choice(idx, size=target - cur, replace=False)
        b[on] = 1.0
    return b


# ---------- binary hill‑climbing ----------

def _binary_hillclimb(h_start: np.ndarray, time_left, target_c5: float) -> tuple:
    """Stochastic hill‑climbing on binary vectors with fixed sum."""
    n = h_start.size
    h = h_start.copy()
    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5

    ones = list(np.flatnonzero(h > 0.5))
    zeros = list(np.flatnonzero(h <= 0.5))

    # cooling schedule parameters
    T_start, T_end = 0.02, 1e-6
    start = time.time()

    while time_left() > 0.05:
        elapsed = time.time() - start
        if elapsed == 0:
            T = T_start
        else:
            T = T_start * (T_end / T_start) ** (elapsed / max(1.0, time_left()))
        # pick a random 1↔0 swap
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i, j = ones[i_idx], zeros[j_idx]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _c5_from_h(h)

        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # Metropolis acceptance
            delta = new_c5 - cur_c5
            if T > 0:
                prob = math.exp(-delta / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
                if best_c5 < target_c5:
                    return best_h, best_c5
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5


# ---------- main driver ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # parent construction, if supplied
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    target = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate grid sizes (even, not too large)
    candidates = []
    if parent_h is not None:
        candidates.append(parent_h.size)
    for n in (80, 96, 112, 128, 144, 160, 192, 224, 256):
        candidates.append(n)
    candidates = sorted({n for n in candidates if n % 2 == 0 and n <= 256})

    for n in candidates:
        if time_left() < 4.0:
            break

        # initialise a binary start vector for this size
        if parent_h is not None and parent_h.size == n:
            start = _make_binary_feasible(parent_h)
        elif parent_h is not None:
            start = _make_binary_feasible(_interpolate_h(parent_h, n))
        else:
            idx = np.random.choice(n, size=n // 2, replace=False)
            start = np.zeros(n, dtype=float)
            start[idx] = 1.0

        # binary hill‑climbing
        h_bin, c5_bin = _binary_hillclimb(start, time_left, target)
        if c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin.copy(), c5_bin, n
            if best_c5 < target:
                break

        # a short SLSQP polish (only if we still have time)
        if time_left() > 2.0:
            h_opt, c5_opt = _solve_h(best_h, max_iter=300)
            h_opt = _local_refine(h_opt, attempts=6, delta=0.012)
            c5_opt = _c5_from_h(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                if best_c5 < target:
                    break

    # fallback in the unlikely case nothing was found
    if best_h is None:
        n = 96
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
