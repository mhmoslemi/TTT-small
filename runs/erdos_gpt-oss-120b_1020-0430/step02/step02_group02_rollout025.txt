def hill_climb(h0, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    current_c = best_c
    n = len(h)
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        # sample i, j such that h[i] > 1e-12, h[j] < 1-1e-12
        # Use np.random.choice with probabilities maybe proportion to h[i], (1-h[j]) to bias?
        # Simpler: i = np.random.randint(n); if h[i]==0: continue else j...
        # We'll attempt up to a few tries.
        i = np.random.randint(n)
        if h[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if h[j] >= 1 - 1e-12:
            continue
        if i == j:
            continue
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            continue
        # epsilon scaling factor
        scale = 0.5 * (0.9 ** (it // 5000))  # decays gradually every 5k iterations
        eps = max_transfer * np.random.rand() * scale
        if eps <= 0:
            continue
        # compute new C5
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        if new_c < current_c - 1e-12:
            h = h_new
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
    return best_h, best_c

import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Helper utilities (projection, upscaling, C5 evaluation)
# ------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust to achieve sum(h)=n/2 via a water‑filling loop."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)

    # tiny correction to get exact sum
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a given step function (assumes correct integral)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ------------------------------------------------------------
# SLSQP optimisation (used on modest sizes)
# ------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    """Run a single SLSQP optimisation, returning (h_opt, c5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)  # start with current max correlation
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

def _guided_move(h: np.ndarray):
    """Return (i, j, eps) targeting the worst shift, or (None,None,None)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    worst_idx = int(np.argmax(corr))
    offset = worst_idx - (n - 1)
    if offset == 0:
        return None, None, None
    idx = np.arange(n)
    partner = idx + offset
    valid = (partner >= 0) & (partner < n)
    high = (h > 0.5) & (h[partner] > 0.5) & valid
    src_candidates = idx[high]
    if src_candidates.size == 0:
        return None, None, None
    i = np.random.choice(src_candidates)
    low_targets = np.where(h < 0.5)[0]
    if low_targets.size == 0:
        return None, None, None
    j = np.random.choice(low_targets)
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return None, None, None
    eps = max_transfer * np.random.rand() * 0.5
    return i, j, eps


def _hill_climb_time(start_h: np.ndarray, time_limit: float) -> tuple:
    """
    Perform many exchange moves while time permits.
    Returns (best_h, best_c5).
    """
    n = start_h.size
    best_h = start_h.copy()
    best_c = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    start = time.time()
    it = 0
    while True:
        if time.time() >= time_limit:
            break
        it += 1

        if np.random.rand() < 0.07:                     # occasional guided move
            i, j, eps = _guided_move(cur_h)
            if i is None:
                continue
        else:                                           # generic random exchange
            i = np.random.randint(n)
            if cur_h[i] <= 1e-12:
                continue
            j = np.random.randint(n)
            if j == i or cur_h[j] >= 1.0 - 1e-12:
                continue
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            # scale epsilon: start large, shrink with progress
            elapsed = time.time() - start
            frac = elapsed / (time_limit - start + 1e-12)
            scale = 0.5 * (1.0 - frac) + 0.01
            eps = max_transfer * np.random.rand() * scale

        # apply move
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = _c5_from_h(cand)
