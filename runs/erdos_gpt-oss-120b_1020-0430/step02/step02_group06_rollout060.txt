def guided_move(h, step):
    n = h.shape[0]
    # compute correlation array
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr)
    s = k - (n - 1)
    # Determine overlapped indices for shift s
    if s >= 0:
        i_start = 0
        i_end = n - s   # exclusive
        i_range = np.arange(i_start, i_end)
        hi = h[i_range]
        hjs = h[i_range + s]
    else:
        i_start = -s
        i_end = n
        i_range = np.arange(i_start, i_end)
        hi = h[i_range]
        hjs = h[i_range + s]  # negative shift => i_range + s < i_range
    # compute contributions
    contrib = hi * (1 - hjs)
    # Choose i (source) where contrib is large and hi>0
    # Instead of just max, we can weight by hi.
    # Let's pick source index i_src from i_range where hi > 1e-12
    # with probability proportional to contrib
    # For simplicity, choose max
    idx_local = np.argmax(contrib)
    i_src = i_range[idx_local]
    # Now find candidate destinations j
    # try outside overlapped region first
    if s >= 0:
        outside = np.arange(n - s, n)  # right side
    else:
        outside = np.arange(0, -s)  # left side
    # filter those where h[j] < 1
    dest_candidates = outside[h[outside] < 1 - 1e-12]
    if dest_candidates.size > 0:
        # choose random dest
        j_dest = np.random.choice(dest_candidates)
        # compute max delta
        max_delta = min(step, h[i_src], 1 - h[j_dest])
        if max_delta > 0:
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i_src] -= delta
            h_new[j_dest] += delta
            return h_new
    # If no outside, pick a destination within overlapped region that has lower h_{j+s}
    # Compute for all potential dest indices in overlapped region
    # We'll compute diff = h[j+s] (since moving from i to j changes correlation by delta*(h[i+s] - h[j+s]))
    # Want h[i+s] > h[j+s] to reduce.
    # We'll restrict dest to indices where h[j] < 1 and h[j+s] is minimal.
    # For s >=0:
    if s >= 0:
        dest_range = i_range  # same as overlapped region
    else:
        dest_range = i_range
    # Compute potential dest candidates
    # Ensure j != i_src
    mask = (dest_range != i_src) & (h[dest_range] < 1 - 1e-12)
    if np.any(mask):
        dest_idx = dest_range[mask]
        # compute h_j_s
        if s >= 0:
            h_js = h[dest_idx + s]
        else:
            h_js = h[dest_idx + s]
        # choose dest with minimal h_js (most decrease)
        j_dest = dest_idx[np.argmin(h_js)]
        max_delta = min(step, h[i_src], 1 - h[j_dest])
        if max_delta > 0:
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i_src] -= delta
            h_new[j_dest] += delta
            return h_new
    # fallback: random move as before
    # choose random source with h>0 and random dest with h<1
    src_indices = np.where(h > 1e-12)[0]
    dest_indices = np.where(h < 1 - 1e-12)[0]
    if src_indices.size == 0 or dest_indices.size == 0:

    i_src = np.random.choice(src_indices)
    j_dest = np.random.choice(dest_indices)
    max_delta = min(step, h[i_src], 1 - h[j_dest])
    if max_delta <= 0:
        return h.copy()
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[i_src] -= delta
    h_new[j_dest] += delta

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 via water-filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    # simple water-filling loop, sufficient for modest n
    for _ in range(500):
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    # final tiny correction (distribute linearly)
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-14:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-14:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    # mid‑point positions of each cell
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _guided_move(h: np.ndarray, step: float) -> np.ndarray:
    """Perform a move that tries to reduce the current worst shift."""
    n = h.size
    # identify worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr)
    s = k - (n - 1)                     # shift (positive = right)

    # overlapping indices for this shift
    if s >= 0:
        i_start, i_end = 0, n - s
        i_range = np.arange(i_start, i_end)
        hi = h[i_range]
        hjs = h[i_range + s]
    else:
        i_start, i_end = -s, n
        i_range = np.arange(i_start, i_end)
        hi = h[i_range]
        hjs = h[i_range + s]

    # contribution of each overlapped position to the worst correlation
    contrib = hi * (1.0 - hjs)
    src_local = np.argmax(contrib)
    i_src = i_range[src_local]

    # try to move mass to a non‑overlapping cell (always beneficial for this shift)
    if s >= 0:
        outside = np.arange(n - s, n)          # right tail
    else:
        outside = np.arange(0, -s)              # left tail

    # candidates that can receive mass
    dest_candidates = outside[h[outside] < 1.0 - 1e-12]
    if dest_candidates.size:
        j_dest = np.random.choice(dest_candidates)
        max_delta = min(step, h[i_src], 1.0 - h[j_dest])
        if max_delta > 0:
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i_src] -= delta
            h_new[j_dest] += delta
            return h_new

    # otherwise move within overlap to a position with smaller h_{j+s}
    # (reduces the worst correlation if possible)
    mask = (i_range != i_src) & (h[i_range] < 1.0 - 1e-12)
    if mask.any():
        dest_idxs = i_range[mask]
        if s >= 0:
            h_js = h[dest_idxs + s]
        else:
            h_js = h[dest_idxs + s]
        j_dest = dest_idxs[np.argmin(h_js)]
        max_delta = min(step, h[i_src], 1.0 - h[j_dest])
        if max_delta > 0:
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i_src] -= delta
            h_new[j_dest] += delta
            return h_new

    # fallback: generic random transfer between any admissible pair
    srcs = np.where(h > 1e-12)[0]
    dsts = np.where(h < 1.0 - 1e-12)[0]
    if srcs.size == 0 or dsts.size == 0:
        return h.copy()
    i_src = np.random.choice(srcs)
    j_dest = np.random.choice(dsts)
    max_delta = min(step, h[i_src], 1.0 - h[j_dest])
    if max_delta <= 0:
        return h.copy()
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[i_src] -= delta
    h_new[j_dest] += delta
    return h_new


def _optimize_on_grid(base_h: np.ndarray, time_budget: float, target: float = 0.3808):
    """Hill‑climbing / simulated‑annealing on a fixed grid size."""
    n = base_h.size
    h = base_h.copy()
    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    it = 0
    # parameters for schedules
    step_max = 0.12
    step_min = 0.001
    T0 = 0.001
    T_min = 1e-7
    # rough upper bound on iterations (will also stop by time)
    max_iters = 200000

    while it < max_iters and (time.time() - start) < time_budget:
        # decay schedules
        frac = it / max_iters
        step = step_max * (1.0 - frac) + step_min
        T = T0 * (1.0 - frac) + T_min

        h_new = _guided_move(h, step)
        new_c5 = _c5_from_h(h_new)

        # simulated‑annealing acceptance
        if new_c5 < cur_c5 or np.random.rand() < np.exp((cur_c5 - new_c5) / T):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                if best_c5 < target:
                    break

        # occasional re‑projection to keep sum exact (tiny floating drift)
        if (it & 511) == 0:
            total = h.sum()
            if abs(total - n / 2.0) > 1e-10:
                # adjust a random index
                idx = np.random.randint(n)
                h[idx] += (n / 2.0) - total
                h = np.clip(h, 0.0, 1.0)

        it += 1

    return best_h, float(best_c5)
