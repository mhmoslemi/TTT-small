        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5(h: np.ndarray) -> float:
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def _ineq_constraint(x: np.ndarray):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1 - h, mode='full')

def _eq_constraint(x: np.ndarray):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)

def _solve_h(initial_h: np.ndarray, max_iter: int = 400):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'ineq', 'fun': _ineq_constraint},
                   {'type': 'eq', 'fun': _eq_constraint}]
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5(h_opt)

def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float):
    n = h.size
    best_h = h.copy()
    best_c = _c5(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

def _binary_search(n: int, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    target_local = 0.3808
    ones_target = n // 2
    while time.time() - start < time_budget:
        # random binary init with exact half ones
        h = np.zeros(n, dtype=np.int8)
        idx = np.random.choice(n, ones_target, replace=False)
        h[idx] = 1
        c = _c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # simple simulated annealing
        T = 0.02
        it = 0
        while time.time() - start < time_budget:
            ones_idx = np.flatnonzero(h)
            zeros_idx = np.flatnonzero(1 - h)
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            h[i], h[j] = 0, 1
            new_c = _c5(h)
            delta = new_c - c
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                c = new_c
                if c < best_c:
                    best_c, best_h = c, h.copy()
                if it % 500 == 0:
                    corr = np.correlate(h, 1 - h, mode='full')
                    max_corr = corr.max()
                    if max_corr == 0:
                        break
                    idx_max = np.argmax(corr)
                    k = idx_max - (n - 1)
                    s = max(0, -k)
                    e = min(n, n - k)
                    cand_i = []
                    cand_j = []
                    for ii in range(s, e):
                        if h[ii] == 1 and h[ii + k] == 0:
                            cand_i.append(ii)
                        elif h[ii] == 0 and h[ii + k] == 1:
                            cand_j.append(ii)
                    if cand_i and cand_j:
                        i2 = np.random.choice(cand_i)
                        j2 = np.random.choice(cand_j)
                        h[i2], h[j2] = 0, 1
                        new_c2 = _c5(h)
                        if new_c2 < c:
                            c = new_c2
                            if c < best_c:
                                best_c, best_h = c, h.copy()
                        else:
                            h[i2], h[j2] = 1, 0
            else:
                h[i], h[j] = 1, 0
            if it % 1000 == 0 and T > 1e-5:
                T *= 0.9
            it += 1
            if best_c < target_local:
                return best_h, best_c
        # end inner while
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))
    target = 0.3808
    best_h = None
    best_c = np.inf
    best_n = None

    # initial construction if provided
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None
    if init_h is not None:
        h_ref, c_ref = _solve_h(init_h, max_iter=400)
        if c_ref < best_c:
            best_h, best_c, best_n = h_ref, c_ref, init_h.size
        if best_c < target:
            return best_h.tolist(), float(best_c), int(best_n)

    # candidate sizes (descending, larger first)
    candidates = [256, 224, 192, 160, 128, 112, 96, 80]
    # reserve a bit for final polishing
    reserve = max(5.0, budget_s * 0.07)
    while remaining() > reserve and candidates:
        n = candidates.pop(0)
        alloc = min(remaining() - reserve, max(5.0, remaining() * 0.3))
        h_bin, c_bin = _binary_search(n, alloc)
        if h_bin is not None and c_bin < best_c:
            best_h = h_bin.astype(float)
            best_c = c_bin
            best_n = n
        if best_c < target:
            break

    # final continuous refinement if time permits
    if best_h is not None and remaining() > 2.0:
        # SLSQP refinement
        h_ref, c_ref = _solve_h(best_h, max_iter=600)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref
        # epsilon hill climbing
        limit = start_time + budget_s - 0.1
        best_h, best_c = _hill_climb(best_h, max_steps=4000, time_limit=limit)

    # fallback
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c = _c5(best_h)
        best_n = n

    return best_h.tolist(), float(best_c), int(best_n)
