def _mass_transfer_climb(h, max_iters, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for it in range(max_iters):
        now = time.time()
        if now > time_limit:
            break

        corr = np.correlate(best_h, 1.0 - best_h, mode='full')

        # Actually best_c is this max_corr.
        s = max_idx - (n - 1)
        shift = abs(s)

        w = np.zeros(n)

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Basic helpers (same semantics as in the previous solution)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a given h (integral over [0,2] already ensured)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Core optimisation: SLSQP (quick coarse optimisation)
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 200) -> tuple:
    """One SLSQP run, returning a feasible h and its C5."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0) * (2.0 / n)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

def _mass_transfer_climb(h: np.ndarray, max_iters: int, time_limit: float) -> tuple:
    """
    Directed hill climbing:
    - focus on the shift where the current C5 is attained;
    - move mass from a low‑impact coordinate to a high‑impact one.
    Returns the best (h, C5) seen before time runs out.
    """
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    no_improve = 0

    for _ in range(max_iters):
        if time.time() > time_limit:
            break

        # current worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = abs(idx_max - (n - 1))

        # compute impact weights w_i = h_{i+shift}+h_{i-shift}
        w = np.zeros(n)
        if shift > 0:
            w[: n - shift] += best_h[shift:]
            w[shift:] += best_h[: n - shift]
        else:                 # shift == 0
            w = 2.0 * best_h

        inc_mask = best_h < 1.0 - 1e-12
        dec_mask = best_h > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            break

        i = int(np.argmax(np.where(inc_mask, w, -np.inf)))
        j = int(np.argmin(np.where(dec_mask, w, np.inf)))

        # if the chosen pair cannot improve, fall back to a random pair
        if w[i] <= w[j] + 1e-12:
            i = np.random.randint(n)
            j = np.random.randint(n)
            while i == j:
                j = np.random.randint(n)

        max_eps = min(1.0 - best_h[i], best_h[j])
        if max_eps <= 0:
            no_improve += 1
            if no_improve > 500:
                break
            continue

        # try decreasing eps until we see improvement
        eps = max_eps * 0.5
        improved = False
        while eps > 1e-9:
            cand = best_h.copy()
            cand[i] += eps
            cand[j] -= eps
            c = _c5_from_h(cand)
            if c < best_c - 1e-12:
                best_h, best_c = cand, c
                no_improve = 0
                improved = True
                break
            eps *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 500:
                break

    return best_h, best_c
