def _solve_sym_h(x0, n, max_iter=400):
    m = n//2
    # x0 length m
    # define initial t from correlation.
    full_h = np.concatenate([x0, x0[::-1]])
    corr = np.correlate(full_h, 1.0 - full_h, mode='full')
    t0 = np.max(corr)
    # x vector = [x0, t]
    x_var0 = np.concatenate([x0, [t0]])
    bounds = [(0.0,1.0)]*m + [(0.0, None)]
    # constraints...
    def ineq(x):
        hfull = np.concatenate([x[:m], x[:m][::-1]])
        corr = np.correlate(hfull, 1.0 - hfull, mode='full')
        return x[-1] - corr
    def eq(x):
        return np.sum(x[:m]) - (n/4.0)
    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]
    res = optimize.minimize(lambda xx: xx[-1], x_var0, method='SLSQP',
          bounds=bounds, constraints=cons, options={'maxiter':max_iter,'ftol':1e-9,'disp':False})
    if not res.success:
        x_opt = x0
    else:
        x_opt = res.x[:m]
    x_opt = _project_to_simplex_box(x_opt, target=n/4.0)
    full_opt = np.concatenate([x_opt, x_opt[::-1]])
    return full_opt, _c5_from_h(full_opt)

import time
import numpy as np

# ---------- utilities ---------------------------------------------------------

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the hyperplane sum(v)=target
    and the box [0,1]^n.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C5 for a step function sampled on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _tent_h(n: int) -> np.ndarray:
    """Triangular (tent) shape h(x)=min(x,2-x)."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _shape_a(n: int, a: float) -> np.ndarray:
    """
    One‑parameter symmetric shape:
        h(x)=min(1, max(0, a - b*|x-1|))
    with b chosen so that the continuous integral equals 1.
    """
    x = (np.arange(n) + 0.5) * (2.0 / n)
    d = np.abs(x - 1.0)
    if a <= 1.0:
        b = a * a
        h = np.maximum(0.0, a - b * d)
    else:
        b = 2.0 * a - 1.0
        h = np.minimum(1.0, np.maximum(0.0, a - b * d))
    return _project_to_simplex_box(h, n / 2.0)


# ---------- SLSQP optimisation (full vector) ------------------------------

def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Run SLSQP from a feasible start and return refined h and its C5."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _find_best_a(n: int, a_lo: float = 0.6, a_hi: float = 1.8, steps: int = 150):
    """Coarse grid + local refinement to find a with minimal C5 for given n."""
    a_vals = np.linspace(a_lo, a_hi, steps)
    best_a, best_c5 = None, float("inf")
    for a in a_vals:
        h = _shape_a(n, a)
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5, best_a = c5, a

    delta = (a_hi - a_lo) / steps
    lo = max(a_lo, best_a - delta)
    hi = min(a_hi, best_a + delta)

    def f(a):
        return _c5_from_h(_shape_a(n, a))

    res = optimize.minimize_scalar(
        f, bounds=(lo, hi), method="bounded", options={"xatol": 1e-5}
    )
    if res.success:
        best_a = res.x
        best_c5 = f(best_a)
    return best_a, best_c5


# ---------- main entry point -------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] achieving a low C5 bound.
    Returns (list_of_h, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # ----------------------------------------------------------------------
    # Phase 1: discover a good shape parameter a on a moderate grid size.
    # ----------------------------------------------------------------------
    n_shape = 256  # comfortable size for scanning
    if time_left() > 30:
        a_opt, _ = _find_best_a(n_shape, a_lo=0.5, a_hi=1.7, steps=180)
    else:
        a_opt = 1.0  # fallback to tent

    # ----------------------------------------------------------------------
    # Phase 2: evaluate the found shape on larger resolutions and polish.
    # ----------------------------------------------------------------------
    candidate_ns = [384, 448, 256, 192, 160]
    for n in candidate_ns:
        if time_left() < 8:
            break

        # Build shape based on the best a.
        h_base = _shape_a(n, a_opt)

        # Initial evaluation.
        c5_val = _c5_from_h(h_base)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_base, c5_val, n
            if best_c5 < target_bound:
                break

        # Local refinements via short SLSQP runs.
        attempts = 6
        sigma = 0.08
        for _ in range(attempts):
            if time_left() < 6:
                break
            # perturbation respecting the box constraints before projection
            if np.random.rand() < 0.7:
                start = h_base + sigma * np.random.randn(n)
            else:
                start = np.random.rand(n)
            start = _project_to_simplex_box(start, n / 2.0)

            h_opt, c5_opt = _solve_h(start, max_iter=350)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                if best_c5 < target_bound:
                    break
            sigma *= 0.88

        if best_c5 < target_bound:
            break

    # ----------------------------------------------------------------------
    # Fallback: if nothing succeeded, use a simple tent shape.
    # ----------------------------------------------------------------------
    if best_h is None:
        n = 256
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
