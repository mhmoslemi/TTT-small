import time

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

def _gradient_from_corr(h: np.ndarray, corr: np.ndarray, eps: float = 1e-8) -> np.ndarray:
    n = h.size
    mx = corr.max()
    thr = mx - max(eps, eps * abs(mx))
    mask = corr >= thr
    idxs = np.where(mask)[0]
    accum = np.zeros(n)
    cnt = 0
    for idx in idxs:
        s = idx - (n - 1)
        if s >= 0:
            k = s
            accum[:n - k] += h[k:]
            accum[k:]   += h[:n - k]
        else:
            k = -s
            accum[k:]   += h[:n - k]
            accum[:n - k] += h[k:]
        cnt += 1
    if cnt:
        accum /= cnt
    return 1.0 - accum

                         target_bound: float = 0.3808):
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur = np.max(corr) * (2.0 / n)
    best_h, best_val = h.copy(), cur
    eta = eta0
    for it in range(max_iter):
        if time_left() < 0.5:
            break
        grad = _gradient_from_corr(h, corr, eps)
        h_new = _project_to_simplex_box(h - eta * grad, target)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        cur_new = np.max(corr_new) * (2.0 / n)
        if cur_new < cur - 1e-12:
            h, corr, cur = h_new, corr_new, cur_new
            if cur_new < best_val:
                best_h, best_val = h_new.copy(), cur_new
                if best_val < target_bound:
                    break
        eta *= decay
        if eta < 1e-6:
            break
    return best_h, best_val

def _solve_h(start_h: np.ndarray, max_iter: int = 250):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    cons = [{"type": "ineq", "fun": _ineq},
            {"type": "eq", "fun": _eq}]
    res = optimize.minimize(lambda x: x[-1], x0,
                            method="SLSQP", bounds=bounds,
                            constraints=cons,
                            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # base size
    if init_h is not None:
        n0 = init_h.size
        base = init_h.copy()
    else:
        n0 = 192
        base = _tent_h(n0)

    # deterministic refinement via descent
    h_opt, c5_opt = _subgradient_descent(base,
                                         max_iter=2500,
                                         eta0=0.12,
                                         decay=0.997,
                                         eps=1e-7,
                                         time_left=time_left,
                                         target_bound=target)
    if c5_opt < best_c5:
        best_h, best_c5, best_n = h_opt, c5_opt, n0
        if best_c5 < target:
            return best_h.tolist(), float(best_c5), int(best_n)

    # random restarts on the same grid
    sigma = 0.18
    for _ in range(12):
        if time_left() < 5:
            break
        start = base + sigma * np.random.randn(n0)
        start = _project_to_simplex_box(start, n0 / 2.0)
        h_opt, c5_opt = _subgradient_descent(start,
                                             max_iter=1500,
                                             eta0=0.1,
                                             decay=0.996,
                                             eps=1e-7,
                                             time_left=time_left,
                                             target_bound=target)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n0
            if best_c5 < target:
                break
        sigma *= 0.93

    # optional fine‑tune with SLSQP (one call)
    if best_h is not None and time_left() > 10:
        h_ref, c5_ref = _solve_h(best_h, max_iter=200)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    # try a finer grid if still above target and time permits
    if best_c5 >= target and time_left() > 30:
        n2 = 256
        base2 = _upscale_h(base, n2)
        h_opt, c5_opt = _subgradient_descent(base2,
                                             max_iter=2500,
                                             eta0=0.12,
                                             decay=0.997,
                                             eps=1e-7,
                                             time_left=time_left,
                                             target_bound=target)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n2
            if time_left() > 10:
                h_ref, c5_ref = _solve_h(best_h, max_iter=200)
                if c5_ref < best_c5:
                    best_h, best_c5 = h_ref, c5_ref

    # fallback
    if best_h is None:
        n = 192
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
