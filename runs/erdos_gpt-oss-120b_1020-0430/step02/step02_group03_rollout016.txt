def full_corr_fft(h):
    n = h.size
    L = 2*n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G.conj(), n=L)[:2*n-1]  # real values
    return conv

def smooth_obj(h, beta):
    corr = full_corr_fft(h)
    bc = beta * corr
    max_bc = bc.max()
    return (max_bc + np.log(np.exp(bc - max_bc).sum())) / beta

def perturb_h(h, rng, scale=0.2):
    n = len(h)
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # Transfer epsilon from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # cap within [-max_eps, max_eps] but already limited
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # ensure bounds due to rounding
    new_h[i] = np.clip(new_h[i], 0.0, 1.0)
    new_h[j] = np.clip(new_h[j], 0.0, 1.0)
    return new_h

def compute_c5(h):
    n = len(h)
    # Using FFT for speed
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G.conj(), n=L)[:2*n-1]
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # modify sum exactly
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-15:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

def random_feasible_vector(n, rng):
    """Random feasible vector (0<=h_i<=1, sum = n/2)."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = enforce_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # clip for safety
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h

def full_corr_fft(h):
    """Full linear correlation h * (1-h) (with flip) via FFT."""
    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G_rev = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G_rev, n=L)[:2 * n - 1]
    return conv

def compute_c5(h):
    """Exact discrete C₅ for step function h."""
    n = h.size
    conv = full_corr_fft(h)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

# ---------- surrogate optimisation ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = full_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny drift)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def hill_climb(start_h, rng, max_iters=1500, start_step=0.5):
    """Simple deterministic hill‑climb with decreasing step size."""
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
