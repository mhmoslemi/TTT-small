def max_corr(h):
    n = len(h)
    N = 1 << ((2*n-1).bit_length())
    fft_h = np.fft.rfft(h, N)
    fft_g = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fft_h * np.conj(fft_g), N)
    # Only first 2*n-1 values correspond to linear correlation
    max_val = np.max(corr[:2*n-1])
    return max_val

def hill_climb_binary(initial_h, max_no_improve=5000):
    h = initial_h.copy()
    n = len(h)
    cur_max = max_corr(h)
    no_improve = 0
    # maintain indices lists
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    while no_improve < max_no_improve:
        # pick random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0, 1
        # compute new max
        new_max = max_corr(h)
        if new_max <= cur_max:
            cur_max = new_max
            # update ones/zeros
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            no_improve = 0
        else:
            # revert
            h[i], h[j] = 1, 0
            no_improve += 1
    return h, cur_max
```

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Core helper functions
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and force sum(h)=n/2 by a simple water‑filling loop."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / free.sum()
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction (rare)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _max_corr_fft(h: np.ndarray) -> float:
    """Maximum of linear correlation of h with (1‑h) using FFT (fast)."""
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    hf = np.fft.rfft(h, N)
    gf = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(hf * np.conj(gf), N)
    return float(np.max(corr[: 2 * n - 1]))


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible step function h."""
    n = h.size
    return _max_corr_fft(h) * (2.0 / n)


# --------------------------------------------------------------
# SLSQP refinement (continuous)
# --------------------------------------------------------------
def _ineq_constraint(x: np.ndarray):
    """t - correlation >= 0 for each shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300):
    """SLSQP optimisation starting from an (infeasible) vector."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# --------------------------------------------------------------
# Greedy hill‑climbing for binary vectors
# --------------------------------------------------------------
def _hill_climb_binary(h_bin: np.ndarray, max_no_improve: int = 2000):
    """Swap a 1 and a 0 while it does not increase the max correlation."""
    n = h_bin.size
    cur = _max_corr_fft(h_bin)
    no_improve = 0
    ones = np.flatnonzero(h_bin == 1)
    zeros = np.flatnonzero(h_bin == 0)

    while no_improve < max_no_improve:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_bin[i], h_bin[j] = 0, 1  # perform swap
        new = _max_corr_fft(h_bin)
        if new <= cur:
            cur = new
            ones = np.flatnonzero(h_bin == 1)
            zeros = np.flatnonzero(h_bin == 0)
            no_improve = 0
        else:
            h_bin[i], h_bin[j] = 1, 0  # revert
            no_improve += 1
    return h_bin, cur
