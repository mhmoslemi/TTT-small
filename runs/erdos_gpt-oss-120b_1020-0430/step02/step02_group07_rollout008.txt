def compute_c5_fft(h):
    n = h.size
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1-h, N)
    conv = np.fft.irfft(H*G, N)[:L]
    return conv.max() * (2.0/n)

def project_onto_feasible(z, target):
    # Clip to [0,1]  
    x = np.clip(z, 0.0, 1.0)
    # Compute sum
    s = x.sum()
    if np.isclose(s, target):
        return x
    # Need to adjust.
    # We'll use algorithm where we treat free variables and adjust to meet sum while staying within bounds.
    # This is like solving: x_i = clip(z_i + λ, 0, 1) where λ is Lagrange multiplier for sum constraint.
    # Because projection onto box with sum constraint is equivalent to adding constant λ and clipping.
    # Indeed, solution of min ||x - z||^2 s.t. sum x = target, 0 <= x <= 1 can be expressed via lambda.
    # So we find λ such that sum(clip(z_i + λ, 0,1)) = target. This is a monotonic function in λ; we can binary search.
    lo = -1.0
    hi = 1.0
    # find adequate bounds: because if we shift too negative, all clipped to 0 => sum=0. If too positive, all to1 => sum=n.
    # We'll compute until max possible sum >= target.
    while np.clip(z + hi, 0, 1).sum() < target:
        hi *= 2
    while np.clip(z + lo, 0, 1).sum() > target:
        lo *= 2
    for _ in range(50):
        mid = (lo + hi) / 2
        sm = np.clip(z + mid, 0, 1).sum()
        if sm > target:
            hi = mid
        else:
            lo = mid
    return np.clip(z + (lo+hi)/2, 0, 1)

def reduce_max_shift(h, iters=200, step=0.1):
    for _ in range(iters):
        # compute correlation vector
        corr = np.correlate(h, 1-h, mode="full")
        k_idx = np.argmax(corr)  # int
        k = k_idx - (h.size - 1)
        # compute shifted arrays:
        if k >= 0:
            h_shift = np.concatenate([np.zeros(k), h[:len(h)-k]])
        else:
            h_shift = np.concatenate([h[-k:], np.zeros(-k)])
        # compute 1-h_shift
        one_minus_shift = 1 - h_shift
        # Gradient direction to reduce correlation: we want to increase h where one_minus_shift is small, decrease where large.
        # Choose i where one_minus_shift minimal, j where maximal.
        i = np.argmin(one_minus_shift)  # index with smallest 1 - h_shift => largest h_shift
        j = np.argmax(one_minus_shift)  # index with largest 1 - h_shift => smallest h_shift
        # Transfer epsilon from j to i (increase h_i, decrease h_j)
        # epsilon limited by h_j and 1 - h_i
        max_eps = min(h[j], 1 - h[i])
        eps = step * max_eps * np.random.random()
        if eps > 1e-12:
            h[i] += eps
            h[j] -= eps
        else:
            break
    return h

def full_corr(h):
    n = h.size
    L = 2 * n - 1
    # next power of two
    N = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    conv = np.fft.irfft(H * G, N)[:L]
    return conv

# Global placeholders for constraints used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # compute full linear correlation of x with (1-x)
    n = x.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    H = np.fft.rfft(x, N)
    G = np.fft.rfft(1.0 - x, N)
    corr = np.fft.irfft(H * G, N)[:L]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    n = h.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(H * G, N)[:L]
    return conv.max() * (2.0 / n)

def project_onto_feasible(z, target):
    lo, hi = -1.0, 1.0
    # expand bounds until sum at extremes encloses target
    while np.clip(z + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    while np.clip(z + lo, 0.0, 1.0).sum() > target:
        lo *= 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid

CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    n = x.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    H = np.fft.rfft(x, N)
    G = np.fft.rfft(1.0 - x, N)
    corr = np.fft.irfft(H * G, N)[:L]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5_fft(h):
    """C5 using FFT (fast, same scaling as the reference implementation)."""
    n = h.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, N)

        if cand_c < cur_c:
            cur, cur_c = cand, cand_c
    return cur, cur_c


def simulated_annealing_swap(h, rng, max_iters=60000, init_temp=0.03, decay=0.9995):
    """Stochastic annealing with swaps."""
    cur = h.copy()
    cur_c = compute_c5_fft(cur)
    best, best_c = cur.copy(), cur_c
    temp = init_temp
    for _ in range(max_iters):
        cand = random_swap(cur, rng)
        cand_c = compute_c5_fft(cand)
        if cand_c < best_c:
            best, best_c = cand, cand_c
        if cand_c < cur_c or rng.random() < np.exp((cur_c - cand_c) / temp):
            cur, cur_c = cand, cand_c
        temp *= decay
    return best, best_c


def refine_candidate(h0, rng):
    """Full refinement pipeline for a single start vector."""
    n = h0.size
    target = n / 2.0
    h = project_onto_feasible(h0, target)
    h, _ = hill_climb_swap(h, rng, max_iters=1500)
    h, _ = simulated_annealing_swap(h, rng, max_iters=80000, init_temp=0.03, decay=0.9995)
    h = progressive_optimize_continuous(h)
    h = project_onto_feasible(h, target)
    return h, compute_c5_exact(h)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Initialise base vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_feasible(h0, n / 2.0)
    else:
        n = 360
        h0 = random_binary_vector(n, rng)

    best_h, best_c = h0, compute_c5_exact(h0)

    # ----------------------------------------------------------------------
    # Try Legendre construction when available
    # ----------------------------------------------------------------------
    leg = legendre_pattern(n)
    if leg is not None:
        leg = project_onto_feasible(leg, n / 2.0)
        lc = compute_c5_exact(leg)
        if lc < best_c:
            best_h, best_c = leg, lc

    # ----------------------------------------------------------------------
    # Pool of starting candidates (few random plus the current best)
    # ----------------------------------------------------------------------
    candidates = [best_h.copy()]
    for _ in range(6):
        candidates.append(random_binary_vector(n, rng))

    # ----------------------------------------------------------------------
    # Refine each candidate while we have the majority of the budget
    # ----------------------------------------------------------------------
    for h_start in candidates:
        if time.time() - start > budget_s * 0.65:
            break
        h_ref, c_ref = refine_candidate(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----------------------------------------------------------------------
    # Optional up‑sampling phase (if time left)
    # ----------------------------------------------------------------------
    if time.time() - start < budget_s * 0.90 and n * 2 <= 800:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        h_up_ref, c_up_ref = refine_candidate(h_up, rng)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = best_h.size

    # ----------------------------------------------------------------------
    # Final exact evaluation and return
    # ----------------------------------------------------------------------
    best_h = project_onto_feasible(best_h, best_h.size / 2.0)
    final_c = compute_c5_exact(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
