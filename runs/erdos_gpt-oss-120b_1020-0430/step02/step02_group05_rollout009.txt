def slsqp_refine(start_h, max_iter=200):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n/2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n/2.0

    cons = [{'type':'ineq', 'fun':ineq},
            {'type':'eq', 'fun':eq}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter':max_iter, 'ftol':1e-9, 'disp':False})
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n/2.0)
    return h_opt, _c5_from_h(h_opt)

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    if not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)

def _piecewise_h(params: np.ndarray, K: int, n: int) -> np.ndarray:
    """Construct a piecewise-constant h from params."""
    h_vals = np.clip(params[:K], 0.0, 1.0)
    raw = params[K:]
    raw = np.clip(raw, -10.0, 10.0)
    exp_raw = np.exp(raw)
    lengths = exp_raw / exp_raw.sum() * 2.0
    cum = np.concatenate(([0.0], np.cumsum(lengths)))
    pos = (np.arange(n) + 0.5) * (2.0 / n)
    idx = np.searchsorted(cum, pos, side="right") - 1
    return h_vals[idx]

def _init_params_from_h(init_h: np.ndarray, K: int) -> np.ndarray:
    """Fit K equal-length segments to init_h and return a param vector."""
    n = init_h.size
    pos = (np.arange(n) + 0.5) * (2.0 / n)
    edges = np.linspace(0.0, 2.0, K + 1)
    h_seg = np.empty(K)
    for i in range(K):
        mask = (pos >= edges[i]) & (pos < edges[i + 1])
        if mask.any():
            h_seg[i] = np.mean(init_h[mask])
        else:
            h_seg[i] = 0.5
    raw = np.zeros(K - 1)
    return np.concatenate([h_seg, raw])

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function for the Erdős C5 problem.
    Returns (list_of_h, C5_value, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # try to load a parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ---- piecewise-constant search (small dimensional optimisation) ----
    n_coarse = 400
    target = 0.3808
    ks = (5, 6, 7, 8, 9)
    for K in ks:
        if time_left() < 15:
            break
        dim = 2 * K - 1
        # base deterministic start (all 0.5 heights, equal lengths)
        base = np.concatenate([np.full(K, 0.5), np.zeros(K - 1)])

        # add a start derived from the parent if available
        starts = [base]
        if init_h is not None:
            starts.append(_init_params_from_h(_upscale_h(init_h, n_coarse), K))

        # random restarts
        for _ in range(20):
            if time_left() < 5:
                break
            # random perturbation of the deterministic base
            pert_h = np.clip(base[:K] + 0.2 * np.random.randn(K), 0.0, 1.0)
            pert_r = base[K:] + 0.5 * np.random.randn(K - 1)
            starts.append(np.concatenate([pert_h, pert_r]))

        for x0 in starts:
            if time_left() < 5:
                break

            def obj(p):
                h = _piecewise_h(p, K, n_coarse)
                return _c5_from_h(h)

            def eq_con(p):
                h_vals = np.clip(p[:K], 0.0, 1.0)
                raw = np.clip(p[K:], -10.0, 10.0)
                exp_raw = np.exp(raw)
                lengths = exp_raw / exp_raw.sum() * 2.0
                return np.sum(h_vals * lengths) - 1.0

            cons = [{"type": "eq", "fun": eq_con}]
            bounds = [(0.0, 1.0)] * K + [(-10.0, 10.0)] * (K - 1)

            res = opt.minimize(
                obj,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 120, "ftol": 1e-8, "disp": False},
            )
            if not res.success:
                continue
            h_opt = _piecewise_h(res.x, K, n_coarse)
            c5_val = _c5_from_h(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                best_n = n_coarse
                if best_c5 < target:
                    break
        if best_c5 < target:
            break

    # ---- evaluate the parent construction directly (coarse) ----
    if init_h is not None:
        if init_h.size != n_coarse:
            init_coarse = _upscale_h(init_h, n_coarse)
        else:
            init_coarse = init_h.copy()
        c5_init = _c5_from_h(init_coarse)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = init_coarse
            best_n = n_coarse

    # ---- final refinement on a finer grid ----
    if best_h is not None and time_left() > 15:
        n_fine = min(800, max(400, best_n * 2))
        # upscale piecewise (or parent) to fine grid via linear interpolation
        x_cur = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
        h_fine = np.interp(x_fine, x_cur, best_h)
        h_fine = np.clip(h_fine, 0.0, 1.0)
        h_fine = _project_to_simplex_box(h_fine, n_fine / 2.0)

        h_ref, c5_ref = _solve_h(h_fine, max_iter=300)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = n_fine

        # optionally also refine the raw parent construction if still time
        if (
            init_h is not None
            and time_left() > 10
            and init_h.size != best_n
        ):
            parent_fine = _upscale_h(init_h, best_n)
            parent_fine = _project_to_simplex_box(parent_fine, best_n / 2.0)
            h_par_ref, c5_par_ref = _solve_h(parent_fine, max_iter=250)
            if c5_par_ref < best_c5:
                best_c5 = c5_par_ref
                best_h = h_par_ref
                best_n = best_n  # unchanged

    # ---- fallback ----
    if best_h is None:
        best_n = 256
        best_h = _tent_h(best_n)
        best_c5 = _c5_from_h(best_h)

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_simplex_box(best_h, best_n / 2.0)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
