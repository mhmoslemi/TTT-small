def smooth_max_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    C = (2.0 / n) * corr
    bc = beta * C
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w_sum = w.sum()
    w = w / w_sum
    # compute gradient as per earlier g_raw
    g_raw = np.ones(n)
    idx_offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - idx_offset
        if k >= 0:
            # k shift positive
            g_raw[:n - k] -= wk * h[k:]
            g_raw[k:] -= wk * h[:n - k]
        else:
            kk = -k
            # shift negative
            g_raw[kk:] -= wk * h[:n - kk]
            g_raw[:n - kk] -= wk * h[kk:]
    grad = (2.0 / n) * g_raw
    obj = (max_bc + np.log(w_sum)) / beta   # This is smooth max value
    return obj, grad

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo = -1e9
    hi = 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Generate a random feasible h (0≤h≤1, sum = n/2)."""
    v = rng.random(n)
    v = project_onto_box_sum(v, n / 2.0)
    return v


def perturb_h(h, rng, scale=0.2):
    """Swap a random amount between two coordinates while keeping sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    """Simple hill‑climb using pairwise swaps."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # gradually shrink step
        if (it + 1) % max(1, max_iters // 6) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth‑max gradient ----------
def smooth_max_obj_and_grad(h, beta):
    """Return smooth‑max value and its gradient w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    C = (2.0 / n) * corr                     # scaled overlaps
    bc = beta * C
    m = np.max(bc)
    w = np.exp(bc - m)
    w_sum = w.sum()
    w = w / w_sum                            # soft‑max weights

    g_raw = np.ones(n)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            nk = n - k
            g_raw[:nk] -= wk * h[k:]
            g_raw[k:]  -= wk * h[:nk]
        else:
            kk = -k
            nk = n - kk
            g_raw[kk:] -= wk * h[:nk]
            g_raw[:nk] -= wk * h[kk:]
    grad = (2.0 / n) * g_raw
    obj = (m + np.log(w_sum)) / beta
    return obj, grad


def grad_desc_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        # step size inversely related to beta (roughly)

            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
            h = new_h

    return best_h, best_c
