def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def pgd_optimize(h0, target_sum, beta_schedule, max_iters, rng):
    """Projected gradient descent using smooth surrogate."""
    h = h0.copy()
    n = h.size
    # momentum vector
    m = np.zeros_like(h)
    # Momentum coefficient
    mom = 0.9
    # Step size rotation: initial step
    step = 0.5
    # Determine how many iters per beta
    if len(beta_schedule) == 0:
        beta_schedule = [200]  # fallback
    iters_per_beta = max_iters // len(beta_schedule)
    total_iters = 0
    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            # compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = corr.max()
            # compute softmax weights
            w = np.exp(beta * (corr - max_corr))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # compute convolution contributions
            conv1 = np.convolve(w, h, mode='full')
            conv2 = np.convolve(w[::-1], h, mode='full')
            start = n - 1
            part1 = conv1[start:start + n]
            part2 = conv2[start:start + n]
            G = part1 + part2
            grad = 1.0 - G
            # momentum update
            m = mom * m + grad
            # gradient step
            y = h - step * m
            # project onto constraints
            h = project_onto_box_hyperplane(y, target_sum)
            # decay step
            step *= 0.995
            total_iters += 1
    return h

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_onto_box_hyperplane(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto the set {x : sum(x)=target, 0<=x<=1}."""
    lower, upper = -10.0, 10.0  # wide enough to bracket the solution
    for _ in range(60):
        lam = (lower + upper) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lower = lam
        else:
            upper = lam
    lam = (lower + upper) * 0.5
    return np.clip(y - lam, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with sum = n/2."""
    y = rng.random(n)
    return project_onto_box_hyperplane(y, n / 2.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_local(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 500, start_step: float = 0.2):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 5)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def pgd_optimize(h0: np.ndarray, target_sum: float,
                 beta_schedule, max_iters: int,
                 rng: np.random.Generator) -> np.ndarray:
    """Projected gradient descent on the smooth log‑sum‑exp surrogate."""
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)               # momentum
    mom = 0.9
    step = 0.5
    # distribute iterations over the β‑schedule
    if len(beta_schedule) == 0:
        beta_schedule = [200.0]
    per = max(1, max_iters // len(beta_schedule))
    iters_done = 0
    for beta in beta_schedule:
        for _ in range(per):
            # correlation and soft‑max weights
            corr = np.correlate(h, 1.0 - h, mode="full")
            mxc = corr.max()
            w = np.exp(beta * (corr - mxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # convolution terms giving Σ w_s h_{j±s}
            conv1 = np.convolve(w, h, mode="full")
            conv2 = np.convolve(w[::-1], h, mode="full")
            start = n - 1
            part1 = conv1[start:start + n]
            part2 = conv2[start:start + n]
            G = part1 + part2
            grad = 1.0 - G                     # ∇ f
            # momentum & step
            m = mom * m + grad
            y = h - step * m
            h = project_onto_box_hyperplane(y, target_sum)
            step *= 0.995
            iters_done += 1
            if iters_done >= max_iters:
                break
        if iters_done >= max_iters:
            break
    return h


def upscale(h: np.ndarray, factor: int = 2) -> np.ndarray:
    """Linear interpolation to a finer grid."""
    n = h.size
    n2 = n * factor
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, n2)
    h2 = np.interp(x_new, x_old, h)
    return h2

    # ------------------------------------------------------------------

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
