```python
import time, numpy as np, scipy.optimize as opt

def project_to_feasible(h, target, tol=1e-9, max_iter=200):
    h = np.clip(h,0.,1.)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h>0)[0]
        else:
            free = np.where(h<1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h,0.,1.)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual across any non-bound vars
        for i in range(len(h)):
            if residual > 0 and h[i] < 1-1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h
```

def _solve_h(initial_h, max_iter=500):
    n = len(initial_h)
    h0 = project_to_feasible(initial_h, n/2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode='full')
    def eq(x):
        return np.sum(x[:n]) - n/2.0
    def obj(x):
        return x[-1]
    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]
    opts = {'maxiter':max_iter,'ftol':1e-9,'disp':False}
    res = opt.minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.x is not None else h0
    h_opt = project_to_feasible(h_opt, n/2.0)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

import time
import numpy as np
from scipy import optimize as opt

# ----- feasibility projection -------------------------------------------------
def _project_to_feasible(h: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip h to [0,1] and adjust sum(h) to target via water-filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(h.size):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----- exact C5 evaluation ---------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----- fast C5 using FFT (used inside GA) ------------------------------------
def _max_corr_fft(h: np.ndarray) -> float:
    n = h.shape[0]
    g = 1.0 - h
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(g, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    return np.max(corr[: 2 * n - 1])


def _c5_fast(h: np.ndarray) -> float:
    return _max_corr_fft(h) * (2.0 / h.size)

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size

    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)

def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400):
    """Run a short SLSQP refinement starting from initial_h."""
    n = int(initial_h.size)
    h0 = _project_to_feasible(np.asarray(initial_h, dtype=float), n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.x is not None else h0
    h_opt = _project_to_feasible(h_opt, n / 2.0)

def _tournament(pop: np.ndarray, fit: np.ndarray, k: int, rng) -> np.ndarray:
    idx = rng.choice(pop.shape[0], size=k, replace=False)
