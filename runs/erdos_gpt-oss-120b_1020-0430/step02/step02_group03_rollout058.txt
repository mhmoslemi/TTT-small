import time
import numpy as np
from scipy.optimize import minimize

# globals used by the smooth surrogate and the equality constraint
CURRENT_BETA = None
CURRENT_TARGET = None


def sum_eq_constraint_global(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def smooth_obj_global(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    # stable log‑sum‑exp
    exps = np.exp(CURRENT_BETA * (corr - m))
    return m + np.log(np.sum(exps)) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σ h = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def binary_from_continuous(v):
    """Return a binary vector with exactly half the entries equal to 1,
    choosing the largest values from v."""
    n = v.size
    k = n // 2
    idx = np.argpartition(-v, k)[:k]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

        c = compute_c5(h)
        if c < best_c:
            best_c, best = c, h.copy()
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # -------------------------------------------------------------
    # 1. Initialise a binary starting point on a modest grid.
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        # aim for ~360 points (still cheap) – repeat if possible
        desired = max(180, min(360, n0 * 2))
        factor = max(1, desired // n0)
        h0 = np.repeat(h0, factor)
        n = h0.size
        h_bin = binary_from_continuous(h0)
    else:
        n = 2 * rng.integers(90, 181)          # even size between 180 and 360
        h_bin = np.zeros(n, dtype=float)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1.0

    # index lists for fast swaps
    ones_idx = list(np.where(h_bin == 1)[0])
    zeros_idx = list(np.where(h_bin == 0)[0])
    cur_c = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c = cur_c

    # -------------------------------------------------------------
    # 2. Simulated‑annealing on the binary space.
    # -------------------------------------------------------------
    phase1_end = start_time + total_budget * 0.55
    T_start, T_end = 0.01, 0.0001
    stagnation = 0
    while time.time() < phase1_end:
        elapsed = time.time() - start_time
        frac = elapsed / (total_budget * 0.55)
        T = T_start * (T_end / T_start) ** frac

        i_pos = rng.integers(len(ones_idx))
        j_pos = rng.integers(len(zeros_idx))
        i = ones_idx[i_pos]
        j = zeros_idx[j_pos]

        # perform the swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c = compute_c5(h_bin)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True

        if accept:
            # keep swap and update the index lists
            ones_idx[i_pos] = j
            zeros_idx[j_pos] = i
            cur_c = new_c
            stagnation = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_bin.copy()
        else:
            # revert swap
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            stagnation += 1

        # occasional random restart to avoid dead‑ends
        if stagnation > 50000:
            h_bin[:] = 0.0
            idx = rng.choice(n, n // 2, replace=False)
            h_bin[idx] = 1.0
            ones_idx = list(idx)
            zeros_idx = list(np.setdiff1d(np.arange(n), idx))
            cur_c = compute_c5(h_bin)
            stagnation = 0

    # -------------------------------------------------------------
    # 3. Refinement with a smooth surrogate (SLSQP ladder).
    # -------------------------------------------------------------
    global CURRENT_BETA, CURRENT_TARGET
    CURRENT_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint_global}
    beta_schedule = [2, 5, 15, 40, 100, 250, 500]
    x = best_h.astype(float).copy()

    for beta in beta_schedule:
        if time.time() >= start_time + total_budget * 0.95:
            break
        CURRENT_BETA = beta
        try:
            res = minimize(
                smooth_obj_global,
                x,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 500, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        x = np.clip(res.x, 0.0, 1.0)
        # enforce exact sum (tiny drift)
        diff = CURRENT_TARGET - x.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = CURRENT_TARGET - x.sum()
        c_now = compute_c5(x)
        if c_now < best_c:
            best_c = c_now
            best_h = x.copy()

    # -------------------------------------------------------------
    # 4. Optional up‑sampling and another SLSQP pass (if time permits).
    # -------------------------------------------------------------
    if time.time() < start_time + total_budget * 0.98 and n * 2 <= 500:
        n_up = n * 2
        up_h = np.repeat(best_h, 2)
        CURRENT_TARGET = n_up / 2.0
        bounds_up = [(0.0, 1.0)] * n_up
        cons_up = {"type": "eq", "fun": sum_eq_constraint_global}
        x_up = up_h.astype(float).copy()
        for beta in beta_schedule:
            if time.time() >= start_time + total_budget:
                break
            CURRENT_BETA = beta
            try:
                res = minimize(
                    smooth_obj_global,
                    x_up,
                    method="SLSQP",
                    bounds=bounds_up,
                    constraints=cons_up,
                    options={"maxiter": 500, "ftol": 1e-12, "disp": False},
                )
            except Exception:
                continue
            x_up = np.clip(res.x, 0.0, 1.0)
            diff = CURRENT_TARGET - x_up.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    if abs(diff) < 1e-12:
                        break
                    i = rng.integers(0, n_up)
                    add = min(diff, 1.0 - x_up[i]) if diff > 0 else max(diff, -x_up[i])
                    x_up[i] += add
                    diff = CURRENT_TARGET - x_up.sum()
            c_up = compute_c5(x_up)
            if c_up < best_c:
                best_c = c_up
                best_h = x_up.copy()
                n = n_up

    # -------------------------------------------------------------
    # 5. Final tiny continuous hill‑climb (fast improvement).
    # -------------------------------------------------------------
    if time.time() < start_time + total_budget * 0.99:
        h_fc, c_fc = hill_climb_float(best_h, rng, steps=3000, scale=0.05)
        if c_fc < best_c:
            best_c, best_h = c_fc, h_fc

    # -------------------------------------------------------------
    # 6. Clean‑up and return.
    # -------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
