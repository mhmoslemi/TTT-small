import time
import numpy as np
from scipy.optimize import minimize

# ---------- Core utilities ----------
def compute_c5(h):
    """Exact C5 for a step function h sampled on [0,2] with dx=2/n."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ---------- Equality constraint for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log‑sum‑exp (used as surrogate for SLSQP)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (simple random redistribution)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Binary helpers ----------
def random_binary_vector(n, rng):
    """Return a 0/1 vector of length n with exactly n/2 ones."""
    bits = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    bits[ones] = 1.0
    return bits


def perturb_binary(bits, rng, swap_fraction=0.1):
    """Swap a fraction of ones with zeros (preserves count)."""
    n = bits.size
    max_swaps = max(1, int(swap_fraction * n // 2))
    ones = np.where(bits == 1.0)[0]
    zeros = np.where(bits == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return bits
    swaps = min(max_swaps, len(ones), len(zeros))
    swap_ones = rng.choice(ones, size=swaps, replace=False)
    swap_zeros = rng.choice(zeros, size=swaps, replace=False)
    bits[swap_ones] = 0.0
    bits[swap_zeros] = 1.0
    return bits


def binary_hill_climb(bits, rng, max_iters=500, trials_per_iter=30):
    """
    Greedy hill‑climb on binary vectors.
    At each iteration examine a few random swaps; keep the best improving one.
    """
    cur = bits.copy()
    cur_c = compute_c5(cur)
    n = cur.size
    for _ in range(max_iters):
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_swap = None
        best_c = cur_c
        for _ in range(trials_per_iter):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = cur.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            cur[i] = 0.0
            cur[j] = 1.0
            cur_c = best_c
        else:
            break
    return cur, cur_c


def gradient_descent_refine(h, rng, step=0.02, max_iter=30):
    """Gradient descent on the worst shift (sub‑gradient of max)."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # ∂/∂h_i of term h_i(1-h_{i+k})
            g[k:]   -= h[:upper]           # ∂/∂h_{i+k}
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def enforce_sum(arr, target, rng, max_iter=5000):
    """Adjust arr in‑place so that its sum equals target (respecting bounds)."""
    diff = target - arr.sum()
    n = arr.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - arr[i]) if diff > 0 else max(diff, -arr[i])
        arr[i] += add
        diff = target - arr.sum()
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Use any supplied parent solution as a baseline
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, n0 / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    candidate_ns = [48, 60, 72, 84, 96, 108, 120]
    if best_n is not None and best_n not in candidate_ns:
        candidate_ns = [best_n] + candidate_ns

    # Main loop over discretisations
    for n in candidate_ns:
        if time.time() - start > budget_s * 0.94:
            break
        attempts = 0
        max_attempts = 12
        while attempts < max_attempts and time.time() - start < budget_s * 0.94:
            attempts += 1

            # ---- initialise a random feasible binary vector ----
            bits = random_binary_vector(n, rng)

            # ---- aggressive hill‑climb ----
            bits, c = binary_hill_climb(bits, rng, max_iters=600, trials_per_iter=40)

            # ---- perturb + re‑climb cycles ----
            for _ in range(3):
                bits = perturb_binary(bits, rng, swap_fraction=0.15)
                bits, c_tmp = binary_hill_climb(bits, rng, max_iters=300, trials_per_iter=20)
                if c_tmp < c:
                    c = c_tmp

            # update global best if binary solution wins
            if c < best_c:
                best_h = bits.astype(float).copy()
                best_c = c
                best_n = n

            # ---- gradient refinement (fractional) ----
            h_frac = best_h.copy()
            h_frac = gradient_descent_refine(h_frac, rng, step=0.02, max_iter=30)
            c_frac = compute_c5(h_frac)
            if c_frac < best_c:
                best_h, best_c, best_n = h_frac, c_frac, n

            # ---- SLSQP refinement with increasing β ----
            if time.time() - start < budget_s * 0.96:
                h_start = best_h.copy()
                for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                    if time.time() - start > budget_s * 0.97:
                        break
                    try:
                        h_ref = slsqp_optimize(h_start, beta, rng, max_iter=500)
                    except Exception:
                        continue
                    c_ref = compute_c5(h_ref)
                    if c_ref < best_c:
                        best_h, best_c, best_n = h_ref.copy(), c_ref, n
                        h_start = h_ref.copy()

    # Final polishing if time permits
    if best_h is not None and time.time() - start < budget_s * 0.99:
        # One more round of aggressive gradient + SLSQP
        best_h = gradient_descent_refine(best_h, rng, step=0.015, max_iter=40)
        for beta in (200.0, 500.0, 1000.0, 2000.0):
            if time.time() - start > budget_s:
                break
            try:
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=600)
            except Exception:
                continue

    # Fallback if nothing was found
    if best_h is None:
        best_n = 44
        best_h = np.full(best_n, 0.5)

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_n / 2.0, rng)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
