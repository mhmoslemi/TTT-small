def pgd_optimize(init_h, n_iter, step0, step_end, beta0, beta_end):
    h = init_h.copy()
    n = len(h)
    for t in range(n_iter):
        # maybe compute progress ratio
        ratio = t / n_iter
        step = step0 * ((step_end / step0) ** ratio)
        beta = beta0 * ((beta_end / beta0) ** ratio)
        # compute corr
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # compute weights
        exp_vals = np.exp(beta * (corr - max_corr))
        w = exp_vals / exp_vals.sum()
        # compute phi via correlation
        phi1 = np.correlate(w, h, mode='full')[n-1:n-1+n]
        phi2 = np.correlate(w[::-1], h, mode='full')[n-1:n-1+n]
        grad = 1.0 - (phi1 + phi2)
        h = h - step * grad
        h = _project_to_feasible(h, n)
    # final corr and C5
    max_corr = np.max(np.correlate(h, 1 - h, mode='full'))
    c5 = max_corr * (2.0 / n)
    return h, c5

import time
import numpy as np
from scipy import optimize

# -------------------- Helper utilities -------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """Inequality: t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Equality: sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Objective is the auxiliary variable t (to be minimized)."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Run SLSQP from `initial_h`. Return (h_opt, C5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02) -> np.ndarray:
    """Simple stochastic hill‑climbing around h to squeeze a few extra digits."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


def _pgd_optimize(
    h0: np.ndarray,
    n_iter: int,
    step0: float,
    step_end: float,
    beta0: float,
    beta_end: float,
) -> tuple:
    """Projected gradient descent on a smoothed max‑overlap objective."""
    h = h0.copy()
    n = h.size
    for t in range(n_iter):
        if n_iter > 1:
            ratio = t / (n_iter - 1)
        else:
            ratio = 0.0
        step = step0 * ((step_end / step0) ** ratio)
        beta = beta0 * ((beta_end / beta0) ** ratio)

        # correlation and softmax weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        exp_vals = np.exp(beta * (corr - max_corr))
        w = exp_vals / exp_vals.sum()  # length 2n‑1, indexed by shift -(n‑1)…+(n‑1)

        # phi = Σ w_k (h_{j+k}+h_{j‑k})
        phi1 = np.correlate(w, h, mode="full")[n - 1 : n - 1 + n]
        phi2 = np.correlate(w[::-1], h, mode="full")[n - 1 : n - 1 + n]
        grad = 1.0 - (phi1 + phi2)

        h = h - step * grad
        h = _project_to_feasible(h, n)
    c5 = _c5_from_h(h)
    return h, c5


# -------------------- Main optimisation driver -------------------- #
