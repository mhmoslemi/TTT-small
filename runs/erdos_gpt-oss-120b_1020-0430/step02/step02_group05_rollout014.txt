import time
import numpy as np
from scipy import optimize

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x in [0,1]^n : sum(x)=target}."""
    lo, hi = np.min(v) - 1.0, np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _make_full_from_half(half: np.ndarray) -> np.ndarray:
    """Reflect half to obtain a symmetric full vector of length 2*len(half)."""
    return np.concatenate([half, half[::-1]])


def _c5_from_full(full: np.ndarray) -> float:
    """Compute C5 for a full symmetric step function."""
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    return np.max(corr) * (2.0 / n)


def _c5_from_half(half: np.ndarray) -> float:
    return _c5_from_full(_make_full_from_half(half))


def _solve_half(start_half: np.ndarray, max_iter: int = 300):
    """SLSQP refinement for a symmetric half-vector."""
    m = start_half.size
    n = 2 * m
    target = n / 4.0  # required sum of half
    half0 = _project_to_simplex_box(start_half, target)

    full0 = _make_full_from_half(half0)
    t0 = np.max(np.correlate(full0, 1.0 - full0, mode="full"))
    x0 = np.concatenate([half0, [t0]])

    bounds = [(0.0, 1.0)] * m + [(0.0, None)]

    def ineq(x):
        half = x[:m]
        t = x[-1]
        corr = np.correlate(_make_full_from_half(half), 1.0 - _make_full_from_half(half), mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:m]) - target

    def obj(x):
        return x[-1]

    res = optimize.minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}],
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    half_opt = res.x[:m] if res.success else half0
    half_opt = _project_to_simplex_box(half_opt, target)
    return half_opt, _c5_from_half(half_opt)


def _hill_climb_full(full: np.ndarray, init_c5: float, time_left):
    """Simple pairwise mass-transfer hill climbing, preserving sum and bounds."""
    best = full.copy()
    best_c5 = init_c5
    n = best.size
    # parameters
    max_step = 0.02
    min_time = 1.0
    while time_left() > min_time:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        hi, hj = best[i], best[j]
        if hi <= hj:
            continue  # transfer from higher to lower
        step = min(hi, 1.0 - hj, max_step)
        if step <= 1e-12:
            continue
        delta = np.random.rand() * step
        trial = best.copy()
        trial[i] -= delta
        trial[j] += delta
        c5_new = _c5_from_full(trial)
        if c5_new + 1e-12 < best_c5:
            best = trial
            best_c5 = c5_new
    return best, best_c5


def _interp_half(src: np.ndarray, m_target: int) -> np.ndarray:
    """Interpolate a half-vector to a different length."""
    m_src = src.size
    x_src = (np.arange(m_src) + 0.5) / m_src
    x_tgt = (np.arange(m_target) + 0.5) / m_target
    return np.interp(x_tgt, x_src, src)
