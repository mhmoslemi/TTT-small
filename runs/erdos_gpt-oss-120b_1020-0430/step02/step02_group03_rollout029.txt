def project_simplex(v, target):
    # v shape (n,)
    # return x with sum=target and 0<=x<=1
    lo = -np.max(v)  # ensures all v+lo <=0
    hi = 1 - np.min(v)  # ensures all v+hi >=1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if np.abs(s - target) < 1e-9:
            return x
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def hill_climb_half(v_start, rng, max_iters, start_step):

   v_current = v_start.copy()

   best_v = v_current.copy()

   best_c = compute_c5_half(v_current)

   step = start_step

   decay_interval = max(1, max_iters // 10)

   for it in range(max_iters):

       new_v = perturb_h_half(v_current, rng, scale=step)

       c = compute_c5_half(new_v)

       if c < best_c:

           best_c = c; best_v = new_v.copy(); v_current = new_v

       if (it+1) % decay_interval == 0:

           step = max(step*0.7, 1e-4)

   return best_v, best_c

def perturb_h_half(v, rng, scale=0.2):

    n = v.size

    i = rng.integers(0,n); j = rng.integers(0,n)

    if i==j: return v.copy()

    max_eps = min(v[i], 1.0 - v[j])

    if max_eps <= 0: return v.copy()

    eps = (rng.random() - 0.5)*2*max_eps*scale

    # clip eps within [-max_eps, max_eps]

    eps = max(-max_eps, min(eps, max_eps))

    new_v = v.copy()

    new_v[i] -= eps

    new_v[j] += eps

    return new_v

import time
import numpy as np
from scipy.optimize import minimize

# global placeholders used by the SLSQP surrogate
CURRENT_BETA = None
CURRENT_TARGET = None


def project_to_sum(v, target, max_iter=50, tol=1e-9):
    """Project vector v onto the capped simplex {x∈[0,1]^n, sum(x)=target}."""
    v = np.clip(v, 0.0, 1.0)
    lo = -np.max(v)               # ensures all v+lo ≤ 0
    hi = 1.0 - np.min(v)          # ensures all v+hi ≥ 1
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def half_to_full(v):
    """Mirror a half‑vector to obtain a symmetric full vector on [0,2]."""
    return np.concatenate([v, v[::-1]])


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_half(v):
    """C₅ for a symmetric function defined by half‑vector v."""
    return compute_c5(half_to_full(v))


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj_half(v):
    """Smooth surrogate of maxₖ Σᵢ vᵢ (1−v_{i+k}) for symmetric h."""
    full = half_to_full(v)
    corr = np.correlate(full, 1 - full, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_half(v0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a half‑vector."""
    global CURRENT_TARGET, CURRENT_BETA
    n = v0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj_half,
        v0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_to_sum(x, CURRENT_TARGET)


def perturb_h_half(v, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = v.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0.0:
        return v.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_v = v.copy()
    new_v[i] -= eps
    new_v[j] += eps
    return new_v


def hill_climb_half(v_start, rng, start_time, budget_s, max_iters=2000, start_step=0.5):
    """Simple hill‑climb on the half‑vector (accept only improvements)."""
    v = v_start.copy()
    best_v = v.copy()
    best_c = compute_c5_half(v)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
        new_v = perturb_h_half(v, rng, scale=step)
        c = compute_c5_half(new_v)
        if c < best_c:
            best_c = c
            best_v = new_v.copy()
            v = new_v
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-6)
    return best_v, best_c


def random_feasible_half(n_half, target_half, rng):
    """Generate a random feasible half‑vector."""
    v = rng.random(n_half)
    return project_to_sum(v, target_half)
