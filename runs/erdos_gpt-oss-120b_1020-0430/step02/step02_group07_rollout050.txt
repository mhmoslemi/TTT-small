    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    indices_one = np.where(h>0.5)[0]
    indices_zero = np.where(h<0.5)[0]

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the SLSQP surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log-sum-exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h_continuous(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5,
                          time_limit=None):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        new_h = perturb_h_continuous(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # decay step occasionally
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.8, 1e-6)
    return best_h, best_c


def simulated_annealing_continuous(h_start, rng, max_iters=2000,
                                   init_temp=0.02, decay=0.999,
                                   scale=0.3, time_limit=None):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        new_h = perturb_h_continuous(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the sum exactly (numeric drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def quick_refine_continuous(h, rng, deadline):
    """Fast refinement with a few β values and a short hill‑climb."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        if time.time() > deadline:
            break
        trial = slsqp_optimize(best_h, beta=beta, max_iter=300)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    if time.time() < deadline:
        hc, cc = hill_climb_continuous(best_h, rng, max_iters=800,
                                       start_step=0.4, time_limit=deadline)
        if cc < best_c:
            best_h, best_c = hc, cc
    return best_h, best_c


def deep_refine_continuous(h, rng, deadline):
    """Intensive refinement on high‑resolution data."""
    best_h, best_c = h.copy(), compute_c5(h)
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0)
    slsqp_iters = 250 if h.size > 200 else 300
    for beta in betas:
        if time.time() > deadline:
            break
        trial = slsqp_optimize(best_h, beta=beta, max_iter=slsqp_iters)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        if time.time() < deadline:
            hc, cc = hill_climb_continuous(best_h, rng, max_iters=2500,
                                           start_step=0.3, time_limit=deadline)
            if cc < best_c:
                best_h, best_c = hc, cc
    if time.time() < deadline:
        sa_h, sa_c = simulated_annealing_continuous(
            best_h, rng, max_iters=5000, init_temp=0.015,
            decay=0.998, scale=0.2, time_limit=deadline
        )
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, keeping the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def binary_hill_climb(h, rng, max_iters=2000, time_limit=None):
    """Hill‑climb on binary vectors via swaps."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # --------------------------------------------------------------
    # Initialise from the supplied parent solution (if any)
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = quick_refine_continuous(h0, rng, deadline)

    # --------------------------------------------------------------
    # Phase 1 – random restarts with quick refinement (≈50% of budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.55
    while time.time() < phase1_deadline:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine_continuous(h_start, rng, phase1_deadline)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # --------------------------------------------------------------
    # Phase 2 – up‑sample to higher resolution and deep refinement
    # --------------------------------------------------------------
    if time.time() < start_time + budget_s * 0.85:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        deep_deadline = start_time + budget_s * 0.95
        best_h, best_c = deep_refine_continuous(h_up, rng, deep_deadline)

    # --------------------------------------------------------------
    # Phase 3 – binary rounding + swap hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        n = best_h.size
        target = n // 2
        # greedy binary rounding: set the largest target entries to 1
        idx_desc = np.argsort(-best_h)
        bin_h = np.zeros_like(best_h)
        bin_h[idx_desc[:target]] = 1.0
        # binary hill‑climb
        bin_deadline = time.time() + remaining - 5.0
        bin_h, bin_c = binary_hill_climb(bin_h, rng,
                                         max_iters=int(remaining * 8),
                                         time_limit=bin_deadline)
        if bin_c < best_c:
            best_h, best_c = bin_h, bin_c

    # --------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    if abs(best_h.sum() - target) > 1e-9:
        best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
