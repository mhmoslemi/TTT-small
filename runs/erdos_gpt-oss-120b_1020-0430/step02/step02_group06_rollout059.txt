def improve_h(h, max_iter=1000, sigma=0.02, target=None, time_limit=None):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = len(h)
    for it in range(max_iter):
        if time_limit and time.time() - start > time_limit:
            break
        corr = np.correlate(best_h, 1-best_h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx]
        # compute contributions
        shift = max_idx - (n - 1)  # because full correlation length = 2n-1, index offset
        # shift may be negative: we need to map i-> h[i] * (1 - h[i+shift]) for overlapping indices.
        # compute overlapped i range
        if shift >= 0:
            idx_i = np.arange(0, n-shift)
            idx_j = idx_i + shift
        else:
            shift_pos = -shift
            idx_i = np.arange(shift_pos, n)
            idx_j = idx_i - shift_pos
        contrib = best_h[idx_i] * (1 - best_h[idx_j])
        # we want to reduce max_c, so move mass from i where contributions high to i where contributions low (maybe at same j for opponent)
        # Choose i_high: top p% maybe 5% (but at least 1)
        p = max(1, int(0.02*n))
        # sort indices by contrib
        sorted_idx = np.argsort(contrib)  # ascending
        low_idx = idx_i[sorted_idx[:p]]
        high_idx = idx_i[sorted_idx[-p:]]
        # compute step size epsilon (small)
        eps = sigma * np.random.rand()
        # Ensure we have capacity to add at low_idx, subtract at high_idx
        # Choose random i_low from low_idx, i_high from high_idx
        i_low = np.random.choice(low_idx)
        i_high = np.random.choice(high_idx)
        # move mass from high to low
        delta = min(eps, best_h[i_high], 1 - best_h[i_low])
        if delta > 0:
            trial_h = best_h.copy()
            trial_h[i_high] -= delta
            trial_h[i_low] += delta
            trial_h = project_to_feasible(trial_h) # though sum unchanged; maybe clip needed (but we ensure not out-of-bounds)
            trial_c5 = compute_c5(trial_h)
            if trial_c5 < best_c5:
                best_h = trial_h
                best_c5 = trial_c5
        # also occasionally random perturbation:
        if np.random.rand() < 0.1:
            trial_h = best_h + sigma * np.random.randn(n)
            trial_h = project_to_feasible(trial_h)
            trial_c5 = compute_c5(trial_h)

                best_h = trial_h
                best_c5 = trial_c5
        # optionally adjust sigma gradually (cooling)
        sigma *= 0.9995
        if target and best_c5 < target:
            break
    return best_h, best_c5

def _hill_climb(h, target, time_budget):
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    sigma = 0.03
    start = time.time()
    while time.time() - start < time_budget:
        # compute correlation
        # Note: correlation needed to find shift
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr_full)
        max_idx = np.argmax(corr_full)
        current_c5 = max_corr * (2.0 / n)
        # update best (should be same)
        best_c5 = current_c5
        # choose move
        if np.random.rand() < 0.5:
            # random perturb
            trial = best_h + sigma * np.random.randn(n)
            trial = _project_to_feasible(trial, n)
            trial_c5 = _c5_from_h(trial)
            if trial_c5 < best_c5 - 1e-12:
                best_h, best_c5 = trial, trial_c5
        else:
            # targeted move
            shift = max_idx - (n - 1)
            if shift >= 0:
                i = np.arange(0, n - shift)
                j = i + shift
            else:
                i = np.arange(-shift, n)
                j = i + shift
            if i.size > 0:
                contrib = best_h[i] * (1 - best_h[j])
                # number of extreme indices
                num = max(1, int(0.02 * contrib.size))
                # get sorted indices
                sorted_idx = np.argsort(contrib)
                low_i = i[sorted_idx[:num]]
                high_i = i[sorted_idx[-num:]]
                i_low = np.random.choice(low_i)
                i_high = np.random.choice(high_i)
                max_move = min(sigma, best_h[i_high], 1 - best_h[i_low])

def _solve_h(initial_h, max_iter=400):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1 - h, mode='full')
    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)
    def obj(x):
        return x[-1]

    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    opt_res = opt.minimize(obj, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    if opt_res.x is not None:
        h_opt = opt_res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

import time
import numpy as np
import scipy.optimize as opt

# ---------- feasibility utilities ----------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _local_refine(h: np.ndarray, attempts: int = 15, delta: float = 0.015) -> np.ndarray:
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


# ---------- optimization primitives ----------

def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def _obj(x):
        return x[-1]

    cons = [{"type": "ineq", "fun": _ineq}, {"type": "eq", "fun": _eq}]
    res = opt.minimize(
        _obj, x0, method="SLSQP", bounds=bounds, constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.x is not None else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h: np.ndarray, target: float, time_budget: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    sigma = 0.03
    start = time.time()
    while time.time() - start < time_budget:
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = np.argmax(corr_full)
        shift = max_idx - (n - 1)          # actual integer shift
        current_c5 = corr_full[max_idx] * (2.0 / n)
        best_c5 = current_c5

        if np.random.rand() < 0.5:
            trial = best_h + sigma * np.random.randn(n)
            trial = _project_to_feasible(trial, n)
            trial_c5 = _c5_from_h(trial)
            if trial_c5 < best_c5 - 1e-12:
                best_h, best_c5 = trial, trial_c5
        else:
            # contributions for the worst shift
            if shift >= 0:
                i = np.arange(0, n - shift)
                j = i + shift
            else:
                i = np.arange(-shift, n)
                j = i + shift
            if i.size:
                contrib = best_h[i] * (1.0 - best_h[j])
                cnt = max(1, int(0.02 * contrib.size))
                ord_idx = np.argsort(contrib)
                low_i = i[ord_idx[:cnt]]
                high_i = i[ord_idx[-cnt:]]
                i_low = np.random.choice(low_i)
                i_high = np.random.choice(high_i)
                max_move = min(sigma, best_h[i_high], 1.0 - best_h[i_low])
                if max_move > 0:
                    delta = np.random.rand() * max_move
                    trial = best_h.copy()
                    trial[i_high] -= delta
                    trial[i_low] += delta
                    trial_c5 = _c5_from_h(trial)
                    if trial_c5 < best_c5 - 1e-12:
                        best_h, best_c5 = trial, trial_c5
        sigma *= 0.9995
        if target is not None and best_c5 < target:
            break
    return best_h, best_c5
