def gradient_step(h, beta, lr, rng):
    n = h.shape[0]
    f = np.correlate(h, 1 - h, mode='full')
    max_f = np.max(f)
    p = np.exp(beta * (f - max_f))
    p_sum = p.sum()
    if p_sum == 0:
        p = np.ones_like(p) / p.size
    else:
        p /= p_sum

    g = np.zeros_like(h)

    # loops
    # shift index mapping: shift = idx - (n-1)
    # precompute (1 - h) into variable one_minus_h for speed
    one_minus_h = 1.0 - h

    for idx, pk in enumerate(p):
        if pk == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            k = shift
            # a term
            g[:n - k] += pk * one_minus_h[k:]
            # b term
            g[k:] -= pk * h[:n - k]
        else:
            k = -shift
            # shift negative
            g[k:] += pk * one_minus_h[:n - k]
            g[:n - k] -= pk * h[k:]

    # update
    h_new = h - lr * g
    # clip
    np.clip(h_new, 0.0, 1.0, out=h_new)
    # project sum
    target = n / 2.0
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        # add diff to random indices up to bound
        # Use while loop up to max attempts
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
    return h_new

def random_feasible_vector(n, rng):
    # generate uniform random between 0 and 1, then adjust sum.
    h = rng.random(n)
    target = n / 2.0
    # scaling then clipping
    scale = target / h.sum()
    h *= scale
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)

                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def hill_climb(h_start, rng, max_iters=1000, step_scale=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    step = step_scale
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        # perturb via moving epsilon between two coord (preserve sum)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        if eps > max_eps:
            eps = max_eps
        if eps < -max_eps:
            eps = -max_eps
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # ensure bounds
        if np.any(h_new < 0) or np.any(h_new > 1):
            continue
        # sum preserved automatically
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
        if (it + 1) % decay == 0:
            step *= 0.9
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact C5 for a discrete step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Continuous feasible vector: 0≤h≤1, Σh=n/2."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_binary_vector(n, rng):
    """Binary feasible vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample_vector(h, factor):
    """Integer up‑sampling by nearest‑neighbour repeat."""
    return np.repeat(h, factor)


# ------------------------------------------------------------
# gradient descent on soft‑max surrogate
# ------------------------------------------------------------
def _softmax_weights(f, beta):
    """Return soft‑max probabilities for vector f with temperature beta."""
    m = np.max(f)
    w = np.exp(beta * (f - m))
    s = w.sum()
    if s == 0.0:                     # under‑flow – fall back to one‑hot
        w = np.zeros_like(f)
        w[np.argmax(f)] = 1.0
        s = 1.0
    return w / s


def gradient_step(h, beta, lr, rng):
    """One gradient step on the smooth‑max objective."""
    n = h.size
    f = np.correlate(h, 1.0 - h, mode="full")          # shape 2n‑1
    w = _softmax_weights(f, beta)

    one_minus_h = 1.0 - h
    g = np.zeros_like(h)

    # vectorised slice updates for each shift
    # shift = idx - (n-1)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            k = shift
            g[: n - k] += wk * one_minus_h[k:]
            g[k:]      -= wk * h[: n - k]
        else:
            k = -shift
            g[k:]      += wk * one_minus_h[: n - k]
            g[: n - k] -= wk * h[k:]

    # gradient descent update
    h_new = h - lr * g
    np.clip(h_new, 0.0, 1.0, out=h_new)

    # enforce exact sum constraint by a cheap redistribution
    target = n / 2.0
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
    return h_new


def smooth_descent(h0, rng, beta_sched, lr0, iter_per_beta):
    """Run a full annealing schedule of gradient steps."""
    h = h0.copy()
    lr = lr0
    for beta, iters in zip(beta_sched, iter_per_beta):
        for _ in range(iters):
            h = gradient_step(h, beta, lr, rng)
        lr *= 0.9                     # slowly decay learning rate
    return h


# ------------------------------------------------------------
# local improvement utilities
# ------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=800, step_scale=0.5):
    """Simple hill‑climb that preserves the sum exactly."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 8)
    n = h.size
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
        if (it + 1) % decay == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def gradient_refine_worst(h, rng, step=0.02, max_iter=6):
    """Few deterministic gradient steps targeting the current worst shift."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:]    -= h[:upper]
        h = h - step * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- start from a supplied parent if available -----------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        np.clip(h0, 0.0, 1.0, out=h0)
        target0 = n0 / 2.0
        diff = target0 - h0.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n0)
                if diff > 0:
                    add = min(diff, 1.0 - h0[i])
                else:
                    add = max(diff, -h0[i])
                h0[i] += add
                diff = target0 - h0.sum()
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, n0

    # ---- grid sizes to explore ------------------------------------
    candidates = []
    if best_n is not None:
        candidates.append(best_n)
        if best_n * 2 <= 400:
            candidates.append(best_n * 2)
        if best_n * 3 <= 400:
            candidates.append(best_n * 3)
    else:
        candidates = [120, 180, 240, 300]

    candidates = sorted(set(candidates))

    # ---- annealing schedule ----------------------------------------
    beta_sched = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0]
    # iterations per beta (more for smaller beta)
    iter_per_beta = [500, 400, 300, 250, 200, 150]

    # ---------------------------------------------------------------
    for n in candidates:
        if time.time() - start > budget_s * 0.94:
            break
        # number of independent starts for this grid size
        attempts = 8 if n <= 240 else 5
        for att in range(attempts):
            if time.time() - start > budget_s * 0.94:
                break

            # ----- initialise -------------------------------------------------
            if best_n == n and att == 0:
                h = best_h.copy()
            elif "initial_h_values" in globals() and n % best_n == 0:
                factor = n // best_n
                h = upsample_vector(best_h, factor)
            else:
                if rng.random() < 0.5:
                    h = random_binary_vector(n, rng)
                else:
                    h = random_feasible_vector(n, rng)

            # ----- smooth‑max gradient descent --------------------------------
            lr0 = 0.02 / np.sqrt(n)                # scale learning rate with dimension
            h = smooth_descent(h, rng, beta_sched, lr0, iter_per_beta)

            # ----- deterministic refinement on the worst shift ----------------
            h = gradient_refine_worst(h, rng, step=0.02, max_iter=6)

            # ----- short hill‑climb for fine tuning -------------------------
            if time.time() - start < budget_s * 0.96:
                h, c = hill_climb(h, rng, max_iters=1200, step_scale=0.4)
            else:
                c = compute_c5(h)

            # ----- keep the best ------------------------------------------------
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # -----------------------------------------------------------------------
    # final feasibility enforcement
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5, dtype=float)

    np.clip(best_h, 0.0, 1.0, out=best_h)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
