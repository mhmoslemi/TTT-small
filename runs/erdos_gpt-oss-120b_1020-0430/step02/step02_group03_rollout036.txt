import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def greedy_worst_shift(h, rng, max_iters=2000):
    """Iteratively move mass from low‑V to high‑V for the worst autocorrelation shift."""
    n = h.size
    for _ in range(max_iters):
        auto = np.correlate(h, h, mode='full')
        min_idx = auto.argmin()
        shift = min_idx - (n - 1)
        # compute V_i = h[i+shift] + h[i-shift] with zero padding
        v = np.zeros(n)
        k = shift
        if k >= 0:
            if k < n:
                v[:n - k] += h[k:]
                v[k:] += h[:n - k]
        else:
            k = -k
            if k < n:
                v[k:] += h[:n - k]
                v[:n - k] += h[k:]
        # choose candidates
        i_candidates = np.where(h > 1e-12)[0]
        j_candidates = np.where(h < 1 - 1e-12)[0]
        if i_candidates.size == 0 or j_candidates.size == 0:
            break
        i = i_candidates[np.argmin(v[i_candidates])]
        j = j_candidates[np.argmax(v[j_candidates])]
        eps_max = min(h[i], 1 - h[j])
        if eps_max <= 1e-12:
            # fallback random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

            if eps_max <= 0:
                continue
        eps = eps_max * rng.random() * 0.5
        if eps < 1e-12:
            continue
        h[i] -= eps
        h[j] += eps
    return h

from scipy.optimize import minimize

# ----- global placeholders for the smooth surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_{k} Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)

    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

    """Run SLSQP on the smooth surrogate for a given β."""

def greedy_worst_shift(h, rng, max_iters=200):
    """
    Move mass from indices with low contribution to the worst autocorrelation shift
    towards indices with high contribution.
    """
    n = h.size
    for _ in range(max_iters):
        auto = np.correlate(h, h, mode="full")
        min_idx = auto.argmin()
        shift = min_idx - (n - 1)               # shift corresponding to the worst S_k
        v = np.zeros(n)                         # V_i(shift)
        k = shift
        if k >= 0:
            if k < n:
                v[: n - k] += h[k:]
                v[k:] += h[: n - k]
        else:
            k = -k
            if k < n:
                v[k:] += h[: n - k]
                v[: n - k] += h[k:]
        # viable donors / recipients
        donors = np.where(h > 1e-12)[0]
        recips = np.where(h < 1 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = donors[np.argmin(v[donors])]
        j = recips[np.argmax(v[recips])]
        eps_max = min(h[i], 1 - h[j])

    return h
