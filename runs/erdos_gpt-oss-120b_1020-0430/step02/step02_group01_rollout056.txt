def transfer_move(h, rng, max_step):

def targeted_move(h, rng, step):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n-1)   # shift
    # neighbor sum g_i
    g = np.zeros(n)
    if k >= 0:
        # for i in 0..n-1-k: contributions from i+k
        g[:n-k] += h[k:]
        g[k:] += h[:n-k]
    else:
        k2 = -k
        g[k2:] += h[:n-k2]
        g[:n-k2] += h[k2:]
    # pick inc and dec indices
    inc_candidates = (1.0 - h) * g
    dec_candidates = h * g
    i_inc = np.argmax(inc_candidates)
    i_dec = np.argmin(dec_candidates)
    # limit epsilon
    max_inc = min(step, 1.0 - h[i_inc])
    max_dec = min(step, h[i_dec])
    eps = min(max_inc, max_dec)
    if eps > 0:
        h[i_inc] += eps
        h[i_dec] -= eps
    return h

import time, math, numpy as np

# ------------------------ core utilities ---------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function h on [0,2] using FFT correlation."""
    n = h.size
    dh = 1.0 - h
    rev = dh[::-1]
    n2 = 1 << ((2 * n - 1).bit_length())          # power‑of‑two >=2n‑1
    H = np.fft.rfft(h, n=n2)
    R = np.fft.rfft(rev, n=n2)
    corr = np.fft.irfft(H * R, n=n2)[: 2 * n - 1]
    return float(np.max(corr) * (2.0 / n))


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def transfer_move(h: np.ndarray, rng: np.random.Generator, max_step: float) -> np.ndarray:
    """Move a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h  # no feasible move
    eps = (rng.random() - 0.5) * 2.0 * max_step * max_eps
    eps = max(-max_eps, min(eps, max_eps))
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


def targeted_move(h: np.ndarray, rng: np.random.Generator, step: float) -> np.ndarray:
    """Deterministic move aimed at the current worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr) - (n - 1)  # shift that attains the maximum
    g = np.zeros(n)
    if k >= 0:
        g[: n - k] += h[k:]          # h[i+k] term
        g[k:] += h[: n - k]          # h[i-k] term
    else:
        kk = -k
        g[kk:] += h[: n - kk]        # h[i-k]
        g[: n - kk] += h[kk:]        # h[i+k]
    inc_score = (1.0 - h) * g
    dec_score = h * g
    i_inc = int(np.argmax(inc_score))
    i_dec = int(np.argmin(dec_score))
    eps = min(step, 1.0 - h[i_inc], h[i_dec])
    if eps > 0.0:
        h[i_inc] += eps
        h[i_dec] -= eps
    return h


def hill_climb(
    h0: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 5000,
    start_step: float = 0.5,
    decay: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Simple hill‑climb that only accepts improvements."""
    h = h0.copy()
    best = h.copy()
    best_val = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        cand = transfer_move(h, rng, step)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            h = cand
        # occasional deterministic tweak
        if (it + 1) % 100 == 0:
            h = targeted_move(h, rng, step * 0.5)
        step = max(step * decay, 1e-4)
    return best, best_val


def simulated_annealing(
    h0: np.ndarray, rng: np.random.Generator, time_limit: float, n: int
) -> tuple[np.ndarray, float]:
    """Annealing using transfer moves and occasional targeted updates."""
    start = time.time()
    target = n / 2.0
    h = h0.copy()
    cur = compute_c5(h)
    best_h, best_val = h.copy(), cur
    # initial temperature as a fraction of the current value
    T0 = max(1e-6, cur * 0.05)
    it = 0
    while time.time() - start < time_limit:
        it += 1
        T = T0 * (0.9995 ** it)
        step = max(0.01, 0.5 * (T / T0))
        cand = transfer_move(h, rng, step)
        cand_val = compute_c5(cand)
        delta = cand_val - cur
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h, cur = cand, cand_val
            if cur < best_val:
                best_val, best_h = cur, h.copy()
        # every few iterations try a deterministic move
        if it % 200 == 0:
            h = targeted_move(h, rng, step)
            cur = compute_c5(h)
            if cur < best_val:
                best_val, best_h = cur, h.copy()
    return best_h, best_val


def upsample_linear(h: np.ndarray, new_n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation to a finer grid, with sum repair."""
    n = h.size
    old_x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = new_n / 2.0
    diff = target - h_new.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, new_n)
        if diff > 0:
            add = min(diff, 1.0 - h_new[i])
        else:
            add = max(diff, -h_new[i])
        h_new[i] += add
        diff = target - h_new.sum()
    return h_new


# ---------------------------- main driver --------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimize a step function h to lower the Erdős C5 constant."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # ---------- initialise from parent if available ----------
    best_h = None
    best_val = np.inf
    best_n = None
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        tgt = n0 / 2.0
        diff = tgt - h0.sum()
        while abs(diff) > 1e-12:
            i = rng.integers(0, n0)
            if diff > 0:
                add = min(diff, 1.0 - h0[i])
            else:
                add = max(diff, -h0[i])
            h0[i] += add
            diff = tgt - h0.sum()
        best_h, best_val, best_n = h0, compute_c5(h0), n0

    # ---------- search over several discretisations ----------
    candidates = [60, 84, 96, 108, 120, 144, 168, 192, 240]
    if best_n is not None:
        candidates = [best_n] + [n for n in candidates if n != best_n]

    for n in candidates:
        if time_left() < 5.0:
            break
        attempts = 0
        max_attempts = 4
        while attempts < max_attempts and time_left() > 5.0:
            attempts += 1
            # initialise a feasible vector for this n
            if best_n == n and best_h is not None and attempts == 1:
                h = best_h.copy()
            else:
                if rng.random() < 0.5:
                    h = np.zeros(n)
                    idx = rng.choice(n, n // 2, replace=False)
                    h[idx] = 1.0
                else:
                    h = random_feasible(n, rng)

            # local hill‑climb
            h, val = hill_climb(h, rng, max_iters=4000, start_step=0.5)
            if val < best_val:
                best_h, best_val, best_n = h.copy(), val, n

            # simulated annealing (time‑boxed)
            anneal_time = min(30.0, 0.2 * time_left())
            if anneal_time > 2.0:
                h_sa, val_sa = simulated_annealing(h, rng, anneal_time, n)
                if val_sa < best_val:
                    best_h, best_val, best_n = h_sa.copy(), val_sa, n
                    h = h_sa
                else:
                    h = h_sa

            # final refinement hill‑climb
            h, val = hill_climb(h, rng, max_iters=2000, start_step=0.3)
            if val < best_val:
                best_h, best_val, best_n = h.copy(), val, n

    # ---------- up‑sample refinement if time permits ----------
    if best_h is not None and time_left() > 10.0:
        up_n = best_n * 2
        if up_n <= 240:
            h_up = upsample_linear(best_h, up_n, rng)
            h_up, val_up = hill_climb(h_up, rng, max_iters=5000, start_step=0.3)
            if val_up < best_val:
                best_h, best_val, best_n = h_up, val_up, up_n

    # ---------- final feasibility enforcement ----------
    if best_h is None:  # fallback to a simple half‑filled vector
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, best_n)
        if diff > 0:
            add = min(diff, 1.0 - best_h[i])
        else:
            add = max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()
    final_val = compute_c5(best_h)

    return best_h.tolist(), float(final_val), int(best_n)
