def binaryize(h):
    n = len(h)
    half = n // 2
    # start with floor/h threshold 0.5
    bin_h = (h > 0.5).astype(float)
    current = int(bin_h.sum())
    if current < half:
        # need to add ones: choose zeros with highest original values
        zeros = np.where(bin_h == 0)[0]
        # sort zeros by original h descending
        order = np.argsort(-h[zeros])
        bin_h[zeros[order[:half-current]]] = 1
    elif current > half:
        # need to remove ones: choose ones with lowest original values
        ones = np.where(bin_h == 1)[0]
        order = np.argsort(h[ones])  # ascending
        bin_h[ones[order[:current-half]]] = 0
    return bin_h
```

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a (already feasible) step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_h(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    """SLSQP optimisation of the max‑overlap problem starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq}
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# --------------------------------------------------------------
# Small utilities for binary handling and simulated annealing
# --------------------------------------------------------------

def _binary_from_continuous(h: np.ndarray) -> np.ndarray:
    """Round a continuous feasible vector to a binary vector with exact half‑ones."""
    n = h.size
    half = n // 2
    bin_h = (h > 0.5).astype(float)
    cur = int(bin_h.sum())
    if cur < half:
        zeros = np.where(bin_h == 0)[0]
        # prefer zeros that had larger original values
        order = np.argsort(-h[zeros])
        bin_h[zeros[order[:half - cur]]] = 1.0
    elif cur > half:
        ones = np.where(bin_h == 1)[0]
        order = np.argsort(h[ones])  # smallest originals become zeros
        bin_h[ones[order[:cur - half]]] = 0.0
    return bin_h


def _random_binary(n: int) -> np.ndarray:
    """Uniform random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


def _compute_c5(h: np.ndarray) -> float:
    """Plain fast C5 computation (no feasibility check)."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _simulated_annealing_binary(start_h: np.ndarray, max_seconds: float,
                                target: float = 0.3808) -> tuple:
    """
    Simple SA on binary vectors (exact half‑ones). Returns best (h, C5) found.
    """
    n = start_h.size
    h = start_h.copy()
    best = h.copy()
    cur_val = _compute_c5(h)
    best_val = cur_val

    start = time.time()
    T0, T1 = 0.02, 1e-5

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        frac = elapsed / max_seconds
        T = T0 * (1 - frac) + T1 * frac

        # propose a swap of a 1 and a 0
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_val = _compute_c5(h)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = h.copy()
                if best_val < target:
                    break
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    return best, best_val


# --------------------------------------------------------------
# Main driver
# --------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Retrieve optional initial construction
    # ------------------------------------------------------------------
    try:
        init_cont = np.array(initial_h_values, dtype=float)
    except NameError:
        init_cont = None

    # ------------------------------------------------------------------
    # Decide on grid size
    # ------------------------------------------------------------------
    if init_cont is not None:
        n0 = init_cont.size
    else:
        n0 = 256
    candidates = [n0]
    if n0 < 512:
        candidates.append(min(512, n0 * 2))
    if n0 < 1024:
        candidates.append(min(1024, n0 * 4))
    candidates = [c for c in candidates if c % 2 == 0]

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Main optimisation loop over resolutions
    # ------------------------------------------------------------------
    for n in candidates:
        if time_left() < 30.0:
            break

        # ------------------------------------------------------------------
        # Build initial binary vector for this resolution
        # ------------------------------------------------------------------
        if best_h is not None and best_n == n:
            init_bin = best_h.copy()
        elif best_h is not None:
            # upscale previous best and binarise
            cont = _upscale_h(best_h, n)
            init_bin = _binary_from_continuous(cont)
        elif init_cont is not None and init_cont.size == n:
            init_bin = _binary_from_continuous(init_cont)
        elif init_cont is not None and init_cont.size != n:
            # upscale/downscale the provided construction
            cont = _upscale_h(init_cont, n)
            init_bin = _binary_from_continuous(cont)
        else:
            init_bin = _random_binary(n)

        # ------------------------------------------------------------------
        # Allocate time for binary simulated annealing
        # ------------------------------------------------------------------
        reserve = 30.0  # keep for final SLSQP and safety
        sa_time = max(5.0, time_left() - reserve)
        sa_time = min(sa_time, 300.0)   # avoid spending all budget on SA at once

        # Run a few SA attempts (each ~sa_time/num_attempts)
        attempts = max(1, int(sa_time // 20))
        per_attempt = sa_time / attempts

        for _ in range(attempts):
            if time_left() < reserve + 5.0:
                break
            bin_res, val_res = _simulated_annealing_binary(init_bin, per_attempt, target=0.3808)
            if val_res < best_c5:
                best_c5 = val_res
                best_h = bin_res.copy()
                best_n = n
            # use the result as the next start
            init_bin = bin_res

            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final polishing with continuous SLSQP (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time_left() > 15.0:
        # start from the binary solution but allow fractional adjustments
        h_start = best_h.astype(float)
        # give a bit more iterations if enough time
        max_iter = 2000
        h_opt, c5_val = _solve_h(h_start, max_iter=max_iter)
        if c5_val < best_c5:
            best_h = h_opt
            best_c5 = c5_val
            best_n = h_opt.size

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    # ------------------------------------------------------------------
    # Ensure feasibility and final C5 evaluation
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
