import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Feasibility projection (clip + water‑filling)
# ------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ------------------------------------------------------------
# C5 evaluation
# ------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ------------------------------------------------------------
# Upscaling (linear interpolation)
# ------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ------------------------------------------------------------
# SLSQP refinement (same as previous program)
# ------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 200) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ------------------------------------------------------------
# Gradient of the maximal‑overlap shift
# ------------------------------------------------------------
def _gradient_max_shift(h: np.ndarray) -> tuple:
    """Return (gradient vector, shift index, full correlation array)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)  # signed shift
    grad = np.zeros(n)
    if k >= 0:
        # i from 0 .. n-1-k  -> (1 - h[i+k])
        grad[: n - k] += (1.0 - h[k:])
        # i from k .. n-1  -> -h[i-k]
        grad[k:] += -h[: n - k]
    else:
        p = -k
        # i from p .. n-1  -> (1 - h[i-p])
        grad[p:] += (1.0 - h[: n - p])
        # i from 0 .. n-1-p -> -h[i+p]
        grad[: n - p] += -h[p:]
    return grad, k, corr


# ------------------------------------------------------------
# Simple projected gradient descent on the max‑overlap
# ------------------------------------------------------------
def _grad_descent(h_start: np.ndarray, time_limit: float) -> np.ndarray:
    """Return a feasible h after a timed projected‑gradient run."""
    n = h_start.size
    h = h_start.copy()
    best_h = h.copy()
    best_c = _c5_from_h(h)
    cur_c = best_c
    it = 0
    while time.time() < time_limit and it < 3000:
        grad, _, corr = _gradient_max_shift(h)
        cur_c = np.max(corr) * (2.0 / n)
        # step size schedule
        alpha = 0.02 * (0.985 ** it)
        h_new = h - alpha * grad
        h_new = _project_to_feasible(h_new, n)

        # evaluate new
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_c = np.max(corr_new) * (2.0 / n)

        # keep best ever
        if new_c < best_c:
            best_c, best_h = new_c, h_new.copy()

        # accept if improvement or tiny uphill chance
        if new_c < cur_c or np.random.rand() < 0.01:
            h, cur_c = h_new, new_c

        it += 1
    return best_h


# ------------------------------------------------------------
# Pairwise hill climbing (from the previous program)
# ------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    while max_steps > 0 and time.time() < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]          # can take at most this
        max_up = 1.0 - best_h[j]      # can give at most this
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
        max_steps -= 1
    return best_h, best_c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # -----------------------------------------------------------------
    # Load possible parent construction
    # -----------------------------------------------------------------
    try:
        GLOBAL_INITIAL_H = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL_H = None

    target = 0.3808
    best_h = None
    best_c = float("inf")
    best_n = None

    # grid sizes to explore
    n_candidates = [80, 96, 112, 128, 160, 192, 224, 256, 320]
    if GLOBAL_INITIAL_H is not None:
        n_candidates.append(GLOBAL_INITIAL_H.size)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 8.0:
            break

        # ---- base construction -------------------------------------------------
        if GLOBAL_INITIAL_H is not None and GLOBAL_INITIAL_H.size == n:
            base = GLOBAL_INITIAL_H.copy()
        elif GLOBAL_INITIAL_H is not None:
            base = _upscale_h(GLOBAL_INITIAL_H, n)
        else:
            base = np.full(n, 0.5)

        # ---- deterministic SLSQP refinement ------------------------------------
        h_opt, c_val = _solve_h(base, max_iter=200)
        if c_val < best_c:
            best_h, best_c, best_n = h_opt, c_val, n

        # ---- projected gradient descent ----------------------------------------
        limit = start + budget_s - 0.5
        h_gd = _grad_descent(best_h, time_limit=limit)
        c_gd = _c5_from_h(h_gd)
        if c_gd < best_c:
            best_h, best_c, best_n = h_gd, c_gd, n

        # ---- final hill‑climbing refinement ------------------------------------
        limit = start + budget_s - 0.2
        h_final, c_final = _hill_climb(best_h, max_steps=2000, time_limit=limit)
        if c_final < best_c:
            best_h, best_c, best_n = h_final, c_final, n

        if best_c < target:
            break

    # fallback (unlikely)
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c), int(best_n)
