import time
import numpy as np
from scipy import optimize

# ----- global variable used by the SLSQP constraints -----
_solve_n = None  # will be set before each call to _solve_h


def _c5_of_seq(seq: np.ndarray) -> float:
    """Overlap bound for a (binary or real) step function."""
    n = seq.size
    return np.max(np.correlate(seq, 1.0 - seq, mode="full")) * (2.0 / n)


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every integer shift."""
    n = _solve_n
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h)=n/2."""
    n = _solve_n
    return np.sum(x[:n]) - n / 2.0


def _objective(x: np.ndarray) -> float:
    """Auxiliary variable t to be minimised."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    """Run a short SLSQP optimisation starting from initial_h."""
    global _solve_n
    n = initial_h.size
    _solve_n = n
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=constraints, options=opts
    )
    h_opt = _project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    c5_val = _c5_of_seq(h_opt)
    return h_opt, c5_val


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    xs_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xs_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xs_f, xs_c, coarse_h)


def _binary_sa(n: int, start_seq: np.ndarray, deadline: float) -> tuple:
    """Simulated‑annealing style hill‑climbing on binary vectors."""
    cur = start_seq.copy()
    cur_val = _c5_of_seq(cur)
    best_seq = cur.copy()
    best_val = cur_val
    it = 0
    T0 = 0.02
    while time.time() < deadline:
        it += 1
        # linear cooling (very slow, enough for our short runs)
        T = max(T0 * (1.0 - it / 200000.0), 1e-6)
        i, j = np.random.randint(n, size=2)
        if cur[i] == cur[j]:
            continue
        # propose swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = _c5_of_seq(cur)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]
    return best_seq, best_val
