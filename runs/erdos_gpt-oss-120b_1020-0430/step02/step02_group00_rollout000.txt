# ----------------------------------------------------------------------
# Feasibility projection (bounds + sum constraint)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if still off
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation (assumes feasibility)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Constraints for the auxiliary-variable formulation
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


# ----------------------------------------------------------------------
# Single SLSQP optimisation starting from any vector (feasible or not)
# ----------------------------------------------------------------------
def _solve_h(start_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = start_h.size
    h0 = _project_to_feasible(start_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        # fallback to projected start
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Upscale a coarse step function onto a finer grid by linear interpolation
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse_h)


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # ------------------------------------------------------------------
    # Baseline start vector
    # ------------------------------------------------------------------
    if init_h is not None:
        n0 = init_h.size
        best_h = _project_to_feasible(init_h, n0)
        best_n = n0
    else:
        # default: half ones, half zeros randomised
        n0 = 256
        base = np.zeros(n0, dtype=float)
        base[: n0 // 2] = 1.0
        np.random.shuffle(base)
        best_h = base
        best_n = n0

    best_c5 = _c5_from_h(best_h)

    # ------------------------------------------------------------------
    # Helper to possibly update the current best solution
    # ------------------------------------------------------------------
    def maybe_update(new_h, new_c5, n):
        nonlocal best_h, best_c5, best_n
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = float(new_c5)
            best_n = n

    # ------------------------------------------------------------------
    # Initial local refinement of the starting vector
    # ------------------------------------------------------------------
    h_opt, c5_val = _solve_h(best_h, max_iter=800)
    maybe_update(h_opt, c5_val, best_n)

    # ------------------------------------------------------------------
    # Set of grid sizes to explore (including the base size)
    # ------------------------------------------------------------------
    candidate_ns = [256, 280, 300, 320, 340, 360, 380, 400, 432, 456]
    # keep only even sizes and below 1000
    candidate_ns = [n for n in candidate_ns if n % 2 == 0 and n <= 1000]
    candidate_ns = sorted(set(candidate_ns + ([best_n] if best_n else [])))

    # ------------------------------------------------------------------
    # Main multi‑resolution search
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time_left() < 12.0:
            break

        # initialise a starting point for this resolution
        if best_n == n:
            base_h = best_h.copy()
        else:
            base_h = _upscale_h(best_h, n)

        base_h = _project_to_feasible(base_h, n)

        # decide how many attempts we can afford at this resolution
        remaining = time_left()
        # rough estimate: each SLSQP run ~ 0.2–0.3 s for n≈300
        # reserve a safety margin for final polish
        max_attempts = max(5, int(remaining // (0.25 * len(candidate_ns))))
        max_attempts = min(max_attempts, 30)

        for att in range(max_attempts):
            if time_left() < 5.0:
                break

            # start point: either fresh random feasible or a perturbed base
            if att % 5 == 0:
                start_h = np.random.rand(n)
                start_h = _project_to_feasible(start_h, n)
            else:
                sigma = 0.35 * (0.82 ** att)   # decreasing magnitude
                pert = sigma * np.random.randn(n)
                start_h = base_h + pert
                start_h = _project_to_feasible(start_h, n)

            # local optimisation
            h_opt, c5_val = _solve_h(start_h, max_iter=600)
            maybe_update(h_opt, c5_val, n)

            # keep a good candidate as new base for subsequent attempts
            if c5_val < best_c5 * 0.999:
                base_h = h_opt

            # early exit if already under target
            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Fine‑grained hill climbing on a binary rounding of the best vector
    # ------------------------------------------------------------------
    # round to 0/1 while preserving exact half‑mass
    bin_h = np.rint(best_h).astype(float)
    diff = int(best_n // 2 - bin_h.sum())
    if diff > 0:
        zero_idx = np.where(bin_h == 0)[0]
        if zero_idx.size >= diff:
            choose = np.random.choice(zero_idx, size=diff, replace=False)
            bin_h[choose] = 1.0
    elif diff < 0:
        one_idx = np.where(bin_h == 1)[0]
        if one_idx.size >= -diff:
            choose = np.random.choice(one_idx, size=-diff, replace=False)
            bin_h[choose] = 0.0
    bin_h = bin_h.astype(float)

    best_bin_h = bin_h.copy()
    best_bin_c5 = _c5_from_h(best_bin_h)

    # hill climbing swaps
    possible_swaps = int(min(20000, time_left() * 1500))
    for _ in range(possible_swaps):
        if time_left() < 0.5:
            break
        ones = np.flatnonzero(best_bin_h == 1)
        zeros = np.flatnonzero(best_bin_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best_bin_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = _c5_from_h(cand)
        if cand_c5 < best_bin_c5:
            best_bin_h = cand
            best_bin_c5 = cand_c5

    # Run a final SLSQP refinement starting from the best binary candidate
    h_refined, c5_refined = _solve_h(best_bin_h, max_iter=800)
    maybe_update(h_refined, c5_refined, h_refined.size)

    # ------------------------------------------------------------------
    # Final feasibility enforcement and return
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_h.size)
    final_c5 = _c5_from_h(best_h)
    return best_h.tolist(), float(final_c5), int(best_h.size)
