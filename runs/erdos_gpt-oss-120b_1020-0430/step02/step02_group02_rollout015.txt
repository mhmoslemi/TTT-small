import time
import numpy as np

# ----------------------------------------------------------------------
# Feasibility utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9,
                         max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0.0
        else:
            free = h < 1.0
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C₅ from a feasible h on the discretised grid."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Gradient of a single shifted correlation
# ----------------------------------------------------------------------
def _grad_corr_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of C_k where k = shift (relative to centre of full correlation).
    For shift s: dC/dh_j = 1 - h_{j+s} - h_{j-s}, with out‑of‑range terms omitted.
    """
    n = h.size
    grad = np.ones(n, dtype=float)
    if shift >= 0:
        s = shift
        # forward term
        grad[: n - s] -= h[s:]
        # backward term
        grad[s:] -= h[: n - s]
    else:
        s = -shift
        # forward term (j-s)
        grad[s:] -= h[: n - s]
        # backward term (j+s)
        grad[: n - s] -= h[s:]
    return grad


# ----------------------------------------------------------------------
# Simple hill‑climb refinement (pairwise epsilon moves)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    rng = np.random.default_rng()
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c

# Main optimisation kernel – balance max/min correlations
# ----------------------------------------------------------------------
def _descend_maxmin(start_h: np.ndarray, n: int, time_limit: float,
                    rng: np.random.Generator) -> tuple:
    """Projected gradient descent on the spread between max and min correlations."""
    h = start_h.copy()
    best_h = h.copy()
    best_c = _c5_from_h(h)
    step = 0.5
    while time.time() < time_limit:
        # full correlation C_k = Σ h_i (1 - h_{i+k})
        C_raw = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(C_raw))
        min_idx = int(np.argmin(C_raw))
        shift_max = max_idx - (n - 1)
        shift_min = min_idx - (n - 1)

        # gradients for the worst and best shifts
        g_max = _grad_corr_shift(h, shift_max)
        g_min = _grad_corr_shift(h, shift_min)

        # direction that lowers the worst and raises the best
        d = -g_max + g_min
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            d = rng.normal(size=n)
            d_norm = np.linalg.norm(d)
        # spread between extremes
        spread = C_raw[max_idx] - C_raw[min_idx]
        if spread <= 1e-12:
            break
        # step size scaled to spread
        alpha = step * spread / (n * d_norm + 1e-12)

        h_new = h + alpha * d
        h_new = _project_to_feasible(h_new, n)

        c_new = _c5_from_h(h_new)
        # keep global best
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new

        # accept if non‑worsening, occasional acceptance of worse moves
        if c_new <= _c5_from_h(h) or rng.random() < 0.01:
            h = h_new

        step *= 0.9995
        if step < 1e-6:
            break
        # safety break
        if time.time() + 0.2 > time_limit:
            break
    return best_h, best_c


# ----------------------------------------------------------------------
# Orchestrator
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # possible initial construction
    try:
        GLOBAL_INITIAL_H = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL_H = None

    # grid sizes to explore (even numbers, modest)
    n_candidates = [80, 96, 112, 128, 160, 192, 224, 256, 320, 384, 512]

    best_h = None
    best_c = float("inf")
    best_n = None

    for n in n_candidates:
        if time_left() < 6.0:
            break

        # ------------------------------------------------------------------
        # Build a viable starting vector for this n
        # ------------------------------------------------------------------
        if GLOBAL_INITIAL_H is not None and GLOBAL_INITIAL_H.size == n:
            base_h = GLOBAL_INITIAL_H.copy()
        elif GLOBAL_INITIAL_H is not None:
            # linear interpolation to target size
            x_old = (np.arange(GLOBAL_INITIAL_H.size) + 0.5) * (2.0 / GLOBAL_INITIAL_H.size)
            x_new = (np.arange(n) + 0.5) * (2.0 / n)
            base_h = np.interp(x_new, x_old, GLOBAL_INITIAL_H)
        else:
            # random feasible start centred at 0.5
            base_h = rng.random(n)
            base_h = _project_to_feasible(base_h, n)

        # number of random restarts for this size
        restarts = 4 if n <= 256 else 2
        for r in range(restarts):
            if time_left() < 4.0:
                break
            # optionally perturb for non‑first restarts
            if r > 0:
                perturb = rng.normal(scale=0.2, size=n)
                start = base_h + perturb
                start = _project_to_feasible(start, n)
            else:
                start = base_h.copy()

            # ------------------------------------------------------------------
            # Gradient descent on max/min spread
            # ------------------------------------------------------------------
            limit = start_time + budget_s - 1.0
            h_opt, c_opt = _descend_maxmin(start, n, limit, rng)

            # ------------------------------------------------------------------
            # Local hill‑climb fine tuning
            # ------------------------------------------------------------------
            if time.time() < limit:
                h_opt, c_opt = _hill_climb(h_opt, max_steps=2000, time_limit=limit)

            # keep global best
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt, c_opt, n
                if best_c <= 0.3808:
                    break
        if best_c <= 0.3808:
            break

    # fallback if nothing succeeded
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_h = _project_to_feasible(best_h, n)
        best_c = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c), int(best_n)
