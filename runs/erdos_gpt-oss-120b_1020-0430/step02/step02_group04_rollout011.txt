def smooth_optimize(h0, rng, n, beta_sched, steps_per_beta, step0=1.0, mu=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in beta_sched:
        step = step0
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = np.sum(w)
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            # momentum update
            v = mu * v - step * grad
            new_h = h + v
            # projection
            new_h = project_onto_box_sum(new_h, n/2.0)
            # maybe clip for numeric stability
            new_h = np.clip(new_h, 0.0, 1.0)
            # evaluate
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                # maybe accept new_h as current
                h = new_h
                # reset velocity?
                v = np.zeros_like(h)
            else:
                # Accept if improvement in objective? Already considered.
                # Maybe still accept move with small probability to allow exploration?
                # Let's use simple rule: if np.random.rand() < 0.001: accept.
                if rng.random() < 0.001:
                    h = new_h
                # else keep h unchanged (velocity already updated)
            # adapt step:
            step *= 0.9995
        # after each beta maybe reduce step further.
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -10.0, 10.0
    for _ in range(max_iter):
        lam = 0.5 * (lo + hi)
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + 0.5 * (lo + hi), lower, upper)
    # tiny residual correction
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Generate a symmetric feasible h (0<=h<=1, sum = n/2)."""
    half = n // 2
    target_half = n / 4.0
    x = rng.random(half)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # -------------------------------------------------
    # target dimensionality (must be even for easy symmetry)
    n = 200
    # build an initial feasible symmetric h
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            # linear interpolation to desired size
            x_old = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, parent)
        else:
            h0 = parent.copy()
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, n / 2.0)
        h0 = symmetrize(h0)
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        h0 = random_feasible_symmetric(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # -------------------------------------------------
    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    steps_per_beta = 200   # enough for each schedule

    restarts = 30
    for r in range(restarts):
        if time.time() - start > budget_s * 0.85:
            break

        if r == 0:
            h_start = h0.copy()
        else:
            h_start = random_feasible_symmetric(n, rng)
            # occasional blend with current best
            if best_h is not None and rng.random() < 0.2:
                h_start = (h_start + best_h) * 0.5
                h_start = project_onto_box_sum(h_start, n / 2.0)
            h_start = symmetrize(h_start)
            h_start = project_onto_box_sum(h_start, n / 2.0)

        # smooth‑max gradient descent
        h_opt, c_opt = smooth_optimize(h_start, rng, n, beta_sched, steps_per_beta)

        # short SLSQP refinement at higher beta
        try:
            h_opt = slsqp_optimize(h_opt, beta=500, max_iter=150)
        except Exception:
            pass

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            # early exit if we beat the target comfortably
            if best_c < 0.3807:
                break

    # -------------------------------------------------
    # final fine‑tuning with hill‑climb
    if time.time() - start < budget_s * 0.96:
        h_hc, c_hc = hill_climb_sym(best_h, rng, max_iters=2500, step_scale=0.4)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # final projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
