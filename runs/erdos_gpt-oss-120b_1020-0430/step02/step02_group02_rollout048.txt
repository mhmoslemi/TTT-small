import time
import numpy as np
from scipy import optimize

# -------------------------------------------------------------
#  Helper utilities
# -------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water-filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for a given discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """
    Constrained SLSQP optimisation of max_{k} correlation.
    Returns (h_opt, C5_value).
    """
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _local_reduce_max(h: np.ndarray, time_limit: float) -> tuple:
    """
    Targeted hill‑climbing: repeatedly pick the shift that attains the
    current maximum correlation and move a small amount of mass from an
    index that helps the most to an index that hurts the least.
    """
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    eps_min = 1e-12

    while time.time() < time_limit:
        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = np.argmax(corr_full)
        s = max_idx - (n - 1)

        # shifted version of h needed for the gradient of the max shift
        if s >= 0:
            shifted = np.concatenate([np.zeros(s), cur_h[: n - s]])
        else:
            shifted = np.concatenate([cur_h[-s :], np.zeros(-s)])

        # feasible source / destination indices
        src = np.where(cur_h > eps_min)[0]
        dst = np.where(cur_h < 1 - eps_min)[0]
        if src.size == 0 or dst.size == 0:
            break

        # choose source with smallest shifted value, destination with largest
        i = src[np.argmin(shifted[src])]
        j = dst[np.argmax(shifted[dst])]

        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        # move a random fraction (1%‑50%) of the allowed amount
        eps = max_eps * (np.random.rand() * 0.49 + 0.01)

        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = _c5_from_h(cand)

        if cand_c + 1e-12 < cur_c:
            cur_h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
        else:
            # occasional random perturbation to escape shallow traps
            if np.random.rand() < 0.02:
                i2, j2 = np.random.randint(n, size=2)
                if i2 != j2:
                    max_eps2 = min(cur_h[i2], 1.0 - cur_h[j2])
                    if max_eps2 > 0:
                        eps2 = max_eps2 * (np.random.rand() * 0.5)
                        cand2 = cur_h.copy()
                        cand2[i2] -= eps2
                        cand2[j2] += eps2
                        cand2_c = _c5_from_h(cand2)
                        if cand2_c < cur_c:
                            cur_h, cur_c = cand2, cand2_c
                            if cand2_c < best_c:
                                best_h, best_c = cand2.copy(), cand2_c
    # final clean‑up
    best_h = _project_to_feasible(best_h, n)
    best_c = _c5_from_h(best_h)
    return best_h, best_c
