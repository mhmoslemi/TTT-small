    lo = -1.0
    hi = 1.0
    # we need to ensure lo, hi cover solution
    # compute s_lo = sum(min(1, max(0, v+lo))) etc.
    for _ in range(max_iter):
        mid = (lo+hi)/2
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

import time, numpy as np
from scipy.optimize import minimize

# ----------- basic utilities -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Continuous feasible vector (0≤h≤1, Σh = n/2)."""
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def random_binary_vector(n, rng):
    """Binary feasible vector with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return h


def upsample_to_target(h, target_n, rng):
    """Linear up‑sampling to target_n points while keeping integral."""
    n = h.size
    if target_n == n:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = target_n / 2.0
    return enforce_sum(h_new, target, rng)


# ----------- continuous local search ----------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly (in case of tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def quick_refine_continuous(h, rng):
    """Fast refinement: few β values and a short hill‑climb."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=250)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    trial_h, c_h = hill_climb(best_h, rng, max_iters=1500, start_step=0.35)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def deep_refine_continuous(h, rng):
    """Intensive refinement with larger β ladder and longer local searches."""
    n = h.size
    max_iter_slsqp = 300 if n <= 300 else 200
    hill_iters = 2500 if n <= 300 else 1500
    anneal_iters = 4000 if n <= 300 else 2000
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0)
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=max_iter_slsqp)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.25)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
    trial_sa, c_sa = simulated_annealing(best_h, rng, max_iters=anneal_iters,
                                         init_temp=0.03, decay=0.998, scale=0.2)
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


# ----------- binary local search -------------------------------------------
def binary_hill_climb(h, rng, max_iters=2000):
    """Hill‑climb on binary vectors (swap 1↔0) to minimise C5."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    ones = list(np.flatnonzero(best_h > 0.5))
    zeros = list(np.flatnonzero(best_h < 0.5))
    for _ in range(max_iters):
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        c = compute_c5(new_h)
        if c < best_c - 1e-9:
            best_h = new_h
            best_c = c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
    return best_h, best_c


# ----------- main driver ----------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- initial vector ---------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target = h0.size / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        h0 = random_feasible_vector(120, rng)

    # ---- up‑sample to a finer grid (max 720 points) -----------------------
    target_n = min(720, h0.size * 2)
    h = upsample_to_target(h0, target_n, rng)
    n = h.size

    # ---- quick refinement -------------------------------------------------
    best_h, best_c = quick_refine_continuous(h, rng)

    # ---- Phase 1: many short random restarts ------------------------------
    while time.time() - start_time < budget_s * 0.45:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine_continuous(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ---- Phase 2: deep refinement -----------------------------------------
    if time.time() - start_time < budget_s * 0.85:
        h_deep, c_deep = deep_refine_continuous(best_h, rng)
        if c_deep < best_c:
            best_h, best_c = h_deep, c_deep

    # ---- Phase 3: binary rounding and binary hill‑climb ------------------
    half = n // 2
    idx_top = np.argpartition(best_h, -half)[-half:]
    binary_h = np.zeros_like(best_h)
    binary_h[idx_top] = 1.0
    c_bin = compute_c5(binary_h)
    if c_bin < best_c:
        best_h, best_c = binary_h, c_bin

    if time.time() - start_time < budget_s * 0.97:
        bh, cb = binary_hill_climb(best_h, rng, max_iters=8000)
        if cb < best_c:
            best_h, best_c = bh, cb

    # ---- Optional extra binary random restarts if time permits -----------
    while time.time() - start_time < budget_s * 0.995:
        hb = random_binary_vector(n, rng)
        bh, cb = binary_hill_climb(hb, rng, max_iters=3000)
        if cb < best_c:
            best_h, best_c = bh, cb

    # ---- final feasibility enforcement ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
