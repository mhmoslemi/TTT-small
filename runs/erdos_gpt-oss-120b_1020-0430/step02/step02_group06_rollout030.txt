def compute_corr_max(h):
    n = h.shape[0]
    L = 2 * n
    Ha = np.fft.rfft(h, L)

import time
import numpy as np
from scipy import optimize

# -------------------- Global placeholder for SLSQP constraints --------------------
_cur_n = None  # will be set before each SLSQP call


# -------------------- Feasibility projection --------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# -------------------- Correlation utilities (FFT) --------------------
def _corr_vector(h: np.ndarray) -> np.ndarray:
    """Linear correlation vector of h with (1‑h) (zero‑padded)."""
    n = h.shape[0]
    L = 2 * n
    Ha = np.fft.rfft(h, L)
    Hb = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(Ha * np.conj(Hb), L)[: 2 * n - 1]
    return corr.real


def _max_corr(h: np.ndarray) -> float:
    """Maximum correlation value for vector h."""
    return _corr_vector(h).max()


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C₅ for discretised step function h."""
    n = h.shape[0]
    return _max_corr(h) * (2.0 / n)


# -------------------- Simulated annealing on binary vectors --------------------
def _anneal_binary(start_h: np.ndarray, max_steps: int, time_left_func) -> tuple:
    """Stochastic hill‑climbing on binary {0,1} vectors."""
    h = start_h.astype(float)
    n = h.size
    # enforce exactly n/2 ones
    ones_needed = n // 2
    current_ones = int(h.sum())
    if current_ones != ones_needed:
        idx = np.argsort(-h)
        h[:] = 0.0
        h[idx[:ones_needed]] = 1.0

    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for step in range(max_steps):
        if time_left_func() <= 0:
            break

        # pick a random 1 and a random 0
        i = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _c5_from_h(h)
        delta = new_c5 - cur_c5

        # linear cooling schedule
        T = 0.04 * (1.0 - step / max_steps)

        if delta <= 0 or (T > 0 and np.random.rand() < np.exp(-delta / T)):
            # accept
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5


# -------------------- Local random refinement (continuous) --------------------
def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02) -> np.ndarray:
    """Perturb a few coordinates and keep improvements."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):

        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h

    n = h_initial.size
    h0 = _project_to_feasible(h_initial, n)
