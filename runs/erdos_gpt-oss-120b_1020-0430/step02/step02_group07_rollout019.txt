```python
def compute_corr(h):
    n = h.size
    # Use FFT for n >= 200
    if n >= 200:
        m = 1 << ((2*n-1).bit_length())
        # zero pad to length m
        H = np.fft.rfft(h, n=m)
        K = np.fft.rfft(1 - h, n=m)
        corr = np.fft.irfft(H * np.conj(K), n=m)[:2*n-1]
        return corr
    else:
        return np.correlate(h, 1 - h, mode='full')
```

import time
import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d

# --------------------------- basic helpers ---------------------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place to satisfy Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Return a random vector in [0,1]^n with sum exactly n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


# --------------------------- correlation ------------------------------------
def compute_corr(h):
    """Linear correlation h * (1-h) (mode='full'). FFT for speed when n large."""
    n = h.size
    if n >= 200:
        m = 1 << ((2 * n - 1).bit_length())
        H = np.fft.rfft(h, n=m)
        K = np.fft.rfft(1.0 - h, n=m)
        corr = np.fft.irfft(H * np.conj(K), n=m)[: 2 * n - 1]
        return corr.real
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    """Exact discrete C5 for a step function with spacing 2/n."""
    corr = compute_corr(h)
    return np.max(corr) * (2.0 / h.size)


# --------------------------- perturbations ---------------------------------
def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one index to another (preserve sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        # slowly shrink step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=3000, init_temp=0.03,
                        decay=0.998, scale=0.2):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h, cur_c = new_h, new_c
        temp *= decay
    return best_h, best_c


# --------------------------- SLSQP surrogate -------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair any drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------- refinement pipelines --------------------------
def continuation_optimize(h, betas, slsqp_iter):
    """Apply SLSQP sequentially for a list of betas."""
    cur = h.copy()
    for b in betas:
        cur = slsqp_optimize(cur, beta=b, max_iter=slsqp_iter)
    return cur


def fast_refine(h, rng):
    """Quick surrogate‑based refinement used for many restarts."""
    betas = (5.0, 20.0, 80.0)
    cur = continuation_optimize(h, betas, slsqp_iter=200)
    cur, c = hill_climb(cur, rng, max_iters=800, start_step=0.4)
    cur, c2 = simulated_annealing(cur, rng, max_iters=2000,
                                  init_temp=0.02, decay=0.999, scale=0.3)
    return (cur, min(c, c2))


def intensive_refine(h, rng):
    """Deep refinement with more β values and longer local searches."""
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0)
    cur = continuation_optimize(h, betas, slsqp_iter=300)
    # Hill climb with more iterations
    cur, c_h = hill_climb(cur, rng, max_iters=2000, start_step=0.3)
    # Simulated annealing with longer run
    cur, c_sa = simulated_annealing(cur, rng, max_iters=5000,
                                    init_temp=0.03, decay=0.998, scale=0.2)
    return cur, min(c_h, c_sa)


def upsample_h(h, factor, rng):
    """Linear up‑sampling by integer factor, preserving integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    return enforce_sum(h_up, target_up, rng)


def smooth_start(h, rng, sigma=0.8):
    """Apply small Gaussian blur then re‑project onto feasible set."""
    h_s = gaussian_filter1d(h, sigma=sigma, mode='wrap')
    h_s = np.clip(h_s, 0.0, 1.0)
    return enforce_sum(h_s, h.size / 2.0, rng)
