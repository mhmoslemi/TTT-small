def grad_for_shift(h, k):
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        # i from 0 to n-k-1 contribute (1 - h[i+k]) to g[i]
        # and -h[i] to g[i+k]
        g[:n - k] += 1 - h[k:]
        g[k:] += -h[:n - k]
    else:
        # k negative, shift to left: i + k < i
        k_abs = -k
        # For shift -k, the expression sum_i h[i] * (1 - h[i-k]) i >= k => i from k to n-1 (since i-k >=0)
        # Equivalent to reversed case: treat positive shift k_abs with indices swapped.
        # Compute using same logic with sign reversed? Let's derive.
        # sum_{i=k}^{n-1} h[i] * (1 - h[i - k]).
        # Let j = i - k => j from 0 to n-k-1.
        # Then term = h[j + k] * (1 - h[j]).
        # Derivative w.r.t h[j]:
        #   contribution from term where it's (1 - h[j]), coefficient -h[j + k].
        #   also from term where it's factor h[j] when i=j (if k==0) but not here.
        # Derivative w.r.t h[j + k]:
        #   contribution (1 - h[j]).
        # So g[j] += -h[j + k]   (note h index j+k)
        #    g[j + k] += 1 - h[j]
        # That is symmetric to k positive but with signs swapped w.r.t position.
        # Equivalent to:
        #   g[k:] += 1 - h[:n - k]   (since j in [0, n-k-1] -> j+k = k..n-1)
        #   g[:n - k] += -h[k:]     (since j = 0..n-k-1)
        # Wait we must careful: for k negative, k_abs = -k > 0.
        # So:
        #   g[k_abs:] (i.e., indices from k_abs to end) add (1 - h[:n - k_abs])
        #   g[:n - k_abs] add -h[k_abs:]
        # This is same as case k positive but swapped; implement accordingly.

        g[k_abs:] += 1 - h[:n - k_abs]
        g[:n - k_abs] += -h[k_abs:]

def worst_shift_gradient_descent(h, rng, target, max_iter=200, step0=0.5, decay=0.95):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    step = step0
    for it in range(max_iter):
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_val = np.max(corr)
        ks = np.where(corr == max_val)[0] - (n - 1)
        g = np.zeros_like(cur_h)

def repair_sum(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            # can add up to capacity
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])  # diff negative, -h[i] is negative or zero
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

CURRENT_TARGET = None

def _sum_eq_constraint(x):

    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr

    return np.max(corr) * (2.0 / h.size)


def grad_for_shift(h, k):
    """Gradient of the overlap for a fixed shift k (k may be negative)."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        g[: n - k] += 1 - h[k:]
        g[k:] += -h[: n - k]
    else:
        k = -k
        g[k:] += 1 - h[: n - k]
        g[: n - k] += -h[k:]
    return g


def _repair_sum(v, target, rng, max_iter=2000):
    """Adjust v (already in [0,1]) so that sum(v) == target."""
    diff = target - v.sum()
    n = v.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        if add == 0:
            continue
        v[i] += add
        diff = target - v.sum()
    return v


def random_feasible(n, rng):
    """Generate a random feasible vector (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    return _repair_sum(h, target, rng)


def hill_climb(h0, rng, max_iters=1500, start_step=0.5):
    """Simple stochastic hill‑climb that preserves the sum."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    n = h.size
    target = n / 2.0

    for it in range(max_iters):
        # choose two indices and transfer a random ε
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = (rng.random() - 0.5) * 2.0 * max_eps * step
                eps = max(-max_eps, min(eps, max_eps))
                h[i] -= eps
                h[j] += eps
        # enforce bounds and sum (cheap repair)
        h = np.clip(h, 0.0, 1.0)
        h = _repair_sum(h, target, rng, max_iter=500)

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP on the smooth soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return _repair_sum(x, CURRENT_TARGET, rng, max_iter=2000)


def worst_gradient_descent(h0, rng, max_iter=200, step0=0.5, decay=0.93):
    """Projected gradient descent targeting the current worst shift(s)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        ks = np.where(corr == max_val)[0] - (n - 1)

        g = np.zeros_like(h)
        for k in ks:
            g += grad_for_shift(h, k)

        if np.linalg.norm(g) == 0:
            break

        new_h = h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = _repair_sum(new_h, target, rng, max_iter=1000)

        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c <= compute_c5(h):
            h = new_h
        step *= decay
        if step < 1e-6:
            break
    return best_h, best_c


def upsample(h):
    """Duplicate each entry – keeps the integral exactly the same."""
    return np.repeat(h, 2)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Parent solution, if any
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _repair_sum(h0, n0 / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, n0

    # ------------------------------------------------------------------
    # Candidate discretisations (keep them modest)
    # ------------------------------------------------------------------
    candidate_ns = [120, 144, 168, 192, 240]
    # ensure parent size is present first
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() - start > budget_s * 0.94:
            break

        # how many restarts for this size
        restarts = 6 if n <= 144 else 4 if n <= 192 else 3

        for r in range(restarts):
            if time.time() - start > budget_s * 0.94:
                break

            # ---- initialise -------------------------------------------------
            if best_n == n and best_h is not None and r == 0:
                h = best_h.copy()
            else:
                if rng.random() < 0.5:
                    # binary start (exactly n/2 ones)
                    h = np.zeros(n)
                    idx = rng.choice(n, n // 2, replace=False)
                    h[idx] = 1.0
                else:
                    h = random_feasible(n, rng)

            # ---- stochastic hill‑climb --------------------------------------
            h, c = hill_climb(h, rng, max_iters=1500, start_step=0.5)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ---- SLSQP passes with increasing beta ---------------------------
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                if time.time() - start > budget_s * 0.96:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=250 if beta < 200 else 400)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ---- worst‑shift gradient descent --------------------------------
            if time.time() - start < budget_s * 0.97:
                h, c = worst_gradient_descent(h, rng, max_iter=200, step0=0.5)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ---- final short hill‑climb --------------------------------------
            if time.time() - start < budget_s * 0.98:
                h, c = hill_climb(h, rng, max_iters=800, start_step=0.3)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # Upsample the current best (if room) and polish further
    # ------------------------------------------------------------------
    if best_h is not None and best_n * 2 <= 240 and time.time() - start < budget_s * 0.99:
        h2 = upsample(best_h)
        n2 = h2.size
        # small hill‑climb then surrogate optimisation
        h2, c2 = hill_climb(h2, rng, max_iters=1200, start_step=0.4)
        for beta in (20.0, 80.0, 200.0, 500.0):
            if time.time() - start > budget_s:
                break
            try:
                h2 = slsqp_optimize(h2, beta, rng, max_iter=300)
            except Exception:
                pass
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_h, best_c, best_n = h2.copy(), c2, n2

    # ------------------------------------------------------------------
    # Final feasibility enforcement (just in case)
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, best_n / 2.0, rng, max_iter=5000)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
