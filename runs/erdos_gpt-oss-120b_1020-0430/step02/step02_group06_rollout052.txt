import time
import numpy as np

# -------------------- Helper functions -------------------- #

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a (possibly binary) step function."""
    n = h.shape[0]
    # use int16 to avoid overflow when h is integer
    corr = np.correlate(h.astype(np.int16), (1 - h).astype(np.int16), mode="full")
    return float(corr.max() * (2.0 / n))


def enforce_half_ones(h: np.ndarray) -> np.ndarray:
    """Given a binary array, flip bits to make exactly n/2 ones."""
    n = h.size
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        # too many ones – turn some to zero
        idx = np.where(h == 1)[0]
        to_flip = np.random.choice(idx, size=cur - target, replace=False)
        h[to_flip] = 0
    elif cur < target:
        # too few ones – turn some zeros to one
        idx = np.where(h == 0)[0]
        to_flip = np.random.choice(idx, size=target - cur, replace=False)
        h[to_flip] = 1
    return h


def random_binary_half(n: int) -> np.ndarray:
    """Generate a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    h[: n // 2] = 1
    np.random.shuffle(h)
    return h


def reduce_max_corr(
    h: np.ndarray,
    max_allowed: int,
    time_left,
    max_iter: int = 5000,
) -> np.ndarray:
    """
    Greedy descent: repeatedly swap a (1,0) pair that contributes to the
    current maximal overlap until the maximal count is ≤ max_allowed
    or time runs out.
    """
    n = h.shape[0]
    it = 0
    while time_left() > 0.001 and it < max_iter:
        corr = np.correlate(h.astype(np.int16), (1 - h).astype(np.int16), mode="full")
        cur_max = int(corr.max())
        if cur_max <= max_allowed:
            break

        shift_idx = int(corr.argmax())
        d = shift_idx - (n - 1)  # signed shift amount

        if d >= 0:
            cand = np.where((h[: n - d] == 1) & (h[d:] == 0))[0]
        else:
            d_abs = -d
            cand = np.where((h[d_abs:] == 1) & (h[: n - d_abs] == 0))[0] + d_abs

        if cand.size:
            i1 = np.random.choice(cand)
            i2 = i1 + d
        else:
            # fallback: any opposite bits
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i1 = np.random.choice(ones)
            i2 = np.random.choice(zeros)

        # swap
        h[i1], h[i2] = h[i2], h[i1]
        it += 1
    return h


# -------------------- Main optimisation driver -------------------- #

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimises
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    target_c5 = 0.3808  # early‑stop target

    # collect candidate grid sizes
    n_candidates = [96, 112, 128, 144, 160, 176, 192, 224, 256]
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        if parent_h.size % 2 == 0:
            n_candidates.append(parent_h.size)
    except NameError:
        parent_h = None

    n_candidates = sorted(set([n for n in n_candidates if n % 2 == 0]))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in n_candidates:
        if time_left() < 5.0:
            break

        max_allowed = int(np.floor(target_c5 * n / 2.0 + 1e-9))

        # decide how many restarts we can afford for this n
        # each restart (including reduction) typically costs <0.5 s for n≤256
        max_restarts = max(5, int(time_left() // 0.7))

        for _ in range(max_restarts):
            if time_left() < 4.0:
                break

            # initialise h
            if parent_h is not None and parent_h.size == n:
                h = (parent_h > 0.5).astype(np.int8)
                h = enforce_half_ones(h)
            else:
                h = random_binary_half(n)

            # greedy improvement
            h = reduce_max_corr(h, max_allowed, time_left)

            c5 = compute_c5(h.astype(float))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.astype(float).copy()
                best_n = n
                if best_c5 < target_c5:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # fallback – simple uniform construction if nothing better was found
    if best_h is None:
        n = 96
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
