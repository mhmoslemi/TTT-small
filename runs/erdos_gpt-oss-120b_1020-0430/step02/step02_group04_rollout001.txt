import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------
# Basic utilities
# -------------------------------------------------------
def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -1e9, 1e9
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    v = np.clip(v, 0.0, 1.0)
    return project_onto_box_sum(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a small random amount from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb accepting improving moves (occasionally uphill)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        else:
            # occasional uphill acceptance
            if rng.random() < 0.001:
                h = new_h
        # gradually shrink step size
        if (it + 1) % max(1, max_iters // 10) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# -------------------------------------------------------
# Smooth‑max gradient descent
# -------------------------------------------------------
def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    step = 0.5
    for beta in beta_sched:
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)                     # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()                              # soft‑max weights

            # gradient of the smooth max (raw, before scaling)
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    # h_i interacts with h_{i+k}
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)

            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # occasional uphill move to escape shallow traps
                if rng.random() < 0.005:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# -------------------------------------------------------
# SLSQP refinement (soft‑max surrogate)
# -------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using a smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

# -------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- initialise ----------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        # Resample to a convenient size (>=200) if needed
        target_n = max(200, n0)
        if n0 != target_n:
            x_old = np.linspace(0.0, 2.0, n0)
            x_new = np.linspace(0.0, 2.0, target_n)
            h0 = np.interp(x_new, x_old, h0)
        n = target_n
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 200
        h0 = random_feasible(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------- quick local improvement ----------
    if time.time() - start < budget_s * 0.15:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2500, start_step=0.5)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---------- main stochastic search ----------
    beta_sched = [5, 20, 80, 200, 500, 1000]
    steps_per_beta = 300

    while time.time() - start < budget_s * 0.70:
        # generate a start point – mostly random, sometimes blended with best
        if rng.random() < 0.2:
            h_start = (random_feasible(n, rng) + best_h) * 0.5
        else:
            h_start = random_feasible(n, rng)

        # smooth‑max gradient descent
        h_opt, c_opt = grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng)

        # short hill‑climb refinement
        h_opt2, c_opt2 = hill_climb(h_opt, rng, max_iters=1500, start_step=0.3)
        if c_opt2 < c_opt:
            h_opt, c_opt = h_opt2, c_opt2

        # occasional SLSQP polishing
        if rng.random() < 0.25:
            try:
                h_opt = slsqp_optimize(h_opt, beta=800, max_iter=250)
            except Exception:
                pass
            c_opt = compute_c5(h_opt)

        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---------- up‑sampling phase ----------
    if time.time() - start < budget_s * 0.94 and n * 2 <= 300:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up = project_onto_box_sum(h_up, n_up / 2.0)
        # refine upsampled version
        h_up, c_up = grad_descent_optimize(h_up, beta_sched, 200, rng)
        h_up, c_up = hill_climb(h_up, rng, max_iters=1200, start_step=0.3)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ---------- final clean‑up ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
