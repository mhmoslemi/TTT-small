def enforce_sum(x, target, rng=None):
    # fast projection onto box and sum
    low = -1.0
    high = 1.0
    for _ in range(40):
        mid = (low + high) / 2
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high)/2, 0.0, 1.0)
    # due to numeric tolerance maybe sum not exactly target, adjust small remainder
    diff = target - y.sum()
    # distribute diff by adding to entries that are not at bound
    if abs(diff) > 1e-12:
        # find non-bound indices (0<y<1)
        mask = (y > 1e-12) & (y < 1 - 1e-12)
        if mask.any():
            # add equally to these
            add = diff / mask.sum()
            y[mask] += add
            y = np.clip(y, 0.0, 1.0)
        else:
            # fallback simple adjust first index
            i = rng.integers(0, len(x))
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y
```

def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.2):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = random_move(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def enforce_sum(x, target, rng):
    """Project x onto [0,1] box with exact sum = target (binary simplex)."""
    low, high = -1.0, 1.0
    for _ in range(40):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            inc = diff / mask.sum()
            y[mask] += inc
            y = np.clip(y, 0.0, 1.0)
        else:
            i = rng.integers(0, len(x))
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y

def random_feasible_vector(n, rng):
    """Draw a random h with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)

# ----------------------------------------------------------------------
# Fast C5 evaluation (FFT based)
# ----------------------------------------------------------------------
def fast_c5(h):
    n = h.size
    fa = np.fft.rfft(h, 2 * n)
    fb = np.fft.rfft(1.0 - h, 2 * n)
    corr = np.fft.irfft(fa * np.conj(fb), 2 * n)
    return corr[:2 * n - 1].max() * (2.0 / n)

def exact_c5(h):
    """Exact (np.correlate) evaluation used for the final reported bound."""
    return np.correlate(h, 1 - h, mode="full").max() * (2.0 / h.size)

# ----------------------------------------------------------------------
# Soft‑max surrogate (FFT based)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def full_corr(x):
    n = x.size
    fa = np.fft.rfft(x, 2 * n)
    fb = np.fft.rfft(1.0 - x, 2 * n)
    corr = np.fft.irfft(fa * np.conj(fb), 2 * n)
    return corr[:2 * n - 1]

def softmax_obj(x):
    corr = full_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair tiny sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ----------------------------------------------------------------------
# Local moves
# ----------------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def random_move(h, rng, scale=0.2):
    if rng.random() < 0.5:
        return perturb_h(h, rng, scale)
    # swap two entries (preserves sum)
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    nh = h.copy()
    nh[i], nh[j] = nh[j], nh[i]
    return nh

# ----------------------------------------------------------------------
# Local optimisation primitives
# ----------------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.2):
    h = h_start.copy()
    best_h = h.copy()
    best_c = fast_c5(h)
    step = start_step
    for it in range(max_iters):
        nh = random_move(h, rng, scale=step)
        c = fast_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.8, 1e-4)
    return best_h, best_c

def simulated_annealing(h_start, rng, max_iters=2000,
                        init_temp=0.02, decay=0.999, scale=0.2):
    h = h_start.copy()
    cur_c = fast_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = init_temp
    for it in range(max_iters):
        nh = random_move(h, rng, scale=scale)
        nc = fast_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()
        if nc < cur_c or rng.random() < np.exp((cur_c - nc) / temp):
            h, cur_c = nh, nc
        temp *= decay
    return best_h, best_c

def quick_refine(h, rng):
    best_h, best_c = h.copy(), fast_c5(h)
    for beta in (5.0, 20.0, 80.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=200)

    trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c

def deep_refine_timebound(h, rng, deadline):
    """Intensive refinement until deadline."""
    best_h, best_c = h.copy(), fast_c5(h)
    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    bi = 0
    while time.time() < deadline:
        beta = betas[bi % len(betas)]
        bi += 1
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=400)
        except Exception:
            continue
        c = fast_c5(trial)
        if c < best_c:
            best_h, best_c = trial, c
        # hill‑climb
        remain = deadline - time.time()
        if remain <= 0:
            break
        hc_iters = int(min(5000, max(800, remain * 15)))
        trial_h, c_h = hill_climb(best_h, rng, max_iters=hc_iters, start_step=0.3)
        if c_h < best_c:
            best_h, best_c = trial_h, c_h
        remain = deadline - time.time()
        if remain <= 0:
            break
        # simulated annealing
        sa_iters = int(min(15000, max(3000, remain * 30)))
        trial_sa, c_sa = simulated_annealing(best_h, rng,
                                            max_iters=sa_iters,
                                            init_temp=0.05,
                                            decay=0.9995,
                                            scale=0.2)
        if c_sa < best_c:
            best_h, best_c = trial_sa, c_sa
    return best_h, best_c

def upsample_h(h, factor, rng):
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target = (n * factor) / 2.0
    return enforce_sum(h_up, target, rng)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n / 2.0, rng)

    best_h, best_c = quick_refine(h0, rng)

    # ------------------------------------------------------------------
    # Phase A – many random restarts (≈30% of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start + budget_s * 0.30
    while time.time() < phase_a_deadline:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase B – upsample to 2× and deep refine (≈50% of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        dline = start + budget_s * 0.80
        best_h, best_c = deep_refine_timebound(h_up, rng, dline)

    if time.time() < deadline - 5 and n * 2 <= 2000:

    # Final tidy‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c = exact_c5(best_h)  # use exact evaluation for the reported bound
