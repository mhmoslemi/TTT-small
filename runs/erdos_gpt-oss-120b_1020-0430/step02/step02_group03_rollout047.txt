def repair(h, target, rng, max_iter=2000):
    # Clip to [0,1] then adjust sum by moving mass from random indices with spare capacity.
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = len(h)
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Choose random index that can increase if diff>0, else decrease.
        if diff > 0:
            # Choose index with h[i] < 1.
            idx = rng.integers(0,n)
            if h[idx] >= 1.0:
                continue
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.integers(0,n)
            if h[idx] <= 0.0:
                continue
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    # After loop, adjust any residual small diff by distributing to all indices proportionally?
    # Not needed if diff small, we can adjust first index.
    if abs(diff) > 1e-9:
        # Fix by adjusting one index within bounds.
        i = rng.integers(0,n)
        if diff > 0:
            h[i] = min(1.0, h[i] + diff)
        else:
            h[i] = max(0.0, h[i] + diff) # diff negative
        # reclip
        h = np.clip(h,0.0,1.0)
        # final check:
        # If still sum off due to clipping, we ignore; error will be small.
    return h

def propose_move(h, rng, scale=0.1):
    # Choose two distinct indices i,j.
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # Transfer epsilon from i to j.
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 1e-12:
        # try reverse direction
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:
            return h.copy()
        # transfer from j to i
        eps = rng.random() * max_transfer * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h
    eps = rng.random() * max_transfer * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# globals for surrogate optimisation
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp in a numerically stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Project h onto [0,1] with Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:  # need to increase sum
            i = rng.integers(0, n)
            if h[i] >= 1.0:
                continue
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:  # need to decrease sum
            i = rng.integers(0, n)
            if h[i] <= 0.0:
                continue
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    # Small residual fix by uniform scaling (rare)
    if abs(diff) > 1e-9:
        total = h.sum()
        if total > 0:
            h *= target / total
        else:
            h = np.full_like(h, target / n)
        h = np.clip(h, 0.0, 1.0)
        # final exact correction on a random index
        i = rng.integers(0, n)
        remaining = target - h.sum()
        h[i] = np.clip(h[i] + remaining, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Create a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)  # uniform in [0,1)
    target = n / 2.0
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        # try opposite direction
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0.0:
            return h.copy()
        eps = rng.random() * max_transfer * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h
    eps = rng.random() * max_transfer * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple deterministic hill‑climb (only accept improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h via SLSQP on the smoothed surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # Repair possible tiny sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

    return np.interp(xs_up, xs, h)
