def project_feasible(h, target_sum, tol=1e-9, max_iter=1000):
        h = np.clip(h, 0, 1)
        # compute diff
        diff = h.sum() - target_sum
        # if diff small return
        for _ in range(max_iter):
            if abs(diff) <= tol:
                break
            # Determine eligible indices (those not at bounds or need to adjust)
            if diff > 0:
                # need to reduce sum
                free = h > 0
                if not np.any(free): break
                delta = diff / free.sum()
                h[free] -= delta
                # clip
                h = np.clip(h, 0, 1)
            else:
                # need to increase sum
                free = h < 1
                if not np.any(free): break
                delta = diff / free.sum()
                h[free] -= delta  # diff negative, so adding
                h = np.clip(h, 0, 1)
            diff = h.sum() - target_sum
        # final tweak:
        residual = target_sum - h.sum()
        if abs(residual) > 1e-12:
            # adjust small residual by adding to first non-full or subtract from first non-zero
            idx = np.where((residual > 0) & (h < 1))[0]
            if idx.size>0:
                add = min(residual, 1 - h[idx[0]])
                h[idx[0]] += add
                residual -= add
            else:
                idx = np.where((residual < 0) & (h>0))[0]
                if idx.size>0:
                    sub = min(-residual, h[idx[0]])
                    h[idx[0]] -= sub
                    residual += sub
        return h

def _project_to_feasible(h, target_sum, tol=1e-9, max_iter=500):
    # Clip and adjust sum with water-filling approach
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / free.sum()
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target_sum
    # small residual fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        idx = np.where((residual > 0) & (h < 1))[0]
        if idx.size:
            add = min(residual, 1 - h[idx[0]])
            h[idx[0]] += add
            residual -= add
        else:
            idx = np.where((residual < 0) & (h > 0))[0]
            if idx.size:
                sub = min(-residual, h[idx[0]])
                h[idx[0]] -= sub
                residual += sub
    return np.clip(h, 0.0, 1.0)

    else:

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Feasibility projection: clip to [0,1] and enforce sum = target_sum
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, target_sum: float,
                         tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = h.sum() - target_sum
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0.0
        else:
            free = h < 1.0
        if not np.any(free):
            break
        delta = diff / free.sum()
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where((h < 1.0))[0]
            if idx.size:
                add = min(residual, 1.0 - h[idx[0]])
                h[idx[0]] += add
                residual -= add
        else:
            idx = np.where((h > 0.0))[0]
            if idx.size:
                sub = min(-residual, h[idx[0]])
                h[idx[0]] -= sub
                residual += sub
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation (max correlation * 2/n)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Zero‑padding shift (no periodic wrap‑around)
# ----------------------------------------------------------------------
def _shift(arr: np.ndarray, k: int) -> np.ndarray:
    n = arr.size
    if k > 0:
        out = np.empty_like(arr)
        out[:k] = 0.0
        out[k:] = arr[: n - k]
    elif k < 0:
        k = -k
        out = np.empty_like(arr)
        out[-k:] = 0.0
        out[: n - k] = arr[k:]
    else:
        out = arr.copy()
    return out


# ----------------------------------------------------------------------
# Simple gradient‑descent on the max‑correlation objective
# ----------------------------------------------------------------------
def _gradient_descent(start_h: np.ndarray,
                     time_left,
                     target_c5: float = 0.3808):
    h = start_h.copy()
    n = h.size
    target_sum = n / 2.0

    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    lr = 0.2               # initial step size
    decay = 0.9995
    min_lr = 0.001
    it = 0
    no_improve = 0

    while time_left() > 0.3 and lr >= min_lr and it < 20000:
        # correlation and shift giving the current maximum
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)          # signed shift
        c5_val = corr[max_idx] * (2.0 / n)

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_c5 <= target_c5:
            break

        # gradient of the max‑correlation term (sub‑gradient)
        h_plus = _shift(h, -k)   # h_{i+k}
        h_minus = _shift(h, k)   # h_{i−k}
        grad = 1.0 - (h_plus + h_minus)

        # descent step + feasibility projection
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_feasible(h_new, target_sum)

        # occasional random kick to escape plateaus
        if no_improve >= 200 and np.random.rand() < 0.3:
            noise = 0.05 * np.random.randn(n)
            h_new = np.clip(h_new + noise, 0.0, 1.0)
            h_new = _project_to_feasible(h_new, target_sum)

        h = h_new
        lr *= decay
        it += 1

    return best_h, best_c5


# ----------------------------------------------------------------------
# Local refinement with SLSQP (same formulation as the reference code)
# ----------------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 400):
    n = initial_h.size
    target_sum = n / 2.0
    h0 = _project_to_feasible(initial_h, target_sum)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

        return np.sum(x[:n]) - target_sum

        return x[-1]

    cons = [{"type": "ineq", "fun": _ineq},

    h_opt = res.x[:n] if res.success else h0
