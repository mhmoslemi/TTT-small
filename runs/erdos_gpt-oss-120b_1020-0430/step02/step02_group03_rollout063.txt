```
def targeted_perturb_h(h, rng, scale=0.5):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr) - (n - 1)
    # contributions for shift k_max:
    i_vals = np.arange(n)
    target_indices = i_vals + k_max
    # mask where target indices in [0,n)
    mask = (target_indices >= 0) & (target_indices < n)
    contributions = np.zeros(n)
    contributions[mask] = h[i_vals[mask]] * (1 - h[target_indices[mask]])
    if not np.any(mask):
        # fallback to random perturb
        return perturb_h(h, rng, scale)
    i = np.argmax(contributions)  # index with largest contribution
    j = i + k_max
    # ensure j in range
    if j < 0 or j >= n:
        return perturb_h(h, rng, scale)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def targeted_perturb_h(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr) - (n - 1)
    if k_max == 0:
        # shift zero means product h[i]*(1-h[i]), contributions largest where h[i] ~ 0.5
        # fallback to random perturb
        return perturb_h(h, rng, scale)
    i_vals = np.arange(n)
    j_vals = i_vals + k_max
    mask = (j_vals >= 0) & (j_vals < n)
    contributions = np.zeros(n)
    contributions[mask] = h[i_vals[mask]] * (1 - h[j_vals[mask]])
    if not np.any(mask):
        return perturb_h(h, rng, scale)
    i = np.argmax(contributions)
    j = i + k_max
    # ensure j valid
    if not (0 <= j < n):
        return perturb_h(h, rng, scale)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # ensure no out of bounds through floating rounding
    new_h[i] = max(0.0, min(1.0, new_h[i]))
    new_h[j] = max(0.0, min(1.0, new_h[j]))
    return new_h
```

```python
import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # simple waterfill-style correction
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb_h(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr) - (n - 1)
    if k_max == 0:
        return perturb_h(h, rng, scale)
    i_vals = np.arange(n)
    j_vals = i_vals + k_max
    mask = (j_vals >= 0) & (j_vals < n)
    if not np.any(mask):
        return perturb_h(h, rng, scale)
    contributions = np.zeros(n)
    contributions[mask] = h[i_vals[mask]] * (1 - h[j_vals[mask]])
    i = np.argmax(contributions)
    j = i + k_max
    if not (0 <= j < n):
        return perturb_h(h, rng, scale)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    TOTAL = budget_s

    # ----- initial feasible vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n / 2.0
        h0 = enforce_sum(h0, target_sum, rng)
    else:
        n = 180
        target_sum = n / 2.0
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, target_sum, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- binary variant from current best -----
    # create exact binary vector with n/2 ones
    binary_h = np.zeros(n, dtype=int)
    idx_one = np.argpartition(-best_h, n // 2)[: n // 2]
    binary_h[idx_one] = 1
    binary_h = enforce_sum(binary_h.astype(float), target_sum, rng).astype(int)
    binary_c = compute_c5(binary_h.astype(float))
    if binary_c < best_c:
        best_h = binary_h.astype(float)
        best_c = binary_c

    # ----- binary hill climbing -----
    is_one = binary_h.astype(bool)
    binary_deadline = start_time + TOTAL * 0.25
    TARGET = 0.3808
    while time.time() < binary_deadline:
        # pick a random 1 and a random 0
        ones_idx = np.flatnonzero(is_one)
        zeros_idx = np.flatnonzero(~is_one)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i_one = rng.choice(ones_idx)
        i_zero = rng.choice(zeros_idx)
        new_is_one = is_one.copy()
        new_is_one[i_one] = False
        new_is_one[i_zero] = True
        new_h = new_is_one.astype(float)

import time

from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb_h(h, rng, scale=0.2):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr) - (n - 1)
    if k_max == 0:
        return perturb_h(h, rng, scale)
    i_vals = np.arange(n)
    j_vals = i_vals + k_max
    mask = (j_vals >= 0) & (j_vals < n)
    if not np.any(mask):
        return perturb_h(h, rng, scale)
    contributions = np.zeros(n)
    contributions[mask] = h[i_vals[mask]] * (1 - h[j_vals[mask]])
    i = np.argmax(contributions)
    j = i + k_max
    if not (0 <= j < n):
        return perturb_h(h, rng, scale)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
