def gradient_descent_smoothmax(h0, betas, steps_per_beta, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in betas:
        step = 0.5  # initial step size per beta
        for s in range(steps_per_beta):
            # compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            C = corr * (2.0 / n)  # scaled overlaps
            bc = beta * C
            # log-sum-exp weights
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
                w_sum = 1.0
            w /= w_sum
            # compute gradient via convolution
            w_rev = w[::-1]
            conv1 = np.convolve(w_rev, 1 - h, mode='full')
            conv2 = np.convolve(w, h, mode='full')
            mid = n - 1
            grad = conv1[mid:mid+n] - conv2[mid:mid+n]
            grad *= (2.0 / n)
            # step
            new_h = h - step * grad
            # projection
            new_h = project_onto_box_sum(new_h, n/2.0)
            # compute new C5
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
                # maybe increase step? not needed
            else:
                # occasionally accept uphill move
                if rng.random() < 0.01:
                    h = new_h
            # decay step
            step *= 0.9995
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step function h on [0,2] with dx=2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Generate a random feasible h (0<=h<=1, sum=n/2)."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def upsample_h(h_old, n_new):
    """Linear interpolation of h from its current size to n_new points on [0,2)."""
    n_old = h_old.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


# ---------- gradient descent on smooth max ----------
def grad_descent_smoothmax(h0, betas, steps_per_beta, rng):
    """Projected gradient descent on log-sum-exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in betas:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)
            bc = beta * C
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w_sum = w.sum()
            if w_sum == 0:
                w[:] = 1.0 / w.size
                w_sum = 1.0
            w /= w_sum
            # gradient via convolution
            w_rev = w[::-1]
            conv1 = np.convolve(w_rev, 1 - h, mode="full")
            conv2 = np.convolve(w, h, mode="full")
            mid = n - 1
            grad = (conv1[mid:mid + n] - conv2[mid:mid + n]) * (2.0 / n)
            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c, best_h = c_new, new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
        # occasional jitter to escape plateaus
        if rng.random() < 0.15:
            h += rng.normal(scale=0.001, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_sum(h, n / 2.0)
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, n / 2.0)


# ---------- simple hill‑climb (optional) ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
