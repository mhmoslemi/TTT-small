def binary_sa(h, rng, max_iters, init_temp=0.1, decay=0.9999):
    # h: binary array (np.uint8) of shape (n,)
    n = h.size
    pad = n - 1
    idx = np.arange(2*n-1)
    two_pad = 2*pad
    padded_h = np.pad(h.astype(np.int64), (pad, pad), constant_values=0)
    padded_c = np.pad((1 - h).astype(np.int64), (pad, pad), constant_values=0)
    O = np.correlate(h.astype(np.int64), (1 - h).astype(np.int64), mode='full')
    cur_max = O.max()
    best_h = h.copy()
    best_max = cur_max
    best_O = O.copy()
    temp = init_temp
    for it in range(max_iters):
        # pick random swap i (h=1) and j (h=0)
        # we can pick random i, then find j such that h[j]==0; while loops
        # to reduce loops, we can sample i,j and skip if same.
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1, j is 0 for delta formula consistency (i is currently 1, j is 0)
        if h[i] == 0:
            i, j = j, i
        # compute delta
        delta_i = -padded_c[i + idx] + padded_h[two_pad + i - idx]
        delta_j = padded_c[j + idx] - padded_h[two_pad + j - idx]
        delta = delta_i + delta_j
        new_max = (O + delta).max()
        # acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # SA acceptance probability
            if rng.random() < np.exp(-(new_max - cur_max) / temp):
                accept = True
        if accept:
            # apply swap
            h[i] = 0
            h[j] = 1
            # update padded arrays
            padded_h[pad + i] = 0
            padded_h[pad + j] = 1
            padded_c[pad + i] = 1
            padded_c[pad + j] = 0
            # update O
            O = O + delta
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                best_O = O.copy()
        # decay temperature
        temp *= decay
        # maybe break early if time exceeded (we will check time outside)
    return best_h, best_max, best_O

def binary_hill_climb_time(h_start, rng, time_budget):
    h = h_start.copy()
    n = h.size
    pad = n - 1
    idx = np.arange(2 * n - 1)
    two_pad = 2 * pad
    padded_h = np.pad(h.astype(np.int64), (pad, pad), constant_values=0)
    padded_c = np.pad((1 - h).astype(np.int64), (pad, pad), constant_values=0)
    O = np.correlate(h.astype(np.int64), (1 - h).astype(np.int64), mode='full')
    cur_max = int(O.max())
    best_h = h.copy()
    best_max = cur_max
    # Simulated annealing parameters
    init_temp = cur_max * 0.1 + 0.01
    temp = init_temp
    decay = 0.9995
    start = time.time()
    # iteration loop
    while time.time() - start < time_budget:
        # propose swap
        i = rng.integers(0, n)
        if h[i] == 0:
            # need i with 1
            # find a 1 index
            # we could sample until find
            while True:
                i = rng.integers(0, n)
                if h[i] == 1:
                    break
        j = rng.integers(0, n)
        if h[j] == 1:
            while True:
                j = rng.integers(0, n)
                if h[j] == 0:
                    break
        # compute delta using vectorized indexing
        delta_i = -padded_c[i + idx] + padded_h[two_pad + i - idx]
        delta_j = padded_c[j + idx] - padded_h[two_pad + j - idx]
        delta = delta_i + delta_j
        new_max = int((O + delta).max())
        # acceptance
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / temp):
            # apply swap
            h[i] = 0
            h[j] = 1
            # update padded arrays
            padded_h[pad + i] = 0
            padded_h[pad + j] = 1
            padded_c[pad + i] = 1
            padded_c[pad + j] = 0
            # update O
            O = O + delta
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        temp *= decay
    return best_h, best_max

def binary_hill_climb_time(h_start, rng, time_budget):
    h = h_start.copy()
    n = h.size
    pad = n - 1
    idx = np.arange(2*n - 1)
    two_pad = 2*pad
    padded_h = np.pad(h.astype(np.int64), (pad, pad), constant_values=0)
    padded_c = np.pad((1 - h).astype(np.int64), (pad, pad), constant_values=0)
    O = np.correlate(h.astype(np.int64), (1 - h).astype(np.int64), mode='full')
    cur_max = int(O.max())
    best_h = h.copy()
    best_max = cur_max
    # SA parameters
    temp = cur_max * 0.1 + 0.01
    decay = 0.9995
    start = time.time()
    while time.time() - start < time_budget:
        # propose swap
        i = rng.integers(0, n)
        # Ensure i is 1
        if h[i] == 0:
            # find one
            ones_idx = np.where(h == 1)[0]
            if len(ones_idx) == 0:
                continue
            i = rng.choice(ones_idx)
        j = rng.integers(0, n)
        if h[j] == 1:
            zeros_idx = np.where(h == 0)[0]
            if len(zeros_idx) == 0:
                continue
            j = rng.choice(zeros_idx)
        # compute delta
        delta = -padded_c[i + idx] + padded_c[j + idx] + padded_h[two_pad + i - idx] - padded_h[two_pad + j - idx]
        # corrections
        delta[pad] -= 2
        delta[i - j + pad] += 1
        delta[j - i + pad] += 1
        # compute new_max
        new_O = O + delta
        new_max = int(new_O.max())
        # acceptance
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / temp):
            # accept swap, update structures
            # swap bits
            h[i] = 0
            h[j] = 1
            # update padded arrays
            padded_h[pad + i] = 0
            padded_h[pad + j] = 1
            padded_c[pad + i] = 1
            padded_c[pad + j] = 0
            # update O and cur_max
            O = new_O
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        temp *= decay
    return best_h, best_max

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities for binary hill‑climbing with exact O(n) updates
# ----------------------------------------------------------------------
def _prepare_state(h):
    """Return padded arrays and pre‑computed indices for a binary vector."""
    n = h.size
    pad = n - 1
    idx = np.arange(2 * n - 1)               # index -> shift k = idx - pad
    two_pad = 2 * pad
    # pad with zeros on both sides (length = 3n‑2)
    ph = np.pad(h.astype(np.int64), (pad, pad), constant_values=0)
    pc = np.pad((1 - h).astype(np.int64), (pad, pad), constant_values=0)
    # initial correlation O_k = sum_i h_i * (1-h_{i+k})
    O = np.correlate(h.astype(np.int64), (1 - h).astype(np.int64), mode="full")
    return ph, pc, O, idx, pad, two_pad


def _delta_swap(ph, pc, i, j, idx, pad, two_pad):
    """
    Compute the exact change of the full correlation after swapping a 1 at i
    with a 0 at j (i!=j).  Returned array has length 2n‑1 and dtype int64.
    """
    # baseline delta (independent changes)
    delta = (
        -pc[i + idx]
        + pc[j + idx]
        + ph[two_pad + i - idx]
        - ph[two_pad + j - idx]
    )
    # corrections for overlapping terms:
    delta[pad] -= 2                     # k = 0
    delta[i - j + pad] += 1             # k = i - j
    delta[j - i + pad] += 1             # k = j - i
    return delta


def binary_hill_climb_time(h_start, rng, time_budget):
    """
    Simulated annealing on a binary step function.
    Returns the best binary vector seen and its maximal overlap count.
    """
    h = h_start.copy()
    n = h.size
    ph, pc, O, idx, pad, two_pad = _prepare_state(h)
    cur_max = int(O.max())
    best_h = h.copy()
    best_max = cur_max

    # annealing schedule
    temp = cur_max * 0.1 + 0.01
    decay = 0.9995
    start = time.time()

    # fast checks for existence of 1/0 positions
    while time.time() - start < time_budget:
        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        if h[i] == 0:
            # find a 1
            ones = np.flatnonzero(h)
            if ones.size == 0:
                continue
            i = int(rng.choice(ones))
        j = rng.integers(0, n)
        if h[j] == 1:
            zeros = np.flatnonzero(h == 0)
            if zeros.size == 0:
                continue
            j = int(rng.choice(zeros))

        delta = _delta_swap(ph, pc, i, j, idx, pad, two_pad)
        new_O = O + delta
        new_max = int(new_O.max())

        accept = False
        if new_max < cur_max:
            accept = True
        elif rng.random() < np.exp(-(new_max - cur_max) / temp):
            accept = True

        if accept:
            # perform the swap
            h[i] = 0
            h[j] = 1
            ph[pad + i] = 0
            ph[pad + j] = 1
            pc[pad + i] = 1
            pc[pad + j] = 0
            O = new_O
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        temp *= decay
    return best_h, best_max


def random_binary(n, rng):
    """Generate a random binary vector of length n with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=np.uint8)
    ones = rng.choice(n, half, replace=False)
    h[ones] = 1
    return h


def upsample_binary(h, factor):
    """Integer up‑sampling by repetition (preserves exact 0/1 values)."""
    return np.repeat(h, factor)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős C5 problem.
    Returns (list_of_h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---------- initialise ------------------------------------------------
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.size
        if n % 2 == 1:                     # enforce even length for binary
            n -= 1
            init_arr = init_arr[:n]
        half = n // 2
        # top‑half thresholding to obtain a binary start
        order = np.argsort(-init_arr)
        h0 = np.zeros(n, dtype=np.uint8)
        h0[order[:half]] = 1
    else:
        n = 180
        if n % 2:
            n += 1
        h0 = random_binary(n, rng)

    # ---------- primary hill‑climb --------------------------------------
    time_left = lambda: budget_s - (time.time() - start_time)
    # allocate ~30 % of the budget for the first intensive climb
    primary_time = min(time_left() * 0.3, 30.0)
    best_h, best_max = binary_hill_climb_time(h0, rng, primary_time)

    # ---------- random restarts (short climbs) --------------------------
    # use remaining budget until ~60 % of total time is spent
    while time_left() > budget_s * 0.4:
        h_rand = random_binary(n, rng)
        # very short climb, a few seconds each
        t_restart = min(5.0, time_left() * 0.05)
        h_opt, max_opt = binary_hill_climb_time(h_rand, rng, t_restart)
        if max_opt < best_max:
            best_h, best_max = h_opt, max_opt

    # ---------- up‑sample & deep refinement -----------------------------
    # try to double resolution if time permits
    while time_left() > budget_s * 0.2 and best_h.size * 2 <= 2000:
        h_up = upsample_binary(best_h, 2)
        n_up = h_up.size
        # allocate a chunk of remaining time for this level
        up_time = min(time_left() * 0.4, 60.0)
        h_up_opt, max_up = binary_hill_climb_time(h_up, rng, up_time)
        if max_up < best_max:
            best_h, best_max = h_up_opt, max_up
        else:
            # if no improvement, break to avoid wasteful up‑sampling
            break

    # ---------- final bound ---------------------------------------------
    final_c = best_max * (2.0 / best_h.size)
    h_list = best_h.astype(float).tolist()
    return h_list, float(final_c), int(best_h.size)
