def simulated_annealing(h, rng, max_iters, start_temp, end_temp):
    n = h.size
    temp = start_temp
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    for t in range(max_iters):
        # reduce temp linearly
        temp = start_temp * (1 - t / max_iters) + end_temp * (t / max_iters)
        i = rng.integers(0,n)
        j = rng.integers(0,n-1)
        if j >= i: j+=1
        # swap if bits differ or do epsilon if continuous?
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        c_new = compute_c5(cur_h)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# Globals for the smooth surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_i Σ_j x_j (1 - x_{j+i}) using log-sum-exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp, numerically stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iters=5000):
    """
    Adjust h so that Σh = target while respecting 0≤h≤1.
    Simple random walk that moves mass between entries.
    """
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iters):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n, rng):
    """Generate a binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)
    half = n // 2
    h[idx[:half]] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """
    Transfer a small amount ε from one coordinate to another (preserves sum).
    Returns a new vector; h is not modified.
    """
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_binary(h_start, rng, max_iters=2000):
    """
    Simple hill‑climb for binary vectors: swap a 0 and a 1 if it improves C5.
    Returns the best vector seen and its C5.
    """
    h = h_start.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < best_c:
            best_c = c
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return h, best_c


def hill_climb_float(h_start, rng, max_iters=2000, start_step=0.5):
    """
    Hill‑climb for continuous vectors using the epsilon‑transfer perturbation.
    Accepts only improving moves; step size decays.
    """
    h = h_start.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def is_binary_vec(h):
    """Return True if all entries are (approximately) 0 or 1."""
    return np.all((h <= 1e-9) | (h >= 1 - 1e-9))


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Main optimisation routine for the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # -----------------------------------------------------------------
    # 1. Base dimension and initial candidates
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180  # default even size
        h0 = random_binary_vector(n, rng).astype(float)

    candidates = [h0]

    # sinusoidal candidates
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    for f in (1, 2, 3):
        h = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))
        h = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(h, 0, 1), n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))

    # a few random binary vectors
    for _ in range(5):
        candidates.append(random_binary_vector(n, rng).astype(float))

    # -----------------------------------------------------------------
    # 2. Pick the best initial candidate
    # -----------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # -----------------------------------------------------------------
    # 3. Global search: hill‑climb + SLSQP ladder
    # -----------------------------------------------------------------
    beta_list = [5.0, 20.0, 80.0, 200.0, 500.0]

    while time.time() - start_time < budget_s * 0.85:
        # starting point: perturbed best or fresh random binary
        if rng.random() < 0.6:
            start = best_h.copy()
            if is_binary_vec(start):
                # a few random swaps to diversify
                for _ in range(3):
                    i = rng.integers(0, n)
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    start[i], start[j] = start[j], start[i]
            else:
                start = perturb_h(start, rng, scale=0.2)
        else:
            start = random_binary_vector(n, rng).astype(float)

        # local hill‑climb
        if is_binary_vec(start):
            h_local, c_local = hill_climb_binary(start, rng, max_iters=2500)
        else:
            h_local, c_local = hill_climb_float(start, rng, max_iters=2500, start_step=0.4)

        # update global best if needed
        if c_local < best_c:
            best_h, best_c = h_local.copy(), c_local

        # SLSQP refinement sequence
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.9:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

        # also try refining the fresh local solution
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.9:
                break
            try:
                h_ref = slsqp_optimize(h_local, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # -----------------------------------------------------------------
    # 4. Optional up‑sampling and fine‑tuning
    # -----------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.96 and n * 2 <= 300:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up = enforce_sum(h_up, n_up / 2.0, rng)
        # intensive hill‑climb on finer grid
        h_up, c_up = hill_climb_float(h_up, rng, max_iters=5000, start_step=0.2)
        # sequential SLSQP with increasing β
        for beta in [20, 80, 200, 500, 1000]:
            if time.time() - start_time > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, max_iter=600)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, n = h_up.copy(), c_up, n_up
        # final high‑β polish
        if time.time() - start_time < budget_s * 0.99:
            try:
                h_up = slsqp_optimize(best_h, beta=2000, max_iter=800)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, n = h_up.copy(), c_up, n
            except Exception:
                pass

    # -----------------------------------------------------------------
    # 5. Final cleaning
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
