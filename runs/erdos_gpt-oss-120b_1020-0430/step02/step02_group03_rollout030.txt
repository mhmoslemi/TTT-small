def enforce_sum(h, target):
    # Clip h to [0,1], then adjust sum via random adjustments
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    while abs(diff) > 1e-12:
        # Choose i where we can add or subtract
        if diff > 0:
            # need to add diff
            idx_possible = np.where(h < 1.0)[0]
            if idx_possible.size == 0:
                break
            i = rng.choice(idx_possible)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            idx_possible = np.where(h > 0.0)[0]
            if idx_possible.size == 0:
                break
            i = rng.choice(idx_possible)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h
```

def propose_move(h, rng, scale):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i: 
        j += 1
    # Transfer epsilon from i to j
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    # clip eps to ensure within bounds
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def simulated_annealing(h0, rng, time_limit):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    n = h0.size
    target_sum = n/2.0
    start_time = time.time()
    iter = 0
    T0 = 0.01
    T_min = 1e-6
    gamma = 0.9995  # decay factor per iteration.
    T = T0
    while time.time() - start_time < time_limit:
        iter += 1
        scale = max(0.2 * T / T0, 0.02) # scaling factor descending
        new_h = propose_move(cur_h, rng, scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_h = cur_h.copy()
                best_c = cur_c
        # Cooling schedule
        T = max(T * gamma, T_min)
    return best_h, best_c

def slsqp_refine(x0, beta, max_iter=200):
    n = len(x0)
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def fun(x):
        corr = np.correlate(x, 1 - x, mode='full')
        # use log-sum-exp
        # Using max may cause overflow; use scipy.special.logsumexp
        from scipy.special import logsumexp
        return logsumexp(beta * corr) / beta
    res = minimize(fun, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    x = enforce_sum(x, target)
    return x

def enforce_sum(h, target, rng, max_iter=10000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            # need to add
            idx = rng.integers(0, n)
            add = min(diff, 1.0 - h[idx])
            if add == 0:
                it += 1
                continue
            h[idx] += add
            diff -= add
        else:
            idx = rng.integers(0, n)
            sub = min(-diff, h[idx])
            if sub == 0:
                it += 1
                continue
            h[idx] -= sub
            diff += sub
        it += 1
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # In rare case diff remains, adjust first element
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff proportionally?
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def slsqp_refine(x0, beta, max_iter=400):
    n = len(x0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def fun(x):
        corr = np.correlate(x, 1 - x, mode='full')
        # note: use logsumexp for smoothing
        # logsumexp(beta * corr) / beta
        return logsumexp(beta * corr) / beta
    res = minimize(fun, x0, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    x = enforce_sum(x, target, np.random.default_rng())
    return x

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng, max_iter=10000):
    """Clip to [0,1] and then adjust to hit the exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = rng.integers(0, n)
            add = min(diff, 1.0 - h[idx])
            if add == 0:
                it += 1
                continue
            h[idx] += add
            diff -= add
        else:
            idx = rng.integers(0, n)
            sub = min(-diff, h[idx])
            if sub == 0:
                it += 1
                continue
            h[idx] -= sub
            diff += sub
        it += 1
    # final tiny correction (should be negligible)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Create a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = rng.random(n)
    scale = target / h.sum()
    h = np.clip(h * scale, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def propose_move(h, rng, scale=0.2):
    """Transfer a small amount ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ----------------------------------------------------------------------
# Gradient descent on the piecewise‑linear max‑overlap objective
# ----------------------------------------------------------------------
def gradient_descent(start, rng, max_iters=2000, step_init=0.2):
    """Projected sub‑gradient descent using the worst shift's gradient."""
    h = start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = step_init
    for it in range(max_iters):
        # correlation and worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)          # shift value (can be negative)

        # gradient of the active overlap term: g_j = 1 - h_{j+k} - h_{j-k}
        g = np.ones(n)
        if k >= 0:
            g[: n - k] -= h[k:]
            g[k:]      -= h[: n - k]
        else:
            kk = -k
            g[kk:]    -= h[: n - kk]
            g[: n - kk] -= h[kk:]

        # try a descent step
        new_h = h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target, rng)

        new_c = compute_c5(new_h)
        cur_c = compute_c5(h)

        if new_c < cur_c - 1e-12:
            h = new_h
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
            step = min(step * 1.05, 0.5)
        else:
            step = max(step * 0.7, 1e-4)

        if step < 1e-6:
            break
    return best_h, best_c

# ----------------------------------------------------------------------
# Smooth surrogate + SLSQP refinement
# ----------------------------------------------------------------------
def slsqp_refine(x0, beta, max_iter=300):
    """Minimise a smooth log‑sum‑exp surrogate of max_{k} overlap."""
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def surrogate(x):
        corr = np.correlate(x, 1 - x, mode="full")
        # smooth approximation of max using log‑sum‑exp
        return logsumexp(beta * corr) / beta

    res = minimize(
        surrogate,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, target, np.random.default_rng())
    return x
