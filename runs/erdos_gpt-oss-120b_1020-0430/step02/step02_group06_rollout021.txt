def tournament_select(pop, fitnesses, k=3):
    idx = np.random.choice(len(pop), k, replace=False)
    best = idx[np.argmin([fitnesses[i] for i in idx])]
    return pop[best]

def crossover(p1, p2):
    # uniform crossover: child = np.where(random<0.5, p1, p2)
    mask = np.random.rand(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    return child

def mutate(child, mutation_rate=0.1, sigma=0.1):
    # For each gene, with probability mutation_rate, add Gaussian noise
    mask = np.random.rand(child.size) < mutation_rate
    n_changes = mask.sum()
    child[mask] += sigma * np.random.randn(n_changes)
    # optionally clip after
    child = np.clip(child, 0.0, 1.0)
    return child

def _ga_optimize(n, base_h, time_left_func, target_bound, max_total_gens=2000):
    pop_size = 30
    elite_size = 2
    mutation_rate = 0.1
    sigma_start = 0.2
    sigma_end = 0.02
    # compute sigma schedule: linear decay based on generation count.
    generation = 0
    best_h = None
    best_c5 = float('inf')
    # initial population
    pop = []
    # include base_h as first candidate
    if base_h is not None:
        pop.append(base_h.copy())
    while len(pop) < pop_size:
        cand = np.random.rand(n)
        cand = _project_to_feasible(cand, n)
        pop.append(cand)
    # evaluate
    fitness = [_c5_from_h(ind) for ind in pop]
    # track best
    idx_best = np.argmin(fitness)
    best_h = pop[idx_best].copy()
    best_c5 = fitness[idx_best]
    # early check
    if best_c5 < target_bound:
        return best_h, best_c5

    # begin generations
    max_gens = max_total_gens
    # but we may break early due to time
    while generation < max_gens and time_left_func() > 1.0:
        # compute sigma for this generation
        sigma = sigma_start - (sigma_start - sigma_end) * generation / max_gens
        # selection for offsprings
        new_pop = []
        # elitism
        elite_idxs = np.argsort(fitness)[:elite_size]
        for idx in elite_idxs:
            new_pop.append(pop[idx].copy())
        # generate rest
        while len(new_pop) < pop_size:
            # tournament selection
            def tournament():
                idxs = np.random.choice(pop_size, 3, replace=False)
                return pop[idxs[np.argmin([fitness[i] for i in idxs])]]
            parent1 = tournament()
            parent2 = tournament()
            # crossover
            if np.random.rand() < 0.7:
                # uniform crossover
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            # mutation
            if np.random.rand() < 0.9:
                # mutate each gene with prob mutation_rate
                mask_mut = np.random.rand(n) < mutation_rate
                n_mut = mask_mut.sum()
                if n_mut > 0:
                    child[mask_mut] += sigma * np.random.randn(n_mut)
                child = np.clip(child, 0.0, 1.0)
            # project finally
            child = _project_to_feasible(child, n)
            new_pop.append(child)
        pop = new_pop
        # evaluate fitness of new pop
        fitness = [_c5_from_h(ind) for ind in pop]
        # update best
        idx_best = np.argmin(fitness)
        if fitness[idx_best] < best_c5:
            best_c5 = fitness[idx_best]
            best_h = pop[idx_best].copy()
            # if beat target bound break early
            if best_c5 < target_bound:
                break
        generation += 1
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# ---------- feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        for i in range(n):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- objective ----------
def _c5_from_h(h: np.ndarray) -> float:
    """C5 for a discretised step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- local continuous optimisation ----------
def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n * 0.5

    def obj(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(obj, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    h_opt = _project_to_feasible(res.x[:n] if res.x is not None else h0, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- interpolation ----------
def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


# ---------- GA core ----------
def _ga_optimize(
    n,
    base_h,
    time_left,
    target,
    max_gens=5000,
    pop_size=40,
    elite_sz=2,
    mutation_rate=0.12,
):
    # initialise population
    pop = []
    if base_h is not None:
        pop.append(base_h.copy())
    # add a sinusoidal seed (often good)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    sin_seed = 0.5 + 0.5 * np.sin(np.pi * x)
    pop.append(_project_to_feasible(sin_seed, n))
    while len(pop) < pop_size:
        cand = np.random.rand(n)
        pop.append(_project_to_feasible(cand, n))

    # initial fitness
    fitness = [_c5_from_h(ind) for ind in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]

    if best_c5 < target:
        return best_h, best_c5

    # schedule for mutation magnitude
    sigma_start, sigma_end = 0.25, 0.03

    generation = 0
    while generation < max_gens and time_left() > 2.0:
        sigma = sigma_start - (sigma_start - sigma_end) * generation / max_gens

        # elitism
        elite_idxs = np.argsort(fitness)[:elite_sz]
        new_pop = [pop[i].copy() for i in elite_idxs]

        # helper for tournament selection
        def tournament():
            idxs = np.random.choice(pop_size, 3, replace=False)
            return pop[idxs[np.argmin([fitness[i] for i in idxs])]]

        while len(new_pop) < pop_size:
            p1 = tournament()
            p2 = tournament()
            # uniform crossover
            if np.random.rand() < 0.7:
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, p1, p2)
            else:
                child = p1.copy()

            # mutation
            if np.random.rand() < 0.9:
                mask_mut = np.random.rand(n) < mutation_rate
                n_mut = mask_mut.sum()
                if n_mut:
                    child[mask_mut] += sigma * np.random.randn(n_mut)
                child = np.clip(child, 0.0, 1.0)

            child = _project_to_feasible(child, n)
            new_pop.append(child)

        pop = new_pop
        fitness = [_c5_from_h(ind) for ind in pop]

        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_c5:
            best_c5 = fitness[cur_best]
            best_h = pop[cur_best].copy()
            if best_c5 < target:
                break

        generation += 1

    return best_h, best_c5

        parent_h = None

    target = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # grid sizes to explore
    n_cand = set()
    if parent_h is not None:
        n_cand.add(parent_h.size)
    for mul in (1, 2, 3, 4):
        base = parent_h.size if parent_h is not None else 64
        n = int(base * mul)
        if n % 2 == 0 and n <= 280:
            n_cand.add(n)
    # extra handcrafted
    n_cand.update([80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256, 280])
    n_list = sorted([n for n in n_cand if n % 2 == 0 and n <= 280])

    for n in n_list:
        if time_left() < 8.0:
            break

        # base seed for this resolution
        if parent_h is not None:
            if parent_h.size == n:
                base_h = parent_h.copy()
            else:
                base_h = _project_to_feasible(_upscale_h(parent_h, n), n)
        else:
            base_h = np.full(n, 0.5)

        # GA optimisation
        local_h, local_c5 = _ga_optimize(
            n, base_h, time_left, target, max_gens=8000, pop_size=40, elite_sz=2
        )
        # local refinement via SLSQP
        refined_h, refined_c5 = _solve_h(local_h, max_iter=800)
        # pick the better of GA and refined
        if refined_c5 < local_c5:
            local_h, local_c5 = refined_h, refined_c5

        if local_c5 < best_c5:
            best_h, best_c5, best_n = local_h, local_c5, n
            if best_c5 < target:
                break

        # occasional random restart on same n if time permits
        attempts = int(min(4, time_left() // 3))
        for _ in range(attempts):
            if time_left() < 5.0:
                break
            rand_start = _project_to_feasible(np.random.rand(n), n)
            h2, c2 = _ga_optimize(
                n, rand_start, time_left, target, max_gens=3000, pop_size=30, elite_sz=2
            )
            h2, c2 = _solve_h(h2, max_iter=600)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2, c2, n
                if best_c5 < target:
                    break

    # fallback if nothing found
    if best_h is None:
        n = 96
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
