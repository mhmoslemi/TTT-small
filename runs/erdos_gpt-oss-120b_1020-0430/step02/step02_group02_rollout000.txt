def solve_binary_milp(n, time_limit):
    import cvxpy as cp, numpy as np, math, time
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2, t >= 0]

    # list for w variables maybe use 2D bool?
    # Use dictionary of w for each shift k
    w_vars = {}
    for k in range(1, n): # shift positive
        m = n - k
        w = cp.Variable(m, boolean=True)
        w_vars[k] = w
        # linearization constraints for each i: w[i] <= h[i], w[i] <= h[i+k], w[i] >= h[i] + h[i+k] - 1
        # we need base arrays for index.
        constraints += [w <= h[:m]]
        constraints += [w <= h[k:]]
        constraints += [w >= h[:m] + h[k:] - 1]

        # mismatch = sum_i h[i] - sum_i w[i]; require <= t
        constraints += [cp.sum(h[:m]) - cp.sum(w) <= t]

    # objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve with time limit: param
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit, verbose=False) # if CBC accepts.
    except cp.error.SolverError:
        # fallback to ECOS_BB
        prob.solve(solver=cp.ECOS_BB, max_iters=int(1e6), verbose=False)
    return h.value, t.value, prob.status
```

import time
import numpy as np
from scipy import optimize
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible h (integral already satisfied)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all lags."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """SLSQP optimisation of C5 from a feasible start."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Pairwise epsilon swaps (preserving sum) with strict improvement."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# MILP for binary (0/1) solutions
# ----------------------------------------------------------------------
def _solve_milp_binary(n: int, time_limit: float):
    """Mixed‑integer formulation with binary h and auxiliary w variables."""
    N1 = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    cons = [cp.sum(h) == N1, t >= 0]

    for k in range(1, n):
        m = n - k
        w = cp.Variable(m, boolean=True)
        cons.append(w <= h[:m])
        cons.append(w <= h[k:])
        cons.append(w >= h[:m] + h[k:] - 1)
        # mismatch N1 - sum(w) <= t  <=> sum(w) >= N1 - t
        cons.append(cp.sum(w) >= N1 - t)

    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS_BB, max_iters=int(1e6), verbose=False)
        except Exception:
            return None, None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None, prob.status
    h_val = np.array(h.value).astype(float)
    t_val = int(round(t.value))
    return h_val, t_val, prob.status

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except NameError:
        n = 80
        init_h = np.full(n, 0.5)

    dx = 2.0 / n
    best_h = None
    best_c5 = float("inf")
    best_n = n

    # ------------------------------------------------------------------
    # 1) Binary MILP attempt (quickly finds a strong integer bound)
    # ------------------------------------------------------------------
    milp_time = max(0.0, min(600.0, time_left() - 5.0))
    if milp_time > 5.0:
        h_bin, t_bin, status = _solve_milp_binary(n, milp_time)
        if status in ("optimal", "optimal_inaccurate") and h_bin is not None:
            c5_bin = _c5_from_h(h_bin)  # exact value, should equal t_bin*dx
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin, c5_bin
                best_n = n
            # Refine the binary solution with continuous optimisation
            if time_left() > 20.0:
                h_ref, c5_ref = _solve_h(best_h, max_iter=800)
                if c5_ref < best_c5:
                    best_h, best_c5 = h_ref, c5_ref

    # ------------------------------------------------------------------
    # 2) Deterministic local optimisation from the best known start
    # ------------------------------------------------------------------
    base_h = best_h if best_h is not None else init_h.copy()
    base_h = _project_to_feasible(base_h, n)

    if time_left() > 10.0:
        h_opt, c5_val = _solve_h(base_h, max_iter=500)
        if c5_val < best_c5:
            best_h, best_c5 = h_opt, c5_val

        # hill‑climb refinement
        limit = start + budget_s - 0.5
        best_h, best_c5 = _hill_climb(best_h, max_steps=20000, time_limit=limit)

    # ------------------------------------------------------------------
    # 3) Randomised restarts with SLSQP + hill‑climb until time expires
    # ------------------------------------------------------------------
    while time_left() > 5.0:
        # generate a random feasible vector (continuous)
        rnd = np.random.rand(n)
        rnd = _project_to_feasible(rnd, n)
        h_opt, c5_val = _solve_h(rnd, max_iter=300)
        if c5_val < best_c5:

            limit = start + budget_s - 0.5
            best_h, best_c5 = _hill_climb(best_h, max_steps=5000, time_limit=limit)

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
