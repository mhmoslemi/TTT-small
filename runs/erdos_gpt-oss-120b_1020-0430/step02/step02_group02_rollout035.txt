def gradient_for_lag(h, k):
    n = h.size
    g = np.zeros_like(h)
    if k == 0:
        # derivative of Q_0 = sum_i h_i^2, gradient = 2*h_i
        g = 2*h
    else:
        # contribution from h[i] * h[i+k]
        g[:n - k] += h[k:]         # h_{i+k} term for i in 0..n-k-1
        g[k:] += h[:n - k]         # h_{i-k} term for i in k..n-1
    return g

def gradient_move(h, k, step):
    n = h.size
    g = gradient_for_lag(h, k)
    # want to move mass from indices with low g to high g.
    # Choose source idx_i where g is minimal among those with h_i > epsilon.
    # Choose target idx_j where g is maximal among those with h_j < 1 - epsilon.
    # epsilon maybe step size? Actually we will move amount eps = min(step, h_i, 1 - h_j).
    # step can be max move amount (some fraction).
    # Could also choose multiple top sources/targets; but simple pair.
    src_candidates = np.where(h > 1e-12)[0]
    dst_candidates = np.where(h < 1 - 1e-12)[0]
    if src_candidates.size == 0 or dst_candidates.size == 0:
        return h
    # select src = argmin g among src candidates
    src = src_candidates[np.argmin(g[src_candidates])]
    # select dst = argmax g among dst candidates
    dst = dst_candidates[np.argmax(g[dst_candidates])]

def _gradient_ascent_step(h, step):
    # compute autocorrelation via FFT to get minQ and argmin lag
    n = h.size
    L = 2*n
    H = np.fft.fft(h, L)
    auto = np.fft.ifft(H * np.conj(H)).real[:n]
    # min Q
    k = int(np.argmin(auto))
    # gradient for this lag
    g = _gradient_for_lag(h, k)
    # choose src and dst
    # indices where moving mass up reduces objective?
    # We want to increase Q_k, so move from low g to high g.
    # Thus src has low g, dst has high g.
    # Also must ensure movement possible.
    src_candidates = np.where(h > 1e-12)[0]
    dst_candidates = np.where(h < 1 - 1e-12)[0]
    if src_candidates.size == 0 or dst_candidates.size == 0:
        return h
    src = src_candidates[np.argmin(g[src_candidates])]
    dst = dst_candidates[np.argmax(g[dst_candidates])]

    h_new[dst] += eps

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Projection onto the feasible set  (0<=h<=1, sum(h)=n/2)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# Linear interpolation to a finer grid
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# C5 evaluation (fast via FFT)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a given h."""
    n = h.size
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    max_corr = np.max(corr)
    return float((2.0 / n) * max_corr)


# ----------------------------------------------------------------------
# Constraints for the SLSQP optimisation
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all lags (vectorised)."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Sum(h) = n/2."""
    n = x.size - 1
    return float(np.sum(x[:n]) - (n / 2.0))


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """SLSQP refinement of a feasible h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    # initialise t with the current max correlation
    L = 2 * n
    H = np.fft.fft(h0, L)
    G = np.fft.fft(1.0 - h0, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    t0 = float(np.max(corr))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Simple hill‑climbing (random epsilon exchanges)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.4
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c


# ----------------------------------------------------------------------
# Gradient step that reduces the worst lag correlation
# ----------------------------------------------------------------------
def _descend_worst_corr(h: np.ndarray, step: float) -> np.ndarray:
    """Move a small amount of mass from a low‑gradient index to a high‑gradient one."""
    n = h.size
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)          # signed lag
    k_abs = abs(k)

    # neighbour‑sum = h[i+k] + h[i‑k] (zero outside)
    if k_abs == 0:
        neigh = 2.0 * h
    else:
        neigh = np.zeros_like(h)
        neigh[: n - k_abs] = h[k_abs:]
        neigh[k_abs:] += h[: n - k_abs]

    src_cand = np.where(h > 1e-12)[0]
    dst_cand = np.where(h < 1 - 1e-12)[0]
    if src_cand.size == 0 or dst_cand.size == 0:
        return h
    src = src_cand[np.argmin(neigh[src_cand])]
    dst = dst_cand[np.argmax(neigh[dst_cand])]
    eps = min(step, h[src], 1.0 - h[dst])
    if eps <= 0:
        return h
    h_new = h.copy()
    h_new[src] -= eps
    h_new[dst] += eps
    return h_new
