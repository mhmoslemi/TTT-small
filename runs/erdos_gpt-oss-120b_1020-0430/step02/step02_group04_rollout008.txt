```
def binary_from_continuous(h):
    n = len(h)
    K = n // 2
    idx = np.argpartition(-h, K-1)[:K]   # top K entries
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b
```

Now binary hill climb:

```
def binary_hill_climb(b, rng, max_iters=2000):
    n = len(b)
    best = b.copy()
    best_c = compute_c5(b)
    for it in range(max_iters):
        # pick random 1 at idx1 and 0 at idx2
        ones = np.flatnonzero(b == 1)
        zeros = np.flatnonzero(b == 0)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        b2 = b.copy()
        b2[i] = 0.0
        b2[j] = 1.0
        c2 = compute_c5(b2)
        if c2 < best_c:
            best_c = c2
            best = b2.copy()
            b = b2
        # optionally occasionally accept worse?
    return best, best_c
```

```
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0

        lam = (lo + hi) * 0.5

def compute_c5(h):
    """Discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """
    Project vector v onto the box [lower,upper] while forcing the sum to be exactly target.
    """
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()

    return proj

        if rng.random() < 0.5:
            h_start = random_feasible_symmetric(n, rng)
        else:
            b = np.zeros(n, dtype=float)
            ones = rng.choice(n, size=n // 2, replace=False)
            b[ones] = 1.0
            h_start = b

        # occasional blending with current best
        if rng.random() < 0.2:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, n / 2.0)
            h_start = enforce_symmetry(h_start)
            h_start = project_onto_box_sum(h_start, n / 2.0)

        # ----- continuous refinement -----
        h_ref, c_ref = grad_descent_optimize(
            h_start, beta_sched, steps_per_beta, rng, sym=True
        )
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

        # ----- binary projection + hill‑climb -----
        b_proj = binary_from_continuous(h_ref)
        b_best, c_b = binary_hill_climb(b_proj, rng, max_iters=2500)
        if c_b < best_c:
            best_h, best_c = b_best.astype(float), c_b

        # ----- occasional SLSQP refinement -----
        if rng.random() < 0.1:
            try:
                h_slsqp = slsqp_optimize(best_h, beta=200.0, max_iter=400)
                c_slsqp = compute_c5(h_slsqp)
                if c_slsqp < best_c:
                    best_h, best_c = h_slsqp, c_slsqp
            except Exception:
                pass

    # ------------------------------------------------------------------
    # optional up‑sampling for finer resolution
    # ------------------------------------------------------------------
    if (
        time.time() - start_time < budget_s * 0.95
        and n * 2 <= 260
    ):
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up, _ = grad_descent_optimize(
            h_up, beta_sched, steps_per_beta, rng, sym=True
        )
        for beta in (200, 500, 1000):
            if time.time() - start_time > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta=beta, max_iter=300)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # final clean‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
```
