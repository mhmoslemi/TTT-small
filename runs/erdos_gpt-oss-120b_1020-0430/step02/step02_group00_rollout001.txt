import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
def _max_corr(h: np.ndarray) -> int:
    """Maximum correlation of h with 1-h (full mode)."""
    return int(np.max(np.correlate(h, 1.0 - h, mode="full")))


def _project_sum_half(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=len(h)/2 (water‑filling)."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(15):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _binary_sa(start_h: np.ndarray, max_seconds: float) -> tuple:
    """Simulated‑annealing on binary vectors (half ones)."""
    n = start_h.size
    h = start_h.copy()
    cur = _max_corr(h)
    best_h, best_val = h.copy(), cur
    t0 = time.time()
    while time.time() - t0 < max_seconds:
        elapsed = time.time() - t0
        T = 2.0 * (1.0 - elapsed / max_seconds)  # linear cooling
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.random.randint(ones.size)]
        j = zeros[np.random.randint(zeros.size)]
        h[i], h[j] = h[j], h[i]                 # swap
        new = _max_corr(h)
        delta = new - cur
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            cur = new
            if new < best_val:
                best_val, best_h = new, h.copy()
        else:
            h[i], h[j] = h[j], h[i]             # revert
    return best_h, best_val


def _continuous_optimize(h0: np.ndarray, max_iter: int, time_budget: float) -> tuple:
    """SLSQP refinement allowing non‑binary values."""
    n = h0.size
    h0 = _project_sum_half(h0)
    t0 = float(np.max(np.correlate(h0, 1.0 - h0, mode="full")))
    x0 = np.concatenate([h0, [t0]])

    def obj(x):
        return x[-1]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{"type": "ineq", "fun": ineq},
                   {"type": "eq",   "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(obj, x0, method="SLSQP",
                                bounds=bounds,
                                constraints=constraints,
                                options=opts)
        h_opt = _project_sum_half(res.x[:n])
        val = _max_corr(h_opt) * (2.0 / n)
        return h_opt, float(val)
    except Exception:
        return h0, _max_corr(h0) * (2.0 / n)


def _upscale_binary(h: np.ndarray, new_n: int) -> np.ndarray:
    """Integer up‑scaling by repetition (new_n must be multiple of len(h))."""
    factor = new_n // h.size
    if factor * h.size == new_n:
        return np.repeat(h, factor).astype(float)
    # fall back to linear interpolation then threshold to half ones
    x_old = np.linspace(0, 2, h.size, endpoint=False) + 1.0 / h.size
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    interp = np.interp(x_new, x_old, h)
    thresh = np.partition(interp, -new_n // 2)[-new_n // 2]
    bin_up = (interp > thresh).astype(float)
    # ensure exact half ones
    cur = bin_up.sum()
    need = new_n // 2 - cur
    if need > 0:
        zeros = np.where(bin_up == 0)[0]
        pick = np.random.choice(zeros, need, replace=False)
        bin_up[pick] = 1.0
    elif need < 0:
        ones = np.where(bin_up == 1)[0]
        pick = np.random.choice(ones, -need, replace=False)
        bin_up[pick] = 0.0
    return bin_up
