    target_one = n // 2
    h = h0.copy()
    cur_O, _ = max_overlap(h)
    best_h, best_O = h.copy(), cur_O
    T = 2.0
    it = 0
    while True:
        if time.time() - start_time > budget_s * 0.90:
            break
        if it >= max_iters:
            break
        # pick a random 1 and a random 0
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        # swap
        h[i] = 0
        h[j] = 1
        new_O, _ = max_overlap(h)
        delta = new_O - cur_O
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_O = new_O
            if new_O < best_O:
                best_O, best_h = new_O, h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        it += 1
        if it % 1000 == 0:
            T *= 0.95
    return best_h, best_O

# ---------- deterministic refinement focusing on the worst shift ----------
def refine_worst_shift(h, rng, start_time, budget_s):
    n = h.size
    while time.time() - start_time < budget_s * 0.98:
        Omax, corr = max_overlap(h)
        if Omax == 0:
            break
        idx = int(corr.argmax())
        k = idx - (n - 1)                # shift corresponding to Omax
        # indices that actually contribute to Omax
        if k >= 0:
            i_range = np.arange(0, n - k)
            contrib = (h[i_range] == 1) & (h[i_range + k] == 0)
        else:
            kabs = -k
            i_range = np.arange(kabs, n)
            contrib = (h[i_range] == 1) & (h[i_range - kabs] == 0)
        i_cand = i_range[contrib]
        if i_cand.size == 0:
            break
        i_cand = rng.permutation(i_cand)
        zeros = np.flatnonzero(h == 0)
        improved = False
        for i in i_cand:
            # try a bounded random sample of zero positions
            if zeros.size > 30:
                js = rng.choice(zeros, 30, replace=False)
            else:
                js = zeros
            for j in js:
                h[i] = 0
                h[j] = 1
                new_O = max_overlap(h)[0]
                if new_O < Omax:
                    improved = True
                    break
                h[i] = 1
                h[j] = 0
            if improved:
                break
        if not improved:
            break
    return h, max_overlap(h)[0]

# ---------- continuous refinement via SLSQP (log‑sum‑exp surrogate) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h0, beta, rng, max_iter=200):
    """One call of SLSQP on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible tiny sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720                     # chosen resolution (≤ 1000)
    M = n // 2

    # start from a binary parent (or random)
    best_h = initial_binary(rng, n)
    best_O, _ = max_overlap(best_h)

    # ---------- multi‑restart simulated annealing ----------
    while time.time() - start < budget_s * 0.95:
        # occasional refinement of current best
        if rng.random() < 0.15:
            h0 = best_h.copy()
        else:
            h0 = np.zeros(n, dtype=np.int8)
            h0[rng.choice(n, M, replace=False)] = 1
        cand_h, cand_O = anneal_binary(h0, rng, start, budget_s)
        if cand_O < best_O:
            best_h, best_O = cand_h, cand_O

    # ---------- deterministic knock‑down of the worst shift ----------
    best_h, best_O = refine_worst_shift(best_h, rng, start, budget_s)

    # ---------- optional continuous polishing ----------
    try:
        h_float = best_h.astype(float)
        # a few β stages, modest iteration count
        for beta in (5.0, 20.0, 80.0, 200.0):
            h_float = slsqp_refine(h_float, beta, rng, max_iter=150)
        O_float = max_overlap(h_float)[0]
        if O_float < best_O:
            best_h, best_O = h_float, O_float
    except Exception:
        pass

    final_c5 = best_O * (2.0 / n)
    # ensure feasibility before returning
    if isinstance(best_h, np.ndarray) and best_h.dtype != float:
        best_h = best_h.astype(float)
    best_h = np.clip(best_h, 0.0, 1.0)
    # fix tiny sum errors
    target_sum = 1.0
    cur_sum = best_h.sum() * (2.0 / n)
    if abs(cur_sum - target_sum) > 1e-12:
        # simple scaling (preserves bounds because values are already near 0/1)
        best_h *= target_sum / cur_sum
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h.tolist(), float(final_c5), int(n)
