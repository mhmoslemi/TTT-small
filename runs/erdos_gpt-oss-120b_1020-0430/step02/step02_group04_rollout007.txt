    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        # compute raw conv
        conv = np.correlate(h, 1 - h, mode='full')
        max_val = conv.max()
        # compute gradient for raw conv
        g = np.zeros_like(h)
        # determine which ks achieving max within tolerance
        # tolerance relative maybe 1e-12 absolute?
        idxs = np.where(conv >= max_val - 1e-12)[0]

                g[:len(h)-k] += 1 - h[k:]
                g[k:] -= h[:len(h)-k]

                kk = -k
                g[kk:] += 1 - h[:len(h)-kk]
                g[:len(h)-kk] -= h[kk:]

    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    # compute max overlap scaled by dx = 2/n
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

def random_feasible_symmetric(n, rng):
    # n must be even
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    full = np.concatenate([x, x[::-1]])
    return full

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale

    new_h[j] += eps
    new_h = (new_h + new_h[::-1]) * 0.5

        max_val = conv.max()
        # compute subgradient of the scaled max

        g = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:

            else:

# SLSQP helpers
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp safely
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n/2.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Scaled maximum overlap for step function h (dx = 2 / len(h))."""
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project onto [lower,upper]^n with exact sum = target (box–sum projection)."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible_symmetric(n, rng):
    """Random symmetric vector of length n with entries in [0,1] and sum = n/2."""
    m = n // 2
    target_half = n / 4.0
    x = rng.random(m)
    x = project_onto_box_sum(x, target_half)
    return np.concatenate([x, x[::-1]])


# ---------- local moves ----------
def perturb_symmetric(h, rng, scale=0.2):
    """Transfer a small amount of mass between two coordinates, keep symmetry."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    new_h = (new_h + new_h[::-1]) * 0.5
    new_h = project_onto_box_sum(new_h, n / 2.0)
    return new_h


def hill_climb_sym(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb on the symmetric feasible set."""
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_symmetric(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- subgradient descent on the max overlap ----------
def subgrad_optimize(h0, rng, max_iters=1200, step0=0.2, decay=0.9995):
    """Projected subgradient descent on the max‑overlap (non‑smooth) objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        # raw correlation (without scaling)
        conv = np.correlate(h, 1 - h, mode="full")
        max_val = conv.max()
        # indices attaining the maximum
        idxs = np.where(conv >= max_val - 1e-12)[0]
        # subgradient of the scaled maximum
        g = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                g[: n - k] += 1 - h[k:]
                g[k:] -= h[: n - k]
            else:
                kk = -k
                g[kk:] += 1 - h[: n - kk]
                g[: n - kk] -= h[kk:]
        # scale because compute_c5 multiplies by 2/n
        g *= (2.0 / n)

        # projected step
        h_new = h - step * g
        h_new = project_onto_box_sum(h_new, target)
        h_new = (h_new + h_new[::-1]) * 0.5

        c_new = compute_c5(h_new)
        c_cur = max_val * (2.0 / n)

        # accept if not worse, otherwise with tiny probability
        if c_new <= c_cur or rng.random() < 0.01:
            h = h_new

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

        step *= decay
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
