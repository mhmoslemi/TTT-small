import time
import numpy as np
from scipy import optimize
from scipy.signal import fftconvolve

# ----------------------------------------------------------------------
# Feasibility projection
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# Objective evaluation (FFT based for speed)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    conv = fftconvolve(h, 1.0 - h, mode='full')
    return float(conv.max() * (2.0 / n))


# ----------------------------------------------------------------------
# Linear up‑sampling of a coarse step function
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# SLSQP refinement (t‑variable formulation)
# ----------------------------------------------------------------------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode='full')


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h_slsqp(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = float(corr0.max())
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Simple hill‑climb (always improving)
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    n = h.size
    step = 0
    while step < max_steps and time.time() < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        max_eps = min(max_down, max_up)
        if max_eps <= 0:
            continue
        eps = np.random.rand() * max_eps * 0.5
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
        step += 1
    return best_h, best_c


# ----------------------------------------------------------------------
# Simulated annealing (allows temporary worsening)
# ----------------------------------------------------------------------
def _simulated_annealing(h: np.ndarray, c0: float, time_limit: float, max_iters: int = 200000) -> tuple:
    best_h = h.copy()
    best_c = c0
    cur_h = h.copy()
    cur_c = c0
    n = h.size
    start_T = 1e-4
    end_T = 1e-7
    it = 0
    while it < max_iters and time.time() < time_limit:
        # temperature schedule (linear)
        T = start_T + (end_T - start_T) * (it / max_iters)

        # propose move
        if np.random.rand() < 0.8:  # epsilon exchange
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            max_down = cur_h[i]
            max_up = 1.0 - cur_h[j]
            max_eps = min(max_down, max_up)
            if max_eps <= 0:
                it += 1
                continue
            eps = np.random.rand() * max_eps * 0.5
            cand = cur_h.copy()
            cand[i] -= eps
            cand[j] += eps
        else:  # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]

        new_c = _c5_from_h(cand)
        delta = new_c - cur_c
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = cand, new_c
            if new_c < best_c:
                best_h, best_c = cand.copy(), new_c
        it += 1
    return best_h, best_c

        if prev_best_h is not None:
            base_h = _upscale_h(prev_best_h, n)
        elif GLOBAL_INITIAL_H is not None and GLOBAL_INITIAL_H.size != n:
            base_h = _upscale_h(GLOBAL_INITIAL_H, n)
        elif GLOBAL_INITIAL_H is not None and GLOBAL_INITIAL_H.size == n:
            base_h = GLOBAL_INITIAL_H.copy()
        else:
            base_h = np.full(n, 0.5)

        # Allocate most of the remaining time to this size, leaving a small reserve
        lim = start_time + budget_s - 0.3
        h_opt, c_opt = _optimize_n(n, base_h, lim)

        if c_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c_opt, n

        # Early exit if we beat the target
        if best_c5 < target_bound:
            break

        # Use the found solution as a seed for finer grids
        prev_best_h = best_h

    # Fallback (should never be needed)
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    # Final projection to guarantee feasibility
    best_h = _project_to_feasible(best_h, best_n)

    return best_h.tolist(), float(best_c5), int(best_n)
