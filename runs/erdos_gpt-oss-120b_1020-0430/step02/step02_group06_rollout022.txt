import time
import numpy as np

# ---------- Feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h) = n/2 via water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- Correlation helpers (FFT) ----------
def _full_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT, returns full array."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())  # next power‑of‑two ≥ 2n‑1
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * G, m)[: 2 * n - 1]
    return corr


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a discretised h."""
    n = h.size
    corr = _full_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def _max_corr_and_idx(h: np.ndarray):
    """Return (maximum correlation value, index of first max) for h."""
    corr = _full_corr_fft(h)
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx


# ---------- Simple up‑sampling ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- Local stochastic refinement ----------
def _local_refine(h: np.ndarray, attempts: int = 12, delta: float = 0.02) -> np.ndarray:
    """Random small perturbations with feasibility projection."""
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for _ in range(attempts):
        idx = np.random.choice(n, size=min(5, n), replace=False)
        trial = best_h.copy()
        trial[idx] += delta * np.random.randn(idx.size)
        trial = _project_to_feasible(trial, n)
        c5 = _c5_from_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h


# ---------- Random feasible vector ----------
def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1] with sum = n/2."""
    h = np.random.rand(n)
    return _project_to_feasible(h, n)


# ---------- Gradient‑descent optimiser ----------
def _gd_optimize(h0: np.ndarray, max_iters: int, time_limit: float):
    """Simple gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    lr = 0.05
    decay = 0.9995
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, idx = _max_corr_and_idx(h)
        c5 = max_val * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            if best_c5 < 0.3808:
                break
        # gradient of the (hard) max
        shift = idx - (n - 1)
        i = np.arange(n)
        j = i + shift
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(h)
        grad[mask] = 1.0 - 2.0 * h[j[mask]]
        # step + projection
        h_new = h - lr * grad
        h_new = _project_to_feasible(h_new, n)
        new_c5 = _c5_from_h(h_new)
        if new_c5 <= c5:
            h = h_new
        else:
            lr *= 0.9
        lr *= decay
        # occasional jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            h = _project_to_feasible(h + 0.02 * np.random.randn(n), n)
    return best_h, best_c5


# ---------- Main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

        if parent_h is not None:

        else:

        base_h = _project_to_feasible(base_h, n)

        alloc = max(5.0, time_left() / (remaining + 1))
