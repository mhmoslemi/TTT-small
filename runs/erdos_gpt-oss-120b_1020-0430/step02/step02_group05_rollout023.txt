def _solve_h(start_h, max_iter=300):
    n = start_h.size
    target = n / 2.0
    h0 = _project_to_simplex_box(start_h, target)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # inequality constraints: t - corr >= 0 for each shift
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr
    # equality: sum h = target
    def eq(x):
        return np.sum(x[:n]) - target

    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]

    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    try:
        res = optimize.minimize(_objective, x0, method='SLSQP', bounds=bounds,
                                constraints=cons, options=opts)
    except Exception:
        res = None

    if res is None or not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, target)
    return h_opt, _c5_from_h(h_opt)

import time
import numpy as np
from scipy import optimize


# ----------------------------------------------------------------------
# Helper classes for SLSQP constraints (no closures / lambdas)
# ----------------------------------------------------------------------
class _IneqConstraint:
    """t - correlation >= 0 for each shift."""
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        h = x[: self.n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr


class _EqConstraint:
    """sum(h) = n/2."""
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x[: self.n]) - (self.n / 2.0)


def _objective(x):
    """Minimise auxiliary variable t."""
    return x[-1]


# ----------------------------------------------------------------------
# Projection onto box [0,1] and hyperplane sum = target
# ----------------------------------------------------------------------
def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# ----------------------------------------------------------------------
# C5 evaluation
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


# ----------------------------------------------------------------------
# SLSQP refinement for a given start point
# ----------------------------------------------------------------------
def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    n = start_h.size
    target = n / 2.0
    h0 = _project_to_simplex_box(start_h, target)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _IneqConstraint(n)},
        {"type": "eq", "fun": _EqConstraint(n)},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(
            _objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=opts,
        )
    except Exception:
        res = None

    if res is None or not res.success:
        h_opt = h0
    else:
        h_opt = res.x[:n]

    h_opt = _project_to_simplex_box(h_opt, target)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Gradient of the active quadratic term (shift that attains max)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.size
    if shift >= 0:
        left = np.concatenate((np.zeros(shift), h[: n - shift]))
        right = np.concatenate((h[shift:], np.zeros(shift)))
    else:
        s = -shift
        left = np.concatenate((h[s:], np.zeros(s)))
        right = np.concatenate((np.zeros(s), h[: n - s]))
    return 1.0 - (left + right)


# ----------------------------------------------------------------------
# Projected gradient descent on max‑correlation objective
# ----------------------------------------------------------------------
def _proj_grad_descent(
    h: np.ndarray,
    max_iter: int,
    start_time: float,
    budget_s: float,
    target: float,
) -> tuple[np.ndarray, float]:
    n = h.size
    cur_c5 = _c5_from_h(h)
    alpha = 0.1
    min_alpha = 1e-6

    for _ in range(max_iter):
        if budget_s - (time.time() - start_time) < 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr) - (n - 1))
        grad = _gradient_for_shift(h, shift)
        h_new = _project_to_simplex_box(h - alpha * grad, target)
        new_c5 = _c5_from_h(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            alpha = min(alpha * 1.05, 0.2)
        else:
            alpha = max(alpha * 0.5, min_alpha)

    return h, cur_c5


# ----------------------------------------------------------------------
# Random hill‑climbing perturbations (mass transfer between two points)
# ----------------------------------------------------------------------
def _hill_climb_random(
    h: np.ndarray,
    cur_c5: float,
    start_time: float,
    budget_s: float,
    target: float,
    steps: int = 5000,
) -> tuple[np.ndarray, float]:
    n = h.size
    max_delta = 0.2

    for _ in range(steps):
        if budget_s - (time.time() - start_time) < 0.5:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        lower = max(-hj, hi - 1.0, -max_delta)
        upper = min(hi, 1.0 - hj, max_delta)
        if lower >= upper:
            continue
        delta = np.random.uniform(lower, upper)
        h_new = h.copy()
        h_new[i] = hi - delta
        h_new[j] = hj + delta
        new_c5 = _c5_from_h(h_new)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            max_delta = max_delta * 0.99  # slowly shrink step size
    return h, cur_c5


# ----------------------------------------------------------------------
# Simple linear upscale of a coarse vector
# ----------------------------------------------------------------------
def _upscale_h(src: np.ndarray, n: int) -> np.ndarray:
    src_n = src.size
    x_src = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_tgt = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x_tgt, x_src, src)


# ----------------------------------------------------------------------
# Tent shape (symmetric linear ramp) as a generic start
# ----------------------------------------------------------------------
def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


# ----------------------------------------------------------------------
# Optimisation routine for a single resolution n
# ----------------------------------------------------------------------
def _optimize_n(
    n: int,
    base_h: np.ndarray,
    start_time: float,
    budget_s: float,
) -> tuple[np.ndarray, float]:
    target = n / 2.0
    h_cur = _project_to_simplex_box(base_h, target)
    best_h = h_cur
    best_c5 = _c5_from_h(best_h)

    # few SLSQP runs with different starts
    for attempt in range(4):
        if budget_s - (time.time() - start_time) < 8.0:
            break
        if attempt == 0:
            start = best_h.copy()
        else:
            if np.random.rand() < 0.5:
                start = best_h + 0.2 * np.random.randn(n)
            else:
                start = np.random.rand(n)
            start = _project_to_simplex_box(start, target)

        h_opt, c5_val = _solve_h(start, max_iter=400)
        if c5_val < best_c5:
            best_h, best_c5 = h_opt, c5_val

    # gradient descent on the max‑correlation
    if budget_s - (time.time() - start_time) > 5.0:
        best_h, best_c5 = _proj_grad_descent(
            best_h,
            max_iter=2000,
            start_time=start_time,
            budget_s=budget_s,
            target=target,
        )

    # final random hill‑climb
    if budget_s - (time.time() - start_time) > 5.0:
        best_h, best_c5 = _hill_climb_random(
            best_h,
            best_c5,
            start_time,
            budget_s,
            target=target,
            steps=10000,
        )

    return best_h, best_c5


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # obtain possible parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target_bound = 0.3808

    # candidate grid sizes (include parent size if available)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (64, 96, 128, 160, 192, 224, 256, 288, 320, 384, 448):
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if budget_s - (time.time() - start_time) < 10.0:
            break

        # build base h at this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        # optimise at this resolution
        h_opt, c5_val = _optimize_n(n, base_h, start_time, budget_s)

        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target_bound:
                break

    # extra refinement on the global best if time permits
    if best_h is not None and budget_s - (time.time() - start_time) > 15.0:
        best_h, best_c5 = _proj_grad_descent(
            best_h,
            max_iter=5000,
            start_time=start_time,
            budget_s=budget_s,
            target=best_n / 2.0,
        )
        best_h, best_c5 = _hill_climb_random(
            best_h,
            best_c5,
            start_time,
            budget_s,
            target=best_n / 2.0,
            steps=20000,
        )

    # fallback in the unlikely event that everything failed
    if best_h is None:
        best_n = 128
        best_h = _tent_h(best_n)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
