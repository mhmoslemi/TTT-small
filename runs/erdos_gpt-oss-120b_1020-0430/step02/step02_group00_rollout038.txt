def _solve_h_slsqp(initial_h, max_iter=500):
    n = initial_h.size
    # project initial feasible
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1-h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    def ineq(x):
        h = x[:n]
        t = x[-1]
        # compute correlations
        corr = np.correlate(h, 1-h, mode='full')
        return t - corr
    def eq(x):
        h = x[:n]
        return np.sum(h) - n/2.0
    def obj(x):
        return x[-1]
    cons = [{'type':'ineq','fun':ineq}, {'type':'eq','fun':eq}]
    res = optimize.minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter, 'ftol':1e-9, 'disp':False})
    if not res.success:
        # fallback to h0
        h_opt = h0
        t_opt = t0
    else:
        x_opt = res.x
        h_opt = x_opt[:n]
        t_opt = x_opt[-1]
    # ensure feasible
    h_opt = _project_to_feasible(h_opt, n)
    c5 = np.max(np.correlate(h_opt, 1-h_opt, mode='full')) * (2.0 / n)
    return h_opt, c5

def _project_to_feasible(h, n, tol=1e-9, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need decrease sum
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust a single entry
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h
```

def _binary_hill_climb(n, time_budget):
    target = n // 2
    best_vec = None
    best_M = n  # high
    end = time.time() + time_budget
    # Pre-calc max number of restarts
    while time.time() < end:
        # random start
        vec = np.zeros(n, dtype=np.int8)
        vec[:target] = 1
        np.random.shuffle(vec)
        M = _binary_max_corr(vec)
        # local search
        improved = True
        while improved and time.time() < end:
            improved = False
            # try a number of random swaps
            for _ in range(4 * n):
                # pick random 1 and 0
                ones = np.where(vec == 1)[0]
                zeros = np.where(vec == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                # swap
                vec[i] = 0
                vec[j] = 1
                M_new = _binary_max_corr(vec)
                if M_new <= M:
                    M = M_new
                    improved = True
                else:
                    # revert
                    vec[i] = 1
                    vec[j] = 0
                if M < best_M:
                    best_M = M
                    best_vec = vec.copy()
                    # if found good enough, maybe break
                    if best_M <= 49:
                        # we can break early
                        return best_vec, best_M
                # optional break if time low
                if time.time() >= end:
                    break
        # optional early exit if best already low
        if best_M <= 49:
            break
    return best_vec, best_M

def _solve_h_slsqp(initial_h, max_iter=500):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full')
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    def obj(x):
        return x[-1]

    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    try:
        res = opt.minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options=opts)
        if res.success:
            h_opt = res.x[:n]
            # ensure feasibility after rounding errors
            h_opt = _project_to_feasible(h_opt, n)
        else:
            h_opt = h0
    except Exception:
        h_opt = h0
    c5 = _c5_from_h(h_opt)
    return h_opt, c5

def _gradient_descent_opt(start_h, time_limit):
    n = start_h.size
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    end = time.time() + time_limit
    step = 0.5
    alpha = 1.0
    # schedule parameters
    iter_count = 0
    while time.time() < end:
        iter_count += 1
        # compute correlation array
        A = np.correlate(h, 1 - h, mode='full')
        max_A = A.max()
        # smoothing
        # subtract max for numerical stability
        exp_vals = np.exp(alpha * (A - max_A))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        g = np.ones(n)
        offset = n - 1
        for d, wk in enumerate(w):
            if wk == 0:
                continue
            k = d - offset
            if k >= 0:
                if n - k > 0:
                    g[:n - k] -= wk * h[k:]
            else:
                kk = -k
                if n - kk > 0:
                    g[kk:] -= wk * h[:n - kk]
        # update
        h_new = h - step * g
        # project
        h_new = _project_to_feasible(h_new, n)
        # evaluate
        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # optionally reduce step for finer progress
            step *= 0.95
        else:
            # maybe increase step if not improving? keep.
            step *= 0.99
        h = h_new
        # adjust alpha periodically
        if iter_count % 200 == 0:
            alpha = min(alpha * 1.2, 50.0)
        # early exit if best_c5 below target
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

    best_vec = None
    best_M = n
    end = time.time() + time_budget
    while time.time() < end:
        vec = np.zeros(n, dtype=np.int8)
        vec[:target] = 1
        np.random.shuffle(vec)
        M = _binary_max_corr(vec)
        improved = True
        while improved and time.time() < end:
            improved = False
            for _ in range(4 * n):
                ones = np.where(vec == 1)[0]
                zeros = np.where(vec == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                vec[i] = 0
                vec[j] = 1
                Mnew = _binary_max_corr(vec)
                if Mnew <= M:
                    M = Mnew
                    improved = True
                else:
                    vec[i] = 1
                    vec[j] = 0
                if M < best_M:
                    best_M = M
                    best_vec = vec.copy()
                    if best_M <= 49:
                        return best_vec, best_M
                if time.time() >= end:
                    break
        if best_M <= 49:
            break
    return best_vec, best_M

def _solve_h_slsqp(initial_h, max_iter=800):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full'))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    def obj(x):
        return x[-1]

    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    try:
        res = opt.minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options=opts)
        if res.success:
            h_opt = res.x[:n]
            h_opt = _project_to_feasible(h_opt, n)
        else:
            h_opt = h0
    except Exception:
        h_opt = h0
    c5 = _c5_from_h(h_opt)
    return h_opt, c5

def _gradient_descent_opt(start_h, time_limit):
    n = start_h.size
    h = _project_to_feasible(start_h, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    end = time.time() + time_limit
    step = 0.3
    alpha = 1.0
    offset = n - 1
    it = 0
    while time.time() < end:
        it += 1
        A = np.correlate(h, 1.0 - h, mode='full')
        max_A = A.max()
        exp_vals = np.exp(alpha * (A - max_A))
        w = exp_vals / exp_vals.sum()
        g = np.ones(n)
        for d, wk in enumerate(w):
            if wk == 0.0:
                continue
            k = d - offset
            if k == 0:
                g -= 2.0 * wk * h
            elif k > 0:
                g[:n - k] -= wk * h[k:]
            else:
                kk = -k
                g[kk:] -= wk * h[:n - kk]
        h_new = h - step * g
        h_new = _project_to_feasible(h_new, n)
        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step *= 0.96
        else:
            step *= 0.99
        h = h_new
        if it % 200 == 0:
            alpha = min(alpha * 1.2, 40.0)
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
    except NameError:
        n0 = 200
        h0 = np.random.rand(n0).astype(float)

    if n0 % 2 != 0:
        n0 += 1
        h0 = np.resize(h0, n0)
    h0 = _project_to_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = _c5_from_h(best_h)
    best_n = n0

    # binary hill climb (use part of budget)
    bin_budget = min(300.0, time_left() * 0.3)
    if bin_budget > 10 and best_n <= 300:
        bin_vec, bin_M = _binary_hill_climb(best_n, bin_budget)
        if bin_vec is not None:
            h_bin = bin_vec.astype(float)
            c5_bin = _c5_from_h(h_bin)
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin, c5_bin
            # polish with SLSQP
            for _ in range(2):
                if time_left() < 20:
                    break
                h_opt, c5_opt = _solve_h_slsqp(best_h, max_iter=800)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt

    # gradient descent / simulated annealing phase
    while time_left() > 8:
        # start from current best with small random perturbation
        sigma = 0.15
        start_vec = best_h + sigma * np.random.randn(best_n)
        start_vec = _project_to_feasible(start_vec, best_n)
        remaining = time_left()
        h_opt, c5_opt = _gradient_descent_opt(start_vec, time_limit=remaining * 0.5)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        # occasional SLSQP refinement
        if time_left() > 12:
            h_opt2, c5_opt2 = _solve_h_slsqp(best_h, max_iter=600)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

            break

    best_c5 = _c5_from_h(best_h)
    return best_h.tolist(), float(best_c5), int(best_h.size)

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h
