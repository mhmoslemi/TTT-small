import time
import numpy as np
from scipy.optimize import minimize

# globals for the smooth max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint: sum(x) = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp surrogate for max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def fix_sum(h, target, rng):
    """Adjust a feasible vector so that its sum equals target."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP optimisation of the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (small numerical drift)
    x = fix_sum(x, CURRENT_TARGET, rng)
    return x


def random_binary(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def hill_climb_binary(h_start, rng, max_iters=2000):
    """Deterministic hill climb on binary vectors (swap moves)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        # pick a random 1 and a random 0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if c < compute_c5(best_h):
            # keep improvement
            pass
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c


def simulated_annealing_binary(h_start, rng, time_limit,
                               max_iters=40000, init_temp=0.02, final_temp=1e-5):
    """Simple SA for binary vectors; returns best found before time_limit."""
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = cur_h.size
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        # linear temperature decay in log‑space
        t = init_temp * ((final_temp / init_temp) ** (it / max_iters))
        # propose a swap
        i = rng.integers(n)
        while cur_h[i] == 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] == 1.0:
            j = rng.integers(n)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / t):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_c


def upsample(h):
    """Repeated up‑sampling by factor 2 (preserves sum constraint)."""
    return np.repeat(h, 2)


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Initialise from parent if provided
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = fix_sum(h0, n0 / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, n0

    # ------------------------------------------------------------------
    # Main optimisation on the original resolution
    # ------------------------------------------------------------------
    if best_n is None:
        best_n = 120
    target_n = best_n
    limit = start_time + budget_s * 0.85  # reserve some time for up‑sampling

    while time.time() < limit:
        # random binary start
        h_start = random_binary(target_n, rng)
        # optional: also try a copy of the current best (binary version)
        if best_h is not None and rng.random() < 0.2:
            # convert best to binary by taking the largest entries
            order = np.argsort(best_h)
            h_start = np.zeros(target_n, dtype=float)
            h_start[order[-target_n // 2:]] = 1.0

        # Simulated annealing
        sa_deadline = min(limit, time.time() + 2.0)  # ≤2 s per SA run
        h_sa, c_sa = simulated_annealing_binary(
            h_start, rng, sa_deadline, max_iters=30000,
            init_temp=0.03, final_temp=1e-5
        )
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

        # Deterministic binary hill‑climb refinement
        h_hc, c_hc = hill_climb_binary(h_sa, rng, max_iters=1500)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

        # Continuous SLSQP passes on the (possibly fractional) solution
        for beta in (5.0, 20.0, 80.0, 200.0):
            if time.time() > limit:
                break
            try:
                h_cont = slsqp_optimize(best_h, beta, rng, max_iter=250)
            except Exception:
                continue
            c_cont = compute_c5(h_cont)
            if c_cont < best_c:
                best_h, best_c = h_cont.copy(), c_cont

    # ------------------------------------------------------------------
    # Optional up‑sampling and refinement (if time permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.98:
        if best_n * 2 <= 300:
            h_up = upsample(best_h)
            n_up = h_up.size
            # a quick SA on the up‑sampled vector
            up_deadline = start_time + budget_s * 0.99
            h_up, c_up = simulated_annealing_binary(
                h_up, rng, up_deadline, max_iters=20000,
                init_temp=0.02, final_temp=1e-5
            )
            # final continuous polishing
            for beta in (5.0, 20.0, 80.0):
                if time.time() > up_deadline:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, rng, max_iter=200)
                except Exception:
                    pass
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final feasibility enforcement
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
