def make_feasible(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    # adjust sum
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # scale factor to approach target
        if s > 0:
            factor = target / s
        else:
            factor = 0.0
        v = v * factor
        v = np.clip(v, 0.0, 1.0)
    # final small correction via transfer
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # perform transfers
        n = len(v)
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            # capacity to add
            if diff > 0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            else:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
    return v

import time, numpy as np
from scipy.optimize import minimize

TARGET_HALF = None
CURRENT_BETA = None

def full_from_half(v):
    return np.concatenate([v, v[::-1]])

def compute_c5_from_h(h):
    # correlation of h and (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def compute_c5_half(v):
    h = full_from_half(v)
    return compute_c5_from_h(h)

def sum_eq_constraint_half(v):
    return np.sum(v) - TARGET_HALF

def softmax_obj_half(v):
    h = full_from_half(v)
    corr = np.correlate(h, 1 - h, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def enforce_sum_full(h, target, rng):
    """Adjust h in-place so that sum(h) == target, keeping 0<=h<=1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def enforce_sum_half(v, target, rng):
    diff = target - v.sum()
    n = v.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v

def random_feasible_half(n_half, target, rng):
    # start with uniform random numbers
    v = rng.random(n_half)
    s = v.sum()
    if s > 0:
        v *= target / s
    v = np.clip(v, 0.0, 1.0)
    # fix possible sum error
    v = enforce_sum_half(v, target, rng)

def perturb_half(v, rng, scale=0.2):

    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])

    best_c = compute_c5_half(v)

def slsqp_optimize_half(v0, beta, max_iter=500):
    global TARGET_HALF, CURRENT_BETA
    TARGET_HALF = v0.size * 0.5  # since target = n/4 = n_half/2

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 180
    n_half = n // 2
    target_half = n / 4.0  # 45.0

    # ----- Initial vector -----
    if "initial_h_values" in globals():

        # ensure correct length
        if h0.size != n:
            # if length differs, resample by linear interpolation to n points
            xs_old = np.linspace(0, 2, h0.size, endpoint=False) + (2.0/h0.size)/2
            xs_new = np.linspace(0, 2, n, endpoint=False) + (2.0/n)/2
            h0 = np.interp(xs_new, xs_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        # symmetrize
        h0 = 0.5 * (h0 + h0[::-1])
        h0 = enforce_sum_full(h0, n/2.0, rng)
        v0 = h0[:n_half]  # already symmetric, first half suffices
    else:
        v0 = random_feasible_half(n_half, target_half, rng)

    best_v = v0.copy()
    best_c = compute_c5_half(best_v)

    # Main optimization loop
    beta_schedule = [10.0, 40.0, 160.0, 640.0, 2560.0]
    # we keep track of iteration count for adaptive restarts
    while time.time() - start_time < budget_s * 0.9:

        if rng.random() < 0.65:
            start = perturb_half(best_v, rng, scale=0.35)
        else:
            start = random_feasible_half(n_half, target_half, rng)

        # hill climbing
        v_opt, c_opt = hill_climb_half(start, rng, max_iters=3500, start_step=0.6)

        # occasional swap refinement
        if rng.random() < 0.3:
            v_opt, c_opt = try_swaps(v_opt, rng, trials=300)

        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.9:

                continue
            c_opt = compute_c5_half(v_opt)
            if c_opt < best_c:
                best_c, best_v = c_opt, v_opt.copy()

        # record any improvement found
        if c_opt < best_c:
            best_c, best_v = c_opt, v_opt.copy()

    # Final fine‑grained hill‑climb on the best solution
    if time.time() - start_time < budget_s * 0.95:
        v_best, c_best = hill_climb_half(best_v, rng, max_iters=5000, start_step=0.3)
        if c_best < best_c:
            best_c, best_v = c_best, v_best

    # Ensure feasibility
    best_v = np.clip(best_v, 0.0, 1.0)
    best_v = enforce_sum_half(best_v, target_half, rng)

    # Build full vector for output
    h_final = full_from_half(best_v)
    final_c = compute_c5_from_h(h_final)

    return h_final.tolist(), float(final_c), int(n)

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP surrogate
TARGET_HALF = None
CURRENT_BETA = None


def full_from_half(v):
    """Construct the full symmetric vector from its first half."""
    return np.concatenate([v, v[::-1]])


def compute_c5_full(h):
    """Exact discrete C5 for a full vector h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_half(v):
    """C5 evaluated on a symmetric half‑vector."""
    return compute_c5_full(full_from_half(v))


def sum_eq_constraint_half(v):
    return np.sum(v) - TARGET_HALF


def softmax_obj_half(v):
    """Log‑sum‑exp smooth approximation of max_k Σ_i v_i(1‑v_{i+k}) on the symmetric vector."""
    h = full_from_half(v)
    corr = np.correlate(h, 1 - h, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum_full(h, target, rng):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def enforce_sum_half(v, target, rng):
    """Same as above but for a half‑vector."""
    diff = target - v.sum()
    n = v.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def random_feasible_half(n_half, target, rng):
    """Generate a random feasible half‑vector (0≤v≤1, Σv=target)."""
    v = rng.random(n_half)                     # uniform [0,1)
    s = v.sum()
    if s > 0:
        v *= target / s                         # scale to the right total
    v = np.clip(v, 0.0, 1.0)
    v = enforce_sum_half(v, target, rng)
    return v


def perturb_half(v, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = v.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0.0:
        # try the opposite direction
        max_eps = min(v[j], 1.0 - v[i])
        if max_eps <= 0.0:
            return v.copy()
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        eps = max(-max_eps, min(eps, max_eps))
        nv = v.copy()
        nv[j] -= eps
        nv[i] += eps
        return nv
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nv = v.copy()
    nv[i] -= eps
    nv[j] += eps
    return nv


def hill_climb_half(v_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    v = v_start.copy()
    best_v = v.copy()
    best_c = compute_c5_half(v)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_v = perturb_half(v, rng, scale=step)
        c = compute_c5_half(new_v)
        if c < best_c:
            best_c = c
            best_v = new_v.copy()
            v = new_v
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_v, best_c


def slsqp_optimize_half(v0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global TARGET_HALF, CURRENT_BETA
    TARGET_HALF = v0.size * 0.5         # n/4
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * v0.size
    cons = {"type": "eq", "fun": sum_eq_constraint_half}
    res = minimize(
        softmax_obj_half,
        v0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    v = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = TARGET_HALF - v.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        v = enforce_sum_half(v, TARGET_HALF, rng)
    return v


def try_swaps(v, rng, trials=200):
    """Randomly swap pairs of coordinates – a cheap diversification."""
    best_v = v.copy()
    best_c = compute_c5_half(v)
    n = v.size
    for _ in range(trials):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        nv = v.copy()
        nv[i], nv[j] = nv[j], nv[i]
        c = compute_c5_half(nv)
        if c < best_c:
            best_c = c
            best_v = nv
    return best_v, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a symmetric step function on [0,2] to minimise the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    n = 180                      # total number of samples
    n_half = n // 2
    target_half = n / 4.0        # each half must sum to 45

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            # resample to the required size
            xs_old = np.linspace(0, 2, h0.size, endpoint=False) + (2.0 / h0.size) / 2
            xs_new = np.linspace(0, 2, n, endpoint=False) + (2.0 / n) / 2
            h0 = np.interp(xs_new, xs_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = 0.5 * (h0 + h0[::-1])                 # enforce symmetry
        h0 = enforce_sum_full(h0, n / 2.0, rng)
        v0 = h0[:n_half]                           # first half suffices
    else:
        v0 = random_feasible_half(n_half, target_half, rng)

    best_v = v0.copy()
    best_c = compute_c5_half(best_v)

    # ----- main optimisation loop ------------------------------------
    beta_schedule = [10.0, 40.0, 160.0, 640.0, 2560.0]

    while time.time() - start_time < budget_s * 0.88:
        # choose a start point
        if rng.random() < 0.65:
            start = perturb_half(best_v, rng, scale=0.35)
        else:
            start = random_feasible_half(n_half, target_half, rng)

        # local hill‑climb
        v_opt, c_opt = hill_climb_half(start, rng, max_iters=3500, start_step=0.6)

        # occasional swap diversification
        if rng.random() < 0.3:
            v_opt, c_opt = try_swaps(v_opt, rng, trials=300)

        # SLSQP ladder with increasing β
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.88:
                break
            try:
                v_opt = slsqp_optimize_half(v_opt, beta, max_iter=400)
            except Exception:
                continue
            c_opt = compute_c5_half(v_opt)
            if c_opt < best_c:
                best_c, best_v = c_opt, v_opt.copy()

        if c_opt < best_c:
            best_c, best_v = c_opt, v_opt.copy()

    # ----- final fine tune -------------------------------------------
    if time.time() - start_time < budget_s * 0.96:
        v_opt, c_opt = hill_climb_half(best_v, rng, max_iters=5000, start_step=0.3)
        if c_opt < best_c:
            best_c, best_v = c_opt, v_opt

    # Ensure exact feasibility
    best_v = np.clip(best_v, 0.0, 1.0)
    best_v = enforce_sum_half(best_v, target_half, rng)

    h_final = full_from_half(best_v)
    final_c = compute_c5_full(h_final)

    return h_final.tolist(), float(final_c), int(n)
