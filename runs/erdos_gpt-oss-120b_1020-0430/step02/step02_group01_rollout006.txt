def project_onto_constraints(v, target):
    # Clip v to [0,1]
    v = np.clip(v, 0, 1)
    # Then need sum(v) = target while staying in [0,1].
    # Use algorithm for projection onto simplex with upper bound.
    # Equivalent to find lambda such that sum(min(1, max(0, v - lambda))) = target.
    # This is a root-finding on lambda.
    # Since sum is monotonic decreasing in lambda, we can binary search.
    low = -1.0  # lambda lower bound
    high = 1.0  # high bound
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2.0, 0, 1)

import time
import numpy as np
from scipy.optimize import minimize

# globals for soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def repair_vector(h, target, rng):
    """Clip to [0,1] and adjust sum to target by random moves."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    return repair_vector(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a random amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (random redistribution)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def worst_shift_and_gradient(h):
    """Return correlation, signed worst shift, and gradient of that shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)          # signed shift
    s = shift
    g = np.zeros_like(h)
    if s >= 0:
        g[: n - s] += 1 - h[s:]
        g[s:] -= h[: n - s]
    else:
        s = -s
        g[s:] += 1 - h[: n - s]
        g[: n - s] -= h[s:]
    return corr, shift, g


def subgradient_descent(h, rng, max_iters=80, step0=1.0, target=None):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h.size
    if target is None:
        target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(max_iters):
        corr, shift, g = worst_shift_and_gradient(h)
        step = step0 / np.sqrt(t + 1.0)
        h = h - step * g
        h = repair_vector(h, target, rng)
        if (t + 1) % 20 == 0:
            c = np.max(corr) * (2.0 / n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    # final check
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by integer factor, followed by feasibility repair."""
    n = h.size
    n_up = n * factor
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)

    h_up = np.interp(x_new, x_old, h)
    return repair_vector(h_up, n_up / 2.0, rng)
