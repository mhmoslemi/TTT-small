    while len(pop) < pop_size:
        # random uniform, then project to sum
        rand = np.random.rand(n)
        proj = _project_to_feasible(rand, n)
        pop.append(proj)

    pop = np.array(pop) # shape (pop_size, n)
    # compute fitness
    scores = np.array([_c5_from_h(ind) for ind in pop])
    best_idx = np.argmin(scores)
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    # DE hyperparameters
    F = 0.7
    CR = 0.9

        for i in range(pop_size):

def _project_to_feasible(h, n, tol=1e-12, max_iter=500):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    # Quick check
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final fix residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual to non-saturated entries
        free = ( (residual > 0) & (h < 1) ) | ( (residual < 0) & (h > 0) )
        idx = np.where(free)[0]
        if idx.size > 0:
            # distribute equally
            inc = residual / idx.size
            h[idx] += inc
            h = np.clip(h, 0.0, 1.0)
            # may still have tiny residual, just ignore
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h):
    n = h.size
    # correlation h * (1-h shifted) across full lags (from -n+1 to n-1)
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _local_refine(h, attempts=20, delta=0.02):

import time
import numpy as np
from scipy import optimize

# ---------- Feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 500) -> np.ndarray:
    """Clamp to [0,1] and adjust sum to n/2 using water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0.0
        else:
            free = h < 1.0
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        free = ((residual > 0) & (h < 1.0)) | ((residual < 0) & (h > 0.0))
        idx = np.where(free)[0]
        if idx.size:
            inc = residual / idx.size
            h[idx] += inc
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


# ---------- Objective ----------
def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 for discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- Simple up‑sampling ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- Local stochastic refinement ----------
def _local_refine(h: np.ndarray, attempts: int = 20, delta: float = 0.02) -> np.ndarray:
    n = h.size
    best = h.copy()
    best_val = _c5_from_h(best)
    for _ in range(attempts):
        k = max(2, int(0.04 * n))
        idx = np.random.choice(n, size=k, replace=False)
        trial = best.copy()
        trial[idx] += delta * np.random.randn(k)
        trial = _project_to_feasible(trial, n)
        val = _c5_from_h(trial)
        if val < best_val:
            best, best_val = trial, val
    return best


# ---------- SLSQP polishing ----------
def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def obj(x):
        return x[-1]

    cons = [{"type": "ineq", "fun": ineq},
            {"type": "eq", "fun": eq}]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(obj, x0, method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)

    if res.success and res.x is not None:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- Differential Evolution ----------
def _de_optimize(initial_h: np.ndarray, n: int, time_limit: float) -> tuple:
    start = time.time()
    pop_size = 12 if n <= 128 else 16
    pop = []
    if initial_h is not None:
        pop.append(_project_to_feasible(initial_h.copy(), n))
    while len(pop) < pop_size:
        rand = np.random.rand(n)
        pop.append(_project_to_feasible(rand, n))
    pop = np.array(pop)
    scores = np.array([_c5_from_h(v) for v in pop])
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.6
    CR = 0.9

    while time.time() - start < time_limit:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = _project_to_feasible(mutant, n)

            cross = np.random.rand(n) < CR
            if not np.any(cross):
                cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _project_to_feasible(trial, n)

            trial_score = _c5_from_h(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best, best_score = trial.copy(), trial_score
        if best_score < 0.3805:
            break
    return best, best_score
