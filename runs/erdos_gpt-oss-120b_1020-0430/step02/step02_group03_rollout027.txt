def hill_climb_a(a, rng, max_iters, start_scale):
    best_a = a.copy()
    best_c = compute_c5_full(np.concatenate([best_a, 1-best_a]))
    scale = start_scale
    decay = max_iters // 10
    for it in range(max_iters):
        i = rng.integers(0, a.size)
        eps = (rng.random()*2 - 1) * scale
        new_val = best_a[i] + eps
        new_val = np.clip(new_val, 0.0, 1.0)
        if new_val == best_a[i]:
            continue
        new_a = best_a.copy()
        new_a[i] = new_val
        c = compute_c5_full(np.concatenate([new_a, 1-new_a]))
        if c < best_c:
            best_c = c
            best_a = new_a
        if (it+1) % decay == 0:
            scale *= 0.7
            if scale < 1e-4:
                scale = 1e-4
    return best_a, best_c
```

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    N = 180
    M = N // 2

    # Build initial a
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # Ensure length N
        if h0.size != N:
            # truncate or pad (by repeating)
            if h0.size > N:
                h0 = h0[:N]
            else:
                # pad with 0.5 to fill length
                pad = np.full(N - h0.size, 0.5)
                h0 = np.concatenate([h0, pad])
        a0 = np.clip(h0[:M], 0.0, 1.0)
    else:
        a0 = rng.random(M)
    # Ensure a0 in bounds (0,1) already.

    # Function for DE
    def obj(a):
        h = np.concatenate([a, 1-a])
        return np.max(np.correlate(h, 1-h, mode='full')) * (2.0 / h.size)

    # Run DE
    bounds = [(0.0, 1.0)] * M
    # Provide initial population with a0 as one candidate
    # SciPy DE does not allow custom init, but can use init='latinhypercube', okay.

    de_result = differential_evolution(
        obj,
        bounds,
        seed=int(rng.integers(0, 2**32-1)),
        maxiter=150,
        popsize=10,
        polish=False,
        updating='deferred',
    )
    a_best = de_result.x
    c_best = obj(a_best)

    # Hill climbing refinement
    remaining_time = budget_s - (time.time() - start_time)
    # allocate half of remaining time for hill climbing, ensure enough loops.
    max_iters = int(3e5)  # huge but break by time
    # but we must break early if time exceed
    loop = 0
    scale = 0.5
    decay_interval = max(1, max_iters // 10)
    while time.time() - start_time < budget_s * 0.99 and loop < max_iters:
        i = rng.integers(0, M)
        eps = (rng.random()*2 - 1) * scale
        new_val = a_best[i] + eps
        if new_val < 0.0:
            new_val = 0.0
        elif new_val > 1.0:
            new_val = 1.0
        if new_val == a_best[i]:
            loop += 1
            continue
        a_new = a_best.copy()
        a_new[i] = new_val
        c_new = obj(a_new)
        if c_new < c_best:
            a_best = a_new
            c_best = c_new
        loop += 1
        if loop % decay_interval == 0:
            scale *= 0.7
            if scale < 1e-4:
                scale = 1e-4
    # After refinement, produce final h
    h_best = np.concatenate([a_best, 1-a_best])
    h_best = np.clip(h_best, 0.0, 1.0)
    final_c = obj(h_best)
    return h_best.tolist(), float(final_c), int(N)

            it += 1
            continue
        new_a = best_a.copy()
        new_a[i] = new_val
        c = obj_a(new_a)
        if c < best_c:
            best_c = c
            best_a = new_a
        it += 1
        if it % decay_interval == 0:
            scale = max(scale * 0.7, 1e-6)
    return best_a, best_c


def sum_eq_constraint_full(x: np.ndarray) -> float:
    """Equality constraint Σx = CURRENT_TARGET."""
    global CURRENT_TARGET
    return np.sum(x) - CURRENT_TARGET


def softmax_obj_full(x: np.ndarray) -> float:
    """Smooth log‑sum‑exp surrogate of max ∑ xᵢ(1‑xᵢ₊ₖ)."""
    global CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA, GLOBAL_N
    GLOBAL_N = h0.size
    CURRENT_TARGET = GLOBAL_N / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * GLOBAL_N
    cons = {"type": "eq", "fun": sum_eq_constraint_full}
    res = minimize(
        softmax_obj_full,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny drift correction to satisfy the exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, GLOBAL_N)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise a step function for the Erdős C₅ bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------
    # 1. Choose number of points and initialise a
    # -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N = h0.size
        if N % 2 != 0:          # enforce even length
            h0 = h0[:-1]
            N -= 1
        if N < 20:              # fallback if the parent is tiny
            N = 180
            h0 = None
    else:
        N = 180
        h0 = None

    if N % 2 != 0:
        N += 1
    M = N // 2

    if h0 is not None:
        a0 = np.clip(h0[:M], 0.0, 1.0)
    else:
        a0 = rng.random(M)

    # -------------------------------------------------
    # 2. Global search with Differential Evolution
    # -------------------------------------------------
    bounds = [(0.0, 1.0)] * M
    maxiter = 500          # generous but still well within the budget
    popsize = 12
    try:
        de_res = differential_evolution(
            obj_a,
            bounds,
            seed=int(rng.integers(0, 2**32 - 1)),
            maxiter=maxiter,
            popsize=popsize,
            polish=False,
            updating="deferred",
        )
        a_best = de_res.x
    except Exception:
        a_best = a0.copy()

    # -------------------------------------------------
    # 3. Local stochastic hill‑climb
    # -------------------------------------------------
    deadline = start_time + budget_s * 0.96
    a_best, _ = hill_climb_a(a_best, rng, start_time, deadline)

    # -------------------------------------------------
    # 4. Build full h and polish with SLSQP surrogate
    # -------------------------------------------------
    h_best = np.concatenate([a_best, 1 - a_best])
    beta_list = [5.0, 20.0, 80.0, 200.0]
    for beta in beta_list:
        if time.time() - start_time > budget_s * 0.98:
            break
        try:
            h_best = slsqp_refine(h_best, beta, max_iter=400)
        except Exception:
            continue

    # -------------------------------------------------
    # 5. Final feasibility clean‑up
    # -------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    target_sum = N / 2.0
    diff = target_sum - h_best.sum()
    if abs(diff) > 1e-12:
        # simple deterministic correction
        for i in range(N):
            if diff > 0 and h_best[i] < 1.0:
                add = min(diff, 1.0 - h_best[i])
                h_best[i] += add
                diff -= add
            elif diff < 0 and h_best[i] > 0.0:
                sub = min(-diff, h_best[i])
                h_best[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    final_c = compute_c5_full(h_best)
    return h_best.tolist(), float(final_c), int(N)
