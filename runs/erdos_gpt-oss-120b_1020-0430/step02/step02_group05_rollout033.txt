def _smooth_optimize(h0, target_sum, max_iters, time_left):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_from_h(h)
    lam = 0.5
    lam_factor = 1.03
    lam_max = 20.0
    step = 0.2
    step_decay = 0.998
    # Precompute maybe 1.0 - h each iteration needed for corr. We'll compute as needed.
    for it in range(max_iters):
        if time_left() < 0.5:
            break
        # compute corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        # use lam to compute weights
        max_corr = corr.max()
        w = np.exp(lam * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # compute conv1 and conv2
        conv1 = fftconvolve(w, h, mode='full')
        conv2 = fftconvolve(w[::-1], h, mode='full')
        offset = len(w) - 1
        conv1_center = conv1[offset: offset + n]
        conv2_center = conv2[offset: offset + n]
        grad = 1.0 - conv1_center - conv2_center
        # update
        h = h - step * grad
        # project onto feasible set
        h = _project_to_simplex_box(h, target_sum)
        # update lam and step
        lam = min(lam * lam_factor, lam_max)
        step *= step_decay
        # evaluate periodically
        if (it % 20) == 0:
            val = _c5_from_h(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                # maybe reset lam and step? not needed
    # final evaluation
    final_val = _c5_from_h(best_h)
    return best_h, final_val

import time
import numpy as np
from scipy import optimize
from scipy.signal import fftconvolve  # not used directly but imported for safety


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum x_i = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _tent_h(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    return _project_to_simplex_box(h, n / 2.0)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(start_h: np.ndarray, max_iter: int = 300):
    """Polish a candidate using SLSQP (few iterations)."""
    n = start_h.size
    h0 = _project_to_simplex_box(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq_constraint(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr

    def _eq_constraint(x):
        h = x[:n]
        return np.sum(h) - n / 2.0

    def _objective(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    h_opt = h0 if not res.success else res.x[:n]
    h_opt = _project_to_simplex_box(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


def _smooth_optimize(h0: np.ndarray, target_sum: float, max_iters: int, time_left):
    """Projected gradient descent on a soft‑max approximation of the max‑overlap."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_from_h(h)
    lam = 0.5                # start temperature
    lam_factor = 1.03
    lam_max = 20.0
    step = 0.2               # initial step size
    step_decay = 0.998

    for it in range(max_iters):
        if time_left() < 0.3:
            break
        # correlation (full) for the soft‑max weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        # soft‑max probabilities
        exp_arg = lam * (corr - max_corr)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:               # extreme under‑flow safeguard
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # convolution to obtain Σ_s w_s h_{j‑s} and Σ_s w_s h_{j+s}
        conv_full1 = np.convolve(w, h)
        conv_full2 = np.convolve(w[::-1], h)
        offset = n - 1
        conv1 = conv_full1[offset: offset + n]
        conv2 = conv_full2[offset: offset + n]

        grad = 1.0 - conv1 - conv2
        h = h - step * grad
        h = _project_to_simplex_box(h, target_sum)

        # annealing and step decay
        lam = min(lam * lam_factor, lam_max)
        step *= step_decay

        if (it & 15) == 0:  # evaluate every 16 steps
            val = _c5_from_h(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()

    final_val = _c5_from_h(best_h)
    return best_h, final_val

    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # try to obtain the parent construction if present

    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None
    prev_best_h = None
    prev_n = None

    # candidate grid sizes (coarse → fine)
    cand_n = [64, 96, 128, 160, 192, 224, 256, 320, 384]
    if init_h is not None:
        cand_n.append(init_h.size)
    cand_n = sorted(set(cand_n))

    for n in cand_n:
        if time_left() < 5.0:
            break

        # build starting function for this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif prev_best_h is not None:
            base_h = _upscale_h(prev_best_h, n)
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        target_sum = n / 2.0

        # phase 1 – deterministic smooth optimisation
        max_iters = max(2000, n * 12)
        h_opt, c5_val = _smooth_optimize(base_h, target_sum, max_iters, time_left)
        if c5_val < best_c5:

            prev_best_h, prev_n = best_h, n
            if best_c5 < target_bound:
                break

        # phase 2 – random restarts with decreasing noise
        attempts = 8
        sigma = 0.18
        for _ in range(attempts):
            if time_left() < 4.0:
                break
            start = base_h + sigma * np.random.randn(n)
            start = _project_to_simplex_box(start, target_sum)
            h_opt, c5_val = _smooth_optimize(start, target_sum, max_iters // 2, time_left)

                if refined_c5 < best_c5:
                    best_h, best_c5 = refined_h, refined_c5
                    if best_c5 < target_bound:

            except Exception:
                pass

            break

    # fallback safety net
    if best_h is None:
        n = 128
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
