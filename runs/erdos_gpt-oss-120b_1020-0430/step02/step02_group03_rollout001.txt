import time, numpy as np
from scipy.optimize import minimize, differential_evolution

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum adjust
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # clip eps
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # occasionally try random restart? not needed
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# Global placeholders used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng):
    """Adjust h in place so that Σh = target while 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact C₅ for the discrete step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---- SLSQP surrogate helpers -------------------------------------------------
def sum_eq_constraint(x):
    """Equality constraint Σx_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate of max_k Σ_i x_i (1‑x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny numerical drift – enforce the sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---- Simple hill‑climb -------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---- Main optimisation routine -----------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------
    # 1. Base dimension and initial vector
    # -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
        n = h0.size
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 180
        h0 = np.full(n, 0.5)
        h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # -------------------------------------------------
    # 2. Symmetric search via differential evolution
    # -------------------------------------------------
    if n % 2 == 0:
        m = n // 2

        def obj_half(a):
            # a has length m, build symmetric h with h_i + h_{n-1-i}=1
            h = np.concatenate([a, 1.0 - a[::-1]])
            return compute_c5(h)

        bounds = [(0.0, 1.0)] * m

        # callback stops early when time is nearly exhausted
        def de_callback(xk, convergence):
            return (time.time() - start_time) > budget_s * 0.78

        # modest population and iteration count – cheap but expressive
        try:
            result = differential_evolution(
                obj_half,
                bounds,
                maxiter=250,
                popsize=6,
                seed=seed,
                polish=False,
                callback=de_callback,
                updating="deferred",
                workers=1,
            )
        except Exception:
            result = None

        if result is not None and hasattr(result, "x"):
            a_best = result.x
            h_sym = np.concatenate([a_best, 1.0 - a_best[::-1]])
            c_sym = compute_c5(h_sym)
            if c_sym < best_c:
                best_h, best_c = h_sym, c_sym

    # -------------------------------------------------
    # 3. Local refinement loop (hill climb + SLSQP)
    # -------------------------------------------------
    while time.time() - start_time < budget_s * 0.92:
        # a quick hill‑climb from the current best
        h_opt, c_opt = hill_climb(best_h, rng, max_iters=3500, start_step=0.3)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

        # finer SLSQP polishing with increasing β
        for beta in (5.0, 20.0, 80.0, 200.0):
            if time.time() - start_time > budget_s * 0.95:
                break
            h_opt = slsqp_optimize(best_h, beta, max_iter=600)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

        # occasional random‑restart diversification
        if rng.random() < 0.25:
            h_rand = np.full(n, 0.5)
            for _ in range(5 * n):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                max_eps = min(h_rand[i], 1.0 - h_rand[j])
                if max_eps <= 0.0:
                    continue
                eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.4
                h_rand[i] -= eps
                h_rand[j] += eps
            h_rand = enforce_sum(h_rand, n / 2.0, rng)
            h_tmp, c_tmp = hill_climb(h_rand, rng, max_iters=2500, start_step=0.5)
            if c_tmp < best_c:
                best_h, best_c = h_tmp, c_tmp

    # -------------------------------------------------
    # 4. Final clean‑up
    # -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
