def subgradient_opt(h_init, max_iter, step0, decay, min_step=1e-6):
    h = h_init.copy()
    best_h, best_c5 = h, compute_c5(h)
    step = step0
    n = h.size
    for it in range(max_iter):
        # compute gradient for max shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idxs = np.where(corr == max_corr)[0]
        idx = idxs[0]  # pick first
        shift = idx - (n - 1)
        # gradient
        grad = np.ones(n)
        if shift >= 0:
            s = shift
            if s < n:
                grad[:n - s] -= h[s:]
                grad[s:] -= h[:n - s]
        else:
            d = -shift
            if d < n:
                grad[d:] -= h[:n - d]
                grad[:n - d] -= h[d:]
        # take step
        h_new = h - step * grad
        h_new = _project_to_feasible(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-12:
            best_h, best_c5 = h_new.copy(), new_c5
            h = h_new
            # optional step increase?
            step *= 1.02  # slight increase?
        else:
            # reject move? (like gradient descent)
            # could still keep h_new to explore.
            # but if not improving, we may reduce step.
            step *= decay
        if step < min_step:
            break
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# -------------------- Core utilities -------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 400) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C5 from step vector h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> np.ndarray:
    """Run a brief SLSQP optimisation starting from `initial_h`."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.x is not None else h0
    return _project_to_feasible(h_opt, n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _gradient_descent_opt(h_init: np.ndarray, max_iter: int, time_check=None):
    """Projected sub‑gradient descent on max‑correlation."""
    n = h_init.size
    h = _project_to_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    step = 0.5
    decay = 0.85
    min_step = 1e-5

    for _ in range(max_iter):
        if time_check is not None and time_check() <= 0:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        shift = idx - (n - 1)

        # sub‑gradient of the active shift
        grad = np.ones(n)
        if shift >= 0:
            s = shift
            if s < n:
                grad[: n - s] -= h[s:]
                grad[s:] -= h[: n - s]
        else:
            d = -shift
            if d < n:
                grad[d:] -= h[: n - d]
                grad[: n - d] -= h[d:]

        # trial move
        h_new = _project_to_feasible(h - step * grad, n)
        c5_new = _c5_from_h(h_new)

        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new.copy(), c5_new
            h = h_new
            step = min(step / decay, 1.0)   # enlarge a bit after success
        else:
            step *= decay

        if step < min_step:
            # occasional random shake to escape flat regions
            d = np.random.randn(n)
            d -= d.mean()
            sigma = 0.3
            h_shake = _project_to_feasible(h + sigma * d, n)
            c5_shake = _c5_from_h(h_shake)
            if c5_shake < best_c5:
                h, best_h, best_c5 = h_shake, h_shake.copy(), c5_shake
            step = 0.5
    return best_h, best_c5


# -------------------- Main driver -------------------- #

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise the Erdős C5 bound and return (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Grab parent construction if provided.
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    target = 0.3808  # early‑stop threshold
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Candidate grid sizes (focus on even sizes up to 256)
    n_candidates = []
    if parent_h is not None:
        n_candidates.append(parent_h.size)
    for mul in (1, 2, 3, 4):
        if parent_h is not None:
            n = int(parent_h.size * mul)
        else:
            n = 64 * mul
        if n % 2 == 0 and n <= 256:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 6.0:
            break

        # Build a base vector for this size.
        if parent_h is not None and parent_h.size != n:
            base_h = _upscale_h(parent_h, n)
        elif parent_h is not None:
            base_h = parent_h.copy()
        else:
            # half‑ones random binary start
            base_h = np.zeros(n)
            idx = np.random.choice(n, n // 2, replace=False)
            base_h[idx] = 1.0

        # ---- 1) quick SLSQP pass ----
        h_slsqp = _solve_h(base_h, max_iter=300)
        h_opt, c5_opt = _gradient_descent_opt(
            h_slsqp, max_iter=2500, time_check=time_left
        )
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < target:
                break

        # ---- 2) stochastic restarts with gradient descent ----
        # Rough budget for extra attempts
        max_attempts = int(min(80, max(5, time_left() // 0.12)))
        for _ in range(max_attempts):
            if time_left() < 5.0:
                break
            # Random feasible start
            rnd = np.random.rand(n)
            rnd = _project_to_feasible(rnd, n)
            h_cur, c5_cur = _gradient_descent_opt(
                rnd, max_iter=1200, time_check=time_left
            )
            if c5_cur < best_c5:
                best_h, best_c5, best_n = h_cur, c5_cur, n
                if best_c5 < target:
                    break

        if best_c5 < target:
            break

    # Fallback if nothing succeeded
    if best_h is None:
        n = 96
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
