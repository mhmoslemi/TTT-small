def optimize_h(initial_h, n, time_limit):
    h = project(initial_h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    start = time.time()
    iteration = 0
    eta = 0.5  # initial step size
    while time.time() < time_limit:
        # compute correlation
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)  # shift amount
        # compute gradient for that shift
        grad = np.zeros_like(h)
        dx = 2.0 / n
        if k >= 0:
            # A: 1 - h[i+k] for i=0..n-1-k
            grad[:n-k] += (1 - h[k:]) * dx
            # B: h[i - k] for i=k..n-1
            grad[k:] -= h[:n-k] * dx
        else:
            kk = -k
            # shift negative: i+k = i - kk
            grad[kk:] += (1 - h[:n-kk]) * dx
            grad[:n-kk] -= h[kk:] * dx
        # gradient ascent: we want to reduce max correlation -> decrease C_k
        # gradient of C_k is grad; to reduce C_k we move opposite to grad.
        h_new = h - eta * grad
        # Clip and project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_feasible(h_new, n)  # using same water-filling with target sum n/2
        c_new = compute_c5(h_new)
        # accept if improvement else maybe accept with probability?
        if c_new < best_c:
            best_h, best_c = h_new, c_new
            h = h_new
        else:
            # maybe accept with probability
            # compute diff
            diff = c_new - best_c
            T = max(0.01, (time_limit - time.time()) / (time_limit - start))  # simple schedule
            # compute probability
            if np.random.rand() < np.exp(-diff / (0.001 + T)):
                h = h_new

                # maybe revert to best
                h = best_h

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 value for a given discrete h on [0,2]."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _project_to_feasible(x: np.ndarray, n: int, target: float = None) -> np.ndarray:
    """
    Project vector x onto the feasible set:
        0 ≤ x_i ≤ 1  and  sum(x) = target (default n/2).
    Uses a simple bisection on the Lagrange multiplier λ for the box‑constrained projection.
    """
    if target is None:
        target = n / 2.0
    # Bisection on λ such that sum(clip(x - λ,0,1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction (should be rare)
    res = target - y.sum()
    if abs(res) > 1e-12:
        free = np.where((y > 0.0) & (y < 1.0))[0]
        if free.size:
            y[free] += res / free.size
            y = np.clip(y, 0.0, 1.0)
    return y


# ----------------------------------------------------------------------
# Gradient‑descent optimizer for the max‑correlation objective
# ----------------------------------------------------------------------
def _optimize_h(initial_h: np.ndarray, n: int, deadline: float) -> tuple:
    """
    Reduce the maximal overlap by a projected gradient descent on the
    most violating shift. Returns (h_opt, c5_val).
    """
    h = _project_to_feasible(initial_h, n)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    dx = 2.0 / n
    eta = 0.5                # initial step size
    it = 0

    while time.time() < deadline:
        # correlation for all integer shifts
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)                # shift amount (may be negative)

        # gradient of the worst shift
        grad = np.zeros_like(h)
        if k >= 0:
            grad[: n - k] += (1.0 - h[k:]) * dx
            grad[k:]      -= h[: n - k] * dx
        else:
            kk = -k
            grad[kk:]     += (1.0 - h[: n - kk]) * dx
            grad[: n - kk] -= h[kk:] * dx

        # projected gradient step
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_feasible(h_new, n)

        c_new = _c5_from_h(h_new)

        # accept if improvement, otherwise keep current (no uphill moves)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
            h = h_new
            eta = min(1.0, eta * 1.04)      # modestly increase step when progressing
        else:
            # reduce step size after non‑improvement
            eta = max(1e-4, eta * 0.96)

        it += 1

        # occasional random perturbation to escape stagnation
        if it % 250 == 0 and time.time() < deadline:
            if np.random.rand() < 0.25:
                noise = (np.random.rand(n) - 0.5) * 0.2   # ±0.1 amplitude
                h_tmp = np.clip(h + noise, 0.0, 1.0)
                h_tmp = _project_to_feasible(h_tmp, n)
                c_tmp = _c5_from_h(h_tmp)
                if c_tmp < best_c:
                    best_c, best_h, h = c_tmp, h_tmp.copy(), h_tmp
                    eta = min(1.0, eta * 1.02)

    return best_h, best_c


# ----------------------------------------------------------------------
# Simple pairwise hill climber for fine refinement
# ----------------------------------------------------------------------
def _hill_climb(h: np.ndarray, steps: int, deadline: float) -> tuple:
    """Small ε exchanges between two coordinates, accepting only improvements."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)

    while steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]          # how much we can take from i
        max_up = 1.0 - best_h[j]      # how much we can give to j
        eps = np.random.rand() * min(max_down, max_up) * 0.5
        if eps <= 0.0:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
        steps -= 1
    return best_h, best_c
