def gradient_transfer_descent(h, rng, max_iters=2000, step_frac=0.5):
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (h.size - 1)
        k = idx
        g = grad_for_shift(h, k)  # gradient of overlap O(k)
        # find indices with positive and negative gradients
        pos_idx = np.argmax(g)  # max positive
        neg_idx = np.argmin(g)  # most negative
        if g[pos_idx] <= 0 or g[neg_idx] >= 0:
            break  # gradient signs not opposite => optimum?
        max_transfer = min(h[pos_idx], 1 - h[neg_idx])
        if max_transfer <= 0:
            break
        delta = max_transfer * step_frac
        h[pos_idx] -= delta
        h[neg_idx] += delta
        # compute new worst shift; if no improvement, might reduce step size.
        # optional early break if improvement stagnates.
    return h

def grad_for_shift(h, k):
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        # i in 0..n-1-k: derivative w.r.t h_i: 1 - h_{i+k}
        g[:n - k] += 1.0 - h[k:]
        # derivative w.r.t h_{i+k}: -h_i
        g[k:] -= h[:n - k]
    else:
        k = -k
        # For negative shift, O(-k) = sum_{i=0}^{n-1-k} h_{i+k} (1 - h_i)
        # derivative w.r.t h_i: -h_{i+k}
        g[:n - k] -= h[k:]
        # derivative w.r.t h_{i+k}: 1 - h_i
        g[k:] += 1.0 - h[:n - k]
    return g
```

def gradient_transfer_descent(h, rng, max_iters=2000, init_step=0.5):
    h = h.copy()
    c = compute_c5(h)
    n = h.size
    step_factor = init_step
    for it in range(max_iters):
        # stop if time? We'll handle externally
        # compute gradient for worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = grad_for_shift(h, k)
        # positive and negative gradient indices
        pos = np.argmax(g)
        neg = np.argmin(g)
        if g[pos] <= 0 or g[neg] >= 0:
            break
        max_transfer = min(h[pos], 1.0 - h[neg])
        if max_transfer <= 0:
            break
        improved = False
        # try decreasing transfer factors
        for factor in (step_factor, step_factor*0.5, step_factor*0.25, step_factor*0.125):
            delta = max_transfer * factor
            if delta <= 0:
                continue
            h_new = h.copy()
            h_new[pos] -= delta
            h_new[neg] += delta
            c_new = compute_c5(h_new)
            if c_new < c:
                h = h_new
                c = c_new
                improved = True
                break
        if not improved:
            # reduce step factor and continue
            step_factor *= 0.5
            if step_factor < 1e-4:
                break
    return h, c
```

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb(h, rng, max_iters=2000, start_step=0.5):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global vars used by the soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def grad_for_shift(h, k):
    """Gradient of overlap O(k) = Σ h_i (1‑h_{i+k}) w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        g[: n - k] += 1.0 - h[k:]
        g[k:] -= h[: n - k]
    else:  # k negative
        k = -k
        g[: n - k] -= h[k:]
        g[k:] += 1.0 - h[: n - k]
    return g


def make_feasible(h, rng):
    """Clip to [0,1] and enforce Σ h_i = n/2 by random redistribution."""
    np.clip(h, 0.0, 1.0, out=h)
    target = h.size / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, h.size)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

    """Random continuous feasible vector (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()

            diff = target - h.sum()
    return h


def binary_random_vector(n, rng):
    """Binary feasible vector with exactly n/2 ones."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """Move a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h, rng, max_iters=1500, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c


def gradient_transfer_descent(h, rng, max_iters=2000, init_step=0.5):
    """
    Reduce the worst shift by transferring mass from a positive‑gradient
    coordinate to a negative‑gradient coordinate.
    """
    h = h.copy()
    c = compute_c5(h)
    n = h.size
    step_factor = init_step
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = grad_for_shift(h, k)
        pos = np.argmax(g)  # largest positive gradient
        neg = np.argmin(g)  # most negative gradient
        if g[pos] <= 0 or g[neg] >= 0:
            break
        max_transfer = min(h[pos], 1.0 - h[neg])
        if max_transfer <= 0:
            break
        improved = False
        for factor in (step_factor, step_factor * 0.5, step_factor * 0.25, step_factor * 0.125):
            delta = max_transfer * factor
            if delta <= 0:
                continue
            h_new = h.copy()
            h_new[pos] -= delta
            h_new[neg] += delta
            c_new = compute_c5(h_new)
            if c_new < c:
                h = h_new
                c = c_new
                improved = True
                break
        if not improved:
            step_factor *= 0.5
            if step_factor < 1e-4:
                break
    return h, c


def binarize_top_n(h):
    """Return a binary vector with exactly n/2 ones (top values become 1)."""
    n = h.size
    N = n // 2
    idx = np.argpartition(-h, N - 1)[:N]
    bin_vec = np.zeros_like(h)
    bin_vec[idx] = 1.0
    return bin_vec


def binary_hill_climb(h_bin, rng, max_trials=20000):
    """Swap a 1 with a 0 to reduce C5 (pure binary local search)."""
    current = h_bin.copy()
    current_c = compute_c5(current)
    n = current.size
    ones = np.where(current == 1)[0]
    zeros = np.where(current == 0)[0]
    for _ in range(max_trials):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        current[i], current[j] = 0.0, 1.0
        c_new = compute_c5(current)
        if c_new < current_c:
            current_c = c_new
            # update sets
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
        else:
            # revert
            current[i], current[j] = 1.0, 0.0
    return current, current_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            i = rng.integers(0, n)

                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
