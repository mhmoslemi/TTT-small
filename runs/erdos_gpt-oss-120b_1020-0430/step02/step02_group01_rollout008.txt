def simulated_annealing(h, n_iter, init_temp, final_temp, rng):
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy(); best_c = cur_c
    for i in range(n_iter):
        t = init_temp * ((final_temp / init_temp) ** (i / n_iter))  # exponential cooling.
        # propose swap:
        idx1 = rng.choice(np.where(cur == 1)[0])
        idx0 = rng.choice(np.where(cur == 0)[0])
        # swap:
        new = cur.copy()
        new[idx1] = 0
        new[idx0] = 1
        # compute score:
        new_c = compute_c5(new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / t):
            cur = new
            cur_c = new_c
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
        # maybe early break if time near limit
    return best, best_c

def binary_local_search(h, rng, max_swaps=2000):
    n = len(h)
    cur_c = compute_c5(h)
    for _ in range(max_swaps):
        # choose random 1-index and 0-index
        one_idx = rng.choice(np.where(h==1)[0])
        zero_idx = rng.choice(np.where(h==0)[0])
        # swap
        h[one_idx], h[zero_idx] = 0, 1
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            # keep swap
        else:
            # revert
            h[one_idx], h[zero_idx] = 1, 0
    return h, cur_c

import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum_and_bounds(h, target_sum, rng, max_iter=2000):
    """Clip to [0,1] and repair the linear sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 0:
            h[i] += add
            diff = target_sum - h.sum()
    return h


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Single SLSQP optimisation of the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # precise sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Gradient descent on the worst shift (continuous version)."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k]) for i< n-k
            g[k:] -= h[:upper]                # - h[i] for i>=k
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def binary_local_search_random_swaps(h, rng, max_swaps=2000):
    """Improve a binary vector by swapping a 1 with a 0 only if it lowers C5."""
    cur_c = compute_c5(h)
    for _ in range(max_swaps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # swap
        h[i1] = 0.0
        h[i0] = 1.0
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
        else:
            # revert
            h[i1] = 1.0
            h[i0] = 0.0
    return h, cur_c


def binary_target_swap_optimize(h, rng, max_iters=200):
    """
    Focused swaps: swap a (1,0) pair that contributes to the current worst shift.
    Stops when no improving swap is found.
    """
    n = h.size
    cur_c = compute_c5(h)
    for _ in range(max_iters):
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(conv)
        shift = idx - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0 or shift >= n:
            break
        cand = np.where((h[: n - shift] == 1) & (h[shift:] == 0))[0]
        if cand.size == 0:
            break

        j = i + shift
        h[i], h[j] = h[j], h[i]   # swap
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
        else:
            h[i], h[j] = h[j], h[i]   # revert
            break
    return h, cur_c
