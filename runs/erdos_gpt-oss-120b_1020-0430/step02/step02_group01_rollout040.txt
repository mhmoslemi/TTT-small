import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def clip_and_repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Simple greedy fix
    if abs(diff) > 1e-12:
        # Identify indices with room to add or remove
        if diff > 0:
            # need to add mass
            # order by how far from 1
            idx = np.argsort(1 - h)  # smallest capacity first? Actually we want ones with space
            for i in idx:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            # need to remove mass
            idx = np.argsort(h)  # smallest values first ensure removal from large ones? Actually need capacity to subtract; choose largest h to remove
            for i in idx[::-1]:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
        # If still not exact, perform random adjustments
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 2000:
            i = rng.integers(0, h.size)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff = target - h.sum()
            attempts += 1
    return h

def random_feasible_vector(n, rng):
    # generate random numbers, scale to target, then clip and repair
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # add/sub randomly
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
    return h

def perturb_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    # transfer epsilon from i to j, preserving bounds
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_transfer(h, rng, scale=step)
        # no need to reproject sum because perturb_transfer preserves sum
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        # optionally occasionally accept slightly worse? Not needed.
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # use log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol":1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum exactly
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
            diff = target - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=8):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip and repair sum
        target = n / 2.0
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the soft‑max surrogate for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def clip_and_repair_sum(h, target, rng):
    """Clip to [0,1] and adjust to meet the exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # greedy repair: add to places with capacity, remove from large entries
        if diff > 0:
            idx = np.argsort(1 - h)                     # most room first
            for i in idx:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            idx = np.argsort(h)                         # largest first
            for i in idx[::-1]:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
        # fallback random adjustments if still off
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 2000:
            i = rng.integers(0, h.size)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
            attempts += 1
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
    return h


def perturb_transfer(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        return h
    # pick a signed amount within [-max_transfer*scale, max_transfer*scale]
    eps = (rng.random() - 0.5) * 2.0 * max_transfer * scale
    eps = np.clip(eps, -max_transfer, max_transfer)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements) with feasibility repair."""
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_transfer(h, rng, scale=step)
        new_h = clip_and_repair_sum(new_h, target, rng)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
            diff = target - x.sum()
    return x


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift, with feasibility repair."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = clip_and_repair_sum(h, target, rng)
    return h
