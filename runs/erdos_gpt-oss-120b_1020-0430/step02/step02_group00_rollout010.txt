def smooth_obj_and_grad(h, n, beta, lam):
    # ensure float
    h = np.asarray(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # stable exp
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, h, mode='full')
    term1 = conv1[2*n-2 : 2*n-2 + n]
    conv2 = np.convolve(w, h, mode='full')
    term2 = conv2[2*n-2 : 2*n-2 + n]
    # gradient of smooth part (before penalty)
    grad_smooth = (2.0 / n) * (1.0 - term1 - term2)
    # penalty gradient
    sum_h = np.sum(h)
    diff = sum_h - n/2.0
    grad_pen = 2.0 * lam * diff
    grad = grad_smooth + grad_pen
    # objective
    f_smooth = (2.0 / n) * (max_corr + np.log(sum_exp) / beta)
    f_pen = lam * diff * diff
    f = f_smooth + f_pen
    return f, grad

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic feasibility projection (water-filling)
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# Exact C5 evaluation (full correlation)
# ----------------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Upscaling (linear interpolation)
# ----------------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Random balanced binary vector
# ----------------------------------------------------------------------
def _random_balanced_binary(n: int) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


# ----------------------------------------------------------------------
# Thue‑Morse sequence (balanced for powers of two)
# ----------------------------------------------------------------------
def _thue_morse(n: int) -> np.ndarray:
    seq = np.zeros(1, dtype=int)
    while seq.size < n:
        seq = np.concatenate([seq, 1 - seq])
    return seq[:n].astype(float)


# ----------------------------------------------------------------------
# Smooth max‑approximation objective + gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(h: np.ndarray, n: int, beta: float, lam: float):
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    # stable softmax weights
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    w_rev = w[::-1]

    # convolutions (full)
    conv1 = np.convolve(w_rev, h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    offset = 2 * n - 2
    term1 = conv1[offset: offset + n]
    term2 = conv2[offset: offset + n]

    grad_smooth = (2.0 / n) * (1.0 - term1 - term2)
    diff = h.sum() - n / 2.0
    grad = grad_smooth + 2.0 * lam * diff

    f_smooth = (2.0 / n) * (max_corr + np.log(sum_exp) / beta)
    f = f_smooth + lam * diff * diff
    return f, grad


def _smooth_obj_wrapper(x, n, beta, lam):
    f, g = _smooth_obj_and_grad(x, n, beta, lam)
    return f, g


# ----------------------------------------------------------------------
# Continuous refinement using a sequence of beta values
# ----------------------------------------------------------------------
def _refine_h_continuous(h0: np.ndarray, n: int, betas, lam: float = 500.0, maxiter: int = 120):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        res = opt.minimize(
            _smooth_obj_wrapper,
            h,
            args=(n, beta, lam),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
        )
        h = np.clip(res.x, 0.0, 1.0)
    return h


# ----------------------------------------------------------------------
# SLSQP polish (exact max‑over‑shifts formulation)
# ----------------------------------------------------------------------
def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    res = opt.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

    n = h.size
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[: n // 2]] = 1.0
    return out

def _hill_climb_binary(h_bin: np.ndarray, steps: int = 1500):
    n = h_bin.size
    best = h_bin.copy()
    best_val = _c5_from_h(best)
    ones = np.flatnonzero(best == 1)
    zeros = np.flatnonzero(best == 0)
    for _ in range(steps):
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = _c5_from_h(cand)
        if val < best_val:
            best = cand
            best_val = val
            # update index lists
            ones = np.flatnonzero(best == 1)
            zeros = np.flatnonzero(best == 0)
    return best, best_val

        have_init = False

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # sizes to explore (even)
    sizes = [256, 512, 1024]

    for n in sizes:
        # quick time check
        if time.time() - start_time > budget_s - 5:
            break

        candidates = []

        # upscaled parent if available
        if have_init:
            if init_h.size != n:
                candidates.append(_upscale_h(init_h, n))
            else:
                candidates.append(init_h.copy())

        # Thue‑Morse for powers of two
        if (n & (n - 1)) == 0:
            candidates.append(_thue_morse(n))

        # a few random balanced binaries
        for _ in range(3):
            candidates.append(_random_balanced_binary(n))

        for cand in candidates:
            if time.time() - start_time > budget_s - 5:
                break

            cand = _project_to_feasible(cand, n)

            # continuous refinement with smooth max approximation
            refined = _refine_h_continuous(
                cand,
                n,
                betas=[1.0, 5.0, 10.0, 20.0],
                lam=500.0,
                maxiter=120,
            )
            refined = _project_to_feasible(refined, n)
            c5_val = _c5_from_h(refined)

            if c5_val < best_c5:
                best_h, best_c5, best_n = refined.copy(), c5_val, n

            # binary rounding & hill‑climb
            h_bin = _binary_round(refined)
            h_bin, c5_bin = _hill_climb_binary(h_bin, steps=1200)
            if c5_bin < best_c5:
                best_h, best_c5, best_n = h_bin.copy(), c5_bin, n

            if best_c5 < 0.3808:
                break

        if best_c5 < 0.3808:
            break

    # final polish with SLSQP if time permits
    if best_h is not None and (time.time() - start_time) < budget_s - 4:
        h_opt, c5_opt = _solve_h(best_h, max_iter=800)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, h_opt.size

    # fallback (should never happen)
    if best_h is None:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    # final safety projection and accurate C5
    best_h = _project_to_feasible(best_h, best_h.size)
    best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_h.size)
