import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x

def hill_climb_generic(h_start, max_iters=2000, step_start=0.6):
    rng = np.random.default_rng()
    n = h_start.size
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_start
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng):
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()

    v = np.zeros_like(h)

    for beta in beta_sched:

            corr = np.correlate(h, 1 - h, mode="full")
            C_raw = corr

    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


# ---- SLSQP (soft‑max) utilities ------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, n / 2.0)


# ---- simple hill‑climb ---------------------------------------------------------
def hill_climb_generic(h_start, max_iters=2000, step_start=0.6):
    rng = np.random.default_rng()
    n = h_start.size
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_start
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# ---- projected gradient descent on smooth max ---------------------------------
def grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng):
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros_like(h)
    momentum = 0.9
    for beta in beta_sched:
        step = 0.6
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C_raw = corr
            bc = beta * C_raw
            m = bc.max()
            w = np.exp(bc - m)
            w /= w.sum()

            # gradient of (2/n)*C_raw
            g = np.ones(n)
            # non‑negative shifts
            for k in range(n):
                wk = w[n - 1 + k]
                if wk == 0.0:
                    continue
                g[: n - k] -= wk * h[k:]
                g[k:] -= wk * h[: n - k]
            # negative shifts
            for kk in range(1, n):
                wk = w[n - 1 - kk]
                if wk == 0.0:
                    continue
                g[kk:] -= wk * h[: n - kk]
                g[: n - kk] -= wk * h[kk:]

            grad = (2.0 / n) * g
            v = momentum * v - step * grad
            h_new = h + v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, target)

            # enforce symmetry (helps reduce dimensionality)
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = project_onto_box_sum(h_new, target)

            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                if rng.random() < 0.01:
                    h = h_new
            step *= 0.998
    return best_h, best_c


# ---- main entry point ---------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # problem size – a moderate even number to keep evaluation cheap
    n = 240
    target = n / 2.0

    # ----- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        # linear interpolation onto our grid
        x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_target, x_parent, parent, period=2.0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        h0 = (h0 + h0[::-1]) * 0.5
        h0 = project_onto_box_sum(h0, target)
    else:
        half = n // 2
        x = rng.random(half)
        x = project_onto_box_sum(x, target / 2.0)
        h0 = np.concatenate([x, x[::-1]])
        h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # quick hill‑climb on the parent solution
    if time.time() - start < budget_s * 0.15:
        h_tmp, c_tmp = hill_climb_generic(best_h, max_iters=3000, step_start=0.8)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    beta_sched = [2.0, 10.0, 40.0, 150.0, 500.0, 1200.0]
    steps_per_beta = 120

    # ----- main stochastic search --------------------------------------------
    while time.time() - start < budget_s * 0.85:
        # random symmetric feasible start
        half = n // 2
        x = rng.random(half)
        x = project_onto_box_sum(x, target / 2.0)
        h_start = np.concatenate([x, x[::-1]])

        # occasionally blend with the current best
        if rng.random() < 0.2:
            h_start = (h_start + best_h) * 0.5
            h_start = project_onto_box_sum(h_start, target)
            h_start = (h_start + h_start[::-1]) * 0.5
            h_start = project_onto_box_sum(h_start, target)

        # smooth‑max gradient descent
        h_opt, c_opt = grad_descent_optimize(h_start, beta_sched, steps_per_beta, rng)

        # short SLSQP refinement (good for polishing)
        try:
            for beta in (150.0, 500.0, 1200.0):
                h_opt = slsqp_optimize(h_opt, beta=beta, max_iter=250)
        except Exception:
            pass

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            # a brief local hill‑climb after a genuine improvement
            if time.time() - start < budget_s * 0.7:
                h2, c2 = hill_climb_generic(best_h, max_iters=2000, step_start=0.6)
                if c2 < best_c:
                    best_h, best_c = h2, c2

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
