def _gradient_wrt_shift(h, ell):
    n = h.shape[0]
    g = np.zeros_like(h)
    # term1 for j such that j+ell in range [0, n-1]
    if ell >= 0:
        # j from 0 to n-ell-1
        j_range = slice(0, n-ell)
        g[j_range] += 1 - h[ell:]
    else:
        d = -ell
        # j from d to n-1, because j - d >= 0? Wait term1: j+ell = j-d >=0 => j >= d
        j_range = slice(d, n)
        g[j_range] += 1 - h[:n-d]

    if ell >= 0:
        # j-ell >=0 => j >= ell

        g[j_range] -= h[:n-ell]

        d = -ell
        # j - (-d) = j + d < n => j < n-d

def _smooth_grad_step(h, beta, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * (2.0 / n)
    # softmax weights
    nb = 2*n - 1
    max_b = np.max(beta * corr_scaled)
    exps = np.exp(beta * corr_scaled - max_b)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for l in range(nb):
        w_l = w[l]
        if w_l == 0:
            continue
        ell = l - (n - 1)
        if ell >= 0:
            # (1 - h[ell:]) added to grad[:n-ell]
            grad[:n-ell] += w_l * (1 - h[ell:])
            # - h[:n-ell] sub from grad[ell:]
            grad[ell:] -= w_l * h[:n-ell]
        else:
            d = -ell
            # + (1 - h[:n-d]) for indices d:
            grad[d:] += w_l * (1 - h[:n-d])
            # - h[d:] for indices :n-d
            grad[:n-d] -= w_l * h[d:]
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_to_feasible(h_new, n)
    return h_new

def _targeted_hill_climb(h, max_steps, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for step in range(max_steps):
        if time.time() > deadline:
            break
        # compute worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        ell = max_idx - (n - 1)
        # compute gradient
        g = _gradient_wrt_shift(cur_h, ell)
        # find i with positive gradient (decrease reduces C5)
        pos = np.where(g > 0)[0]
        neg = np.where(g < 0)[0]
        if pos.size == 0 or neg.size == 0:
            # fallback to random swap
            i = np.random.randint(n)
            j = np.random.randint(n)
        else:
            i = np.random.choice(pos)
            j = np.random.choice(neg)
        # decide eps
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        # choose epsilon proportionally to gradient magnitude
        eps = min(max_eps, 0.5 * max_eps) * np.random.rand()
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = _project_to_feasible(cand, n)  # ensure feasibility
        cand_c5 = _c5_from_h(cand)
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
        else:
            # with small probability accept
            if np.random.rand() < 0.001:
                cur_h, cur_c5 = cand, cand_c5
    return best_h, best_c5

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 300) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _gradient_wrt_shift(h: np.ndarray, ell: int) -> np.ndarray:
    """Gradient of C5 for a fixed shift ell (lag). Returns unscaled gradient."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if ell >= 0:
        # (1 - h[j+ell]) part
        g[: n - ell] += 1 - h[ell:]
        # -h[j-ell] part
        g[ell:] -= h[: n - ell]
    else:
        d = -ell
        # (1 - h[j- d]) where j-d >=0  -> j >= d
        g[d:] += 1 - h[: n - d]
        # -h[j + d] where j+d < n -> j < n-d
        g[: n - d] -= h[d:]
    return g


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (global exploration)
# ----------------------------------------------------------------------
def _gd_optimize(start_h: np.ndarray, max_iters: int, deadline: float):
    """Projected gradient descent on a smooth approximation of max C5."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_c5 = best_c5

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # schedule: beta grows, step shrinks
        beta = 5.0 + 45.0 * (it / max_iters)      # up to ~50
        lr = 0.2 * (1.0 - it / max_iters) + 0.01  # 0.21 → 0.01

        corr = np.correlate(h, 1.0 - h, mode="full")
        corr_s = corr * dx
        # stable soft‑max
        max_bc = np.max(beta * corr_s)
        exp_vals = np.exp(beta * corr_s - max_bc)
        w = exp_vals / np.sum(exp_vals)          # weights over all lags

        grad = np.zeros_like(h)
        # accumulate weighted gradient
        nb = 2 * n - 1
        for l in range(nb):
            w_l = w[l]
            if w_l == 0.0:
                continue
            ell = l - (n - 1)
            if ell >= 0:
                grad[: n - ell] += w_l * dx * (1 - h[ell:])
                grad[ell:] -= w_l * dx * h[: n - ell]
            else:
                d = -ell
                grad[d:] += w_l * dx * (1 - h[: n - d])
                grad[: n - d] -= w_l * dx * h[d:]

        # gradient step + projection
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_feasible(h_new, n)

        new_c5 = _c5_from_h(h_new)
        if new_c5 < best_c5:
            best_h, best_c5 = h_new.copy(), new_c5
        # accept if improvement, else allow occasional uphill move
        if new_c5 < cur_c5:
            h, cur_c5 = h_new, new_c5
        elif np.random.rand() < 0.001:
            h, cur_c5 = h_new, new_c5

    return best_h, best_c5


# ----------------------------------------------------------------------
# Targeted hill climbing (local refinement)
# ----------------------------------------------------------------------
def _targeted_hill_climb(start_h: np.ndarray, max_steps: int, deadline: float):
    """Move mass from positions that increase the worst shift to those that decrease it."""
    n = start_h.size
    h = start_h.copy()
    cur_c5 = _c5_from_h(h)
    best_h, best_c5 = h.copy(), cur_c5

    for step in range(max_steps):
        if time.time() > deadline:
            break

        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        # occasionally pick a random shift to avoid over‑focusing
        if np.random.rand() < 0.2:
            max_idx = np.random.randint(2 * n - 1)
        ell = max_idx - (n - 1)

        g = _gradient_wrt_shift(h, ell)
        pos = np.where(g > 0)[0]
        neg = np.where(g < 0)[0]

        if pos.size == 0 or neg.size == 0:
            # fallback: random swap of two coordinates
            i, j = np.random.randint(n, size=2)
        else:
            i = np.random.choice(pos)
            j = np.random.choice(neg)

        # maximal amount we can transfer while staying feasible
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # step size shrinks with progress
        eps = max_eps * np.random.rand() * (0.5 * (1.0 - step / max_steps) + 0.01)

        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        # sum unchanged, bounds respected → no projection needed

        cand_c5 = _c5_from_h(cand)
        if cand_c5 < cur_c5:
            h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
        elif np.random.rand() < 0.001:
            h, cur_c5 = cand, cand_c5

    return best_h, best_c5


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function on [0,2] that yields a low C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load any provided initial construction
    # ------------------------------------------------------------------
    try:
        GLOBAL_INITIAL_H = np.array(initial_h_values, dtype=float)
    except NameError:
        GLOBAL_INITIAL_H = None

    # ------------------------------------------------------------------
    # Candidate grid sizes
    # ------------------------------------------------------------------
    n_candidates = []
    if GLOBAL_INITIAL_H is not None:
        n_candidates.append(GLOBAL_INITIAL_H.size)
    # hand‑picked even sizes (kept modest for speed)
    n_candidates.extend([80, 96, 112, 128, 160, 192, 224, 256])
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Search loop over grid sizes
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Prepare a collection of starting vectors for this size
        start_vectors = []

        # Upscaled initial construction (if any)
        if GLOBAL_INITIAL_H is not None:
            if GLOBAL_INITIAL_H.size != n:
                start_vectors.append(_upscale_h(GLOBAL_INITIAL_H, n))
            else:
                start_vectors.append(GLOBAL_INITIAL_H.copy())

        # Random starts – ensure feasibility via projection
        needed = max(10 - len(start_vectors), 0)
        for _ in range(needed):
            rand = np.random.rand(n)
            start_vectors.append(_project_to_feasible(rand, n))

        # ------------------------------------------------------------------
        # Optimise each start vector
        # ------------------------------------------------------------------
        for init_h in start_vectors:
            if time_left() < 2.0:
                break

            dead = start_time + budget_s - 0.5  # cushion for final returns

            # Global‑search phase
            h_gd, c5_gd = _gd_optimize(init_h, max_iters=200, deadline=dead)

            if time.time() > dead:
                break

            # Local refinement phase
            h_ref, c5_ref = _targeted_hill_climb(
                h_gd, max_steps=2500, deadline=dead
            )

            if c5_ref < best_c5:
                best_h, best_c5, best_n = h_ref.copy(), c5_ref, n
                # early exit if we beat the target bound
                if best_c5 < 0.3808:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
