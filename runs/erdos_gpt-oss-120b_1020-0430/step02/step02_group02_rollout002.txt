def _solve_h_slsqp(h0, max_iter=200):
    n = h0.size
    h0 = _project_to_feasible(h0, n)
    corr = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr) * (2.0/n) # Actually in same units? Wait t variable is correlation value * (2/n). Use raw correlation to maintain scaling consistency. In previous code they used raw correlation to compare. We'll keep with raw correlation.
    t0_raw = np.max(corr)
    x0 = np.concatenate([h0, [t0_raw]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        # return t - corr_i
        corr = np.correlate(h, 1 - h, mode='full')
        return t - corr

    def eq_fun(x):
        h = x[:n]
        return np.sum(h) - n/2.0

    constraints = [{'type': 'ineq', 'fun': ineq_fun},
                   {'type': 'eq', 'fun': eq_fun}]
    options = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                            constraints=constraints, options=options)
    h_opt = res.x[:n]
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt

def _hillclimb(h, steps, time_limit):
    n = h.size
    best = h.copy()
    bestc = _c5_from_h(best)
    start = time.time()
    for _ in range(steps):
        if time.time() > time_limit:
            break
        # decide move type
        if np.random.rand() < 0.2:
            # targeted move reducing max correlation
            corr = np.correlate(best, 1 - best, mode='full')
            k = np.argmax(corr) - (n-1) # shift index to lag
            # compute indices for positive lag
            if k >= 0:
                # i in [0, n-1-k]
                candidates_i = np.where(best[:n - k] > 0)[0]
                if candidates_i.size == 0:
                    continue
                i = np.random.choice(candidates_i)
                j = i + k
            else:
                # k negative, i+k => i-k
                kk = -k
                candidates_i = np.where(best[kk:] > 0)[0]
                if candidates_i.size == 0:
                    continue
                i = candidates_i[0] + kk
                j = i + k  # = i - kk
            max_eps = min(best[i], 1 - best[j])
            if max_eps <= 0:
                continue
            eps = np.random.rand() * max_eps
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
        else:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_eps = min(best[i], 1 - best[j])
            if max_eps <= 0:
                continue
            eps = np.random.rand() * max_eps * 0.5
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
        c = _c5_from_h(cand)
        if c < bestc:
            best, bestc = cand, c
    return best, bestc

    best_h = None

    best_n = None

    try:
        init_h = np.array(initial_h_values, dtype=float)

        init_h = None

def _project_to_feasible(h, n, tol=1e-12, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust first feasible entries
        if residual > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        for i in idx:
            if residual > 0:
                add = min(1.0 - h[i], residual)
                h[i] += add
                residual -= add
            else:
                sub = min(h[i], -residual)
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy import optimize

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        for i in idx:
            if residual > 0:
                add = min(1.0 - h[i], residual)
                h[i] += add
                residual -= add
            else:
                sub = min(h[i], -residual)
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _solve_h_slsqp(h0: np.ndarray, max_iter: int = 300) -> np.ndarray:
    n = h0.size
    h0 = _project_to_feasible(h0, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt


def _hillclimb(h: np.ndarray, steps: int, time_limit: float, rng: np.random.Generator):
    n = h.size
    best = h.copy()
    bestc = _c5_from_h(best)
    for _ in range(steps):
        if time.time() > time_limit:
            break
        if rng.random() < 0.2:
            corr = np.correlate(best, 1 - best, mode="full")
            k = int(np.argmax(corr) - (n - 1))
            if k >= 0:
                i_cand = np.where(best[: n - k] > 0)[0]
                if i_cand.size == 0:
                    continue
                i = rng.choice(i_cand)
                j = i + k
            else:
                kk = -k
                i_cand = np.where(best[kk:] > 0)[0]
                if i_cand.size == 0:
                    continue
                i = i_cand[0] + kk
                j = i + k
            max_eps = min(best[i], 1.0 - best[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_eps = min(best[i], 1.0 - best[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * 0.5
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
        c = _c5_from_h(cand)
        if c < bestc:
            best, bestc = cand, c
