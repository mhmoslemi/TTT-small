import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _binary_random(n, target, rng):
    """Balanced binary vector of length n with exactly target ones."""
    idx = rng.choice(n, size=target, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def _upscale_initial(init_vec, n, target):
    """Upscale a given step vector (length m) to length n and keep the largest target entries."""
    init = np.asarray(init_vec, dtype=np.float64)
    m = init.size
    if m == n:
        # just take the biggest target entries
        idx = np.argsort(-init)[:target]
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
        return h
    # linear interpolation to new grid
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    interp = np.interp(x_new, x_old, init)
    idx = np.argsort(-interp)[:target]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with small C5."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- problem size -------------------------------------------------
    n = 200                         # number of samples (kept modest)
    target = n // 2                 # required sum of h (exactly half)

    # ---- initial candidate --------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h_best = _upscale_initial(init_vec, n, target)
    else:
        h_best = _binary_random(n, target, rng)

    best_c5 = _evaluate_c5(h_best)

    # ---- helper for a cheap random balanced swap ----------------------
    def random_swap(h):
        """Return a new vector after swapping a random 1 with a random 0."""
        i1 = rng.integers(n)
        while h[i1] == 0:
            i1 = rng.integers(n)
        i2 = rng.integers(n)
        while h[i2] == 1:
            i2 = rng.integers(n)
        h2 = h.copy()
        h2[i1] = 0.0
        h2[i2] = 1.0
        return h2, i1, i2

    # ---- main search (simulated annealing + restarts) -----------------
    margin = 5.0                    # reserve time for final polishing
    while time.time() < start + budget_s - margin:
        # start from best with a few random jitter
        if rng.random() < 0.25:
            # occasionally start from a fresh random solution
            cur = _binary_random(n, target, rng)
        else:
            cur = h_best.copy()
            # apply a couple of random swaps to diversify
            for _ in range(3):
                cur, _, _ = random_swap(cur)

        cur_c5 = _evaluate_c5(cur)

        # inner annealing loop
        while time.time() < start + budget_s - margin:
            # temperature decreases linearly with elapsed fraction
            elapsed = time.time() - start
            frac = elapsed / (budget_s - margin)
            T = 0.02 * (1 - frac) + 0.0005 * frac

            cand, _, _ = random_swap(cur)
            cand_c5 = _evaluate_c5(cand)
            delta = cand_c5 - cur_c5
            if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
                cur, cur_c5 = cand, cand_c5
                if cand_c5 < best_c5:
                    h_best, best_c5 = cand.copy(), cand_c5
            # stop inner loop early if we have found an excellent bound
            if best_c5 < 0.3810:
                break

        # early exit if global budget nearly exhausted
        if time.time() >= start + budget_s - margin:
            break

    # ---- final deterministic hill climbing -----------------------------
    deadline = start + budget_s - 0.5
    attempts_without_improve = 0
    while time.time() < deadline and attempts_without_improve < 50000:
        cand, _, _ = random_swap(h_best)
        cand_c5 = _evaluate_c5(cand)
        if cand_c5 < best_c5 - 1e-12:
            h_best, best_c5 = cand, cand_c5
            attempts_without_improve = 0
        else:
            attempts_without_improve += 1

    # ---- final sanity checks -------------------------------------------
    # ensure feasibility
    h_best = np.clip(h_best, 0.0, 1.0)
    # adjust any tiny numerical drift in the sum
    cur_sum = h_best.sum()
    if abs(cur_sum - target) > 1e-9:
        # fix by toggling a few entries (very unlikely)
        diff = int(round(target - cur_sum))
        if diff > 0:
            zero_idx = np.flatnonzero(h_best == 0)
            h_best[rng.choice(zero_idx, size=diff, replace=False)] = 1.0
        elif diff < 0:
            one_idx = np.flatnonzero(h_best == 1)
            h_best[rng.choice(one_idx, size=-diff, replace=False)] = 0.0

    best_c5 = float(_evaluate_c5(h_best))
    return h_best, best_c5, n
