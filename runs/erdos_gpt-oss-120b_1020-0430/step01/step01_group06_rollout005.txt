def project_to_sum_and_bounds(v, target):
    lo = -1.0  # maybe lower bound for lambda (ensuring all v-lambda >= 0)
    hi = 1.0   # ensure v-lambda <= 1

    for _ in range(30):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return w
```

import numpy as np

def _c5_of(v):
    """Maximum overlap integral for a step vector v."""
    n = v.shape[0]
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(corr.max() * (2.0 / n))

def _project_sum_bounds(v, target):
    """Euclidean projection of v onto 0≤v≤1 and sum(v)=target."""
    lo, hi = -1.0, 2.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _gradient(v, shift):
    """Gradient of the overlap for a given shift (derivative of I_shift)."""
    n = v.shape[0]
    g = np.zeros(n)
    if shift >= 0:
        i = np.arange(n - shift)
        j = i + shift
        g[i] += (1.0 - v[j])
        g[j] -= v[i]
    else:
        d = -shift
        i = np.arange(d, n)
        j = i - d
        g[i] += (1.0 - v[j])
        g[j] -= v[i]
    return g

                    nh[j_best] += eps
                    new_c5 = _c5_of(nh)
                    if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                        h = nh
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                            last_improve = it

        # --------------------------------------------------------------
        # 2. Random two‑coordinate exchange preserving the sum
        # --------------------------------------------------------------
        elif move_type < p_target + p_swap:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            max_up   = min(1.0 - hi, hj)   # increase i, decrease j
            max_down = min(hi, 1.0 - hj)   # decrease i, increase j
            if max_up + max_down > 0.0:
                delta = (random.random() * (max_up + max_down)) - max_down
                if abs(delta) > 1e-14:
                    nh = h.copy()
                    nh[i] = hi + delta
                    nh[j] = hj - delta
                    new_c5 = _c5_of(nh)
                    if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                        h = nh
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                            last_improve = it

        # --------------------------------------------------------------
        # 3. Gradient descent on the worst shift (with projection)
        # --------------------------------------------------------------
        else:
            g = _gradient(h, s_max)
            if np.any(np.abs(g) > 0):
                step = T * 0.5
                step = min(step, 0.12)            # safety cap
                nh = h - step * g
                nh = _project_sum_bounds(nh, target_sum)
                new_c5 = _c5_of(nh)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        last_improve = it

        # --------------------------------------------------------------
        # temperature schedule & occasional restart
        # --------------------------------------------------------------
        it += 1
        if it % 1000 == 0:
            T *= decay

        # simple restart if stagnation lasts too long
        if it - last_improve > 200_000 and time.time() - start < budget_s * 0.90:
            # re‑initialise near the best solution with a small jitter
            h = best_h.copy()
            jitter = np.random.uniform(-0.03, 0.03, size=n)
            h = np.clip(h + jitter, 0.0, 1.0)
            h = _project_sum_bounds(h, target_sum)
            cur_c5 = _c5_of(h)
            last_improve = it

        # early exit if we already beat the target
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Final safety correction of the best vector (tiny drift)
    # ------------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        # distribute the error uniformly across non‑saturated entries
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size > 0:
            correction = err / free.size
            best_h[free] -= correction
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
