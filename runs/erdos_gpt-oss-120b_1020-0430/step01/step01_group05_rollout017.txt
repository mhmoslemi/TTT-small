import time, random, math
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while forcing sum(x)=target."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth objective used by SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def make_triangle(n: int) -> np.ndarray:
    """Symmetric triangular profile, height 1 at centre."""
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(xs - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2), already integrates to 1."""
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * xs / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin_pow(n: int, p: float = 4.0) -> np.ndarray:
    """h(x)=sin^p(πx/2), gives a peaked shape."""
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * xs / 2.0) ** p
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Derivative of C(k) w.r.t. h for the given integer shift."""
    n = h.shape[0]
    term1 = np.ones(n)
    term2 = np.zeros(n)
    if shift >= 0:
        term1[: n - shift] = 1.0 - h[shift:]
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = 1.0 - h[: n - s]
        term2[: n - s] = h[s:]
    return term1 - term2


# ---------- deterministic greedy improvement ----------
def greedy_optimize(h: np.ndarray, time_limit: float) -> tuple:
    """Focused descent that always tries to reduce the current worst lag."""
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    start = time.time()

    while time.time() - start < time_limit:
        # current worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        cur_val = corr[idx_max] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        if best_val <= 0.3808:
            break

        shift = idx_max - (n - 1)          # integer lag corresponding to idx_max
        grad = gradient_for_shift(h, shift)

        # indices offering maximal reduction when moving mass from j to i
        i = int(np.argmin(grad))          # most negative gradient -> increase
        j = int(np.argmax(grad))          # most positive gradient -> decrease

        if grad[i] >= -1e-12 or grad[j] <= 1e-12:
            # no clear direction, fall back to a small random move
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta_max = min(1.0 - h[i], h[j])
            if delta_max <= 0:
                continue
            delta = np.random.uniform(0.0, delta_max)
            cand = h.copy()
            cand[i] += delta
            cand[j] -= delta
        else:
            delta_max = min(1.0 - h[i], h[j])
            if delta_max <= 0:
                continue
            # a moderate step (half of the feasible amount)
            delta = 0.5 * delta_max
            cand = h.copy()
            cand[i] += delta
            cand[j] -= delta

        # evaluate candidate
        new_val = compute_c5(cand)
        if new_val < cur_val - 1e-10 or new_val < best_val - 1e-10:
            h = cand
        # else reject – stay with current h

    return best_h, best_val


# ---------- main routine ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] that yields a low C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)

    start_total = time.time()
    # problem size (trade‑off between flexibility and evaluation cost)
    n = 300                     # even number ensures exact half‑mass constraint
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum

    bounds = [(0.0, 1.0)] * n

    # ---- collect initial seeds ----
    seed_vectors = []
    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))
    seed_vectors.append(make_sin_pow(n, p=4.0))
    seed_vectors.append(make_sin_pow(n, p=6.0))

    best_h = None
    best_val = np.inf
    restart = 0

    # ---- SLSQP phase (global search) ----
    while time.time() - start_total < 0.55 * budget_s:
        if restart < len(seed_vectors):
            h0 = seed_vectors[restart]
        else:
            rnd = np.random.rand(n)
            h0 = project_onto_bounds_and_sum(rnd, target_sum)
        restart += 1

        # smooth‑max parameter; we slowly increase it with each restart
        alpha = 20.0 + 0.5 * restart

        try:
            res = minimize(
                objective_smooth,
                h0,
                args=(dx, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": lambda x: np.sum(x) - target_sum},
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue

        h_opt = project_onto_bounds_and_sum(res.x, target_sum)
        val = compute_c5(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        if best_val <= 0.3808:
            break

    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    # ---- deterministic greedy refinement ----
    remaining = budget_s - (time.time() - start_total)
    if remaining > 2.0:
        best_h, best_val = greedy_optimize(best_h, remaining - 1.0)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
