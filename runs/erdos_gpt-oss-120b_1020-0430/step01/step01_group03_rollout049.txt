    # solve p = clip(v - lambda, 0,1) so that sum(p)=target
    # bisection on lambda
    lo = -1e6; hi = 1e6

        s = p.sum()
        if s > target:

            hi = lam

def gradient_descent_proj(h0, rng, max_iters=2000, step0=0.2):
    n = h0.size
    target = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    eta = step0
    for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        k_idx = np.argmax(corr)
        # compute shift k
        k = k_idx - (n - 1)
        # compute gradient
        g = np.zeros_like(h)
        if k >= 0:
            # i:0..n-1-k -> contributor
            g[:n-k] += (1 - h[k:])
            # -h[i-k] for i>=k
            g[k:] -= h[:n-k]
        else:
            kk = -k
            g[kk:] += (1 - h[:n-kk])
            g[:n-kk] -= h[kk:]
        # update
        v = h - eta * g
        h_new = project_onto_box_and_sum(v, target)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()

        # shrink step

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for SLSQP (soft‑max surrogate)
CURRENT_TARGET = None
CURRENT_BETA = None
GLOBAL_RNG = None  # will be set in run()


def sum_eq_constraint(x):
    """Equality Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max_{k} Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
# Simple feasibility utilities
def enforce_feasibility(h, rng):
    """Clip to [0,1] and adjust sum to n/2 exactly."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                add = max(diff, -h[i])
                h[i] += add
                diff -= add
    return h


def random_feasible_vector(n, rng):
    """Generate a random point in the polytope (0≤h≤1, Σh=n/2)."""
    target = n / 2.0
    v = rng.random(n)
    v = v / v.sum() * target
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        idx = (v > 0) & (v < 1)
        if not np.any(idx):
            break
        i = rng.choice(np.where(idx)[0])
        if diff > 0:
            add = min(diff, 1.0 - v[i])
            v[i] += add
            diff -= add
        else:
            add = max(diff, -v[i])
            v[i] += add
            diff -= add
    return v


def random_binary_vector(n, rng):
    """Binary feasible start (exactly n/2 ones)."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

# Local search utilities
def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # shrink step periodically
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def project_onto_box_and_sum(v, target):
    """Projection of v onto {0≤x≤1, Σx=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        p = np.clip(v - lam, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        idx = (p > 0) & (p < 1)
        if np.any(idx):
            p[idx] += diff / np.count_nonzero(idx)
            p = np.clip(p, 0.0, 1.0)
    return p


def gradient_descent_proj(h0, rng, max_iters=1500, step0=0.3):
    """Projected gradient descent on the max‑correlation (sub‑gradient)."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    eta = step0
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)          # shift amount (negative allowed)
        g = np.ones_like(h)          # derivative of Σ h_i part
        if k >= 0:
            # -h_{i+k}
            g[:n - k] -= h[k:]
            # -h_{i-k}
            g[k:]   -= h[:n - k]
        else:
            kk = -k
            # -h_{i+k} = -h_{i-kk}
            g[kk:]  -= h[:n - kk]
            # -h_{i-k} = -h_{i+kk}
            g[:n - kk] -= h[kk:]
        # descent step and projection
        v = h - eta * g
        h_new = project_onto_box_and_sum(v, target)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            eta = min(eta * 1.1, 1.0)   # gently increase step after success
        else:
            eta *= 0.7
            if eta < 1e-5:
                break
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=250):
    """SLSQP refinement of h using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA, GLOBAL_RNG
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum correction (rare)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = GLOBAL_RNG.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                add = max(diff, -x[i])
                x[i] += add
                diff -= add
    return x

    start_time = time.time()
