def binary_hill_climb(start_h, rng, max_iters, best_c_initial):
    cur_h = start_h.copy()
    cur_c = best_c_initial
    best_h = cur_h.copy()
    best_c = cur_c
    # keep list of indices for ones and zeros for quick random selection
    for it in range(max_iters):
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand_h = cur_h.copy()
        cand_h[i] = 0
        cand_h[j] = 1
        c = compute_c5(cand_h)
        if c < cur_c:
            cur_h = cand_h
            cur_c = c
        if c < best_c:
            best_h = cand_h
            best_c = c
    return best_h, best_c
```

def mismatched_indices(h, L):
    n = h.size
    if L >= 0:
        i_start = 0
        i_end = n - 1 - L
        i_range = np.arange(i_start, i_end+1)
        cond = (h[i_range] == 1) & (h[i_range + L] == 0)
    else:
        L = -L
        i_start = L
        i_end = n - 1
        i_range = np.arange(i_start, i_end+1)
        # condition for original shift L negative: i + (-L) = i - L
        cond = (h[i_range] == 1) & (h[i_range - L] == 0)
    return i_range[cond]

import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_{k} Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=200):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size

    n = h.size

        c_val = (2.0 / n) * max_corr
        if c_val < best_c:
            best_h, best_c = h_opt.copy(), c_val
        if best_c <= TARGET_C:
            break

    # ------------------------------------------------------------------
    # Continuous refinement with soft‑max SLSQP
    # ------------------------------------------------------------------
    if time.time() - start_time < MAX_CONTINUOUS_TIME:
        # guarantee feasibility before feeding to optimizer
        best_h = make_feasible(best_h.astype(float), target_sum)
        betas = [5.0, 20.0, 80.0, 200.0]
        for beta in betas:
            if time.time() - start_time > budget_s * 0.99:
                break
            try:
                trial = slsqp_optimize(best_h, beta=beta, max_iter=400)
                c_trial = compute_c5(trial)
                if c_trial < best_c:
                    best_h, best_c = trial, c_trial
            except Exception:
                continue

    # ------------------------------------------------------------------
    # Final feasibility clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = make_feasible(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
