def compute_gradient(h, w):
    n = len(h)
    g = np.zeros(n)
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            # first term
            i_end = n - s
            if i_end > 0:
                g[:i_end] += weight * (1 - h[s:s + i_end])
            # second term
            if s < n:
                g[s:] -= weight * h[:n - s]
        else:
            s_abs = -s
            # first term
            start = s_abs
            if start < n:

def run(seed=42, budget_s=1100, **kwargs):
    import time, random, math
    import numpy as np

    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def evaluate(h):
        """C5 value for a given vector h (no rounding)."""
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(corr.max() * (2.0 / len(h)))

    def proj_capped_simplex(v, target):
        """
        Euclidean projection of v onto the set:
            0 <= x_i <= 1,   sum_i x_i = target
        Uses a simple binary search for the Lagrange multiplier.
        """
        lo = np.min(v) - 1.0            # ensures x_i = 1 for all i
        hi = np.max(v)                  # ensures x_i = 0 for all i
        for _ in range(60):
            mid = (lo + hi) * 0.5
            x = np.clip(v - mid, 0.0, 1.0)
            s = x.sum()
            if s > target:
                lo = mid
            else:
                hi = mid
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)

        # tiny correction for numerical drift
        err = x.sum() - target
        if abs(err) > 1e-10:
            # adjust deterministically on the first free coordinate(s)
            free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
            if free.size == 0:
                free = np.arange(len(x))
            # distribute error evenly (sign aware)
            delta = err / free.size
            x[free] -= delta
            # clip once more
            x = np.clip(x, 0.0, 1.0)
        return x

    def soft_weights(c, T):
        """Softmax weights for the vector c with temperature T."""
        if T <= 0.0:
            # pure max weight
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            return w
        m = np.max(c)
        exp_vals = np.exp((c - m) / T)
        s = exp_vals.sum()
        if s == 0:
            # fallback uniform
            return np.full_like(c, 1.0 / c.size)
        return exp_vals / s

    def gradient_weighted(h, w):
        """
        Weighted subgradient of  max_{shift} sum_i h_i(1-h_{i+shift})
        using weights w over all shifts.
        """
        n = len(h)
        g = np.zeros(n, dtype=float)
        # w length = 2*n-1, index 0 corresponds to shift = -(n-1)
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            s = idx - (n - 1)          # actual shift
            if s >= 0:
                # first term: (1 - h[i+s]) for i=0..n-s-1
                end = n - s
                if end > 0:
                    g[:end] += weight * (1.0 - h[s:s + end])
                # second term: -h[i] for i=0..n-s-1 added to positions i+s
                if s < n:
                    g[s:] -= weight * h[:end]
            else:
                s_abs = -s
                # first term for indices i >= s_abs
                if s_abs < n:
                    g[s_abs:] += weight * (1.0 - h[:n - s_abs])
                # second term for indices i < n - s_abs
                end = n - s_abs
                if end > 0:
                    g[:end] -= weight * h[s_abs:s_abs + end]
        return g

    def random_feasible(n, target):
        """Generate a random vector in [0,1] with given sum (projected)."""
        v = np.random.rand(n)
        return proj_capped_simplex(v, target)

    # ----------------------------------------------------------------------
    # Problem dimensions
    # ----------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = int(h0.shape[0])
    except NameError:
        n = 70
        h0 = random_feasible(n, n / 2.0)

    target_sum = n / 2.0

    # make sure the starting vector respects constraints
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = proj_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_c5 = evaluate(best_h)

    # ----------------------------------------------------------------------
    # Search parameters
    # ----------------------------------------------------------------------
    max_outer_iters = 30                     # number of restarts
    max_inner_iters = 3000                   # iterations per restart
    T0 = 1.0                                 # initial temperature for softmax
    temp_decay = 0.995                       # geometric decay per inner iteration
    base_alpha = 0.12                        # base step size (will be scaled)

    # candidate alphas tried in a decreasing list whenever a move fails
    alpha_candidates = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01]

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    iteration = 0
    while time.time() - start_time < budget_s * 0.93:
        # initialise a fresh start (skip if we already used the given one)
        if iteration == 0:
            h = h0.copy()
        else:
            h = random_feasible(n, target_sum)

        cur_c5 = evaluate(h)

        T = T0
        for it in range(max_inner_iters):
            if time.time() - start_time > budget_s * 0.93:
                break

            # compute weighted gradient
            corr = np.correlate(h, 1.0 - h, mode="full")
            w = soft_weights(corr, T)
            g = gradient_weighted(h, w)

            improved = False
            for alpha in alpha_candidates:
                step = base_alpha * alpha
                h_new = proj_capped_simplex(h - step * g, target_sum)
                new_c5 = evaluate(h_new)
                if new_c5 < cur_c5 - 1e-9:
                    h, cur_c5 = h_new, new_c5
                    improved = True
                    break
                # occasional acceptance of worse moves (simulated annealing)
                if new_c5 > cur_c5:
                    prob = math.exp(-(new_c5 - cur_c5) / max(T, 1e-8))
                    if random.random() < prob:
                        h, cur_c5 = h_new, new_c5
                        improved = True
                        break

            # if none of the deterministic alphas succeeded, try a tiny random perturbation
            if not improved:
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                # move a tiny epsilon from i to j keeping bounds
                eps = min(hi, 1.0 - hj) * 0.02
                if eps > 1e-12:
                    h[i] = hi - eps
                    h[j] = hj + eps
                    cur_c5 = evaluate(h)

            # update global best
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()

            # temperature schedule (for softmax and SA acceptance)
            T = max(0.001, T * temp_decay)

            iteration += 1

        # after one restart, check budget
        if time.time() - start_time > budget_s * 0.93:
            break

    # ----------------------------------------------------------------------
    # Final safety correction (ensure exact sum)
    # ----------------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        # deterministic correction on the first free coordinate
        for idx in range(n):
            if err > 0 and best_h[idx] > err:
                best_h[idx] -= err
                err = 0.0
                break
            if err < 0 and best_h[idx] < 1.0 + err:
                best_h[idx] -= err   # err negative -> addition
                err = 0.0
                break
        # final re‑projection if still off
        if abs(err) > 1e-12:
            best_h = proj_capped_simplex(best_h, target_sum)

    final_c5 = evaluate(best_h)

    return (best_h.tolist(), final_c5, n)
