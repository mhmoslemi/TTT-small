def propose_move(h):
    i, j = random indices distinct.
    hi, hj = h[i], h[j]
    delta_max = min(1 - hi, hj)
    delta_min = -min(hi, 1 - hj)
    if delta_max <= delta_min: return None # can't move
    delta = random.uniform(delta_min, delta_max) // possibly use temperature scaling factor.
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

import time, random, numpy as np

def _project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n ∩ {sum = target} by bisection on a Lagrange multiplier."""
    low, high = -1.0, 1.0          # guarantees sum(low) > target > sum(high)
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def _compute_c5(h: np.ndarray) -> float:
    """Exact discrete objective: max_k ∫ h(x)(1‑h(x+k)) dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return _project_onto_bounds_and_sum(h, n / 2.0)

def _make_triangle(n: int) -> np.ndarray:
    """Triangular profile, peak 1 at the centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return _project_onto_bounds_and_sum(h, n / 2.0)

def _make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2] (already integrates to 1)."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return _project_onto_bounds_and_sum(h, n / 2.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low Erdős C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ---------------------------------------------------------------
    # problem size – larger n gives more flexibility but still cheap
    n = 600                     # even, integral constraint = n/2
    target_sum = n / 2.0

    best_h = None
    best_val = float("inf")

    # ---------------------------------------------------------------
    # assemble a few seed vectors
    seed_vecs = []

    if "initial_h_values" in globals():
        try:
            seed_vecs.append(_upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seed_vecs.append(_make_triangle(n))
    seed_vecs.append(_make_sin2(n))

    for v in seed_vecs:
        val = _compute_c5(v)
        if val < best_val:
            best_val, best_h = val, v.copy()

    # ---------------------------------------------------------------
    # simulated‑annealing style local search
    cur_h = best_h.copy()
    cur_val = best_val

    T0 = 0.005                     # initial temperature

    while True:
        now = time.time()
        if now - start > budget_s - 0.5:
            break

        # ----- propose a move -------------------------------------------------
        if random.random() < 0.9:                     # two‑point mass transfer
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            hi, hj = cur_h[i], cur_h[j]
            delta_max = min(1.0 - hi, hj)
            delta_min = -min(hi, 1.0 - hj)
            if delta_max <= delta_min:
                continue
            delta = random.uniform(delta_min, delta_max)   # full feasible interval
            cand = cur_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
        else:                                         # global Gaussian perturbation
            sigma = 0.02
            cand = cur_h + sigma * np.random.randn(n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_onto_bounds_and_sum(cand, target_sum)

        cand_val = _compute_c5(cand)
        delta_val = cand_val - cur_val

        elapsed = now - start
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)   # linear cooling

        if delta_val < 0 or random.random() < np.exp(-delta_val / (T + 1e-12)):
            cur_h, cur_val = cand, cand_val
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
                if best_val <= 0.3808:
                    break

    # ---------------------------------------------------------------
    # final greedy refinement (only accept improvements)
    while time.time() - start < budget_s - 0.5:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = random.uniform(delta_min, delta_max)
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cval = _compute_c5(cand)
        if cval < best_val:
            best_val, best_h = cval, cand.copy()
            if best_val <= 0.3808:
                break

    # ---------------------------------------------------------------
    # final safety projection and accurate evaluation
    best_h = _project_onto_bounds_and_sum(best_h, target_sum)
    best_val = _compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
