def simulated_annealing(h, rng, max_iters=200000, t0=0.01):
    n = len(h)
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    
    # Precompute indices of ones and zeros for efficient random selection
    for it in range(max_iters):
        # Temperature schedule
        T = t0 * (1 - it / max_iters)
        # Choose random swap
        # Choose random 1 index:
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            # update best
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c
```

import time, numpy as np, math
from scipy.optimize import minimize

# Global vars for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def binary_from_continuous(v):
    n = v.size
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros_like(v, dtype=float)
    h[idx] = 1.0
    return h

def upsample_pattern(h, factor):
    return np.repeat(h, factor)

def simulated_annealing(h0, rng, max_iters=100000, t0=0.02):
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # maintain lists of indices where h==1 and h==0
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(1 - cur_h))
    for it in range(max_iters):
        T = t0 * (1.0 - it / max_iters)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        # update index lists
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            # use standard Metropolis criterion
            accept = rng.random() < math.exp(-delta / T)
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            # revert list updates
            ones.append(i)
            zeros.append(j)
            ones.remove(j)
            zeros.remove(i)
    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

        cand_h, cand_c = simulated_annealing(h_start, rng, max_iters=max_iters, t0=0.015)
        if cand_c < best_c:

        # occasional random restart

            cand_h, cand_c = simulated_annealing(rand_h, rng, max_iters=max_iters//2, t0=0.02)
            if cand_c < best_c:

    # Final feasibility clean‑up

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the SLSQP surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run a single SLSQP optimisation with a soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the integral exactly (numerical drift may appear)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def binary_from_continuous(v):
    """Return a binary vector with exactly half the entries equal to 1."""
    n = v.size
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros_like(v, dtype=float)
    h[idx] = 1.0
    return h


def upsample_pattern(h, factor):
    """Repeat each entry ``factor`` times."""
    return np.repeat(h, factor)


def simulated_annealing(h0, rng, max_iters=100000, t0=0.02):
    """
    Simple SA on a binary vector that respects the exact sum constraint.
    Swaps one 1 with one 0 at each iteration.
    """
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(1 - cur_h))

    for it in range(max_iters):
        T = t0 * (1.0 - it / max_iters)
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)

        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            accept = rng.random() < math.exp(-delta / T)

        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            ones.append(i)
            zeros.append(j)
            ones.remove(j)
            zeros.remove(i)

    return best_h, best_c


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb on a continuous vector (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
