def ascend_h(h, max_iter=2000, step0=0.5):
    n = len(h)
    target = n/2
    for it in range(max_iter):
        # compute self-correlation
        corr = np.correlate(h, h, mode='full')
        min_corr = corr.min()
        # index of min
        idx = corr.argmin()
        shift = idx - (n - 1)  # shift range negative..positive

        # compute gradient g
        g = np.zeros_like(h)
        if shift >= 0:
            # positive shift
            # for i = 0..n-1-shift:
            g[:n - shift] += h[shift:]  # neighbor on right
            # for i = shift..n-1:
            g[shift:] += h[:n - shift]  # neighbor on left
        else:
            s = -shift
            g[s:] += h[:n - s]         # neighbor on left (since shift negative)
            g[:n - s] += h[s:]         # neighbor on right
        # optional scaling: since each term appears twice, we counted each pair twice (except when shift=0). That's okay.
        # Perform step.
        alpha = step0 / np.sqrt(it + 1)  # decreasing step
        h_new = h + alpha * g
        # Clip
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to sum
        h_new = _project_to_feasible(h_new, n)
        # Evaluate improvement:
        new_corr = np.correlate(h_new, h_new, mode='full')
        new_min = new_corr.min()
        if new_min > min_corr + 1e-12:
            h = h_new
        else:
            # maybe reduce step size or break
            # if not improved for some iterations break

    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    corr = np.fft.irfft(H * np.conj(H), n=N)
    # corr[0..n-1] corresponds to lag 0..n-1 (positive shift); corr[N-n+1..N-1] corresponds to negative lags - (n-1).. -1
    # We'll combine to full length 2n-1: first n entries (lag 0..n-1) and last n-1 entries (negative lags)
    full_corr = np.empty(2 * n - 1, dtype=corr.dtype)
    full_corr[:n] = corr[:n]
    full_corr[n:] = corr[N - n + 1:]  # skip the duplicated zero
    return full_corr

def gradient_step_min_self(h, beta):
    n = len(h)
    # compute self_corr
    self_corr = np.correlate(h, h, mode='full')  # length 2n-1
    # compute weights
    # to avoid overflow, subtract max?
    w = np.exp(-beta * self_corr)  # might be small
    w_sum = w.sum()
    w = w / w_sum
    # compute g = sum_s w_s * (shifted sums)
    g = np.zeros_like(h)
    # shift indices: correlate self_corr index k corresponds to shift s = k - (n-1)
    for idx, w_s in enumerate(w):
        s = idx - (n - 1)
        if w_s == 0:
            continue
        if s >= 0:
            # contributions: for i from 0 to n-1-s: add w_s * h[i+s] to g[i]; for i from s to n-1: add w_s * h[i-s] to g[i]
            g[:n-s] += w_s * h[s:]   # h[i+s] -> g[i]
            g[s:] += w_s * h[:n-s]   # h[i-s] -> g[i]
        else:
            s_abs = -s
            # s negative, same as positive s with shift sign swapped
            # for i from s_abs to n-1: g[i] += w_s * h[i - s_abs]
            # for i from 0 to n-1-s_abs: g[i] += w_s * h[i + s_abs]
            g[s_abs:] += w_s * h[:n-s_abs]
            g[:n-s_abs] += w_s * h[s_abs:]
    return g

def _optimize_h_start(h0, n_iter=2000, beta=10, step0=0.5):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    for it in range(1, n_iter+1):
        # compute self correlation
        self_corr = np.correlate(h, h, mode='full')
        min_self = self_corr.min()
        diff = self_corr - min_self
        w = np.exp(-beta * diff)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        # compute gradient
        g = np.zeros_like(h)
        n = h.size
        for idx, w_s in enumerate(w):
            if w_s == 0:
                continue
            s = idx - (n - 1)
            if s >= 0:
                g[:n-s] += w_s * h[s:]
                g[s:] += w_s * h[:n-s]
            else:
                s_abs = -s
                g[s_abs:] += w_s * h[:n-s_abs]
                g[:n-s_abs] += w_s * h[s_abs:]
        # step size schedule
        alpha = step0 / np.sqrt(it)
        h = h + alpha * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # project to sum = n/2
        h = _project_to_feasible(h, n)
        # compute C5
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            # early exit if below target?
        # optional: if it%100 ==0: maybe reduce beta?
    return best_h, best_c5

def _project_to_feasible(h, n, tol=1e-12, max_iter=100):
    target = n/2.0
    # Clip
    h = np.clip(h, 0.0, 1.0)
    low = -1.0
    high = 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        h_proj = np.clip(h + lam, 0.0, 1.0)
        s = h_proj.sum()
        if abs(s - target) < tol:
            return h_proj
        if s > target:
            high = lam
        else:
            low = lam
    # final projection
    return np.clip(h + lam, 0.0, 1.0)

def _c5_from_h(h):
    n = h.size
    min_self = np.min(np.correlate(h, h, mode='full'))
    return 1.0 - (2.0 * min_self) / n

def _gradient_optimize(h0, n_iter=1500, beta=10.0, step0=0.5):
    h = np.copy(h0)
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    for it in range(1, n_iter+1):
        # compute self-correlation
        self_corr = np.correlate(h, h, mode='full')
        min_self = self_corr.min()
        diff = self_corr - min_self  # non-negative
        # compute weights
        # to prevent underflow, maybe compute w = np.exp(-beta * diff)
        # but diff may be up to maybe 30, beta*diff large => zero; will cause w sum maybe only for min_self entries
        w = np.exp(-beta * diff)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # compute gradient
        g = np.zeros_like(h)
        # shift index mapping: idx 0 -> shift = -(n-1)
        offset = n - 1
        for idx, w_s in enumerate(w):
            if w_s == 0.0:
                continue
            s = idx - offset
            if s >= 0:
                g[:n - s] += w_s * h[s:]
                g[s:] += w_s * h[:n - s]
            else:
                s_abs = -s
                g[s_abs:] += w_s * h[:n - s_abs]
                g[:n - s_abs] += w_s * h[s_abs:]

        alpha = step0 / np.sqrt(it)
        h = h + alpha * g
        # Clip and project
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_feasible(h, n)
        # evaluate
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # early stop if reaches target
        # optional early break if not improving for many iterations?
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Projection onto the feasible set: 0 ≤ h ≤ 1 and sum(h) = n/2
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 80) -> np.ndarray:
    """Euclidean projection onto the capped simplex."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0  # enough to drive sum to 0 or n after clipping
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(h + lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(h + lam, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation (max overlap)
def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 = max_x ∫ h(x)(1−h(x+x)) dx via discrete correlation."""
    n = h.size
    # max_{k} Σ h[i] (1−h[i+k]) = Σ h[i] − min_{k} Σ h[i]h[i+k]
    # Σ h[i] = n/2 by construction
    min_self = np.min(np.correlate(h, h, mode="full"))
    return 1.0 - (2.0 * min_self) / n


# ----------------------------------------------------------------------
# SLSQP polishing (optional, used only on the best candidate)
def _objective(x: np.ndarray) -> float:
    return x[-1]  # minimise the auxiliary variable t


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    # t - corr >= 0  (elementwise)
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """Run a brief SLSQP optimisation starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----------------------------------------------------------------------
# Gradient ascent on the smooth approximation of min_self
def _gradient_optimize(
    start_h: np.ndarray,
    n_iter: int = 1500,
    beta: float = 12.0,
    step0: float = 0.5,
) -> tuple:
    """Projected gradient ascent that maximises the minimal self‑correlation."""
    h = start_h.copy()
    n = h.size
    best_h, best_c5 = h.copy(), _c5_from_h(h)

    for it in range(1, n_iter + 1):
        # self‑correlation (full)
        self_corr = np.correlate(h, h, mode="full")
        min_self = self_corr.min()
        diff = self_corr - min_self  # non‑negative

        # soft‑min weights
        w = np.exp(-beta * diff)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # gradient = Σ w_s (h shifted by +s + h shifted by −s)
        g = np.zeros_like(h)
        offset = n - 1
        for idx, ws in enumerate(w):
            if ws == 0.0:
                continue
            s = idx - offset
            if s >= 0:
                g[: n - s] += ws * h[s:]
                g[s:] += ws * h[: n - s]
            else:
                s_abs = -s
                g[s_abs:] += ws * h[: n - s_abs]
                g[: n - s_abs] += ws * h[s_abs:]

        # diminishing step size
        alpha = step0 / np.sqrt(it)
        h = h + alpha * g
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_feasible(h, n)

        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    return best_h, best_c5


# ----------------------------------------------------------------------
# Simple random feasible starter
def _random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    return _project_to_feasible(h, n)


# ----------------------------------------------------------------------
# Parameterised cosine mixture generator
def _cosine_mixture_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random mixture of a few low‑frequency cosine terms."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    coeffs = rng.uniform(-0.5, 0.5, size=3)  # a1,a2,a3
    h = (
        0.5
        + coeffs[0] * np.cos(np.pi * x)
        + coeffs[1] * np.cos(2 * np.pi * x)
        + coeffs[2] * np.cos(3 * np.pi * x)
    )
    h = np.clip(h, 0.0, 1.0)
    return _project_to_feasible(h, n)

# Main entry point
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as a list), the achieved C5 bound, and n_points.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Pull optional initial construction if supplied by the harness
    try:
        initial_h = np.array(initial_h_values, dtype=float)  # type: ignore
    except NameError:
        initial_h = None

    target_bound = 0.3808

    best_h, best_c5, best_n = None, float("inf"), None

    # Candidate grid sizes (small, then a couple of finer grids)
    n_candidates = [128, 192, 256]
    if initial_h is not None and initial_h.size not in n_candidates:
        n_candidates.append(initial_h.size)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # ----------------------------------------------------------------
        # Assemble a small set of promising starting vectors for this size
        starts = []

        # 1) use supplied construction (up‑scaled / down‑scaled as needed)
        if initial_h is not None:
            if initial_h.size != n:
                # linear interpolation to the new grid
                src_x = (np.arange(initial_h.size) + 0.5) * (2.0 / initial_h.size)
                tgt_x = (np.arange(n) + 0.5) * (2.0 / n)
                interp = np.interp(tgt_x, src_x, initial_h)
                starts.append(_project_to_feasible(interp, n))
            else:
                starts.append(initial_h.copy())

        # 2) a few deterministic cosine shapes (varying amplitude)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        for a in np.linspace(0.1, 0.5, 5):
            h = 0.5 + a * np.cos(np.pi * x)
            h = np.clip(h, 0.0, 1.0)
            starts.append(_project_to_feasible(h, n))

        # 3) random feasible vectors
        for _ in range(6):
            starts.append(_random_start(n, rng))

        # 4) random low‑frequency cosine mixtures
        for _ in range(8):
            starts.append(_cosine_mixture_start(n, rng))

        # ----------------------------------------------------------------
        # Optimise each starter
        for h0 in starts:
            if time_left() < 3.0:
                break

            # Gradient ascent on smooth min‑self‑correlation
            h_opt, c5_opt = _gradient_optimize(h0, n_iter=1800, beta=12.0, step0=0.6)

            # Polishing with a short SLSQP run if we have an improvement
            if c5_opt < best_c5:
                h_pol, c5_pol = _solve_h(h_opt, max_iter=150)
                if c5_pol < c5_opt:
                    h_opt, c5_opt = h_pol, c5_pol

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                if best_c5 < target_bound:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # ----------------------------------------------------------------------
    # Fallback – uniform construction (should never be needed)
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
