def project_to_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # Scale proportionally
    if cur > 0:
        v = v * (target / cur)
    v = np.clip(v, 0.0, 1.0)
    # adjust residual
    diff = v.sum() - target
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need decrease
            idx = np.where(v > 1e-12)[0]
            if len(idx) == 0:
                break
            i = random.choice(idx.tolist())
            delta = min(v[i], diff)
            v[i] -= delta
        else:
            idx = np.where(v < 1 - 1e-12)[0]
            if len(idx) == 0:
                break
            i = random.choice(idx.tolist())
            delta = min(1 - v[i], -diff)
            v[i] += delta
        diff = v.sum() - target
    return v

import random
import time
import math
import numpy as np

def _full_corr(v: np.ndarray) -> np.ndarray:
    """Linear cross–correlation of v with (1‑v) using FFT (full length)."""
    n = v.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    vr = v[::-1]
    f1 = np.fft.rfft(vr, size)
    f2 = np.fft.rfft(1.0 - v, size)
    conv = np.fft.irfft(f1 * f2, size)[: 2 * n - 1]
    return conv


def _c5_of(v: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k))dx for the discretised vector v."""
    return float(_full_corr(v).max() * (2.0 / len(v)))


def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust (by transfers) so that sum(v)=target exactly."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if cur > 0:
        v = v * (target / cur)
    v = np.clip(v, 0.0, 1.0)
    # fix any remaining tiny deviation by local transfers
    for _ in range(200):
        diff = v.sum() - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:  # need to decrease total mass
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            i = random.choice(cand.tolist())
            delta = min(v[i], diff)
            v[i] -= delta
        else:        # need to increase total mass
            cand = np.where(v < 1 - 1e-12)[0]
            if cand.size == 0:
                break
            i = random.choice(cand.tolist())
            delta = min(1.0 - v[i], -diff)
            v[i] += delta
    return v
