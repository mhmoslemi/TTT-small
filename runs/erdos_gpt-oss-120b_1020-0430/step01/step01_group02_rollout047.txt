def _adjust_sum(x, target):
    """Project x onto [0,1] with sum(x)=target (simple iterative redistribution)."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        per = diff / idx.size

            diff -= add.sum()
