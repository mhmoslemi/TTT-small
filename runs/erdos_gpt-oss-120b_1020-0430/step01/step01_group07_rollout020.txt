    idx = np.argmax(c)
    k = idx - (n - 1)
    # compute contributions vector

    end = min(n, n - k)  # exclusive
    # slice region
    lo = start
    hi = end
    # compute contributions as h[lo:hi] * (1 - h[lo+k:hi+k])
    contrib = np.zeros_like(h)
    contrib[lo:hi] = h[lo:hi] * (1 - h[lo+k:hi+k])

    i_max = np.argmax(contrib)

    # feasible delta
    delta = min(h[i_max], 1 - h[i_min])
    if delta <= 0: return False

    # to avoid overshooting, move a fraction like 0.5 * delta
    delta_move = delta * 0.5
    h[i_max] -= delta_move

def anneal(h, max_seconds):
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c

    start = time.time()
    T0 = 0.02
    T_end = 1e-6
    while time.time() - start < max_seconds:
        # temperature schedule linear in log-space
        progress = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        i, j = random.sample(range(n), 2)
        hi = cur_h[i]
        hj = cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # limit step size
        max_step = 0.2
        low_c = max(low, -max_step)
        high_c = min(high, max_step)
        delta = random.uniform(low_c, high_c) if low_c < high_c else random.uniform(low, high)
        # move mass from j to i (i gets delta, j loses delta)
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_c = compute_c5(cur_h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = hi
            cur_h[j] = hj
    return best_h, best_c

import time
import random
import math
import numpy as np

def _fft_len(m: int) -> int:
    """Next power‑of‑two length for an FFT of size m."""
    return 1 << ((m - 1).bit_length())

def _correlation_full(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1‑h) using FFT (zero‑padded)."""
    n = h.shape[0]
    m = 2 * n - 1
    nfft = _fft_len(m)
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft((1.0 - h)[::-1], nfft)
    corr = np.fft.irfft(H * G, nfft)[:m]
    return corr

def _c5_metric(h: np.ndarray) -> float:
    """C₅ value for a step function sampled by array h."""
    n = h.shape[0]
    corr = _correlation_full(h)
    return np.max(corr) * (2.0 / n)

def _anneal(start_h: np.ndarray, max_seconds: float):
    """Simple simulated‑annealing refinement."""
    h = start_h.copy()
    n = h.size
    cur_c = _c5_metric(h)
    best_h = h.copy()
    best_c = cur_c

    t0, t_end = 0.04, 1e-6
    max_step = 0.15
    start = time.time()
    while time.time() - start < max_seconds:
        prog = (time.time() - start) / max_seconds
        T = t0 * (t_end / t0) ** prog

        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        lo_cap = max(low, -max_step)
        hi_cap = min(high, max_step)
        delta = random.uniform(lo_cap, hi_cap) if lo_cap < hi_cap else random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c

    random.seed(seed)

    else:

    cur = h.sum()
    if cur != 0:

            add = min(room, deficit)
