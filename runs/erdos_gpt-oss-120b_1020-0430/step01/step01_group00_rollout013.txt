def _project_to_feasible(h, target_sum, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    # compute sum
    cur = h.sum()
    diff = cur - target_sum
    if abs(diff) <= tol:
        return h
    # Use simple iterative approach: while diff not zero:
    for _ in range(max_iter):
        # find indices that can be decreased/increased
        if diff > 0:
            # need to decrease sum, choose indices >0
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]

        diff = cur - target_sum
        if abs(diff) <= tol:
            break
    # final leftover fix if still off by tiny amount

import time
import numpy as np
from scipy import optimize

# ---------- feasibility projection ----------
def _project_to_feasible(x: np.ndarray, target_sum: float, tol: float = 1e-12,
                         max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and adjust to obtain exactly target_sum (water‑filling)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = cur - target_sum
    if abs(diff) <= tol:
        return x
    for _ in range(max_iter):
        if diff > 0:
            free = np.where(x > 0)[0]
        else:
            free = np.where(x < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        if diff > 0:
            x[free] = np.maximum(x[free] - delta, 0.0)
        else:
            x[free] = np.minimum(x[free] - delta, 1.0)
        cur = x.sum()
        diff = cur - target_sum
        if abs(diff) <= tol:
            break
    # tiny final correction (should be negligible)
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        for i in range(len(x)):
            if residual > 0 and x[i] < 1 - 1e-12:
                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 1e-12:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


# ---------- C5 evaluation ----------
def _c5_from_h(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


# ---------- SLSQP optimiser ----------
def _solve_slsqp(start_h: np.ndarray, max_iter: int = 400) -> tuple:
    n = start_h.size
    h0 = _project_to_feasible(start_h, n / 2.0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = corr0.max()
    x0 = np.concatenate([h0, [t0]])

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n / 2.0)
    return h_opt, _c5_from_h(h_opt)


# ---------- gradient descent refinement ----------
def _gradient_refine(h: np.ndarray, time_left, max_iter: int = 2000,
                    init_step: float = 0.02) -> tuple:
    n = h.size
    target = n / 2.0
    step = init_step
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    for it in range(max_iter):
        if time_left() < 0.5:
            break
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_corr = corr.max()
        idxs = np.where(corr >= max_corr - 1e-12)[0]
        grad = np.zeros_like(cur_h)
        for idx in idxs:
            shift = idx - (n - 1)
            if shift >= 0:
                grad[:n - shift] += cur_h[shift:]
                grad[shift:] += cur_h[: n - shift]
            else:
                s = -shift
                grad[s:] += cur_h[: n - s]
                grad[: n - s] += cur_h[s:]
        new_h = cur_h + step * grad
        new_h = _project_to_feasible(new_h, target)
        new_c5 = _c5_from_h(new_h)

        if new_c5 < cur_c5 - 1e-9:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
            step = min(step * 1.1, 0.2)
        else:
            step *= 0.7
            if np.random.rand() < 0.1:
                pert = (np.random.rand(n) - 0.5) * step * 0.5
                cand = _project_to_feasible(cur_h + pert, target)
                cand_c5 = _c5_from_h(cand)
                if cand_c5 < cur_c5:
                    cur_h, cur_c5 = cand, cand_c5
                    if cand_c5 < best_c5:
                        best_h, best_c5 = cand.copy(), cand_c5
        if step < 1e-6:
            break
    return best_h, best_c5


# ---------- simulated annealing refinement ----------
def _anneal_refine(h: np.ndarray, time_left, max_steps: int = 4000,
                  init_temp: float = 0.005, decay: float = 0.999) -> tuple:
    n = h.size
    target = n / 2.0
    cur_h = h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5
    temp = init_temp

    for _ in range(max_steps):
        if time_left() < 0.5:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # allowable transfer from i to j
        max_plus = min(cur_h[i], 1.0 - cur_h[j])
        max_minus = min(cur_h[j], 1.0 - cur_h[i])
        # choose direction uniformly
        if np.random.rand() < 0.5:
            delta = np.random.rand() * max_plus
        else:
            delta = -np.random.rand() * max_minus
        if abs(delta) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = _c5_from_h(new_h)
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
        temp *= decay
        if temp < 1e-8:
            temp = init_temp
    return best_h, best_c5


# ---------- up‑sampling helper ----------
def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse to a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)
