def softmax_c5(h, alpha):
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(conv)
    # We'll compute log-sum-exp with scaling to avoid overflow:
    exp_terms = np.exp(alpha * (conv - max_val))
    sum_exp = np.sum(exp_terms)
    result = (max_val + np.log(sum_exp) / alpha)
    # Because we used scaling, ensure correct: we used shift by max_val to improve numeric stability:
    # log(sum_i exp(alpha*conv_i)) = max_val*alpha + log(sum_i exp(alpha*(conv_i - max_val))) => final = (max_val*alpha + log(...))/alpha = max_val + log(...)/alpha.
    return result

def softmax_c5_grad(h, alpha):
    N = len(h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(conv)
    exp_terms = np.exp(alpha * (conv - max_val))
    sum_exp = np.sum(exp_terms)
    # probabilities:
    p = exp_terms / sum_exp
    # compute g = sum_i p_i * grad(conv_i)
    g = np.zeros(N)
    # Compute gradient for each i (lag L)
    # conv index i from 0 to M-1, L = i - (N-1)
    for idx, weight in enumerate(p):
        L = idx - (N - 1)
        # first contribution:
        # for j where j+L < N
        if L >= 0:
            # j from 0 to N-1-L: j+L inside
            j_end = N - L
            # compute 1 - h[j+L] for these j
            # g[0:j_end] += weight * (1 - h[ L : N ])
            g[:j_end] += weight * (1.0 - h[L:])
            # for j where j+L >= N (i.e., j from N-L to N-1), contribution is weight * 1
            if L > 0:
                g[j_end:] += weight * 1.0
        else:
            # L negative, so j+L may be <0 for some j: j from -L to N-1, j+L >=0
            # For j from -L to N-1: j+L index valid
            j_start = -L
            # g[0:j_start] => j+L < 0 => contribution = weight * 1
            if j_start > 0:
                g[:j_start] += weight * 1.0
            # for j >= j_start:
            g[j_start:] += weight * (1.0 - h[:N + L])  # because L negative, N+L < N
        # second contribution: - weight * h[j-L] where j-L >=0
        if L <= 0:
            # j - L >=0 => j >= -L
            j_start = -L
            # for j from j_start to N-1, subtract weight * h[j-L]

            g[j_start:] -= weight * h[:N - L]  # because j-L in [0,N-1+L] (since L<=0)

            # j - L >=0 => j >= L
            # for j from L to N-1 subtract weight * h[j-L] which index 0..N-1-L
            g[L:] -= weight * h[:N - L]
    return g

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0,2.0,old_n,endpoint=False) + 1.0/old_n
    new_x = np.linspace(0.0,2.0,new_n,endpoint=False) + 1.0/new_n
    up = np.interp(new_x, old_x, arr)
    # adjust sum
    target = new_n/2.0
    up = up * (target / up.sum())
    # ensure bound
    up = np.clip(up,0.0,1.0)
    # project if needed
    cur = up.sum()
    if abs(cur - target) > 1e-12:
        lo = np.min(up - 1.0)
        hi = np.max(up)
        for _ in range(30):
            lam = (lo+hi)/2
            proj = np.clip(up - lam,0.0,1.0)
            s = proj.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        up = np.clip(up - ((lo+hi)/2),0.0,1.0)

def project_to_feasible(h, target):
    # ensure within [0,1] and sum = target via binary search on lambda
    # Input h may be slight outside range
    # We'll first clip to [0,1] then do binary search
    h = np.clip(h, 0.0, 1.0)
    # if sum already = target close, return
    cur = h.sum()
    if abs(cur - target) <= 1e-12:
        return h
    # Determine search bounds for lambda
    lo = np.min(h - 1.0)  # subtract up to make all 1 -> sum=n
    hi = np.max(h)        # subtract enough to make all 0 -> sum=0
    # Actually if we add lambda (subtraction) we want projection = clip(h - lam,0,1)
    # So lam small => close to original, lam large => pushes towards 0.
    # For lam = lo (most negative) -> h - lo = h + |lo| > 1 for all => clip=1 => sum=n.
    # For lam = hi => h - hi <= 0 for all => clip=0 => sum=0.
    # So target between 0 and n inclusive.
    for _ in range(40):
        lam = (lo + hi) / 2.0
        proj = np.clip(h - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(h - ((lo+hi)/2.0), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    conv = np.correlate(h, 1.0 - h, mode='full')
    # conv shape len 2n-1
    c5 = conv.max() * dx
    return float(c5)

def softmax_c5_and_grad(h, alpha):
    # Use scaling for log-sum-exp.
    N = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_val = conv.max()
    exp_terms = np.exp(alpha * (conv - max_val))
    sum_exp = exp_terms.sum()
    # probabilities:
    p = exp_terms / sum_exp
    # gradient
    g = np.zeros(N)
    # loop over idx and weight
    # compute L = idx - (N - 1)
    for idx, weight in enumerate(p):
        L = idx - (N - 1)
        # First term f1 (1 - h[j+L] if in range else 1)
        if L >= 0:
            # part where j+L < N
            if L < N:
                limit = N - L
                # f1 for j in [0, limit)
                g[:limit] += weight * (1.0 - h[L:])
                # for remaining j, add weight * 1
                if limit < N:
                    g[limit:] += weight * 1.0
            else:
                # L >= N => all j+L out of range
                g += weight * 1.0
        else:
            shift = -L
            # for j < shift, contribution = 1
            if shift > 0:
                g[:shift] += weight * 1.0
                # for j >= shift, contribution = 1 - h[j - shift]
                g[shift:] += weight * (1.0 - h[:N - shift])
            else:
                # shift==0 => L=0; treat as same as L>=0 case
                # Actually L=0 falls in forward case already.
                pass

        # Second term f2 = h[j-L] if j-L >=0 else 0, subtract.
        if L >= 0:
            # subtract h[j-L] for j >= L
            if L < N:
                g[L:] -= weight * h[:N - L]
        else:
            shift = -L
            # subtract h[j+shift] for j < N - shift
            if shift < N:
                g[:N - shift] -= weight * h[shift:]
    # objective value
    value = max_val + np.log(sum_exp) / alpha
    return value, g

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def softmax_c5_and_grad(h: np.ndarray, alpha: float):
    """Smooth max (log‑sum‑exp) of the correlation together with its gradient."""
    N = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = conv.max()
    # stable log‑sum‑exp
    exp_terms = np.exp(alpha * (conv - m))
    s = exp_terms.sum()
    probs = exp_terms / s

    # gradient: g = Σ p_i ∇conv_i
    g = np.zeros(N)
    for idx, w in enumerate(probs):
        L = idx - (N - 1)                     # lag = i - (N‑1)
        if L >= 0:
            limit = N - L
            if limit > 0:
                g[:limit] += w * (1.0 - h[L:])
                if limit < N:
                    g[limit:] += w               # out‑of‑range part (value 1)
            else:                                 # L ≥ N, all out‑of‑range
                g += w
            if L < N:
                g[L:] -= w * h[:N - L]            # second term
        else:                                       # L < 0
            shift = -L
            if shift > 0:
                limit = N - shift
                if limit > 0:
                    g[:shift] += w               # out‑of‑range part (value 1)
                    g[shift:] += w * (1.0 - h[:limit])
                else:                            # shift ≥ N
                    g += w
            if shift < N:
                g[:N - shift] -= w * h[shift:]
    value = m + np.log(s) / alpha
    return float(value), g


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    v, _ = softmax_c5_and_grad(h, alpha)
    return v


def softmax_c5_grad(h: np.ndarray, alpha: float = 30.0):
    _, g = softmax_c5_and_grad(h, alpha)
    return g


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0 ≤ h ≤ 1, Σh = target} via a simple binary search."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) <= 1e-12:
        return h
    lo = np.min(h - 1.0)          # enough to make all entries 1
    hi = np.max(h)                # enough to make all entries 0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        proj = np.clip(h - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point in the box [0,1]^n with exact sum = n/2."""
    target = n / 2.0
    remaining = target
    a = np.empty(n)
    for i in range(n - 1):
        lo = max(0.0, remaining - (n - i - 1))
        hi = min(1.0, remaining)
        a[i] = rng.uniform(lo, hi)
        remaining -= a[i]
    a[-1] = remaining
    return a


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape min(x,2‑x), rescaled to satisfy the integral."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    return project_to_feasible(tri, target)


def generate_sine(n: int, amplitude: float = 0.4) -> np.ndarray:
    """Sinusoidal bump h=0.5+amp·sin(πx/2), then projected to feasibility."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    h = 0.5 + amplitude * np.sin(np.pi * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    return project_to_feasible(h, target)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, followed by feasibility projection."""
    old_n = arr.shape[0]
    old_x = np.arange(old_n) * (2.0 / old_n) + (1.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n) + (1.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    return project_to_feasible(up, target)


def optimize_h(start: np.ndarray, n: int, alphas, deadline: float) -> np.ndarray:
    """Stage‑wise SLSQP optimisation using a smooth max surrogate."""
    h = start.copy()
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    for a in alphas:
        if time.time() > deadline:
            break
        res = minimize(
            fun=lambda x, aa=a: softmax_c5(x, aa),
            x0=h,
            jac=lambda x, aa=a: softmax_c5_grad(x, aa),
            method="SLSQP",
            bounds=bounds,
            constraints=eq_cons,
            options={"maxiter": 150, "ftol": 1e-9, "disp": False},
        )
        h = np.clip(res.x, 0.0, 1.0)
        h = project_to_feasible(h, target)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + 0.95 * budget_s  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Candidate discretisations (even numbers give a perfectly feasible binary point)
    candidate_ns = [120, 150, 180, 200]

    # Parent construction, if supplied by the harness

    except NameError:
        parent_h = None
        parent_n = None

    # Smooth‑max schedule
    alpha_schedule = [5.0, 12.0, 25.0, 50.0]

    for n in candidate_ns:
        if time.time() > time_limit:
            break
        target = n / 2.0
        starts = []

        # 1. Upsampled parent (or direct copy)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # 2. Classical triangular shape
        starts.append(generate_triangular(n))

        # 3. Sinusoidal shapes with two amplitudes
        for amp in (0.35, 0.45):
            starts.append(generate_sine(n, amplitude=amp))

        # 4. Several random feasible vectors
        for _ in range(3):
            starts.append(random_feasible(n, rng))

        # 5. Perturbations of the triangular shape
        tri = generate_triangular(n)
        for _ in range(2):
            pert = tri + rng.uniform(-0.15, 0.15, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_to_feasible(pert, target)
            starts.append(pert)

        # Optimize each start; keep the best for this resolution
        for s in starts:
            if time.time() > time_limit:
                break
            h_opt = optimize_h(s, n, alpha_schedule, time_limit)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5 - 1e-9:
                best_h = h_opt
                best_c5 = c5_val
                best_n = n

    # ------------------------------------------------------------
    # Second‑stage refinement on a finer grid (if time permits)
    # ------------------------------------------------------------
    if best_h is not None and time.time() < time_limit:
        high_n = best_n * 2
        if high_n <= 500:                     # keep the problem tractable
            h_up = upsample(best_h, high_n)
            h_refined = optimize_h(h_up, high_n, alpha_schedule, time_limit)
            c5_ref = compute_c5(h_refined)
            if c5_ref < best_c5:
                best_h, best_c5, best_n = h_refined, c5_ref, high_n

    # Final safety checks / fallback
    # ------------------------------------------------------------
    if best_h is None:
        # Simple binary block (guaranteed feasible)
        n = 120
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_n = n
        best_c5 = compute_c5(best_h)
    else:
        best_h = project_to_feasible(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
