def local_optimize(start_h, max_iter, time_limit):
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    eta = 0.5
    decay = 0.9995
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # compute correlation and find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        s = idx_max - (n - 1)
        # compute gradient
        grad = np.zeros(n)
        if s >= 0:
            L = n - s
            i = np.arange(L)
            grad[i] += 1.0 - h[i + s]
            grad[i + s] -= h[i]
        else:
            s_abs = -s
            L = n - s_abs
            i = np.arange(L)
            grad[i] -= h[i + s_abs]
            grad[i + s_abs] += 1.0 - h[i]
        # gradient step
        y = h - eta * grad
        h = project_simple(y)
        # evaluate
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()
        eta *= decay
    return best_h, best_c5

def run(...):
    import time, numpy as np, random, math
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # decide n
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    except NameError:
        n = 70
        init = np.random.rand(n)
    target_sum = n/2.0

    # projection function
    def project(vec):
        # clip then waterfilling
        lo, hi = 0.0, 1.0
        x = np.clip(vec, lo, hi)
        diff = target_sum - x.sum()
        if abs(diff) < 1e-12:
            return x
        # water-filling
        lam_low = np.min(vec - hi) - 1.0
        lam_high = np.max(vec - lo) + 1.0
        for _ in range(30):
            lam = (lam_low + lam_high) / 2.0
            x = np.clip(vec - lam, lo, hi)
            s = x.sum()
            if s > target_sum:
                lam_low = lam
            else:
                lam_high = lam
        x = np.clip(vec - (lam_low+lam_high)/2.0, lo, hi)
        # ensure sum accurate
        # tiny correction:
        err = target_sum - x.sum()
        if abs(err) > 1e-9:
            # adjust a random non-bound index
            idx = random.choice(np.where((x>lo+1e-12) & (x<hi-1e-12))[0].tolist()
                                or list(range(n)))
            x[idx] += err
            x = np.clip(x, lo, hi)
        return x

    # Ensure start feasible
    h0 = project(init)

    # compute c5 function
    def compute_c5(v):
        corr = np.correlate(v, 1 - v, mode='full')
        return float(corr.max() * (2.0 / n))

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # define optimizer for a given start
    def gradient_opt(start_vec, time_limit):
        h = start_vec.copy()
        eta = 0.5
        decay = 0.9995
        local_best_h = h.copy()
        local_best_c5 = compute_c5(h)
        iter_count = 0
        while time.time() - start_time < time_limit:
            # compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            idx_max = np.argmax(corr)
            s = idx_max - (n - 1)
            # gradient
            grad = np.zeros(n)
            if s >= 0:
                L = n - s
                i = np.arange(L)
                grad[i] += 1.0 - h[i + s]
                grad[i + s] -= h[i]
            else:
                s_abs = -s
                L = n - s_abs
                i = np.arange(L)
                grad[i] -= h[i + s_abs]
                grad[i + s_abs] += 1.0 - h[i]
            # step
            y = h - eta * grad
            h = project(y)
            cur_c5 = compute_c5(h)
            if cur_c5 < local_best_c5:
                local_best_c5, local_best_h = cur_c5, h.copy()
            eta *= decay
            iter_count += 1
            # optionally break after many iterations
            if iter_count >= 5000:
                # early exit if time left maybe start new restart
                break
        return local_best_h, local_best_c5

    # overall loop with restarts
    time_per_restart = budget_s * 0.9 / 8  # say 8 restarts
    # Ensure at least some minimal time per restart
    while time.time() - start_time < budget_s * 0.95:
        # create random start around best or initial
        # Add perturbation to best_h
        noise = np.random.normal(scale=0.2, size=n)
        start = best_h + noise

    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # Starting vector (use the provided one if available)
    # -----------------------------------------------------------

    target_sum = n / 2.0

    # -----------------------------------------------------------
    # Projection onto the bounded simplex {0<=x_i<=1, sum x_i = target}
    # -----------------------------------------------------------
    def project(vec):
        lo, hi = 0.0, 1.0
        x = np.clip(vec, lo, hi)
        s = x.sum()
        if abs(s - target_sum) < 1e-12:
            return x
        # water‑filling via bisection on λ where x_i = clip(vec_i-λ, lo, hi)
        lam_low = np.min(vec - hi) - 1.0
        lam_high = np.max(vec - lo) + 1.0
        for _ in range(30):
            lam = (lam_low + lam_high) / 2.0
            x = np.clip(vec - lam, lo, hi)
            s = x.sum()
            if s > target_sum:
                lam_low = lam
            else:
                lam_high = lam
        x = np.clip(vec - (lam_low + lam_high) / 2.0, lo, hi)

        # tiny correction if rounding left a residue
        err = target_sum - x.sum()
        if abs(err) > 1e-9:
            free = np.where((x > lo + 1e-12) & (x < hi - 1e-12))[0]
            if free.size == 0:  # all at a bound – pick any index
                free = np.arange(n)
            i = random.choice(free.tolist())

        return x

    # C5 evaluation
    # -----------------------------------------------------------
    def c5_of(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(corr.max() * (2.0 / n))   # dx = 2/n
