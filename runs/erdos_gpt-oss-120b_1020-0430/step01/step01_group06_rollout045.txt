    random.seed(seed)

    start = time.time()
    # initialize

        idx = np.arange(n)
        np.random.shuffle(idx)
        h[idx[: n // 2]] = 1.0

    h = np.clip(h, 0.0, 1.0)

        diff = target_sum - cur_sum
        # we will adjust small amounts to first entries not at bounds
        for i in range(n):
            if diff == 0:
                break
            if diff > 0:
                # increase h[i] up to 1
                inc = min(1.0 - h[i], diff)
                h[i] += inc
                diff -= inc
            else:
                dec = min(h[i], -diff)
                h[i] -= dec
                diff += dec

    dx = 2.0 / n
    def c5_of(v):
        return float(np.correlate(v, 1 - v, mode="full").max() * dx)

    cur_c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # Simulated annealing parameters
    T0 = max(1e-3, cur_c5 * 0.02)
    T = T0
    decay = 0.9998
    it = 0

    while time.time() - start < budget_s * 0.95:
        it += 1
        # select move type
        r = random.random()
        if r < 0.4:
            # binary swap (or any swap preserving sum)
            # find indices where h is close to 0 or 1
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:

            i = random.choice(ones)
            j = random.choice(zeros)
            nh = h.copy()
            nh[i], nh[j] = nh[j], nh[i]

            max_down = min(hi, 1.0 - hj) # decrease i, increase j

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ---------- initialise ----------
    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
        # up‑sample if possible (keep size reasonable)
        up = 2
        n = n0 * up if n0 * up <= 200 else n0
        if n == n0:
            h = init.copy()
        else:
            h = np.repeat(init, up)
    except NameError:
        n = 140
        h = np.zeros(n)
        idx = np.arange(n)
        np.random.shuffle(idx)
        h[idx[: n // 2]] = 1.0

    h = np.clip(h, 0.0, 1.0)

    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # simple projection onto the hyperplane while respecting bounds
        for i in range(n):
            if diff == 0:
                break
            if diff > 0:
                inc = min(1.0 - h[i], diff)
                h[i] += inc
                diff -= inc
            else:
                dec = min(h[i], -diff)
                h[i] -= dec
                diff += dec

    dx = 2.0 / n

    def c5_of(v):
        """max correlation * dx"""
        return float(np.correlate(v, 1 - v, mode="full").max() * dx)

    cur_c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ---------- optimisation ----------
    # temperature schedule
    T = max(1e-4, cur_c5 * 0.02)
    decay = 0.9998
    it = 0
    # target bound to stop early
    target_bound = 0.38075

    while time.time() - start < budget_s * 0.95:
        it += 1
        # choose move
        r = random.random()
        if r < 0.4:
            # ----- binary‑like swap -----
            # treat values >0.5 as 1, <0.5 as 0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i, j = random.choice(ones), random.choice(zeros)
            nh = h.copy()
            nh[i], nh[j] = nh[j], nh[i]
        elif r < 0.7:
            # ----- small epsilon transfer -----
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            max_up = min(1.0 - hi, hj)      # increase i, decrease j
            max_down = min(hi, 1.0 - hj)    # decrease i, increase j
            if max_up + max_down == 0:
                continue
            eps = (random.random() * (max_up + max_down)) - max_down
            if abs(eps) < 1e-14:
                continue
            nh = h.copy()
            nh[i] = hi + eps
            nh[j] = hj - eps
        else:
            # ----- targeted move on worst shift -----
            corr = np.correlate(h, 1 - h, mode="full")
            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)               # shift attaining max
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:
                i_inds = np.arange(-s, n)
                p_inds = i_inds + s
            # look for a pair that can improve the worst contribution
            contrib = h[i_inds] * (1.0 - h[p_inds])
            k = int(np.argmax(contrib))
            i_best, p_best = i_inds[k], p_inds[k]
            eps_max = min(h[i_best], 1.0 - h[p_best])
            if eps_max <= 1e-12:
                # fallback: try a mismatched‑pair swap (increase overlap)
                mism_a = np.where((h[i_inds] > 0.5) & (h[p_inds] < 0.5))[0]
                mism_b = np.where((h[i_inds] < 0.5) & (h[p_inds] > 0.5))[0]
                if mism_a.size and mism_b.size:
                    ia = i_inds[random.choice(mism_a.tolist())]
                    ib = i_inds[random.choice(mism_b.tolist())]
                    nh = h.copy()
                    nh[ia], nh[ib] = nh[ib], nh[ia]
                else:
                    continue
            else:
                eps = eps_max * 0.5
                nh = h.copy()
                nh[i_best] -= eps
                nh[p_best] += eps

        new_c5 = c5_of(nh)
        # Metropolis acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 <= target_bound:
                    break

        # temperature decay
        if it % 250 == 0:
            T *= decay

    # ---------- final correction ----------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-10:
        # distribute the tiny error to random free coordinates
        for _ in range(1000):
            k = random.randrange(n)
            if err > 0 and best_h[k] > 0:
                delta = min(best_h[k], err)
                best_h[k] -= delta
                err -= delta
            elif err < 0 and best_h[k] < 1:
                delta = min(1 - best_h[k], -err)
                best_h[k] += delta
                err += delta
            if abs(err) < 1e-12:
                break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
