def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # compute softmax weights
    exp_term = np.exp(alpha * (corr - max_corr))
    w = exp_term / exp_term.sum()
    # accumulate gradient
    g = np.zeros_like(h)
    # iterate through shift indexes
    for i, wi in enumerate(w):
        if wi == 0: continue  # optional optimization
        s = i - (n - 1)
        # compute gradient for shift s (same as earlier)
        # using the _gradient_for_shift()
        g += wi * _gradient_for_shift(h, s)
    return g

def local_refine(h, iterations, dx, rng):
   best = h.copy()
   best_val = _compute_c5(best, dx)
   for _ in range(iterations):
       i, j = rng.randint(0, n, size=2)
       if i == j:
            continue

       max_delta = min(best[i], 1 - best[j])

def _smooth_gradient(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # compute stable softmax
    shifted = corr - max_corr
    exp_term = np.exp(alpha * shifted)
    w = exp_term / np.sum(exp_term)
    # accumulate gradient
    g = np.zeros_like(h)
    # loops over all shifts
    for idx, wi in enumerate(w):
        if wi == 0:
            continue
        s = idx - (n - 1)
        # compute gradient for this shift
        # reuse _gradient_for_shift
        g += wi * _gradient_for_shift(h, s)
    return g

import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} using a bisection on the dual."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with prescribed sum (projected)."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Exact C5 value for a discretised h (full correlation, then max)."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation for a given integer shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 800,
                 step0: float = 0.5, decay: float = 0.995) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑correlation objective."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max using log‑sum‑exp (shifted for stability)."""
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float, alpha: float = 30.0,
                    max_iter: int = 1000) -> np.ndarray:
    """SLSQP on a smooth max objective."""
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12,
                            'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


def _smooth_gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the smooth‑max of correlations (soft‑max weighted)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exp_term = np.exp(alpha * shifted)
    w = exp_term / np.sum(exp_term)
    g = np.zeros_like(h)
    # accumulate weighted sub‑gradients
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        g += wi * _gradient_for_shift(h, shift)
    return g


def _smooth_optimize(h0: np.ndarray, target: float,
                     alphas, steps_per_alpha: int, step0: float) -> np.ndarray:
    """Gradient descent on the smooth‑max (incrementally increase alpha)."""
    h = h0.copy()
    for alpha in alphas:
        step = step0
        for _ in range(steps_per_alpha):
            g = _smooth_gradient(h, alpha)
            h = h - step * g
            h = _project_onto_capped_simplex(h, target)
            step *= 0.995   # gentle decay inside each alpha‑stage
    return h
