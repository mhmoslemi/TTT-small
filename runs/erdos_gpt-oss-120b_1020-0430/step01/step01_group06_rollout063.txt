def project_onto_bounds_and_sum(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v - lo, 0.0, 1.0)
    # ensure exact sum maybe slight error adjust:
    err = target - x.sum()
    if abs(err) > 1e-8:
        # adjust by adding err to variable not at bounds
        # find index with 0 < x < 1
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            i = free[0]
            x[i] += err
        else:
            # assign to index with room
            if err > 0:
                i = np.argmax(x < 1 - 1e-12)
                x[i] = min(1.0, x[i] + err)
            else:
                i = np.argmin(x > 1e-12)
                x[i] = max(0.0, x[i] + err)
    return x
