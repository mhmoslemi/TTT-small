def _smooth_gd_optimize(h0, target, max_iter, step0, decay, alpha):
    h = h0.copy()
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # compute softmax weights
        m = np.max(corr)
        exp_weights = np.exp(alpha * (corr - m))
        w = exp_weights / np.sum(exp_weights)
        # compute weighted gradient
        n = h.shape[0]
        g = np.zeros_like(h)
        # shift indices range: from -(n-1) to (n-1)
        # compute once all shift gradients weighted by w
        for shift_idx, shift in enumerate(range(-n+1, n)):
            w_shift = w[shift_idx]
            if w_shift == 0:
                continue
            # compute gradient for this shift
            if shift >= 0:
                s = shift
                # term1: 1 - h[i+s] for i < n-s
                g[: n - s] += w_shift * (1.0 - h[s:])
                g[n - s :] += w_shift * 1.0
                # term2: -h[i-s] for i >= s
                g[s:] += w_shift * (-h[: n - s])
            else:
                s = -shift
                g[s:] += w_shift * (1.0 - h[: n - s])
                g[: s] += w_shift * 1.0
                g[: n - s] += w_shift * (-h[s:])
        # update
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

import time, numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v,target):
    # same as earlier but may improve robustness
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(70):
        lam = (lo+hi)/2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s-target) <= 1e-10:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng,n,target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _compute_c5(h,dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr*dx).max())

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _gd_submomentum(h0, target, max_iter=1500, step0=0.8, decay=0.997, momentum=0.9):
    h = h0.copy()
    step = step0
    v = np.zeros_like(h)
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        v = momentum * v - step * g
        h = h + v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

def _slsqp_optimize(h0, target, alpha=30.0, max_iter=400):
    n = h0.shape[0]
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        # smooth max using log-sum-exp
        shifted = corr - m
        # use alpha scaling
        return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    cons = ({'type':'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0,1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
