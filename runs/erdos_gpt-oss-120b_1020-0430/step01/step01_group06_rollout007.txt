            hi, hj = h[i], h[j]
            max_up = min(1.0 - hi, hj)      # increase i, decrease j
            max_down = min(hi, 1.0 - hj)    # decrease i, increase j
            if max_up + max_down == 0:
                it += 1
                continue
            delta = (random.random() * (max_up + max_down)) - max_down
            if abs(delta) < 1e-14:
                it += 1
                continue
            nh = h.copy()
            nh[i] = hi + delta
            nh[j] = hj - delta
            new_c5 = c5_of(nh)

        elif r < 0.8:
            # ---------- swap two entries ----------
            i, j = random.sample(range(n), 2)
            if h[i] == h[j]:
                it += 1
                continue
            nh = h.copy()
            nh[i], nh[j] = nh[j], nh[i]
            new_c5 = c5_of(nh)

        else:
            # ---------- targeted move ----------
            corr = np.correlate(h, 1 - h, mode="full")
            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)               # shift attaining max
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:
                s_abs = -s
                i_inds = np.arange(s_abs, n)
                p_inds = i_inds + s
            contrib = h[i_inds] * (1.0 - h[p_inds])
            k = int(np.argmax(contrib))
            i_best, p_best = i_inds[k], p_inds[k]
            eps_max = min(h[i_best], 1.0 - h[p_best])
            if eps_max <= 1e-12:
                it += 1
                continue
            # try a half‑step and a full‑step, accept whichever improves
            accepted = False
            for factor_eps in (0.5, 1.0):
                eps = eps_max * factor_eps
                nh = h.copy()
                nh[i_best] -= eps
                nh[p_best] += eps
                new_c5 = c5_of(nh)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        best_it = it
                    accepted = True
                    break
            if not accepted:
                it += 1
                continue

        # ------------------------------------------------------------------
        # Acceptance test for non‑targeted moves
        # ------------------------------------------------------------------
        if r < 0.55 or (0.55 <= r < 0.8):
            if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                h = nh
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    best_it = it

        # ------------------------------------------------------------------
        # Temperature schedule & occasional reheat
        # ------------------------------------------------------------------
        it += 1
        if it % 1000 == 0:
            T *= decay
        if it % 20000 == 0:
            T = max(T0, T * 10)

        # ------------------------------------------------------------------
        # Stagnation handling – small perturbation around best
        # ------------------------------------------------------------------
        if it - best_it > 25000:
            h = best_h.copy()
            # small random nudges
            for _ in range(5):
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                max_up = min(1.0 - hi, hj)
                max_down = min(hi, 1.0 - hj)
                rng = max_up + max_down
                if rng == 0:
                    continue
                delta = (random.random() * rng) - max_down
                h[i] = hi + delta
                h[j] = hj - delta
            cur_c5 = c5_of(h)
            T = T0
            best_it = it

    # ----------------------------------------------------------------------
    # Final projection onto the hyperplane (tiny numerical drift)
    # ----------------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        for _ in range(200):
            k = random.randrange(n)
            if err > 0 and best_h[k] >= err:
                best_h[k] -= err
                err = 0.0
                break
            if err < 0 and best_h[k] <= 1 + err:
                best_h[k] -= err   # err negative => addition
                err = 0.0
                break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
