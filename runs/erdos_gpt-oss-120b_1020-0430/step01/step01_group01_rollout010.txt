import time, numpy as np

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _compute_c5_fft(h, dx, nfft):
    # compute linear correlation via FFT, h and 1-h
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(1.0 - h, nfft)
    corr = np.fft.ifft(H * np.conj(G)).real
    # linear correlation length = 2*n-1, start at 0 index
    # Our convolution yields output length nfft; first 2*n-1 entries correspond to lags 0..2n-2 (i.e., shift from -(n-1) to n-1 somehow). But max is unaffected.
    # We just take the max of the first 2*n-1 entries.
    return float(corr[:2*len(h)-1].max() * dx)

```

    for i in range(popsize):

    c2 = 1.49618

    while time.time() - start < budget_s * 0.94 and iter_count < max_iters:

        r1 = rng.rand(popsize, n)
        r2 = rng.rand(popsize, n)

            if val < pbest_vals[i]:
                pbest_vals[i] = val
                pbest[i] = positions[i].copy()
                if val < best_val:
                    best_val = val
                    best_h = positions[i].copy()

import time
import numpy as np

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {0 <= x <= 1, sum(x) = target} using bisection."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum(target)."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5_fft(h: np.ndarray, dx: float, nfft: int) -> float:
    """Correlation via FFT: max_{k} ∫ h(x)(1-h(x+k)) dx."""
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(1.0 - h, nfft)
    corr = np.fft.ifft(np.conj(H) * G).real
    n = h.shape[0]
    # linear correlation for lags -(n-1)..(n-1)
    corr_full = np.concatenate((corr[-(n - 1):], corr[:n]))
    return float(corr_full.max() * dx)

    """Find a step function h on [0,2] giving a low C5 bound."""
    rng = np.random.RandomState(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Determine discretisation size (use parent if provided)
    # --------------------------------------------------------------
    try:
        base = np.asarray(initial_h_values, dtype=float)
        n = base.size
    except Exception:
        base = None
        n = 200

    target_sum = n / 2.0          # because ∫h = 1 and dx = 2/n
    dx = 2.0 / n
    nfft = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n-1

    # --------------------------------------------------------------

    popsize = 80

    # first particle = parent (if any) otherwise random feasible
    if base is not None:
        positions[0] = _project_onto_capped_simplex(base, target_sum)
    else:

    for i in range(1, popsize):
        positions[i] = _random_feasible(rng, n, target_sum)

    # personal bests
    pbest = positions.copy()
    pbest_vals = np.full(popsize, np.inf)

    # global best
    best_h = None
    best_val = np.inf

    # evaluate initial population
    for i in range(popsize):

        if val < best_val:

            best_h = positions[i].copy()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    w = 0.7298
    c1 = 1.49618
