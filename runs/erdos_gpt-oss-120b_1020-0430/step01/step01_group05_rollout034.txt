def shape_h(n, s, p):
    # s: half-support width (>=0, <=1)
    # p: plateau length (>= 2-2s)
    # a = 2/(2*s + p) <= 1 as ensured by constraint.
    a = 2.0/(2.0*s + p)
    dx = 2.0/n
    # positions of cell centers:
    x = (np.arange(n) + 0.5) * dx
    d = np.abs(x - 1.0)
    # outside support:
    h = np.zeros_like(x)
    # inside support:
    mask_support = d <= s
    # inside plateau region:
    half_p = p/2
    mask_plateau = d <= half_p
    # compute slope region values:
    # For points where half_p < d <= s:
    mask_slope = mask_support & ~mask_plateau
    # linear drop from a at d=half_p to 0 at d=s.
    if s - half_p > 0:
        h[mask_plateau] = a
        h[mask_slope] = a * (1.0 - (d[mask_slope] - half_p) / (s - half_p))
    else:
        # p == 2*s, plateau covers support fully (no slope)
        h[mask_support] = a
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Might not exactly meet integral due to discretization; we can project onto sum target using project_onto_bounds_and_sum.
    return h

```python
import time, random, math
import numpy as np

TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x, target):
    # Simple projection onto [0,1]^n intersect sum=target via bisection for Lagrange multiplier.
    # Assumes target between 0 and len(x)
    low, high = -1.5, 1.5
    for _ in range(60):
        lam = (low + high) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def shape_h(n, s, p):
    # symmetric shape: half-support s (0 < s <= 1), plateau length p (>= max(0,2-2s), <= 2s)
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    d = np.abs(x - 1.0)
    a = 2.0 / (2.0 * s + p)
    # ensure a <= 1 (by construction of p)
    h = np.zeros_like(x)
    half_p = p * 0.5
    mask_support = d <= s
    mask_plateau = d <= half_p
    h[mask_plateau] = a
    mask_slope = mask_support & (~mask_plateau)
    if s - half_p > 0:
        h[mask_slope] = a * (1.0 - (d[mask_slope] - half_p) / (s - half_p))
    # else plateau covers whole support -> already set above
    # Clip to [0,1] for safety
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_initial(init, n):
    old_n = init.shape[0]
    # sample points at centers of intervals for both old and new
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x_new, x_old, init)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # choose resolution
    n = 500   # moderate size balancing quality and speed
    target_sum = n / 2.0

    # initialise best vector
    if 'initial_h_values' in globals():
        try:
            h0 = upsample_initial(np.array(initial_h_values, dtype=float), n)
        except Exception:
            h0 = np.full(n, 0.5)  # fallback
    else:
        # sin^2 shape as simple fallback
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx
        h0 = np.sin(np.pi * x / 2.0) ** 2

    best_h = project_onto_bounds_and_sum(h0, target_sum)
    best_val = compute_c5(best_h)

    # --------------------------------------------------
    # 1) Random structured shapes (triangle with plateau)
    # number of samples scaled by remaining budget
    max_shape_samples = int(5000 * min(1.0, budget_s / 1100.0))
    for _ in range(max_shape_samples):
        s = random.uniform(0.5, 1.0)            # half‑support width
        min_p = max(0.0, 2.0 - 2.0 * s)        # enforce height ≤ 1
        max_p = 2.0 * s
        p = random.uniform(min_p, max_p)
        h_candidate = shape_h(n, s, p)
        h_candidate = project_onto_bounds_and_sum(h_candidate, target_sum)
        val = compute_c5(h_candidate)
        if val < best_val:
            best_val = val
            best_h = h_candidate.copy()
            if best_val <= 0.3808:
                break

    # --------------------------------------------------
    # 2) Simulated annealing refinement on the current best
    # parameters
    T0 = 0.1                         # initial temperature
    min_T = 1e-5
    # rough estimate of max iterations based on time left
    while time.time() - start < budget_s - 0.5:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.5)
        T = max(min_T, T0 * (1.0 - frac))
        # propose a mass transfer between two positions
        i = random.randrange(n)
        j = random.randrange(n-1)
        if j >= i:
            j += 1
        hi = best_h[i]
        hj = best_h[j]
        # allowed delta so that after transfer both stay in [0,1]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        # magnitude scaled by temperature to encourage larger moves early
        scale = 1.0
        delta = random.uniform(delta_min, delta_max) * scale
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        # clip just in case numerical
        cand = np.clip(cand, 0.0, 1.0)
        # sum is preserved analytically, but we re‑project to be safe
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand
            if best_val <= 0.3808:
                break
        else:
            # Metropolis acceptance
            if random.random() < math.exp(-(val - best_val) / (T + 1e-12)):
                best_h = cand   # move without improving best_val
        # occasional random restart from shape family (helps escape plateaus)
        if random.random() < 0.002:
            s = random.uniform(0.5, 1.0)
            min_p = max(0.0, 2.0 - 2.0 * s)
            max_p = 2.0 * s
            p = random.uniform(min_p, max_p)
            rand_h = shape_h(n, s, p)
            rand_h = project_onto_bounds_and_sum(rand_h, target_sum)
            rand_val = compute_c5(rand_h)
            if rand_val < best_val:
                best_val = rand_val
                best_h = rand_h

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))

import time, random, math
import numpy as np
from scipy.optimize import minimize

TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n with sum = target via bisection on the Lagrange multiplier."""
    low, high = -1.5, 1.5
    for _ in range(60):
        lam = (low + high) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def shape_h(n: int, s: float, p: float) -> np.ndarray:
    """
    Symmetric shape with half‑support width s (0 < s ≤ 1) and plateau length p.
    Height a = 2/(2s+p) ≤ 1 by construction (p ≥ 2‑2s).
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    d = np.abs(x - 1.0)                     # distance from centre
    a = 2.0 / (2.0 * s + p)                # peak (or plateau) height
    half_p = 0.5 * p
    h = np.zeros_like(x)
    mask_support = d <= s
    mask_plateau = d <= half_p
    h[mask_plateau] = a
    mask_slope = mask_support & (~mask_plateau)
    if s - half_p > 0:
        h[mask_slope] = a * (1.0 - (d[mask_slope] - half_p) / (s - half_p))
    return np.clip(h, 0.0, 1.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new length n."""
    old_n = init.shape[0]
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x_new, x_old, init)


def smooth_max(arr: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h, dx, alpha=30.0):
    """Smooth objective for SLSQP (approximation of the max)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # --------------------------------------------------
    # discretisation
    n = 500                     # trade‑off between resolution and speed
    dx = 2.0 / n
    target_sum = n / 2.0

    # --------------------------------------------------
    # initialise from parent if possible
    if "initial_h_values" in globals():
        try:
            h0 = upsample_initial(np.array(initial_h_values, dtype=float), n)
        except Exception:
            h0 = np.full(n, 0.5)
    else:
        x = (np.arange(n) + 0.5) * dx
        h0 = np.sin(np.pi * x / 2.0) ** 2

    best_h = project_onto_bounds_and_sum(h0, target_sum)
    best_val = compute_c5(best_h)

    # --------------------------------------------------
    # 1) random structured shapes (triangular with optional plateau)
    max_shape_samples = int(5000 * min(1.0, budget_s / 1100.0))
    for _ in range(max_shape_samples):
        s = random.uniform(0.5, 1.0)
        min_p = max(0.0, 2.0 - 2.0 * s)   # enforce height ≤ 1
        max_p = 2.0 * s
        p = random.uniform(min_p, max_p)
        h_cand = shape_h(n, s, p)
        h_cand = project_onto_bounds_and_sum(h_cand, target_sum)
        val = compute_c5(h_cand)
        if val < best_val:
            best_val = val
            best_h = h_cand.copy()
            if best_val <= 0.3808:
                break

    # --------------------------------------------------
    # 2) simulated‑annealing refinement
    curr_h = best_h.copy()
    curr_val = best_val
    T0 = 0.12
    min_T = 1e-5
    while time.time() - start < budget_s - 0.5:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.5)
        T = max(min_T, T0 * (1.0 - frac))

        # propose a transfer of mass between two points
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1
        hi, hj = curr_h[i], curr_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = random.uniform(delta_min, delta_max)
        cand = curr_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand_val = compute_c5(cand)

        # Metropolis acceptance
        if cand_val < curr_val or random.random() < math.exp(-(cand_val - curr_val) / (T + 1e-12)):
            curr_h, curr_val = cand, cand_val
            if cand_val < best_val:

        # occasional random restart from the shape family
        if random.random() < 0.001:

            min_p = max(0.0, 2.0 - 2.0 * s)

            rand_h = project_onto_bounds_and_sum(rand_h, target_sum)
            rand_val = compute_c5(rand_h)
            if rand_val < curr_val:
                curr_h, curr_val = rand_h, rand_val
            if rand_val < best_val:
                best_val, best_h = rand_val, rand_h.copy()
                if best_val <= 0.3808:
                    break

    # --------------------------------------------------
    # 3) final local optimisation with smooth max (SLSQP)
    bounds = [(0.0, 1.0)] * n

    def sum_eq_constraint(x):
        return np.sum(x) - target_sum

    try:
        res = minimize(
            objective_smooth,
            best_h,
            args=(dx, 30.0),
            method="SLSQP",
            bounds=bounds,
            constraints={"type": "eq", "fun": sum_eq_constraint},
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        h_opt = project_onto_bounds_and_sum(res.x, target_sum)
        val_opt = compute_c5(h_opt)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()
    except Exception:
        pass  # if optimiser fails we keep the current best

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
