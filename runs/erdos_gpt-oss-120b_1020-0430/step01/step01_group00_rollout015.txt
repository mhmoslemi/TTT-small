import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a given h (integral over [0,2] is already ensured)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _binary_round(h: np.ndarray) -> np.ndarray:
    """Return the binary (0/1) vector with exactly half 1's that is closest to h."""
    n = h.size
    idx = np.argsort(h)[::-1]          # descending order
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx[: n // 2]] = 1.0
    return bin_vec


# ------------------------------------------------------------
# SLSQP refinement (non‑linear constraints)
# ------------------------------------------------------------

def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 600) -> tuple:
    """Run a single SLSQP optimisation for the supplied initial h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = minimize(_objective, x0, method="SLSQP", bounds=bounds,
                   constraints=constraints, options=opts)

    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


# ------------------------------------------------------------
# Evolutionary search utilities
# ------------------------------------------------------------

def _random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    return _project_to_feasible(h, n)


def _perturb_h(h: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    pert = rng.normal(0.0, sigma, size=n)
    h2 = h + pert
    return _project_to_feasible(h2, n)

    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # ----------------------------------------------------------------
    # Load possible parent construction
    # ----------------------------------------------------------------
    try:
        parent_h = np.array(initial_h_values, dtype=float)  # may raise NameError
    except NameError:
        parent_h = None

    target_bound = 0.3808

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Grid sizes to explore (increase progressively)
    candidate_ns = []
    if parent_h is not None:
        candidate_ns.append(parent_h.size)
    # add a sequence of powers of two up to ~512
    for m in (128, 256, 512):
        if m not in candidate_ns:
            candidate_ns.append(m)
    # a few intermediate even sizes
    for m in (160, 200, 300, 384):
        if m not in candidate_ns:
            candidate_ns.append(m)
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]
    candidate_ns = sorted(set(candidate_ns))

    # ----------------------------------------------------------------
    # Search loop over grid sizes
    # ----------------------------------------------------------------
    for n in candidate_ns:
        if time_left() < 10.0:
            break

        # ----- build initial population -----
        popsize = 30
        population = []
        # include upsized parent if possible
        if parent_h is not None:
            if parent_h.size != n:
                base = _upscale_h(parent_h, n)
            else:
                base = parent_h.copy()
            base = _project_to_feasible(base, n)
            population.append(base)
        # uniform baseline
        population.append(np.full(n, 0.5))
        # fill the rest randomly
        while len(population) < popsize:
            population.append(_random_feasible_h(n, rng))

        scores = [_c5_from_h(h) for h in population]

        elite_size = 5
        mutation_sigma = 0.15
        generation = 0
        max_generations = 300

        # ----- evolutionary loop -----
        while time_left() > 5.0 and generation < max_generations:
            # sort by score (lower is better)
            order = np.argsort(scores)
            population = [population[i] for i in order]
            scores = [scores[i] for i in order]

            # update global best
            if scores[0] < best_c5:
                best_h, best_c5, best_n = population[0].copy(), scores[0], n
                if best_c5 < target_bound:
                    return best_h.tolist(), float(best_c5), int(best_n)

            # elite selection
            elite = population[:elite_size]

            # produce next generation
            new_pop = elite.copy()
            while len(new_pop) < popsize:
                # crossover between two elite members
                idx1, idx2 = rng.choice(elite_size, size=2, replace=False)
                p1, p2 = elite[idx1], elite[idx2]
                alpha = rng.random()
                child = alpha * p1 + (1.0 - alpha) * p2
                # mutation
                child = _perturb_h(child, mutation_sigma, rng)
                new_pop.append(child)

            population = new_pop
            scores = [_c5_from_h(h) for h in population]

            # decay mutation size slowly
            mutation_sigma *= 0.995
            generation += 1

        # ----- optional binary rounding of the current best for this n -----
        if best_h is not None and best_n == n:
            bin_h = _binary_round(best_h)
            bin_c5 = _c5_from_h(bin_h)
            if bin_c5 < best_c5:
                best_h, best_c5 = bin_h, bin_c5
                if best_c5 < target_bound:
                    return best_h.tolist(), float(best_c5), int(best_n)

        # ----- local SLSQP refinement of the best candidate for this n -----
        if best_h is not None and best_n == n:
            refined_h, refined_c5 = _solve_h(best_h, max_iter=800)
            if refined_c5 < best_c5:
                best_h, best_c5 = refined_h, refined_c5
                if best_c5 < target_bound:
                    return best_h.tolist(), float(best_c5), int(best_n)

            # also try to refine a few random perturbations of the best
            for _ in range(4):
                if time_left() < 5.0:
                    break
                pert = _perturb_h(best_h, 0.07, rng)
                refined_h2, refined_c5_2 = _solve_h(pert, max_iter=400)
                if refined_c5_2 < best_c5:
                    best_h, best_c5 = refined_h2, refined_c5_2
                    if best_c5 < target_bound:

        # early termination if we already beat the target
        if best_c5 < target_bound:

    # Fallback (should not happen)

    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
