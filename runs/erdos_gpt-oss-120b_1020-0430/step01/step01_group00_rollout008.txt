def _project_to_feasible(h, target_sum):

   # clip
   h = np.clip(h, 0.0, 1.0)
   # adjust sum via water-filling
   s = h.sum()
   diff = s - target_sum
   if abs(diff) < 1e-12:
        return h
   # while loop
   for _ in range(1000):
        if diff > 0:
            # reduce from elements > 0
            mask = h > 0
            n_free = mask.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            # subtract delta from those elements, ensure not negative
            h[mask] -= delta
            h = np.clip(h, 0.0, 1.0)
        else:
            # diff < 0: need to increase total
            mask = h < 1
            n_free = mask.sum()
            if n_free == 0:
                break
            delta = diff / n_free  # diff negative, delta negative
            h[mask] -= delta  # subtracting negative -> add
            h = np.clip(h, 0.0, 1.0)
        new_s = h.sum()
        new_diff = new_s - target_sum
        if abs(new_diff) < 1e-12 or abs(new_diff - diff) < 1e-15:
            break
        diff = new_diff
   # final small correction if still off: distribute among first few
   residual = target_sum - h.sum()
   if abs(residual) > 1e-9:
        for i in range(h.size):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
   return h

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    # Clip values
    h = np.clip(h, 0.0, 1.0, out=h)
    target = n * 0.5
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = h > 0.0
        else:
            mask = h < 1.0
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[mask] -= delta
        np.clip(h, 0.0, 1.0, out=h)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1.0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0.0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h

            else:
                l = -lag
                grad[l:] += 1.0 - h[: n - l]
                grad[: n - l] -= h[l:]
        grad /= max_idxs.size

        step = step0 / (1.0 + 0.001 * it)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            h_cand = _project_to_feasible(h - step * factor * grad, n)
            cand_val = _c5_from_h(h_cand)
            if cand_val < cur_val - 1e-12:
                h = h_cand
                improved = True
                break
        if not improved:
            step0 *= 0.7
            if it % 250 == 0:
                h = _project_to_feasible(h + 0.015 * np.random.randn(n), n)

        if it % 400 == 0:
            h = _local_swap_refine(h, n, cur_val)

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with C5 as small as possible."""
    np.random.seed(seed)
    start = time.time()

    # optional parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
        init_n = init_h.size
    except NameError:
        init_h = None
        init_n = None

    target = 0.3808
    best_h, best_val, best_n = None, np.inf, None

    # candidate grid sizes (≤500 points)
    cand = []
    if init_n:
        cand.append(init_n)
    for mul in (1, 2, 3, 4):
        n = (init_n if init_n else 64) * mul
        if n <= 500:
            cand.append(int(n))
    for n in (150, 200, 256, 300, 384, 448, 500):
        if n <= 500:
            cand.append(n)
    cand = sorted(set(cand))

    # keep best solution to upscale later
    reference = init_h.copy() if init_h is not None else None

    for n in cand:
        if time.time() - start > budget_s - 6.0:
            break

        # build a base h for this resolution
        if reference is not None and reference.size == n:
            base = reference.copy()
        elif reference is not None:
            base = _upscale(reference, n)
        else:
            base = np.full(n, 0.5)

        # decide number of restarts and iterations
        remaining = budget_s - (time.time() - start)
        restarts = 5 if n <= 200 else 3
        iters = max(2000, int(min(12000, remaining * 15 / n)))

        for r in range(restarts):
            # perturbed start
            h0 = base + 0.05 * np.random.randn(n)
            h0 = _project_to_feasible(h0, n)
            h_opt, val_opt = _gradient_optimize(h0, n, iters, start, budget_s)
            if val_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), val_opt, n
                reference = best_h.copy()
                if best_val <= target:
                    return best_h.tolist(), float(best_val), int(best_n)

            # first restart also tries a binary random start (exact half ones)
            if r == 0:
                perm = np.random.permutation(n)
                h_bin = np.zeros(n)
                h_bin[perm[: n // 2]] = 1.0
                h_bin = _project_to_feasible(h_bin, n)
                h_opt, val_opt = _gradient_optimize(h_bin, n, iters, start, budget_s)
                if val_opt < best_val:
                    best_h, best_val, best_n = h_opt.copy(), val_opt, n
                    reference = best_h.copy()
                    if best_val <= target:
                        return best_h.tolist(), float(best_val), int(best_n)

    # fallback if nothing better was found
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_val = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_val), int(best_n)
