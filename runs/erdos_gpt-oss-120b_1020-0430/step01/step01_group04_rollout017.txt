def project_onto_capped_simplex(v, target):
    n = v.shape[0]
    # if sum after clipping to [0,1] is already target, return clipped.
    # compute low and high bound for lambda
    # low ensures all v+lam <= 0: lam_low = - (v.max() + 1)
    # high ensures all v+lam >= 1: lam_high = 1 - v.min() + 1 (or 2 - v.min())
    # Actually we need bounds that guarantee sum <= target and >= target.
    # Use low = -np.max(v) - 1 (makes all v+low <= -1), after clipping yields zeros => sum=0.
    # Use high = 1 - np.min(v) + 1 (makes v+high >=2, after clipping yields 1) => sum=n.
    # Then binary search for lambda.

    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # To improve accuracy: adjust by small correction to meet sum exactly (if off by >1e-9) by adding diff/num_free where free indices 0<x<1.
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0) & (x < 1)
        if free.any():
            # distribute diff uniformly across free indices
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:

    n = h.shape[0]
    # phases = [(5, 400, 0.2), (30, 800, 0.12), (80, 1200, 0.07)] maybe.

        for i in range(iters):
            if time.time() > time_deadline: break outer loops via raising or returning early
            # compute gradient & update
            O = np.correlate(h, 1 - h, mode='full')

            # subtract max for stability
            max_alpha_O = np.max(alpha * O)

            # gradient descent step
            h_new = h - lr * grad
            h = project_onto_capped_simplex(h_new, target)
            # optionally adapt step size if gradient norm small etc.
    return h

def project_onto_capped_simplex(v, target):
    n = v.shape[0]
    # quick check: clip then sum:
    v_clipped = np.clip(v, 0.0, 1.0)
    s = v_clipped.sum()
    if abs(s - target) <= 1e-12:
        return v_clipped

    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # adjust final sum if needed
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff evenly among non-boundary entries
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: distribute across all
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x

def generate_triangular(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0/n)  # midpoints approx
    tri = np.minimum(x, 2.0 - x)
    # Normalize sum to target = n/2
    target = n / 2.0
    # Scale
    tri = tri * (target / tri.sum())
    # Clip to [0,1]
    tri = np.clip(tri, 0.0, 1.0)
    # Ensure sum exact after clipping by adjusting uniformly
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + (1.0/old_n)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + (1.0/new_n)
    up = np.interp(new_x, old_x, arr)
    # renorm
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {x | 0≤x_i≤1, sum x_i = target}."""
    n = v.shape[0]
    # quick clipping
    x = np.clip(v, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) <= 1e-12:
        return x

    lo = -np.max(v) - 1.0          # makes all v+lo ≤ 0  → sum 0
    hi = 1.0 - np.min(v) + 1.0     # makes all v+hi ≥ 1  → sum n
    for _ in range(60):
        mid = (lo + hi) * 0.5
        xmid = np.clip(v + mid, 0.0, 1.0)
        smid = xmid.sum()
        if smid > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # final tiny correction
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the capped simplex."""
    v = rng.random(n)                     # in [0,1)
    target = n / 2.0
    return project_onto_capped_simplex(v, target)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular profile (min(x,2-x)) sampled on n points, normalized."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new resolution, preserving the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def _smooth_gradient_step(h: np.ndarray, alpha: float, target: float) -> np.ndarray:
    """One gradient descent step for the soft‑max approximation."""
    n = h.shape[0]
    # correlation h with (1‑h)
    O = np.correlate(h, 1.0 - h, mode="full")
    aO = alpha * O
    max_aO = np.max(aO)
    w = np.exp(aO - max_aO)
    w /= w.sum()                     # softmax weights, sum to 1

    # convolution w * h  and (reverse w) * h
    conv1 = np.convolve(w, h, mode="full")
    conv2 = np.convolve(w[::-1], h, mode="full")
    offset = len(w) - 1               # = 2*n‑2
    term1 = conv1[offset: offset + n]
    term2 = conv2[offset: offset + n]
    grad = 1.0 - (term1 + term2)      # ∂f/∂h
    return grad


def local_optimize(start_h: np.ndarray,
                   target: float,
                   time_deadline: float) -> np.ndarray:
    """Projected gradient descent on a sequence of smooth approximations."""
    h = start_h.copy()
    n = h.shape[0]

    # (alpha, max_iters, learning_rate) triples
    phases = [(5.0, 500, 0.30),
              (30.0, 800, 0.15),
              (80.0, 1200, 0.07)]

    for alpha, max_iters, lr in phases:
        for _ in range(max_iters):
            if time.time() > time_deadline:
                return h
            grad = _smooth_gradient_step(h, alpha, target)
            h = project_onto_capped_simplex(h - lr * grad, target)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidate_ns = []
    if parent_n is not None:
        candidate_ns.append(parent_n)
    candidate_ns.extend([120, 160, 200, 240])
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    # -----------------------------------------------------------------
    # Loop over grid sizes
    # -----------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > time_limit:
            break
        target = n / 2.0

        # -----------------------------------------------------------------
        # Assemble a modest collection of diverse starting points
        # -----------------------------------------------------------------
        starts = []

        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # binary block (first half ones, second half zeros)
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)], dtype=float)
        starts.append(block)

        # classic triangular shape
        starts.append(generate_triangular(n))

        # a few random feasible vectors
        for _ in range(2):
            starts.append(random_feasible(n, rng))

        # Optimise each starting point
        # -----------------------------------------------------------------
        for s in starts:

            h_opt = local_optimize(s, target, time_limit)
            if h_opt is None:

            c5 = compute_c5(h_opt)
            if c5 < best_c5 - 1e-9:

                best_c5 = c5

    target = best_n / 2.0
    best_h = project_onto_capped_simplex(best_h, target)
    best_c5 = compute_c5(best_h)
