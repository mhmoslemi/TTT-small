def compute_c5(h, dx):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    h_fft = np.fft.rfft(h, n=L)
    g = 1.0 - h
    g_rev = g[::-1]
    g_fft = np.fft.rfft(g_rev, n=L)
    corr = np.fft.irfft(h_fft * g_fft, n=L)[:2*n-1] * dx
    max_corr = corr.max()
    return float(max_corr)

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- helper utilities -------------------------------------------

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} by bisection on the dual variable."""
    lo, hi = -np.max(v) - 1.0, np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _corr_full_fft(h: np.ndarray) -> np.ndarray:
    """Full (zero‑padded) correlation of h with 1‑h using FFT."""
    n = h.shape[0]
    m = 2 * n - 1
    L = 1 << ((m - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)[:m]
    return conv


def _c5_of(h: np.ndarray, dx: float) -> float:
    """Integral max‑overlap value for a step function."""
    return float(_corr_full_fft(h).max() * dx)


# ---------- main optimisation routine -----------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # ---- grid size ----------------------------------------------------
    try:

    except Exception:
        base = None
    n = max(200, len(base) if base is not None else 0)
    if n % 2:                     # ensure half‑integer sum is exact
        n += 1
    dx = 2.0 / n
    target_sum = n / 2.0

    # ---- initialise a binary candidate --------------------------------
    if base is not None:
        # upscale if needed and take the largest half as ones
        if len(base) != n:
            x_old = np.linspace(0.0, 2.0, len(base), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            base = np.interp(x_new, x_old, base)
        idx = np.argsort(base)[-int(target_sum):]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=int(target_sum), replace=False)
        h[ones] = 1.0

    best_h = h.copy()
    best_val = _c5_of(best_h, dx)
    cur_h = h.copy()
    cur_val = best_val

    # ---- simulated annealing on the binary space ----------------------
    # temperature schedule (linear decay)
    T0 = 0.02
    sa_deadline = start + budget_s * 0.70   # 70 % of the budget
    n_range = np.arange(n)

    while time.time() < sa_deadline:
        # correlation to locate the current worst shift
        corr = _corr_full_fft(cur_h)
        max_idx = int(corr.argmax())
        shift = max_idx - (n - 1)

        # find positions that are harmful/good for this shift
        if shift >= 0:
            rng_idx = np.arange(0, n - shift)
            partner = rng_idx + shift
        else:
            s = -shift
            rng_idx = np.arange(s, n)
            partner = rng_idx - s

        bad = rng_idx[(cur_h[rng_idx] == 1) & (cur_h[partner] == 0)]
        good = rng_idx[(cur_h[rng_idx] == 0) & (cur_h[partner] == 1)]

        if bad.size and good.size:
            i = rng.choice(bad)
            j = rng.choice(good)
        else:
            # fallback: random opposite-valued swap
            i = rng.randint(n)
            # ensure opposite value
            if cur_h[i] == 1:
                zeroes = n_range[cur_h == 0]
                if zeroes.size == 0:
                    continue
                j = rng.choice(zeroes)
            else:
                ones = n_range[cur_h == 1]
                if ones.size == 0:
                    continue
                j = rng.choice(ones)

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        new_val = _c5_of(cur_h, dx)
        delta = new_val - cur_val
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        accept = False
        if delta <= 0:
            accept = True
        elif T > 0 and rng.rand() < math.exp(-delta / T):
            accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # ---- optional smooth optimisation (continuous refinement) ---------
    # use the best binary vector as a starting point
    opt_deadline = start + budget_s * 0.95   # reserve a little margin
    while time.time() < opt_deadline:
        # smooth max objective (log‑sum‑exp)
        def obj(v):
            corr = np.correlate(v, 1.0 - v, mode='full') * dx
            return _smooth_max(corr, alpha=30.0)

        cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target_sum},)
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-10, 'disp': False})
        if not res.success:
            break
        cand = _project_onto_capped_simplex(np.clip(res.x, 0.0, 1.0), target_sum)
        cand_val = _c5_of(cand, dx)
        if cand_val < best_val:
            best_val, best_h = cand_val, cand
        else:
            # no further improvement – break early
            break

    # final safety projection
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _c5_of(best_h, dx)

    return (best_h, float(best_val), n)
