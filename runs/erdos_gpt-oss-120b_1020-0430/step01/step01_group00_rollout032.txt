                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ------------------------------------------------------------
# SLSQP local refinement (same as the parent implementation)
# ------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """Run a single SLSQP optimisation for the supplied initial h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ------------------------------------------------------------
# Simulated‑annealing optimiser on the original discrete problem
# ------------------------------------------------------------
def _sa_optimize(h_start: np.ndarray, time_limit: float) -> tuple:
    n = h_start.size
    h = h_start.copy()
    cur_c5 = _c5_from_h(h, n)
    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    end = start + time_limit
    T0, Tf = 0.02, 1e-6  # temperature schedule parameters

    while time.time() < end:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        if frac > 1.0:
            frac = 1.0
        T = T0 * (Tf / T0) ** frac

        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        eps = np.random.uniform(low, high)

        # apply move
        h[i] = hi + eps
        h[j] = hj - eps
        new_c5 = _c5_from_h(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

    return best_h, best_c5


# ------------------------------------------------------------
# Main driver
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdos C5 bound using a combination of
    simulated annealing and local SLSQP refinement.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # ----------------------------------------------------------------
    # Obtain initial construction if supplied
    # ----------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        init_n = init_h.size
    except NameError:
        init_h = None
        init_n = None

    # ----------------------------------------------------------------
    # Candidate grid sizes (prefer the supplied size, then a few larger)
    # ----------------------------------------------------------------
    candidates = []
    if init_n:
        candidates.append(init_n)
    for mul in (1.5, 2, 2.5, 3):
        if init_n:
            n = int(round(init_n * mul))
        else:
            n = int(round(128 * mul))
        if n % 2:
            n += 1
        if n <= 300:
            candidates.append(n)
    for n in (150, 200, 256):
        if n <= 300:
            candidates.append(n)
    candidates = sorted(set(candidates))

    best_h, best_c5, best_n = None, float("inf"), None
    target = 0.3808

    # ----------------------------------------------------------------
    # Optimisation loop over grid sizes
    # ----------------------------------------------------------------
    for n in candidates:
        if time_left() < 6.0:
            break

        # Build start vector for this grid size
        if init_h is not None and init_n != n:
            base_h = _upscale_h(init_h, n)
        elif init_h is not None:
            base_h = init_h.copy()
        else:
            base_h = np.full(n, 0.5)
        base_h = _project_to_feasible(base_h, n)

        # Allocate time (half of remaining) to SA on this size
        alloc = min(0.45 * time_left(), 30.0)
        h_sa, c5_sa = _sa_optimize(base_h, alloc)

        # Local refinement from the SA result
        h_ref, c5_ref = _solve_h(h_sa, max_iter=300)

        # Record best
        if c5_ref < best_c5:
            best_h, best_c5, best_n = h_ref, c5_ref, n
            if best_c5 < target:
                break

        # Occasionally try a fresh random start (if time allows)
        if time_left() > 5.0 and np.random.rand() < 0.2:
            rnd = np.random.rand(n)
            rnd = _project_to_feasible(rnd, n)
            # short SA run
            h_tmp, c5_tmp = _sa_optimize(rnd, min(5.0, 0.1 * time_left()))
            h_tmp_ref, c5_tmp_ref = _solve_h(h_tmp, max_iter=150)
            if c5_tmp_ref < best_c5:
                best_h, best_c5, best_n = h_tmp_ref, c5_tmp_ref, n
                if best_c5 < target:
                    break

    # ----------------------------------------------------------------
    # Fallback (should never happen)
    # ----------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h, n)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
