import time, numpy as np

def _evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Use simple redistribution loop:
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        indices = np.where(mask)[0]
        per = diff / len(indices)
        for i in indices:
            if diff > 0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n and target
    init_vec = globals().get('initial_h_values')
    if init_vec is not None:
        n = len(init_vec)
    else:
        n = 45
    target = n / 2.0

    # Helper to adjust sum
    def adjust(vec):
        return _adjust_sum(np.clip(vec, 0.0, 1.0), target)

    # Evaluate function
    def evaluate(vec):
        return _evaluate_c5(vec)

    # Generate initial population
    popsize = 30
    pop = []

    # include initial if available
    if init_vec is not None:
        vec0 = adjust(np.asarray(init_vec, dtype=np.float64))
        pop.append(vec0)

    # cosine shape
    cos_vec = adjust(0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n))
    pop.append(cos_vec)

    # binary vector
    def binary_vec():
        w = int(np.floor(target))
        idx = rng.choice(n, w, replace=False)
        vec = np.zeros(n)
        vec[idx] = 1.0
        leftover = target - w
        if leftover > 0:
            avail = np.where(vec == 0)[0]
            extra = rng.choice(avail)
            vec[extra] = leftover
        return adjust(vec)
    pop.append(binary_vec())

    # add some random vectors to fill popsize
    while len(pop) < popsize:

    # GA parameters

    while time.time() - start < budget_s - 1.0:

        cur_min_idx = np.argmin(c5_vals)

import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _adjust_sum(x, target):
    """Project x onto {0≤x≤1, sum(x)=target}."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple redistribution loop
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Genetic + hill‑climbing search for a low C₅ step function.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        n = len(init_vec)
    else:
        n = 45                     # fallback resolution
    target = n / 2.0               # required sum of h

    def _proj(v):
        return _adjust_sum(np.clip(v, 0.0, 1.0), target)

    # ------------------------------------------------------------------
    # helpers to construct starting vectors
    # ------------------------------------------------------------------
    def _random_vec():
        return _proj(rng.random(n))

    def _binary_vec():
        w = int(np.floor(target))
        idx = rng.choice(n, w, replace=False)
        v = np.zeros(n)
        v[idx] = 1.0
        leftover = target - w
        if leftover > 1e-12:
            free = np.where(v == 0)[0]
            v[rng.choice(free)] = leftover
        return _proj(v)

    def _cosine_vec():
        v = 0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n) / n)
        return _proj(v)

    def _tent_vec():
        dx = 2.0 / n
        x = np.arange(n) * dx
        v = np.minimum(x, 2.0 - x)
        return _proj(v)

    # ------------------------------------------------------------------
    # initialise population
    # ------------------------------------------------------------------
    popsize = 30
    population = []

    if init_vec is not None:
        population.append(_proj(np.asarray(init_vec, dtype=np.float64)))
    population.append(_cosine_vec())
    population.append(_binary_vec())
    population.append(_tent_vec())
    while len(population) < popsize:
        population.append(_random_vec())

    # ------------------------------------------------------------------
    # evaluate fitness (lower is better)
    # ------------------------------------------------------------------
    def _c5_of(pop):
        return np.array([_evaluate_c5(v) for v in pop])

    fitness = _c5_of(population)
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    elite_frac = 0.2
    elite_num = max(1, int(popsize * elite_frac))
    mutate_prob = 0.9
    parent_pool = max(2, int(popsize * 0.5))

    # ------------------------------------------------------------------
    # main evolutionary loop
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 1.0:
        fitness = _c5_of(population)
        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_c5 - 1e-12:
            best_c5 = float(fitness[cur_best])
            best_h = population[cur_best].copy()

        order = np.argsort(fitness)
        elites = [population[i].copy() for i in order[:elite_num]]
        new_pop = elites

        while len(new_pop) < popsize:

            child = (population[p1] + population[p2]) * 0.5
            child = _proj(child)                     # keep feasibility

                if rng.random() < 0.5:

                    src, dst = j, i

                    child[src] -= delta
                    child[dst] += delta
            new_pop.append(child)

    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5

    while time.time() < deadline:

        cand[dst] += delta

                best_h, best_c5 = cand.copy(), cand_c5

    # final projection and return
    # ------------------------------------------------------------------
    best_h = _proj(best_h)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
