import time
import random
import numpy as np
from scipy.optimize import minimize

TARGET_SUM_GLOBAL = None  # will be set inside run()


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while enforcing sum(x)=target via bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a (possibly fractional) step function, using FFT for speed."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, L)
    B = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(B), L)[: 2 * n - 1]
    return np.max(corr) * dx


def compute_min_overlap(h: np.ndarray) -> float:
    """Return the minimal linear auto‑correlation value of a binary vector h."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    corr = np.fft.irfft(H * np.conj(H), L)[: 2 * n - 1]
    # rounding removes tiny numerical noise (binary case)
    corr = np.rint(corr)
    return corr.min()


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Smooth approximation of max via log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def objective_smooth(x: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth objective for SLSQP: smoothed max correlation of h with (1‑h)."""
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(x, L)
    B = np.fft.rfft(1.0 - x, L)
    corr = np.fft.irfft(H * np.conj(B), L)[: 2 * n - 1]
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape, peak at centre, linearly to 0 at the ends."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin²(πx/2) on [0,2]; already has integral 1."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(arr: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n, then re‑project."""
    old_n = arr.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, arr)
    return project_onto_bounds_and_sum(h, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low C5 value."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # problem size (larger n gives more freedom – keep even)
    n = 500
    if n % 2:
        n += 1
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # collect seed vectors (parent if available, plus analytic starts)
    seed_vecs = []
    if "initial_h_values" in globals():
        try:
            seed_vecs.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seed_vecs.append(make_triangle(n))
    seed_vecs.append(make_sin2(n))

    # ------------------------------------------------------------------
    best_h = None
    best_val = np.inf

    def update_best(candidate):
        nonlocal best_h, best_val
        val = compute_c5(candidate)
        if val < best_val - 1e-12:
            best_val = val
            best_h = candidate.copy()
        return val

    # evaluate the initial seeds
    for vec in seed_vecs:
        update_best(vec)

    # ------------------------------------------------------------------
    # main optimisation loop – random starts + staged SLSQP + local tweaks
    max_restarts = 30
    for restart in range(max_restarts):
        if time.time() - start > budget_s * 0.90:
            break

        # random feasible start (continuous)
        rnd = np.random.rand(n)
        h_cur = project_onto_bounds_and_sum(rnd, target_sum)

        # staged SLSQP with increasing smoothness
        for alpha, maxit in ((5.0, 200), (20.0, 200), (50.0, 200)):
            try:
                res = minimize(
                    objective_smooth,
                    h_cur,
                    args=(dx, alpha),
                    method="SLSQP",
                    bounds=bounds,
                    constraints={"type": "eq", "fun": sum_eq_constraint},
                    options={"maxiter": maxit, "ftol": 1e-9, "disp": False},
                )
                h_cur = project_onto_bounds_and_sum(res.x, target_sum)
                update_best(h_cur)
            except Exception:
                break

        # modest random pairwise tweaks (continuous)
        for _ in range(5000):
            if time.time() - start > budget_s * 0.90:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            hi, hj = h_cur[i], h_cur[j]
            delta_max = min(1.0 - hi, hj)
            delta_min = -min(hi, 1.0 - hj)
            if delta_max < delta_min:
                continue
            delta = np.random.uniform(delta_min, delta_max)
            if abs(delta) < 1e-12:
                continue
            cand = h_cur.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_bounds_and_sum(cand, target_sum)
            val = update_best(cand)
            # keep improvement for the next iteration
            if val < compute_c5(h_cur):
                h_cur = cand
            if best_val <= 0.3808:
                break

        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # binary hill‑climbing refinement (may raise min‑overlap)
    if best_h is not None:
        binary = (best_h > 0.5).astype(float)
        # enforce exact half‑mass
        cur_ones = int(binary.sum())
        need = n // 2 - cur_ones
        if need > 0:
            idx_zero = np.flatnonzero(1 - binary)
            to_one = np.random.choice(idx_zero, need, replace=False)
            binary[to_one] = 1.0
        elif need < 0:
            idx_one = np.flatnonzero(binary)
            to_zero = np.random.choice(idx_one, -need, replace=False)
            binary[to_zero] = 0.0
        binary = project_onto_bounds_and_sum(binary, n / 2.0)

        min_ov = compute_min_overlap(binary)
        val_bin = (n / 2.0 - min_ov) * dx
        if val_bin < best_val:
            best_val = val_bin
            best_h = binary.copy()

        ones_idx = np.flatnonzero(binary)
        zeros_idx = np.flatnonzero(1 - binary)

        for _ in range(20000):
            if time.time() - start > budget_s * 0.97:
                break
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            binary[i] = 0.0
            binary[j] = 1.0
            new_min_ov = compute_min_overlap(binary)
            new_val = (n / 2.0 - new_min_ov) * dx
            if new_val < best_val:
                best_val = new_val
                best_h = binary.copy()
            if new_val < val_bin:
                val_bin = new_val
                ones_idx = np.flatnonzero(binary)
                zeros_idx = np.flatnonzero(1 - binary)
            else:
                # revert the swap
                binary[i] = 1.0
                binary[j] = 0.0
            if best_val <= 0.3808:
                break

    # ------------------------------------------------------------------
    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, n / 2.0)
    final_val = compute_c5(best_h)

    return (best_h.tolist(), float(final_val), int(n))
