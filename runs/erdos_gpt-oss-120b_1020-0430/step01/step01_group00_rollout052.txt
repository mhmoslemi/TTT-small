def gradient_descent(y, target_sum, n, dx, max_iter, alpha0):

    y_cur = y.copy()

    best_y = y_cur.copy()
    best_c5 = compute_c5_from_y(y_cur)

    alpha = alpha0

    for it in range(max_iter):
        # compute h
        h = 1.0 / (1.0 + np.exp(-y_cur))

        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n - 1)

        # compute shifted arrays:
        if k >= 0:
            h_forward = np.empty_like(h)
            h_forward[:n - k] = h[k:]
            h_forward[n - k:] = 0.0
            h_backward = np.empty_like(h)
            h_backward[k:] = h[:n - k]
            h_backward[:k] = 0.0
        else:
            s = -k
            h_forward = np.empty_like(h)
            h_forward[s:] = h[:n - s]
            h_forward[:s] = 0.0

def _adjust_sum(y, target, tol=1e-9, max_iter=50):

    low = -10.0
    high = 10.0
    # Expand bounds if needed:
    while np.sum(1.0/(1.0+np.exp(-(y+low)))) > target:
        low *= 2
    while np.sum(1.0/(1.0+np.exp(-(y+high)))) < target:
        high *= 2
    for _ in range(max_iter):

def _adjust_sum(y, target, tol=1e-9, max_iter=30):
    delta = 0.0
    for _ in range(max_iter):
        sig = 1.0/(1.0+np.exp(-(y+delta)))
        s = sig.sum()
        diff = s - target
        if abs(diff) < tol:
            break
        der = (sig * (1 - sig)).sum()
        if der == 0:
            break
        delta -= diff / der
    return y + delta

def _solve_h(initial_h, max_iter=200):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full')) * (2.0 / n)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr * (2.0 / n)
    def eq(x):
        return np.sum(x[:n]) - n / 2.0
    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter':max_iter,'ftol':1e-9,'disp':False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5

import time
import numpy as np
from scipy import optimize

# ---------- helper utilities ----------
def _sigmoid(y: np.ndarray) -> np.ndarray:
    return 1.0 / (1.0 + np.exp(-y))

def _logit(p: np.ndarray) -> np.ndarray:
    eps = 1e-12
    p = np.clip(p, eps, 1.0 - eps)
    return np.log(p / (1.0 - p))

def _adjust_sum(y: np.ndarray, target: float, tol: float = 1e-10, max_iter: int = 30) -> np.ndarray:
    """Find delta so that sum(sigmoid(y+delta)) == target (monotone, use Newton)."""
    delta = 0.0
    for _ in range(max_iter):
        s = _sigmoid(y + delta)
        cur = s.sum()
        diff = cur - target
        if abs(diff) <= tol:
            break
        der = (s * (1.0 - s)).sum()
        if der == 0:
            break
        delta -= diff / der
    return y + delta

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where((diff > 0) & (h > 0) | (diff < 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)

def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# ---------- gradient‑descent optimiser ----------
def _gd_optimize(y0: np.ndarray, target: float, n: int, max_iter: int,
                 alpha0: float, time_left) -> tuple:
    y = y0.copy()
    dx = 2.0 / n
    best_y = y.copy()
    best_c5 = _c5_from_h(_sigmoid(y))
    alpha = alpha0
    for it in range(max_iter):
        if time_left() < 0.05:
            break
        h = _sigmoid(y)

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)

        # shifted copies with zero padding
        if k >= 0:
            forward = np.empty_like(h)
            forward[:n - k] = h[k:]
            forward[n - k:] = 0.0
            backward = np.empty_like(h)
            backward[k:] = h[:n - k]
            backward[:k] = 0.0
        else:
            s = -k
            forward = np.empty_like(h)
            forward[s:] = h[:n - s]
            forward[:s] = 0.0
            backward = np.empty_like(h)
            backward[:n - s] = h[s:]
            backward[n - s:] = 0.0

        g = 1.0 - forward - backward
        gy = g * h * (1.0 - h)

        y_trial = y - alpha * gy
        y_trial = _adjust_sum(y_trial, target)
        h_trial = _sigmoid(y_trial)
        c5_trial = np.max(np.correlate(h_trial, 1.0 - h_trial, mode="full")) * dx
        c5_cur = np.max(corr) * dx

        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_y = y_trial.copy()

        if c5_trial <= c5_cur:
            y = y_trial
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                alpha = alpha0
    return best_y, best_c5

# ---------- local SLSQP refinement ----------
def _solve_h(initial_h: np.ndarray, max_iter: int = 200) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full")) * (2.0 / n)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [{"type": "ineq", "fun": ineq},
            {"type": "eq", "fun": eq}]
    res = optimize.minimize(lambda x: x[-1],
                            x0,
                            method="SLSQP",
                            bounds=bounds,
                            constraints=cons,
                            options={"maxiter": max_iter,
                                     "ftol": 1e-9,
                                     "disp": False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

# ---------- simple hill‑climb ----------
def _hill_climb(h: np.ndarray, cur_c5: float,
                time_left, max_trials: int = 4000, eps: float = 0.02) -> tuple:
    n = h.size
    target = n / 2.0
    best = h.copy()
    best_val = cur_c5
    for _ in range(max_trials):
        if time_left() < 0.1:
            break
        i, j = np.random.choice(n, 2, replace=False)
        max_inc = min(eps, 1.0 - best[i])
        max_dec = min(eps, best[j])
        delta = np.random.rand() * min(max_inc, max_dec)
        if delta == 0.0:
            continue
        h_new = best.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = _c5_from_h(h_new)
        if c5_new < best_val:
            best = h_new
            best_val = c5_new
    return best, best_val

# ---------- main routine ----------
