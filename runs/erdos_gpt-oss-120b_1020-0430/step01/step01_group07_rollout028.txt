def _solve_optimal_delta(low, high, corr, dvec, tol=1e-12):
    # convex piecewise linear min of max.
    # We'll binary search based on sign of subgradient.
    # Ensure low<high.
    while high - low > tol:
        mid = (low + high) * 0.5
        vals = corr + mid * dvec
        t = vals.max()
        mask = np.isclose(vals, t, atol=1e-12)
        max_slope = dvec[mask].max()
        min_slope = dvec[mask].min()
        # If zero is between min_slope and max_slope -> optimal
        if min_slope <= 0 <= max_slope:
            return mid
        # If all slopes >0 => decreasing Δ reduces max
        if min_slope > 0:
            high = mid
        else:  # all slopes <0
            low = mid
    return (low + high) * 0.5

def improve_h(h, time_limit):
    n = h.shape[0]
    pad_len = n-1
    dx = 2.0/n
    # initial
    s = 1.0 - h
    ext_h = np.pad(h, (pad_len, pad_len), mode='constant', constant_values=0)
    ext_s = np.pad(s, (pad_len, pad_len), mode='constant', constant_values=1)
    corr = np.correlate(h, s, mode='full')  # length 2n-1
    best_h = h.copy()
    best_val = corr.max()
    start = time.time()
    offsets = np.arange(-n+1, n, dtype=int)

    # Rerun loops
    while time.time() - start < time_limit:
        # compute current max
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx]
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
        shift = max_idx - (n - 1)

        # Compute gradient
        idxs = np.arange(n) + pad_len
        grad = ext_s[idxs + shift] - ext_h[idxs - shift]

        # find inc/dec indices
        inc_candidates = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
        dec_candidates = np.where((grad > 1e-12) & (h > 1e-12))[0]

        if inc_candidates.size == 0 or dec_candidates.size == 0:
            # fallback random pair
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i: j+=1
            # ensure feasible movement (swap?) We'll just try binary random delta within bounds.
            i_inc, i_dec = i, j
        else:
            i_inc = inc_candidates[np.argmin(grad[inc_candidates])]
            i_dec = dec_candidates[np.argmax(grad[dec_candidates])]

        # bound
        bound = min(1.0 - h[i_inc], h[i_dec])
        if bound <= 1e-12:
            # try another pair
            # maybe random attempt
            continue

        # compute dvec
        # using offsets vector
        dvec = ext_s[i_inc + pad_len + offsets] - ext_h[i_inc + pad_len - offsets] - \
               ext_s[i_dec + pad_len + offsets] + ext_h[i_dec + pad_len - offsets]

        delta_opt = _solve_optimal_delta(0.0, bound, corr, dvec)

        # Evaluate at delta_opt, and maybe at bounds to pick best

        new_corr_opt = corr + delta_opt * dvec
        new_max_opt = new_corr_opt.max()

        new_corr_low = corr  # delta 0
        new_max_low = corr.max()
        new_corr_high = corr + bound * dvec
        new_max_high = new_corr_high.max()

        # choose best
        if new_max_opt < new_max_low - 1e-12 and new_max_opt < new_max_high - 1e-12:
            delta = delta_opt
            new_corr = new_corr_opt
        elif new_max_low < new_max_high - 1e-12:
            delta = 0.0
            new_corr = corr
        else:
            delta = bound
            new_corr = new_corr_high

            # For now break

        h[i_inc] += delta
        h[i_dec] -= delta

        ext_h[i_inc + pad_len] += delta
        ext_s[i_inc + pad_len] -= delta
        ext_h[i_dec + pad_len] -= delta
        ext_s[i_dec + pad_len] += delta
        # update corr
        corr = new_corr

    final_val = best_val  # raw sum value

    best_h = h.copy()

    start = time.time()
    steps_without_improve = 0

import time, random, math
import numpy as np

def _c5_metric(h):
    """Discrete C5 metric for array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _solve_optimal_delta(lo, hi, corr, dvec, tol=1e-8):
    """Minimise max(corr + delta*dvec) for delta in [lo,hi] (convex, piecewise‑linear)."""
    if hi - lo < tol or np.all(dvec == 0):
        return lo
    while hi - lo > tol:
        mid = (lo + hi) * 0.5
        vals = corr + mid * dvec
        t = vals.max()
        mask = np.isclose(vals, t, atol=1e-12)
        max_slope = dvec[mask].max()
        min_slope = dvec[mask].min()
        if min_slope <= 0 <= max_slope:
            return mid
        if min_slope > 0:          # all slopes positive → move left
            hi = mid
        else:                       # all slopes negative → move right
            lo = mid
    return (lo + hi) * 0.5


def _initial_h(n):
    """Create a feasible start from the global initial_h_values if present."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        m = base.size
        if m != n:
            x0 = np.linspace(0.0, 2.0, m, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x, x0, base)
        else:
            h = base.copy()
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for i in np.random.permutation(n):
            if 0.0 <= h[i] + deficit <= 1.0:
                h[i] += deficit
                break
    return h


def _random_binary_h(n):
    """Binary (0/1) start with exactly n/2 ones (fractional if needed)."""
    h = np.zeros(n, dtype=np.float64)
    half = n // 2
    idx = np.random.permutation(n)
    h[idx[:half]] = 1.0
    if n % 2:
        extra = (n / 2.0) - half
        if extra > 1e-12:
            h[idx[half]] = extra
    return h

    offsets = np.arange(-n + 1, n, dtype=int)

    # enforce exact mass (tiny numerical drift)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        for i in np.random.permutation(n):
            if 0.0 <= best_h[i] + diff <= 1.0:
                best_h[i] += diff
                break

    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
