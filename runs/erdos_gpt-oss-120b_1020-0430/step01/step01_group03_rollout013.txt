def subgradient_descent(h, rng, max_iter, step_size=0.1):
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        cvals = corr * (2.0 / h.size)
        M = np.max(cvals)
        # find worst shift indices in full array:
        idxs = np.where(np.abs(cvals - M) < 1e-12)[0]
        # compute w vector length 2n-1 for each shift
        # sign to map to shift value: L = idx - (n-1). We'll handle indexing.
        n = h.size
        # w array for shift index L: w[L + n -1] = 1/len(idxs) if L in worst_shifts else 0
        w = np.zeros(2*n-1)
        for idx in idxs:
            w[idx] = 1.0/len(idxs)
        # compute term1 for each i:
        # term1 = 1 - (convolution of h with w reversed?) Let's compute convolution via np.convolve.
        # Using full convolution: conv = np.convolve(h, w, mode='full'), length n + 2n -2 = 3n -2. But we want for each i, sum over k w_k * h_{i+k}. This corresponds to conv where we align h with w.
        # Actually conv[i + (len(w)-1)] = Σ_j h_j w_{i + (len(w)-1) - j}. That's messy.
        # Simpler: compute term1_i = sum_{k} w_{k + n -1} * h_{i+k} where i+k in [0,n-1]. This is like sliding dot product.
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        # naive loops:
        for i in range(n):
            s1 = 0.0
            s2 = 0.0
            for L in range(-n+1, n):
                wk = w[L + n - 1]
                if wk == 0:
                    continue
                j = i + L
                if 0 <= j < n:
                    s1 += wk * h[j]  # this is sum over h[i+L]
                j2 = i - L
                if 0 <= j2 < n:
                    s2 += wk * h[j2]
            term1[i] = s1
            term2[i] = s2

        step = step_size / (it+1)**0.5  # decreasing

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- Constraint and objective for SLSQP ----------
def sum_eq_constraint(x):
    """Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- Core evaluation ----------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- SLSQP refinement ----------
def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce the sum exactly (tiny drift may appear)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Helpers for random feasible vectors ----------
def random_feasible_vector(n, rng):
    """Create a random vector in [0,1] with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- Simulated annealing ----------
def simulated_annealing(h, rng, max_steps, start_time, budget_s):
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    # temperature schedule (relative to current value)
    t0 = max(0.02, cur_c * 0.2)
    t1 = max(1e-6, cur_c * 0.001)
    for step in range(max_steps):
        if step % 200 == 0 and time.time() - start_time > budget_s * 0.98:
            break
        t = t0 * (1 - step / max_steps) + t1 * (step / max_steps)
        cand = perturb_h(cur, rng, scale=0.3)
        cand_c = compute_c5(cand)
        if cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c) / t):
            cur, cur_c = cand, cand_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
    return best, best_c


# ---------- Sub‑gradient descent on the true max ----------
def subgradient_descent(h, rng, max_iter=500):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        vals = corr * (2.0 / n)
        M = np.max(vals)
        # indices of worst shifts (full‑mode order)
        worst = np.where(np.abs(vals - M) < 1e-12)[0]
        w = np.zeros(2 * n - 1)
        w[worst] = 1.0 / len(worst)

        term1 = np.zeros(n)
        term2 = np.zeros(n)
        # vectorised slice updates
        for L in range(-n + 1, n):
            wk = w[L + n - 1]
            if wk == 0.0:
                continue
            if L >= 0:
                term1[: n - L] += wk * h[L:]
                term2[L:] += wk * h[: n - L]
            else:
                d = -L
                term1[d:] += wk * h[: n - d]
                term2[: n - d] += wk * h[d:]
        grad = (1.0 - term1) - term2
        step = 0.1 / np.sqrt(it + 1.0)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust the sum to stay feasible
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[i])
                else:
                    add = max(diff, -h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()
        h = h_new
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c
