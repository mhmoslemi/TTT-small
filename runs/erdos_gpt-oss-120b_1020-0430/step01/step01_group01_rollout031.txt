    rng = np.random.RandomState(seed)
    start = time.time()

    # Determine base vector
    try:
        base = np.array(initial_h_values, dtype=float)
    except NameError:
        base = None

    # Choose n = max(200, n0*2) but limit maybe 800
    n = min(max(200, n0*2), 800)   # ensures not huge
    target_sum = n/2.0
    dx = 2.0 / n

    def upscale(vec):
        if len(vec) == n:
            return vec.copy()
        # linear interpolation over points
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    if base is not None:
        h0 = upscale(base)
        h0 = _project_onto_capped_simplex(h0, target_sum)
    else:
        h0 = _random_feasible(rng, n, target_sum)

    best_h = h0.copy()
    best_val = _compute_c5(best_h, dx)

    # Optimization schedule
    schedule = [
        (3000, 10.0, 0.2),
        (3000, 30.0, 0.1),
        (3000, 80.0, 0.05),
    ]

    # Optimization function
    def run_opt(h_start, schedule):
        h = h_start.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        best_local_h = h.copy()
        best_local_val = _compute_c5(h, dx)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        for num_iters, alpha, lr in schedule:
            for _ in range(num_iters):
                # check time
                if time.time() - start > budget_s * 0.95:
                    break
                t += 1
                # compute gradient via smooth max
                corr = np.correlate(h, 1.0 - h, mode='full')
                max_corr = np.max(corr)
                shifted = corr - max_corr
                exp_alpha = np.exp(alpha * shifted)
                weights = exp_alpha / np.sum(exp_alpha)
                # compute weighted gradient
                grad = np.zeros_like(h)
                # loop over shifts
                n_len = len(h)
                for idx, w in enumerate(weights):
                    if w == 0.0:
                        continue
                    s = idx - (n_len - 1)
                    if s >= 0:
                        t_shift = s
                        # part 1
                        grad[:n_len - t_shift] += w * (1.0 - h[t_shift:])

                        grad[n_len - t_shift:] += w * 1.0

                        grad[t_shift:] += -w * h[:n_len - t_shift]
                    else:
                        t_shift = -s
                        grad[t_shift:] += w * (1.0 - h[:n_len - t_shift])
                        grad[:t_shift] += w * 1.0
                        grad[:n_len - t_shift] += -w * h[t_shift:]

                h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
                # projection onto feasible set
                h = _project_onto_capped_simplex(h, target_sum)
                # occasional evaluation
                if (_ % 20) == 0:
                    val = _compute_c5(h, dx)
                    if val < best_local_val:
                        best_local_val = val
                        best_local_h = h.copy()
        return best_local_h, best_local_val

    best_h = _project_onto_capped_simplex(best_h, target_sum)

import time, numpy as np

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=50):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

    return _project_onto_capped_simplex(v, target)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _optimize_adam(h0, target, dx, schedule, start_time, budget):
    h = h0.copy()
    n = len(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_h = h.copy()
    best_val = _compute_c5(h, dx)
    for num_iters, alpha, lr in schedule:
        for _ in range(num_iters):
            if time.time() - start_time > budget * 0.95:
                return best_h, best_val
            t += 1
            # correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            shifted = corr - max_corr
            exp_alpha = np.exp(alpha * shifted)
            weights = exp_alpha / np.sum(exp_alpha)
            # gradient
            grad = np.zeros_like(h)
            for idx, w in enumerate(weights):
                if w == 0.0:
                    continue
                s = idx - (n - 1)
                if s >= 0:
                    ts = s
                    # 0..n-ts-1
                    grad[:n - ts] += w * (1.0 - h[ts:])
                    grad[n - ts:] += w * 1.0
                    grad[ts:] += -w * h[:n - ts]
                else:
                    ts = -s
                    grad[ts:] += w * (1.0 - h[:n - ts])
                    grad[:ts] += w * 1.0
                    grad[:n - ts] += -w * h[ts:]
            # adam step
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)

                    best_val = val
                    best_h = h.copy()

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 50) -> np.ndarray:
    """Projection onto {0<=x<=1, sum(x)=target} using dual bisection."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Draw a random vector and project onto the feasible set."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Exact C5 value for the step function defined by h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# --------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate of the max‑overlap
# --------------------------------------------------------------

def _adam_optimize(h0: np.ndarray, target: float, dx: float,
                   schedule, start_time, budget):
    """
    Adam optimisation of  f(h)=smooth_max_{k}  ∑ h_i(1‑h_{i+k})
    schedule = list of (iterations, alpha, learning_rate)
    """
    h = h0.copy()
    n = h.shape[0]
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_h = h.copy()
    best_val = _compute_c5(h, dx)

    for iters, alpha, lr in schedule:
        for _ in range(iters):
            # time guard
            if time.time() - start_time > budget * 0.95:
                return best_h, best_val
            t += 1

            # correlation and softmax weights
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            shifted = corr - max_corr                     # ≤ 0
            exp_alpha = np.exp(alpha * shifted)
            weights = exp_alpha / exp_alpha.sum()

            # weighted gradient (smooth‑max)
            grad = np.zeros_like(h)
            for idx, w in enumerate(weights):
                if w == 0.0:
                    continue
                s = idx - (n - 1)        # shift corresponding to this entry
                if s >= 0:
                    ts = s
                    # 1‑h_{i+ts} part
                    grad[: n - ts] += w * (1.0 - h[ts:])
                    # edge‑only ones
                    grad[n - ts:] += w * 1.0
                    # –h_{i‑ts} part
                    grad[ts:] += -w * h[: n - ts]
                else:
                    ts = -s
                    grad[ts:] += w * (1.0 - h[: n - ts])
                    grad[:ts] += w * 1.0
                    grad[: n - ts] += -w * h[ts:]

            # Adam step
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep feasibility
            h = _project_onto_capped_simplex(h, target)

            # occasional exact evaluation
            if (t % 50) == 0:
                cur = _compute_c5(h, dx)
                if cur < best_val:
                    best_val = cur
                    best_h = h.copy()
    return best_h, best_val
