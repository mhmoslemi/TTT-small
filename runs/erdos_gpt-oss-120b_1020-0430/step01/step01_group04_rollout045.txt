import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_to_bounds_and_sum(v, target, tol=1e-12, max_iter=20):
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = v.sum()
        diff = target - cur
        if abs(diff) <= tol:
            break
        # variables not at a bound can absorb the diff
        mask = (v > 0.0) & (v < 1.0)
        free_idx = np.where(mask)[0]
        if free_idx.size == 0:
            # all at bounds, distribute uniformly (will be re‑clipped)
            delta = diff / v.size
            v += delta
            v = np.clip(v, 0.0, 1.0)
        else:
            delta = diff / free_idx.size
            v[free_idx] += delta
            v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    residual = target - v.sum()
    if abs(residual) > tol:
        v += residual / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def generate_triangular(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    return project_to_bounds_and_sum(tri, target)

def generate_sine_shape(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    # base sine that vanishes at the borders
    base = np.sin(np.pi * x / 2.0)  # max 1 at x=1
    scale = (np.pi / 4.0) / base.max()  # ensure integral roughly 1 after scaling later
    h = scale * base
    target = n / 2.0
    # rescale to meet integral exactly while staying inside [0,1]
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return project_to_bounds_and_sum(h, target)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    return project_to_bounds_and_sum(up, target)

def random_uniform_feasible(n, rng):
    target = n / 2.0
    v = rng.random(n)
    v = v * (target / v.sum())
    v = np.clip(v, 0.0, 1.0)
    return project_to_bounds_and_sum(v, target)

def random_binary_feasible(n, rng):
    target = n // 2  # integer number of ones required
    v = np.zeros(n)
    idx = rng.choice(n, target, replace=False)
    v[idx] = 1.0
    # may not hit sum exactly due to target being integer, but ensures sum = target
    return v.astype(float)

def smooth_max_obj(h, alpha=80.0):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return logsumexp(alpha * conv) / alpha

        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, final_c5, best_n

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def project_to_bounds_and_sum(v: np.ndarray, target: float,
                             tol: float = 1e-12, max_iter: int = 20) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(v) == target (simple water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = v.sum()
        diff = target - cur
        if abs(diff) <= tol:
            break
        mask = (v > 0.0) & (v < 1.0)
        free = np.where(mask)[0]
        if free.size == 0:
            delta = diff / v.size
            v += delta
            v = np.clip(v, 0.0, 1.0)
        else:
            delta = diff / free.size
            v[free] += delta
            v = np.clip(v, 0.0, 1.0)
    residual = target - v.sum()
    if abs(residual) > tol:
        v += residual / v.size
        v = np.clip(v, 0.0, 1.0)
    return v


def generate_triangular(n: int) -> np.ndarray:
    """Triangular bump h(x)=min(x,2−x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    return project_to_bounds_and_sum(tri, target)


def generate_sine_shape(n: int) -> np.ndarray:
    """Half‑sine bump vanishing at the ends, scaled to satisfy the integral."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    base = np.sin(np.pi * x / 2.0)          # 0 at 0 and 2, 1 at 1
    scale = np.pi / 4.0                     # makes integral ≈1 before discretisation
    h = scale * base
    target = n / 2.0
    h = h * (target / h.sum())
    return project_to_bounds_and_sum(h, target)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, keeping the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    return project_to_bounds_and_sum(up, target)


def random_uniform_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector on the bounded simplex."""
    target = n / 2.0
    v = rng.random(n)
    v = v * (target / v.sum())
    return project_to_bounds_and_sum(v, target)


def random_binary_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary feasible vector (exactly n/2 ones, the rest zeros)."""
    ones = n // 2
    v = np.zeros(n)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v.astype(float)


def smooth_max_obj(h: np.ndarray, alpha: float = 80.0) -> float:
    """Smooth approximation of the max overlap via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return logsumexp(alpha * conv) / alpha
