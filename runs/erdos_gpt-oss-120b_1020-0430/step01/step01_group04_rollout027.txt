def compute_softmax_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0/n
    conv = np.correlate(h, 1 - h, mode='full')
    # conv length 2n-1
    # compute max via softmax
    # (max) approx not needed for gradient? we compute objective = softmax(conv, alpha).
    # but we can compute f = logsumexp(alpha*conv)/alpha.
    # compute weights w = exp(alpha*conv - max)/sum(exp(alpha*conv - max))
    # each weight corresponds to shift index.
    max_val = np.max(alpha * conv)
    w = np.exp(alpha * conv - max_val)
    w_sum = np.sum(w)
    w /= w_sum

    # gradient of objective:
    # f = logsumexp(alpha*conv)/alpha
    # df/dconv_k = w_k
    # dconv_k/dh_j = 1 - (h_{j+shift} + h_{j-shift})
    # So grad_j = sum_k w_k * (1 - (h_{j+shift} + h_{j-shift}))
    # = sum_k w_k - sum_k w_k*(h_{j+shift}+h_{j-shift}) = 1 - contribution
    # Since sum_k w_k = 1.
    # So compute contribution = sum_k w_k * (h_{j+shift}+h_{j-shift})
    # Then grad = 1 - contribution.

    # Compute contribution efficiently:
    contribution = np.zeros(n)
    # pad h
    pad = np.zeros(n)
    hp = np.concatenate([pad, h, pad])
    N = len(hp)
    # shift indices range -(n-1) to (n-1). We loop over idx, shift = idx - (n-1).
    for idx, wk in enumerate(w):
        s = idx - (n - 1)
        # forward slice = h_{j+s}
        # slice where j runs 0..n-1, slice = hp[n + s : n + s + n]
        # Note hp start index = n -> corresponds to h[0]
        f_slice = hp[n + s : n + s + n]
        b_slice = hp[n - s : n - s + n]  # corresponds to h_{j-s}
        contribution += wk * (f_slice + b_slice)
    grad = 1.0 - contribution
    # gradient of f w.r.t h = grad (since f not multiplied by dx)
    return f, grad

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Global for the smooth objective
# ------------------------------------------------------------
_current_alpha = 5.0


def compute_c5(h: np.ndarray) -> float:
    """Exact Erdős C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def compute_softmax_c5_and_grad(h: np.ndarray, alpha: float):
    """Smooth max (log‑sum‑exp) of the overlap and its gradient."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * conv
    a_max = np.max(a)
    exp_a = np.exp(a - a_max)
    w = exp_a / np.sum(exp_a)                     # softmax weights, sum to 1

    # padded version of h for easy shifting
    pad = np.zeros(n)
    hp = np.concatenate([pad, h, pad])
    contrib = np.zeros(n)
    centre = n - 1  # index of zero shift in the full correlation

    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        s = idx - centre                       # shift amount (can be negative)
        start = n + s
        f_slice = hp[start:start + n]          # h_{j+s}
        b_slice = hp[n - s:n - s + n]          # h_{j-s}
        contrib += wk * (f_slice + b_slice)

    grad = 1.0 - contrib                        # ∂f/∂h_j
    logsumexp = a_max + np.log(np.sum(exp_a))
    val = logsumexp / alpha
    return val, grad


def softmax_obj(h):
    """Objective wrapper used by scipy.optimize (reads global alpha)."""
    val, _ = compute_softmax_c5_and_grad(h, _current_alpha)
    return val


def softmax_grad(h):
    """Gradient wrapper used by scipy.optimize (reads global alpha)."""
    _, g = compute_softmax_c5_and_grad(h, _current_alpha)
    return g


class SumConstraint:
    """Linear equality ∑h = target."""
    def __init__(self, target):
        self.target = target

    def fun(self, h):
        return np.sum(h) - self.target

    def jac(self, h):
        return np.ones_like(h)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape (tent) normalised to integral 1."""
    dx = 2.0 / n
    x = np.linspace(0.0, 2.0, n, endpoint=False) + dx / 2.0
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (bounds + exact sum)."""
    target = n / 2.0
    a = rng.random(n)
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old grid to a finer grid."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def solve_one_start(start: np.ndarray, n: int, deadline: float):
    """Two‑stage continuation SLSQP using increasing smoothness."""
    target = n / 2.0
    eq = SumConstraint(target)
    cons = {"type": "eq", "fun": eq.fun, "jac": eq.jac}
    bounds = [(0.0, 1.0)] * n

    alphas = [5.0, 15.0, 30.0, 80.0, 200.0]
    h = start.copy()

    for alp in alphas:
        if time.time() > deadline:
            return None, None
        global _current_alpha
        _current_alpha = alp
        res = minimize(
            softmax_obj,
            h,
            method="SLSQP",
            jac=softmax_grad,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            # still keep the iterate, it may be useful
            h = res.x
        else:
            h = res.x

        # Project back onto the feasible set (tiny numerical drifts)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)

    return h, compute_c5(h)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.94  # safety margin

    # possible resolutions (keep them modest for speed)
    candidate_ns = [120, 150, 180, 200, 240]

    # incorporate parent solution if available
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # construct diverse starts
        starts = []

        if parent_h is not None:
            starts.append(parent_h.copy() if parent_n == n else upsample(parent_h, n))

        starts.append(generate_triangular(n))

        for _ in range(5):
            starts.append(random_start(n, rng))

        tri = generate_triangular(n)
        for _ in range(3):
            pert = tri + rng.uniform(-0.1, 0.1, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            diff = target - pert.sum()
            if abs(diff) > 1e-12:
                pert += diff / n
                pert = np.clip(pert, 0.0, 1.0)
            starts.append(pert)

        # optimise each start
        for s in starts:
            if time.time() > deadline:
                break
            h_opt, c5_opt = solve_one_start(s, n, deadline)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_opt, n

    # fallback in the unlikely event nothing succeeded
    if best_h is None:
        best_n = 120
        best_h = generate_triangular(best_n)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------
    # Local refinement: simple pairwise swaps (hill climbing)
    # ------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        h = best_h.copy()
        cur_c5 = best_c5
        n = best_n
        target = n / 2.0
        max_iters = int(remaining * 200)  # rough budget
        for _ in range(max_iters):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # amount that can be transferred from j to i
            max_inc = min(1.0 - h[i], h[j])
            if max_inc <= 0.0:
                continue
            delta = max_inc * rng.random()
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            new_c5 = compute_c5(h_new)
            if new_c5 < cur_c5 - 1e-10:
                h, cur_c5 = h_new, new_c5
        best_h, best_c5 = h, cur_c5

    # final safety projection onto the feasible set
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-12:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
