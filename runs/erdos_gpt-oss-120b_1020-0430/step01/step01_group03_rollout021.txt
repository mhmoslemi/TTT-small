def targeted_move(h, rng, scale):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    low = max(0, -shift)
    high = n - max(0, shift)
    i_range = np.arange(low, high)
    j_range = i_range + shift
    contrib = h[i_range] * (1.0 - h[j_range])
    src_local = np.argmax(contrib)
    tgt_local = np.argmin(contrib)
    src = i_range[src_local]
    tgt = i_range[tgt_local]
    max_eps = min(h[src], 1.0 - h[tgt])
    if max_eps <= 0:
        # fallback random perturb
        return perturb_h(h, rng, scale)
    eps = max_eps * rng.random() * scale
    new_h = h.copy()
    new_h[src] -= eps
    new_h[tgt] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the SLSQP surrogate (soft‑max of the maximum)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for a step function (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
# Constraints / surrogate for SLSQP
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # numerically stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """One SLSQP optimisation using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce exact sum (numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ----------------------------------------------------------------------
def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break

            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Targeted move based on the worst shift
# ----------------------------------------------------------------------
def targeted_move(h, rng, scale):
    """Transfer ε from a high‑contribution index to a low‑contribution one."""
    n = h.size
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr_raw)
    # choose uniformly among all worst shifts
    best_idxs = np.flatnonzero(corr_raw == max_val)
    idx = rng.choice(best_idxs)
    shift = idx - (n - 1)  # convert index to actual shift

    # avoid shift == 0 (which would increase variance)
    if shift == 0:
        return perturb_h(h, rng, scale)

    low = max(0, -shift)
    high = n - max(0, shift)
    i_range = np.arange(low, high, dtype=int)
    j_range = i_range + shift

    contrib = h[i_range] * (1.0 - h[j_range])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)

    src_local = np.argmax(contrib)
    tgt_local = np.argmin(contrib)
    src = i_range[src_local]
    tgt = i_range[tgt_local]

    max_eps = min(h[src], 1.0 - h[tgt])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)

    eps = max_eps * rng.random() * scale
    new_h = h.copy()
    new_h[src] -= eps
    new_h[tgt] += eps
    return new_h


def hill_climb_targeted(h_start, rng, max_iters=5000, start_scale=0.5, decay=0.97):
    """Greedy hill‑climb using the targeted move."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = start_scale
    no_improve = 0

    for it in range(max_iters):
        # occasionally fall back to a random perturb to escape plateaus
        if rng.random() < 0.05:
            new_h = perturb_h(h, rng, scale=scale)
        else:
            new_h = targeted_move(h, rng, scale)

        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
            no_improve = 0
        else:
            no_improve += 1
            # if we stagnate, reduce step size a bit more aggressively
            if no_improve > 200:
                scale = max(scale * 0.5, 1e-5)

        # schedule for scale reduction
        if (it + 1) % (max_iters // 10 + 1) == 0:
            scale = max(scale * decay, 1e-5)

    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise from the supplied parent solution (if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h0[i])
                else:
                    add = max(diff, -h0[i])
                h0[i] += add
                diff = target - h0.sum()
    else:
        n = 90
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1 – focused targeted hill‑climb
    # ------------------------------------------------------------------
    time_used = time.time() - start_time
    if time_used < budget_s * 0.65:
        hc_h, hc_c = hill_climb_targeted(
            best_h, rng, max_iters=8000, start_scale=0.5, decay=0.96
        )
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

    # ------------------------------------------------------------------
    # Phase 2 – SLSQP refinement with an expanding beta ladder
    # ------------------------------------------------------------------
    betas = [5.0, 20.0, 80.0, 200.0]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.85:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=400)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            continue

    # ------------------------------------------------------------------
    # Phase 3 – random restarts with brief targeted climb + SLSQP
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.96:
        h_start = random_feasible_vector(n, rng)
        # brief climb
        h_tmp, c_tmp = hill_climb_targeted(h_start, rng, max_iters=1500,
                                           start_scale=0.4, decay=0.95)
        # short SLSQP passes
        for beta in betas[:2]:
            try:
                h_tmp = slsqp_optimize(h_tmp, beta=beta, max_iter=150)
            except Exception:
                pass
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ------------------------------------------------------------------
    # Optional up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.99 and n * 2 <= 200:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # enforce exact sum after up‑sampling
        diff = n_up / 2.0 - h_up.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n_up)
                if diff > 0:
                    add = min(diff, 1.0 - h_up[i])
                else:
                    add = max(diff, -h_up[i])
                h_up[i] += add
                diff = n_up / 2.0 - h_up.sum()
        # targeted climb on finer grid
        h_up, c_up = hill_climb_targeted(h_up, rng, max_iters=4000,
                                         start_scale=0.3, decay=0.97)
        for beta in betas:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta=beta, max_iter=250)
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final clean‑up (ensure exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
