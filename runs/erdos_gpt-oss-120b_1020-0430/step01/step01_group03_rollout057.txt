def simulated_annealing(h, rng, max_iters, start_temp, end_temp):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(max_iters):
        t = start_temp * ( (end_temp/start_temp) ** (it / max_iters) )
        # propose perturbation
        new_h = perturb_h(cur_h, rng, scale=0.2)  # could adapt scale based on t
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / t):
            cur_h = new_h
            cur_c = new_c
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        # optionally occasionally adjust scale
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the soft‑max surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality: Σ_i h_i = CURRENT_TARGET (i.e. n/2)."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i h_i (1‑h_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for a step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP with a soft‑max surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce exact sum (tiny numerical drift may appear)
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 6):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (sum conserved)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements) with step decay."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # exponential decay of the step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def simulated_annealing(h0, rng, max_iters=20000, start_temp=0.01, end_temp=1e-6):
    """Simple SA using the same perturbation (sum is preserved)."""
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for it in range(max_iters):
        # geometric temperature schedule
        t = start_temp * ((end_temp / start_temp) ** (it / max_iters))
        new_h = perturb_h(cur_h, rng, scale=0.25)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c


def _adjust_sum(h, target, rng):
    """Redistribute a tiny mass to achieve an exact target sum."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

        for beta in betas[:3]:
            try:
                h_tmp = slsqp_optimize(h_tmp, beta=beta, max_iter=200)
            except Exception:
                pass
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ------------------------------------------------------------------
    # Phase 4 – up‑sample (if room) and fine‑tune on the finer grid
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.85 and base_n * 2 <= 400:
        h_fine = np.repeat(best_h, 2)
        n_fine = h_fine.size
        h_fine = _adjust_sum(h_fine, n_fine / 2.0, rng)

        # a longer hill‑climb on the fine grid
        h_fine, c_fine = hill_climb(h_fine, rng, max_iters=8000, start_step=0.4)

        # simulated annealing – run until time runs low
        while time.time() - start_time < budget_s * 0.97:
            remaining = budget_s - (time.time() - start_time)
            # allocate a modest chunk of SA iterations
            iters = min(20000, int(remaining * 3000))
            if iters < 1000:
                break
            h_fine, c_fine = simulated_annealing(
                h_fine, rng, max_iters=iters, start_temp=0.02, end_temp=1e-6
            )
            if c_fine < best_c:
                best_h, best_c = h_fine, c_fine
        # keep the finer result if it improved
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine

    # ------------------------------------------------------------------
    # Final clean‑up (enforce feasibility exactly)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum(best_h, best_h.size / 2.0, rng)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
