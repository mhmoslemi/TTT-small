def make_trapezoidal(n, a, s):
    # a is ramp length, s total support (a+b?), where b = s - 2*a
    b = s - 2*a
    if b < 0: return None
    h_max = 1.0 / s
    # generate x positions at centers of intervals: x_i = (i+0.5)/n*2
    i = np.arange(n)
    x = 2.0 * (i + 0.5) / n
    left = 1.0 - s/2.0
    # compute piecewise:
    # Use np.where for each region
    h = np.zeros_like(x)
    # left ramp region:
    mask = (x >= left) & (x < left + a)
    # linearly map from [left, left+a] to [0,h_max]
    h[mask] = h_max * (x[mask] - left) / a  # if a>0 else ignore
    # plateau region:
    mask2 = (x >= left + a) & (x < left + a + b)
    h[mask2] = h_max
    # right ramp region:
    mask3 = (x >= left + a + b) & (x <= left + s)  # left+s = 1+ s/2?
    # However left + s = left + s = 1 - s/2 + s = 1 + s/2.
    # mapping: at left + a + b, value h_max; at left + s, 0.
    if a>0:
        h[mask3] = h_max * (left + s - x[mask3]) / a
    else:
        # if a==0, ramp length zero, just plateau.
        h[mask3] = 0.0  # but mask3 would be empty anyway maybe
    return h

def trapezoidal_shape(n, a, s):
    b = s - 2*a
    h_max = 1.0/s
    # positions at midpoints:
    pos = (np.arange(n) + 0.5) * 2.0 / n
    left = 1.0 - s/2.0
    h = np.zeros_like(pos)
    # left ramp:
    if a > 1e-12:
        idx = (pos >= left) & (pos < left + a)
        h[idx] = h_max * (pos[idx] - left) / a
    # plateau:
    if b > 1e-12:
        idx = (pos >= left + a) & (pos < left + a + b)
        h[idx] = h_max
    # right ramp:
    if a > 1e-12:
        idx = (pos >= left + a + b) & (pos <= left + s)  # inclusive right
        h[idx] = h_max * (left + s - pos[idx]) / a
    # else a=0 no ramp region.
    return h

def normalize_h(h, target):
    cur_sum = h.sum()
    if cur_sum == 0:
        return h
    h = h * (target / cur_sum)
    # Clip
    np.clip(h, 0.0, 1.0, out=h)
    # After clipping may deviate from target, so adjust difference
    diff = target - h.sum()
    if abs(diff) > 1e-12:

def local_optimize(h0, n, time_deadline):
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    # Stage 1: soft objective low alpha
    opts = {"maxiter": 600, "ftol": 1e-9, "disp": False}
    res1 = opt.minimize(lambda h: softmax_c5(h, alpha=10.0), h0,
                        method='SLSQP', bounds=bounds, constraints=cons, options=opts)
    if not res1.success:
        return None, None
    if time.time() > time_deadline:
        return None, None
    # Stage 2: higher alpha
    res2 = opt.minimize(lambda h: softmax_c5(h, alpha=40.0), res1.x,
                        method='SLSQP', bounds=bounds, constraints=cons,
                        options={"maxiter": 600, "ftol": 1e-9, "disp": False})
    if not res2.success:
        return None, None
    h_opt = np.clip(res2.x, 0.0, 1.0)
    # Ensure sum
    cur = h_opt.sum()
    diff = target - cur
    if abs(diff) > 1e-9:
        h_opt += diff/n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5 = compute_c5(h_opt)
    return h_opt, c5

def generate_trapezoidal(n, a, s):
    # a: ramp length, s: total support length (a+b+a)
    # compute b = s - 2*a (>=0)
    b = max(s - 2*a, 0.0)
    h_max = 1.0 / s
    x = (np.arange(n) + 0.5) * 2.0 / n
    left = 1.0 - s / 2.0
    h = np.zeros_like(x)
    # left ramp
    if a > 1e-12:
        mask = (x >= left) & (x < left + a)
        h[mask] = h_max * (x[mask] - left) / a
    # plateau
    if b > 1e-12:
        mask = (x >= left + a) & (x < left + a + b)
        h[mask] = h_max
    # right ramp
    if a > 1e-12:
        mask = (x >= left + a + b) & (x <= left + s)  # inclusive
        h[mask] = h_max * (left + s - x[mask]) / a
    return h

def adjust_sum(h, target):
    current = h.sum()
    if current == 0:
        return h
    scale = target / current
    h = h * scale
    # Ensure within [0,1]; after scaling some may exceed 1; clip and re-balance.
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff uniformly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max_k ∫ h(x)(1‑h(x+k)) using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def _adjust_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Scale and clip h so that sum(h) == target while staying in [0,1]."""
    cur = h.sum()
    if cur == 0:
        return h
    h = h * (target / cur)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------
def _trapezoidal(n: int, a: float, s: float, offset: float) -> np.ndarray:
    """
    Symmetric (or shifted) trapezoid.
    a   – length of each rising/falling ramp (>=0)
    s   – total support length (a+b+a) with b = s‑2a (>=0)
    offset – leftmost point of the support (0 ≤ offset ≤ 2‑s)
    """
    b = max(s - 2.0 * a, 0.0)
    h_max = 1.0 / s
    x = (np.arange(n) + 0.5) * 2.0 / n
    left = offset
    right = offset + s
    h = np.zeros_like(x)

    if a > 1e-12:  # left ramp
        mask = (x >= left) & (x < left + a)
        h[mask] = h_max * (x[mask] - left) / a
    if b > 1e-12:  # plateau
        mask = (x >= left + a) & (x < left + a + b)
        h[mask] = h_max
    if a > 1e-12:  # right ramp
        mask = (x >= left + a + b) & (x <= right)
        h[mask] = h_max * (right - x[mask]) / a
    return h


def _power_shape(n: int, p: float) -> np.ndarray:
    """
    h(x) = max(0, 1 - (|x‑1| / a)^p) with a = (p+1)/(2p) (so that ∫h = 1).
    p > 0.
    """
    a = (p + 1.0) / (2.0 * p)
    a = min(a, 1.0)            # cannot exceed half the interval
    x = (np.arange(n) + 0.5) * 2.0 / n
    r = np.abs(x - 1.0)
    with np.errstate(divide="ignore", invalid="ignore"):
        h = np.maximum(0.0, 1.0 - (r / a) ** p)
    return h


# ----------------------------------------------------------------------
# Local continuous optimisation (SLSQP)
# ----------------------------------------------------------------------
def _local_optimize(start: np.ndarray, n: int, deadline: float) -> tuple:
    """Two‑stage SLSQP from a feasible start; returns (h, c5) or (None, None)."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    opts = {"maxiter": 600, "ftol": 1e-9, "disp": False}

    # stage 1 – smoother objective
    res1 = minimize(lambda h: softmax_c5(h, alpha=10.0), start,
                    method="SLSQP", bounds=bounds, constraints=cons,
                    options=opts)
    if not res1.success or time.time() > deadline:
        return None, None

    # stage 2 – sharper approximation
    res2 = minimize(lambda h: softmax_c5(h, alpha=40.0), res1.x,
                    method="SLSQP", bounds=bounds, constraints=cons,
                    options=opts)
    if not res2.success or time.time() > deadline:
        return None, None

    h_opt = np.clip(res2.x, 0.0, 1.0)
    # final projection onto the exact simplex (tiny correction)
    cur = h_opt.sum()
    if abs(cur - target) > 1e-9:
        h_opt += (target - cur) / n
        np.clip(h_opt, 0.0, 1.0, out=h_opt)
    return h_opt, compute_c5(h_opt)
