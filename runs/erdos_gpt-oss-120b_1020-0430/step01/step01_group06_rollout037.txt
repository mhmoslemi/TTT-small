def project_simplex_box(v, target_sum):
    # water-filling algorithm; find lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    lo = -1.0
    hi = 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

        if move_type < 0.6:
            # random epsilon swap
            i, j = random.sample(range(n), 2)
            hi, hj = cur_h[i], cur_h[j]
            max_up = min(1.0 - hi, hj)
            max_down = min(hi, 1.0 - hj)

            nh = cur_h.copy()

            nh[j] = hj - eps

            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)
            if s >= 0:

                p_inds = i_inds + s
            else:

                p_inds = i_inds - s_abs
            contrib = cur_h[i_inds] * (1.0 - cur_h[p_inds])
            k = int(np.argmax(contrib))
            i_best = i_inds[k]
            p_best = p_inds[k]

            nh[p_best] += eps

            noise = np.random.randn(n) * sigma
            nh = cur_h + noise
            nh = project(nh)

        # acceptance

            cur_h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = nh.copy()
                last_improve_iter = it

        it += 1
        if it % 1000 == 0:
            T *= decay

        # restart if stagnated
        if it - last_improve_iter > 20000:
            # reinitialize around best with perturbation
            cur_h = best_h.copy()
            # add small random noise
            cur_h = project(cur_h + np.random.randn(n) * 0.02)
            cur_c5 = c5(cur_h)
            last_improve_iter = it

    # final adjustment: ensure sum correct and clip
    best_h = np.clip(best_h, 0.0, 1.0)
    err = best_h.sum() - target
    if abs(err) > 1e-8:

                best_h[k] -= err
                break
            if err < 0 and best_h[k] <= 1 + err:
                best_h[k] -= err
                break

    final_c5 = c5(best_h)
    return (best_h.tolist(), final_c5, n)

def run(seed=42, budget_s=1100, **kwargs):

    random.seed(seed)

    start = time.time()

    try:
        init = np.array(initial_h_values, dtype=float).copy()
        n = init.shape[0]
    except NameError:
        n = 70
        init = None

    target_sum = n / 2.0
    dx = 2.0 / n

    def project(v):
        """ Euclidean projection onto {0<=x_i<=1, sum x = target_sum}. """
        v = np.clip(v, 0.0, 1.0)
        lo, hi = -1.0, 1.0
        for _ in range(30):
            mid = (lo + hi) / 2.0
            w = np.clip(v - mid, 0.0, 1.0)
            s = w.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    def c5(v):
        """ compute the Erdős C5 value for a vector v """
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(corr.max() * dx)
