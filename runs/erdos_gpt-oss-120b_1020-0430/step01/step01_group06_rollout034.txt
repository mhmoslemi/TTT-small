def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s * 0.95      # leave some margin

    # ----------------------------------------------------------------------
    # Parameters (feel free to tune)
    # ----------------------------------------------------------------------
    n = 200                                 # number of samples ( < 1000 )
    target_sum = n / 2.0                    # integral condition
    step_initial = 0.2                      # base step length
    decay = 0.9995                          # temperature decay factor
    # ----------------------------------------------------------------------
    # Helper: projection onto [0,1] with exact sum = target_sum
    # ----------------------------------------------------------------------
    def project_to_sum_box(x):
        lo, hi = -2.0, 2.0                  # search interval for the offset τ
        for _ in range(40):                 # enough for 1e-12 accuracy on our scale
            mid = (lo + hi) * 0.5
            y = np.clip(x - mid, 0.0, 1.0)
            s = y.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    # ----------------------------------------------------------------------
    # Helper: evaluate C5 for a vector of length n
    # ----------------------------------------------------------------------
    def c5_of(vec):
        return float(np.correlate(vec, 1.0 - vec, mode="full").max() *
                     (2.0 / n))

    # ----------------------------------------------------------------------
    # Initialise candidate solutions
    # ----------------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")

    # 1) use the provided construction if it exists (interpolate to n points)
    try:
        init = np.array(initial_h_values, dtype=float)

        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init, left=0.0, right=0.0)
        h0 = project_to_sum_box(h0)
        v = c5_of(h0)
        if v < best_c5:
            best_c5, best_h = v, h0.copy()
    except NameError:
        pass

    # 2) a simple triangular shape (height 1, base 2)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    tri = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    tri = project_to_sum_box(tri)
    v = c5_of(tri)
    if v < best_c5:
        best_c5, best_h = v, tri.copy()

    # 3) a random start if nothing else succeeded
    if best_h is None:
        rnd = np.random.rand(n)

    # make a working copy

    # temperature for occasional uphill moves
    T = max(cur_c5 * 0.02, 1e-8)

    it = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop (gradient + occasional random swaps)
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # -------------------- evaluate worst shift -----------------------
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr_full))
        cur_c5 = corr_full[idx_max] * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()

        # -------------------- gradient step targeting worst shift -------
        s = idx_max - (n - 1)                     # signed shift producing max
        g = np.zeros_like(h)                     # gradient of the worst term
        overlap_len = n - abs(s)
        i_start = 0 if s >= 0 else -s
        i = np.arange(i_start, i_start + overlap_len, dtype=int)
        j = i + s
        g[i] += 1.0 - h[j]                        # ∂/∂h_i (1 – h_j)
        g[j] += -h[i]                             # ∂/∂h_j (–h_i)

        # direction (steepest descent) – normalise to keep step size stable
        norm = np.linalg.norm(g)
        if norm > 0.0:
            d = -g / norm
        else:
            d = np.random.randn(n)
            d /= np.linalg.norm(d)

    # ----------------------------------------------------------------------
    best_h = project_to_sum_box(best_h)     # enforce constraints exactly
    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
