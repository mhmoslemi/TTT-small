def _evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = per if per < cap else cap
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = -per if -per < cap else cap
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x, target):
    """Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple redistribution loop
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = per if per < cap else cap
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = -per if -per < cap else cap
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _hill_climb(h, cur_score, deadline, rng):
    """Fine‑grained random hill‑climbing using small mass exchanges."""
    n = len(h)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # choose direction uniformly
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_score = _evaluate_c5(h_new)
        if new_score + 1e-12 < cur_score:
            h, cur_score = h_new, new_score
    return h, cur_score
