def project_onto_box_simplex(v, target, lower=0.0, upper=1.0, tol=1e-10, max_iter=100):
    # binary search for λ such that sum_clipped = target
    lo = v.min() - upper
    hi = v.max() - lower
    for _ in range(max_iter):
        mid = (lo + hi)/2
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    # final projection after loop
    return np.clip(v - lo, lower, upper)

def proj_simplex_box(v, target, lo=0, hi=1, eps=1e-12):
    """project onto {x: lo <= x <= hi, sum x = target}"""
    # initial λ bounds:
    # When λ = -inf, v - λ -> +inf, clipped to hi => sum = n*hi
    # When λ = +inf, v - λ -> -inf, clipped to lo => sum = n*lo
    # So we can binary search λ between -M and M such that sum=target.
    # Choose bounds based on v's range.

    lambda_low = -1e6
    lambda_high = 1e6
    for _ in range(100):
        lam = (lambda_low + lambda_high)/2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            # want larger lam to reduce values
            lambda_low = lam
        else:
            lambda_high = lam

import numpy as np, random, time, math

def project_onto_box_simplex(v, target, low=0.0, high=1.0, tol=1e-12, max_iter=60):
    n = v.shape[0]
    # fast checks
    if target <= n * low:
        return np.full_like(v, low)
    if target >= n * high:
        return np.full_like(v, high)
    lo = np.min(v) - high
    hi = np.max(v) - low
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, low, high)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - lo, low, high)

def c5_value(v):
    n = v.shape[0]
    dx = 2.0 / n
    w = np.correlate(v, 1.0 - v, mode='full')
    return float(w.max() * dx)

def smooth_max_and_grad(h, gamma):
    n = h.shape[0]
    w = np.correlate(h, 1.0 - h, mode='full')
    gw = gamma * w
    max_gw = np.max(gw)
    exp_vals = np.exp(gw - max_gw)
    sum_exp = exp_vals.sum()
    probs = exp_vals / sum_exp
    grad = np.zeros_like(h)
    # iterate over shifts
    for shift in range(-n + 1, n):
        p = probs[shift + n - 1]
        if p == 0.0:
            continue
        if shift >= 0:
            i_slice = slice(0, n - shift)
            j_slice = slice(shift, n)
        else:
            i_slice = slice(-shift, n)
            j_slice = slice(0, n + shift)
        grad[i_slice] += p * (1.0 - h[j_slice])
        grad[j_slice] += p * (-h[i_slice])
    smooth = (max_gw + math.log(sum_exp)) / gamma
    return smooth, grad

def gradient_descent_phase(h0, target, time_budget):
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    cur_c5 = c5_value(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = 0.5
    step = 0.2
    T = 0.01
    it = 0
    start = time.time()
    while time.time() - start < time_budget:
        smooth, grad = smooth_max_and_grad(h, gamma)
        nh = h - step * grad
        nh = project_onto_box_simplex(nh, target)
        new_c5 = c5_value(nh)
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = nh.copy()
        it += 1
        if it % 200 == 0:
            gamma = min(gamma * 1.07, 30.0)
            step *= 0.97
            T *= 0.97
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

    for i in range(n):

import numpy as np, random, time, math

def project_onto_box_simplex(v, target, low=0.0, high=1.0, tol=1e-12, max_iter=60):
    """Euclidean projection onto {x: sum(x)=target, low<=x<=high}."""
    n = v.shape[0]
    if target <= n * low:
        return np.full_like(v, low)
    if target >= n * high:
        return np.full_like(v, high)
    lo = np.min(v) - high
    hi = np.max(v) - low
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, low, high)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - lo, low, high)


def c5_value(v):
    """Exact C5 for a step function represented by the vector v."""
    n = v.shape[0]
    dx = 2.0 / n
    w = np.correlate(v, 1.0 - v, mode="full")
    return float(w.max() * dx)


def smooth_max_and_grad(h, gamma):
    """Smooth approximation of max_s w_s and its gradient."""
    n = h.shape[0]
    w = np.correlate(h, 1.0 - h, mode="full")          # w_s = Σ h_i (1−h_{i+s})
    gw = gamma * w
    max_gw = np.max(gw)
    exp_vals = np.exp(gw - max_gw)                    # avoid overflow
    sum_exp = exp_vals.sum()
    probs = exp_vals / sum_exp                         # softmax weights

    grad = np.zeros_like(h)
    for shift in range(-n + 1, n):
        p = probs[shift + n - 1]
        if p == 0.0:
            continue
        if shift >= 0:
            i_slice = slice(0, n - shift)
            j_slice = slice(shift, n)
        else:
            i_slice = slice(-shift, n)
            j_slice = slice(0, n + shift)
        grad[i_slice] += p * (1.0 - h[j_slice])
        grad[j_slice] += p * (-h[i_slice])
    smooth = (max_gw + math.log(sum_exp)) / gamma
    return smooth, grad


def gradient_descent_phase(h0, target, time_budget):
    """Continuous optimisation using a soft‑max surrogate."""
    h = h0.copy()
    cur_c5 = c5_value(h)
    best_h, best_c5 = h.copy(), cur_c5

    gamma = 0.5
    step = 0.2
    T = 0.01
    it = 0
    start = time.time()

    while time.time() - start < time_budget:
        _, grad = smooth_max_and_grad(h, gamma)
        nh = h - step * grad
        nh = project_onto_box_simplex(nh, target)
        new_c5 = c5_value(nh)

        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            h, cur_c5 = nh, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = nh.copy(), new_c5

        it += 1
        if it % 200 == 0:
            gamma = min(gamma * 1.07, 30.0)
            step *= 0.97
            T *= 0.97
        if best_c5 < 0.3808:
            break
    return best_h, best_c5


def local_swap_phase(h0, target, time_budget):
    """Fine‑grained hill climbing with random two‑coordinate moves."""
    h = h0.copy()
    cur_c5 = c5_value(h)
    best_h, best_c5 = h.copy(), cur_c5

    T = 0.01
    decay = 0.9995
    it = 0
    start = time.time()
    n = h.shape[0]

    while time.time() - start < time_budget:
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)
        max_down = min(hi, 1.0 - hj)
        if max_up + max_down == 0:
            continue
        delta = (random.random() * (max_up + max_down)) - max_down
        if abs(delta) < 1e-14:
            continue
        nh = h.copy()
        nh[i] = hi + delta
        nh[j] = hj - delta
        new_c5 = c5_value(nh)

        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            h, cur_c5 = nh, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = nh.copy(), new_c5

        it += 1
        if it % 1000 == 0:
            T *= decay
        if best_c5 < 0.3808:
            break
    return best_h, best_c5


def generate_thue_morse(n, target):
    """Binary Thue‑Morse word (0/1) projected onto the feasible simplex."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count("1") % 2
    return project_onto_box_simplex(seq, target)

    rng = np.random.default_rng(seed)
    start = time.time()

    n = 70
    target_sum = n / 2.0

    # ------------------------------------------------------------------

        rand = project_onto_box_simplex(rand, target_sum)
        starts.append(rand)

    best_h = None
    best_c5 = float("inf")
    total_budget = budget_s * 0.95

    # ------------------------------------------------------------------
    # Optimise each start in turn, sharing the overall time budget.
    # ------------------------------------------------------------------
    for h0 in starts:
        if time.time() - start >= total_budget:
            break
        remaining = total_budget - (time.time() - start)
        # 70 % of the remaining budget for the continuous phase,
        # 30 % for the discrete refinement.
        grad_time = remaining * 0.7
        swap_time = remaining * 0.3

                best_h, best_c5 = h_ref.copy(), c5_ref

    if best_h is None:

        best_h[: n // 2] = 1.0
        best_h = project_onto_box_simplex(best_h, target_sum)

    best_h = project_onto_box_simplex(best_h, target_sum)
    final_c5 = c5_value(best_h)

    return (best_h.tolist(), final_c5, n)
