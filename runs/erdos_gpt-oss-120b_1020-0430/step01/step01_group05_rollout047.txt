def gradient_c5(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n-1)  # can be negative
    g = np.zeros_like(h)
    if lag >= 0:
        # contribution from i where i+lag < n
        g[:n-lag] -= h[lag:]  # -h[i+lag]
        # contribution from i where i-lag >=0
        g[lag:] -= h[:n-lag]  # -h[i-lag]
    else:
        d = -lag
        g[d:] -= h[:n-d]   # -h[i-d]
        g[:n-d] -= h[d:]   # -h[i+d]
    return g
```

def gradient_descent_optimize(h0, target, max_iter, step_init, start_time, budget):
    h = h0.copy()
    step = step_init
    val = compute_c5(h)
    n = len(h)
    for it in range(max_iter):
        if time.time() - start_time > budget - 1:
            break
        # compute correlation with 1-h
        corr_full = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr_full)
        lag = idx_max - (n - 1)
        # gradient
        g = np.zeros_like(h)
        if lag >= 0:
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
        else:
            d = -lag
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
        # step
        h_new = h - step * g
        h_new = project_to_bounds_and_sum(h_new, target)
        val_new = compute_c5(h_new)
        if val_new < val:
            h = h_new
            val = val_new
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.5
            if step < 1e-6:
                break
    return h

def slsqp_refine(h0, target, dx, bounds, alpha, max_iter, start_time, budget):
    # define smooth objective
    def smooth_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        # to avoid overflow
        # use np.exp(alpha*shifted) safe because shifted <= 0
        return (m + np.log(np.exp(alpha * shifted).sum()) / alpha) * dx
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    # set options with maxiter, but we also need to limit time later.
    try:
        res = scopt.minimize(smooth_obj, h0, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
        h_opt = project_to_bounds_and_sum(res.x, target)
    except Exception:
        h_opt = h0
    return h_opt
```

import time, random, math
import numpy as np
import scipy.optimize as scopt

def project_to_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n ∩ {sum = target} by bisection on the Lagrange multiplier."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        p = np.clip(x - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_to_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_to_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.where((pts >= 0.5) & (pts <= 1.5), 1.0, 0.0)
    return project_to_bounds_and_sum(h, n / 2.0)


def make_double_rectangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    cond = ((pts >= 0.25) & (pts <= 0.75)) | ((pts >= 1.25) & (pts <= 1.75))
    h = np.where(cond, 1.0, 0.0)
    return project_to_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    if old_n == n:
        return project_to_bounds_and_sum(init.astype(float), n / 2.0)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_to_bounds_and_sum(h, n / 2.0)


def gradient_descent_optimize(h0, target, max_iter, step_init, start, budget):
    """Projected gradient descent on the exact C5 objective."""
    h = h0.copy()
    step = step_init
    val = compute_c5(h)
    n = h.shape[0]
    for _ in range(max_iter):
        if time.time() - start > budget - 1:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        g = np.zeros_like(h)
        if lag >= 0:
            g[: n - lag] -= h[lag:]
            g[lag:] -= h[: n - lag]
        else:
            d = -lag
            g[d:] -= h[: n - d]
            g[: n - d] -= h[d:]
        h_new = h - step * g
        h_new = project_to_bounds_and_sum(h_new, target)
        val_new = compute_c5(h_new)
        if val_new < val:
            h, val = h_new, val_new
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.5
            if step < 1e-7:
                break
    return h


def slsqp_refine(h0, target, dx, bounds, alpha, max_iter, start, budget):
    """Smooth–max SLSQP refinement."""
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = corr - m                # ≤0
        return (m + np.log(np.exp(alpha * shifted).sum()) / alpha) * dx
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    try:
        res = scopt.minimize(
            smooth_obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
        )
        h_opt = project_to_bounds_and_sum(res.x, target)
    except Exception:
        h_opt = h0
    return h_opt
