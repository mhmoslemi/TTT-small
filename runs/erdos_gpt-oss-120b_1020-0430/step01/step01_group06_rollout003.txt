def gradient_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    # contributions where both i and i+s valid
    if s >= 0:
        i_range = np.arange(n - s)
        i_plus = i_range + s
        # gradient w.r.t h[i]
        g[i_range] += 1 - h[i_plus]
        # gradient w.r.t h[i_plus] has -h[i]
        g[i_plus] -= h[i_range]
    else:
        s_abs = -s
        i_range = np.arange(s_abs, n)
        i_minus = i_range - s_abs
        # Now i = i_range (i >= s_abs) and i-s = i_minus
        # For shift -s (negative) = -s_abs, we treat same logic: contributions a_i = h_i*(1 - h[i+s]) but s negative => i+s = i - s_abs.

        g[i_range] += 1 - h[i_range - s_abs]  # equals 1 - h[i_minus]
        g[i_minus] -= h[i_range]               # -h_i

def local_optimize(h0, time_limit):
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_local_h = h.copy()
    best_local_c5 = cur_c5
    n = len(h)
    T = max(1e-6, cur_c5 * 0.02)
    decay = 0.9996
    iteration = 0
    start_local = time.time()
    while time.time() - start_local < time_limit:
        # choose move type
        if random.random() < 0.6:  # targeted move
            # compute correlation and worst shift
            corr = np.correlate(h, 1 - h, mode='full')
            s_idx = np.argmax(corr)
            s = s_idx - (n - 1)
            # compute gradient and propose
            g = gradient_for_shift(h, s)
            # selection
            pos_idx = np.where(g > 1e-12)[0]
            neg_idx = np.where(g < -1e-12)[0]
            if pos_idx.size and neg_idx.size:
                i = np.random.choice(pos_idx)
                j = np.random.choice(neg_idx)
                max_delta = min(h[i], 1 - h[j])
                if max_delta > 1e-12:
                    delta = max_delta * (np.random.rand() * 0.5 + 0.25)  # between 0.25*max_delta to 0.75*max_delta maybe.
                    nh = h.copy()
                    nh[i] -= delta
                    nh[j] += delta
                else:
                    nh = None
            else:
                nh = None
        else:
            # random move
            i, j = np.random.choice(n, 2, replace=False)
            hi, hj = h[i], h[j]
            max_up = min(1 - hi, hj)
            max_down = min(hi, 1 - hj)
            if max_up + max_down == 0:
                nh = None
            else:
                delta = (np.random.rand() * (max_up + max_down)) - max_down
                if abs(delta) < 1e-12:
                    nh = None
                else:
                    nh = h.copy()
                    nh[i] = hi + delta
                    nh[j] = hj - delta
        if nh is None:
            continue
        new_c5 = compute_c5(nh)
        # Simulated annealing acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):

            if new_c5 < best_local_c5:
                best_local_c5 = new_c5
                best_local_h = h.copy()
        iteration += 1
        if iteration % 500 == 0:
            T *= decay
    return best_local_h, best_local_c5

import numpy as np
import random
import math
import time

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def gradient_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of overlap for shift s (s = shift index, negative allowed)."""
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if s >= 0:
        i = np.arange(n - s)
        ip = i + s
        g[i] += 1.0 - h[ip]          # d/dh_i term
        g[ip] -= h[i]                # d/dh_{i+s} term
    else:
        sa = -s
        i = np.arange(sa, n)
        im = i - sa
        g[i] += 1.0 - h[im]
        g[im] -= h[i]
    return g


def random_binary_h(n: int) -> np.ndarray:
    """Binary vector with exactly n/2 ones (n assumed even)."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def get_initial_h():
    """Return an initial feasible vector and its length."""
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            if cur > 0:
                h0 *= target / cur
            h0 = np.clip(h0, 0.0, 1.0)
            cur = h0.sum()
            while abs(cur - target) > 1e-9:
                diff = cur - target
                if diff > 0:
                    cand = np.where(h0 > 1e-12)[0]
                    i = random.choice(cand.tolist())
                    delta = min(h0[i], diff)
                    h0[i] -= delta
                else:
                    cand = np.where(h0 < 1 - 1e-12)[0]
                    i = random.choice(cand.tolist())
                    delta = min(1.0 - h0[i], -diff)
                    h0[i] += delta
                cur = h0.sum()
        return h0, n
    except NameError:
        n = 120
        return random_binary_h(n), n


def local_optimize(start_h: np.ndarray, run_time: float) -> tuple:
    """Simulated‑annealing style local search limited by run_time seconds."""
    h = start_h.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    T = max(1e-6, cur_c5 * 0.02)
    decay = 0.9996
    it = 0
    start = time.time()

    while time.time() - start < run_time:
        # decide move type
        if random.random() < 0.55:          # targeted move
            corr = np.correlate(h, 1.0 - h, mode="full")
            s_idx = int(np.argmax(corr))
            s = s_idx - (n - 1)

            g = gradient_for_shift(h, s)
            pos = np.where(g > 1e-12)[0]
            neg = np.where(g < -1e-12)[0]
            if pos.size and neg.size:
                i = pos[np.argmax(g[pos])]
                j = neg[np.argmin(g[neg])]
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta > 1e-12:
                    delta = max_delta * 0.5
                    nh = h.copy()
                    nh[i] -= delta
                    nh[j] += delta
                else:
                    nh = None
            else:
                nh = None
        else:                               # random mass transfer
            i, j = np.random.choice(n, 2, replace=False)
            hi, hj = h[i], h[j]
            max_up = min(1.0 - hi, hj)      # increase i, decrease j
            max_down = min(hi, 1.0 - hj)    # decrease i, increase j
            if max_up + max_down == 0:
                nh = None
            else:
                delta = (random.random() * (max_up + max_down)) - max_down
                if abs(delta) < 1e-14:
                    nh = None
                else:
                    nh = h.copy()
                    nh[i] = hi + delta
                    nh[j] = hj - delta

        if nh is None:
            # nothing changed – just iterate
            it += 1
            if it % 500 == 0:
                T = max(1e-12, T * decay)
            continue

        new_c5 = compute_c5(nh)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_c5 - cur_c5) / T)
            if random.random() < prob:
                accept = True

            h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:

                best_h = h.copy()

        it += 1
        if it % 500 == 0:
            T = max(1e-12, T * decay)

    return best_h, best_c5


def hill_climb(h: np.ndarray) -> np.ndarray:
    """Deterministic refinement using the worst‑shift gradient."""
    n = h.shape[0]
    while True:
        cur_c5 = compute_c5(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        s = int(np.argmax(corr)) - (n - 1)
        g = gradient_for_shift(h, s)
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(g[pos])]
        j = neg[np.argmin(g[neg])]
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta < 1e-12:
            break
        delta = max_delta * 0.5
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        if cur_c5 - new_c5 < 1e-9:
            break
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    np.random.seed(seed)
    start_total = time.time()
    time_budget = budget_s * 0.95          # safety margin

    best_h = base_h.copy()
    best_c5 = compute_c5(best_h)

    # Main multi‑restart loop
    while time.time() - start_total < time_budget:
        remaining = time_budget - (time.time() - start_total)
        if remaining < 1.0:
            break

        # allocate at most 30 s for a single local search run
        run_time = min(30.0, remaining * 0.6)

        # decide starting point for this run
        if best_h is None or random.random() < 0.25:
            h0 = random_binary_h(n)
        else:
            # perturb the current best a little
            h0 = best_h.copy()
            for _ in range(3):
                i, j = np.random.choice(n, 2, replace=False)
                hi, hj = h0[i], h0[j]
                max_up = min(1.0 - hi, hj)
                max_down = min(hi, 1.0 - hj)
                if max_up + max_down == 0:
                    continue
                delta = (random.random() * (max_up + max_down)) - max_down
                if abs(delta) < 1e-14:
                    continue
                h0[i] = hi + delta
                h0[j] = hj - delta

            # tiny projection to fix possible drift
            cur_sum = h0.sum()
            if abs(cur_sum - target_sum) > 1e-9:
                diff = cur_sum - target_sum
                while abs(diff) > 1e-9:
                    if diff > 0:
                        cand = np.where(h0 > 1e-12)[0]
                        i = random.choice(cand.tolist())
                        delta = min(h0[i], diff)
                        h0[i] -= delta
                    else:
                        cand = np.where(h0 < 1 - 1e-12)[0]
                        i = random.choice(cand.tolist())
                        delta = min(1.0 - h0[i], -diff)
                        h0[i] += delta
                    cur_sum = h0.sum()
                    diff = cur_sum - target_sum

        local_best_h, local_best_c5 = local_optimize(h0, run_time)

        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best_h.copy()

    # final deterministic refinement
    best_h = hill_climb(best_h.copy())

    # final feasibility repair (tiny numerical errors)
    err = best_h.sum() - target_sum
    if abs(err) > 1e-8:
        # distribute the error to a random feasible coordinate
        if err > 0:
            cand = np.where(best_h > 1e-12)[0]
        else:
            cand = np.where(best_h < 1 - 1e-12)[0]
        if cand.size:
            i = random.choice(cand.tolist())
            best_h[i] -= err

    final_c5 = compute_c5(best_h)

    return best_h.tolist(), final_c5, n
