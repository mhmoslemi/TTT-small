import time, random, math
import numpy as np
from scipy.optimize import minimize

# ----- utility functions ----------------------------------------------------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n ∩ {sum = target} by bisection on the Lagrange multiplier."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretised."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth version of the C5 objective for the optimizer."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular bump, peak = 1 at centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin_pow(n: int, p: float) -> np.ndarray:
    """h(x)=sin^p(πx/2) on [0,2]; then projected to the feasible set."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** p
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rect(n: int) -> np.ndarray:
    """Rectangular step of width 1 centred at x=1."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = ((pts >= 0.5) & (pts <= 1.5)).astype(float)
    # exact half‑mass is already satisfied for even n, otherwise project
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n and projection."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    rnd = np.random.rand(n)
    return project_onto_bounds_and_sum(rnd, n / 2.0)


# ----- optimisation driver --------------------------------------------------
def optimise_multi_alpha(start_h: np.ndarray, dx: float, bounds, maxiter=300) -> np.ndarray:
    """Run SLSQP with a schedule of smoothing parameters."""
    h = start_h.copy()
    for alpha in (10.0, 50.0, 200.0, 500.0):
        try:
            res = minimize(
                objective_smooth,
                h,
                args=(dx, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
            )
            h = project_onto_bounds_and_sum(res.x, n / 2.0)
        except Exception:
            pass
    return h


def local_refine(h: np.ndarray, dx: float, budget: float, start_time: float) -> np.ndarray:
    """Tiny random pairwise tweaks while time permits."""
    n = h.shape[0]
    target = n / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    while time.time() - start_time < budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            h = cand  # keep the improved candidate for further moves
        if best_val <= 0.3808:
            break
    return best


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with low Erdős C₅ bound."""
    global TARGET_SUM_GLOBAL, n
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    n = 250                     # number of discretisation points
    dx = 2.0 / n
    TARGET_SUM_GLOBAL = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = None
    best_val = np.inf

    # ----- construct a diverse set of seeds ---------------------------------
    seed_vectors = []

    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_rect(n))
    for p in (0.5, 1.0, 2.0, 3.0, 5.0):
        seed_vectors.append(make_sin_pow(n, p))

    # ----- main optimisation loop -------------------------------------------
    restart = 0
    while time.time() - start < budget_s - 0.5:
        # pick a start vector
        if restart < len(seed_vectors):
            h0 = seed_vectors[restart]
        else:
            h0 = random_feasible(n)
        restart += 1

        # multi‑alpha SLSQP optimisation
        h_opt = optimise_multi_alpha(h0, dx, bounds)

        # compute exact C5
        val = compute_c5(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val <= 0.3808:
                break

        # modest local random refinement
        remaining = budget_s - (time.time() - start) - 0.4
        if remaining > 0:
            h_opt = local_refine(h_opt, dx, remaining, start)
            val = compute_c5(h_opt)
            if val < best_val:
                best_val = val
                best_h = h_opt.copy()
                if best_val <= 0.3808:
                    break

    # ----- fallback -----------------------------------------------------------
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    # final safety projection and exact evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, n / 2.0)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
