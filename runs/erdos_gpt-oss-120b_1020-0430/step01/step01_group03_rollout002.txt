import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2 / n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=400):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny fix of the sum (numerical drifts)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Uniform sampling from the bounded simplex Σh_i = n/2, 0≤h_i≤1."""
    target = n / 2.0
    h = np.empty(n)
    remaining = target
    for i in range(n - 1):
        low = max(0.0, remaining - (n - i - 1) * 1.0)
        high = min(1.0, remaining)
        hi = rng.uniform(low, high)
        h[i] = hi
        remaining -= hi
    h[-1] = remaining
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = 0.9995
    min_step = 1e-5
    for _ in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        step = max(step * decay, min_step)
    return best_h, best_c


def make_feasible(vec, rng):
    """Clip to [0,1] and enforce the exact integral constraint."""
    n = vec.size
    vec = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    diff = target - vec.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - vec[i])
            else:
                add = max(diff, -vec[i])
            vec[i] += add
            diff = target - vec.sum()
    return vec


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---------- initial construction ----------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        # aim for 180 points (or 2*n0 if smaller) but keep ≤200
        target_n = min(200, max(180, n0 * 2))
        if target_n != n0:
            old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
            new_x = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h0 = np.interp(new_x, old_x, base)
        else:
            h0 = base.copy()
    else:
        # sine wave with zero mean, automatically satisfies integral=1
        n0 = 120
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        h0 = 0.5 + 0.5 * np.sin(np.pi * x)
        target_n = n0

    n = h0.size
    h0 = make_feasible(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------- Phase 1: many hill‑climbs ----------
    phase1_end = start_time + budget_s * 0.6
    while time.time() < phase1_end:
        # choose a starter
        if rng.random() < 0.6:
            h_start = random_feasible_vector(n, rng)
        elif rng.random() < 0.5:
            # a perturbation of current best
            h_start = perturb_h(best_h, rng, scale=0.3)
        else:
            # sinusoidal seed with random frequencies
            freqs = rng.choice([1, 2, 3, 4, 5], size=3, replace=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            amp = 0.2
            h_start = 0.5
            for f in freqs:
                h_start += amp * np.sin(np.pi * f * x / 2.0)  # period 4/f
            h_start = make_feasible(h_start, rng)

        max_it = int(1500 + 2500 * rng.random())
        h_opt, c_opt = hill_climb(h_start, rng, max_iters=max_it, start_step=0.5)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---------- Phase 2: SLSQP refinements ----------
    betas = [5.0, 20.0, 80.0, 200.0, 500.0]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.9:
            break
        try:
            cand = slsqp_optimize(best_h, beta=beta, max_iter=500)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_h, best_c = cand, c_cand
        except Exception:
            continue

    # ---------- Phase 3: optional finer up‑sampling ----------
    if time.time() - start_time < budget_s * 0.99 and n * 2 <= 200:
        n2 = n * 2
        old_x = np.linspace(0.0, 2.0, n, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, rng)
        h_up, c_up = hill_climb(h_up, rng, max_iters=3500, start_step=0.4)
        for beta in betas[:3]:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta=beta, max_iter=400)
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n2

    # ---------- final clean‑up ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
