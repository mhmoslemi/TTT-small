def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector x onto the box [0,1]^n intersected with the hyperplane sum(x)=target.
    Uses bisection on the Lagrange multiplier.
    """
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    m = np.max(arr)
    return m + np.log(np.exp(alpha * (arr - m)).sum()) / alpha


def objective_smooth_corr(x: np.ndarray) -> float:
    """
    Smooth objective for SLSQP:
    (TARGET_SUM + smooth_max(-corr(h,h))) * dx  ≈  C5
    """
    d = np.correlate(x, x, mode="full")
    sm = smooth_max(-d, ALPHA_SMOOTH)
    return (GLOBAL_TARGET_SUM + sm) * GLOBAL_DX


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM."""
    return np.sum(x) - GLOBAL_TARGET_SUM


def compute_c5_and_shift(h: np.ndarray):
    """
    Return C5 and the integer shift (relative to centre) where the minimum
    self‑correlation occurs.
    """
    d = np.correlate(h, h, mode="full")
    min_idx = int(np.argmin(d))
    d_min = d[min_idx]
    c5 = (GLOBAL_TARGET_SUM - d_min) * GLOBAL_DX
    shift = min_idx - (len(h) - 1)  # can be negative

            current_shift = cand_shift
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val <= 0.3808:
                    break

        iter_cnt += 1

        # occasional diversification
        if iter_cnt % 50000 == 0:
            ii, jj = np.random.randint(0, n, size=2)
            hi, hj = best_h[ii], best_h[jj]
            dmx = min(1.0 - hi, hj)
            dmn = -min(hi, 1.0 - hj)
            if dmx >= dmn:
                d = np.random.uniform(dmn, dmx) * 0.5
                best_h[ii] = hi + d
                best_h[jj] = hj - d
                best_h = project_onto_bounds_and_sum(best_h, target_sum)
                best_val, _ = compute_c5_and_shift(best_h)
                current_h = best_h.copy()
                current_val = best_val
                current_shift = _

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val, _ = compute_c5_and_shift(best_h)

    return (best_h.tolist(), float(best_val), int(n))
