def compute_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Use max for numeric stability:
    max_corr = corr.max()
    # shift indexes: s from -(n-1) .. n-1, idx = s + n -1.
    exp_beta = np.exp(beta * (corr - max_corr))  # subtract max to avoid overflow
    probs = exp_beta / exp_beta.sum()
    grad = np.zeros(n)
    # s = 0
    s_idx = n - 1
    p0 = probs[s_idx]
    grad += p0 * (1 - 2 * h)
    # for s >0
    for s in range(1, n):
        p = probs[n - 1 + s]  # positive shift
        # term1: positions i where i+s in range => i=0..n-1-s:
        # grad[i] += p * (1 - h[i+s]) for i in 0..n-1-s
        # term2: positions i where i-s in range => i=s..n-1:
        # grad[i] -= p * h[i-s] for i=s..n-1
        # Implement with slices
        grad[:n - s] += p * (1 - h[s:])
        grad[s:] -= p * h[:n - s]
        # negative shift s (i.e., shift = -s)
        p_neg = probs[n - 1 - s]
        # For negative shift (s <0) formula is similar, but we can reuse symmetry: gradient for negative shift corresponds to swapping indices.
        # Let's derive: shift = -s, with s>0.
        # Overlap_{-s} = sum_i h_i * (1 - h_{i - s}) where i-s in range => i from s to n-1.
        # gradient w.r.t h_j: term1: (1 - h_{j - s}) if j - s >=0 else 1 (since out-of-range). term2: -h_{j + s} if j + s < n else 0.
        # So we can compute similar slices.
        # For j >= s: term1 = (1 - h[j - s]), else term1 = 1.
        # For j <= n-1-s: term2 = -h[j + s].
        # We'll implement using slices for vectorizable.
        # term1:
        grad[s:] += p_neg * (1 - h[:n - s])   # for j >= s, index maps j-s = i, so h[i] = h[j-s]
        # term2:
        grad[:n - s] -= p_neg * h[s:]
    return grad

def triangular_h(n):
    indices = np.arange(n)
    center = (n - 1) / 2.0
    h = 1.0 - (2.0 / n) * np.abs(indices - center)
    h = np.clip(h, 0.0, 1.0)
    # adjust sum exactly
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # Because of rounding errors, adjust using simple distributed approach.
        # We will increase h by diff/N evenly for all indices not at boundaries, then clip.
        # Simpler: add diff / n to each entry then clip and renormalize iterative.
        # Let's implement simple loop like earlier.
        for _ in range(100):
            cur = h.sum()
            diff = target - cur
            if abs(diff) < 1e-12:
                break
            # Identify indices we can adjust in direction of diff.
            if diff > 0:
                slack = 1.0 - h
                total_slack = slack.sum()
                if total_slack < 1e-12:
                    break
                delta = diff * slack / total_slack
                h = np.clip(h + delta, 0.0, 1.0)
            else:
                slack = h
                total_slack = slack.sum()
                if total_slack < 1e-12:
                    break
                delta = diff * slack / total_slack
                h = np.clip(h + delta, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    # ------------------------------------------------------------------
    # Determine number of sample points
    # ------------------------------------------------------------------
    try:
        n = len(initial_h_values)          # use the size supplied by the harness
    except NameError:
        n = 200                             # a reasonable default if not provided

    # ------------------------------------------------------------------
    # Construct a triangular step function.
    #   h[i] = 1 - (2/n) * |i - (n-1)/2|
    # This satisfies 0 ≤ h ≤ 1 and Σ h = n/2 exactly (up to floating error).
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    h = 1.0 - (2.0 / n) * np.abs(idx - center)
    h = np.clip(h, 0.0, 1.0)

    # Very small numerical drift may break the sum constraint; fix it deterministically.
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # distribute the correction uniformly over non‑saturated entries
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:                     # fallback – adjust any element
            free = np.arange(n)
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)               # re‑clip in case of tiny overshoot

    # ------------------------------------------------------------------
    # Helper: compute C5 for a given vector
    # ------------------------------------------------------------------
    def c5_of(v):
        # linear correlation of v with 1‑v, then scale by dx = 2/n
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(corr.max() * (2.0 / len(v)))

    # ------------------------------------------------------------------
    # (Optional) tiny local refinement – a few gradient steps.
    # This can only improve or keep the value, and costs almost nothing.
    # ------------------------------------------------------------------
    def gradient(v, beta):
        """∇_v  ⁢  log‑sum‑exp(beta·overlap_s) ."""
        n = len(v)
        # over‑laps for all shifts
        ov = np.correlate(v, 1.0 - v, mode="full")
        # soft‑max weights
        max_ov = ov.max()
        w = np.exp(beta * (ov - max_ov))
        w /= w.sum()
        grad = np.zeros_like(v)

        # shift 0 term (special case)
        p0 = w[n - 1]
        grad += p0 * (1.0 - 2.0 * v)

        for s in range(1, n):
            # positive shift s
            p = w[n - 1 + s]
            grad[: n - s] += p * (1.0 - v[s:])
            grad[s:]       -= p * v[: n - s]

            # negative shift -s
            pn = w[n - 1 - s]
            grad[s:]       += pn * (1.0 - v[: n - s])
            grad[: n - s]  -= pn * v[s:]

        return grad

    # simple projected gradient descent
    beta = 0.5
    step = 0.05
    for it in range(200):
        g = gradient(h, beta)
        # make the update have zero mean so the sum stays unchanged
        g -= g.mean()
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)

        cur = h_new.sum()

    # ------------------------------------------------------------------
