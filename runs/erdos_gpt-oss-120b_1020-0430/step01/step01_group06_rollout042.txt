def solve_lp(n, target_sum, time_limit):
    import cvxpy as cp
    import numpy as np

    h = cp.Variable(n)
    t = cp.Variable()

    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum, t >= 0]

    # Build w variables
    for s in range(-(n-1), n):
        if s == 0:
            continue
        if s >= 0:
            i_idx = np.arange(n - s)
            j_idx = i_idx + s
        else:
            i_idx = np.arange(-s, n)
            j_idx = i_idx + s
        m = i_idx.shape[0]
        if m == 0:
            continue
        y = cp.Variable(m)
        # constraints:
        constraints.append(y <= h[i_idx])
        constraints.append(y <= h[j_idx])
        constraints.append(y >= h[i_idx] + h[j_idx] - 1)
        constraints.append(cp.sum(y) >= target_sum - t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    # Solve with time limit
    try:
        prob.solve(solver=cp.GLPK, glpk={'msg_lev': 'GLP_MSG_OFF'}, max_iters=100000, # hmm GLPK doesn't have max_iters
    except Exception:
        prob.solve(solver=cp.ECOS, max_iters=5000)

def build_mip_model(n, init_h=None):
    target = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    cons = [cp.sum(h) == target,
            t >= 0,
            h[0] == 1]  # symmetry breaking
    for s in range(1, n):
        m = n - s
        i_idx = np.arange(m)
        j_idx = i_idx + s
        y = cp.Variable(m)
        cons.append(y <= h[i_idx])
        cons.append(y <= h[j_idx])
        cons.append(y >= h[i_idx] + h[j_idx] - 1)
        cons.append(y >= 0)
        cons.append(cp.sum(y) >= target - t)
    prob = cp.Problem(cp.Minimize(t), cons)
    if init_h is not None:
        h.value = init_h.astype(int)
    return prob, h, t
```

def gradient_descent(h, max_iter, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() - start > time_limit: break
        # compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        s = idx_max - (n - 1)
        # compute gradient for all i
        if s >= 0:
            # indices where i+s valid
            i_plus = np.arange(n - s)
            i_minus = np.arange(s, n)
            # compute g:
            g = np.zeros(n)
            # for i in i_plus: contribution (1 - h[i+s])
            g[i_plus] = (1 - h[i_plus + s])
            # for i in i_minus: subtract h[i - s]
            g[i_minus] -= h[i_minus - s]
        else:
            s_abs = -s
            i_plus = np.arange(s_abs, n)
            i_minus = np.arange(n - s_abs)
            g = np.zeros(n)
            # i + s valid? define s negative: i+s = i - s_abs
            # for i in i_plus: (1 - h[i+s]) = (1 - h[i - s_abs])
            g[i_plus] = (1 - h[i_plus - s_abs])
            # for i in i_minus: subtract h[i - s] = h[i + s_abs]
            g[i_minus] -= h[i_minus + s_abs]
        # choose i_max with max g > 0, j_min with min g <0
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if len(pos)==0 or len(neg)==0:
            # cannot improve further
            break
        i = pos[np.argmax(g[pos])]
        j = neg[np.argmin(g[neg])]
        eps = min(h[i], 1 - h[j]) * 0.5
        if eps < 1e-12:

        h[i] -= eps
        h[j] += eps
        # enforce bounds
        h = np.clip(h, 0, 1)
        # recompute c5
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def project_h(h, target):
    # h already within [0,1]
    cur = h.sum()
    diff = cur - target
    if abs(diff) < 1e-12:
        return h
    # while diff not zero, adjust
    while abs(diff) > 1e-9:
        if diff > 0:
            # need to reduce sum: find indices with h > 0
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            i = random.choice(idx)
            delta = min(h[i], diff)
            h[i] -= delta
            diff -= delta
        else:
            idx = np.where(h < 1 - 1e-12)[0]
            if len(idx) == 0:
                break
            i = random.choice(idx)
            delta = min(1 - h[i], -diff)
            h[i] += delta
            diff += delta
    return h

                    i_neg = np.arange(s, n)
                    g[i_pos] += (1.0 - h[i_pos + s])
                    g[i_neg] -= h[i_neg - s]
                else:
                    s_abs = -s
                    i_pos = np.arange(s_abs, n)
                    i_neg = np.arange(n - s_abs)
                    g[i_pos] += (1.0 - h[i_pos - s_abs])
                    g[i_neg] -= h[i_neg + s_abs]
                pos_idx = np.where(g > 1e-12)[0]
                neg_idx = np.where(g < -1e-12)[0]
                if pos_idx.size and neg_idx.size:
                    i = pos_idx[int(np.argmax(g[pos_idx]))]
                    j = neg_idx[int(np.argmin(g[neg_idx]))]
                    eps_max = min(h[i], 1.0 - h[j])
                    if eps_max > 1e-12:
                        eps = eps_max * random.random()
                        h[i] -= eps
                        h[j] += eps
                        new_c5 = c5_of(h)
                        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                            cur_c5 = new_c5
                            if new_c5 < best_c5:
                                best_h, best_c5 = h.copy(), new_c5

                # ----- c) generic random pair move -----
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                max_up = min(1.0 - hi, hj)
                max_down = min(hi, 1.0 - hj)
                if max_up + max_down > 0:
                    delta = (random.random() * (max_up + max_down)) - max_down
                    if abs(delta) > 1e-14:
                        h[i] = hi + delta
                        h[j] = hj - delta
                        new_c5 = c5_of(h)
                        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                            cur_c5 = new_c5
                            if new_c5 < best_c5:
                                best_h, best_c5 = h.copy(), new_c5

                # temperature cooling (slow)
                T = max(T * decay, 1e-7)

        # early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    # --------------------------------------------------------------
    # Final clean‑up of the best vector (tiny numerical drift fix)
    # --------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-8:
        for _ in range(100):
            k = random.randrange(n)
            if err > 0 and best_h[k] >= err:
                best_h[k] -= err
                break
            if err < 0 and best_h[k] <= 1.0 + err:
                best_h[k] -= err   # err is negative ⇒ addition
                break

    final_c5 = c5_of(best_h)

    return (best_h.tolist(), final_c5, n)
