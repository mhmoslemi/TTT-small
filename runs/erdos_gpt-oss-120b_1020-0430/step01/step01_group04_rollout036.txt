import time
import numpy as np

# ----------------------------------------------------------------------
# Basic evaluation utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function (full correlation)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


# ----------------------------------------------------------------------
# Feasible constructions
# ----------------------------------------------------------------------
def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape (tent) normalised to area 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1] with exact sum = n/2."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to another grid size while preserving the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


# ----------------------------------------------------------------------
# Continuous optimisation with a continuation on the smoothing parameter
# ----------------------------------------------------------------------
def continuous_optimize(start: np.ndarray, n: int, deadline: float,
                        rng: np.random.Generator) -> tuple:
    """Run a two‑stage continuation of L‑BFGS‑B on the smooth max objective."""
    target = n / 2.0
    penalty_weight = 1e3

    def obj(vec, alpha):
        h = np.clip(vec, 0.0, 1.0)
        conv = np.correlate(h, 1.0 - h, mode="full")
        smooth = logsumexp(alpha * conv) / alpha
        pen = penalty_weight * (np.sum(h) - target) ** 2
        return smooth + pen

    # schedule of smoothing strengths
    alphas = [5.0, 15.0, 30.0, 60.0]
    cur = start.copy()
    for a in alphas:
        if time.time() > deadline:
            return None, None
        res = minimize(
            lambda v: obj(v, a),
            cur,
            method="L-BFGS-B",
            bounds=[(0.0, 1.0)] * n,
            options={"maxiter": 300, "ftol": 1e-10, "disp": False},
        )
        if not res.success:
            # fall back to the current point anyway
            cur = res.x
        else:
            cur = res.x
    # final projection onto the exact feasible set
    h = np.clip(cur, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h, compute_c5(h)


# ----------------------------------------------------------------------
# Binary hill‑climbing (swap‑based local search)
# ----------------------------------------------------------------------
def hill_climb_binary(start_bin: np.ndarray, n: int,
                      deadline: float, rng: np.random.Generator,
                      max_iters: int = 500_000) -> tuple:
    """Greedy (with optional plateau) swap‑based improvement for binary vectors."""
    h = start_bin.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    # Pre‑compute list of positions containing 1’s and 0’s each iteration
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # attempt swap
        h[i], h[j] = h[j], h[i]
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_val


def binarize_top_half(h: np.ndarray) -> np.ndarray:
    """Return a binary vector with exactly half of the entries equal to 1 (the largest values)."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    bin_vec = np.zeros_like(h)
    bin_vec[idx] = 1.0
    return bin_vec


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Determine candidate grid sizes (even numbers only)
    # ------------------------------------------------------------------
    candidate_ns = [140, 160, 180, 200, 240]

    # incorporate parent size if available
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    # ------------------------------------------------------------------
    # Continuous optimisation phase
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > time_limit:
            break

        target = n / 2.0
        starts = []

        # 1. Parent (upsampled / same)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # 2. Simple triangular baseline
        starts.append(generate_triangular(n))

        # 3. Random feasible vectors
        for _ in range(8):
            starts.append(random_feasible(n, rng))

        # 4. Perturbations of the triangular shape
        tri = generate_triangular(n)
        for _ in range(4):
            pert = tri + rng.uniform(-0.25, 0.25, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            diff = target - pert.sum()
            if abs(diff) > 1e-12:
                pert += diff / n
                pert = np.clip(pert, 0.0, 1.0)
            starts.append(pert)

        # 5. Perturbations of the parent (if present)
        if parent_h is not None:
            base = upsample(parent_h, n) if parent_n != n else parent_h.copy()
            for _ in range(4):
                pert = base + rng.uniform(-0.2, 0.2, size=n)
                pert = np.clip(pert, 0.0, 1.0)
                diff = target - pert.sum()
                if abs(diff) > 1e-12:
                    pert += diff / n
                    pert = np.clip(pert, 0.0, 1.0)
                starts.append(pert)

        # ------------------------------------------------------------------
        # Optimise each start
        # ------------------------------------------------------------------
        for s in starts:
            if time.time() > time_limit:
                break
            h_opt, c5_opt = continuous_optimize(s, n, time_limit, rng)
            if h_opt is None:
                continue
            if c5_opt < best_c5 - 1e-9:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = n

    # ------------------------------------------------------------------
    # If continuous phase failed to produce anything, fall back to a simple binary block
    # ------------------------------------------------------------------
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Binary refinement (hill climbing) using the best vector found so far
    # ------------------------------------------------------------------
    # a) create a binary version via top‑half rounding
    binary_start = binarize_top_half(best_h)

    # b) allocate remaining time to hill climbing
    hill_deadline = time.time() + (time_limit - time.time()) * 0.5
    # try a few random binary seeds as well
    binary_candidates = [binary_start]
    for _ in range(4):
        binary_candidates.append(binarize_top_half(random_feasible(best_n, rng)))

    for bin_start in binary_candidates:
        if time.time() > hill_deadline:
            break
        h_bin, c5_bin = hill_climb_binary(bin_start, best_n, hill_deadline, rng)
        if c5_bin < best_c5 - 1e-9:
            best_h = h_bin.astype(float)
            best_c5 = c5_bin
            best_n = best_n  # unchanged

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    # enforce feasibility one last time (tiny numeric drift possible)
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
