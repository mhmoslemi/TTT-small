def simulated_annealing(h0, rng, max_iters, init_temp, cooling_factor):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_temp
    n = h.size
    for it in range(max_iters):
        # propose move
        i = rng.integers(n); j = rng.integers(n)
        if i == j: continue
        # direction random: either transfer from i to j or j to i, but unify as signed eps:
        # compute max_transfer (source->target)
        if rng.random() < 0.5:
            src, tgt = i, j
        else:
            src, tgt = j, i
        max_eps = min(h[src], 1 - h[tgt])
        if max_eps <= 0:
            continue
        # scale step size: we could use step based on T or decaying factor
        eps = rng.random() * max_eps * 0.5  # 0..0.5*mxt
        # apply
        h_new = h.copy()
        h_new[src] -= eps
        h_new[tgt] += eps
        # compute c

        delta = c_new - cur_c

            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        # update temperature
        T *= cooling_factor

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Create a random feasible vector (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    # random mixing
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def anneal_optimize(h0, rng, deadline, start_temp=0.02):
    """Simple SA that preserves the sum exactly (mass transfer + swaps)."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c

    n = h.size
    # number of iterations is generous; we stop by time.
    max_iters = 10 ** 7
    # exponential cooling to ~1e-6
    decay = (1e-6 / start_temp) ** (1.0 / max_iters)
    T = start_temp

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ---- propose a move ------------------------------------------------
        if rng.random() < 0.1:          # swap move (larger jump)
            i, j = rng.choice(n, size=2, replace=False)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:                           # transfer move
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # random direction
            if rng.random() < 0.5:
                src, tgt = i, j
            else:
                src, tgt = j, i
            max_eps = min(h[src], 1.0 - h[tgt])
            if max_eps <= 0.0:
                continue
            # step size adapts with temperature
            scale = min(1.0, T * 5.0)   # larger when hot, tiny when cold
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:

            h_new[src] -= eps

        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, cur_c = h_new, c_new
            if cur_c < best_c:

        # ---- cool -----------------------------------------------------------
        T *= decay
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0, beta, max_iter=300):
    """Refine a feasible vector using SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny fix for any residual sum error
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
