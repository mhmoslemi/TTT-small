        if i == j: continue
        hi = h[i]; hj = h[j]

import time, random
import numpy as np

# ---------- helper utilities ----------

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while keeping sum(x)=target using bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape, peak at the centre."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(pi x / 2) on [0,2] (already unit integral)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_to_n(arr: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation from current length to n points."""
    old_n = arr.shape[0]
    if old_n == n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, arr)
    return h


def fill_delta(p: int, d: float, h: np.ndarray, b: np.ndarray,
               t_arr: np.ndarray, out: np.ndarray) -> None:
    """Write the correlation change caused by adding d to h[p] (linear, zero‑pad)."""
    out.fill(0.0)
    N = h.shape[0]
    idx = p + t_arr
    mask = (idx >= 0) & (idx < N)
    out[mask] = b[idx[mask]]
    idx2 = p - t_arr
    mask2 = (idx2 >= 0) & (idx2 < N)
    out[mask2] -= h[idx2[mask2]]
    out *= d


def refine_h(h: np.ndarray, time_budget: float) -> tuple:
    """
    Hill‑climbing refinement for a fixed n.
    Returns (best_h, best_val, final_corr).
    """
    N = h.shape[0]
    dx = 2.0 / N
    target_sum = N / 2.0
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    best_val = corr.max() * dx
    best_h = h.copy()

    t_arr = np.arange(-N + 1, N, dtype=np.int32)          # length 2N‑1
    shift0 = N - 1                                        # index of shift 0
    delta_i = np.empty_like(corr)
    delta_j = np.empty_like(corr)

    end_time = time.time() + time_budget
    no_improve = 0

    while time.time() < end_time:
        i = np.random.randint(0, N)
        j = np.random.randint(0, N - 1)
        if j >= i:
            j += 1

        hi, hj = h[i], h[j]
        dmin = max(-hi, hj - 1.0)
        dmax = min(1.0 - hi, hj)
        if dmax <= dmin:
            continue

        # try a few candidate deltas (both extremes and occasional random)
        candidates = [dmin, dmax]
        if random.random() < 0.2:
            candidates.append(random.uniform(dmin, dmax))

        improved = False
        for d in candidates:
            if abs(d) < 1e-14:
                continue
            fill_delta(i, d, h, b, t_arr, delta_i)
            fill_delta(j, -d, h, b, t_arr, delta_j)
            new_corr = corr + delta_i + delta_j
            # cross‑terms from the bilinear product
            new_corr[shift0] -= 2.0 * d * d
            idx_ij = j - i + shift0
            idx_ji = i - j + shift0
            new_corr[idx_ij] += d * d
            new_corr[idx_ji] += d * d

            new_val = new_corr.max() * dx
            if new_val < best_val - 1e-14:
                # Accept move
                h[i] = hi + d
                h[j] = hj - d
                b = 1.0 - h
                corr = new_corr
                best_val = new_val
                best_h = h.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        # occasional random perturbation to escape plateaus
        if no_improve >= 5000:
            noise = (np.random.rand(N) - 0.5) * 0.02
            h_cand = np.clip(h + noise, 0.0, 1.0)
            h_cand = project_onto_bounds_and_sum(h_cand, target_sum)
            b_cand = 1.0 - h_cand
            corr_cand = np.correlate(h_cand, b_cand, mode="full")
            val_cand = corr_cand.max() * dx
            if val_cand < best_val:
                h, b, corr, best_val, best_h = (
                    h_cand,
                    b_cand,
                    corr_cand,
                    val_cand,
                    h_cand.copy(),
                )
            no_improve = 0

        if best_val <= 0.3808:
            break

    return best_h, best_val, corr

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    total_budget = budget_s

    # ----- initial discretisation size -----
    n0 = 250
    seed_vecs = []

    if "initial_h_values" in globals():
        try:
            seed_vecs.append(
                upsample_to_n(np.array(initial_h_values, dtype=float), n0)
            )
        except Exception:
            pass
    # deterministic analytic starts
    seed_vecs.append(make_triangle(n0))
    seed_vecs.append(make_sin2(n0))

    # allocate fractions of the total budget for each seed
    fractions = [0.6, 0.2, 0.2]
    best_h_global = None
    best_val_global = np.inf

    for idx, init_h in enumerate(seed_vecs):
        elapsed = time.time() - start
        if elapsed >= total_budget - 0.5:
            break
        remaining = total_budget - elapsed
        # give a share of the remaining time to this seed
        frac = fractions[idx] / sum(fractions[idx:])
        alloc = remaining * frac
        h0 = project_onto_bounds_and_sum(init_h, n0 / 2.0)
        h_opt, val_opt, _ = refine_h(h0, alloc)
        if val_opt < best_val_global:
            best_val_global = val_opt
            best_h_global = h_opt.copy()
        if best_val_global <= 0.3808:
            break

    # ----- upscale and continue refinement -----
    # progressive up‑sampling (250 → 512 → 1024) while time permits
    target_sizes = [512, 1024]
    for new_n in target_sizes:
        if best_h_global is None:
            break
        if time.time() - start >= total_budget - 0.5:
            break
        # up‑sample preserving shape
        h_up = upsample_to_n(best_h_global, new_n)
        h_up = project_onto_bounds_and_sum(h_up, new_n / 2.0)
        # allocate a portion of the remaining budget
        remaining = total_budget - (time.time() - start)
        if remaining <= 0.2:

        alloc = remaining * 0.8
        h_opt, val_opt, _ = refine_h(h_up, alloc)
        if val_opt < best_val_global:
            best_val_global = val_opt
            best_h_global = h_opt.copy()
        if best_val_global <= 0.3808:
            break

    # final safety projection and exact evaluation
    if best_h_global is None:
        best_h_global = make_triangle(n0)
    best_h_global = np.clip(best_h_global, 0.0, 1.0)
    best_h_global = project_onto_bounds_and_sum(
        best_h_global, best_h_global.shape[0] / 2.0
    )
    final_val = compute_c5(best_h_global)

    return (best_h_global.tolist(), float(final_val), int(best_h_global.shape[0]))
