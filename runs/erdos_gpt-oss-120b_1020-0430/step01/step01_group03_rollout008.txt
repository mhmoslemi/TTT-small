            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def hill_climb(h, rng, max_iters=2000, start_step=0.5, target_step_factor=0.7, target_step_min=1e-4, use_targeted=False):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        # if use_targeted and iteration%10==0: use targeted move
        if use_targeted and (it % 10 == 0):
            # compute lag with max overlap
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            lag = max_idx - (n - 1)
            # if lag positive and < n, do directed move
            if lag != 0 and -n < lag < n:
                # choose i such that i+lag within bounds
                if lag > 0:
                    i = rng.integers(0, n - lag)
                else:
                    i = rng.integers(-lag, n)
                j = i + lag
                # try moving some epsilon from higher to lower
                if h[i] > h[j]:
                    max_eps = min(h[i], 1.0 - h[j])
                    direction = -1  # move from i to j
                else:
                    max_eps = min(h[j], 1.0 - h[i])
                    direction = 1   # move from j to i
                if max_eps > 0:
                    eps = rng.random() * max_eps * step
                    new_h = h.copy()
                    if direction == -1:
                        new_h[i] -= eps
                        new_h[j] += eps
                    else:
                        new_h[i] += eps
                        new_h[j] -= eps
                else:
                    new_h = None
            else:
                new_h = None
        else:
            # regular random perturb
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            min_eps = -min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = (rng.random() * 2 - 1) * max_eps * step  # random in [-max_eps, max_eps]
            eps = np.clip(eps, min_eps, max_eps)
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        if new_h is None:
            continue
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        else:
            # optionally accept if improvement small relative to step? Not needed.
            # Optionally probabilistic acceptance can help escape.
            pass
        # shrinking step
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * target_step_factor, target_step_min)
    return best_h, best_c

    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    # compute schedule for shrinking step: maybe after each quarter of iterations.
    for it in range(max_iters):
        if use_targeted and (it % 10 == 0):
            # target move
            corr = np.correlate(h, 1 - h, mode='full')
            idx = np.argmax(corr)
            lag = idx - (n - 1)
            if lag != 0:
                if lag > 0:
                    i = rng.integers(0, n - lag)
                else:
                    i = rng.integers(-lag, n)
                j = i + lag
                # decide direction
                if h[i] > h[j]:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps > 0:
                        eps = rng.random() * max_eps * step
                        new_h = h.copy()
                        new_h[i] -= eps
                        new_h[j] += eps
                    else:
                        continue
                elif h[j] > h[i]:
                    max_eps = min(h[j], 1.0 - h[i])

                        eps = rng.random() * max_eps * step
                        new_h = h.copy()

                        new_h[j] -= eps

                        continue
                else:
                    continue

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce integral exactly via redistribution
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = x.copy()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=200):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce exact sum (possible numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = x.copy()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def make_feasible(h, target, rng):
    """Clip to [0,1] and adjust the sum to exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, h.size)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_binary_vector(n, rng):
    """Binary feasible vector with exactly n/2 ones."""
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5, use_targeted=True):
    """
    Simple hill‑climb (accept only improvements).  If use_targeted,
    every 10‑th iteration performs a move directed at the current worst lag.
    """
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    shrink_every = max_iters // 8 + 1

    for it in range(max_iters):
        if use_targeted and (it % 10 == 0):
            # targeted move based on the lag with maximal overlap
            corr = np.correlate(h, 1 - h, mode="full")
            idx = int(np.argmax(corr))
            lag = idx - (n - 1)
            if lag != 0:
                if lag > 0:
                    i = rng.integers(0, n - lag)
                else:
                    i = rng.integers(-lag, n)
                j = i + lag
                # move mass from larger to smaller entry
                if h[i] > h[j]:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps > 0:
                        eps = rng.random() * max_eps * step
                        new_h = h.copy()
                        new_h[i] -= eps
                        new_h[j] += eps
                        candidate = new_h
                    else:
                        candidate = None
                elif h[j] > h[i]:
                    max_eps = min(h[j], 1.0 - h[i])
                    if max_eps > 0:
                        eps = rng.random() * max_eps * step
                        new_h = h.copy()
                        new_h[i] += eps
                        new_h[j] -= eps
                        candidate = new_h
                    else:
                        candidate = None
                else:
                    candidate = None
            else:
                candidate = None
        else:
            # random transfer preserving total mass
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            min_eps = -min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = (rng.random() * 2.0 - 1.0) * max_eps * step
            eps = np.clip(eps, min_eps, max_eps)
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            candidate = new_h

        if candidate is None:
            continue
        c = compute_c5(candidate)
        if c < best_c:
            best_c = c
            best_h = candidate.copy()
            h = candidate
        # shrink step size periodically
        if (it + 1) % shrink_every == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
