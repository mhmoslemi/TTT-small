def neighbor_sum(h, k):
    n = h.shape[0]
    w = np.zeros(n)
    if k >= 0:
        # i+k valid for i=0..n-1-k
        w[:n - k] += h[k:]
        # i-k valid for i=k..n-1
        w[k:] += h[:n - k]
    else:
        k = -k
        # i+k (i+(-k)) = i - k valid for i=k..n-1
        w[k:] += h[:n - k]
        # i-k (i - (-k)) = i + k valid for i=0..n-1-k
        w[:n - k] += h[k:]
    return w

def _greedy_refine(h, max_seconds, max_step=0.2):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_h = h.copy()
    cur_c = best_c
    while time.time() - start < max_seconds:
        # compute overlap array (full)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        # find worst shift
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)   # shift
        # compute w_i
        w = _neighbor_sum(cur_h, k)
        # find candidates
        # get top indices by w among those not at 1
        cand_i = np.where(cur_h < 1.0 - 1e-12)[0]
        cand_j = np.where(cur_h > 1e-12)[0]
        if cand_i.size == 0 or cand_j.size == 0:
            break
        # sort candidates by w
        # top few highest w among cand_i
        top_i = cand_i[np.argsort(-w[cand_i])][:5]
        # bottom few lowest w among cand_j
        low_j = cand_j[np.argsort(w[cand_j])][:5]
        improved = False
        for i in top_i:
            for j in low_j:
                if i == j:
                    continue
                # compute possible delta
                delta = min(max_step, 1.0 - cur_h[i], cur_h[j])
                if delta <= 1e-12:
                    continue
                # apply
                cur_h[i] += delta
                cur_h[j] -= delta
                new_c = _c5_metric(cur_h)
                if new_c < cur_c - 1e-10:
                    # accept
                    cur_c = new_c
                    improved = True
                    if new_c < best_c:
                        best_c = new_c
                        best_h = cur_h.copy()
                    break
                else:
                    # revert
                    cur_h[i] -= delta
                    cur_h[j] += delta
            if improved:
                break
        if not improved:
            # try random small move as fallback
            i, j = random.sample(range(n), 2)
            hi, hj = cur_h[i], cur_h[j]
            # feasible interval for delta (i receives)
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low < high:
                delta = random.uniform(low, high) * 0.1
                if abs(delta) > 1e-12:
                    cur_h[i] = hi + delta
                    cur_h[j] = hj - delta
                    new_c = _c5_metric(cur_h)
                    if new_c < cur_c:
                        cur_c = new_c
                        if new_c < best_c:
                            best_c = new_c
                            best_h = cur_h.copy()
                    else:
                        cur_h[i] = hi
                        cur_h[j] = hj
    return best_h, best_c

def _c5_metric(h):
    """C5 value for a step function sampled by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _neighbor_sum(h, k):
    """w_i = h_{i+k}+h_{i-k} with zero outside the interval."""
    n = h.shape[0]
    w = np.zeros(n, dtype=np.float64)
    if k >= 0:
        w[: n - k] += h[k:]
        w[k:] += h[: n - k]
    else:
        k = -k
        w[k:] += h[: n - k]
        w[: n - k] += h[k:]
    return w
