def project(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    diff = target_sum - current
    # adjust greedily:
    if abs(diff) < 1e-12:
        return v
    # If diff > 0: need to add mass; choose random indices with capacity.
    # We'll add to indices in random order.
    inds = np.where(v < 1)[0]
    np.random.shuffle(inds)
    for i in inds:
        capacity = 1.0 - v[i]
        add = min(capacity, diff)
        v[i] += add
        diff -= add
        if abs(diff) < 1e-12:

    if diff > 1e-12:

            capacity = v[i]
            sub = min(capacity, -diff)

            diff += sub

            v[i] -= sub
            diff += sub

            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)          # shift giving the max overlap
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:
                s_abs = -s
                i_inds = np.arange(s_abs, n)
                p_inds = i_inds - s_abs
            contrib = h[i_inds] * (1.0 - h[p_inds])
            order = np.argsort(-contrib)    # descending
            moved = False
            for k in order:
                i = i_inds[k]
                p = p_inds[k]
                eps_max = min(h[i], 1.0 - h[p])
                if eps_max > 1e-12:
                    eps = eps_max                # full possible transfer
                    nh = h.copy()
                    nh[i] -= eps
                    nh[p] += eps
                    new_c5 = evaluate(nh)
                    moved = True
                    break
            if not moved:
                # fallback random two‑coordinate move
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                up = min(1.0 - hi, hj)
                down = min(hi, 1.0 - hj)
                if up + down == 0:
                    continue
                delta = random.random() * (up + down) - down
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_c5 = evaluate(nh)
        else:
            # ----- random two‑coordinate move -----
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up + down == 0:
                continue
            delta = random.random() * (up + down) - down
            if abs(delta) < 1e-14:
                continue
            nh = h.copy()
            nh[i] = hi + delta
            nh[j] = hj - delta
            new_c5 = evaluate(nh)

        # ---------- acceptance ----------
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

        it += 1
        if it % 200 == 0:
            T = max(1e-8, T * decay)

        # occasional jitter around the best solution to escape plateaus
        if it % 2000 == 0:
            h = best_h.copy()
            for _ in range(5):
                i, j = random.sample(range(n), 2)
                eps = (random.random() - 0.5) * 0.02
                if eps > 0:
                    eps = min(eps, 1.0 - h[i], h[j])
                else:
                    eps = -min(-eps, h[i], 1.0 - h[j])
                h[i] += eps
                h[j] -= eps
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                if diff > 0:
                    idx = np.where(h < 1.0 - 1e-12)[0][0]
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                else:
                    idx = np.where(h > 1e-12)[0][0]
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
            cur_c5 = evaluate(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # -------- final greedy refinement (only improving moves) --------
    for _ in range(5000):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx_max = int(np.argmax(corr))
        s = idx_max - (n - 1)
        if s >= 0:
            i_inds = np.arange(n - s)
            p_inds = i_inds + s
        else:
            s_abs = -s
            i_inds = np.arange(s_abs, n)
            p_inds = i_inds - s_abs
        contrib = best_h[i_inds] * (1.0 - best_h[p_inds])
        k = int(np.argmax(contrib))
        i = i_inds[k]
        p = p_inds[k]
        eps_max = min(best_h[i], 1.0 - best_h[p])
        if eps_max <= 1e-12:
            break
        nh = best_h.copy()
        nh[i] -= eps_max
        nh[p] += eps_max
        new_c5 = evaluate(nh)
        if new_c5 < best_c5 - 1e-12:
            best_h = nh
            best_c5 = new_c5
        else:
            break

    # ---- final feasibility enforcement ----
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            idx = np.where(best_h < 1.0 - 1e-12)[0][0]
            best_h[idx] += diff
        else:
            idx = np.where(best_h > 1e-12)[0][0]
            best_h[idx] += diff   # diff negative ⇒ subtraction
    final_c5 = evaluate(best_h)

    return (best_h.tolist(), final_c5, n)
