def _anneal(start_h, max_seconds, T0=0.05, T_end=1e-6, max_step=0.3):
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    t0 = time.time()
    while time.time() - t0 < max_seconds:
        # compute progress
        elapsed = time.time() - t0
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        # optionally adapt step size with temperature
        step = max_step * (T / T0)  # decaying step
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)   # minimal delta to keep bounds
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # restrict step size
        low_cap = max(low, -step)
        high_cap = min(high, step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)
        # apply move on a copy
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        new_c = _c5_metric(h_new)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
    return best_h, best_c

import time, random, math
import numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """C₅ value for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _initial_vector(target_n: int) -> np.ndarray:
    """Build a feasible start vector of length target_n."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        n0 = base.size
        if n0 == target_n:
            h = base.copy()
        else:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h = np.interp(x, x0, base)
    else:
        h = np.full(target_n, 0.5, dtype=np.float64)

    target = target_n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur                       # simple scaling
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(target_n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _anneal(start_h: np.ndarray, max_seconds: float,
            T0: float = 0.06, T_end: float = 1e-6,
            max_step: float = 0.30, focus_prob: float = 0.12) -> tuple:
    """Simulated‑annealing improvement."""
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        step = max_step * (1.0 - progress) + 1e-6

        if random.random() < focus_prob:
            # four‑point move aimed at the worst shift
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)
            if shift >= 0:
                i = random.randint(0, n - 1 - shift)
            else:
                i = random.randint(-shift, n - 1)
            i2 = i + shift
            donors = list(set(range(n)) - {i, i2})
            if len(donors) < 2:
                continue
            j, k = random.sample(donors, 2)
            # feasible delta for the four‑point exchange
            max_delta = min(step,
                            1.0 - h[i],
                            h[j],
                            1.0 - h[i2],
                            h[k])
            if max_delta <= 0.0:
                continue
            delta = random.uniform(0.0, max_delta)

            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            h_new[i2] += delta
            h_new[k] -= delta
        else:
            # simple two‑point exchange
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            low_cap = max(low, -step)
            high_cap = min(high, step)
            if low_cap < high_cap:
                delta = random.uniform(low_cap, high_cap)
            else:
                delta = random.uniform(low, high)

            h_new = h.copy()
            h_new[i] = hi + delta
            h_new[j] = hj - delta

        new_c = _c5_metric(h_new)
        dC = new_c - cur_c
        if dC < 0.0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
    return best_h, best_c


def _refine_worst_shift(h: np.ndarray, time_limit: float) -> tuple:
    """Deterministic refinement targeting the shift with maximal overlap."""
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        if shift >= 0:
            i = random.randint(0, n - 1 - shift)
        else:
            i = random.randint(-shift, n - 1)
        i2 = i + shift
        donors = list(set(range(n)) - {i, i2})
        if len(donors) < 2:

        new_c = _c5_metric(new_h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
            best_h = new_h
    return best_h, best_c


def _fix_sum(h: np.ndarray) -> np.ndarray:
    """Enforce exact sum = n/2 without breaking box constraints."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for idx in np.random.permutation(n):
        if 0.0 <= h[idx] + diff <= 1.0:
            h[idx] += diff
            break
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C₅ bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # ------------------------------------------------------------------ #
    # 1. Choose discretisation size (even) and build a first candidate.
    # ------------------------------------------------------------------ #
    n = 400
    if n % 2:
        n += 1
    h0 = _initial_vector(n)

    # ------------------------------------------------------------------ #
    # 2. Primary annealing (takes the bulk of the time budget).
    # ------------------------------------------------------------------ #
    primary_time = total_budget * 0.62
    best_h, best_c = _anneal(h0, primary_time)

    # ------------------------------------------------------------------ #
    # 3. Random restarts (binary starts) while time permits.
    # ------------------------------------------------------------------ #
    while True:
        elapsed = time.time() - start_all
        remaining = total_budget - elapsed
        if remaining < 12.0:                     # not enough time for another run
            break

        h_start = np.zeros(n, dtype=np.float64)

        cand_h, cand_c = _anneal(h_start, slice_time)

    # ------------------------------------------------------------------ #

    # ------------------------------------------------------------------ #
    remaining = total_budget - (time.time() - start_all)

    # ------------------------------------------------------------------ #
    best_h = _fix_sum(best_h)

    try:
        final_c = evaluate_erdos_solution(best_h)   # provided helper, if any
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
