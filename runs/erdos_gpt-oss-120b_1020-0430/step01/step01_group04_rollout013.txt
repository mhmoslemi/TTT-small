def compute_c5_fast(h: np.ndarray) -> float:
    """Exact max overlap for a step function given by samples h."""
    n = h.shape[0]
    # convolution via FFT (zero‑padding)
    L = 2 * n - 1
    M = 1 << (L - 1).bit_length()          # next power of two
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    conv = np.fft.irfft(H * G, n=M)[:L]
    dx = 2.0 / n
    return float(conv.max() * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    # stable log‑sum‑exp
    m = np.max(conv)
    return float((np.log(np.sum(np.exp(alpha * (conv - m)))) + alpha * m) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular profile h(x)=min(x,2−x) sampled on n points and renormalised."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector with values in [0,1] and exact sum n/2."""
    target = n / 2.0
    a = rng.random(n)
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
        # tiny residual fix
        diff2 = target - a.sum()
        if abs(diff2) > 1e-12:
            idx = rng.integers(n)
            a[idx] = np.clip(a[idx] + diff2, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, keeping the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def slsqp_refine(start: np.ndarray, n: int, target_sum: float,
                 deadline: float) -> np.ndarray | None:
    """Two‑stage SLSQP minimisation of a smooth surrogate of C₅."""
    if time.time() > deadline:
        return None

    def eq_constr(x):
        return np.sum(x) - target_sum

    cons = {"type": "eq", "fun": eq_constr}
    bounds = [(0.0, 1.0)] * n
    # first stage – gentle smoothing
    res1 = minimize(lambda x: softmax_c5(x, alpha=5.0),
                    start, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 400, "ftol": 1e-9, "disp": False})
    if not res1.success or time.time() > deadline:
        return None
    # second stage – sharper smoothing
    res2 = minimize(lambda x: softmax_c5(x, alpha=30.0),
                    res1.x, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 400, "ftol": 1e-9, "disp": False})
    if not res2.success or time.time() > deadline:
        return None
    h_opt = np.clip(res2.x, 0.0, 1.0)
    return h_opt


# ------------------------------------------------------------
# Main optimisation driver
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1 that minimises
    C₅ = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    time_limit = t0 + budget_s * 0.95   # safety margin

    # try to reuse a parent construction if supplied
    parent_h = None
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    # candidate discretisations (even sizes give a perfectly binary feasible point)
    candidate_ns = [80, 100, 120, 160, 180, 200, 240, 300]

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in candidate_ns:
        if time.time() > time_limit:
            break
        target_sum = n / 2.0

        # ---- build a collection of starting points ----
        starts = []

        # parent (upsampled if needed)
        if parent_h is not None:
            if parent_h.shape[0] == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # triangular profile
        starts.append(generate_triangular(n))

        # several random feasible vectors
        for _ in range(6):
            starts.append(random_start(n, rng))

        # perturbed triangular shapes
        for _ in range(3):
            pert = generate_triangular(n) + rng.uniform(-0.1, 0.1, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = pert * (target_sum / pert.sum())
            diff = target_sum - pert.sum()
            if abs(diff) > 1e-12:
                pert += diff / n
                pert = np.clip(pert, 0.0, 1.0)
            starts.append(pert)

        # ---- optimise each start ----
        for s in starts:
            if time.time() > time_limit:
                break

            # local SLSQP refinement (fast surrogate)
            h_local = slsqp_refine(s, n, target_sum, time_limit)
            if h_local is not None:
                cur_h = h_local
                cur_c5 = compute_c5_fast(cur_h)
                # record if improvement
                if cur_c5 < best_c5 - 1e-9:
                    best_h, best_c5, best_n = cur_h.copy(), cur_c5, n
            else:
                cur_h = s.copy()
                cur_c5 = compute_c5_fast(cur_h)

            # ---- simulated annealing / greedy descent ----
            max_iter = 20000
            T0 = 0.001
            decay = 0.99995
            step_factor = 0.5

            for it in range(max_iter):
                if time.time() > time_limit:
                    break

                # temperature schedule
                T = T0 * (decay ** it)

                # propose a mass‑transfer move
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                max_transfer = min(cur_h[i], 1.0 - cur_h[j])
                if max_transfer <= 0.0:
                    continue
                delta = max_transfer * step_factor * rng.random()
                if delta == 0.0:
                    continue
                h_new = cur_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_c5 = compute_c5_fast(h_new)

                # accept if better, otherwise Metropolis criterion
                if new_c5 < cur_c5 - 1e-12:
                    cur_h, cur_c5 = h_new, new_c5
                else:
                    if T > 0.0 and rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                        cur_h, cur_c5 = h_new, new_c5

                # occasional local refinement to escape plateaus
                if (it + 1) % 5000 == 0:
                    refined = slsqp_refine(cur_h, n, target_sum, time_limit)
                    if refined is not None:
                        refined_c5 = compute_c5_fast(refined)
                        if refined_c5 < cur_c5:
                            cur_h, cur_c5 = refined, refined_c5

            # final update for this start
            if cur_c5 < best_c5 - 1e-9:
                best_h, best_c5, best_n = cur_h.copy(), cur_c5, n

    # ----- fallback in the unlikely case nothing succeeded -----
    if best_h is None:
        n = 100
        best_h = random_start(n, rng)
        best_c5 = compute_c5_fast(best_h)
        best_n = n

    # enforce the integral constraint exactly (tiny numerical drift possible)
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    # final accurate bound
    final_c5 = compute_c5_fast(best_h)

    return best_h, final_c5, best_n
