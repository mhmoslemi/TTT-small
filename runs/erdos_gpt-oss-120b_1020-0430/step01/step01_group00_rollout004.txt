import time

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, target: float = None,
                         tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=target (default n/2) by water‑filling."""
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Calculate C5 for a discrete step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


# ----------------------------------------------------------------------
# Core optimizer (smooth‑max gradient descent)
# ----------------------------------------------------------------------
def _optimize_h(start_h: np.ndarray, n: int, time_left) -> tuple:
    """Run smooth‑max gradient descent from start_h."""
    h = _project_to_feasible(start_h, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    max_iter = 4000
    lr_start, lr_end = 0.7, 0.02          # learning‑rate schedule
    alpha_start, alpha_end = 0.3, 80.0   # soft‑max sharpness schedule

    for it in range(max_iter):
        if time_left() < 0.8:
            break

        # correlation and current objective
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_c5 = corr.max() * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            # early exit if already excellent
            if best_c5 < 0.3808:
                break

        # smooth‑max weights
        max_corr = corr.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        w = np.exp(alpha * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via convolutions (linear, zero‑padded)
        f = 1.0 - h
        w_rev = w[::-1]

        conv_f_rev = np.convolve(w_rev, f, mode="full")
        conv_h = np.convolve(w, h, mode="full")
        off = n - 1
        term1 = conv_f_rev[off:off + n]
        term2 = conv_h[off:off + n]
        grad = term1 - term2

        # learning‑rate schedule
        lr = lr_start + (lr_end - lr_start) * (it / max_iter)

        # gradient step + projection
        h = h - lr * grad
        h = _project_to_feasible(h, n)

    return best_h, best_c5
