def gradient_smooth(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    maxval = np.max(alpha * corr)
    w = np.exp(alpha * corr - maxval)
    w /= w.sum()
    # w_rev
    w_rev = w[::-1]
    conv_w_h = np.convolve(w, h, mode='full')
    conv_rev_w_h = np.convolve(w_rev, h, mode='full')
    offset = len(h) - 1
    grad = 1.0 - conv_rev_w_h[offset:offset+len(h)] - conv_w_h[offset:offset+len(h)]
    return grad

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-9,
                                 max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ----------------------------------------------------------------------
# Random feasible point generation
# ----------------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Exact C5 evaluation (max correlation)
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective (no dx factor, absorbed in step)
# ----------------------------------------------------------------------
def _grad_smooth_max_corr(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of  α⁻¹ log ∑exp(α·corr)  w.r.t. h (corr = h⋆(1−h))."""
    n = h.shape[0]
    # unscaled correlation (dx omitted – constant factor)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # stabilized softmax weights
    max_val = np.max(alpha * corr)
    exp_term = np.exp(alpha * corr - max_val)
    w = exp_term / exp_term.sum()                     # shape (2n‑1,)
    w_rev = w[::-1]

    # convolutions (full mode)
    conv_w_h = np.convolve(w, h, mode="full")
    conv_rev_w_h = np.convolve(w_rev, h, mode="full")
    offset = n - 1
    grad = 1.0 - conv_rev_w_h[offset:offset + n] - conv_w_h[offset:offset + n]
    return grad


# ----------------------------------------------------------------------
# Projected Gradient Descent with continuation on alpha
# ----------------------------------------------------------------------
def _pgd_continuation(h0: np.ndarray, target: float,
                      alphas, it_per_alpha: int,
                      step0: float, decay: float) -> np.ndarray:
    h = h0.copy()
    for alpha in alphas:
        step = step0
        for _ in range(it_per_alpha):
            g = _grad_smooth_max_corr(h, alpha)
            h -= step * g
            h = _project_onto_capped_simplex(h, target)
            step *= decay
    return h


# ----------------------------------------------------------------------
# Main routine
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Determine discretisation size
    # --------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        init_vec = None

    n = len(init_vec) if init_vec is not None else 200
    target_sum = n / 2.0                   # because dx = 2/n
    dx = 2.0 / n

    # --------------------------------------------------------------
    # Baseline candidate
    # --------------------------------------------------------------
    if init_vec is not None:
        h_best = init_vec.copy()
        h_best = _project_onto_capped_simplex(h_best, target_sum)
        best_val = _compute_c5(h_best, dx)
    else:
        h_best = None
        best_val = float("inf")

    # --------------------------------------------------------------
    # Hyper‑parameters for the continuation method
    # --------------------------------------------------------------
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0]          # growing sharpness
    it_per_alpha = 120                               # modest iterations per stage
    step0_choices = [0.6, 0.4, 0.3, 0.2]             # allow variability
    decay = 0.996

    # --------------------------------------------------------------
    # Random‑restart loop (time‑aware)
    # --------------------------------------------------------------
    restart = 0

        best_val = _compute_c5(h_best, dx)

    # --------------------------------------------------------------
    # Return result
    # --------------------------------------------------------------
    return (h_best, float(best_val), n)
