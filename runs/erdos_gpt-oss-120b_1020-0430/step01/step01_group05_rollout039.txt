import time
import random
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n ∩ {sum = target} by bisection on the Lagrange multiplier."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretised."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def compute_O_full(h: np.ndarray) -> np.ndarray:
    """Full correlation O[l] = Σ h_i (1‑h_{i+l}) for l = -(n-1)…(n-1)."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())           # power of two ≥ 2n‑1
    H = np.fft.rfft(h, m)
    Hc = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * np.conj(Hc), m)       # circular correlation
    # first n entries → lags 0…n‑1, last n‑1 entries → lags –(n‑1)…-1
    pos = corr[:n]
    neg = corr[m - (n - 1):]
    return np.concatenate((neg, pos))


def gradient_from_weights(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_h Σ w_l O_l  where O_l = Σ_i h_i (1‑h_{i+l}),
    using FFT based convolutions.  w must sum to 1.
    """
    n = h.shape[0]
    m = 1 << ((3 * n - 2).bit_length())          # ≥ 3n‑2 for convolutions
    H = np.fft.rfft(h, m)
    W = np.fft.rfft(w, m)
    conv1 = np.fft.irfft(H * W, m)               # w * h
    rev = w[::-1]
    Wrev = np.fft.rfft(rev, m)
    conv2 = np.fft.irfft(H * Wrev, m)            # w_rev * h
    offset = n - 1
    part1 = conv1[offset: offset + n]
    part2 = conv2[offset: offset + n]
    # Σ w_l = 1, so gradient = 1 – (part1 + part2)
    return 1.0 - (part1 + part2)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new size n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    out = np.interp(x_new, x_old, init)
    target = n / 2.0
    return project_onto_bounds_and_sum(out, target)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    max_time = budget_s - 0.5                     # safety margin

    # problem size – a compromise between resolution and speed
    n = 500
    dx = 2.0 / n
    target_sum = n / 2.0

    # ----------------------------------------------------------------
    # Build a set of initial feasible vectors
    # ----------------------------------------------------------------
    init_vectors = []

    if "initial_h_values" in globals():
        try:
            init_vectors.append(
                upsample_initial(np.array(initial_h_values, dtype=float), n)

            pass

    # analytic starts
    pts = np.arange(n) * dx + dx * 0.5

    init_vectors.append(1.0 - np.sin(np.pi * pts / 2.0) ** 2)      # cosine shape
    # simple triangle
    tri = 1.0 - np.abs(pts - 1.0)
    init_vectors.append(tri)

    # ----------------------------------------------------------------
    # Optimisation utilities

    def evaluate_exact(h_vec):
        return compute_c5(h_vec)

    best_h = None
    best_val = np.inf

    # ----------------------------------------------------------------
    # Core optimisation: projected gradient descent with smooth max
    # ----------------------------------------------------------------
    def descent(start_h, init_alpha):
        h = start_h.copy()
        alpha = init_alpha
        lr = 0.2                     # initial step size
        min_lr = 1e-7
        alpha_factor = 1.5

        while time.time() - start < max_time and alpha <= max_alpha:

            O = compute_O_full(h)

                    lr = step * 1.1
                    break
                step *= 0.5
            else:
                lr *= 0.5
                if lr < min_lr:
                    break

            # track best exact value occasionally
            if random.random() < 0.02:
                val = evaluate_exact(h)
                nonlocal best_h, best_val
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                if best_val <= 0.3808:

            # increase smoothing parameter gradually
            alpha = min(alpha * alpha_factor, max_alpha)

        return h, alpha, lr

    # ----------------------------------------------------------------

    restarts = 0

        else:

            h0 = project_onto_bounds_and_sum(rnd, target_sum)

        h_opt, _, _ = descent(h0, init_alpha=5.0)

    best_val = evaluate_exact(best_h)
