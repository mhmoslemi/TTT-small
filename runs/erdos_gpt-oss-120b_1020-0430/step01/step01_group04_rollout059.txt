import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode="full")
    # using logsumexp from scipy.special
    from scipy.special import logsumexp
    return float(logsumexp(alpha * conv) / alpha)

def upsample(arr, new_n):
    old_n = len(arr)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0/old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

def generate_triangular(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0/n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def generate_U(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0/n
    d = np.minimum(x, 2.0 - x)
    u = 1.0 - d  # 1 at edges, 0 at centre
    target = n / 2.0
    u = u * (target / u.sum())
    u = np.clip(u, 0.0, 1.0)
    diff = target - u.sum()
    if abs(diff) > 1e-12:
        u += diff / n
        u = np.clip(u, 0.0, 1.0)
    return u

def random_feasible(n, rng):

    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a

def binary_feasible(n, rng):
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, n//2, replace=False)
    arr[idx] = 1.0
    return arr

def solve_one(start, n, rng, deadline):
    target = n / 2.0
    def eq_constr(h):
        return np.sum(h) - target
    cons = {"type": "eq", "fun": eq_constr}
    bounds = [(0.0, 1.0)] * n
    h = start.copy()
    for alpha in (5.0, 15.0, 30.0, 80.0):
        if time.time() > deadline:
            break
        res = opt.minimize(lambda h: softmax_c5(h, alpha=alpha),
                           h,
                           method="SLSQP",
                           bounds=bounds,
                           constraints=cons,
                           options={"maxiter": 400, "ftol": 1e-9, "disp": False})
        if not res.success:
            break
        h = np.clip(res.x, 0.0, 1.0)
        # re‑enforce sum exactly (helps numerical drift)
        cur = h.sum()
        if abs(cur - target) > 1e-9:
            h = h + (target - cur) / n
            h = np.clip(h, 0.0, 1.0)
    # final projection to meet constraints
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h = h + (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h, compute_c5(h)

def subgradient_opt(start, n, rng, deadline):
    h = start.copy()
    target = n/2.
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.5
    iteration = 0
    while time.time() < deadline:
        # compute conv and L
        conv = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(conv)
        L = idx - (n - 1)
        # gradient
        g = np.zeros_like(h)
        i_start = max(0, -L)
        i_end = min(n, n - L)
        g[i_start:i_end] += 1.0
        # -h_{i+L}
        if L >= 0:
            g[i_start:i_end] -= h[i_start+L:i_end+L]
            g[i_start+L:i_end+L] -= h[i_start:i_end]
        else:
            g[i_start:i_end] -= h[i_start+L:i_end+L]
            g[i_start+L:i_end+L] -= h[i_start:i_end]
        # step size schedule
        step_factor = 0.5 / np.sqrt(iteration + 1)
        h_new = h - step_factor * g
        # project onto box [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust sum to target
        cur_sum = h_new.sum()
        diff = target - cur_sum
        if abs(diff) > 1e-12:
            h_new += diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
        # evaluate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        iteration += 1
    return best_h, best_c5

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length,
    then re‑normalise to satisfy the integral constraint."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x), normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def generate_U(n: int) -> np.ndarray:
    """U‑shaped profile (high at the ends, low in the centre)."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    d = np.minimum(x, 2.0 - x)          # distance to the nearest edge, 0…1
    u = 1.0 - d                         # 1 at the edges, 0 in the centre
    target = n / 2.0
    u = u * (target / u.sum())
    u = np.clip(u, 0.0, 1.0)
    diff = target - u.sum()
    if abs(diff) > 1e-12:
        u += diff / n
        u = np.clip(u, 0.0, 1.0)
    return u


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random values in [0,1] with exact sum = n/2."""
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def binary_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random binary vector with exactly n/2 ones."""
    b = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    b[idx] = 1.0
    return b

# Optimisation primitives
# ------------------------------------------------------------
def run_slsqp(start: np.ndarray, n: int,
              deadline: float) -> np.ndarray:
    """Two‑stage SLSQP with increasing smoothness."""
    target = n / 2.0

    def eq_constr(h):
        return np.sum(h) - target

    cons = {"type": "eq", "fun": eq_constr}
    bounds = [(0.0, 1.0)] * n
    h = start.copy()

    for alpha in (5.0, 15.0, 30.0, 40.0):
        if time.time() > deadline:
            break
        res = opt.minimize(
            lambda hh: softmax_c5(hh, alpha=alpha),
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 400, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            break
        h = np.clip(res.x, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target) > 1e-12:
            h = h + (target - cur) / n
            h = np.clip(h, 0.0, 1.0)
    # final projection onto the feasible simplex
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h = h + (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def subgradient_opt(h: np.ndarray, n: int,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 2000) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑overlap objective."""
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(conv))
        L = idx - (n - 1)          # shift corresponding to the worst overlap
        # gradient of conv(L)
        g = np.zeros_like(h)
        i_start = max(0, -L)
        i_end = min(n, n - L)
        g[i_start:i_end] += 1.0
        if L >= 0:
            g[i_start:i_end] -= h[i_start + L:i_end + L]
            g[i_start + L:i_end + L] -= h[i_start:i_end]
        else:
            g[i_start:i_end] -= h[i_start + L:i_end + L]
            g[i_start + L:i_end + L] -= h[i_start:i_end]
        # step size decays like 1/sqrt(iter)
        step = 0.5 / np.sqrt(it + 1.0)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        cur = h_new.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            h_new = h_new + diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        it += 1
    return best_h


def solve_one(start: np.ndarray, n: int,
              rng: np.random.Generator,
              deadline: float) -> tuple:
    """Run SLSQP then a brief sub‑gradient refinement."""
    h = run_slsqp(start, n, deadline)
    if time.time() < deadline:
        h = subgradient_opt(h, n, rng, deadline, max_iters=1500)
    c5 = compute_c5(h)
    return h, c5
