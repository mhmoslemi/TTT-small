def local_search(h, target, dx, time_limit, start_time):
    best = h.copy()
    best_val = compute(best)
    temp_start = 0.01
    temp_end = 1e-5
    total_iters = 0
    while time.time() - start_time < time_limit:
        # schedule temperature decaying linearly
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        temp = temp_start * (1 - frac) + temp_end * frac
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # choose delta limit
        max_delta_i = min(0.05, 1 - best[i])  # step magnitude bounded
        min_delta_i = -min(0.05, best[i])

        min_delta_j = -min(0.05, 1 - best[j])

        delta_low = max(min_delta_i, -max_delta_j)

import time, numpy as np

def _project_onto_capped_simplex(v, target, tol=1e-10, max_iter=100):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[:n - s] = 1.0 - h[s:]
        g[n - s:] = 1.0
        g[s:] += -h[:n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[:n - s]
        g[:s] = 1.0
        g[:n - s] += -h[s:]
    return g

def _gd_momentum(h0, target, iters=2000, step0=0.5, decay=0.998, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    n = h.shape[0]
    for _ in range(iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        v = momentum * v + g
        h = h - step * v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

def _hill_climb(h_start, target, dx, time_limit, start_time, rng):
    h = h_start.copy()
    n = h.shape[0]
    best_val = _compute_c5(h, dx)
    # parameters
    max_step = 0.05
    # simple annealing schedule based on remaining time
    while time.time() - start_time < time_limit:
        i, j = rng.choice(n, size=2, replace=False)
        hi = h[i]
        hj = h[j]
        low = max(-min(hi, max_step), hj - 1.0, -max_step)
        high = min(min(1.0 - hi, max_step), hj, max_step)
        if low >= high:
            continue
        delta = rng.uniform(low, high)
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        val = _compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            h = h_new
    return h, best_val

    best_val = _compute_c5(best_h, dx)
    # allocate time for gradient descent stage

import time
import numpy as np


def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} via bisection on the dual variable."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation at a given integer shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_momentum(h0: np.ndarray, target: float,
                 iters: int = 2000,
                 step0: float = 0.5,
                 decay: float = 0.998,
                 momentum: float = 0.9) -> np.ndarray:
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    n = h.shape[0]
    for _ in range(iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        v = momentum * v + g
        h = h - step * v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def _hill_climb(h_start: np.ndarray, target: float, dx: float,
                time_limit: float, start_time: float,
                rng: np.random.RandomState) -> tuple[np.ndarray, float]:
    """Greedy local search that preserves sum and box constraints."""
    h = h_start.copy()
    n = h.shape[0]
    best_val = _compute_c5(h, dx)
    max_step = 0.05
    while time.time() - start_time < time_limit:
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]

        low = max(-min(hi, max_step), hj - 1.0, -max_step)
        high = min(min(1.0 - hi, max_step), hj, max_step)
        if low >= high:
            continue
        delta = rng.uniform(low, high)
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        val = _compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            h = h_new
    return h, best_val
