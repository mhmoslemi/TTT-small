def project_to_feasible(h, target, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some values
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)

    # Not required likely.
    return h

import time
import numpy as np

def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_to_feasible(h: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to the target by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    # tiny residual correction
    err = target - h.sum()
    if abs(err) > 1e-12:
        for i in range(h.size):
            if err > 0 and h[i] < 1.0:
                add = min(err, 1.0 - h[i])
                h[i] += add
                err -= add
            elif err < 0 and h[i] > 0.0:
                sub = min(-err, h[i])
                h[i] -= sub
                err += sub
            if abs(err) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _gradient_descent(h0: np.ndarray, time_left, target_bound=0.3808) -> tuple:
    """Smooth‑max gradient descent with projection."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    # algorithmic parameters
    alpha = 5.0          # smoothing parameter
    step = 0.6           # initial learning rate
    max_iter = 2000

    it = 0
    while it < max_iter and time_left() > 0.5:
        f = 1.0 - h
        conv = np.correlate(h, f, mode="full")
        max_corr = conv.max()
        cur_c5 = max_corr * (2.0 / n)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            if best_c5 < target_bound:
                break

        # smooth‑max weights
        m = max_corr
        w = np.exp(alpha * (conv - m))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient components A and B
        wf_corr = np.correlate(w, f, mode="full")
        A = wf_corr[(n - 1):(n - 1) + n]          # length n
        B = np.convolve(w, h, mode="full")[:n]   # length n
        grad = A - B

        # tentative update
        h_new = h - step * grad
        h_new = _project_to_feasible(h_new, target_sum)

        new_c5 = _c5_from_h(h_new)
        # adapt step size
        if new_c5 > cur_c5:
            step *= 0.7
        else:
            step = min(step * 1.02, 1.0)
            # slowly sharpen the smooth max
            alpha = min(alpha * 1.01, 30.0)
            h = h_new

        it += 1

    return best_h, best_c5

    best_h = init_h
    best_c5 = _c5_from_h(best_h)
    best_n = n0

    # candidate grid sizes (prefer even numbers, not too large)
    cand = [n0]
    for mul in (2, 3):
        n = n0 * mul
        if n <= 600:
            cand.append(int(n))
    for n in (200, 256, 300, 384, 500):
        if n <= 600:
            cand.append(int(n))
    cand = sorted(set(cand))

    for n in cand:

        base_h = _project_to_feasible(base_h, n / 2.0)

    # final safeguard – if nothing improved, return whatever we have
    return best_h.tolist(), float(best_c5), int(best_n)
