        # compute overlap correlation

        min_idx = overlap.argmin()
        shift = min_idx - (n - 1)
        # Compute gradient g
        g = np.zeros_like(h)
        if shift >= 0:

            g[shift:] += h[:n - shift]

            g[s:] += h[:n - s]
            g[:n - s] += h[s:]

    corr = np.correlate(x, 1 - x, mode='full')

    m = np.max(bc)

    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def gradient_improve(h, rng, max_iters, time_limit):
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        # compute overlap correlation to find minimal overlap shift
        overlap = np.correlate(h, h, mode='full')
        min_idx = overlap.argmin()
        min_shift = min_idx - (n - 1)
        # gradient g
        g = np.zeros_like(h)
        if min_shift >= 0:
            s = min_shift
            g[:n-s] += h[s:]
            g[s:] += h[:n-s]
        else:
            s = -min_shift
            g[s:] += h[:n-s]
            g[:n-s] += h[s:]
        # select candidates
        # sort indices by g values
        sorted_idx = np.argsort(g)
        low = sorted_idx[:max(1, n//10)]
        high = sorted_idx[-max(1, n//10):]
        # filter feasible
        low = low[h[low] > 0]
        high = high[h[high] < 1]
        if low.size == 0 or high.size == 0:
            # fallback random transfer
            i = rng.integers(n)
            j = rng.integers(n)
        else:
            i = int(rng.choice(high))
            j = int(rng.choice(low))
        max_eps = min(1.0 - h[i], h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.5  # up to half of max
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-10:  # improvement
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        # occasional random perturbation
        if it % 200 == 0 and rng.random() < 0.2:
            # random transfer between any index
            a = rng.integers(n)
            b = rng.integers(n)
            if a == b:
                continue
            max_eps2 = min(1.0 - h[a], h[b])
            if max_eps2 <= 0:
                continue
            eps2 = max_eps2 * rng.random() * 0.5
            h[a] += eps2
            h[b] -= eps2
    return best_h, best_c

def adjust_sum_to_target(h, target, rng):

    diff = target - h.sum()

def gradient_improve(h, rng, max_iters, end_time):
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        if time.time() > end_time:
            break
        # compute overlap
        overlap = np.correlate(h, h, mode='full')
        min_idx = overlap.argmin()
        shift = min_idx - (n - 1)
        # compute gradient g
        g = np.zeros_like(h)
        if shift >= 0:
            s = shift
            g[:n-s] += h[s:]
            g[s:] += h[:n-s]
        else:
            s = -shift
            g[s:] += h[:n-s]
            g[:n-s] += h[s:]
        # Choose candidates
        # We'll pick high from top 20% of g that are <1
        # low from bottom 20% of g that are >0
        sorted_idx = np.argsort(g)
        low_cut = max(1, n // 10)
        low_candidates = sorted_idx[:low_cut]
        high_candidates = sorted_idx[-low_cut:]
        low_candidates = low_candidates[h[low_candidates] > 1e-12]
        high_candidates = high_candidates[h[high_candidates] < 1 - 1e-12]
        if low_candidates.size == 0 or high_candidates.size == 0:
            # random transfer
            i = rng.integers(n)
            j = rng.integers(n)
        else:
            i = int(rng.choice(high_candidates))
            j = int(rng.choice(low_candidates))
        max_eps = min(1.0 - h[i], h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        c_new = compute_c5(h_new)
        if c_new + 1e-12 < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        # occasional random perturbation

            a = rng.integers(n)
            b = rng.integers(n)
            if a != b:
                max_eps2 = min(1.0 - h[a], h[b])
                if max_eps2 > 0:
                    eps2 = max_eps2 * rng.random() * 0.5
                    h[a] += eps2
                    h[b] -= eps2
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = 0.0
CURRENT_BETA = 1.0

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate of max C5."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # fix any tiny sum error
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def adjust_sum_to_target(h, target, rng):
    """Redistribute mass to hit the exact target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_improve(h, rng, max_iters, deadline):
    """Iteratively move mass from low‑ to high‑gradient positions to raise min overlap."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # overlap with itself, min gives the worst shift
        ov = np.correlate(h, h, mode="full")
        min_idx = ov.argmin()
        shift = min_idx - (n - 1)

        # gradient of the overlap at this shift
        g = np.zeros_like(h)
        if shift >= 0:
            s = shift
            g[: n - s] += h[s:]
            g[s :] += h[: n - s]
        else:
            s = -shift
            g[s :] += h[: n - s]
            g[: n - s] += h[s :]

        # choose candidates
        order = np.argsort(g)
        cut = max(1, n // 10)
        low = order[:cut][h[order[:cut]] > 1e-12]
        high = order[-cut:][h[order[-cut:]] < 1 - 1e-12]

        if low.size and high.size:
            i = int(rng.choice(high))
            j = int(rng.choice(low))
        else:
            i = rng.integers(n)
            j = rng.integers(n)

        max_eps = min(1.0 - h[i], h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        c_new = compute_c5(h_new)

        if c_new < best_c - 1e-12:
            best_c, best_h, h = c_new, h_new.copy(), h_new

        # occasional random shuffle to avoid stagnation
        if it % 400 == 0 and rng.random() < 0.15:
            a, b = rng.integers(n, size=2)
            if a != b:
                max_eps2 = min(1.0 - h[a], h[b])
                if max_eps2 > 0:
                    eps2 = max_eps2 * rng.random() * 0.5
                    h[a] += eps2
                    h[b] -= eps2

    return best_h, best_c


def random_binary(n, rng):
    """Generate a feasible binary vector with exactly n/2 ones."""
    target = int(n // 2)
    h = np.zeros(n)
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    end = start + budget_s * 0.99  # a small safety margin

    # ----------------------------------------------------------------------

    if "initial_h_values" in globals():

    # ----------------------------------------------------------------------
    # Phase 1 – aggressive gradient driven search with restarts
    # ----------------------------------------------------------------------
    search_deadline = start + budget_s * 0.70
    while time.time() < search_deadline:
        # start from a perturbed best or a fresh random binary
        if rng.random() < 0.5:
            start_vec = best_h.copy()
            # small random shuffle to escape local traps
            for _ in range(6):
                i, j = rng.integers(n, size=2)
                if i == j:
                    continue
                max_eps = min(start_vec[i], 1.0 - start_vec[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * 0.5
                start_vec[i] -= eps
                start_vec[j] += eps

        h_opt, c_opt = gradient_improve(start_vec, rng, max_iters=4000,
                                         deadline=search_deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----------------------------------------------------------------------
    # Phase 2 – SLSQP refinements with an increasing beta ladder
    # ----------------------------------------------------------------------
    for beta in [5.0, 20.0, 80.0, 200.0, 800.0]:
        if time.time() > end:

            trial = slsqp_optimize(best_h, beta=beta, max_iter=400)
            c_trial = compute_c5(trial)

        except Exception:

    # ----------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_to_target(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
