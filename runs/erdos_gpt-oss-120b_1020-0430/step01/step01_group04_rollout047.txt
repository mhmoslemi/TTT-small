def softmax_c5_grad(x, alpha=30.0):
    # compute conv
    conv = np.correlate(x, 1.0 - x, mode='full')
    # exponentials scaled:
    max_val = np.max(alpha * conv)  # for stability
    # compute softmax weights
    exp_scaled = np.exp(alpha * conv - max_val)
    p = exp_scaled / exp_scaled.sum()
    n = len(x)
    grad = np.zeros(n)
    # shift range
    for shift in range(-(n-1), n):
        idx = shift + (n - 1)
        w = p[idx]
        # compute range of i where both i and i+shift valid
        if shift >= 0:
            i_start = 0
            i_end = n - shift  # exclusive
            i = np.arange(i_start, i_end)
            # contribution for i (h_i term)
            grad[i] += w * (1.0 - x[i + shift])
            # contribution for i+shift (h_{i+shift} term)
            grad[i + shift] += w * (-x[i])
        else:
            s = -shift
            i_start = 0
            i_end = n - s
            i = np.arange(i_start, i_end)
            # now shift negative: conv term = sum_{i} h_{i+s} * (1 - h_i)
            # derivative wrt h_i gives -h_{i+s}, derivative wrt h_{i+s} gives (1 - h_i)
            grad[i] += w * (-x[i + s])
            grad[i + s] += w * (1.0 - x[i])
    return grad

def upsample(arr, new_n):

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n

    target = new_n / 2.0

def solve_one_start(start, n, target_sum, time_deadline):
    h = project_to_feasible(start, target_sum)
    # schedule alphas
    alphas = [5, 15, 50, 200, 800]
    # maxiters maybe decreasing for later alphas
    maxiters = [400, 400, 400, 600, 800]
    for alpha, maxit in zip(alphas, maxiters):
        if time.time() > time_deadline:
            break
        # define obj and grad
        def obj(x):
            return softmax_c5(x, alpha)
        def jac(x):
            return softmax_c5_grad(x, alpha)
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, method='SLSQP', jac=jac,
                           bounds=bounds, constraints=cons,
                           options={"maxiter": maxit, "ftol": 1e-9, "disp": False})
        except Exception:
            # fallback to no-jac minimize
            res = minimize(lambda x: softmax_c5(x, alpha),
                           h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={"maxiter": maxit, "ftol": 1e-9, "disp": False})
        if not res.success:
            # proceed with current h anyway
            h = project_to_feasible(res.x, target_sum) if hasattr(res, 'x') else h
            continue
        h = project_to_feasible(res.x, target_sum)
    # final compute exact c5
    c5_val = compute_c5(h)
    return h, c5_val

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# Helper utilities

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max‑overlap."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def softmax_c5_grad(h: np.ndarray, alpha: float = 30.0) -> np.ndarray:
    """Gradient of the soft‑max objective."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    # stable softmax weights
    max_val = np.max(alpha * conv)
    exp_scaled = np.exp(alpha * conv - max_val)
    p = exp_scaled / np.sum(exp_scaled)

    grad = np.zeros(n, dtype=np.float64)
    for shift in range(-(n - 1), n):
        idx = shift + (n - 1)
        w = p[idx]
        if shift >= 0:
            end = n - shift
            if end <= 0:
                continue
            i = np.arange(end)
            grad[i] += w * (1.0 - h[i + shift])
            grad[i + shift] += w * (-h[i])
        else:
            s = -shift
            end = n - s
            if end <= 0:
                continue
            i = np.arange(end)
            grad[i] += w * (-h[i + s])
            grad[i + s] += w * (1.0 - h[i])
    return grad


def generate_triangular(n: int) -> np.ndarray:
    """Triangular profile, renormalised to integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1], sum = n/2)."""
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()

        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, preserving the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def project_to_feasible(x: np.ndarray, target: float, max_iter: int = 10) -> np.ndarray:
    """Clip to [0,1] and enforce the sum constraint by uniform correction."""
    for _ in range(max_iter):
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        x = x + diff / len(x)
    return np.clip(x, 0.0, 1.0)


def refine_max_objective(h: np.ndarray, target: float, rng: np.random.Generator,
                         max_iter: int = 200, init_lr: float = 0.1) -> np.ndarray:
    """Projected sub‑gradient steps on the true max‑overlap objective."""
    n = len(h)
    lr = init_lr
    for _ in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(conv))
        shift = idx - (n - 1)
        grad = np.zeros_like(h)
        if shift >= 0:
            end = n - shift
            if end > 0:
                i = np.arange(end)
                grad[i] += (1.0 - h[i + shift])
                grad[i + shift] += (-h[i])
        else:
            s = -shift
            end = n - s
            if end > 0:
                i = np.arange(end)
                grad[i] += (-h[i + s])
                grad[i + s] += (1.0 - h[i])
        # move opposite to the gradient (we minimise the max)
        h = h - lr * grad
        h = project_to_feasible(h, target)
        lr *= 0.995  # gentle decay
    return h


def hill_climb(h: np.ndarray, target: float, rng: np.random.Generator,
               time_deadline: float, max_iter: int = 500) -> (np.ndarray, float):
    """Simple neighbourhood search to polish a candidate."""
    best = h.copy()
    best_c5 = compute_c5(best)
    n = len(h)
    while time.time() < time_deadline and max_iter > 0:
        i, j = rng.choice(n, size=2, replace=False)
        eps = (rng.random() - 0.5) * 0.2  # [-0.1,0.1]
        cand = best.copy()
        cand[i] = np.clip(cand[i] - eps, 0.0, 1.0)
        cand[j] = np.clip(cand[j] + eps, 0.0, 1.0)
        cand = project_to_feasible(cand, target)
        c5 = compute_c5(cand)
        if c5 < best_c5 - 1e-7:
            best, best_c5 = cand, c5
        max_iter -= 1
    return best, best_c5


def solve_one_start(start: np.ndarray, n: int, target: float,
                    time_deadline: float, rng: np.random.Generator):
    """Optimise a single start point via staged smooth‑max then exact sub‑gradient."""
    h = project_to_feasible(start, target)

    # staged soft‑max optimisation
    alphas = [5, 15, 50, 200, 800]
    maxiters = [400, 400, 400, 600, 800]
    for alpha, mxit in zip(alphas, maxiters):
        if time.time() > time_deadline:
            break

        def obj(x):
            return softmax_c5(x, alpha)

        def jac(x):
            return softmax_c5_grad(x, alpha)

        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, method="SLSQP", jac=jac,
                           bounds=bounds, constraints=cons,
                           options={"maxiter": mxit, "ftol": 1e-9, "disp": False})
        except Exception:
            res = minimize(lambda x: softmax_c5(x, alpha), h,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": mxit, "ftol": 1e-9, "disp": False})

        if res.success:
            h = project_to_feasible(res.x, target)
        else:
            # still make a best‑effort projection
            if hasattr(res, "x"):
                h = project_to_feasible(res.x, target)

    if time.time() < time_deadline:

    # final cheap local polishing
    if time.time() < time_deadline:
        h, _ = hill_climb(h, target, rng, time_deadline, max_iter=300)

    return h, compute_c5(h)

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (must be even so that n/2 is integer)
    candidate_ns = [80, 120, 160, 200, 240, 300]

    # parent construction, if provided by the harness
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target_sum = n / 2.0

        # assemble a diverse pool of starting points
        starts = []

        # upsample the parent solution
        if parent_h is not None:
            if parent_n != n:
                starts.append(upsample(parent_h, n))
            else:
                starts.append(parent_h.copy())

        # upsample the best solution found so far
        if best_h is not None:
            if best_n != n:
                starts.append(upsample(best_h, n))
            else:
                starts.append(best_h.copy())

        # classical triangular shape
        starts.append(generate_triangular(n))

        # several random feasible vectors
        for _ in range(5):
            starts.append(random_start(n, rng))

        # perturbations of the current best (if any)
        if best_h is not None:
            for _ in range(3):
                pert = best_h + rng.normal(scale=0.12, size=n)
                pert = np.clip(pert, 0.0, 1.0)
                pert = project_to_feasible(pert, target_sum)
                starts.append(pert)

        # optimise every start, retaining the best overall
        for s in starts:
            if time.time() > deadline:
                break
            h_opt, c5_opt = solve_one_start(s, n, target_sum, deadline, rng)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                # early exit if we beat the target bound
                if best_c5 < 0.3808:
                    deadline = time.time()  # force termination after this iteration

    # fallback – uniform half‑value vector (guaranteed feasible)
    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final sanity projection and C5 recomputation
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), int(best_n)
