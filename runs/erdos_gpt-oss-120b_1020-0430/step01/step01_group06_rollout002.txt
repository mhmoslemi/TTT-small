    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    def compute_c5(v):
        """max correlation multiplied by dx = 2/n"""
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(corr.max() * (2.0 / len(v)))

    def random_binary():
        v = np.zeros(n, dtype=float)
        idx = np.random.choice(n, n // 2, replace=False)
        v[idx] = 1.0
        return v

    def bad_indices(v, shift):
        """positions i with v[i]=1 and v[i+shift]=0 (valid index)"""
        if shift >= 0:
            idx = np.arange(n - shift)
            mask = (v[idx] > 0.5) & (v[idx + shift] < 0.5)
            return idx[mask]
        else:
            s = -shift
            idx = np.arange(s, n)
            mask = (v[idx] > 0.5) & (v[idx - s] < 0.5)
            return idx[mask]

    def greedy_targeted(v):
        """try all swaps that fix a bad pair for the worst shift"""
        corr_full = np.correlate(v, 1.0 - v, mode="full")
        smax = int(np.argmax(corr_full)) - (n - 1)
        cur_val = float(corr_full.max() * (2.0 / n))
        bad = bad_indices(v, smax)
        if bad.size == 0:
            return v, None
        best_v = v
        best_val = cur_val
        improved = False
        for i in bad:
            j = i + smax
            nv = v.copy()
            nv[i], nv[j] = nv[j], nv[i]
            val = compute_c5(nv)
            if val < best_val - 1e-12:
                best_v = nv
                best_val = val
                improved = True
        if improved:
            return best_v, best_val
        return v, None

    def local_search(start_vec, max_iters):
        """greedy + occasional random swaps"""
        v = start_vec.copy()
        cur_val = compute_c5(v)
        best_local = v.copy()
        best_local_val = cur_val
        stagn = 0
        for _ in range(max_iters):
            # attempt targeted improvement
            v_new, val_new = greedy_targeted(v)
            if val_new is not None:
                v, cur_val = v_new, val_new
                if cur_val < best_local_val:
                    best_local = v.copy()
                    best_local_val = cur_val
                stagn = 0
                continue

            # no targeted move: try a random 1↔0 swap
            ones = np.where(v > 0.5)[0]
            zeros = np.where(v < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = random.choice(ones.tolist())
            j = random.choice(zeros.tolist())
            nv = v.copy()
            nv[i], nv[j] = nv[j], nv[i]
            new_val = compute_c5(nv)

            if new_val < cur_val:
                v, cur_val = nv, new_val
                if new_val < best_local_val:
                    best_local = nv.copy()
                    best_local_val = new_val
                stagn = 0
            else:
                stagn += 1
                if stagn > 30:
                    # allow uphill move with tiny probability
                    T = max(1e-4, best_local_val * 0.01)
                    if random.random() < math.exp(-(new_val - cur_val) / T):
                        v, cur_val = nv, new_val
                        stagn = 0
        return best_local, best_local_val

    # ------------------------------------------------------------------
    # Main iterative search (restarts + refinement)
    # ------------------------------------------------------------------
    best_h = init.copy()
    best_val = compute_c5(best_h)

    while time.time() - start_time < budget_s * 0.95:
        # a fresh random start
        v0 = random_binary()
        # adapt iteration count to remaining time
        remaining = budget_s - (time.time() - start_time)
        iters = 2000 if remaining > 30 else 400
        cand_h, cand_val = local_search(v0, iters)
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val

        # try to polish the current global best
        cand_h2, cand_val2 = local_search(best_h.copy(), max(1000, iters // 2))
        if cand_val2 < best_val:
            best_h, best_val = cand_h2, cand_val2

    # ------------------------------------------------------------------
    # Final feasibility correction (tiny numerical drift)
    # ------------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-8:
        if err > 0:
            cand = np.where(best_h > 1e-12)[0]
            if cand.size:
                i = random.choice(cand.tolist())
                best_h[i] -= err
        else:
            cand = np.where(best_h < 1 - 1e-12)[0]
            if cand.size:
                i = random.choice(cand.tolist())
                best_h[i] -= err  # err negative ⇒ addition
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n)
