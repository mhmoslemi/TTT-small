import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

# ----------------------------------------------------------------------
# Correlation based objective
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

# ----------------------------------------------------------------------
# Gradient of the correlation at the maximizing shift (used for local GD)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 400, step0: float = 0.5,
                 decay: float = 0.995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

# ----------------------------------------------------------------------
# Smooth max (log-sum-exp) and SLSQP refinement
# ----------------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 500) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)

# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h on [0,2] that (hopefully) improves the Erdős C5 bound."""
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Determine discretisation size, keeping any parent solution if present.
    # --------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    n = max(200, len(base_h) if base_h is not None else 0)
    target_sum = n / 2.0
    dx = 2.0 / n

    # Helper to upscale/downscale a parent solution to the chosen n.
    def _resize(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # --------------------------------------------------------------
    # Initial feasible vector
    # --------------------------------------------------------------
    if base_h is not None:
        h0 = _resize(base_h)
        h0 = _project_onto_capped_simplex(h0, target_sum)
    else:
        h0 = _random_feasible(rng, n, target_sum)

    best_h = h0.copy()
    best_val = _compute_c5(best_h, dx)
    cur_h = best_h.copy()
    cur_val = best_val

    # --------------------------------------------------------------
    # Simulated Annealing phase (global search)
    # --------------------------------------------------------------
    # Temperature schedule (linear cooling)
    T0 = 0.008          # start temperature
    Tf = 1e-6           # final temperature
    # allocate about 85% of budget to SA
    sa_budget = budget_s * 0.85

    while True:
        elapsed = time.time() - start_time
        if elapsed >= sa_budget:
            break
        # linear temperature
        t_frac = elapsed / sa_budget
        T = max(Tf, (1 - t_frac) * T0 + t_frac * Tf)

        # propose a move: transfer δ from i to j
        i, j = rng.choice(n, 2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        # choose δ; bias towards larger fractions early on
        delta = max_delta * (0.02 + 0.48 * rng.random())  # roughly up to 0.5*max_delta
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = _compute_c5(new_h, dx)

        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()

        # occasional larger perturbation to escape plateaus
        if rng.rand() < 0.001:
            cur_h = best_h + rng.normal(scale=0.05, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_onto_capped_simplex(cur_h, target_sum)
            cur_val = _compute_c5(cur_h, dx)

    # --------------------------------------------------------------
    # Local refinement (gradient descent + optional SLSQP)
    # --------------------------------------------------------------
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        # Gradient descent refinement
        refined = _gd_optimize(best_h, target_sum, max_iter=400,
                               step0=0.5, decay=0.995)
        refined_val = _compute_c5(refined, dx)
        if refined_val < best_val:
            best_val = refined_val
            best_h = refined.copy()

    if time_left > 20.0:
        try:
            refined2 = _slsqp_optimize(best_h, target_sum,
                                       alpha=50.0, max_iter=500)
            refined2_val = _compute_c5(refined2, dx)
            if refined2_val < best_val:
                best_val = refined2_val
                best_h = refined2.copy()
        except Exception:
            pass

    # Final safety projection and evaluation
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
