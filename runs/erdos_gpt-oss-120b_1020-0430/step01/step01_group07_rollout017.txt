import time, math, random, numpy as np

def _c5_metric(h):
    """Discrete C5 value (max correlation) for array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _max_corr_and_shift(h):
    """Return (max_corr, shift) where shift is integer offset (samples)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_corr = corr[idx] * (2.0 / n)
    shift = idx - (n - 1)
    return max_corr, shift

def _binary_from_continuous(v):
    """Force vector v to binary with exactly n/2 ones (n even)."""
    n = v.shape[0]
    target = n // 2
    idx = np.argsort(-v)  # descending
    b = np.zeros(n, dtype=np.float64)
    b[idx[:target]] = 1.0
    return b

def _greedy_local_search(start_h, max_seconds, global_start):
    """
    Greedy improvement: repeatedly swap a 0 and a 1 that most increase the
    overlap for the currently worst shift. Stops after max_seconds or after
    a streak of unsuccessful swaps.
    """
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(h)
    start = time.time()
    no_improve = 0
    while time.time() - start < max_seconds:
        cur_c, s = _max_corr_and_shift(h)          # worst shift
        # neighbor sum for this shift
        N = np.zeros(n)
        if s >= 0:
            if s < n:
                N[s:] += h[:-s]          # left neighbor
                N[:n - s] += h[s:]       # right neighbor
        else:
            a = -s
            if a < n:
                N[:n - a] += h[a:]       # left neighbor (i-s)
                N[a:] += h[:n - a]       # right neighbor (i+s)
        zero_idx = np.where(h < 1e-9)[0]
        one_idx = np.where(h > 1 - 1e-9)[0]
        if zero_idx.size == 0 or one_idx.size == 0:
            break
        i0 = zero_idx[np.argmax(N[zero_idx])]
        i1 = one_idx[np.argmin(N[one_idx])]
        if N[i0] <= N[i1] + 1e-12:
            # no improving direct swap; try a random pair a few times
            trials = 0
            improved = False
            while trials < 20:
                i0 = random.choice(zero_idx)
                i1 = random.choice(one_idx)
                h[i0], h[i1] = 1.0, 0.0
                new_c = _c5_metric(h)
                if new_c + 1e-12 < cur_c:
                    improved = True
                    break
                h[i0], h[i1] = 0.0, 1.0
                trials += 1
            if not improved:
                no_improve += 1
                if no_improve >= 80:
                    break
                continue
        else:
            # perform the promising swap
            h[i0], h[i1] = 1.0, 0.0
            new_c = _c5_metric(h)
            if new_c + 1e-12 < cur_c:
                no_improve = 0
            else:
                # revert – swap not helpful for global max
                h[i0], h[i1] = 0.0, 1.0
                no_improve += 1
                if no_improve >= 80:
                    break
                continue
        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, h.copy()
    return best_h, best_c

    phi = (math.sqrt(5) - 1) / 2.0

        return h

    best_c = float('inf')

    # Allocate most of the budget to refine the three seeds

        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= 0.38075:        # early exit if sufficiently good

    while True:

        remaining = total_budget - elapsed
        if remaining < 5.0:

        # generate new binary start
        h0 = random_binary()
        slice_time = min(remaining * 0.2, 12.0)
        h_opt, c_opt = _greedy_local_search(h0, slice_time, start_all)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= 0.38075:

    # final enforcement of exact sum (tiny rounding)

    if abs(diff) > 1e-10:
