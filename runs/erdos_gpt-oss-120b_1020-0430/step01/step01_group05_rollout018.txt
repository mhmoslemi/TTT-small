def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    # compute correlation between h and (1-h) using FFT linear correlation
    # Use convolution with reversed (1-h)
    rev = (1-h)[::-1]
    N = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(rev, N)
    conv = np.fft.irfft(fa * fb, N)[:2*n -1]
    # This is linear convolution of h and rev => correlation vector same as np.correlate(h, 1-h, mode='full')
    return np.max(conv) * dx

def project_onto_bounds_and_sum(x, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo+hi)/2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    # reversed second vector
    rev = (1.0 - h)[::-1]
    N = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(rev, N)
    conv = np.fft.irfft(fa * fb, N)[:2*n-1]
    # conv is correlation values (maybe need no reversal)
    max_corr = np.max(conv)
    return max_corr * dx

import time, random, math
import numpy as np
from scipy.optimize import differential_evolution

# ---------- helper functions (top level) ----------

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of x onto the intersection of the box [0,1]^n
    and the hyperplane sum(x)=target. Uses a bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0  # guaranteed bounds for the multiplier
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5_fft(h: np.ndarray) -> float:
    """
    Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx for a discretised step function h.
    Uses a linear convolution via FFT (equivalent to np.correlate with mode='full').
    """
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]                 # reversed (1‑h)
    N = 1 << (2 * n - 1).bit_length()    # next power of two ≥ 2n‑1
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(rev, N)
    conv = np.fft.irfft(fa * fb, N)[: 2 * n - 1]
    max_corr = np.max(conv)
    return max_corr * dx


def evaluate_heights(heights: np.ndarray) -> float:
    """Objective for DE: given block heights, build the full step function and evaluate C5."""
    global block_len, target_sum_blocks
    proj = project_onto_bounds_and_sum(heights, target_sum_blocks)
    full = np.repeat(proj, block_len)
    return compute_c5_fft(full)


def upsample_h(h_low: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a lower‑resolution step function onto n_target points."""
    old_n = h_low.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(new_x, old_x, h_low)


# ---------- main driver ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low value of the Erdős constant C5.
    Returns (list_of_values, C5_bound, n_points).
    """
    global n, m, block_len, target_sum, target_sum_blocks

    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # problem size – choose a divisible pair (n points, m blocks)
    n = 512               # total samples on [0,2]
    m = 128               # number of constant blocks
    block_len = n // m    # points per block (must be integer)
    target_sum = n / 2.0                      # integral constraint in sample units
    target_sum_blocks = m / 2.0                # same constraint for block heights

    # --------- initialise a good seed for DE ----------
    # default: random feasible block heights
    init_heights = project_onto_bounds_and_sum(np.random.rand(m), target_sum_blocks)

    # if a parent construction is supplied, up‑sample it and extract block means
    if "initial_h_values" in globals():
        try:
            init_full = upsample_h(np.array(initial_h_values, dtype=float), n)
            # average over each block to obtain a block‑level start
            init_heights = init_full.reshape(m, block_len).mean(axis=1)
            init_heights = project_onto_bounds_and_sum(init_heights, target_sum_blocks)
        except Exception:
            pass  # fall back to random start if anything goes wrong

    # --------- differential evolution (global search) ----------
    bounds = [(0.0, 1.0)] * m
    pop_factor = 8                # actual population size = pop_factor * m
    max_iter = 200                # generations; population*generations ≈ 25 000 evals

    # build an explicit initial population (first individual is our seeded start)
    popsize_total = pop_factor * m
    init_pop = np.empty((popsize_total, m), dtype=float)
    init_pop[0] = init_heights
    for i in range(1, popsize_total):
        init_pop[i] = project_onto_bounds_and_sum(np.random.rand(m), target_sum_blocks)

    result = differential_evolution(
        evaluate_heights,
        bounds,
        strategy="best1bin",
        maxiter=max_iter,
        popsize=pop_factor,
        mutation=0.7,
        recombination=0.7,
        seed=seed,
        init=init_pop,
        polish=False,
        disp=False,
    )
    best_heights = project_onto_bounds_and_sum(result.x, target_sum_blocks)
    best_h = np.repeat(best_heights, block_len)
    best_val = compute_c5_fft(best_h)

    # --------- local refinement: block‑level exchanges ----------
    target_bound = 0.3808  # stop early if we beat the target

        i, j = np.random.randint(m, size=2)
        if i == j:

        hi, hj = best_heights[i], best_heights[j]

        cand_heights[j] = hj - delta
        cand_full = np.repeat(cand_heights, block_len)
        val = compute_c5_fft(cand_full)
        if val < best_val:

    while time.time() - start < budget_s - 0.5:

        delta_max = min(1.0 - hi, hj)

        delta = np.random.uniform(delta_min, delta_max)
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta

        if val < best_val:

            best_val = val
            if best_val <= target_bound:

    # --------- final safety projection and evaluation ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5_fft(best_h)

    return (best_h.tolist(), float(best_val), int(n))
