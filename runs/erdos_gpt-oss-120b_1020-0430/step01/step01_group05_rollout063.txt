import time, numpy as np, random, math

TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x, target):
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    auto = np.fft.irfft(H * np.conj(H), n=L)
    min_autocorr = np.min(auto[:n])
    return 1.0 - min_autocorr * dx

def gradient_worst_shift(h, k):
    n = h.shape[0]
    if k == 0:
        return np.ones_like(h) - 2.0 * h   # gradient of f_0
    else:
        forward = np.zeros_like(h)
        backward = np.zeros_like(h)
        forward[:n-k] = h[k:]
        backward[k:] = h[:n-k]
        return np.ones_like(h) - (forward + backward)

def subgradient_descent(h0, max_iter, init_step):
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    target = n/2
    L = 1 << ((2*n - 1).bit_length())
    for t in range(max_iter):
        # compute autocorr
        H = np.fft.rfft(h, n=L)
        auto = np.fft.irfft(H * np.conj(H), n=L)
        # find worst shift (minimum autocorr)
        k = int(np.argmin(auto[:n]))
        # compute gradient
        if k == 0:
            g = np.ones_like(h) - 2.0 * h
        else:
            forward = np.empty_like(h)
            backward = np.empty_like(h)
            forward[:n-k] = h[k:]
            forward[n-k:] = 0.0
            backward[:k] = 0.0
            backward[k:] = h[:n-k]
            g = np.ones_like(h) - (forward + backward)
        step = init_step / np.sqrt(t+1)
        h_candidate = project_onto_bounds_and_sum(h - step * g, target)
        val = compute_c5(h_candidate)
        if val < cur_val - 1e-12:  # improvement
            h = h_candidate
            cur_val = val
            if val < best_val:
                best_val = val
                best_h = h_candidate.copy()
        # else maybe keep old but also perhaps attempt random perturbation later
        if best_val <= 0.3808:
            break
    return best_h, best_val

def subgradient_descent(h0, max_iter, init_step):
    n = h0.shape[0]
    target = n/2.0
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    L = 1 << ((2*n - 1).bit_length())
    for t in range(max_iter):
        # compute correlation once
        H = np.fft.rfft(h, n=L)
        auto_full = np.fft.irfft(H * np.conj(H), n=L)
        pos = auto_full[:n]
        neg = auto_full[-(n-1):]
        # find worst shift
        min_pos = pos.min()
        min_neg = neg.min()
        if min_pos <= min_neg:
            k = int(pos.argmin())
            min_val = min_pos
        else:
            idx = int(neg.argmin())
            k = -((n-1) - idx)
            min_val = min_neg
        # compute gradient
        shift = abs(k)
        g = np.ones_like(h)
        if shift == 0:
            g -= 2.0 * h
        else:
            # forward (shift positive)
            g[:n-shift] -= h[shift:]
            # backward
            g[shift:] -= h[:n-shift]
        # step size
        step = init_step / np.sqrt(t+1)
        h_candidate = project_onto_bounds_and_sum(h - step * g, target)
        val_candidate = compute_c5(h_candidate)
        if val_candidate < cur_val - 1e-12:
            h = h_candidate
            cur_val = val_candidate
            if val_candidate < best_val:
                best_val = val_candidate
                best_h = h_candidate.copy()
        if best_val <= 0.3808:
            break
    return best_h, best_val

import time, random, math
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto box [0,1]^n with a fixed sum (hyperplane)
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


# --------------------------------------------------------------
# Compute the Erdős overlap constant for a given step function h
def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑2 >= 2n‑1
    H = np.fft.rfft(h, n=L)
    auto_full = np.fft.irfft(H * np.conj(H), n=L)
    # valid shifts are  -(n-1) … (n-1)
    pos = auto_full[:n]                # k = 0 … n‑1
    neg = auto_full[-(n - 1):]         # k = -(n-1) … -1
    min_autocorr = min(pos.min(), neg.min())
    return 1.0 - min_autocorr * dx


# --------------------------------------------------------------
# Analytic seed constructions
def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)     # peak 1 at centre, linearly to 0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rect(n: int) -> np.ndarray:
    h = np.zeros(n)
    block_len = n // 2
    start = (n - block_len) // 2
    h[start:start + block_len] = 1.0
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


# --------------------------------------------------------------
# Sub‑gradient descent for the max‑overlap objective
def subgradient_descent(h0: np.ndarray, max_iter: int, init_step: float):
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    L = 1 << ((2 * n - 1).bit_length())

    for t in range(max_iter):
        # autocorrelation via FFT (real)
        H = np.fft.rfft(h, n=L)
        auto_full = np.fft.irfft(H * np.conj(H), n=L)

        pos = auto_full[:n]                # k = 0 … n‑1
        neg = auto_full[-(n - 1):]         # k = -(n-1) … -1

        min_pos = pos.min()
        min_neg = neg.min()

        if min_pos <= min_neg:
            k = int(pos.argmin())          # non‑negative shift
        else:
            idx = int(neg.argmin())
            k = -((n - 1) - idx)           # negative shift

        shift = abs(k)

        # gradient of the worst term f_k = 1 - Σ h_i h_{i+k}
        g = np.ones_like(h)
        if shift == 0:
            g -= 2.0 * h
        else:
            g[: n - shift] -= h[shift:]
            g[shift:]      -= h[: n - shift]

        step = init_step / math.sqrt(t + 1.0)
        h_candidate = project_onto_bounds_and_sum(h - step * g, target)
        val_candidate = compute_c5(h_candidate)

        if val_candidate < cur_val - 1e-12:
            h = h_candidate
            cur_val = val_candidate
            if val_candidate < best_val:
                best_val = val_candidate
                best_h = h_candidate.copy()

        if best_val <= 0.3808:
            break

    return best_h, best_val


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    n = 500                     # even, good trade‑off between quality and speed
    target_sum = n / 2.0

    # ----- collect diverse starting points ---------------------------------
    seed_vectors = []

    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))
    seed_vectors.append(make_rect(n))

    # a couple of random feasible seeds
    for _ in range(2):
        rnd = np.random.rand(n)
        seed_vectors.append(project_onto_bounds_and_sum(rnd, target_sum))

    best_h = None
    best_val = np.inf

    # ----- main optimisation -------------------------------------------------
    # allocate a generous number of sub‑gradient steps; the loop respects the time budget
    max_iters_per_seed = 8000
    for h0 in seed_vectors:
        if time.time() - start > budget_s - 1.0:
            break
        init_val = compute_c5(h0)
        if init_val < best_val:
            best_val = init_val
            best_h = h0.copy()

        h_opt, val_opt = subgradient_descent(h0, max_iters_per_seed, init_step=1.0)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

        if best_val <= 0.3808:
            break

    # ----- final cheap local refinement --------------------------------------
    while time.time() - start < budget_s - 1.0:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()
            if best_val <= 0.3808:
                break

    # final safety projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
