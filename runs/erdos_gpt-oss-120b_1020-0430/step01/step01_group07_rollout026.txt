```
def _project_feasible(h, target):
    # h may be any floats
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute deficit
    s = h.sum()
    deficit = target - s
    # If deficit approx 0 return
    if abs(deficit) < 1e-12:
        return h
    # distribute plus/minus across free (non-bound) entries
    # We'll do water-filling: iterate.
    max_iter = 10
    for _ in range(max_iter):
        if abs(deficit) < 1e-12:
            break
        free = ((deficit > 0) & (h < 1.0)) | ((deficit < 0) & (h > 0.0))
        idx = np.where(free)[0]
        if idx.size == 0:
            # can't adjust further due to bounds; break
            break
        delta = deficit / idx.size
        # apply delta but ensure still within bounds
        h[idx] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
        deficit = target - h.sum()
    # final fudge if still deficit
    if abs(deficit) > 1e-12:
        # add to random index if possible
        for i in np.random.permutation(len(h)):
            if 0.0 <= h[i] + deficit <= 1.0:
                h[i] += deficit
                break
    return h
```

    return best_h, float(final_c), int(n)

import time
import math
import random
import numpy as np

def _c5_metric(h):
    """C5 value for discrete step function h (dx = 2/n)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _project_feasible(v, target):
    """Project vector v onto {0<=x<=1, sum(x)=target}."""
    v = np.clip(v, 0.0, 1.0)
    deficit = target - v.sum()
    # simple water‑filling
    for _ in range(10):
        if abs(deficit) < 1e-12:
            break
        if deficit > 0:
            free = v < 1.0
        else:
            free = v > 0.0
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = deficit / idx.size
        v[idx] += delta
        v = np.clip(v, 0.0, 1.0)
        deficit = target - v.sum()
    if abs(deficit) > 1e-12:
        for i in np.random.permutation(len(v)):
            if 0.0 <= v[i] + deficit <= 1.0:
                v[i] += deficit
                break
    return v

def _grad_for_shift(h, k):
    """Gradient of correlation for a given shift k (integer, can be negative)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        g[:n - k] += 1.0                # from sum_i h_i
        g[:n - k] -= h[k:]              # -h_{i+k}
        g[k:]   -= h[:n - k]            # -h_{i-k}
    else:
        kp = -k
        g[kp:] += 1.0                   # sum term for i>=kp
        g[kp:] -= h[:n - kp]            # -h_{i-kp}
        g[:n - kp] -= h[kp:]            # -h_{i+kp}
    return g

def _subgradient_descent(start_h, max_seconds):
    """Projected sub‑gradient descent on max‑correlation objective."""
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(h)
    start = time.time()
    step0 = 0.1
    while time.time() - start < max_seconds:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        grad = _grad_for_shift(h, k)
        prog = (time.time() - start) / max_seconds
        step = step0 * (1.0 - prog) + 1e-6
        h_new = _project_feasible(h - step * grad, target)
        new_c = _c5_metric(h_new)
        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, h_new.copy()
        # accept if improvement, otherwise keep previous
        if new_c < _c5_metric(h) - 1e-12:
            h = h_new
        else:
            # occasional random jitter to escape flat regions
            if random.random() < 0.001:
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                lo = max(-hi, hj - 1.0)
                hi_cap = min(1.0 - hi, hj)
                if lo < hi_cap:
                    delta = random.uniform(lo, hi_cap)
                    h[i] = hi + delta
                    h[j] = hj - delta
                    h = _project_feasible(h, target)
    return best_h

def _anneal(start_h, max_seconds):
    """Simple simulated‑annealing refinement."""
    h = start_h.copy()
    n = h.size
    best_h, best_c = h.copy(), _c5_metric(h)
    cur_c = best_c
    start = time.time()
    T0, T_end = 0.04, 1e-6
    max_step = 0.2
    while time.time() - start < max_seconds:
        prog = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** prog
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        lo = max(-hi, hj - 1.0)
        hi_cap = min(1.0 - hi, hj)
        if lo >= hi_cap:
            continue
        lo_c = max(lo, -max_step)
        hi_c = min(hi_cap, max_step)
        delta = random.uniform(lo_c, hi_c) if lo_c < hi_c else random.uniform(lo, hi_cap)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c

def _initial_vector(n):
    """Interpolate provided construction (if any) onto n points."""
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        n0 = base.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

        h = np.interp(x, x0, base)
    else:
        h = np.full(n, 0.5, dtype=np.float64)
    return _project_feasible(h, n / 2.0)

def _random_binary_centered(n, target):
    """Binary vector with half‑ones, biased toward the centre."""
    ones = int(target)
    idx = np.arange(n)
    center = (n - 1) / 2.0
    dist = np.abs(idx - center)
    weights = (n / 2.0) - dist
    weights = np.clip(weights, 0.0, None)
    probs = weights / weights.sum()
    chosen = np.random.choice(n, size=ones, replace=False, p=probs)
    h = np.zeros(n, dtype=np.float64)
    h[chosen] = 1.0
    return h

def _random_binary_uniform(n, target):
    """Pure random binary half‑ones."""
    ones = int(target)
    h = np.zeros(n, dtype=np.float64)
    perm = np.random.permutation(n)
    h[perm[:ones]] = 1.0
    return h

def _triangular_shape(n, target):
    """Piecewise linear triangle (area = 1)."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0 / n)
    h = np.where(xs <= 1.0, xs, 2.0 - xs)
    h = np.clip(h, 0.0, 1.0)
    return _project_feasible(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C5 bound and sample count."""
    random.seed(seed)
    np.random.seed(seed)

    n = 400                     # resolution
    target = n / 2.0

    # --- generate diverse starts ------------------------------------------------
    starts = [
        _initial_vector(n),
        _random_binary_centered(n, target),
        _random_binary_uniform(n, target),
        _triangular_shape(n, target),
        np.full(n, 0.5, dtype=np.float64),
    ]

    # ensure feasibility
    starts = [_project_feasible(h, target) for h in starts]

    total_budget = float(budget_s)
    start_all = time.time()
    phase1 = total_budget * 0.68          # sub‑gradient phase
    per_start = max(phase1 / len(starts), 5.0)

    best_h, best_c = None, float('inf')

    # --- sub‑gradient optimisation on each start -------------------------------
    for h0 in starts:
        if time.time() - start_all >= total_budget * 0.70:
            break
        h_opt = _subgradient_descent(h0, per_start)
        c_val = _c5_metric(h_opt)
        if c_val < best_c:
            best_c, best_h = c_val, h_opt.copy()

    # --- final annealing on the current best ------------------------------------
    remaining = total_budget - (time.time() - start_all)
    if remaining > 10.0 and best_h is not None:
        h_opt, c_val = _anneal(best_h, remaining)
        if c_val < best_c:
            best_c, best_h = c_val, h_opt.copy()

    # --- final projection and evaluation ----------------------------------------
    if best_h is None:
        best_h = _initial_vector(n)
    best_h = _project_feasible(best_h, target)

    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)
