    cur_h = h.copy()

    cur_c = best_c

    start = time.time()

    T0 = 0.1 (starting temperature)
    T_end = 1e-6

    step0 = 0.2

    while time.time() - start < max_seconds:

        elapsed = time.time() - start

        progress = elapsed / max_seconds

        T = T0 * (1 - progress) + T_end * progress   # linear decrease

        max_step = step0 * (1 - progress) + 1e-5 * progress

        i, j = np.random.randint(n, size=2)  # ensure distinct

        if i==j: continue

        hi = cur_h[i]; hj = cur_h[j]

        # compute feasible delta

        low = max(-hi, hj - 1.0)

        high = min(1.0 - hi, hj)

        if low >= high: continue

        # sample delta uniformly within [-max_step, max_step] intersect feasible

        cap_low = max(low, -max_step)

        cap_high = min(high, max_step)

        if cap_low < cap_high:

            delta = np.random.uniform(cap_low, cap_high)

        else:

            delta = np.random.uniform(low, high)

        cur_h[j] = hj - delta

        new_c = metric(cur_h)

                best_c = new_c

                best_h = cur_h.copy()

        else:

            cur_h[i] = hi

import time, math, random
import numpy as np

def _c5_metric(h):
    """C5 metric for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_interpolated(n):
    """Create a feasible start by interpolating the provided construction."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
    return h

def _binary_anneal(start_bin, max_seconds):
    """Simulated annealing on a 0/1 vector (half ones)."""
    n = start_bin.size
    h = start_bin.copy()
    best_h = h.astype(np.float64).copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0, T_end = 0.5, 1e-6
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        T = T0 * (1 - prog) + T_end * prog

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))

        h[i] = 0
        h[j] = 1
        new_c = _c5_metric(h.astype(np.float64))
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.astype(np.float64).copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c

def _run_binary_phase(n, time_budget):
    """Run several short binary anneals and keep the best."""
    best_h, best_c = None, np.inf
    start = time.time()
    while time.time() - start < time_budget:
        bin_vec = np.zeros(n, dtype=np.int8)
        ones = n // 2
        idx = np.random.permutation(n)[:ones]
        bin_vec[idx] = 1
        remain = time_budget - (time.time() - start)
        slice_t = min(20.0, remain)
        h_res, c_res = _binary_anneal(bin_vec, slice_t)
        if c_res < best_c:
            best_c, best_h = c_res, h_res.copy()
        if time.time() - start > time_budget - 5.0:
            break
    return best_h, best_c

def _continuous_targeted_anneal(start_h, max_seconds):
    """Continuous annealing biased towards the current worst shift."""
    n = start_h.size
    dx = 2.0 / n
    target = n / 2.0

    h = start_h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c

    start = time.time()
    T0, T_end = 0.1, 1e-6
    max_step0 = 0.15

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        T = T0 * (1 - prog) + T_end * prog
        max_step = max_step0 * (1 - prog) + 1e-5 * prog

        idx_max = corr.argmax()
        shift = idx_max - (n - 1)

        # build shifted version of h for this shift
        if shift >= 0:
            h_shift = np.empty_like(h)
            h_shift[:n - shift] = h[shift:]
            h_shift[n - shift:] = 0.0
        else:
            sh = -shift
            h_shift = np.empty_like(h)
            h_shift[sh:] = h[:n - sh]
            h_shift[:sh] = 0.0

        contrib = h * (1.0 - h_shift)

        if np.any(contrib > 0):
            i = int(np.argmax(contrib))
        else:
            i = random.randrange(n)

        # pick j where (1 - h_shift) is small and there is capacity
        target = 1.0 - h_shift
        cap_mask = h < 1.0
        if cap_mask.any():
            tm = target.copy()
            tm[~cap_mask] = np.inf
            j = int(np.argmin(tm))
        else:
            j = (i + 1) % n

        if i == j:
            cand = np.where(cap_mask & (np.arange(n) != i))[0]
            if cand.size == 0:
                continue
            j = int(np.random.choice(cand))

        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        lo_cap = max(low, -max_step)
        hi_cap = min(high, max_step)
        if lo_cap < hi_cap:
            delta = random.uniform(lo_cap, hi_cap)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta

        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_c = new_corr.max() * dx
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            corr = new_corr
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c

def _final_refinement(h, time_budget):
    """Tiny random redistributions to polish the solution."""
    n = h.size
    best_h, best_c = h.copy(), _c5_metric(h)
    start = time.time()
    while time.time() - start < time_budget:
        i, j = random.sample(range(n), 2)
        hi, hj = best_h[i], best_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.02
        best_h[i] = hi + delta
        best_h[j] = hj - delta
        new_c = _c5_metric(best_h)
        if new_c + 1e-9 < best_c:
            best_c = new_c
        else:
            best_h[i] = hi
            best_h[j] = hj
    return best_h, best_c
