import time, random, math, numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """C5 value for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _greedy_binary(start_h: np.ndarray, time_limit: float):
    """Monotone greedy improvement for binary vectors."""
    n = start_h.size
    h = start_h.copy()
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    end = time.time() + time_limit

    while time.time() < end:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        if max_val == 0:
            break
        shift = max_idx - (n - 1)

        # overlapping region for this shift
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n
        if i_start >= i_end:
            continue

        i_range = np.arange(i_start, i_end, dtype=np.int64)
        j_range = i_range + shift
        hi = h[i_range]
        hj = h[j_range]

        # positions that contribute to the worst overlap (1→0) and the opposite (0→1)
        A = i_range[(hi == 1) & (hj == 0)]
        C = i_range[(hi == 0) & (hj == 1)]

        moved = False
        if A.size and C.size:
            i_swap = int(np.random.choice(A))
            j_swap = int(np.random.choice(C))
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if not ones.size or not zeros.size:
                break
            i_swap = int(np.random.choice(ones))
            j_swap = int(np.random.choice(zeros))

        # remember old values for a possible revert
        hi_old, hj_old = h[i_swap], h[j_swap]
        h[i_swap], h[j_swap] = 0.0, 1.0
        new_c = _c5_metric(h)

        if new_c <= cur_c + 1e-12:          # accept non‑worsening move
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_c, best_h = new_c, h.copy()
        else:
            h[i_swap], h[j_swap] = hi_old, hj_old   # revert

    return best_h, best_c


def _binary_anneal(start_h: np.ndarray, time_limit: float):
    """Simulated‑annealing on binary vectors (pair swaps)."""
    n = start_h.size
    h = start_h.copy()
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    start = time.time()
    T0, T_end = 0.20, 1e-5

    while time.time() - start < time_limit:
        progress = (time.time() - start) / time_limit
        T = T0 * (T_end / T0) ** progress

        # pick a random 1 and a random 0
        i = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)

        # swap
        h[i], h[j] = 0.0, 1.0
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

    return best_h, best_c
