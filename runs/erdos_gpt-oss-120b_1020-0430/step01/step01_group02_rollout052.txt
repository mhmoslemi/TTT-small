import time
import numpy as np

def _evaluate_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """Project x onto the hyperplane sum(x)=target while respecting 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    # iterative redistribution
    while abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / idx.size
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return np.clip(x, 0.0, 1.0)

def _shifts_matrix(h):
    """Return matrix S where S[i,k]=h[k-i] (zero outside)."""
    n = h.size
    L = 2 * n - 1
    pad = np.concatenate((np.zeros(n - 1, dtype=h.dtype), h, np.zeros(n - 1, dtype=h.dtype)))
    base = n - 1
    S = np.empty((n, L), dtype=h.dtype)
    for i in range(n):
        start = base - i
        S[i] = pad[start:start + L]
    return S

def _greedy_optimize(h, target, deadline, rng):
    """Monotone descent moving full transferable mass between points."""
    n = h.size
    cur_h = h.copy()
    while time.time() < deadline:
        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        cur_max = conv.max()
        S = _shifts_matrix(cur_h)
        # candidates
        i_cand = np.where(cur_h > 1e-12)[0]
        j_cand = np.where(cur_h < 1.0 - 1e-12)[0]
        if i_cand.size == 0 or j_cand.size == 0:
            break
        best_new = cur_max
        best_move = None
        for i in i_cand:
            si = S[i]
            eps_i = cur_h[i]
            for j in j_cand:
                if i == j:
                    continue
                eps_j = 1.0 - cur_h[j]
                eps = eps_i if eps_i < eps_j else eps_j
                if eps <= 0:
                    continue
                d = si - S[j]
                new_max = np.max(conv + 2.0 * eps * d)
                if new_max < best_new - 1e-14:
                    best_new = new_max
                    best_move = (i, j, eps)
        if best_move is None:
            break
        i, j, eps = best_move
        cur_h[i] -= eps
        cur_h[j] += eps
    return cur_h, _evaluate_c5(cur_h)

def _hill_climb(h, cur_c5, deadline, rng):
    """Simple random pairwise mass‑exchange hill climbing."""
    n = h.size
    best_h, best_c5 = h.copy(), cur_c5
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        if best_h[i] <= 1e-12 or best_h[j] >= 1.0 - 1e-12:
            continue
        max_move = min(best_h[i], 1.0 - best_h[j])
        eps = rng.random() * max_move
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c5 = _evaluate_c5(new_h)
        if new_c5 + 1e-12 < best_c5:
            best_h, best_c5 = new_h, new_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    deadline = start + budget_s - 0.05  # tiny safety margin

    init_vec = globals().get("initial_h_values")
    n0 = len(init_vec) if init_vec is not None else None

    # candidate resolutions (try a few, primary is the provided or 45)
    candidates = []
    if n0 is not None:
        candidates.append(n0)
    for n in (45, 48, 60, 72):
        if n not in candidates:
            candidates.append(n)

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in candidates:
        if time.time() >= deadline - 0.5:
            break
        target = n / 2.0

        # first starting vector
        if n0 is not None and n == n0:
            h0 = np.asarray(init_vec, dtype=np.float64)
        else:
            h0 = rng.random(n)
            h0 = _adjust_sum(h0, target)
        h0 = _adjust_sum(np.clip(h0, 0.0, 1.0), target)

        # run greedy descent on the start
        h_opt, c5_opt = _greedy_optimize(h0, target, deadline, rng)

        # a couple of perturbed restarts (if time permits)
        for _ in range(3):
            if time.time() >= deadline - 0.5:

            perturb = rng.normal(scale=0.05, size=n)
            h_start = h_opt + perturb
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = _adjust_sum(h_start, target)
            h_tmp, c5_tmp = _greedy_optimize(h_start, target, deadline, rng)
            if c5_tmp < c5_opt:
                h_opt, c5_opt = h_tmp, c5_tmp

        # final polish with hill climbing (short time slice)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

    # final sanity projection
    if best_h is None:

        best_h = np.full(n, 0.5, dtype=np.float64)
        best_h = _adjust_sum(best_h, target)
        best_c5 = _evaluate_c5(best_h)
        best_n = n

    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, best_n
