import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {0≤x≤1, sum(x)=target} via bisection on the dual variable."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the max-correlation term for a given integer shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 800,
                 step0: float = 0.5, decay: float = 0.995) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑correlation objective."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    def project(v):
        return _project_onto_capped_simplex(v, target_sum)

    # ---------------------------------------------------------------
    # Particle Swarm Optimisation
    # ---------------------------------------------------------------
    n_particles = 60
    positions = np.empty((n_particles, n))
    velocities = np.zeros((n_particles, n))
    for i in range(n_particles):
        if base_h is not None and i == 0:
            positions[i] = project(upscale(base_h))
        else:
            positions[i] = _random_feasible(rng, n, target_sum)
        velocities[i] = (rng.rand(n) - 0.5) * 0.1  # modest start

    vals = np.empty(n_particles)
    for i in range(n_particles):
        vals[i] = _compute_c5(positions[i], dx)

    pbest_pos = positions.copy()
    pbest_val = vals.copy()
    best_idx = int(np.argmin(pbest_val))
    gbest_pos = pbest_pos[best_idx].copy()
    gbest_val = float(pbest_val[best_idx])

    w = 0.7298
    c1 = 1.49618
    c2 = 1.49618
    vmax = 0.2

    # main PSO loop
    while time.time() - start < budget_s * 0.85:
        for i in range(n_particles):
            r1 = rng.rand(n)
            r2 = rng.rand(n)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest_pos[i] - positions[i]) +
                             c2 * r2 * (gbest_pos - positions[i]))
            velocities[i] = np.clip(velocities[i], -vmax, vmax)
            new_pos = positions[i] + velocities[i]
            new_pos = project(new_pos)
            positions[i] = new_pos
            val = _compute_c5(new_pos, dx)
            if val < pbest_val[i]:
                pbest_val[i] = val
                pbest_pos[i] = new_pos.copy()
                if val < gbest_val:
                    gbest_val = val
                    gbest_pos = new_pos.copy()

    # ---------------------------------------------------------------
    # Local refinement with gradient descent if time permits
    # ---------------------------------------------------------------
    time_left = budget_s - (time.time() - start)
    if time_left > 6.0:
        # allocate about 20 GD steps per second of remaining time
        gd_iters = min(2000, int(time_left * 20))
        refined = _gd_optimize(gbest_pos, target_sum,
                               max_iter=gd_iters,
                               step0=0.8, decay=0.997)
        refined = project(refined)
        refined_val = _compute_c5(refined, dx)
        if refined_val < gbest_val:
            gbest_val = refined_val
            gbest_pos = refined

    # ---------------------------------------------------------------
    # Final projection & evaluation
    # ---------------------------------------------------------------
    best_h = project(gbest_pos)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
