def _hill_climb_targeted(h, c5, deadline, rng, target):
    n = len(h)
    while time.time() < deadline:
        # compute raw correlation
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr_raw))
        l = max_idx - (n - 1)  # shift offset
        # compute contributions
        if l >= 0:
            i_vals = np.arange(0, n - l)
            contrib = h[i_vals] * (1.0 - h[i_vals + l])
        else:
            i_vals = np.arange(-l, n)
            contrib = h[i_vals] * (1.0 - h[i_vals + l])
        if contrib.size > 0:
            i_max = int(np.argmax(contrib))
            i = int(i_vals[i_max])
            j = i + l
            # Determine direction: move from larger h to smaller h
            if h[i] > h[j] + 1e-12:
                src, dst = i, j
            elif h[j] > h[i] + 1e-12:
                src, dst = j, i
            else:
                # fallback to random
                src, dst = i, j
        else:
            # fallback random
            src = rng.integers(n)
            dst = rng.integers(n)
        # Ensure we can transfer
        if src == dst:
            continue
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            # random fallback
            # find random src with positive mass, dst with capacity
            src_candidates = np.where(h > 1e-12)[0]
            dst_candidates = np.where(h < 1.0 - 1e-12)[0]
            if src_candidates.size == 0 or dst_candidates.size == 0:
                break
            src = int(rng.choice(src_candidates))
            dst = int(rng.choice(dst_candidates))
            max_move = min(h[src], 1.0 - h[dst])
            if max_move <= 1e-12:
                continue
        delta = rng.uniform(0.0, max_move)
        new_h = h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        new_c5 = _evaluate_c5(new_h)
        if new_c5 + 1e-12 < c5:
            h, c5 = new_h, new_c5
    return h, c5

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x, target):
    """Project x onto {0 ≤ x ≤ 1, sum(x)=target}."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    # iterative redistribution
    while abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return np.clip(x, 0.0, 1.0)


def _propose_move(h, c_raw, rng):
    """
    Propose a mass‑transfer move.
    With probability 0.7 we use the shift that gives the current maximal overlap,
    otherwise we pick a random feasible pair.
    Returns (src, dst, delta) or None if no feasible move exists.
    """
    n = len(h)
    offset = n - 1
    max_idx = int(np.argmax(c_raw))
    l = max_idx - offset

    use_targeted = rng.random() < 0.7
    src = dst = None

    if use_targeted:
        # contributions for the offending shift
        if l >= 0:
            i_vals = np.arange(0, n - l, dtype=int)
            contrib = h[i_vals] * (1.0 - h[i_vals + l])
        else:
            i_vals = np.arange(-l, n, dtype=int)
            contrib = h[i_vals] * (1.0 - h[i_vals + l])
        if contrib.size > 0:
            i_best = int(i_vals[np.argmax(contrib)])
            j_best = i_best + l
            if h[i_best] > h[j_best] + 1e-12:
                src, dst = i_best, j_best
            elif h[j_best] > h[i_best] + 1e-12:
                src, dst = j_best, i_best
            else:
                # fallback to a random feasible pair
                src = i_best
                dst = j_best

    # if targeted failed or was not used, fall back to a random pair
    if src is None or dst is None or src == dst:
        # list of possible sources and destinations
        possible_src = np.where(h > 1e-12)[0]
        possible_dst = np.where(h < 1.0 - 1e-12)[0]
        if possible_src.size == 0 or possible_dst.size == 0:
            return None
        src = int(rng.choice(possible_src))
        dst = int(rng.choice(possible_dst))
        while dst == src:
            dst = int(rng.choice(possible_dst))

    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-12:
        return None
    delta = rng.uniform(0.0, max_move)
    return src, dst, delta

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    it = 0
    while time.time() < slsq_deadline:
        # jitter around the current best, occasionally strong perturbation
        if it % 7 == 0:
            x0 = best_h + rng.normal(scale=0.09, size=n)
        else:
            x0 = best_h + rng.normal(scale=0.03, size=n)
        x0 = np.clip(x0, 0.0, 1.0)
        x0 = _adjust_sum(x0, target)

        res = minimize(
            obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 250, "ftol": 1e-9, "disp": False},
        )
        cand = _adjust_sum(np.clip(res.x, 0.0, 1.0) if res.success else x0, target)
        cand_c5 = _evaluate_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_c5
        it += 1

    # ------------------------------------------------------------------
    # 3. Focused hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    hill_deadline = start_time + budget_s - 0.5
    cur_h = best_h.copy()
    cur_c5 = best_c5
    best_local_h = cur_h.copy()
    best_local_c5 = cur_c5

    # initial correlation array for proposal logic
    c_raw = np.correlate(cur_h, 1.0 - cur_h, mode="full")

    while time.time() < hill_deadline:
        prop = _propose_move(cur_h, c_raw, rng)
        if prop is None:
            continue
        src, dst, delta = prop

        # apply move
        cur_h[src] -= delta
        cur_h[dst] += delta
        c_raw = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        new_c5 = np.max(c_raw) * dx

        # simulated‑annealing acceptance
        elapsed = time.time() - start_time
        T = 0.02 * max(0.0, 1.0 - elapsed / budget_s) + 0.0001
        accept = False
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        elif rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 + 1e-12 < best_local_c5:
                best_local_c5 = new_c5
                best_local_h = cur_h.copy()
        else:
            # revert move
            cur_h[src] += delta
            cur_h[dst] -= delta
            c_raw = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            # cur_c5 unchanged

    # adopt best local solution if it improved the global best
    if best_local_c5 + 1e-12 < best_c5:
        best_h, best_c5 = best_local_h, best_local_c5

    # ------------------------------------------------------------------
    # 4. Final feasibility check and return
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
