        if delta <= 1e-12:

        # compute net gain

def hill_climb(h, time_left, max_iter=5000, step_init=0.1):
    n = h.size
    best_h = h.copy()
    best_min_corr = np.min(compute_self_corr(h))
    step = step_init
    iter_count = 0
    while time_left() > 0.05 and iter_count < max_iter:
        # compute correlation
        corr = np.correlate(h, h, mode='full')
        min_corr = corr.min()
        # If no improvement for a while, break
        # find shift index(s)
        idx_min = np.argmin(corr)  # first occurrence
        shift = idx_min - (n - 1)
        m = abs(shift)
        if m == 0:
            # shift 0 is max; need next min
            # find second smallest
            # We could set m=1 default
            m = 1
        # compute gain vector
        gain = np.zeros(n)
        if m < n:
            # forward contributions
            gain[:n-m] += h[m:]
            # backward contributions
            gain[m:] += h[:n-m]
        # Determine i (increase) and j (decrease)
        inc_mask = h < 1 - 1e-12
        dec_mask = h > 1e-12
        # restrict to allowed
        if not np.any(inc_mask) or not np.any(dec_mask):
            break
        # For i we want max gain among inc_mask
        # set gain where not allowed to -inf
        gain_inc = gain.copy()
        gain_inc[~inc_mask] = -np.inf
        i = np.argmax(gain_inc)
        # For j we want min gain among dec_mask
        gain_dec = gain.copy()
        gain_dec[~dec_mask] = np.inf
        j = np.argmin(gain_dec)
        if i == j:
            # try alternative selections
            # find second best for i or j
            # pick next best inc index
            inc_idx = np.where(inc_mask)[0]
            dec_idx = np.where(dec_mask)[0]
            if inc_idx.size <= 1 or dec_idx.size <= 1:
                break
            # sort inc candidates by gain descending
            inc_order = inc_idx[np.argsort(-gain[inc_idx])]
            dec_order = dec_idx[np.argsort(gain[dec_idx])]
            # attempt pairs
            found = False
            for ii in inc_order:
                for jj in dec_order:
                    if ii != jj:
                        i, j = ii, jj
                        found = True
                        break
                if found:
                    break
            if not found:
                break
        # compute delta
        delta = min(step, 1 - h[i], h[j])
        if delta <= 1e-12:
            # reduce step
            step *= 0.5
            if step < 1e-5:
                break
            continue
        net_gain = (gain[i] - gain[j]) * delta
        if net_gain <= 1e-12:
            # cannot improve for this shift; reduce step
            step *= 0.5
            if step < 1e-5:
                break
            continue
        # apply transfer
        h[i] += delta
        h[j] -= delta
        # ensure feasibility
        # Already maintain sum constant; but rounding may cause small error; we can project occasionally
        if (iter_count % 100) == 0:
            h = _project_to_feasible(h, n)
        # compute new min_corr
        new_min_corr = np.min(np.correlate(h, h, mode='full'))
        if new_min_corr > best_min_corr + 1e-12:
            best_min_corr = new_min_corr
            best_h = h.copy()
        # maybe increase step if improvement
        # warm step for next iteration
        iter_count += 1
    # final projection
    best_h = _project_to_feasible(best_h, n)
    return best_h, best_min_corr

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Get initial h if available
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # candidate sizes
    candidates = []
    if init_h is not None:
        n0 = init_h.size
        candidates.append(n0)
        if n0*2 <= 300:
            candidates.append(n0*2)
    for n in (200, 256, 300):
        if n not in candidates and n <= 300:
            candidates.append(n)
    candidates = sorted(set(candidates))

    best_h = None
    best_c5 = float('inf')
    best_n = None
    # main loops
    for n in candidates:
        if time_left() < 5:
            break
        # Build base h
        if init_h is not None:
            if init_h.size != n:
                base_h = _upscale_h(init_h, n)
            else:
                base_h = init_h.copy()
        else:
            base_h = np.full(n, 0.5)
        # optional: ensure feasibility
        base_h = _project_to_feasible(base_h, n)

        # Determine time budget for this n
        rem_n_candidates = len(candidates) - candidates.index(n)
        time_for_n = max(5, time_left() / (rem_n_candidates+1))
        # But we limit to maybe 100 seconds for n=256 etc.

        # Hill climbing attempts
        attempts = 3
        for attempt in range(attempts):
            if time_left() < 5:
                break
            # choose starting h
            if attempt == 0:
                h_start = base_h.copy()
            else:
                # random binary or continuous start
                if np.random.rand() < 0.5:
                    # binary random
                    h_start = np.zeros(n)
                    ones_idx = np.random.choice(n, n//2, replace=False)
                    h_start[ones_idx] = 1.0
                else:
                    # continuous random
                    h_start = np.random.rand(n)
                    h_start = _project_to_feasible(h_start, n)
            # Run hill climb with time limit per attempt = min(time_for_n/attempts, time_left())
            time_budget_attempt = min(time_for_n/attempts, time_left())
            # define time_left_inner based on remaining time in this attempt
            start_attempt = time.time()
            def inner_time_left():
                return time_budget_attempt - (time.time() - start_attempt)

            h_opt, min_corr = _hill_climb(h_start, inner_time_left, max_iter=2000, step_init=0.2)
            # compute c5
            c5_val = 1 - (2.0 / n) * min_corr  # derived
            # optionally refine with SLSQP
            if c5_val < best_c5:
                # run SLSQP to potentially improve
                h_refined, c5_refined = _solve_h(h_opt, max_iter=250)
                if c5_refined < c5_val:
                    c5_val = c5_refined
                    h_opt = h_refined
                # update global best
                best_h = h_opt.copy()
                best_c5 = c5_val
                best_n = n
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # fallback if not found
    if best_h is None:
        n = 64 if init_h is None else init_h.size
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    # final projection just in case
    best_h = _project_to_feasible(best_h, best_n)
    # compute final C5 accurately
    final_c5 = _c5_from_h(best_h)
    return best_h.tolist(), float(final_c5), int(best_n)

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip values to [0,1] and adjust them so that sum(h) = n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # Which entries can be changed?
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discretised h (max correlation with 1‑h)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """SLSQP refinement for a given starting h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

        idx = np.argmax(corr)

        k = abs(shift)

        # gradient of the worst correlation w.r.t. h
        ind = (np.arange(n) <= n - 1 - k).astype(float)
        forward = np.zeros(n)
        backward = np.zeros(n)
        if k < n:
            forward[: n - k] = h[k:]
            backward[k:] = h[: n - k]
        grad = ind - forward - backward

        # step size (sub‑gradient schedule)
        alpha = step0 / np.sqrt(it)

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_feasible(h_new, n)

        c5 = _c5_from_h(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

        h = h_new

    return best_h, best_c5


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Retrieve optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        init_h = None

    # ------------------------------------------------------------------
    # Candidate grid sizes (even only, ≤300)
    # ------------------------------------------------------------------
    candidates = []
    if init_h is not None:
        n0 = init_h.size
        if n0 % 2 == 0:
            candidates.append(n0)
            if n0 * 2 <= 300:
                candidates.append(n0 * 2)

    # add a few handcrafted even sizes
    for n in (200, 224, 256, 300):
        if n % 2 == 0:
            candidates.append(n)
    candidates = sorted(set(candidates))

    best_h, best_c5, best_n = None, float("inf"), None
    target = 0.3808  # early‑stop target

    # ------------------------------------------------------------------
    # Main search loop over grid sizes
    # ------------------------------------------------------------------
    for idx_n, n in enumerate(candidates):
        if time_left() < 5:
            break

        # construct a base h for this size
        if init_h is not None:
            if init_h.size != n:
                base_h = _upscale_h(init_h, n)
            else:
                base_h = init_h.copy()
        else:
            base_h = np.full(n, 0.5)
        base_h = _project_to_feasible(base_h, n)

        # Number of restarts for this size
        restarts = 4
        for r in range(restarts):
            if time_left() < 5:
                break

            # starting point
            if r == 0:
                h_start = base_h.copy()
            else:
                # mix of random binary / continuous starts
                if np.random.rand() < 0.6:
                    # binary with exactly n/2 ones
                    h_start = np.zeros(n)
                    ones = np.random.choice(n, n // 2, replace=False)
                    h_start[ones] = 1.0
                else:
                    h_start = np.random.rand(n)
                    h_start = _project_to_feasible(h_start, n)

            # allocate a slice of the remaining budget to this attempt
            attempt_budget = min(time_left() / (restcasts:= (len(candidates) - idx_n) + (restarts - r)), time_left())
            attempt_start = time.time()

            def inner_time():
                return attempt_budget - (time.time() - attempt_start)

            # ----- sub‑gradient phase -----
            h_opt, c5_val = _subgradient_optimize(h_start, inner_time)

            # ----- optional local refinement -----
            if c5_val > 0.35:  # cheap guard – only refine if still relatively large
                h_opt, c5_ref = _solve_h(h_opt, max_iter=180)
                c5_val = min(c5_val, c5_ref)

            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_val, n
                if best_c5 < target:
                    break
        if best_c5 < target:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 64
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    # final projection & evaluation for safety
    best_h = _project_to_feasible(np.array(best_h, dtype=float), best_n)
    final_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(final_c5), int(best_n)
