def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    best_h = None
    best_c5 = float('inf')
    best_n = None

    for n in [200, 180, 220]:
        if time.time() > deadline:
            break
        target_sum = n/2.0
        # attempt binary SA
        # Generate initial seeds
        seeds = []
        # use parent if available
        try:
            parent_h = np.asarray(initial_h_values, dtype=float)
            # upsample if needed
            if parent_h.shape[0] != n:
                # upsample using linear interpolation then round
                up = upsample(parent_h, n)   # using function we define
                # round to binary
                bin_h = (up >= 0.5).astype(float)
                # adjust to exact sum
                cur_sum = bin_h.sum()
                if cur_sum > target_sum:
                    idx = np.where(bin_h == 1)[0]
                    flip = rng.choice(idx, int(cur_sum - target_sum), replace=False)
                    bin_h[flip] = 0
                elif cur_sum < target_sum:
                    idx = np.where(bin_h == 0)[0]
                    flip = rng.choice(idx, int(target_sum - cur_sum), replace=False)
                    bin_h[flip] = 1
                seeds.append(bin_h)
            else:
                # round directly
                bin_h = (parent_h >= 0.5).astype(float)
                # adjust similarly
                cur_sum = bin_h.sum()
                if cur_sum > target_sum:
                    idx = np.where(bin_h == 1)[0]
                    flip = rng.choice(idx, int(cur_sum - target_sum), replace=False)
                    bin_h[flip] = 0
                elif cur_sum < target_sum:
                    idx = np.where(bin_h == 0)[0]
                    flip = rng.choice(idx, int(target_sum - cur_sum), replace=False)
                    bin_h[flip] = 1
                seeds.append(bin_h)
        except NameError:
            parent_h = None

        # random binary seeds
        for _ in range(5):
            idx = rng.choice(n, size=int(target_sum), replace=False)
            h = np.zeros(n, dtype=float)
            h[idx] = 1.0
            seeds.append(h)

        # for each seed run SA
        for seed_h in seeds:
            if time.time() > deadline:
                break
            # local SA particularly for binary
            h_cur = seed_h.copy()
            c_cur = compute_c5(h_cur)
            # update best
            if c_cur < best_c5:
                best_h, best_c5, best_n = h_cur.copy(), c_cur, n

            # limit number of iterations to something like 5000 per start
            max_iter = 10000
            temp = 0.05
            decay = 0.9995
            it = 0
            while it < max_iter and time.time() < deadline:
                # maybe early break if c_cur already low enough (<0.3805)
                # propose swap

                zeros_idx = np.where(h_cur == 0)[0]

                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                c_new = compute_c5(h_new)
                if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(temp, 1e-12)):
                    h_cur = h_new
                    c_cur = c_new
                    if c_cur < best_c5 - 1e-12:
                        best_h, best_c5, best_n = h_cur.copy(), c_cur, n

                               constraints=cons,

                if not res.success:

                h_ref = np.clip(res.x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 value for a step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, renormalised to integral 1."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def binary_from_continuous(v: np.ndarray, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Round a continuous vector to {0,1} while preserving the exact sum."""
    b = (v >= 0.5).astype(float)
    cur = b.sum()
    if cur > target_sum:
        idx = np.where(b == 1)[0]
        flip = rng.choice(idx, int(cur - target_sum), replace=False)
        b[flip] = 0.0
    elif cur < target_sum:
        idx = np.where(b == 0)[0]
        flip = rng.choice(idx, int(target_sum - cur), replace=False)
        b[flip] = 1.0
    return b


# ------------------------------------------------------------
# Core optimisation
# ------------------------------------------------------------
def simulated_annealing_binary(start_h: np.ndarray, budget_deadline: float,
                               rng: np.random.Generator) -> tuple:
    """Binary SA that respects the sum constraint."""
    n = start_h.shape[0]
    target = n / 2.0
    h_cur = start_h.copy()
    c_cur = compute_c5(h_cur)

    best_h, best_c = h_cur.copy(), c_cur

    temp = 0.07               # initial temperature (in C5 units)
    decay = 0.9995
    it = 0
    max_iter = 20000

    while it < max_iter and time.time() < budget_deadline:
        # pick a 1 and a 0 to swap
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h_new = h_cur.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c_new = compute_c5(h_new)

        if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(temp, 1e-12)):
            h_cur, c_cur = h_new, c_new
            if c_cur < best_c - 1e-12:
                best_h, best_c = h_cur.copy(), c_cur

        temp *= decay
        it += 1

    return best_h, best_c


def refine_continuous(start_h: np.ndarray, target_sum: float,
                      deadline: float) -> tuple:
    """Sequential SLSQP refinement with increasing softmax sharpness."""
    h = start_h.copy()
    alphas = [5, 15, 50, 150, 400]
    for a in alphas:
        if time.time() >= deadline:
            break
        cons = {"type": "eq", "fun": lambda x, t=target_sum: np.sum(x) - t}
        bounds = [(0.0, 1.0)] * h.shape[0]
        res = minimize(lambda x, aa=a: softmax_c5(x, aa), h,
                       method="SLSQP", bounds=bounds, constraints=cons,
                       options={"maxiter": 500, "ftol": 1e-9, "disp": False})
        if not res.success:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        # enforce the sum exactly (tiny numerical drift)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            h += diff / h.shape[0]
            h = np.clip(h, 0.0, 1.0)
    c = compute_c5(h)
    return h, c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95   # safety margin

    best_h, best_c5, best_n = None, np.inf, None

    # Choose a single resolution that offers fine granularity.
    # 200 points give a step of 0.01 in C5 which matches the target.
    n = 200
    target_sum = n / 2.0

    # -----------------------------------------------------------------
    # Build initial seeds
    # -----------------------------------------------------------------
    seeds = []

    # 1. Parent (if present) – both continuous and binary versions
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.shape[0] != n:
            parent_up = upsample(parent, n)
        else:
            parent_up = parent.copy()
        seeds.append(parent_up)                     # continuous start
        seeds.append(binary_from_continuous(parent_up, target_sum, rng))  # binary start
    except NameError:
        pass

    # 2. Random binary seeds
    for _ in range(6):
        idx = rng.choice(n, size=int(target_sum), replace=False)
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        seeds.append(h)

    # 3. Simple deterministic shapes (triangular, block)
    # triangular approximated then rounded to binary
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    tri = tri * (target_sum / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    tri = (tri >= 0.5).astype(float)
    # enforce exact sum
    cur = tri.sum()
    if cur > target_sum:
        idx = np.where(tri == 1)[0]
        tri[rng.choice(idx, int(cur - target_sum), replace=False)] = 0.0
    elif cur < target_sum:
        idx = np.where(tri == 0)[0]
        tri[rng.choice(idx, int(target_sum - cur), replace=False)] = 1.0
    seeds.append(tri)

    # -----------------------------------------------------------------
    # Main optimisation loop over seeds
    # -----------------------------------------------------------------
    for s in seeds:
        if time.time() >= deadline:
            break

        # ----- binary simulated annealing -----
        if np.all((s == 0) | (s == 1)):
            h_bin, c_bin = simulated_annealing_binary(s, deadline, rng)
        else:
            # round to binary first
            bin_start = binary_from_continuous(s, target_sum, rng)
            h_bin, c_bin = simulated_annealing_binary(bin_start, deadline, rng)

        if c_bin < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_bin.copy(), c_bin, n

        # ----- continuous refinement (SLSQP) -----
        if time.time() >= deadline:
            break
        h_ref, c_ref = refine_continuous(h_bin, target_sum, deadline)
        if c_ref < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_ref.copy(), c_ref, n

        # also try refining the original continuous seed directly
        if not np.all((s == 0) | (s == 1)):
            if time.time() >= deadline:
                break
            h_ref2, c_ref2 = refine_continuous(s, target_sum, deadline)
            if c_ref2 < best_c5 - 1e-12:
                best_h, best_c5, best_n = h_ref2.copy(), c_ref2, n

    # -----------------------------------------------------------------
    # Fallback in the unlikely case nothing succeeded
    # -----------------------------------------------------------------
    if best_h is None:
        n = 100
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_h, best_c5, best_n = h, compute_c5(h), n

    # -----------------------------------------------------------------
    # Final sanity check: enforce sum exactly
    # -----------------------------------------------------------------
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
