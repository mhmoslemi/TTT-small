def upsample(arr, new_n):
    old_n = len(arr)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n  # center points
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    up = np.interp(new_x, old_x, arr)
    # ensure sum exactly new_n/2
    target = new_n/2.0
    up = up * (target / up.sum())
    # clip to [0,1]
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

def improve_local(h_initial, time_deadline, rng):
    h = h_initial.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    # We'll run for loops:
    while time.time() < time_deadline - 0.05:
        # compute correlation conv and find best shift(s)
        conv_full = np.correlate(h, 1-h, mode='full')
        # C5 = max*dx
        # find index of maximum
        max_idx = np.argmax(conv_full)
        # Determine shift L: L = max_idx - (n-1)
        L = max_idx - (n-1)
        # compute overlapping index ranges:
        if L >= 0:
            i_range = np.arange(0, n-L)
            j_range = i_range + L
        else:
            i_range = np.arange(-L, n)
            j_range = i_range + L  # which is smaller
        # compute diff array where h_i > h_j
        diff = h[i_range] - h[j_range]
        # pick indices where diff > tolerance
        pos_idxs = np.where(diff > 1e-9)[0]
        if pos_idxs.size > 0:
            # choose random index among positive diffs
            idx = rng.choice(pos_idxs)
            i = i_range[idx]
            j = j_range[idx]
            # compute possible delta
            max_delta = min(h[i], 1.0 - h[j], diff/2)
            # if max_delta small, maybe skip
            if max_delta > 1e-6:
                # choose delta as random fraction up to max_delta
                delta = max_delta * rng.random()
                # apply
                h[i] -= delta
                h[j] += delta
                # evaluate new C5
                new_c5 = compute_c5(h)
                # accept only if not worse
                if new_c5 <= best_c5 + 1e-12:
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
                    # optionally try random move or keep
                    # continue loop
                    continue
            else:
                # cannot move, maybe skip
                pass
        else:
            # no positive diff for this shift; random generic move
            # choose random i and j distinct uniform
            i, j = rng.choice(n, 2, replace=False)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:

                best_h = h.copy()
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        # Optionally also attempt reverse move (swap from lower to higher) probabilistically for exploration
    return best_h, best_c5

generate_triangular(n):
    x = np.linspace(0,2,n,endpoint=False) + 1/n   # centers
    tri = np.minimum(x, 2-x)
    # Scale to sum = n/2
    target = n/2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0, 1)
    # adjust diff
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0, 1)
    return tri

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x), sampled on n points and normalized."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (box constraints + sum)."""
    target = n / 2.0
    a = rng.random(n)
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    # fix the sum exactly
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a

def improve_local(start_h: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple:
    """
    Greedy pairwise mass‑exchange search guided by the worst overlap shift.
    Returns best (h, C5) found before deadline.
    """
    h = start_h.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    # parameters
    no_improve_limit = 20000
    no_improve_cnt = 0

    while time.time() < deadline - 0.01:
        # find shift with maximal overlap
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(conv))
        L = max_idx - (n - 1)            # lag corresponding to the max

        # overlapping index pairs (i, j=i+L) or (i-L, i) depending on sign
        if L >= 0:
            i_idx = np.arange(0, n - L)
            j_idx = i_idx + L
        else:
            i_idx = np.arange(-L, n)
            j_idx = i_idx + L

        # look for a pair where values differ
        diff = h[i_idx] - h[j_idx]
        pos = np.where(diff > 1e-12)[0]
        neg = np.where(diff < -1e-12)[0]

        moved = False
        if pos.size + neg.size > 0:
            if pos.size:
                k = rng.choice(pos)
                src = i_idx[k]
                dst = j_idx[k]
                delta_max = min(h[src], 1.0 - h[dst], diff[k] / 2.0)
            else:
                k = rng.choice(neg)
                src = j_idx[k]          # larger value
                dst = i_idx[k]          # smaller value
                delta_max = min(h[src], 1.0 - h[dst], -diff[k] / 2.0)

            if delta_max > 1e-12:
                delta = delta_max * rng.random()
                h[src] -= delta
                h[dst] += delta
                new_c5 = compute_c5(h)
                if new_c5 <= best_c5 + 1e-12:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve_cnt = 0
                else:
                    # revert
                    h[src] += delta
                    h[dst] -= delta
                moved = True

        if not moved:
            # fallback: random feasible exchange
            i, j = rng.choice(n, 2, replace=False)
            if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
                src, dst = i, j
            elif h[j] > 1e-12 and h[i] < 1.0 - 1e-12:
                src, dst = j, i
            else:
                # both at bounds – skip
                no_improve_cnt += 1
                if no_improve_cnt >= no_improve_limit:
                    # small random perturbation to shake out of stagnation
                    a, b = rng.choice(n, 2, replace=False)
                    eps = min(0.05, h[a], 1.0 - h[b])
                    if eps > 0:
                        delta = eps * rng.random()
                        h[a] -= delta
                        h[b] += delta
                continue

            delta_max = min(h[src], 1.0 - h[dst])
            delta = delta_max * rng.random()
            h[src] -= delta
            h[dst] += delta
            new_c5 = compute_c5(h)
            if new_c5 <= best_c5 + 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve_cnt = 0
            else:
                h[src] += delta
                h[dst] -= delta

        no_improve_cnt += 1
        if no_improve_cnt >= 5 * n:
            # occasional tiny random nudge to avoid deadlock
            a, b = rng.choice(n, 2, replace=False)
            eps = min(0.02, h[a], 1.0 - h[b])
            if eps > 0:
                delta = eps * rng.random()
                h[a] -= delta
                h[b] += delta
            no_improve_cnt = 0

    # final sanity projection
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h + (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, compute_c5(best_h)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    # Desired resolution – higher gives more freedom but costs a bit more.
    target_n = 200

    # Build initial candidates
    starts = []

    # 1. Parent solution (if supplied) – upsample to target_n
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        if parent_h.shape[0] != target_n:
            parent_h = upsample(parent_h, target_n)
        starts.append(parent_h.copy())
    except NameError:
        parent_h = None

    # 2. Simple triangular shape
    starts.append(generate_triangular(target_n))

    # 3. A few random feasible seeds
    for _ in range(3):
        starts.append(random_feasible(target_n, rng))

    # 4. Slight perturbations of the triangular shape
    tri = generate_triangular(target_n)
    for _ in range(2):
        pert = tri + rng.uniform(-0.15, 0.15, size=target_n)
        pert = np.clip(pert, 0.0, 1.0)
        diff = (target_n / 2.0) - pert.sum()
        if abs(diff) > 1e-12:
            pert += diff / target_n
            pert = np.clip(pert, 0.0, 1.0)
        starts.append(pert)

    best_h = None
    best_c5 = np.inf
    best_n = target_n

    # Run local search from each start while time permits
    for s in starts:
        if time.time() >= deadline:
            break
        h_opt, c5_opt = improve_local(s, deadline, rng)
        if c5_opt < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_opt, c5_opt, target_n

    # Fallback: deterministic half‑ones block (guaranteed feasible)
    if best_h is None:
        n = target_n
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final sanity adjustments
    target_sum = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h + (target_sum - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
