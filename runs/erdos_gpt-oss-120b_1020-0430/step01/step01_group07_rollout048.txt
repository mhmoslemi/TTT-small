    corr = np.correlate(h, 1.0 - h, mode='full')

def anneal(start_h, max_seconds):
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = c5_metric(h)
    cur_c = best_c
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= max_seconds:
            break
        # schedule
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        max_step = max_step_initial * (1 - progress) + max_step_final * progress

        if random.random() < swap_prob:
            # swap move: pick i random, j random
            i = random.randrange(n)
            j = random.randrange(n)
            if h[i] == h[j]:
                continue
            # swap
            h[i], h[j] = h[j], h[i]
            new_c = c5_metric(h)
            dC = new_c - cur_c
            if dC < 0 or random.random() < math.exp(-dC / (T + 1e-12)):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
        else:
            # delta move
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            # sample delta
            # we limit to max_step around 0
            low_cap = max(low, -max_step)
            high_cap = min(high, max_step)
            if low_cap < high_cap:
                delta = random.uniform(low_cap, high_cap)
            else:
                delta = random.uniform(low, high)
            h[i] = hi + delta
            h[j] = hj - delta
            new_c = c5_metric(h)
            dC = new_c - cur_c
            if dC < 0 or random.random() < math.exp(-dC / (T + 1e-12)):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # revert
                h[i] = hi
                h[j] = hj
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    total_budget = float(budget_s)
    # Choose n

    n = desired_n

    best_h = leg_vec.astype(np.float64)
    best_c = leg_c
    # if initial available
    if 'initial_h_values' in globals():
        # resample into n length
        h_init = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h_init.size
        if n0 != n:
            x0 = np.linspace(0,2,n0,endpoint=False)
            x = np.linspace(0,2,n,endpoint=False)
            h_res = np.interp(x, x0, h_init)
        else:
            h_res = h_init.copy()
        # enforce sum
        target = n/2.0
        cur = h_res.sum()
        if cur != 0:
            h_res *= target / cur
        np.clip(h_res,0,1,inplace=True)
        deficit = target - h_res.sum()
        if abs(deficit) > 1e-12:
            for idx in np.random.permutation(n):
                lo,hi = 0.0,1.0
                if lo <= h_res[idx] + deficit <= hi:
                    h_res[idx] += deficit
                    break
        c_init = c5_metric(h_res)
        if c_init < best_c:
            best_h = h_res.copy()
            best_c = c_init
    # allocate time
    elapsed = time.time() - start
    # define function to compute remaining time 
    while True:
        remaining = total_budget - (time.time() - start)
        if remaining < 10: # reserve time for final polish and return
            break
        # Random binary start
        h_rand = np.zeros(n, dtype=np.float64)
        half = n // 2
        idx = np.random.permutation(n)
        h_rand[idx[:half]] = 1.0
        # Possibly add a tiny random fractional mask for half? Not needed.
        # run anneal for slice time (maybe min(remaining*0.15, 120))
        slice_time = min(remaining * 0.15, 120.0)
        cand_h, cand_c = anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_h, best_c = cand_h.copy(), cand_c
        # also anneal from best_h for some medium time to possibly improve further, but limit time
        # optional if remaining > something and best updated recently
        # continue loop
    # final polishing
    remaining = total_budget - (time.time() - start)
    if remaining > 1:
        best_h, best_c = anneal(best_h, remaining)
    # final enforcement sum
    target = n/2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0 <= best_h[idx] + diff <= 1:
                best_h[idx] += diff

    # compute final bound

import time, random, math
import numpy as np

def _c5_metric(h):
    """Discrete C5 value for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_sum(h, target):
    """Adjust one entry to enforce exact sum, staying in [0,1]."""
    deficit = target - h.sum()
    if abs(deficit) < 1e-12:
        return
    # try a random index that can absorb the deficit
    for idx in np.random.permutation(h.size):
        lo, hi = 0.0, 1.0
        if lo <= h[idx] + deficit <= hi:
            h[idx] += deficit
            break


def _legendre_vector(min_n):
    """Return a binary vector based on a Legendre sequence.
       Length will be p‑1 where p is the smallest prime ≥ min_n+1."""
    p = int(min_n) + 1
    while True:
        if p > 2 and all(p % d for d in range(2, int(p ** 0.5) + 1)):
            break
        p += 1
    n = p - 1
    h = np.zeros(n, dtype=np.float64)
    exp = (p - 1) // 2
    for i in range(1, n + 1):
        if pow(i, exp, p) == 1:
            h[i - 1] = 1.0
    return h


def _random_binary(n):
    """Binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=np.float64)
    half = n // 2
    idx = np.random.permutation(n)
    h[idx[:half]] = 1.0
    return h


def _anneal(start_h, max_seconds):
    """Simulated‑annealing improvement."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    # schedule parameters
    T0, T_end = 0.04, 1e-6
    step0, step_end = 0.20, 0.01
    swap_prob = 0.12

    t0 = time.time()
    while True:
        elapsed = time.time() - t0
        if elapsed >= max_seconds:
            break
        prog = elapsed / max_seconds
        T = T0 * (T_end / T0) ** prog
        max_step = step0 * (1 - prog) + step_end * prog

        if random.random() < swap_prob:
            i, j = random.sample(range(n), 2)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_c = _c5_metric(h)
            dC = new_c - cur_c
            if dC < 0 or random.random() < math.exp(-dC / (T + 1e-12)):
                cur_c = new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h.copy()
            else:
                h[i], h[j] = h[j], h[i]
        else:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_cap = min(1.0 - hi, hj)
            if lo >= hi_cap:
                continue
            lo2 = max(lo, -max_step)
            hi2 = min(hi_cap, max_step)
            if lo2 < hi2:
                delta = random.uniform(lo2, hi2)
            else:
                delta = random.uniform(lo, hi_cap)
            h[i] = hi + delta
            h[j] = hj - delta
            new_c = _c5_metric(h)
            dC = new_c - cur_c
            if dC < 0 or random.random() < math.exp(-dC / (T + 1e-12)):
                cur_c = new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h.copy()
            else:
                h[i] = hi
                h[j] = hj
    return best_h, best_c
