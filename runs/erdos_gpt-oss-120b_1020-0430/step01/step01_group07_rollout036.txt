import time
import math
import random
import numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """C5 value for a sampled step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # autocorrelation of h (integral of h(x)h(x+k))
    conv = np.correlate(h, h, mode="full")
    min_val = conv.min() * dx          # smallest overlap
    return 1.0 - min_val               # because C5 = max ∫h(1‑h(x+k)) = 1‑min O(k)

def _initial_vector(n: int) -> np.ndarray:
    """Create a feasible start vector on a grid of size n."""
    if "initial_h_values" in globals():
        src = np.asarray(initial_h_values, dtype=np.float64)
        m = src.size
        x_src = np.linspace(0.0, 2.0, m, endpoint=False)
        x_dst = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_dst, x_src, src)
    else:
        # uniform density 0.5 gives the correct total mass
        h = np.full(n, 0.5, dtype=np.float64)

    # enforce ∑h = n/2 exactly
    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)

    deficit = target - h.sum()
    if abs(deficit) > 1e-12:                     # tiny correction
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + deficit <= hi:
                h[idx] += deficit
                break
    return h

def _gradient_from_low_shifts(conv: np.ndarray, h: np.ndarray, k_low: int) -> np.ndarray:
    """
    Build a gradient that tries to increase the k_low smallest overlap values.
    conv – full autocorrelation (length 2n‑1)
    h    – current function values
    """
    n = h.size
    center = n - 1
    low_idx = np.argpartition(conv, k_low)[:k_low]
    g = np.zeros_like(h)

    for idx in low_idx:
        shift = idx - center
        t = abs(shift)
        if t >= n:
            continue
        if t == 0:
            g += 2.0 * h                       # derivative of h_i^2
        else:
            # contributions from pairs (i,i+t) and (i+t,i)
            g[: n - t] += h[t:]                # h_{i+t}
            g[t:]     += h[: n - t]            # h_{i}
    return g

    # annealing parameters
    T0 = 0.02
    T_end = 1e-6
    K_LOW = 3               # number of low‑overlap shifts to guide the gradient
    max_step_start = 0.15
    max_step_end = 0.02

    while time.time() < end_time:
        # elapsed fraction for schedules
        elapsed = time.time() - start
        progress = elapsed / budget_s
        T = T0 * ((T_end / T0) ** progress)
        max_step = max_step_start * (1.0 - progress) + max_step_end * progress

        # autocorrelation and current C5
        conv = np.correlate(cur_h, cur_h, mode="full")
        cur_c = 1.0 - conv.min() * (2.0 / n)

        # gradient pointing toward raising the worst overlaps
        g = _gradient_from_low_shifts(conv, cur_h, K_LOW)

        # choose a source (decrease) and a sink (increase) according to g
        mask_up = cur_h < 1.0 - 1e-12
        mask_dn = cur_h > 1e-12

        if not mask_up.any() or not mask_dn.any():
            break

        # index to receive mass – highest gradient among admissible positions
        i_up = int(np.argmax(np.where(mask_up, g, -np.inf)))
        # index to give mass – lowest gradient among admissible positions
        i_dn = int(np.argmin(np.where(mask_dn, g, np.inf)))

        if i_up == i_dn:
            # pick a random alternative to avoid stagnation
            cand = np.where(mask_up & (np.arange(n) != i_dn))[0]
            if cand.size == 0:
                break
            i_up = int(np.random.choice(cand))

        # feasible transfer amount
        delta = min(max_step, 1.0 - cur_h[i_up], cur_h[i_dn])
        if delta <= 0.0:
            continue

        # apply tentative move
        cur_h[i_up] += delta
        cur_h[i_dn] -= delta

        new_c = _c5_metric(cur_h)
        dC = new_c - cur_c

        if dC < 0.0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i_up] -= delta
            cur_h[i_dn] += delta

    # ----------------------------------------------------------------------
    # final deterministic polishing – random swaps that preserve the sum
    # ----------------------------------------------------------------------
    cur_h = best_h.copy()
    for _ in range(3000):
        i, j = random.sample(range(n), 2)
        if cur_h[i] == cur_h[j]:
            continue
        # swap the values
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = _c5_metric(cur_h)
        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # ----------------------------------------------------------------------
    # final feasibility correction (tiny rounding errors)
    # ----------------------------------------------------------------------
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= best_h[idx] + diff <= hi:
                best_h[idx] += diff
                break

    # final bound – use the provided helper if available
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
