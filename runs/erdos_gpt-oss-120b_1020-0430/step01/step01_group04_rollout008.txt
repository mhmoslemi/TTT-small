def simulated_annealing(start_h, time_limit, rng):
    n = start_h.shape[0]
    dx = 2.0 / n
    h_cur = start_h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur

    # times
    start = time.time()
    # Temperature scheduling: initial based on c5_cur
    T0 = max(0.001, c5_cur * 0.5)  # can tune
    # we will linearly cool from T0 to T_end = 1e-6 over total iterations.
    iter = 0
    while time.time() < time_limit:
        iter += 1
        # compute T based on elapsed proportion
        elapsed = time.time() - start
        prog = elapsed / (time_limit - start)
        T = T0 * (1 - prog) + 1e-6 * prog
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:

        # compute bounds for delta: move from i to j
        delta_low = -min(h_cur[j], 1 - h_cur[i])
        delta_high = min(h_cur[i], 1 - h_cur[j])
        if delta_low >= delta_high:

        h_new[j] += delta

            c5_cur = c5_new

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, keeping the integral exact."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 0.5 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 0.5 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible start (values in [0,1] and sum = n/2)."""
    target = n / 2.0
    a = rng.random(n)
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def simulated_annealing(start_h: np.ndarray, deadline: float, rng: np.random.Generator):
    """Simple SA that preserves the sum and box constraints."""
    n = start_h.shape[0]
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    # initial temperature proportional to current value
    T0 = max(1e-4, cur_c5 * 0.5)

    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling
        prog = (time.time() - start_time) / max(1e-12, deadline - start_time)
        T = T0 * (1.0 - prog) + 1e-6 * prog

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move delta from i to j (positive delta -> i loses, j gains)
        delta_low = -min(h[j], 1.0 - h[i])
        delta_high = min(h[i], 1.0 - h[j])
        if delta_low >= delta_high:
            continue
        delta = rng.uniform(delta_low, delta_high)
        if abs(delta) < 1e-14:
            continue

        # apply move in‑place
        h[i] -= delta
        h[j] += delta

        new_c5 = compute_c5(h)
        dC = new_c5 - cur_c5
        accept = False
        if dC <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < math.exp(-dC / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta

        # occasional restart from best with a tiny random perturbation
        if it % 5000 == 0:
            if rng.random() < 0.3:
                h[:] = best_h
                # a few tiny swaps
                for _ in range(4):
                    a, b = rng.integers(0, n, size=2)
                    if a == b:
                        continue
                    d_low = -min(h[b], 1.0 - h[a])
                    d_high = min(h[a], 1.0 - h[b])
                    if d_low >= d_high:
                        continue
                    d = rng.uniform(d_low, d_high) * 0.2
                    h[a] -= d
                    h[b] += d
                cur_c5 = compute_c5(h)

    return best_h, best_c5


def local_improve(h: np.ndarray, deadline: float, rng: np.random.Generator):
    """Greedy reduction of the worst‑lag overlap."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    while time.time() < deadline:
        conv = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = int(np.argmax(conv))
        lag = max_idx - (n - 1)

        # indices where the shift overlaps
        if lag >= 0:
            idxs = np.arange(n - lag)
        else:
            idxs = np.arange(-lag, n)

        diff = best_h[idxs] - best_h[idxs + lag]
        pos = diff > 1e-9
        if not np.any(pos):
            break

        # choose i with probability proportional to improvement potential
        probs = diff[pos]
        probs = probs / probs.sum()
        chosen = rng.choice(np.where(pos)[0], p=probs)
        i = idxs[chosen]
        j = i + lag

        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max <= 0.0:
            continue
        # modest step
        delta = delta_max * rng.random() * 0.5

        best_h[i] -= delta
        best_h[j] += delta
        new_c5 = compute_c5(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # revert
            best_h[i] += delta
            best_h[j] -= delta

    return best_h, best_c5
