def project_capped_simplex(v, S, upper=1.0):
    # Solve min ||x - v||^2, s.t. sum x = S, lower=0, upper=upper
    # We'll consider dual variable lambda.
    # Using binary search for lambda in [lo,hi].
    lo = np.min(v - upper)
    hi = np.max(v)
    # iterate:
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, upper)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0, upper)
    # Ensure sum exactly S by small correction pass:
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff across elements not at bounds
        # adjust each element proportionally or simple loop
        # We'll just add diff/N? but may violate bounds.
        for i in range(len(x)):
            if diff == 0:
                break
            if diff > 0 and x[i] < upper - 1e-12:
                add = min(diff, upper - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 1e-12:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, S: float, upper: float = 1.0) -> np.ndarray:
    """Projection of v onto {x : 0≤x≤upper, sum(x)=S}."""
    # simple water‑filling via a dual variable λ, solved by bisection
    lo = np.min(v - upper)
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, upper)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, upper)
    # tiny correction to hit the target sum exactly
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        for i in range(len(x)):
            if diff > 0 and x[i] < upper - 1e-12:
                add = min(diff, upper - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 1e-12:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return x


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Smooth max weights w_k = e^{β conv_k} / Σ e^{β conv}."""
    maxc = np.max(conv)
    exp_vals = np.exp(beta * (conv - maxc))
    return exp_vals / exp_vals.sum()


def _gradient_smoothed_max(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (softmax) objective.
    w has length 2*n-1, indexed by lag = idx-(n-1).
    """
    n = h.size
    g = np.zeros_like(h)
    n2 = 2 * n - 1
    for idx in range(n2):
        weight = w[idx]
        if weight == 0.0:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            i_start = 0
            i_end = n - 1 - lag
        else:
            i_start = -lag
            i_end = n - 1
        i0 = i_start
        i1 = i_end + 1  # exclusive
        # contribution from ∂/∂h_i of the linear term (1)
        g[i0:i1] += weight * (1.0 - h[i0 + lag:i1 + lag])
        # contribution from the product term –h_i h_{i+lag}
        g[i0 + lag:i1 + lag] -= weight * h[i0:i1]
    return g


def _pgd_optimize(
    h0: np.ndarray,
    time_allow: float,
    target: float,
) -> tuple:
    """
    Projected gradient descent on the smoothed max.
    Returns best (h, c5) found within time_allow seconds.
    """
    start = time.time()
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_value(h)
    betas = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0]

    while time.time() - start < time_allow and best_c5 > target:
        for beta in betas:
            # current objective
            cur_c5 = _c5_value(h)
            conv = np.correlate(h, 1.0 - h, mode="full")
            w = _softmax_weights(conv, beta)
            grad = _gradient_smoothed_max(h, w)
            norm = np.linalg.norm(grad)
            if norm == 0.0:
                continue
            direction = -grad / norm
            step = 1.0
            improved = False
            while step > 1e-6:
                h_new = h + step * direction
                h_new = _project_capped_simplex(h_new, n / 2.0, upper=1.0)
                new_c5 = _c5_value(h_new)
                if new_c5 < cur_c5 - 1e-12:
                    h = h_new
                    cur_c5 = new_c5
                    improved = True
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_new.copy()
                    break
                step *= 0.5
            if not improved:
                continue
            if best_c5 < target:
                return best_h, best_c5
            if time.time() - start >= time_allow:
                return best_h, best_c5
    return best_h, best_c5


def _binary_hill_climb(
    n: int,
    time_allow: float,
    target: float,

    """

        idx = np.arange(n)

        h[idx[: n // 2]] = 1.0

    best_h = random_binary()
    best_c5 = _c5_value(best_h)

        new_c5 = _c5_value(best_h)

            # keep the improvement
        else:
            # revert
            best_h[i], best_h[j] = 1.0, 0.0
        # occasional escape from stagnation
        if np.random.rand() < 0.001:
            best_h = random_binary()
            best_c5 = _c5_value(best_h)
    return best_h, best_c5


def _binarize(h: np.ndarray) -> np.ndarray:
    """Convert a (possibly fractional) h to a feasible binary vector."""
    n = h.size
    bin_h = (h >= 0.5).astype(float)
    diff = bin_h.sum() - n / 2.0
    if diff > 0:
        # too many ones → turn the smallest ones to zero
        idx = np.argsort(h)
        for i in idx:
            if bin_h[i] == 1.0:
                bin_h[i] = 0.0
                diff -= 1
                if diff == 0:
                    break
    elif diff < 0:
        # too few ones → turn the largest zeros to one
        idx = np.argsort(-h)
        for i in idx:
            if bin_h[i] == 0.0:
                bin_h[i] = 1.0
                diff += 1
                if diff == 0:
                    break
    return bin_h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound using a mix of projected gradient descent
    (smoothed max) and a binary hill‑climb.  Returns (h_values, c5_bound,
    n_points).  Smaller n is preferred when possible.
    """
    np.random.seed(seed)
    start_all = time.time()
    time_left = lambda: budget_s - (time.time() - start_all)

    # ---- initial construction -------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # type: ignore
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # candidate grid sizes (start with the size of the supplied construction)
    n0 = init_h.size if init_h is not None else 128
    cand_sizes = [n0]
    for mul in (2, 3):
        sz = n0 * mul
        if sz <= 300:
            cand_sizes.append(int(sz))
    for sz in (160, 200, 256, 300):
        if sz <= 300 and sz not in cand_sizes:
            cand_sizes.append(sz)
    cand_sizes = sorted(set(cand_sizes))

    # ---- main optimisation loop -----------------------------------------------
    for n in cand_sizes:
        if time_left() < 5.0:
            break

        # base vector for this resolution
        if init_h is not None:
            if init_h.size != n:
                # linear interpolation to upscale/downscale
                x_old = (np.arange(init_h.size) + 0.5) * (2.0 / init_h.size)
                x_new = (np.arange(n) + 0.5) * (2.0 / n)
                base = np.interp(x_new, x_old, init_h)
            else:
                base = init_h.copy()
        else:
            base = np.full(n, 0.5)

        base = _project_capped_simplex(base, n / 2.0, upper=1.0)

        # --- allocate a slice of the remaining budget -------------------------
        portion = min(time_left() * 0.4, 200.0)

        # Gradient‑descent refinement
        h_opt, c5_opt = _pgd_optimize(base, portion, target_bound)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < target_bound:
                break

        # Try a few random restarts (fast)
        while time_left() > 5.0:
            rand_start = np.random.rand(n)
            rand_start = _project_capped_simplex(rand_start, n / 2.0, upper=1.0)
            sub_allow = min(time_left() * 0.2, 30.0)
            h_r, c5_r = _pgd_optimize(rand_start, sub_allow, target_bound)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r, c5_r, n
                if best_c5 < target_bound:
                    break
            # stop if no noticeable improvement
            if c5_r >= best_c5 - 1e-5:
                break

        # ----- binary hill‑climb on the best continuous solution ---------------
        if time_left() > 10.0 and best_h is not None:
            bin_start = _binarize(best_h)
            hc, c5c = _binary_hill_climb(
                best_n, time_left() * 0.2, target_bound
            )
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c
                if best_c5 < target_bound:
                    break

        if best_c5 < target_bound:
            break

    # ---- fallback (should never be needed) -----------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_value(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
