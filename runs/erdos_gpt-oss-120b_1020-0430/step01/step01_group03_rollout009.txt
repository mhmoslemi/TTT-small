def gradient_refine(h, rng, step=0.01, max_iter=10):
    n = len(h)
    for it in range(max_iter):
        # compute worst shift
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        # compute gradient
        g = np.zeros_like(h)
        if k >= 0:
            max_i = n - k
            # first part
            if max_i > 0:
                i = np.arange(max_i)
                g[i] += 1 - h[i + k]
            # second part
            if k > 0:
                j = np.arange(k, n)
                g[j] -= h[j - k]
        else:
            # k negative -> same as positive with sign
            k_abs = -k
            max_i = n - k_abs
            if max_i > 0:
                i = np.arange(max_i)
                g[i + k_abs] += 1 - h[i]
            if k_abs > 0:
                j = np.arange(k_abs, n)
                g[j - k_abs] -= h[j]
        # perform step
        step_size = step * (0.5 ** it)  # decreasing step
        h = h - step_size * g
        # clip and adjust sum
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        target = n / 2.0
        diff = target - h.sum()
        # distribute diff
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    # simple scaling then redistribution to satisfy bounds
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k]) for i< n-k
            g[k:] -= h[:upper]                # - h[i] for i>=k
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Use supplied parent solution if available
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target0 = n0 / 2.0
        diff = target0 - h0.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n0)
                if diff > 0:
                    add = min(diff, 1.0 - h0[i])
                else:
                    add = max(diff, -h0[i])
                h0[i] += add
                diff = target0 - h0.sum()
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # candidate grid sizes (keep them modest for speed)
    candidate_ns = [45, 60, 72, 84, 90, 96, 108, 120]
    if best_n is not None:
        candidate_ns = [best_n] + [n for n in candidate_ns if n != best_n]

    # ------------------------------------------------------------------
    # Main optimisation loop over different discretisations
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.94:
            break
        attempts = 0
        max_attempts = 15
        while attempts < max_attempts and time.time() - start_time < budget_s * 0.94:
            attempts += 1
            # ---- initialise a feasible vector --------------------------------
            if best_n == n and best_h is not None and attempts == 1:
                h = best_h.copy()
            else:
                # random binary or continuous start (50% chance each)
                if rng.random() < 0.5:
                    h = np.zeros(n)
                    idx = rng.choice(n, n // 2, replace=False)
                    h[idx] = 1.0
                else:
                    h = random_feasible_vector(n, rng)

            # ---- local hill‑climb -------------------------------------------
            h, c = hill_climb(h, rng, max_iters=2500, start_step=0.5)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ---- SLSQP passes with increasing beta ---------------------------
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                if time.time() - start_time > budget_s * 0.96:
                    break
                max_iter = 250 if beta < 100 else 500
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ---- gradient refinement -----------------------------------------
            if time.time() - start_time < budget_s * 0.97:
                h = gradient_refine(h, rng, step=0.02, max_iter=6)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ---- short final hill‑climb --------------------------------------
            if time.time() - start_time < budget_s * 0.98:
                h, c = hill_climb(h, rng, max_iters=1500, start_step=0.3)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # Optional up‑sampling refinement (if budget permits)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        if best_n * 2 <= 200 and time.time() - start_time < budget_s * 0.99:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.3)
            for beta in (5.0, 20.0, 80.0):
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
                except Exception:
                    pass
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final feasibility enforcement
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback: uniform 0.5 function
        best_n = 44
        best_h = np.full(best_n, 0.5)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, best_n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
