def upsample_initial(init, n):

import time, random, math
import numpy as np

# ---------- utility functions ----------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect {sum = target} by bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx (discretised)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_w_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """For a given shift s, return w_i = h[i+s]+h[i-s] (out‑of‑range treated as 0)."""
    n = h.shape[0]
    w = np.zeros(n)
    if s >= 0:
        w[: n - s] += h[s:]
        w[s:] += h[: n - s]
    else:
        sa = -s
        w[sa:] += h[: n - sa]
        w[: n - sa] += h[sa:]
    return w


def apply_delta(conv: np.ndarray, h: np.ndarray, idx: int, delta: float):
    """
    Linear part of the effect on the autocorrelation when h[idx] changes by `delta`.
    Updates `conv` in place.  Uses the identity:
        conv[s] += delta * h[idx + s] + delta * h[idx - s]
    realised by vectorised index arithmetic.
    """
    n = h.shape[0]
    off = n - 1
    # vector of shifts j - idx for every j
    j = np.arange(n)
    shifts = j - idx                    # size n, range [-idx, n-1-idx]
    conv[off + shifts] += delta * h    # first‑factor contribution
    conv[off - shifts] += delta * h    # second‑factor contribution


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n and projection."""
    old_n = init.shape[0]
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(x - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * x / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Block of ones of total mass 1 centred on the interval."""
    h = np.zeros(n)
    block = n // 2                     # exact half the points = unit mass
    start = (n - block) // 2
    h[start : start + block] = 1.0
    return h

    # ---- optimisation over each seed ----
    for seed_vec in seeds:
        if time.time() >= deadline:
            break
        h = project_onto_bounds_and_sum(seed_vec, target)  # safety
        conv = np.correlate(h, h, mode="full")
        val = (target - np.min(conv)) * dx
        if val < best_val:
            best_val, best_h, best_conv = val, h.copy(), conv.copy()
            if best_val <= 0.3808:
                return (best_h.tolist(), float(best_val), int(n))

        # simple deterministic + stochastic hill‑climbing
        no_improve = 0
        while time.time() < deadline:
            if no_improve > 5000:
                break

            # worst shift (the bottleneck)
            idx_min = int(np.argmin(conv))
            s = idx_min - off

            # contribution of each position to that shift
            w = compute_w_for_shift(h, s)

            # feasible directions
            up_mask = h < 1.0 - 1e-12
            down_mask = h > 1e-12
            if not up_mask.any() or not down_mask.any():
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
            else:
                up_idx = np.where(up_mask)[0]
                down_idx = np.where(down_mask)[0]
                i = up_idx[np.argmax(w[up_idx])]    # increase where contribution is large
                j = down_idx[np.argmin(w[down_idx])]  # decrease where contribution is small
                if i == j:
                    # fallback to a random feasible pair
                    i, j = np.random.choice(up_idx, 1)[0], np.random.choice(down_idx, 1)[0]
                    if i == j:
                        continue

            # maximal admissible transfer
            delta_max = min(1.0 - h[i], h[j])
            if delta_max <= 1e-12:
                no_improve += 1
                continue
            # random step size (10‑50 % of the maximum)
            frac = 0.1 + 0.4 * random.random()
            delta = delta_max * frac

            # ----- evaluate the move -----
            conv_cand = conv.copy()
            apply_delta(conv_cand, h, i, delta)
            apply_delta(conv_cand, h, j, -delta)

            d2 = delta * delta
            conv_cand[off] += 2.0 * d2                 # square terms at shift 0
            idx1 = off + (j - i)
            idx2 = off - (j - i)
            conv_cand[idx1] -= d2                      # cross‑term corrections
            conv_cand[idx2] -= d2

            new_val = (target - np.min(conv_cand)) * dx

            if new_val + 1e-12 < best_val:
                # accept improvement
                best_val, best_h, best_conv = new_val, h.copy(), conv.copy()
                best_h[i] += delta
                best_h[j] -= delta
                best_conv = conv_cand
                h[i] += delta
                h[j] -= delta
                conv = conv_cand
                no_improve = 0
                if best_val <= 0.3808:
                    return (best_h.tolist(), float(best_val), int(n))
            else:
                # occasional uphill acceptance (simulated annealing)
                elapsed = time.time() - start
                frac_time = elapsed / budget_s
                T = 0.001 * (1.0 - frac_time) + 1e-7
                if random.random() < math.exp(-(new_val - best_val) / T):
                    # accept anyway (helps escaping)
                    h[i] += delta
                    h[j] -= delta
                    conv = conv_cand
                no_improve += 1

    # ------------------------------------------------------------------
    # final safety checks
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        # correct any tiny mass error
        resid = target - best_h.sum()
        if abs(resid) > 1e-12:
            for k in range(n):
                cap = (1.0 - best_h[k]) if resid > 0 else best_h[k]
                change = min(abs(resid), cap)
                if change < 1e-14:
                    continue
                if resid > 0:
                    best_h[k] += change
                    resid -= change
                else:
                    best_h[k] -= change
                    resid += change
                if abs(resid) < 1e-12:
                    break

    final_val = compute_c5(best_h)
    return (best_h.tolist(), float(final_val), int(n))
