def make_triangle(n):
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx*0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n/2.0)

def make_sin2(n):
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx*0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n/2.0)

def make_rectangle(n):
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx*0.5
    h = np.where((pts >= 0.5) & (pts <= 1.5), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n/2.0)

def make_blocks(num_blocks, n):
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx*0.5
    width = 1.0 / num_blocks
    spacing = 2.0 / num_blocks
    h = np.zeros(n)
    for i in range(num_blocks):
        start = i * spacing
        end = start + width
        mask = (pts >= start) & (pts < end)
        h[mask] = 1.0
    # project (in case discretization makes sum not exact)
    return project_onto_bounds_and_sum(h, n/2.0)

import time, random, math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

TARGET_SUM_GLOBAL = None


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = target."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Orthogonal projection of x onto the box [0,1]^n intersected
    with the hyperplane Σx = target using a simple bisection for the
    Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete objective: max_k ∫h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    return logsumexp(alpha * arr) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth objective for SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.where((pts >= 0.5) & (pts <= 1.5), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_blocks(num_blocks: int, n: int) -> np.ndarray:
    """Equally spaced blocks of height 1, total length 1."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    width = 1.0 / num_blocks
    spacing = 2.0 / num_blocks
    h = np.zeros(n)
    for i in range(num_blocks):
        start = i * spacing
        end = start + width
        mask = (pts >= start) & (pts < end)
        h[mask] = 1.0
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def slsqp_optimize(h0: np.ndarray, target_sum: float, dx: float,
                   alphas, maxiter_phase: int,
                   start: float, budget: float) -> np.ndarray:
    """Run a multi‑phase SLSQP optimisation with increasing smoothness."""
    global TARGET_SUM_GLOBAL
    h = h0.copy()
    bounds = [(0.0, 1.0)] * len(h)
    for alpha in alphas:
        if time.time() - start > budget - 5.0:
            break
        TARGET_SUM_GLOBAL = target_sum
        try:
            res = minimize(
                objective_smooth,
                h,
                args=(dx, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": maxiter_phase, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            break
        h = project_onto_bounds_and_sum(res.x, target_sum)
    return h


def hill_climb(h: np.ndarray, target_sum: float, best_val: float,
               start: float, budget: float, max_iter: int = None):
    """Greedy (with occasional full‑range moves) pairwise mass transfers."""
    n = len(h)
    it = 0
    while time.time() - start < budget - 0.5:
        if max_iter is not None and it >= max_iter:
            break
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        # Occasionally take an extreme move to explore faster
        if np.random.rand() < 0.3:
            delta = delta_max if np.random.rand() < 0.5 else delta_min
        else:
            delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            h = cand
            if best_val <= 0.3808:
                break
        it += 1
    return h, best_val

    np.random.seed(seed)

    start = time.time()

    return (best_h.tolist(), float(best_val), int(n0))
