import time
import numpy as np
from scipy import optimize

# -------------------------------------------------
# Feasibility projection (clip to [0,1] and exact sum)
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction for any leftover rounding error
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# -------------------------------------------------
# C5 evaluation (max_{k} int h(1-h shifted) )
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# -------------------------------------------------
# Linear interpolation for up‑scaling
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# -------------------------------------------------
# SLSQP optimisation (identical to the reference implementation)
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


# -------------------------------------------------
# Autocorrelation helpers for the gradient ascent
def _auto_corr(h: np.ndarray) -> np.ndarray:
    return np.correlate(h, h, mode="full")


def _min_shift_and_val(h: np.ndarray) -> tuple:
    A = _auto_corr(h)
    idx = int(np.argmin(A))
    shift = idx - (len(h) - 1)
    return shift, A[idx]


def _grad_auto_corr_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] += h[shift:]
            g[shift:] += h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += h[: n - s]
            g[: n - s] += h[s:]
    return g  # derivative of Σ_i h_i h_{i+shift}


# -------------------------------------------------
# Simple sub‑gradient ascent on the minimal autocorrelation
def _gradient_ascent(
    h: np.ndarray,
    iters: int,
    lr: float,
    time_ok,
) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    fails = 0
    for _ in range(iters):
        if not time_ok():
            break
        shift, _ = _min_shift_and_val(h)
        g = _grad_auto_corr_shift(h, shift)  # we want to increase A(shift)
        h_new = h + lr * g
        h_new = _project_to_feasible(h_new, n)
        c5 = _c5_from_h(h_new)
        if c5 < best_c5 - 1e-12:
            best_h, best_c5, h = h_new, c5, h_new
            fails = 0
        else:
            fails += 1
            if fails >= 200:
                lr *= 0.7
                fails = 0
    return best_h, best_c5

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # Build a starting h for this resolution
        if init_h is not None and init_h.size != n:
            base_h = _upscale_h(init_h, n)
        elif init_h is not None:
            base_h = init_h.copy()
        else:
            base_h = np.full(n, 0.5)
        base_h = _project_to_feasible(base_h, n)

        # -----------------------------------------------------------------
        # 1) directly evaluate the starting point
        c5_start = _c5_from_h(base_h)
        if c5_start < best_c5:
            best_h, best_c5, best_n = base_h, c5_start, n
            if best_c5 < target:
                break

        # -----------------------------------------------------------------
        # 2) gradient ascent on the starting point
        iters = 4000
        lr = 0.02
        ok = lambda: time_left() > 0
        h_grad, c5_grad = _gradient_ascent(base_h, iters, lr, ok)
        if c5_grad < best_c5:
            best_h, best_c5, best_n = h_grad, c5_grad, n
            if best_c5 < target:
                break

        # -----------------------------------------------------------------
        # 3) short SLSQP refinement (helps tighten the max constraint)
        h_slsqp, c5_slsqp = _solve_h(h_grad, max_iter=300)
        if c5_slsqp < best_c5:
            best_h, best_c5, best_n = h_slsqp, c5_slsqp, n
            if best_c5 < target:
                break

        # -----------------------------------------------------------------
        # 4) a handful of stochastic restarts
        restarts = 8
        for _ in range(restarts):
            if time_left() < 5.0:
                break
            rnd = np.random.rand(n)
            rnd = _project_to_feasible(rnd, n)
            h_r, c5_r = _gradient_ascent(rnd, 2000, 0.02, ok)
            h_r, c5_r = _solve_h(h_r, max_iter=200)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r, c5_r, n
                if best_c5 < target:
                    break

    # -------------------------------------------------
    # Fallback: uniform half‑filled step function
    if best_h is None:
        n = 64
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
