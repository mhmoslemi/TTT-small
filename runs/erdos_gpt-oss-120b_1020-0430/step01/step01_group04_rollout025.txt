def simulated_annealing(start_h, time_deadline):
    n = len(start_h)
    best_h = start_h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    # parameters
    init_temp = 0.02 # scale of C5 changes
    final_temp = 1e-5
    max_step = 0.2  # max delta per move (subject to bounds)
    while time.time() < time_deadline:
        # compute remaining fraction
        t = (time.time() - start_time) / (time_deadline - start_time)
        temp = init_temp * (1 - t) + final_temp * t
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # compute allowed delta
        # possible transfer from i to j
        max_pos = min(cur_h[i], 1 - cur_h[j])
        max_neg = min(cur_h[j], 1 - cur_h[i])
        if max_pos + max_neg == 0:
            continue
        # sample delta in [-max_neg, max_pos]
        # but limit magnitude to max_step
        low = -max_neg
        high = max_pos
        # restrict to [-max_step, max_step] intersection
        low = max(low, -max_step)
        high = min(high, max_step)
        if low >= high:
            continue
        delta = rng.uniform(low, high)
        # apply
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        if delta_c5 <= 0 or rng.random() < np.exp(-delta_c5 / temp):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
    return best_h, best_c5
```

    return best_h, final_c5, best_n

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode='full')
    # log-sum-exp
    return float(logsumexp(alpha * conv) / alpha)

def generate_triangular(n):
    # points at midpoints of subintervals
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    # Scale to meet sum target n/2
    target = n / 2.0
    tri = tri * (target / tri.sum())
    # Clip to [0,1] then re‑adjust sum if needed
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

def random_feasible(n, rng):
    # generate random numbers, scale to target sum, then clip and adjust
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a

def solve_one_start(start, n, rng, deadline):
    target = n / 2.0
    def eq_sum(h):
        return np.sum(h) - target
    cons = {"type":"eq","fun":eq_sum}
    bounds = [(0.0,1.0)] * n

    # first stage
    if time.time() > deadline:
        return None, None
    res1 = minimize(lambda h: softmax_c5(h, alpha=5.0), start,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter':150,'ftol':1e-9,'disp':False})
    if not res1.success or time.time() > deadline:
        return None, None

    # second stage – sharper
    if time.time() > deadline:
        return None, None
    res2 = minimize(lambda h: softmax_c5(h, alpha=30.0), res1.x,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter':150,'ftol':1e-9,'disp':False})
    if not res2.success or time.time() > deadline:
        return None, None

    h_opt = np.clip(res2.x, 0.0, 1.0)
    # final projection onto sum constraint
    cur_sum = h_opt.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        h_opt = h_opt + diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

def simulated_annealing(start_h, deadline, rng):
    n = start_h.shape[0]
    target = n / 2.0
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # parameters
    init_temp = 0.02
    final_temp = 1e-6
    max_step = 0.3   # max delta per move
    # keep track of iterations without improvement to optionally restart
    no_improve = 0

    start_time = time.time()
    while True:
        now = time.time()
        if now >= deadline:
            break
        # linear temperature schedule
        frac = (now - start_time) / (deadline - start_time)
        temp = init_temp * (1 - frac) + final_temp * frac
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        hi = cur_h[i]
        hj = cur_h[j]
        max_pos = min(hi, 1.0 - hj)      # delta >0 moves from i to j
        max_neg = min(hj, 1.0 - hi)      # delta <0 moves from j to i
        if max_pos == 0 and max_neg == 0:
            continue
        low = -max_neg
        high = max_pos
        # limit magnitude
        low = max(low, -max_step)
        high = min(high, max_step)
        if low >= high:
            continue
        delta = rng.uniform(low, high)
        # apply
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # ensure bounds
        # (delta already respects bounds)
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        if delta_c5 <= 0 or rng.random() < np.exp(-delta_c5 / max(temp, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random restart if stuck
        if no_improve > 5000:
            # re‑initialize to a random feasible vector
            cur_h = random_feasible(n, rng)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95

    # Determine base resolution
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    if parent_h is not None:
        n0 = parent_n
        init_h = parent_h.copy()
    else:
        n0 = 100
        init_h = generate_triangular(n0)

    # Ensure even n for convenience
    if n0 % 2 != 0:
        n0 += 1
        if parent_h is not None and parent_n == n0-1:
            # upsample parent to new n0
            init_h = upsample(parent_h, n0)
        else:
            # generate anew at even size
            init_h = generate_triangular(n0)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Stage 1: quick SLSQP refinement
    if time.time() < time_limit:
        h_opt, c5_opt = solve_one_start(init_h, n0, rng, time_limit)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n0

    # Stage 2: simulated annealing at same resolution
    if time.time() < time_limit:
        # allocate half of remaining time to SA
        deadline_sa = time_limit
        start_h = best_h if best_h is not None else init_h
        h_sa, c5_sa = simulated_annealing(start_h, deadline_sa, rng)
        if c5_sa < best_c5:
            best_h, best_c5, best_n = h_sa, c5_sa, n0

    # Stage 3: upsample and refine at higher resolution if time permits
    if time.time() < time_limit:
        # choose a larger even n (roughly double, but cap at 400)
        n1 = min(400, n0 * 2)
        if n1 % 2 != 0:
            n1 += 1
        # upsample best_h to n1
        up_h = upsample(best_h if best_h is not None else init_h, n1)
        # quick local SLSQP (fewer iters) to adapt to new grid
        if time.time() < time_limit:
            h_opt2, c5_opt2 = solve_one_start(up_h, n1, rng, time_limit)
            if h_opt2 is not None and c5_opt2 < best_c5:
                best_h, best_c5, best_n = h_opt2, c5_opt2, n1
        # final SA at higher resolution
        if time.time() < time_limit:

            if c5_sa2 < best_c5:
                best_h, best_c5, best_n = h_sa2, c5_sa2, n1

    # fallback if nothing succeeded
    if best_h is None:
        n_fallback = 100
        best_h = np.concatenate([np.ones(n_fallback//2), np.zeros(n_fallback - n_fallback//2)]).astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n_fallback

    # final projection onto sum constraint
    target = best_n / 2.0
    cur_sum = best_h.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        best_h = best_h + diff / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points, normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1] with prescribed sum n/2."""
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


# ------------------------------------------------------------
# Local optimiser (SLSQP with smooth max)
# ------------------------------------------------------------
def solve_one_start(start: np.ndarray, n: int,
                    rng: np.random.Generator, deadline: float):
    """Two‑stage SLSQP from a feasible start. Returns (h_opt, c5_opt) or (None,None)."""
    target = n / 2.0

    def eq_sum(h):
        return np.sum(h) - target

    cons = {"type": "eq", "fun": eq_sum}
    bounds = [(0.0, 1.0)] * n

    if time.time() > deadline:
        return None, None

    # stage 1 – mild smoothing
    res1 = minimize(lambda h: softmax_c5(h, alpha=5.0),
                    start, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 120, "ftol": 1e-9, "disp": False})
    if not res1.success or time.time() > deadline:
        return None, None

    # stage 2 – sharper smoothing
    if time.time() > deadline:
        return None, None
    res2 = minimize(lambda h: softmax_c5(h, alpha=30.0),
                    res1.x, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 120, "ftol": 1e-9, "disp": False})
    if not res2.success or time.time() > deadline:
        return None, None

    h_opt = np.clip(res2.x, 0.0, 1.0)
    # final sum correction (tiny numerical drift)
    cur_sum = h_opt.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        h_opt = h_opt + diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)

    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt


# ------------------------------------------------------------
# Simulated annealing (pairwise mass exchange)
# ------------------------------------------------------------
def simulated_annealing(start_h: np.ndarray, deadline: float,
                        rng: np.random.Generator):
    """Simple SA that preserves the integral by moving mass between two points."""
    n = start_h.shape[0]
    target = n / 2.0
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    init_temp = 0.02
    final_temp = 1e-6
    max_step = 0.3
    no_improve = 0
    t0 = time.time()

    while True:
        now = time.time()
        if now >= deadline:
            break
        # linear cooling
        frac = (now - t0) / (deadline - t0)
        temp = init_temp * (1 - frac) + final_temp * frac

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        max_pos = min(hi, 1.0 - hj)  # i -> j
        max_neg = min(hj, 1.0 - hi)  # j -> i
        if max_pos == 0 and max_neg == 0:
            continue
        low, high = -max_neg, max_pos
        low = max(low, -max_step)
        high = min(high, max_step)
        if low >= high:
            continue
        delta = rng.uniform(low, high)
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        if delta_c5 <= 0 or rng.random() < np.exp(-delta_c5 / (temp + 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 5000:
            # random restart to re‑inject diversity
            cur_h = random_feasible(n, rng)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    return best_h, best_c5
