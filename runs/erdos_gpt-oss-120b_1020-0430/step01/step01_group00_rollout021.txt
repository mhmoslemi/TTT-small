import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _c5_corr(h: np.ndarray):
    """Return current C5 value and the full correlation array."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * (2.0 / n), corr


def _binary_from_float(arr: np.ndarray) -> np.ndarray:
    """Turn a real vector into a binary one with exactly half entries = 1."""
    n = arr.size
    half = n // 2
    idx = np.argsort(arr)[::-1]  # descending
    h = np.zeros(n, dtype=float)
    h[idx[:half]] = 1.0
    return h


def _random_binary(n: int) -> np.ndarray:
    """Create a random binary vector of length n with n/2 ones."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    np.random.shuffle(h)
    return h


def _propose_swap(h: np.ndarray, k: int):
    """
    Propose a pair of indices to swap.
    If k != 0 we try to increase the overlap at shift |k|
    (i.e. make a 0 whose partner is a 1 become a 1, and a 1 whose
    partner is a 0 become a 0).  If that fails we fall back to a
    random opposite‑bit swap.
    """
    n = h.size
    if k == 0:
        i, j = np.random.choice(n, 2, replace=False)
        while h[i] == h[j]:
            i, j = np.random.choice(n, 2, replace=False)
        return i, j

    if k < 0:
        k = -k
    limit = n - k
    idx = np.arange(limit)

    # indices where a 0 is paired with a 1 at distance k
    cand_i = idx[(h[idx] == 0) & (h[idx + k] == 1)]
    # indices where a 1 is paired with a 0 at distance k
    cand_j = idx[(h[idx] == 1) & (h[idx + k] == 0)]

    if cand_i.size and cand_j.size:
        i = np.random.choice(cand_i)
        j = np.random.choice(cand_j)
        return i, j

    # fallback – any opposite bits
    i, j = np.random.choice(n, 2, replace=False)
    while h[i] == h[j]:
        i, j = np.random.choice(n, 2, replace=False)
    return i, j


def _hill_climb_binary(init_h: np.ndarray, time_limit: float):
    """
    Greedy hill‑climbing on binary vectors.
    Swaps are performed only if they do not increase C5.
    """
    h = init_h.copy()
    n = h.size
    best_h = h.copy()
    best_c5, corr = _c5_corr(h)
    max_idx = int(np.argmax(corr))
    max_corr = float(corr[max_idx])
    start = time.time()

    while time.time() - start < time_limit:
        k = max_idx - (n - 1)          # shift causing the current maximum
        curr_c5 = max_corr * (2.0 / n)

        i, j = _propose_swap(h, k)

        # perform swap
        h[i], h[j] = h[j], h[i]

        # evaluate the new state
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max_idx = int(np.argmax(new_corr))
        new_max_corr = float(new_corr[new_max_idx])
        new_c5 = new_max_corr * (2.0 / n)

        if new_c5 <= curr_c5 + 1e-12:
            # accept move
            max_idx, max_corr = new_max_idx, new_max_corr
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # reject – revert swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_c5
