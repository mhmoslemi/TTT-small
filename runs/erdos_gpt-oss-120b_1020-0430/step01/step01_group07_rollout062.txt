def greedy_optimize(start_h, time_limit):
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_correlation(h).max() * (2.0 / n)
    start = time.time()
    while time.time() - start < time_limit:
        corr = compute_correlation(h)
        max_val = corr.max()
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
        # try to improve
        d_candidates = np.argsort(-corr)[:5]  # top 5 shifts (including maybe zero)
        improved = False
        for idx in d_candidates:
            d = idx - (n - 1)
            if d == 0:
                continue
            bad, good = mismatched_indices(h, d)
            if bad.size and good.size:
                i = int(np.random.choice(bad))
                j = int(np.random.choice(good))
                # swap
                h[i], h[j] = h[j], h[i]
                improved = True
                break
        if not improved:
            # cannot improve with current h, break out
            break
    return best_h, best_c

import time, random, math, numpy as np

def _c5_metric(h):
    """Return C5 for a (float) step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _mismatched_indices(h, d):
    """
    For a binary vector h and integer shift d,
    return two arrays:
      bad  – indices i where h[i]==1 and h[i+d]==0
      good – indices i where h[i]==0 and h[i+d]==1
    Only indices with i+d inside the array are considered.
    """
    n = h.shape[0]
    if d >= 0:
        n_ov = n - d
        hi = h[:n_ov]
        hj = h[d:]
        base = 0
    else:
        dd = -d
        n_ov = n - dd
        hi = h[dd:]
        hj = h[:n_ov]
        base = dd
    bad = np.where((hi == 1) & (hj == 0))[0] + base
    good = np.where((hi == 0) & (hj == 1))[0] + base
    return bad, good


def _targeted_anneal(start_h, max_seconds):
    """Simulated‑annealing that preferentially swaps mismatched pairs."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0, T_end = 0.04, 1e-6               # temperature schedule

    while time.time() - start < max_seconds:
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr_full))
        d = idx_max - (n - 1)

        bad, good = _mismatched_indices(h, d)
        if bad.size and good.size:
            i = int(np.random.choice(bad))
            j = int(np.random.choice(good))
        else:                               # fallback to a random pair
            i, j = random.sample(range(n), 2)

        hi, hj = h[i], h[j]
        h[i], h[j] = hj, hi                # propose swap

        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_c = new_corr.max() * (2.0 / n)

        progress = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        if new_c < cur_c or random.random() < math.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj            # revert

    return best_h, best_c


def _random_binary(n):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = n // 2
    idx = np.random.permutation(n)[:ones]
    h[idx] = 1
    return h
