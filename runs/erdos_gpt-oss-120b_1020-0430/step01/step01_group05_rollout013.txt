```
def simulated_annealing(h, n_iter, target_sum, start_temp, end_temp):
    curr_h = h.copy()
    curr_val = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_val = curr_val
    for iter in range(1, n_iter+1):
        T = start_temp * ( (end_temp/start_temp) ** (iter/n_iter) )
        # propose move
        i, j = np.random.randint(0, len(curr_h), size=2)
        if i==j: continue
        hi, hj = curr_h[i], curr_h[j]
        # max delta
        delta_max = min(1-hi, hj)
        delta_min = -min(hi, 1-hj)
        if delta_max < delta_min: continue
        delta = np.random.uniform(delta_min, delta_max)
        # limit magnitude maybe
        # apply
        cand = curr_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        # ensure within [0,1] anyway.
        cand_val = compute_c5(cand)
        # acceptance
        if cand_val < curr_val or np.random.rand() < np.exp(-(cand_val-curr_val)/T):
            curr_h, curr_val = cand, cand_val
            if curr_val < best_val:
                best_h, best_val = curr_h.copy(), curr_val
    return best_h, best_val
```

```
def project_sum_and_bounds(x, target):
    low, high = -1.0, 1.0
    for _ in range(60):
        lam = (low+high)/2
        proj = np.clip(x - lam, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(x - ((low+high)/2), 0.0, 1.0)
```

def smooth_max(arr, alpha=20.0):
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

    corr = np.correlate(h, 1.0 - h, mode='full')

import time
import random
import math
import numpy as np
from scipy.optimize import minimize

# global for equality constraint used by SLSQP
TARGET_SUM_GLOBAL = None


def project_sum_and_bounds(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while enforcing sum(x)=target."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete objective C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth surrogate of the max‑overlap objective."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular bump of height 1 at centre, linearly to 0 at ends."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + 0.5 * dx
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_sum_and_bounds(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2]."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + 0.5 * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_sum_and_bounds(h, n / 2.0)


def upsample_to(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of arr from its length to new_n points."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h = np.interp(x_new, x_old, arr)
    return project_sum_and_bounds(h, new_n / 2.0)


def _gradient_step(h: np.ndarray, target: float) -> np.ndarray:
    """One projected sub‑gradient step for the max‑overlap objective."""
    n = h.shape[0]
    # locate worst shift (max of h*(1‑h) correlation)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # signed integer shift
    g = np.ones_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]
            g[: s] -= h[s:]
    # step size (will be tuned externally)
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C₅ bound."""
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # --------------------------------------------------------------
    n = 500                      # discretisation points (even)
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    best_h = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # build initial pool
    seeds = []

    if "initial_h_values" in globals():
        try:

        except Exception:
            pass

    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))

    # a few random feasible seeds
    for _ in range(5):
        rnd = np.random.rand(n)
        seeds.append(project_sum_and_bounds(rnd, target_sum))

    # --------------------------------------------------------------
    # helper to possibly exit early
    def maybe_break():
        return time.time() - start > budget_s - 5.0

    # --------------------------------------------------------------
    # main loop over seeds
    for seed_h in seeds:
        if maybe_break():
            break

        # ---- local smooth optimisation (SLSQP) --------------------
        try:
            res = minimize(
                objective_smooth,
                seed_h,
                args=(dx, 20.0),

        h_cur = project_sum_and_bounds(res.x, target_sum)
        cur_val = compute_c5(h_cur)
        if cur_val < best_val:

            if best_val <= 0.3808:

        eta = 0.12
        for it in range(2000):
            if maybe_break():

            g = _gradient_step(h_cur, target_sum)
            cand = h_cur - eta * g

            cand = project_sum_and_bounds(cand, target_sum)

                if eta < 1e-6:
                    break

        max_anneal_iters = 250000
        start_temp, end_temp = 5e-4, 5e-8
        start_move, end_move = 0.12, 0.001

        for it in range(max_anneal_iters):
            if maybe_break():
                break
            frac = it / max_anneal_iters
            T = start_temp * (1 - frac) + end_temp * frac
            max_delta = start_move * (1 - frac) + end_move * frac

            # propose a feasible 2‑point transfer
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            hi, hj = h_cur[i], h_cur[j]
            delta_max = min(1.0 - hi, hj)
            delta_min = -min(hi, 1.0 - hj)
            if delta_max < delta_min:
                continue
            low = max(delta_min, -max_delta)
            high = min(delta_max, max_delta)
            if low >= high:

            delta = np.random.uniform(low, high)

            cand = h_cur.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            # projection just in case of numerical drift
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_sum_and_bounds(cand, target_sum)

            cand_val = compute_c5(cand)
            if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
                h_cur, cur_val = cand, cand_val
                if cur_val < best_val:
                    best_val, best_h = cur_val, h_cur.copy()
                    if best_val <= 0.3808:
                        return (best_h.tolist(), float(best_val), int(n))

        # end of seed processing

    # --------------------------------------------------------------
    # final greedy hill‑climbing on the best found point
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    h_cur = best_h.copy()
    cur_val = best_val
    while not maybe_break():
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        # small perturbations for fine tuning
        delta = np.random.uniform(delta_min, delta_max) * 0.02
        cand = h_cur.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_sum_and_bounds(cand, target_sum)
        cand_val = compute_c5(cand)
        if cand_val < cur_val:
            h_cur, cur_val = cand, cand_val
            if cur_val < best_val:
                best_val, best_h = cur_val, h_cur.copy()
                if best_val <= 0.3808:

    # final safety projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
