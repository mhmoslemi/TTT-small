def compute_c5_fft(h):
    n = len(h)
    L = 1 << ((2*n-1).bit_length())
    # compute convolution h and (1-h) via FFT
    # h and 1-h are real arrays.
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(H * G, L)
    # conv length L. The linear convolution result (full) is conv[:2*n-1]
    corr = conv[:2*n-1]
    max_corr = corr.max()
    return max_corr * (2.0 / n)
```

import time, random, math, numpy as np

def compute_c5_corr(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)
    # take linear part
    corr = corr[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)
```

import time, random, math
import numpy as np

# ----------------------------------------------------------------------
# Core evaluation (same as the harness uses)
def compute_c5(h: np.ndarray) -> float:
    """
    Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using FFT correlation.
    h is a 1‑D array of values on a uniform grid over [0,2].
    """
    n = h.shape[0]
    # length for linear correlation via zero‑padding
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[: 2 * n - 1]
    max_corr = np.max(corr)  # should be integer for binary h
    return float(max_corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Small utilities
def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, init)


def binarize_by_top(v: np.ndarray, weight: int) -> np.ndarray:
    """Return a binary vector with exactly `weight` ones taken from the largest entries."""
    idx = np.argpartition(-v, weight)[:weight]
    h = np.zeros_like(v, dtype=float)
    h[idx] = 1.0
    return h


def make_balanced(n: int, weight: int, alpha: float) -> np.ndarray:
    """Sturmian‑type balanced word (density ½) using rotation by `alpha`."""
    vals = (np.arange(n, dtype=float) * alpha) % 1.0
    h = (vals < 0.5).astype(float)
    cur = int(h.sum())
    if cur > weight:
        # turn off surplus ones
        ones = np.flatnonzero(h)[: cur - weight]
        h[ones] = 0.0
    elif cur < weight:
        zeros = np.flatnonzero(1.0 - h)[: weight - cur]
        h[zeros] = 1.0
    return h


def make_thue_morse(n: int, weight: int) -> np.ndarray:
    """Generate a Thue–Morse word and adjust to exact weight."""
    bits = np.fromiter((bin(i).count("1") % 2 for i in range(n)), dtype=int)
    # Convert to a 0/1 sequence with roughly half ones
    h = (bits == 0).astype(float)
    cur = int(h.sum())
    if cur > weight:
        excess = cur - weight
        ones = np.flatnonzero(h)[:excess]
        h[ones] = 0.0
    elif cur < weight:
        deficit = weight - cur
        zeros = np.flatnonzero(1.0 - h)[:deficit]
        h[zeros] = 1.0
    return h


def make_central_block(n: int, weight: int) -> np.ndarray:
    """Single contiguous block of ones centred in the interval."""
    h = np.zeros(n, dtype=float)
    start = (n - weight) // 2
    h[start : start + weight] = 1.0
    return h


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing for binary vectors
def hill_climb_from(
    h_start: np.ndarray,
    n: int,
    weight: int,
    start_time: float,
    budget_s: int,
    target_val: float,
):
    """
    Random swap hill‑climb with a simple temperature schedule.
    Returns the best binary vector found and its C5 value.
    """
    h = h_start.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    ones = set(map(int, np.flatnonzero(h == 1.0)))
    zeros = set(map(int, np.flatnonzero(h == 0.0)))

    # simple linear cooling (max temp 0.05)
    max_temp = 0.05

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.3:
            break
        if best_val <= target_val:
            break

        # temperature proportional to remaining time
        temp = max_temp * max(0.0, (budget_s - elapsed) / budget_s)

        # pick a random 1 and a random 0 to swap
        i_on = random.choice(tuple(ones))
        i_off = random.choice(tuple(zeros))

        # perform swap
        h[i_on] = 0.0
        h[i_off] = 1.0

        cand_val = compute_c5(h)
        delta = cand_val - cur_val

        accept = False
        if delta <= 0.0:
            accept = True
        elif temp > 0.0:
            if random.random() < math.exp(-delta / temp):
                accept = True

        if accept:
            cur_val = cand_val
            ones.remove(i_on)
            zeros.add(i_on)
            zeros.remove(i_off)
            ones.add(i_off)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert swap
            h[i_on] = 1.0
            h[i_off] = 0.0

    return best_h, best_val


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h=1 that yields a low C5.
    Returns (list_of_h, C5_value, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    target_val = 0.3808

    # problem size – larger n gives finer resolution (still cheap for FFT)
    n = 500                     # even → weight = n/2
    weight = n // 2

    best_h = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Build a list of promising initial binary vectors
    seed_vectors = []

    # parent construction, if available
    if "initial_h_values" in globals():
        try:
            init = np.array(initial_h_values, dtype=float)
            if init.shape[0] != n:
                init = upsample_initial(init, n)
            seed_vectors.append(binarize_by_top(init, weight))
        except Exception:
            pass

    # deterministic seeds
    seed_vectors.append(make_central_block(n, weight))
    seed_vectors.append(make_balanced(n, weight, alpha=(math.sqrt(5) - 1) / 2))  # golden ratio
    seed_vectors.append(make_balanced(n, weight, alpha=math.sqrt(2) - 1))
    seed_vectors.append(make_thue_morse(n, weight))

    # ------------------------------------------------------------------
    # Run hill climbing from each seed (using shared time budget)
    for sv in seed_vectors:
        h_opt, val = hill_climb_from(
            sv, n, weight, start, budget_s, target_val
        )
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        if best_val <= target_val:
            break

    # ------------------------------------------------------------------
    # Additional random restarts if time permits
    while time.time() - start < budget_s - 0.7:
        # random feasible binary vector
        rand_h = np.zeros(n, dtype=float)
        idx = np.random.choice(n, weight, replace=False)
        rand_h[idx] = 1.0
        h_opt, val = hill_climb_from(
            rand_h, n, weight, start, budget_s, target_val
        )
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        if best_val <= target_val:
            break

    # ------------------------------------------------------------------
    # Safety checks / final re‑evaluation
    if best_h is None:
        best_h = make_central_block(n, weight)
        best_val = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact binary nature (tiny numerical drift may appear)
    best_h = (best_h > 0.5).astype(float)

    # final C5 recomputation with the official routine
    final_val = compute_c5(best_h)

    return (best_h.tolist(), float(final_val), int(n))
