def gradient_descent_phase(h, dx, target_sum, time_left):
    alphas = [5, 10, 20, 40]
    for alpha in alphas:
        step = 0.5
        while True:
            # check time
            if time.time() - start > budget_s - 0.5: break outer
            # compute corr and gradient
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            w = np.exp(alpha * (corr - max_corr))
            w_sum = w.sum()
            w /= w_sum
            # convs
            conv_plus = np.convolve(w[::-1], h, mode='full')
            conv_minus = np.convolve(w, h, mode='full')
            slice_start = n-1
            sum_plus = conv_plus[slice_start:slice_start+n]
            sum_minus = conv_minus[slice_start:slice_start+n]
            g = dx * (1.0 - sum_plus - sum_minus)
            # candidate
            h_cand_raw = h - step * g
            h_cand = project_onto_bounds_and_sum(h_cand_raw, target_sum)
            cand_val = compute_c5(h_cand)
            cur_val = compute_c5(h)
            if cand_val < cur_val:
                h = h_cand
                # adjust step
                step *= 1.1
                # update best
                if cand_val < best_val: best_val = cand_val; best_h = h.copy()
            else:
                step *= 0.5
                if step < 1e-6: break # exit inner while for this alpha
    return h

def project_onto_bounds_and_sum(x, target):
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

import time
import random
import numpy as np

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n ∩ {sum = target} using bisection on the Lagrange multiplier."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x) (1‑h(x+k)) dx, discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_triangle(n: int) -> np.ndarray:
    """Triangular shape: peak 1 at centre, linear to 0 at edges, then projected."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2]; then projected to exact integral."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent solution to a new size n, then projected."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

    n = 300
    dx = 2.0 / n
    target_sum = n / 2.0

    if "initial_h_values" in globals():

        except Exception:
            pass

    seed_vectors.append(make_sin2(n))

    # a few random feasible starts
    for _ in range(2):
        rnd = np.random.rand(n)
        seed_vectors.append(project_onto_bounds_and_sum(rnd, target_sum))

    best_h = None
    best_val = np.inf

    # evaluate the initial candidates
    for h0 in seed_vectors:
        val0 = compute_c5(h0)
        if val0 < best_val:
            best_val = val0
            best_h = h0.copy()

    # ------------------------------------------------------------------
    # Core optimisation: projected gradient descent with continuation of α,
    # followed by a biased random local search.
    for h_init in seed_vectors:
        if time.time() - start > budget_s - 0.5:

        h = h_init.copy()

        # ----------- projected gradient descent with increasing α ----------

            if delta < 1e-12:
                continue

            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta

            cand_val = compute_c5(cand)
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                h = cand

                cur_val = compute_c5(h)
                if cand_val < cur_val:
