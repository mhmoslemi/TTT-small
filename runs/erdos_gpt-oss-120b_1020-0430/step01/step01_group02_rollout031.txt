import time, numpy as np, scipy.optimize as opt

def _evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    # same iterative but may be simplified
    # We'll use while loop as earlier, but careful about infinite loops.
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Use simple water-filling: compute free indices
    # We'll attempt to distribute diff equally among free indices.
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        free = np.where(mask)[0]
        per = diff / len(free)
        # adjust each
        if diff > 0:
            add = np.minimum(per, 1.0 - x[free])
            x[free] += add
            diff -= add.sum()
        else:
            sub = np.minimum(-per, x[free])
            x[free] -= sub
            diff += sub.sum()
    return np.clip(x, 0.0, 1.0)

def _slsqp_opt(x0, target, deadline):
    n = len(x0)
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    def obj(x):
        return _evaluate_c5(x)
    # Loop maybe to fit iterations
    res = opt.minimize(obj, x0, method="SLSQP",
                       bounds=bounds, constraints=[cons],
                       options={"maxiter": 200, "ftol": 1e-9, "disp": False})
    # Ensure feasible
    x = np.clip(res.x, 0.0, 1.0)
    x = _adjust_sum(x, target)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step values h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto {0≤x≤1, sum(x)=target}."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative water‑filling
    max_iter = 500
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        free = np.where(mask)[0]
        if free.size == 0:
            break
        per = diff / free.size
        if diff > 0.0:
            add = np.minimum(per, 1.0 - x[free])
            x[free] += add
            diff -= add.sum()
        else:
            sub = np.minimum(-per, x[free])
            x[free] -= sub
            diff += sub.sum()
    return np.clip(x, 0.0, 1.0)


def _slsqp_opt(x0: np.ndarray, target: float, time_deadline: float) -> np.ndarray:
    """Run a short SLSQP optimisation from x0."""
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda v: v.sum() - target}

    def obj(v):
        return _evaluate_c5(v)

    # keep the call cheap – limit iterations
    opts = {"maxiter": 200, "ftol": 1e-9, "disp": False}
    res = minimize(obj, x0, method="SLSQP", bounds=bounds,
                   constraints=[cons], options=opts)
    v = np.clip(res.x, 0.0, 1.0)
    v = _adjust_sum(v, target)
    # honour the deadline (just in case)
    if time.time() > time_deadline:
        raise TimeoutError
    return v


def _hill_climb(h: np.ndarray, c5: float, deadline: float, rng: np.random.Generator):
    """Simple pairwise mass‑exchange hill climbing."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5


def _smooth_gradient_descent(h: np.ndarray, target: float,
                            deadline: float, rng: np.random.Generator):
    """Projected gradient descent on a smooth approximation of max overlap."""
    n = h.size
    step = 0.5
    alpha = 1.0
    best_h = h.copy()
    best_c5 = _evaluate_c5(h)
    cur_c5 = best_c5

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        # softmax weights over shifts
        exp_w = np.exp(alpha * (corr - max_corr))
        w = exp_w / exp_w.sum()

        # term1 = conv(w, h)   (contributes h_{j-s})
        conv1 = np.convolve(w, h, mode="full")
        term1 = conv1[n - 1:n - 1 + n]

        # term2 = conv(w[::-1], h)   (contributes h_{j+s})
        conv2 = np.convolve(w[::-1], h, mode="full")
        term2 = conv2[n - 1:n - 1 + n]

        grad = 1.0 - term1 - term2

        h_candidate = h - step * grad
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate = _adjust_sum(h_candidate, target)

        cand_c5 = _evaluate_c5(h_candidate)
        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = h_candidate.copy(), cand_c5
        if cand_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_candidate, cand_c5
        else:
            # occasional stochastic jump to escape plateaus
            if rng.random() < 0.01:
                h, cur_c5 = h_candidate, cand_c5

        step = max(step * 0.995, 1e-5)
        alpha = min(alpha * 1.02, 30.0)

    return best_h, best_c5
