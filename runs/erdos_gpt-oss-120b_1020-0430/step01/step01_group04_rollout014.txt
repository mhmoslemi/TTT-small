def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def project_onto_box_sum(v: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Euclidean projection onto {0 <= x <= 1, sum x = target}."""
    # binary search for lambda such that sum clip(v+λ) = target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def gradient_softmax(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of log‑sum‑exp (soft‑max) objective w.r.t. h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    # stable soft‑max
    a_conv = alpha * conv
    a_max = np.max(a_conv)
    w = np.exp(a_conv - a_max)
    w /= w.sum()                     # w sums to 1
    # convolutions needed for gradient
    conv1 = np.convolve(w, h, mode="full")          # Σ_s w_s * h_{i‑s}
    w_rev = w[::-1]
    conv2 = np.convolve(w_rev, h, mode="full")      # Σ_s w_s * h_{i+s}
    offset = 2 * n - 2
    grad = alpha * (1.0 - conv1[:n] - conv2[offset:offset + n])
    return grad


def upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving total sum."""
    old_n = v.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, v)
    target = new_n / 2.0
    up = up * (target / up.sum())
    np.clip(up, 0.0, 1.0, out=up)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        np.clip(up, 0.0, 1.0, out=up)
    return up


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random h with values in [0,1] and exact integral 1."""
    a = rng.random(n)
    a *= (n / 2.0) / a.sum()
    np.clip(a, 0.0, 1.0, out=a)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        np.clip(a, 0.0, 1.0, out=a)
    return a


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape, renormalised to integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    np.clip(tri, 0.0, 1.0, out=tri)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        np.clip(tri, 0.0, 1.0, out=tri)
    return tri


def binary_local_search(bin_vec: np.ndarray, rng: np.random.Generator,
                       deadline: float, max_trials: int = 20000) -> tuple:
    """Hill‑climbing on binary vectors (exactly n/2 ones)."""
    n = bin_vec.shape[0]
    target_ones = n // 2
    best = bin_vec.copy()
    best_c5 = compute_c5(best)
    ones = np.flatnonzero(best == 1)
    zeros = np.flatnonzero(best == 0)

    trials = 0
    while time.time() < deadline and trials < max_trials:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = cand_c5
            # update index lists
            ones = np.flatnonzero(best == 1)
            zeros = np.flatnonzero(best == 0)
        trials += 1
    return best, best_c5

        # a few random feasible vectors
        for _ in range(4):
            starts.append(random_feasible(n, rng))

        # ---- process each start ------------------------------------------------
        for h_start in starts:
            if time.time() > time_limit:
                break

            h = h_start.copy()
            # Adam state
            m = np.zeros_like(h)
            v = np.zeros_like(h)
            beta1 = 0.9
            beta2 = 0.999
            beta1_pow = 1.0
            beta2_pow = 1.0
            base_lr = 0.2

            # schedule of α values (smooth to sharp)
            alphas = [5.0, 15.0, 30.0, 60.0, 120.0]

            for alpha in alphas:
                for _ in range(70):                     # moderate number of iterations
                    if time.time() > time_limit:
                        break
                    grad = gradient_softmax(h, alpha)

                    # Adam update
                    m = beta1 * m + (1.0 - beta1) * grad
                    v = beta2 * v + (1.0 - beta2) * (grad * grad)
                    beta1_pow *= beta1
                    beta2_pow *= beta2
                    m_hat = m / (1.0 - beta1_pow)
                    v_hat = v / (1.0 - beta2_pow)

                    step = (base_lr / (1.0 + alpha / 30.0)) * m_hat / (np.sqrt(v_hat) + 1e-8)
                    h = project_onto_box_sum(h - step, target_sum)

                # occasional evaluation
                cur_c5 = compute_c5(h)
                if cur_c5 + 1e-9 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    best_n = n

    # ---- post‑processing: binary polishing ------------------------------------
    if best_h is not None and best_n % 2 == 0:
        # round to a binary vector with exactly n/2 ones
        n = best_n
        ones_needed = n // 2
        idx = np.argpartition(-best_h, ones_needed)[:ones_needed]
        binary_vec = np.zeros(n, dtype=float)
        binary_vec[idx] = 1.0

        # run a short local search if time permits
        if time.time() < time_limit:
            deadline = min(time_limit, time.time() + (budget_s * 0.02))
            refined, refined_c5 = binary_local_search(binary_vec, rng, deadline)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined.astype(float)
                best_n = n

    # ---- fallback if everything failed ----------------------------------------
    if best_h is None:
        n = 80
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final sanity projection (numerical drift)
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = project_onto_box_sum(best_h, target)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
