def solve_slsqp(initial_h, max_iter=200):
    n = len(initial_h)
    h0 = project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0,1)]*n + [(0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1 - h, mode='full')
    def eq(x):
        return np.sum(x[:n]) - n/2

    cons = [{'type':'ineq','fun':ineq}, {'type':'eq','fun':eq}]
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                            constraints=cons, options={'maxiter': max_iter, 'ftol':1e-9})
    h_opt = res.x[:n] if res.success else h0
    h_opt = project_to_feasible(h_opt, n)
    return h_opt, compute_c5(h_opt)

import time
import numpy as np
from scipy import optimize

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given step function (dx scaling already accounted for)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all integer shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce the integral constraint sum(h)=n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Local optimisation using SLSQP, returns (h_opt, c5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _local_random_refine(h: np.ndarray, time_budget: float) -> tuple:
    """Very cheap random perturbation refinement while time permits."""
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    # sigma shrinks as we spend time
    sigma0 = 0.07
    while time.time() - start < time_budget:
        # linear decay of sigma
        elapsed = time.time() - start
        sigma = sigma0 * (1 - elapsed / time_budget)
        cand = best_h + sigma * np.random.randn(n)
        cand = _project_to_feasible(cand, n)
        val = _c5_from_h(cand)
        if val < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val
