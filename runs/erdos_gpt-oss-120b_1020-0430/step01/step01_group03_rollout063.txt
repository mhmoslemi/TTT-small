    g = np.ones_like(h)  # because derivative term 1
    # subtract contributions
    if k >= 0:

        g[:n - k] -= h[k:]  # because for i=0..n-k-1, h[i+k] = h[i+k]

        g[k:] -= h[:n - k]
    else:

        k_abs = -k

def project_to_sum_and_bounds(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # allocate using capacity proportionally
    # while diff not satisfied due to numerical limit
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h
        else:
            cap = -h  # negative values represent ability to decrease
        cap_sum = cap.sum()
        if cap_sum < 1e-12:
            break
        # distribute proportionally
        delta = diff * cap / cap_sum
        # ensure we don't exceed bound due to numeric issues: clamp
        if diff > 0:
            delta = np.minimum(delta, cap)
        else:
            delta = np.maximum(delta, cap)  # cap negative, delta will be negative
        h += delta
        diff = target - h.sum()
    # final small correction
    if abs(diff) > 1e-8:
        # distribute uniformly among non-bound entries
        adjustable = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if adjustable.size > 0:
            per = diff / adjustable.size
            h[adjustable] += per
            # clip again
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Exact C5 for a step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def _project_to_sum_and_bounds(x, target):
    """Clip x to [0,1] and adjust its sum to *target* by proportional redistribution."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # distribute proportionally to the available capacity
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            cap = 1.0 - x
        else:
            cap = -x          # negative capacities for decreasing
        cap_sum = cap.sum()
        if cap_sum == 0.0:
            break
        delta = diff * cap / cap_sum
        # keep inside bounds
        if diff > 0.0:
            delta = np.minimum(delta, cap)
        else:
            delta = np.maximum(delta, cap)
        x += delta
        diff = target - x.sum()
    # final tiny fix (uniformly on non‑saturated entries)
    if abs(diff) > 1e-9:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n, rng):
    """Uniform random feasible vector (0≤h≤1, Σh=n/2)."""
    h = rng.random(n)
    return _project_to_sum_and_bounds(h, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
            h = cand
        # shrink step periodically
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    # final feasibility clean‑up
    best_h = _project_to_sum_and_bounds(best_h, target)
    return best_h, best_c


# ---------- subgradient descent ----------
def _active_shifts(corr, max_corr, tol=1e-12):
    """Indices of shifts attaining the maximum (within tolerance)."""
    return np.where(corr >= max_corr - tol)[0] - (corr.size // 2)


def subgradient_optimize(h0, target, time_limit):
    """Projected sub‑gradient descent on the exact max‑overlap objective."""
    h = h0.copy()
    n = h.size
    factor = 2.0 / n
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    alpha = 0.3
    decay = 0.9998
    min_alpha = 1e-6
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = corr.max()
        cur_c = max_corr * factor
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # active shifts
        ks = _active_shifts(corr, max_corr)
        # accumulate contributions of h[i+k] and h[i‑k]
        contrib = np.zeros_like(h)
        for k in ks:
            if k >= 0:
                contrib[:n - k] += h[k:]
                contrib[k:] += h[:n - k]
            else:
                L = -k
                contrib[L:] += h[:n - L]
                contrib[:n - L] += h[L:]
        g = len(ks) - contrib                     # sub‑gradient of max
        h_new = h - alpha * g
        h_new = _project_to_sum_and_bounds(h_new, target)
        # simple acceptance rule (only if not worse)
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        new_c = corr_new.max() * factor
        if new_c <= cur_c:
            h = h_new
        alpha = max(min_alpha, alpha * decay)
    return best_h, best_c


# ---------- SLSQP with soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP minimisation of the soft‑max surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift may remain)
    return _project_to_sum_and_bounds(x, CURRENT_TARGET)


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- initialise -------------------------------------------------------

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- Phase 1: fast sub‑gradient descent (≈30 % of budget) -------------
    time_left = budget_s - (time.time() - start_time)
    phase1_budget = min(time_left * 0.30, 300.0)
    h_sg, c_sg = subgradient_optimize(best_h, n / 2.0, phase1_budget)
    if c_sg < best_c:
        best_h, best_c = h_sg, c_sg

    # ---- Phase 2: deterministic hill‑climb (short) -----------------------
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30.0:
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.4)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ---- Phase 3: SLSQP ladder with increasingly sharp beta ---------------
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=400)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            continue

    # ---- Phase 4: random restarts with brief sub‑gradient (till time) ------
    while time.time() - start_time < budget_s * 0.98:
        h_rand = random_feasible_vector(n, rng)
        # very short sub‑gradient (≈1 s) to diversify
        sub_time = min(1.0, budget_s - (time.time() - start_time))
        h_tmp, c_tmp = subgradient_optimize(h_rand, n / 2.0, sub_time)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---- Optional up‑sampling (if time permits) -------------------------
    if time.time() - start_time < budget_s * 0.99 and n * 2 <= 200:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = _project_to_sum_and_bounds(h_up, target_up)
        # a quick refinement on the finer grid
        sub_time = min(120.0, budget_s - (time.time() - start_time))
        h_up, c_up = subgradient_optimize(h_up, target_up, sub_time)
        for beta in betas[:4]:
            if time.time() - start_time > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta=beta, max_iter=300)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ---- final clean‑up ----------------------------------------------------
    best_h = _project_to_sum_and_bounds(best_h, n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
