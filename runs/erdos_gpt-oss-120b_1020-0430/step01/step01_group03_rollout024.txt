def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    # random pairwise transfers to mix the vector
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            if diff > 0:

            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# Soft‑max surrogate for max‑overlap
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_{k} Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    diff = CURRENT_TARGET - x.sum()

            x[i] += add
            diff = CURRENT_TARGET - x.sum()

        sm = np.convolve(best, kernel, mode="same")
        sm = np.clip(sm, 0.0, 1.0)
        # renormalise the sum exactly
        diff = target - sm.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, sm.size)
                if diff > 0:
                    add = min(diff, 1.0 - sm[i])
                else:
                    add = max(diff, -sm[i])
                sm[i] += add
                diff = target - sm.sum()
        c = compute_c5(sm)
        if c < best_c:
            best, best_c = sm, c
        passes += 1
    return best, best_c


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    global RNG
    RNG = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Initialise h – upsample parent if present, otherwise random.
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.size
        n = 200  # target resolution (≤200)
        # linear interpolation to 200 points
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
    else:
        n = 200
        h0 = random_feasible_vector(n, RNG)

    # enforce feasibility (clip + exact sum)
    h0 = np.clip(h0, 0.0, 1.0)
    target_sum = n / 2.0
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = RNG.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h0[i])
            else:
                add = max(diff, -h0[i])
            h0[i] += add
            diff = target_sum - h0.sum()

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------------
    # Phase 1 – Simulated annealing on the full grid
    # --------------------------------------------------------------
    phase1_budget = budget_s * 0.58
    h_sa, c_sa = simulated_annealing(
        best_h, RNG, time_limit=phase1_budget, init_temp=0.5, final_temp=1e-4
    )
    if c_sa < best_c:
        best_h, best_c = h_sa, c_sa

    # --------------------------------------------------------------
    # Phase 2 – Increasing‑beta SLSQP refinements
    # --------------------------------------------------------------
    betas = [5, 10, 20, 40, 80, 160]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.90:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, rng=RNG, max_iter=800)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            continue

    # --------------------------------------------------------------
    # Phase 3 – Random‑restart quick searches (if time permits)
    # --------------------------------------------------------------
    restarts = 0
    while time.time() - start_time < budget_s * 0.96 and restarts < 8:
        h_start = random_feasible_vector(n, RNG)
        # short SA (≈200 iterations)
        h_tmp, c_tmp = simulated_annealing(
            h_start, RNG, time_limit=budget_s * 0.04, init_temp=0.4, final_temp=5e-4
        )
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp
        # two quick SLSQP passes
        for beta in (5, 20):
            if time.time() - start_time > budget_s * 0.96:
                break
            try:
                trial = slsqp_optimize(best_h, beta=beta, rng=RNG, max_iter=150)
                c_trial = compute_c5(trial)
                if c_trial < best_c:
                    best_h, best_c = trial, c_trial
            except Exception:
                pass
        restarts += 1

    # --------------------------------------------------------------
    # Phase 4 – Gentle smoothing & final refinement
    # --------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:
        h_smooth, c_smooth = smooth_refine(best_h, RNG, time_limit=remaining * 0.6)
        if c_smooth < best_c:
            best_h, best_c = h_smooth, c_smooth
        # final SLSQP with high beta if time left
        if time.time() - start_time < budget_s * 0.99:
            try:
                final = slsqp_optimize(best_h, beta=300, rng=RNG, max_iter=500)
                c_final = compute_c5(final)
                if c_final < best_c:
                    best_h, best_c = final, c_final
            except Exception:
                pass

    # --------------------------------------------------------------
    # Final feasibility clean‑up
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = RNG.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
