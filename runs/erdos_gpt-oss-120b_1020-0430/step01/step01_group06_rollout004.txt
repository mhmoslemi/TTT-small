def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    end = start + budget_s * 0.97          # tiny safety margin

    # ----------------------------------------------------------------------
    # Problem size and initial candidate(s)
    # ----------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        n = int(init.shape[0])
    except NameError:
        n = 140          # even number, reasonable resolution
        init = None

    target_sum = n / 2.0
    dx = 2.0 / n

    def project_sum(v):
        """adjust v in-place so that sum(v)=target_sum while 0≤v≤1"""
        v[:] = np.clip(v, 0.0, 1.0)
        diff = v.sum() - target_sum
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:                         # need to decrease total
                cand = np.where(v > 1e-12)[0]
                if not cand.size:
                    break
                i = random.choice(cand.tolist())
                delta = min(v[i], diff)
                v[i] -= delta
                diff -= delta
            else:                                 # need to increase total
                cand = np.where(v < 1 - 1e-12)[0]
                if not cand.size:
                    break
                i = random.choice(cand.tolist())
                delta = min(1.0 - v[i], -diff)
                v[i] += delta
                diff += delta
        return v

    # start vectors
    starts = []
    if init is not None:
        starts.append(project_sum(init.copy()))
    # a few binary random starts
    for _ in range(5):
        v = np.zeros(n)
        v[: n // 2] = 1.0
        np.random.shuffle(v)
        starts.append(v)

    # ----------------------------------------------------------------------
    # Core evaluation utilities
    # ----------------------------------------------------------------------
    def c5_of(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(corr.max() * dx)

    def c5_and_shift(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        idx = int(corr.argmax())
        return float(corr[idx] * dx), idx - (n - 1)

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    def local_optimize(v, max_iter=2000):
        for _ in range(max_iter):
