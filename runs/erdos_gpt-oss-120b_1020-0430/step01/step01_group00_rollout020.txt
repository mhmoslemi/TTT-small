    return max_val, shift

def projected_subgradient(h, max_iters, step0):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)

    for t in range(1, max_iters+1):
        _, shift = max_corr_and_shift(h)
        g = gradient_max_shift(h, shift)
        alpha = step0 / np.sqrt(t)  # step schedule
        h_new = h - alpha * g
        h_new = project_to_feasible(h_new, n)
        val = compute_c5(h_new)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

def gradient_max_shift(h, shift):
    n = len(h)
    if shift >= 0:
        k = shift
        term1 = np.empty_like(h)
        term1[:n - k] = 1 - h[k:]
        term1[n - k:] = 0.0
        term2 = np.empty_like(h)
        term2[:k] = 0.0
        term2[k:] = h[:n - k]
    else:
        k = -shift
        term1 = np.empty_like(h)
        term1[k:] = 1 - h[:n - k]
        term1[:k] = 0.0
        term2 = np.empty_like(h)
        term2[:n - k] = h[k:]
        term2[n - k:] = 0.0
    return term1 - term2

def _subgradient_descent(h, n, max_iters, step0, time_left_func):
    best_h = h.copy()
    best_val = _compute_c5(h)
    for it in range(1, max_iters+1):
        if time_left_func() < 0.1:
            break
        # compute gradient for max shift
        _, shift = _max_corr_shift(h)
        grad = _gradient_shift(h, shift)
        alpha = step0 / np.sqrt(it)
        h = _project_to_feasible(h - alpha * grad, n)
        cur_val = _compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
    return best_h, best_val

def _project_to_feasible(h, n, tol=1e-9):
    h = np.clip(h, 0.0, 1.0)

    s = h.sum()
    if abs(s - target) <= tol:
        return h
    if s > target:

        # indices that can be reduced (>0)
        while excess > tol:
            free = np.where(h > 0)[0]
            if free.size == 0:
                break
            # total reducible amount
            reducible = h[free].sum()
            if reducible <= 0:
                break
            if excess <= reducible:
                # reduce proportionally
                reduction = excess / free.size
                # ensure not go below zero
                h[free] -= reduction
                # clip after
                h = np.clip(h, 0.0, 1.0)
                excess = 0.0
                break
            else:
                # zero out all free
                h[free] = 0.0
                excess -= reducible
    else:
        deficit = target - s
        while deficit > tol:
            free = np.where(h < 1)[0]
            if free.size == 0:
                break
            increase_possible = (1.0 - h[free]).sum()
            if deficit <= increase_possible:
                inc = deficit / free.size
                h[free] += inc
                h = np.clip(h, 0.0, 1.0)
                deficit = 0.0
                break
            else:
                h[free] = 1.0
                deficit -= increase_possible
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # add/subtract from any non-saturated element
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:

                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:

                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:

import time, math, numpy as np

# ----- helper utilities -------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) <= tol:
        return h
    if s > target:                      # need to decrease total mass
        excess = s - target
        while excess > tol:
            free = np.where(h > 0)[0]
            if free.size == 0:
                break
            reducible = h[free].sum()
            if excess <= reducible:
                dec = excess / free.size
                h[free] -= dec
                h = np.clip(h, 0.0, 1.0)
                break
            h[free] = 0.0
            excess -= reducible
    else:                               # need to increase total mass
        deficit = target - s
        while deficit > tol:
            free = np.where(h < 1)[0]
            if free.size == 0:
                break
            incable = (1.0 - h[free]).sum()
            if deficit <= incable:
                inc = deficit / free.size
                h[free] += inc
                h = np.clip(h, 0.0, 1.0)
                break
            h[free] = 1.0
            deficit -= incable
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _compute_c5(h: np.ndarray) -> float:
    """Return C5 for a step function h on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _max_corr_shift(h: np.ndarray):
    """Maximum correlation value and its shift (in index units)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # 0‑lag corresponds to index n‑1
    return float(corr[idx]), shift


def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the maximal correlation wrt h."""
    n = h.size
    if shift >= 0:
        k = shift
        term1 = np.empty_like(h)
        term1[: n - k] = 1.0 - h[k:]
        term1[n - k :] = 0.0
        term2 = np.empty_like(h)
        term2[:k] = 0.0
        term2[k:] = h[: n - k]
    else:
        k = -shift
        term1 = np.empty_like(h)
        term1[k:] = 1.0 - h[: n - k]
        term1[:k] = 0.0
        term2 = np.empty_like(h)
        term2[: n - k] = h[k:]
        term2[n - k :] = 0.0
    return term1 - term2


def _upscale_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(x_new, x_old, old_h)


# ----- optimisation primitives -------------------------------------------
def _subgradient_descent(h0: np.ndarray, max_it: int, step0: float, time_left):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)
    for it in range(1, max_it + 1):
        if time_left() < 0.05:
            break
        _, sh = _max_corr_shift(h)
        g = _gradient_shift(h, sh)
        alpha = step0 / math.sqrt(it)
        h = _project_to_feasible(h - alpha * g, n)
        val = _compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val


def _pairwise_refine(h0: np.ndarray, max_it: int, time_left):
    """Local random pairwise mass transfers."""
    n = h0.size
    best_h = h0.copy()
    best_val = _compute_c5(best_h)
    for _ in range(max_it):
        if time_left() < 0.02:

    Returns (list_of_h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # retrieve optional parent construction

    # decide grid sizes (prefer the supplied one, then larger even sizes)

    # multiples of the supplied size, capped at 300
    if init_h is not None:

            n = init_h.size * mul
            if n <= 300:

    # a few handcrafted sizes
    for n in (160, 200, 240, 256, 300):
        if n <= 300:
            candidates.append(n)
    # fallback if nothing else
    if not candidates:
        candidates = [128, 200]

    candidates = sorted(set(candidates))

    # main optimisation loop over grid sizes
    for n in candidates:
        if time_left() < 5.0:
            break

        # initialise base h for this grid
        if init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
