def simulated_annealing(h0, max_seconds):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_metric(h)
    cur_val = best_val

    start = time.time()
    # temperature schedule: T0 = 0.1 maybe, T_end=1e-5
    T0 = 0.1
    T_end = 1e-5

    max_step = 0.3  # allow larger steps early

    iteration = 0
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end/T0) ** progress

        # choose mode
        if random.random() < 0.5:
            # gradient-based pair
            # compute corr array maybe every few steps
            if iteration % 10 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full')
            t_max = corr.argmax()
            s = t_max - (n-1)
            # compute gradient vector
            g = np.zeros(n)
            # compute forward = 1 - h, backward = h
            forward = 1.0 - h
            backward = h
            # first term: (1 - h[i+s])
            if s >= 0:
                # i from 0 to n-1-s have forward[i+s]
                g[:n-s] += forward[s:]  # forward[i+s] for i = 0..n-s-1
                # second term: - h[i - s] for i from s..n-1 (i-s valid)
                g[s:] -= backward[:n-s]   # h[i - s] = back[i-s] where i = s..n-1
            else:
                # s negative => i+s = i - |s|
                # forward term defined for i >= |s|
                g[-s:] += forward[:n+s]  # shift: forward[i+s], i from -s .. n-1
                # second term: -h[i-s] => i-s = i+|s| valid for i from 0..n-1+ s
                g[:n+s] -= backward[-s:]  # backward[i - s] = backward[i+|s|]
            # Now g_i is derivative (positive where decreasing h reduces C)
            # find candidates
            pos_idx = np.where((g > 1e-12) & (h > 1e-12))[0]
            neg_idx = np.where((g < -1e-12) & (h < 1 - 1e-12))[0]
            if pos_idx.size == 0 or neg_idx.size == 0:
                # fallback to random pair
                i, j = random.sample(range(n), 2)
                low = max(-h[i], h[j] - 1.0)
                high = min(1.0 - h[i], h[j])
                if low >= high:
                    continue
                delta = random.uniform(low, high)
            else:
                i = int(rng.choice(pos_idx))
                j = int(rng.choice(neg_idx))
                # allowed delta magnitude
                low = max(-h[i], h[j] - 1.0)
                high = min(1.0 - h[i], h[j])
                # we want delta negative for i (decrease) and positive for j (increase)
                # So delta = -d where d>0 such that i decreases, j increases.
                # low <= delta <= high; low negative, high positive.
                # To move from i to j, we set delta = max(low, -max_step) maybe negative.
                # Let's set delta = np.clip(-min(max_step, h[i], 1-h[j]), low, high)
                delta_target = -min(max_step, h[i], 1.0 - h[j])
                # clamp
                delta = max(low, min(high, delta_target))
        else:
            # random pair
            i, j = random.sample(range(n), 2)
            low = max(-h[i], h[j] - 1.0)
            high = min(1.0 - h[i], h[j])
            if low >= high:
                continue
            delta = random.uniform(low, high)
        # apply move
        h[i] += delta
        h[j] -= delta

        # compute new metric
        new_val = _c5_metric(h)
        dC = new_val - cur_val
        accept = False
        if dC < 0:
            accept = True
        else:
            if T > 0:

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

def _prepare_initial(n):
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)

        for offset in range(n):
            i = (idx + offset) % n

                h[i] += diff

    return h

def _gradient_worst(h, corr=None):
    n = h.shape[0]
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    t_max = corr.argmax()
    s = t_max - (n - 1)
    forward = 1.0 - h
    backward = h
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += forward[s:]
        g[s:] -= backward[:n - s]
    else:
        p = -s
        g[p:] += forward[:n - p]
        g[:n - p] -= backward[p:]
    return s, g, corr
```

def _simulated_annealing(h0, max_seconds):
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_metric(best_h)
    cur_val = best_val
    start = time.time()
    T0 = 0.1
    T_end = 1e-6
    max_step = 0.3
    # We will recompute corr only when needed for gradient moves.
    corr = np.correlate(h, 1.0 - h, mode='full')
    iter_count = 0
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress

        # decide mode
        if random.random() < 0.5:
            # gradient-based
            # maybe recompute corr every few iterations
            if iter_count % 20 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full')
            s, g, _ = _gradient_worst(h, corr)
            pair = _select_pair_by_gradient(h, g)
            if pair is None:
                i, j = random.sample(range(n), 2)
                low = max(-h[i], h[j] - 1.0)
                high = min(1.0 - h[i], h[j])
                if low >= high:
                    continue
                delta = random.uniform(low, high)
            else:
                i, j = pair
                # allowed delta: negative to decrease i, positive to increase j
                max_transfer = min(max_step, h[i], 1.0 - h[j])
                delta = -max_transfer
                low = max(-h[i], h[j] - 1.0)
                high = min(1.0 - h[i], h[j])
                delta = max(low, min(high, delta))
        else:
            # random pair
            i, j = random.sample(range(n), 2)
            low = max(-h[i], h[j] - 1.0)
            high = min(1.0 - h[i], h[j])
            if low >= high:
                continue
            # choose delta; allow larger steps earlier
            # scale step by temperature maybe
            step_scale = max_step * (T / T0 + 0.1)  # ensure >0
            # compute allowed range limited by step_scale
            lim_low = max(low, -step_scale)
            lim_high = min(high, step_scale)
            if lim_low >= lim_high:
                delta = random.uniform(low, high)
            else:
                delta = random.uniform(lim_low, lim_high)

        # apply move
        h[i] += delta
        h[j] -= delta

        new_val = _c5_metric(h)
        dC = new_val - cur_val
        accept = False
        if dC < 0:
            accept = True
        else:
            # accept with probability exp(-dC/T)
            if T > 0 and random.random() < math.exp(-dC / T):
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        iter_count += 1

    return best_h, best_val

import time
import math
import random
import numpy as np

def _c5_metric(h):
    """Compute C_5 for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _prepare_initial(n):
    """Create an initial feasible vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder to a feasible entry
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
    return h

def _gradient_worst(h, corr=None):
    """Return worst shift, its gradient and correlation."""
    n = h.shape[0]
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode="full")
    t = int(np.argmax(corr))
    s = t - (n - 1)                     # shift value (may be negative)
    forward = 1.0 - h
    backward = h
    g = np.zeros(n, dtype=np.float64)

    if s >= 0:
        g[:n - s] += forward[s:]
        g[s:] -= backward[:n - s]
    else:
        p = -s
        g[p:] += forward[:n - p]
        g[:n - p] -= backward[p:]
    return s, g, corr

def _select_pair_by_gradient(h, g):
    """Choose i (to decrease) and j (to increase) based on gradient signs."""
    pos = np.where((g > 1e-12) & (h > 1e-12))[0]
    neg = np.where((g < -1e-12) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return None
    i = int(np.random.choice(pos))
    j = int(np.random.choice(neg))
    return i, j

def _simulated_annealing(start_h, max_seconds):
    """Hybrid SA with gradient‑biased moves."""
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_metric(best_h)
    cur_val = best_val

    start = time.time()
    T0 = 0.2                     # initial temperature (in C‑unit)
    T_end = 1e-8
    max_step = 0.5               # largest absolute delta allowed
    it = 0

    # keep a correlation cache for speed‑up in gradient mode
    corr = np.correlate(h, 1.0 - h, mode="full")

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        T = T0 * (T_end / T0) ** prog

        # ---------- decide move type ----------
        if random.random() < 0.55:        # gradient‑biased move
            if it % 15 == 0:              # refresh correlation periodically
                corr = np.correlate(h, 1.0 - h, mode="full")
            s, g, _ = _gradient_worst(h, corr)
            pair = _select_pair_by_gradient(h, g)
            if pair is None:
                i, j = random.sample(range(n), 2)
                low = max(-h[i], h[j] - 1.0)
                high = min(1.0 - h[i], h[j])
                if low >= high:
                    it += 1
                    continue
                delta = random.uniform(low, high)
            else:
                i, j = pair
                # move as much mass as feasible (most negative delta)
                low = max(-h[i], h[j] - 1.0)
                high = min(1.0 - h[i], h[j])
                delta = low                     # low ≤ 0, transfers from i to j
        else:                                 # pure random pair move
            i, j = random.sample(range(n), 2)
            low = max(-h[i], h[j] - 1.0)
            high = min(1.0 - h[i], h[j])
            if low >= high:
                it += 1
                continue
            # temperature‑scaled step size
            step = max_step * (T / T0 + 0.1)
            lo = max(low, -step)
            hi = min(high, step)
            if lo >= hi:
                delta = random.uniform(low, high)
            else:
                delta = random.uniform(lo, hi)

        # ---------- apply move ----------
        h[i] += delta
        h[j] -= delta

        new_val = _c5_metric(h)
        dC = new_val - cur_val
        accept = False
        if dC < 0:
            accept = True
        else:
            if T > 0 and random.random() < math.exp(-dC / T):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        it += 1
    return best_h, best_val

def _gradient_refine(h, max_iters=5000):
    """Deterministic refinement focusing on the worst shift."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_metric(best_h)

    for _ in range(max_iters):
        s, g, _ = _gradient_worst(best_h)
        pos = np.where((g > 1e-9) & (best_h > 1e-12))[0]
        neg = np.where((g < -1e-9) & (best_h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = int(pos[np.argmax(g[pos])])      # biggest positive gradient
        j = int(neg[np.argmin(g[neg])])      # most negative gradient
        # modest transfer to keep stability
        delta = min(0.02, best_h[i], 1.0 - best_h[j])
        if delta <= 0:
            break
        best_h[i] -= delta
        best_h[j] += delta
        new_val = _c5_metric(best_h)
        if new_val + 1e-9 < best_val:
            best_val = new_val
        else:
            best_h[i] += delta
            best_h[j] -= delta
            break
    return best_h, best_val

def _final_hill_climb(h, cur_val, max_iters=4000):
    """Tiny random adjustments for last‑minute polishing."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = cur_val
    for _ in range(max_iters):
        i, j = random.sample(range(n), 2)
        low = max(-best_h[i], best_h[j] - 1.0)
        high = min(1.0 - best_h[i], best_h[j])
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.02
        best_h[i] += delta
        best_h[j] -= delta
        new_val = _c5_metric(best_h)
        if new_val + 1e-9 < best_val:
            best_val = new_val
        else:
            best_h[i] -= delta
            best_h[j] += delta
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low C₅ bound."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # ---- configuration ----
    n = 300                     # number of sample points (even)
    # ---- initial vector ----
    h0 = _prepare_initial(n)

    # ---- primary simulated annealing ----
    remaining = total_budget - (time.time() - start_all)
    primary_time = remaining * 0.55
    best_h, best_val = _simulated_annealing(h0, primary_time)

    # ---- randomized binary restarts ----
    while True:
        elapsed = time.time() - start_all
        remaining = total_budget - elapsed
        if remaining < 10.0:
            break
        # binary start (exactly n/2 ones)
        target = n / 2.0
        ones = int(target)
        h_rand = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        h_rand[perm[:ones]] = 1.0
        # allocate a slice of the remaining budget
        slice_time = min(remaining * 0.3, 30.0)
        cand_h, cand_val = _simulated_annealing(h_rand, slice_time)
        if cand_val < best_val:

    # ---- gradient‑based refinement ----
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 2.0:

    # ---- final hill climbing if time permits ----
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 1.0:

    # ---- optional binary projection (may improve the bound) ----

                best_h[idx] += diff
                break

    try:

        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
