def c5_metric_bin(h):
    n = len(h)
    s = n/2
    # compute autocorrelation (linear) of h with reverse h
    # Use convolution style
    rev = h[::-1]
    # convolution length 2n-1
    N = 1
    while N < 2*n:
        N <<= 1
    fft_h = np.fft.rfft(h, N)
    fft_rev = np.fft.rfft(rev, N)
    conv = np.fft.irfft(fft_h * fft_rev, N)[:2*n-1]
    # R(k) for shift shift = -(n-1)...(n-1) (the correlation offset)
    # For each shift index j = 0..2n-2 corresponds to shift = j-(n-1)
    # Corresponds to sum_i h[i] * h[i+shift] where out-of-range indices have 0.
    # Then C(k) = s - R(k)
    # Max C(k) = s - min R(k)
    min_R = conv.min()
    maxC = s - min_R
    return maxC * (2.0 / n)  # multiply by dx
```

def subgradient_opt(start_h, time_limit):
    cur = start_h.copy()
    best = cur.copy()
    best_c = eval(cur)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # compute conv and min_R
        conv = fftconvolve(cur, cur[::-1], mode='full')
        min_R = conv.min()
        # compute bound
        c = (len(cur)/2 - min_R) * 2.0 / len(cur)
        if c < best_c:
            best_c = c
            best = cur.copy()
        # shift d
        idx = conv.argmin()
        d = idx - (len(cur) - 1)
        # gradient
        g = np.zeros_like(cur)
        if d > 0:
            g[d:] = cur[:-d]
            g[:-d] = cur[d:]
        elif d < 0:
            dd = -d
            g[:-dd] = cur[dd:]
            g[dd:] = cur[:-dd]
        else:
            g = 2 * cur
        # step size schedule
        it += 1
        alpha = 0.5 / np.sqrt(it)  # or maybe 0.2?
        # update then project
        cur = cur + alpha * g
        # random noise sometimes
        if it % 500 == 0:
            noise = np.random.normal(scale=0.01, size=cur.shape)
            cur += noise
        cur = project_sum_box(cur, len(cur)/2.0)
    return best, best_c

def _project_sum_box(x, target):
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return y

def _c5_metric_fast(h):

    conv = fftconvolve(h, h[::-1], mode='full')

def _subgradient_opt(start_h, max_seconds):
    cur = start_h.copy()
    best = cur.copy()
    best_c = _c5_metric_fast(best)
    start = time.time()
    it = 0
    n = cur.shape[0]
    target = n / 2.0
    while time.time() - start < max_seconds:
        conv = fftconvolve(cur, cur[::-1], mode='full')
        min_R = conv.min()
        c = (target - min_R) * 2.0 / n
        if c < best_c:
            best_c = c
            best = cur.copy()
        idx = conv.argmin()
        d = idx - (n - 1)
        # compute gradient
        g = np.zeros_like(cur)
        if d > 0:
            g[d:] = cur[:-d]
            g[:-d] = cur[d:]
        elif d < 0:
            dd = -d
            g[:-dd] = cur[dd:]
            g[dd:] = cur[:-dd]
        else:
            g = 2 * cur
        it += 1
        alpha = 0.5 / np.sqrt(it)
        cur = cur + alpha * g
        # occasional stochastic jitter
        if it % 500 == 0:
            cur += np.random.normal(scale=0.01, size=n)
        cur = _project_sum_box(cur, target)
    return best, best_c

def _anneal(start_h, max_seconds):

    best = cur.copy()
    best_c = _c5_metric_fast(best)
    cur_c = best_c
    n = cur.shape[0]
    target = n/2.0
    start = time.time()
    T0 = 0.01
    T_end = 1e-6
    max_step = 0.2
    while time.time() - start < max_seconds:
        progress = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        i, j = random.sample(range(n), 2)
        hi, hj = cur[i], cur[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.5
        cur[i] = hi + delta
        cur[j] = hj - delta
        new_c = _c5_metric_fast(cur)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            cur[i] = hi
            cur[j] = hj
    return best, best_c

import time
import random
import math
import numpy as np
from scipy.signal import fftconvolve

def _project_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto the box [0,1] and the affine constraint sum(x)=target."""
    lo, hi = -10.0, 10.0
    # binary search on the Lagrange multiplier λ
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return y

def _c5_metric_fast(h: np.ndarray) -> float:
    """Fast evaluation of C5 using the min autocorrelation."""
    n = h.shape[0]
    conv = fftconvolve(h, h[::-1], mode='full')
    min_R = conv.min()
    target = n * 0.5
    return (target - min_R) * 2.0 / n

def _subgradient_opt(start_h: np.ndarray, max_seconds: float):
    """Subgradient ascent on the minimal autocorrelation."""
    cur = start_h.copy()
    best = cur.copy()
    best_c = _c5_metric_fast(best)
    n = cur.shape[0]
    target = n * 0.5
    it = 0
    t0 = time.time()
    while time.time() - t0 < max_seconds:
        conv = fftconvolve(cur, cur[::-1], mode='full')
        min_R = conv.min()
        c = (target - min_R) * 2.0 / n
        if c < best_c:
            best_c, best = c, cur.copy()
        idx = conv.argmin()
        d = idx - (n - 1)                     # shift corresponding to the worst autocorrelation
        g = np.zeros_like(cur)
        if d > 0:
            g[d:] = cur[:-d]
            g[:-d] = cur[d:]
        elif d < 0:
            dd = -d
            g[:-dd] = cur[dd:]
            g[dd:] = cur[:-dd]
        else:                                 # d == 0 (theoretically never the worst)
            g = 2.0 * cur
        it += 1
        alpha = 0.5 / math.sqrt(it)           # diminishing step size
        cur = cur + alpha * g
        if it % 500 == 0:                     # occasional stochastic jitter
            cur += np.random.normal(scale=0.01, size=n)
        cur = _project_sum_box(cur, target)
    return best, best_c

    T0, T_end = 0.02, 1e-6
    while time.time() - t0 < max_seconds:
        prog = (time.time() - t0) / max_seconds
        T = T0 * (T_end / T0) ** prog
        i, j = random.sample(range(n), 2)
        hi, hj = cur[i], cur[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.5
        cur[i] = hi + delta
        cur[j] = hj - delta
        new_c = _c5_metric_fast(cur)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best = new_c, cur.copy()
        else:
            cur[i], cur[j] = hi, hj
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # choose a modest resolution; larger n gives a smoother search
    n = 500
    if n % 2:
        n += 1
    target_sum = n * 0.5

    # initialise – interpolate the supplied construction if present
    if 'initial_h_values' in globals():
        hv = np.asarray(initial_h_values, dtype=np.float64)
        x_src = np.linspace(0.0, 2.0, hv.size, endpoint=False)
        x_dst = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_dst, x_src, hv)
        h0 = _project_sum_box(h0, target_sum)
    else:
        h0 = np.full(n, 0.5, dtype=np.float64)
        h0 = _project_sum_box(h0, target_sum)

    best_h, best_c = h0.copy(), _c5_metric_fast(h0)

    start_all = time.time()
    total_budget = float(budget_s)

    # ----- main exploration phase (subgradient + random restarts) -----
    while time.time() - start_all < total_budget * 0.80:
        remaining = total_budget - (time.time() - start_all)
        if remaining < 5.0:
            break
        # allocate up to 45 s for a subgradient run
        slice_time = min(remaining * 0.4, 45.0)
        # random feasible start
        start_vec = np.random.rand(n)
        start_vec = _project_sum_box(start_vec, target_sum)
        cand_h, cand_c = _subgradient_opt(start_vec, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()

    # ----- polishing phase (annealing) -----
    remaining = total_budget - (time.time() - start_all)
    if remaining > 5.0:
        cand_h, cand_c = _anneal(best_h, remaining)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()

    # enforce feasibility exactly
    best_h = _project_sum_box(best_h, target_sum)

    # final bound using the harness helper if available
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric_fast(best_h)

    return best_h, float(final_c), int(n)
