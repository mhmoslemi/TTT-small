def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the feasible set: 0<=h<=1 and sum(h)=n/2
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12,
                         max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# --------------------------------------------------------------
# Correlation helpers
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _c5_and_lags(h: np.ndarray, atol: float = 1e-12):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = corr.max()
    # lags where the maximum is attained (within tolerance)
    idx = np.where(np.abs(corr - mx) <= atol)[0]
    lags = (idx - (n - 1)).tolist()
    return float(mx), lags


# --------------------------------------------------------------
# Gradient of the selected lag(s)
def _grad_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        i = np.arange(n - lag)
        g[i] += 1.0 - h[i + lag]
        g[i + lag] -= h[i]
    else:
        k = -lag
        i = np.arange(k, n)               # indices where i+lag is valid
        g[i] += 1.0 - h[i - k]
        j = np.arange(0, n - k)           # indices where i-k is valid
        g[j] -= h[j + k]
    return g


# --------------------------------------------------------------
# Simple projected gradient descent on the max‑correlation objective
def _grad_descent(h0: np.ndarray, max_iter: int,
                  time_left_func) -> tuple:
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(h)
    step = 0.05

    for it in range(max_iter):
        if time_left_func() <= 0:
            break

        val, lags = _c5_and_lags(h)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h.copy()

        # combined gradient of all maximal lags
        g = np.zeros_like(h)
        for lag in lags:
            g += _grad_for_lag(h, lag)
        g /= max(1, len(lags))

        # line search (up to 8 reductions)
        improved = False
        cur_step = step
        for _ in range(8):
            h_new = h - cur_step * g
            h_new = _project_to_feasible(h_new, n)
            new_val = _c5_from_h(h_new)
            if new_val < val:
                h = h_new
                step = cur_step * 1.02
                improved = True
                break
            cur_step *= 0.5

        if not improved:
            step *= 0.5
            if step < 1e-6:
                break
            # occasional random shake to escape flat regions
            if it % 120 == 0:
                h = h + np.random.randn(n) * 0.01
                h = _project_to_feasible(h, n)

    return best_h, best_val


# --------------------------------------------------------------
def _upscale_h(coarse: np.ndarray, n_new: int) -> np.ndarray:
    n_old = coarse.size
    if n_old == n_new:
        return coarse.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, coarse)
