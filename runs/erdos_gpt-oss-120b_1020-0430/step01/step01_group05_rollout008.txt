def compute_corr_and_max(h):
    n = h.shape[0]
    L = n + n - 1
    # reverse of (1-h)
    g = (1.0 - h)[::-1]
    H = np.fft.fft(h, L)
    G = np.fft.fft(g, L)
    corr = np.fft.ifft(H * G).real
    # round small negatives to zero? Not needed.
    idx_max = int(np.argmax(corr))
    max_val = float(corr[idx_max])
    return max_val, idx_max, corr

from scipy.optimize import minimize

# Global for equality constraint
TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while enforcing sum(x)=target via bisection."""
    low, high = -1.0, 1.0  # bounds for lambda; start wide enough
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL

def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Log‑sum‑exp smooth max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth upper bound on max overlap integral."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx

def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_cosine(n: int) -> np.ndarray:
    """Cosine arch: h(x)=0.5+0.5*cos(pi*(x-1)), average 0.5."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 0.5 + 0.5 * np.cos(np.pi * (pts - 1.0))
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, n / 2.0)

def compute_corr_and_max(h: np.ndarray):
    """Full cross‑correlation of h with (1‑h) using FFT; return max value and index."""
    n = h.shape[0]
    L = n + n - 1
    rev_g = (1.0 - h)[::-1]
    H = np.fft.fft(h, L)
    G = np.fft.fft(rev_g, L)
    corr = np.fft.ifft(H * G).real
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx, corr

def compute_c5(h: np.ndarray) -> float:
    max_val, _, _ = compute_corr_and_max(h)
    return max_val * (2.0 / h.shape[0])

def gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the correlation term that currently attains the maximum."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        if L < n:
            g[:n - L] += 1.0 - h[L:]
            g[L:] -= h[:n - L]
    else:
        L = -lag
        if L < n:
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
    return g

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Find a step function h on [0,2] with small max overlap (C5)."""
    global TARGET_SUM_GLOBAL
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    safe_margin = 5.0  # seconds reserved for finalisation

    # problem size – larger n gives finer control, still cheap
    n = 500
    dx = 2.0 / n
    target_sum = n / 2.0
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # Build initial seeds
    seed_vectors = []
    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))
    seed_vectors.append(make_cosine(n))

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Local SLSQP optimisation from each seed
    alpha = 200.0            # smoother approximation of max
    maxiter = 500
    for h0 in seed_vectors:
        if time.time() - start > budget_s - safe_margin:
            break
        try:
            res = minimize(
                objective_smooth,
                h0,
                args=(dx, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue
        h_opt = project_onto_bounds_and_sum(res.x, target_sum)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        if best_val <= 0.3808:
            break

    # fallback if nothing succeeded
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Gradient‑descent / simulated‑annealing refinement
    h_cur = best_h.copy()
    cur_val = best_val
    max_iters = 300000
    init_temp = 0.02
    min_temp = 1e-5
    it = 0
    stagnation = 0

    while time.time() - start < budget_s - safe_margin:
        # compute worst lag
        max_corr, idx_max, _ = compute_corr_and_max(h_cur)
        lag = idx_max - (n - 1)

        # gradient of that lag
        g = gradient_for_lag(h_cur, lag)

        # step size decays linearly
        step = 0.2 * (1.0 - it / max_iters)
        if step < 0.001:
            step = 0.001

        # propose move opposite to gradient
        h_prop = h_cur - step * g
        h_prop = np.clip(h_prop, 0.0, 1.0)
        h_prop = project_onto_bounds_and_sum(h_prop, target_sum)

        new_val = compute_c5(h_prop)

        # temperature schedule
        temp = init_temp * (1.0 - it / max_iters) + min_temp

        # accept if better or with Metropolis probability
        if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / temp):
            h_cur = h_prop
            cur_val = new_val
            stagnation = 0
            if new_val < best_val:
                best_val = new_val
                best_h = h_prop.copy()
                if best_val <= 0.3808:
                    break
        else:
            stagnation += 1

        # occasional random jitter to escape long plateaus
        if stagnation > 4000:
            jitter = np.random.uniform(-0.05, 0.05, size=n)
            h_jit = h_cur + jitter
            h_jit = np.clip(h_jit, 0.0, 1.0)
            h_jit = project_onto_bounds_and_sum(h_jit, target_sum)
            h_cur = h_jit
            cur_val = compute_c5(h_cur)
            stagnation = 0

        it += 1
        if it >= max_iters:
            break

    # ------------------------------------------------------------------
    # Final simple pairwise refinement (pure improvement moves)
    while time.time() - start < budget_s - safe_margin:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cval = compute_c5(cand)
        if cval < best_val:
            best_val = cval
            best_h = cand.copy()
            if best_val <= 0.3808:
                break

    # final safety check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
