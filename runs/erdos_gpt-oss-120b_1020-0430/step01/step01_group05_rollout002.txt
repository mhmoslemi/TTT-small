import time, random, math
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect sum(x)=target via bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 = max_k ∫ h(x)(1‑h(x+k))dx, discretised."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    m = arr.max()
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx

    dx = 2.0 / n
    target_bound = 0.3808        # early‑exit target

    # projection shortcut
    proj = lambda v: project_onto_bounds_and_sum(v, target_sum)

    # ----- initial population -----
    popsize = 30
    pop = []
    # parent if provided
    if "initial_h_values" in globals():
        try:
            pop.append(upsample_initial(np.asarray(initial_h_values, dtype=float), n))
        except Exception:
            pass
    pop.append(make_triangle(n))
    pop.append(make_sin2(n))
    while len(pop) < popsize:
        rnd = rng.random(n)
        pop.append(proj(rnd))

    pop_vals = [compute_c5(v) for v in pop]
    best_idx = int(np.argmin(pop_vals))
    best_h = pop[best_idx].copy()
    best_val = pop_vals[best_idx]

    # ----- differential evolution -----
    F = 0.8
    Cr = 0.9
    max_gen = 2000

    gen = 0
    while time.time() - start < budget_s - 0.5 and gen < max_gen:
        for i in range(popsize):
            # pick three distinct other indices
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = proj(mutant)

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(n)
            cross_mask = rng.random(n) < Cr
            cross_mask[jrand] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = np.clip(trial, 0.0, 1.0)
            trial = proj(trial)

            val = compute_c5(trial)
            if val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val
                if val < best_val:
                    best_h = trial.copy()
                    best_val = val
                    if best_val <= target_bound:
                        break
        if best_val <= target_bound:
            break
        gen += 1

    # ----- local random pairwise refinement -----
    while time.time() - start < budget_s - 0.5:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        # projection should be unnecessary (sum preserved), only clip needed
        cand = np.clip(cand, 0.0, 1.0)
        val = compute_c5(cand)
        if val < best_val:
            best_h = cand
            best_val = val
            if best_val <= target_bound:
                break

    # ----- final smooth‑max SLSQP refinement (if time) -----
    if time.time() - start < budget_s - 0.5:
        try:
            cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
            bounds = [(0.0, 1.0)] * n
            res = minimize(
                lambda x: objective_smooth(x, dx, alpha=30.0),
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            h_opt = proj(res.x)
            val_opt = compute_c5(h_opt)
            if val_opt < best_val:
                best_h, best_val = h_opt, val_opt
        except Exception:
            pass

    # final safety projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj(best_h)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
