def _adam_optimize(h0, target, lr=0.5, steps=1000, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for t in range(1, steps+1):

# ----- projection onto the capped simplex (0<=x<=1, sum = target) -----
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----- correlation and gradient utilities -----
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the worst‑shift correlation."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----- Adam style projected optimisation -----
def _adam_optimize(h0: np.ndarray, target: float, lr: float = 0.5,
                  steps: int = 1500, beta1: float = 0.9,
                  beta2: float = 0.999, eps: float = 1e-8) -> np.ndarray:
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for t in range(1, steps + 1):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (h.size - 1)
        g = _gradient_for_shift(h, shift)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = _project_onto_capped_simplex(h, target)
    return h


# ----- smooth max for SLSQP -----
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 40.0, max_iter: int = 1000) -> np.ndarray:
    n = h0.size

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10,
                            'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ----- main driver -------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    TIME_FRACTION = 0.92      # safety margin

    # --- base solution (if supplied) ---------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    # --- discretisation -----------------------------------------------
    n0 = len(base_h) if base_h is not None else 0
    n = max(400, n0)          # richer grid than the parent
    target_sum = n / 2.0
    dx = 2.0 / n

    # --- helper to resample vectors ------------------------------------
    def _resample(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # --- initialise best candidate --------------------------------------
    if base_h is not None:
        best_h = _resample(base_h)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = None
        best_val = float('inf')

    # --- a good analytic start (cosine) --------------------------------
    a_opt = np.sqrt(0.125 / (1.0 + 1.0 / (2.0 * np.pi)))   # amplitude that equalises f(0) & f(0.5)
    x_grid = np.linspace(0.0, 2.0, n, endpoint=False)
    cos_start = 0.5 + a_opt * np.cos(np.pi * x_grid)
    cos_start = np.clip(cos_start, 0.0, 1.0)
    cos_start = _project_onto_capped_simplex(cos_start, target_sum)
    val_cos = _compute_c5(cos_start, dx)
    if val_cos < best_val:
        best_val = val_cos
        best_h = cos_start.copy()

    # --- start pool ----------------------------------------------------
    start_pool = [cos_start]
    if best_h is not None:
        start_pool.append(best_h)

    # --- main optimisation loop -----------------------------------------
    it = 0
    TARGET_BOUND = 0.3808
    while time.time() - start_time < budget_s * TIME_FRACTION:
        # pick a seed vector
        if it < len(start_pool):
            h0 = start_pool[it]
        else:
            # perturb the current best occasionally
            if best_h is not None and rng.rand() < 0.3:
                h0 = best_h + rng.normal(scale=0.03, size=n)
            else:
                h0 = _random_feasible(rng, n, target_sum)

        # projected Adam optimisation
        remaining = budget_s - (time.time() - start_time)
        # adapt number of steps to remaining time (very coarse)
        steps = 1200 if remaining > 30 else 400
        h_opt = _adam_optimize(h0, target_sum, lr=0.6, steps=steps)

        # occasional SLSQP polishing
        if it % 5 == 0:
            h_opt = _slsqp_optimize(h_opt, target_sum,
                                    alpha=50.0, max_iter=800)

        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            # early exit if we already beat the target
            if best_val <= TARGET_BOUND:
                break

        it += 1

    # --- final safety projection ----------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
    else:
        best_h = _project_onto_capped_simplex(best_h, target_sum)

    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
