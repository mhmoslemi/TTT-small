import time, random, math
import numpy as np

# -------------------------------------------------------------
# basic utilities
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Projection of x onto [0,1]^n intersect {sum = target} using bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact max‑overlap objective (discretised)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


# -------------------------------------------------------------
# deterministic starters
def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)           # peak 1 at centre, linear down to 0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(arr: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    if arr.shape[0] == n:
        return np.clip(arr, 0.0, 1.0).astype(float)
    x_old = np.linspace(0.0, 2.0, arr.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, arr)
    return project_onto_bounds_and_sum(h, n / 2.0)

# targeted local improvement
def improve_worst_shift(h: np.ndarray):
    """Try a small transfer of mass that reduces the current worst‑shift value.
    Returns (new_h, new_val) or (None, None) if no admissible move."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    shift = idx_max - (n - 1)               # can be negative

    # neighbor‑sum (score) for each position w.r.t. this shift
    score = np.zeros(n)
    if shift >= 0:
        score[: n - shift] += h[shift:]          # i+shift term
        score[shift:] += h[: n - shift]          # i-shift term
    else:
        s = -shift
        score[s:] += h[: n - s]                  # i+shift (actually i-s)
        score[: n - s] += h[s:]                  # i-shift (i+s)

    inc = np.where(h < 1.0)[0]
    dec = np.where(h > 0.0)[0]
    if inc.size == 0 or dec.size == 0:
        return None, None

    i = inc[np.argmax(score[inc])]
    j = dec[np.argmin(score[dec])]

    delta_max = min(1.0 - h[i], h[j])
    if delta_max <= 0.0:
        return None, None

    # use a modest fraction of the allowable transfer
    delta = delta_max * (0.05 + 0.45 * np.random.rand())
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    val = compute_c5(h_new)
    return h_new, val
