def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Parameters
    # --------------------------------------------------------------
    n = 140                     # number of samples (even)
    target_sum = n / 2.0        # integral = 1  => sum = n/2
    dx = 2.0 / n                # step width for correlation scaling

    # --------------------------------------------------------------
    # Helper functions
    # --------------------------------------------------------------
    def c5_of(v):
        """max overlap integral for discrete h"""
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(corr.max() * dx)

    def gradient_shift(h, s):
        """gradient of O_s = sum_i h_i (1-h_{i+s}) wrt h for given shift s"""
        nloc = h.shape[0]
        grad = np.zeros(nloc, dtype=float)
        if s >= 0:
            # i from 0 .. n-1-s : + (1 - h_{i+s})
            i1 = np.arange(0, nloc - s)
            grad[i1] += 1.0 - h[i1 + s]
            # i from s .. n-1 : - h_{i-s}
            i2 = np.arange(s, nloc)
            grad[i2] -= h[i2 - s]
        else:
            s = -s
            # i from s .. n-1 : + (1 - h_{i-s})
            i1 = np.arange(s, nloc)
            grad[i1] += 1.0 - h[i1 - s]
            # i from 0 .. n-1-s : - h_{i+s}
            i2 = np.arange(0, nloc - s)
            grad[i2] -= h[i2 + s]
        return grad

    def max_step(h, d):
        """largest alpha so that h+alpha*d stays in [0,1]"""
        pos = d > 1e-12
        neg = d < -1e-12
