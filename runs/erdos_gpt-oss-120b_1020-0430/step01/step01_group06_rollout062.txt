def solve_mip(n, w, time_limit):
    import cvxpy as cp, numpy as np, time
    h = cp.Variable(n, boolean=True)
    total_y = n * (n-1) // 2
    y = cp.Variable(total_y, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == w, t >= 0]
    y_idx = 0
    for s in range(1, n):
        length = n - s
        y_slice = y[y_idx : y_idx + length]
        y_idx += length
        # linearization
        constraints.append(y_slice <= h[:length])
        constraints.append(y_slice <= h[s:])
        constraints.append(y_slice >= h[:length] + h[s:] - 1)
        prefix = cp.sum(h[:length])
        suffix = cp.sum(h[s:])
        sumy = cp.sum(y_slice)
        constraints.append(prefix - sumy <= t)
        constraints.append(suffix - sumy <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    return h, t, prob
```

def local_search(n, w, max_time):
    import numpy as np, random, time, math
    # start with random binary vector of weight w
    h = np.zeros(n)
    ones_idx = random.sample(range(n), w)
    h[ones_idx] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    # Simulated annealing
    T0 = 0.5
    T = T0
    decay = 0.9999
    it = 0
    start = time.time()
    while time.time() - start < max_time:
        # pick two positions i!=j
        i, j = random.sample(range(n), 2)
        # delta to move weight preserving sum: we can exchange values.
        # Actually binary swap: if h[i]==h[j], skip.
        if cur_h[i] == cur_h[j]:
            continue
        # swap them
        nh = cur_h.copy()
        nh[i], nh[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(nh)
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = nh.copy()
        it += 1
        if it % 1000 == 0:
            T *= decay
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    import time, random, math
    import numpy as np

    # ------------------------------------------------------------------
    # Helper to evaluate C5 for a binary (or fractional) step function
    # ------------------------------------------------------------------
    def compute_c5(vec):
        n = len(vec)
        # correlation with 1-vec, full mode – exactly the definition used by the harness
        corr = np.correlate(vec, 1 - vec, mode="full")
        return float(corr.max() * (2.0 / n))

    # ------------------------------------------------------------------
    # Local search / simulated annealing (binary swap moves)
    # ------------------------------------------------------------------
    def local_search(n, w, time_limit, start_vec=None):
        start = time.time()
        rng = random.Random(seed)
        if start_vec is not None:
            cur = np.array(start_vec, copy=True)
        else:
            cur = np.zeros(n)
            cur[rng.sample(range(n), w)] = 1.0

        best = cur.copy()
        best_c5 = compute_c5(best)
        cur_c5 = best_c5

        T = 0.5
        decay = 0.9995
        it = 0
        while time.time() - start < time_limit:
            i, j = rng.sample(range(n), 2)
            if cur[i] == cur[j]:
                continue          # swapping equal values does nothing
            new = cur.copy()
            new[i], new[j] = cur[j], cur[i]
            new_c5 = compute_c5(new)
            if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                cur = new
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best = new.copy()
                    best_c5 = new_c5
            it += 1
            if it % 1000 == 0:
                T *= decay
        return best, best_c5

    # ------------------------------------------------------------------
    # Determine problem size (use parent vector if supplied)
    # ------------------------------------------------------------------
    start_time = time.time()
    try:
        init = np.array(initial_h_values, dtype=float)
        n = int(init.shape[0])
    except NameError:
        n = 70                     # default size
    if n % 2:
        n += 1                    # enforce even length for integral weight
    w = n // 2                    # required number of ones (measure 1 on [0,2])

    # ------------------------------------------------------------------
    # Try a small Mixed‑Integer Linear Program to minimise the maximum overlap
    # ------------------------------------------------------------------
    h_opt = None
    best_c5 = float("inf")
    time_remaining = lambda: budget_s - (time.time() - start_time)

    # allocate most of the remaining time to the MIP, keep a little for fallback
    mip_time = max(5, int(time_remaining() * 0.85))
    try:
        import cvxpy as cp

        # binary indicator vector
        h = cp.Variable(n, boolean=True)
        # binary variables for each pair (i,i+s)
        total_y = n * (n - 1) // 2
        y = cp.Variable(total_y, boolean=True)
        # bound on the maximal overlap
        t = cp.Variable(integer=True)

        constraints = [cp.sum(h) == w,
                       t >= 0,
                       t <= w]               # trivial upper bound

        y_start = 0
        for s in range(1, n):
            length = n - s
            y_slice = y[y_start:y_start + length]
            y_start += length

            # y <= h_i and y <= h_{i+s}
            constraints.append(y_slice <= h[:length])
            constraints.append(y_slice <= h[s:])

            # prefix (left) and suffix (right) contributions
            prefix = cp.sum(h[:length])
            suffix = cp.sum(h[s:])
            sum_y = cp.sum(y_slice)

            # both directions must respect the bound t
            constraints.append(prefix - sum_y <= t)
            constraints.append(suffix - sum_y <= t)

        prob = cp.Problem(cp.Minimize(t), constraints)

        # solve with CBC – give it the reserved time budget
        prob.solve(solver=cp.CBC, maxSeconds=mip_time, verbose=False)

        if h.value is not None:
            # round to exact 0/1 and repair any tiny drift in weight
            h_bin = np.rint(h.value).astype(int)
            diff = w - h_bin.sum()
            if diff > 0:
                zero_idx = np.where(h_bin == 0)[0]
                rng = random.Random(seed)
                chosen = rng.sample(list(zero_idx), diff)
                h_bin[chosen] = 1
            elif diff < 0:
                one_idx = np.where(h_bin == 1)[0]
                rng = random.Random(seed)
                chosen = rng.sample(list(one_idx), -diff)
                h_bin[chosen] = 0
            h_opt = h_bin.astype(float)
            best_c5 = compute_c5(h_opt)
    except Exception:
        # any problem (missing CBC, timeout, etc.) – just continue to fallback
        pass

    # ------------------------------------------------------------------
    # If the MIP gave a solution, or not, run a short local search
    # ------------------------------------------------------------------
    remaining = time_remaining()
    if remaining > 2.0:   # keep a small safety margin
        # give the search at most the remaining time
        search_time = remaining
        start_vec = h_opt if h_opt is not None else None
        h_sa, c5_sa = local_search(n, w, search_time, start_vec=start_vec)
        if c5_sa < best_c5:
            h_opt = h_sa
            best_c5 = c5_sa

    # ------------------------------------------------------------------
    # Final safety: ensure weight and bounds are exact
    # ------------------------------------------------------------------
    if h_opt is None:                     # very unlikely, but guard
        # fall back to a simple deterministic construction
        h_opt = np.zeros(n)
        h_opt[:w] = 1.0

    # repair possible rounding issues
    total = h_opt.sum()
    if abs(total - w) > 1e-6:
        # adjust greedily
        diff = int(round(w - total))
        if diff > 0:
            zero_idx = np.where(h_opt < 0.5)[0]
            rng = random.Random(seed)
            chosen = rng.sample(list(zero_idx), diff)
            h_opt[chosen] = 1.0
        elif diff < 0:
            one_idx = np.where(h_opt > 0.5)[0]
            rng = random.Random(seed)
            chosen = rng.sample(list(one_idx), -diff)
            h_opt[chosen] = 0.0

    final_c5 = compute_c5(h_opt)

    return (h_opt.tolist(), final_c5, n)
