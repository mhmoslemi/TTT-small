import numpy as np
import random
import time
import math

def compute_c5(v):
    return float(np.max(np.correlate(v, 1.0 - v, mode='full')) * (2.0 / len(v)))

def project_h(v, target_sum):
    # ensures bounds and sum
    np.clip(v, 0.0, 1.0, out=v)
    cur = v.sum()
    diff = cur - target_sum
    if abs(diff) < 1e-12:
        return v
    # simple iterative adjustment
    rng = random.Random()  # not using seed, okay
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx.tolist())
            delta = min(v[i], diff)
            v[i] -= delta
            diff -= delta
        else:
            idx = np.where(v < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx.tolist())
            delta = min(1.0 - v[i], -diff)
            v[i] += delta
            diff += delta
    np.clip(v, 0.0, 1.0, out=v)
    return v

import numpy as np
import random
import time
import math

def _c5(v: np.ndarray) -> float:
    """Maximum overlap for a step function."""
    return float(np.max(np.correlate(v, 1.0 - v, mode="full")) * (2.0 / len(v)))


def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to the required target."""
    np.clip(v, 0.0, 1.0, out=v)
    cur = v.sum()
    diff = cur - target
    if abs(diff) < 1e-12:
        return v
    rng = random.Random()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx.tolist())
            d = min(v[i], diff)
            v[i] -= d
            diff -= d
        else:
            idx = np.where(v < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx.tolist())
            d = min(1.0 - v[i], -diff)
            v[i] += d
            diff += d
    np.clip(v, 0.0, 1.0, out=v)
    return v
