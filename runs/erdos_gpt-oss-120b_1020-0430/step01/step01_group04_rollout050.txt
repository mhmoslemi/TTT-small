import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalized."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible start (values in [0,1] and sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length (preserving integral)."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

# Simple stochastic improvement (hill‑climbing with occasional
# probabilistic uphill moves).
# ------------------------------------------------------------
def local_search(h0: np.ndarray, deadline: float, rng: np.random.Generator):
    """Improve h0 by random mass transfers while respecting constraints."""
    cur = h0.copy()
    n = cur.shape[0]
    target = n / 2.0
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    step = 0.25          # maximal transferred mass per move
    temp = 0.02          # temperature for occasional uphill moves
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide direction of transfer
        if cur[i] > 1e-12 and cur[j] < 1.0 - 1e-12:
            max_delta = min(step, cur[i], 1.0 - cur[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        elif cur[j] > 1e-12 and cur[i] < 1.0 - 1e-12:
            max_delta = min(step, cur[j], 1.0 - cur[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = cur.copy()
            cand[j] -= delta
            cand[i] += delta
        else:
            continue

        # enforce bounds (numerical safety)
        np.clip(cand, 0.0, 1.0, out=cand)

        # enforce exact sum (tiny drift may appear)
        diff = target - cand.sum()
        if abs(diff) > 1e-12:
            cand += diff / n
            np.clip(cand, 0.0, 1.0, out=cand)

        new_c5 = compute_c5(cand)
        improve = new_c5 < cur_c5 - 1e-12
        if not improve:
            # metropolis‑like acceptance
            prob = np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12))
            if rng.random() < prob:
                improve = True
        if improve:
            cur = cand
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best = cand.copy()
                best_c5 = new_c5

        it += 1
        if it % 1000 == 0:
            step = max(step * 0.99, 0.02)
            temp = max(temp * 0.995, 1e-4)

    return best, best_c5


def polish_h(h0: np.ndarray, deadline: float):
    """A short SLSQP refinement using the smooth max approximation."""
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x, t=target: np.sum(x) - t}

    def slsqp_pass(x0, alpha):
        return minimize(lambda x: softmax_c5(x, alpha=alpha),
                        x0,
                        method="SLSQP",
                        bounds=bounds,
                        constraints=cons,
                        options={"maxiter": 200, "ftol": 1e-9, "disp": False})

    # first a loose pass, then a sharper one
    try:
        res1 = slsqp_pass(h0, alpha=5.0)
        if not res1.success:
            return h0
        res2 = slsqp_pass(res1.x, alpha=30.0)
        if res2.success:
            h_opt = np.clip(res2.x, 0.0, 1.0)
            # final tiny correction of the integral
            diff = target - h_opt.sum()
            if abs(diff) > 1e-12:
                h_opt += diff / n
                np.clip(h_opt, 0.0, 1.0, out=h_opt)
            return h_opt
    except Exception:
        pass
    return h0

    """

    """

    start_time = time.time()
    # keep a safety margin

    best_c5 = np.inf

    # ------------------------------------------------------------------
    n = 200
    target_sum = n / 2.0

    starts = []

            starts.append(parent.copy())
    except NameError:

    # 2. Classic triangular shape
    starts.append(generate_triangular(n))

    # 3. Several random feasible vectors
    for _ in range(6):
        starts.append(random_start(n, rng))

    # 4. Small perturbations of the triangular shape
    tri = generate_triangular(n)
    for _ in range(4):
        pert = tri + rng.uniform(-0.15, 0.15, size=n)
        np.clip(pert, 0.0, 1.0, out=pert)
        diff = target_sum - pert.sum()
        if abs(diff) > 1e-12:
            pert += diff / n
            np.clip(pert, 0.0, 1.0, out=pert)
        starts.append(pert)

    # ------------------------------------------------------------------
    # Optimise each start.
    # ------------------------------------------------------------------
    for s in starts:
        if time.time() > time_limit:
            break

        # Allocate a slice of the remaining budget for this start.
        remaining = time_limit - time.time()
        # give at most 25 s to the stochastic hill‑climb; the rest (if any) will be used for polishing.
        ls_deadline = time.time() + min(remaining * 0.75, 25.0)

        h_opt, c5_opt = local_search(s, ls_deadline, rng)

        # Optional polish with a few SLSQP iterations (quick, bounded by remaining time)
        if time.time() < time_limit:

            # we don't enforce a strict deadline inside polish_h – it will finish quickly.
            h_polished = polish_h(h_opt, polish_dead)
            c5_polished = compute_c5(h_polished)
        else:
            h_polished = h_opt
            c5_polished = c5_opt

        if c5_polished < best_c5:
            best_h = h_polished
            best_c5 = c5_polished

    # Fallback: a simple binary half‑interval (guaranteed feasible).
    # ------------------------------------------------------------------

        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])

        best_h = best_h * (n / 2.0 / best_h.sum())
        best_c5 = compute_c5(best_h)

    # Final sanity fix (integral constraint).
    # ------------------------------------------------------------------
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
