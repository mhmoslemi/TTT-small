def projected_gradient_descent(h0, max_iters=2000, step=0.1):
    h = h0.copy()
    target = h.sum()  # should be n/2
    for it in range(max_iters):
        # compute correlation and c array
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        # compute shift
        shift = max_idx - (len(h) - 1)
        # compute gradient
        g = np.zeros_like(h)
        n = len(h)
        if shift >= 0:
            i_end = n - shift
            i = slice(0, i_end)
            j = slice(shift, n)
            # i -> i, j -> i+shift
            g[i] += 1 - h[j]
            g[j] -= h[i]
        else:
            s = -shift
            i_start = s
            i = slice(i_start, n)
            j = slice(0, n - s)
            g[i] += 1 - h[j]
            g[j] -= h[i]
        # apply step (moving opposite direction of gradient to reduce max)

def descent_optimize(h0, max_iters, step0, rng):
   n = len(h0)
   target = n/2
   h = h0.copy()
   step = step0
   best_c = compute_c5(h)
   for it in range(max_iters):
       # compute correlation
       corr = np.correlate(h, 1 - h, mode='full')
       max_val = corr.max()
       # find all indices attaining max (or near)
       max_idxs = np.where(corr >= max_val - 1e-12)[0]
       # compute avg gradient over these shifts
       grad = np.zeros_like(h)
       for idx in max_idxs:
           shift = idx - (n - 1)
           # compute gradient for this shift
           # as defined earlier
           if shift >= 0:
               i_end = n - shift
               i = slice(0, i_end)
               j = slice(shift, n)
               grad[i] += 1 - h[j]
               grad[j] -= h[i]
           else:
               s = -shift
               i = slice(s, n)
               j = slice(0, n - s)
               grad[i] += 1 - h[j]
               grad[j] -= h[i]
       grad /= len(max_idxs)  # average

       # step direction opposite gradient to reduce max
       new_h = h - step * grad
       # projection
       new_h = proj_simplex_box(new_h, target)
       # evaluate
       new_c = compute_c5(new_h)

           # maybe increase step a bit

           # reduce step
           step *= 0.5
           # optionally also add small random jitter to escape? Could skip.
       # optionally break if step very small
       if step < 1e-8:
           break

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the SLSQP surrogate (soft‑max) optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate for max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Minor correction of the equality constraint
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def proj_simplex_box(v, target):
    """
    Euclidean projection of v onto {x: sum(x)=target, 0≤x≤1}.
    Uses a simple bisection on the Lagrange multiplier λ such that
    x_i = clip(v_i - λ, 0, 1).
    """
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def random_continuous_vector(n, rng):
    """Generate a random feasible vector (0≤h≤1, Σh = n/2)."""
    v = rng.random(n)
    return proj_simplex_box(v, n / 2.0)


def gradient_of_shift(h, shift):
    """Gradient of the overlap‑complement term for a single shift."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        i = slice(0, end)
        j = slice(shift, n)
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        s = -shift
        i = slice(s, n)
        j = slice(0, n - s)
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g


def descent_optimize(h0, max_iters, step0, rng):
    """
    Projected (sub)gradient descent directly on the max‑over‑k objective.
    Returns a feasible vector and its C5 value.
    """
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    step = step0
    best_c = compute_c5(h)

    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        # all shifts attaining the maximum (within a tiny tolerance)
        max_idxs = np.where(corr >= max_val - 1e-12)[0]

        grad = np.zeros_like(h)
        for idx in max_idxs:
            shift = idx - (n - 1)
            grad += gradient_of_shift(h, shift)
        grad /= len(max_idxs)

        new_h = h - step * grad
        new_h = proj_simplex_box(new_h, target)
        new_c = compute_c5(new_h)

        if new_c < best_c - 1e-12:
            h = new_h
            best_c = new_c
            step = min(step * 1.1, 1.0)   # modest step increase after success
        else:
            step *= 0.5
            if step < 1e-8:
                # inject a tiny random jitter to escape flat regions
                jitter = 0.02 * rng.random(n)
                h = proj_simplex_box(h + jitter, target)
                step = step0
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise base discretisation
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = proj_simplex_box(h0, n / 2.0)
    else:
        n = 120                # even number, modest size
        h0 = random_continuous_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Local refinement from the parent / initial vector
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.85:
        h_opt, c_opt = descent_optimize(best_h, max_iters=2000,
                                        step0=0.5, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ------------------------------------------------------------------
    # Random‑restart gradient descent phase
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.80:
        h_start = random_continuous_vector(n, rng)
        h_tmp, c_tmp = descent_optimize(h_start, max_iters=800,
                                        step0=0.5, rng=rng)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ------------------------------------------------------------------
    # Short SLSQP polishing passes (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    betas = [5.0, 20.0, 80.0, 200.0]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.90:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=300)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if (time.time() - start_time < budget_s * 0.95) and (n * 2 <= 200):
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # refine on the finer grid
        h_up, c_up = descent_optimize(h_up, max_iters=1500,
                                      step0=0.3, rng=rng)
        # final SLSQP passes on the fine grid
        for beta in betas:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta=beta, max_iter=250)
            except Exception:
                pass
        final_c_up = compute_c5(h_up)
        if final_c_up < best_c:
            best_h, best_c, n = h_up, final_c_up, n_up

    # ------------------------------------------------------------------
    # Final clean‑up to guarantee exact feasibility
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    best_h = proj_simplex_box(best_h, target)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
