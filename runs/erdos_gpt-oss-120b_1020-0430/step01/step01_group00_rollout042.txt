import time
import numpy as np
from scipy import optimize

# ---------- helper utilities ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h) = n/2 using a simple water‑filling scheme.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (should be very small)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C₅ value for a given discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- SLSQP local refinement ----------
def _solve_h(initial_h: np.ndarray, max_iter: int = 200) -> tuple:
    """Run a cheap SLSQP optimisation starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        h = x[:n]
        return np.sum(h) - (n / 2.0)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h0, [t0]])

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": _ineq}, {"type": "eq", "fun": _eq}],
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- gradient‑descent based local search ----------
def _gradient_descent(h0: np.ndarray, max_iter: int = 4000) -> tuple:
    """Simple projected gradient ascent on the minimal overlap (maximising product)."""
    n = h0.size
    h = _project_to_feasible(h0, n)
    cur_val = _c5_from_h(h)
    best_h, best_val = h.copy(), cur_val

    step = 0.12          # initial step size
    temp = 0.015         # initial temperature for occasional uphill moves

    for it in range(max_iter):
        if time.time() >= GLOBAL_END_TIME - 0.5:
            break

        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)            # may be negative

        # gradient of the inner product for the worst lag
        grad = np.empty_like(h)
        if lag >= 0:
            grad[: n - lag] = h[lag:]
            grad[lag:] = h[: n - lag]
        else:
            l = -lag
            grad[l:] = h[: n - l]
            grad[: n - l] = h[l:]

        # proposed update
        h_new = h + step * grad
        h_new = _project_to_feasible(h_new, n)
        new_val = _c5_from_h(h_new)

        if new_val < best_val:
            best_h, best_val = h_new.copy(), new_val

        # accept if improvement, otherwise probabilistic uphill move
        if new_val < cur_val:
            h, cur_val = h_new, new_val
        else:
            prob = np.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                h, cur_val = h_new, new_val

        # cooling schedules
        step *= 0.9997
        temp *= 0.9996
