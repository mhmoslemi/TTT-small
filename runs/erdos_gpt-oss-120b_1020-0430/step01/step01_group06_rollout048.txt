def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # Determine n and target sum
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
    except NameError:
        n = 70
        h0 = np.zeros(n)
    target_sum = n / 2.0

    # Helper functions
    def c5_of(v):
        corr = np.correlate(v, 1 - v, mode='full')
        return float(corr.max() * (2.0 / n))

    def compute_corr(v):
        return np.correlate(v, 1 - v, mode='full')

    def proj_sum(v):
        # Project onto [0,1] and sum target
        lo = np.min(v) - 1.0
        hi = np.max(v) + 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            x = np.clip(v - mid, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
        # Fix small rounding error
        err = x.sum() - target_sum
        if abs(err) > 1e-10:
            # distribute error among non-extreme entries
            idx = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
            if idx.size > 0:
                adjust = err / idx.size
                x[idx] -= adjust
                # clip again
                x = np.clip(x, 0.0, 1.0)
        return x

    # Prepare initial vector
    if 'h0' in locals() and h0.size > 0:
        h = np.clip(h0, 0.0, 1.0)
        # project onto sum target keeping bounds
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            h = proj_sum(h)
    else:
        # random binary with exact sum
        h = np.zeros(n)
        idx = np.random.choice(n, n//2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_c5 = c5_of(h)

    # Simulated annealing parameters
    T = max(0.001, best_c5 * 0.05)  # initial temperature
    decay = 0.9997

    it = 0
    while time.time() - start_time < budget_s * 0.97:
        # compute correlation
        corr = compute_corr(h)
        idx_max = int(np.argmax(corr))
        s = idx_max - (n - 1)  # shift for max correlation
        # targeted move
        if s >= 0:
            i_range = np.arange(n - s)
            p_range = i_range + s
        else:

            p_range = i_range + s
        contrib = h[i_range] * (1.0 - h[p_range])
        if contrib.size > 0:

            i_best = i_range[k]
            p_best = p_range[k]
            eps = min(h[i_best], 1.0 - h[p_best])
            if eps > 1e-12:

                nh = h.copy()
                nh[i_best] -= eps
                nh[p_best] += eps
                new_c5 = c5_of(nh)
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_h = nh.copy()
                if new_c5 < c5_of(h) or random.random() < math.exp(-(new_c5 - c5_of(h)) / T):
                    h = nh

        hi, hj = h[i], h[j]

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.97  # reserve a bit for safety

    # ------------------------------------------------------------------
    # Initial vector
    # ------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    except NameError:
        n = 70
        h = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Projection onto {0<=x<=1, sum(x)=target_sum}
    # ------------------------------------------------------------------
    def proj_sum(v):
        lo = np.min(v) - 1.0
        hi = np.max(v) + 1.0
        # bisection on λ so that sum(clip(v-λ,0,1)) = target_sum
        for _ in range(60):
            mid = (lo + hi) * 0.5
            x = np.clip(v - mid, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
        err = x.sum() - target_sum
        if abs(err) > 1e-10:
            idx = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
            if idx.size:
                adjust = err / idx.size
                x[idx] -= adjust
                x = np.clip(x, 0.0, 1.0)
        return x

    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h = proj_sum(h)

    # ------------------------------------------------------------------
    # Objective utilities
    # ------------------------------------------------------------------
    def c5_of(v):
        """max correlation × dx (dx = 2/n)"""
        return float(np.correlate(v, 1 - v, mode="full").max() * (2.0 / n))

    def full_corr(v):
        return np.correlate(v, 1 - v, mode="full")

    cur_c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    T = max(1e-4, cur_c5 * 0.05)   # initial temperature
    decay = 0.9997
    it = 0

    # ------------------------------------------------------------------
    # Main loop: targeted moves + random pair swaps
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        # correlation and worst shift
        corr = full_corr(h)
        idx_max = int(np.argmax(corr))
        s = idx_max - (n - 1)               # shift attaining the max

        # ---------- targeted move (skip shift 0) ----------
        if s != 0:
            if s > 0:
                i_inds = np.arange(n - s)
            else:
                i_inds = np.arange(-s, n)
            p_inds = i_inds + s
            contrib = h[i_inds] * (1.0 - h[p_inds])
            if contrib.size:
                k = int(np.argmax(contrib))
                i_best, p_best = i_inds[k], p_inds[k]
                eps = min(h[i_best], 1.0 - h[p_best])
                if eps > 1e-12:
                    nh = h.copy()
                    nh[i_best] -= eps
                    nh[p_best] += eps
                    new_c5 = c5_of(nh)
                    if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                        h = nh
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()

        # ---------- random two‑coordinate move ----------
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)     # increase i, decrease j
        max_down = min(hi, 1.0 - hj)   # decrease i, increase j
        if max_up + max_down > 0:
            delta = (random.random() * (max_up + max_down)) - max_down
            if abs(delta) > 1e-14:
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_c5 = c5_of(nh)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()

        # ---------- temperature schedule ----------
        it += 1
        if it % 1000 == 0:
            T *= decay

    # ------------------------------------------------------------------
    # Final projection (numerical drift)
    # ------------------------------------------------------------------
    best_h = proj_sum(best_h)
    final_c5 = c5_of(best_h)

    return (best_h.tolist(), final_c5, n)
