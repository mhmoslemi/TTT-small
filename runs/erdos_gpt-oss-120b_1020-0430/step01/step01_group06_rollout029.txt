def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # set n: possibly choose 256 or 128 based on time; we can compute n=128 if budget small else 256.
    # Let's pick n=128 for quick compute, but we can try 256 if fast.
    # We'll target n=256 for better bound.
    n = 256
    # Generate Thue-Morse sequence
    h = np.zeros(n, dtype=float)
    seq = np.array([0], dtype=int)
    while seq.size < n:
        seq = np.concatenate([seq, 1 - seq])
    h = seq[:n].astype(float)   # 0/1

    # Check sum
    target_sum = n / 2.0
    cur_sum = h.sum()
    # Should be exactly target. But ensure due to truncation? For power 2 it's exact.
    # adjust if not.
    if cur_sum != target_sum:
        # adjust by flipping some bits randomly to achieve target.
        diff = int(target_sum - cur_sum)
        # diff may be 0; else adjust.
        idx0 = np.where(h==0)[0]
        idx1 = np.where(h==1)[0]
        if diff > 0:
            # need to add ones
            chosen = np.random.choice(idx0, diff, replace=False)
            h[chosen] = 1
        elif diff < 0:
            chosen = np.random.choice(idx1, -diff, replace=False)
            h[chosen] = 0

    # function to compute c5
    def c5_of(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return float(corr.max() * (2.0 / len(v)))

    cur_c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # simulated annealing parameters
    T = cur_c5 * 0.05 + 1e-4
    decay = 0.9995
    it = 0
    # We'll loop until time budget about 0.94 of total.
    while time.time() - start < budget_s * 0.94:
        # decide move type
        if it % 200 == 0:
            # targeted move using worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)
            # compute valid i indices
            if s >= 0:
                i_start = 0
                i_end = n - s
            else:
                i_start = -s
                i_end = n
            # get list of i where h[i]==1 and h[i+s]==0
            # we can vectorize to speed
            i_range = np.arange(i_start, i_end)
            i_plus_s = i_range + s
            # boolean masks
            mask = (h[i_range] > 0.5) & (h[i_plus_s] < 0.5)
            candidates = i_range[mask]
            if candidates.size > 0:
                i_sel = int(np.random.choice(candidates))
                # swap i_sel and i_sel+s
                nh = h.copy()
                nh[i_sel], nh[i_sel + s] = nh[i_sel + s], nh[i_sel]
                new_c5 = c5_of(nh)
                # accept
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
        else:
            # random swap of a 1 and a 0
            # find random i with h[i]==1
            # fast random selection: sample until find.
            i = random.randrange(n)
            while h[i] < 0.5:
                i = random.randrange(n)
            j = random.randrange(n)
            while h[j] > 0.5:
                j = random.randrange(n)
            if i == j:
                continue
            nh = h.copy()
            nh[i], nh[j] = nh[j], nh[i]
            new_c5 = c5_of(nh)
            if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                h = nh
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        it += 1
        if it % 1000 == 0:
            T *= decay

    # final greedy improvement: try random improving swaps only
    for _ in range(20000):
        i = random.randrange(n)
        while h[i] < 0.5:
            i = random.randrange(n)
        j = random.randrange(n)
        while h[j] > 0.5:
            j = random.randrange(n)
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        new_c5 = c5_of(nh)
        if new_c5 < cur_c5 - 1e-12:
            h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        # optional early break if no improvement for many loops

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)

def targeted_move(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = int(np.argmax(corr))
    s = idx_max - (len(h) - 1)
    n = len(h)
    # neighbor sum as described.
    if s >= 0:
        left = np.concatenate([np.zeros(s), h[:n-s]])
        right = np.concatenate([h[s:], np.zeros(s)])
    else:
        s_abs = -s
        left = np.concatenate([h[s_abs:], np.zeros(s_abs)])
        right = np.concatenate([np.zeros(s_abs), h[:n-s_abs]])
    neighbor = left + right
    # source idx where h>0 (or >1e-12) and neighbor minimal
    src_mask = h > 1e-12
    if not np.any(src_mask):
        return None  # nothing to move
    src_candidates = np.where(src_mask)[0]
    src_idx = src_candidates[np.argmin(neighbor[src_candidates])]
    # target idx where h<1 (or <1-1e-12) and neighbor maximal
    tgt_mask = h < 1 - 1e-12
    tgt_candidates = np.where(tgt_mask)[0]
    if not np.any(tgt_mask):
        return None
    tgt_idx = tgt_candidates[np.argmax(neighbor[tgt_candidates])]
    if src_idx == tgt_idx:
        # pick random different index
        # maybe just return None or pick random pair.
        return None
    eps = min(h[src_idx], 1 - h[tgt_idx])
    if eps < 1e-12:
        return None
    nh = h.copy()
    nh[src_idx] -= eps
    nh[tgt_idx] += eps
    return nh

        if new_c5 < best_c5 - 1e-12:
            best_h = nh
            best_c5 = new_c5
        else:
            break

    # --------------------------------------------------------------
    # Final tiny correction of the integral (numerical drift)
    # --------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        for _ in range(200):
            k = random.randrange(n)
            if err > 0 and best_h[k] >= err:
                best_h[k] -= err
                break
            if err < 0 and best_h[k] <= 1.0 + err:
                best_h[k] -= err   # err negative → addition
                break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
```
