def hill_climb(h, cur_c5, deadline, rng):
    while time.time() < deadline:
        i,j = rng.choice(n, size=2, replace=False)
        # direction random
        if rng.random() < 0.5:

    while time.time() < deadline:
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        # contributions
        if k >= 0:
            contrib = np.empty(n)
            contrib[:n - k] = h[:n - k] * (1 - h[k:])
            contrib[n - k:] = h[n - k:]  # the tail region no overlap

import time
import numpy as np

# ---------------------------------------------------------
# Helper utilities
# ---------------------------------------------------------
def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyperplane sum(x)=target while staying in [0,1]."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0.0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _full_corr(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.size
    b = 1.0 - h
    revb = b[::-1]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(revb, n=L)
    conv = np.fft.irfft(H * B, n=L)[: 2 * n - 1]
    return conv


def _evaluate_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a step vector."""
    n = h.size
    conv = _full_corr(h)
    return float(conv.max() * (2.0 / n))


def _hill_climb(h: np.ndarray, cur_c5: float, deadline: float, rng: np.random.Generator):
    """Simple pairwise mass‑exchange hill climbing (only accept improvements)."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h = h_new
            cur_c5 = new_c5
    return h, cur_c5


def _targeted_refine(h: np.ndarray, cur_c5: float, deadline: float, rng: np.random.Generator):
    """Improve by moving mass from indices that contribute most to the worst shift."""
    n = h.size
    while time.time() < deadline:
        conv = _full_corr(h)
        k_idx = int(conv.argmax())
        k = k_idx - (n - 1)          # signed shift
        # contributions for this shift
        contrib = np.empty(n, dtype=np.float64)
        if k >= 0:
            # overlapping part
            contrib[: n - k] = h[: n - k] * (1.0 - h[k:])
            # tail where shifted part is zero
            contrib[n - k :] = h[n - k :]
        else:
            kk = -k
            contrib[kk:] = h[kk:] * (1.0 - h[: n - kk])
            contrib[:kk] = h[:kk]
        i = int(contrib.argmax())
        # paired index according to shift
        j = i + k if k >= 0 else i - kk
        if not (0 <= j < n):
            # fallback to a random pair
            i, j = rng.choice(n, size=2, replace=False)
        # decide direction that actually moves mass
        if h[i] > 0 and h[j] < 1.0:
            src, dst = i, j
        elif h[j] > 0 and h[i] < 1.0:
            src, dst = j, i
        else:
            continue
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        # pick a moderate step size
        delta = max_move * (0.2 + 0.8 * rng.random())
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h = h_new
            cur_c5 = new_c5
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------
    # 1. Choose resolution and build an initial guess
    # -----------------------------------------------------
    n = 200                               # number of discretisation points
    target = n / 2.0                      # required sum of h
    dx = 2.0 / n

    # start from a Gaussian‑shaped profile; explore a few widths
    best_h = None
    best_c5 = np.inf
    for sigma in np.linspace(0.30, 0.55, 13):
        x = np.arange(n) * dx
        h = np.exp(-((x - 1.0) ** 2) / (2.0 * sigma ** 2))
        h = np.clip(h, 0.0, 1.0)
        h = _adjust_sum(h, target)
        c5 = _evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # If a parent construction is supplied, try to reuse it (resample if necessary)
    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        n0 = parent.size
        if n0 == n:
            cand = _adjust_sum(parent, target)
            cand_c5 = _evaluate_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5
        else:
            # linear interpolation to the new grid
            x0 = np.arange(n0) * (2.0 / n0)
            x_new = np.arange(n) * dx
            interp = np.interp(x_new, x0, parent)
            cand = _adjust_sum(np.clip(interp, 0.0, 1.0), target)
            cand_c5 = _evaluate_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5

    # -----------------------------------------------------
    # 2. Targeted refinement (major part of the budget)
    # -----------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.70
    cur_h = best_h.copy()
    cur_c5 = best_c5
    cur_h, cur_c5 = _targeted_refine(cur_h, cur_c5, phase1_deadline, rng)

    # keep global best
    if cur_c5 < best_c5:
        best_h, best_c5 = cur_h.copy(), cur_c5

    # -----------------------------------------------------
    # 3. Simulated‑annealing style random mass exchange
    # -----------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.90
    while time.time() < phase2_deadline:
        # linear temperature schedule (starts at 0.02, ends at 0)
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.90)
        T = 0.02 * max(0.0, 1.0 - frac)

        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            continue
        delta = max_move * rng.random()
        h_new = cur_h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5

    # -----------------------------------------------------
    # 4. Final pure hill‑climb for polishing
    # -----------------------------------------------------
    final_deadline = start_time + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, final_deadline, rng)

    # -----------------------------------------------------
    # 5. Return feasible solution
    # -----------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
