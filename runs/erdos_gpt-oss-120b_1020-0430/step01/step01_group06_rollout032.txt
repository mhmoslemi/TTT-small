def solve_milp(n):
    w = n // 2
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == w, t >= 0]

def solve_ilp(n, w, time_limit):
    import cvxpy as cp, numpy as np
    v = cp.Variable(n, boolean=True)
    t = cp.Variable()   # real
    constraints = [cp.sum(v) == w]

    y_vars = {}
    for s in range(1, n):
        length = n - s
        y = cp.Variable(length, boolean=True)
        y_vars[s] = y
        # linking
        v_i = v[:length]
        v_j = v[s:]
        constraints += [y <= v_i, y <= v_j, y >= v_i + v_j - 1]
        # correlation constraints
        corr_pos = cp.sum(v_i) - cp.sum(y)
        corr_neg = cp.sum(v_j) - cp.sum(y)
        constraints += [corr_pos <= t, corr_neg <= t]

    # t >= 0 maybe
    constraints += [t >= 0]

        return None, None
    v_val = v.value
    # threshold to binary 0/1

    t_val = t.value
    return v_bin, t_val

def solve_via_milp(n, w, time_limit):
    import cvxpy as cp, numpy as np, time
    v = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(v) == w, t >= 0]

    for s in range(1, n):
        length = n - s
        y = cp.Variable(length, boolean=True)
        # link y with v
        constraints += [y <= v[:length]]
        constraints += [y <= v[s:]]
        constraints += [y >= v[:length] + v[s:] - 1]
        # correlation constraints
        corr_pos = cp.sum(v[:length]) - cp.sum(y)
        corr_neg = cp.sum(v[s:]) - cp.sum(y)
        constraints += [corr_pos <= t, corr_neg <= t]

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)
    except Exception:
        # maybe heuristic
        return None, None

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None

    v_val = v.value
    # threshold
    v_bin = np.where(v_val >= 0.5, 1.0, 0.0)  # convert to binary
    t_val = t.value
    return v_bin, t_val

    # First attempt: solve the exact integer programme
    # --------------------------------------------------------------
    # Time budget: keep a safety margin for fallback routine
    time_for_ip = budget_s * 0.75
    remaining = budget_s - (time.time() - start)
    if remaining > time_for_ip:
        time_for_ip = remaining

    ip_solution = None
    ip_obj = None

    # ------------------------------------------------------------------
    # Build and solve the MILP using cvxpy (CBC or GLPK_MI)
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        # Variables
        v = cp.Variable(n, boolean=True)               # h_i ∈ {0,1}
        t = cp.Variable(integer=True)                  # max correlation (integer)

        constraints = [cp.sum(v) == w, t >= 0]

        # for each shift s = 1 .. n-1 create product variables and correlation constraints
        for s in range(1, n):
            length = n - s
            y = cp.Variable(length, boolean=True)      # y_i = v_i * v_{i+s}
            # linking constraints
            constraints += [y <= v[:length]]
            constraints += [y <= v[s:]]
            constraints += [y >= v[:length] + v[s:] - 1]

            # correlation for +s (left shift) and -s (right shift)
            corr_pos = cp.sum(v[:length]) - cp.sum(y)      # Σ v_i (1‑v_{i+s})
            corr_neg = cp.sum(v[s:]) - cp.sum(y)          # Σ v_{i+s} (1‑v_i)
            constraints += [corr_pos <= t, corr_neg <= t]

        prob = cp.Problem(cp.Minimize(t), constraints)

        # solver choice – CBC preferred, fall back to GLPK_MI if unavailable
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_for_ip)
        except Exception:
            prob.solve(solver=cp.GLPK_MI, verbose=False, max_seconds=time_for_ip)

        if prob.status in ("optimal", "optimal_inaccurate"):
            v_val = v.value
            # round to exact 0/1 (numerical tolerances)
            h_int = np.where(v_val >= 0.5, 1.0, 0.0)
            # sanity check – enforce the exact half‑sum if rounding disturbed it
            cur = int(h_int.sum())
            if cur != w:
                diff = w - cur
                # add or remove ones arbitrarily (few adjustments, negligible effect)
                zeros = np.where(h_int == 0)[0]
                ones = np.where(h_int == 1)[0]
                if diff > 0:
                    sel = np.random.choice(zeros, diff, replace=False)
                    h_int[sel] = 1.0
                elif diff < 0:
                    sel = np.random.choice(ones, -diff, replace=False)
                    h_int[sel] = 0.0
            ip_solution = h_int.tolist()
            ip_obj = float(t.value) * dx          # t is the integer max correlation count
    except Exception:
        # any problem importing/using cvxpy – ignore and fall back
        ip_solution = None

    # --------------------------------------------------------------
    # If IP succeeded and gives a good bound, return it immediately
    # --------------------------------------------------------------
    if ip_solution is not None:
        final_c5 = c5_of(np.array(ip_solution, dtype=float))
        return (ip_solution, final_c5, n)

    # --------------------------------------------------------------
    # Fallback: simple simulated‑annealing/hill‑climbing search
    # --------------------------------------------------------------
    # initialise from provided vector or random binary with exact weight
    if h0.size == n:
        h = np.clip(h0, 0.0, 1.0)
        # project onto the hyperplane sum = w  (preserving bounds)
        cur_sum = h.sum()
        if cur_sum != w:
            # simple deterministic repair
            diff = w - cur_sum
            if diff > 0:
                zero_idx = np.where(h < 0.5)[0]
                chosen = np.random.choice(zero_idx, int(diff), replace=False)
                h[chosen] = 1.0
            else:
                one_idx = np.where(h > 0.5)[0]
                chosen = np.random.choice(one_idx, int(-diff), replace=False)
                h[chosen] = 0.0
    else:
        h = np.zeros(n)
        idx = np.arange(n)
        np.random.shuffle(idx)
        h[idx[:w]] = 1.0

    best_h = h.copy()
    best_val = c5_of(best_h)
    cur_val = best_val

    # annealing parameters (tuned for speed)
    T = max(1e-4, cur_val * 0.02)
    decay = 0.9993
    it = 0

    while time.time() - start < budget_s * 0.95:
        # ---------------------------------------------------------
        # targeted improvement: pick the worst shift and try to fix it
        # ---------------------------------------------------------
        if it % 400 == 0:
            corr = np.correlate(h, 1 - h, mode="full")
            s_idx = int(np.argmax(corr)) - (n - 1)          # shift attaining the max
            if s_idx >= 0:
                i_range = np.arange(n - s_idx)
                j_range = i_range + s_idx
            else:
                s_abs = -s_idx
                i_range = np.arange(s_abs, n)
                j_range = i_range - s_abs
            # contribution of each i to the current max correlation
            contrib = h[i_range] * (1.0 - h[j_range])
            # we would like to turn a 1‑0 pair into 0‑1 (or vice‑versa)
            bad = np.where(contrib > 0)[0]
            if bad.size:
                k = random.choice(bad)
                i, j = i_range[k], j_range[k]
                eps = min(h[i], 1.0 - h[j])
                if eps > 1e-12:
                    nh = h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    new_val = c5_of(nh)
                    if new_val < cur_val or random.random() < np.exp(-(new_val - cur_val) / T):
                        h, cur_val = nh, new_val
                        if new_val < best_val:
                            best_h, best_val = nh.copy(), new_val
        # ---------------------------------------------------------
        # generic two‑coordinate exchange (preserve total sum)
        # ---------------------------------------------------------
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)      # increase i, decrease j
        max_down = min(hi, 1.0 - hj)    # decrease i, increase j
        if max_up + max_down > 0:
            delta = (random.random() * (max_up + max_down)) - max_down
            if abs(delta) > 1e-14:
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_val = c5_of(nh)
                if new_val < cur_val or random.random() < np.exp(-(new_val - cur_val) / T):
                    h, cur_val = nh, new_val
                    if new_val < best_val:
                        best_h, best_val = nh.copy(), new_val

        it += 1
        if it % 1000 == 0:
            T *= decay

    # final tiny correction for possible rounding drift
    err = best_h.sum() - w
    if abs(err) > 1e-8:
        for _ in range(100):
            k = random.randrange(n)
            if err > 0 and best_h[k] >= err:
                best_h[k] -= err
                break
            if err < 0 and best_h[k] <= 1 + err:
                best_h[k] -= err   # err negative => addition
                break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
