def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    # ----------------------------------------------------------------------
    # Setup
    # ----------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95          # safe margin

    # ----------------------------------------------------------------------
    # Helper: C5 evaluation (full correlation, scaled by dx)
    # ----------------------------------------------------------------------
    def c5_of(vec):
        """Return max overlap integral for a step function represented by vec."""
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return float(corr.max() * (2.0 / len(vec)))

    # ----------------------------------------------------------------------
    # Determine problem size
    # ----------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float).copy()
        n = init_vec.shape[0]
    except NameError:
        n = 70
        init_vec = None

    dx = 2.0 / n
    target_sum = n / 2.0                     # ∑h = n/2 exactly

    # ----------------------------------------------------------------------
    # Project a given vector onto the feasible polytope (box + sum)
    # ----------------------------------------------------------------------
    def project(vec):
        vec = np.clip(vec, 0.0, 1.0)
        cur = vec.sum()
        if abs(cur - target_sum) < 1e-12:
            return vec
        # simple scaling then tiny fix‑up
        if cur > 0:
            vec *= target_sum / cur
        vec = np.clip(vec, 0.0, 1.0)
        cur = vec.sum()
        # iterative correction for any residual error
        for _ in range(200):
            err = cur - target_sum
            if abs(err) <= 1e-12:
                break
            if err > 0:
                cand = np.where(vec > 1e-12)[0]
                if len(cand) == 0:
                    break
                i = random.choice(cand.tolist())
                delta = min(vec[i], err)
                vec[i] -= delta
            else:
                cand = np.where(vec < 1.0 - 1e-12)[0]
                if len(cand) == 0:
                    break
                i = random.choice(cand.tolist())
                delta = min(1.0 - vec[i], -err)
                vec[i] += delta
            cur = vec.sum()
        return vec

    # ----------------------------------------------------------------------
    # Build initial candidates
    # ----------------------------------------------------------------------
    candidates = []

    if init_vec is not None:
        # use the supplied vector (after projection)
        init_vec = project(init_vec)
        candidates.append(init_vec)

        # also use a binary version obtained by taking the largest entries
        bins = np.zeros_like(init_vec)
        idx_desc = np.argsort(-init_vec)          # descending order
        bins[idx_desc[: n // 2]] = 1.0
        candidates.append(bins)

    # ----------------------------------------------------------------------
    # Random binary generator
    # ----------------------------------------------------------------------
    def random_binary():
        """Return a random 0/1 vector with exactly n/2 ones."""
        perm = np.random.permutation(n)
        vec = np.zeros(n, dtype=float)
        vec[perm[: n // 2]] = 1.0
        return vec

    # ----------------------------------------------------------------------
    # Core local optimiser: greedy best 1↔0 swap
    # ----------------------------------------------------------------------
    def local_optimize(vec):
        """Repeatedly perform the best improving 1↔0 swap until none exists."""
        cur_c5 = c5_of(vec)
        # pre‑compute positions of ones / zeros
        while True:
            ones = np.where(vec > 0.5)[0]
            zeros = np.where(vec < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            best_delta = 0.0
            best_pair = None
            best_new = cur_c5
            # exhaustive search of all possible swaps
            for i in ones:
                for j in zeros:
                    # perform tentative swap
                    tmp = vec.copy()
                    tmp[i], tmp[j] = tmp[j], tmp[i]
                    new_c5 = c5_of(tmp)
                    delta = cur_c5 - new_c5
                    if delta > best_delta + 1e-12:
                        best_delta = delta
                        best_pair = (i, j)
                        best_new = new_c5
                # early exit if a perfect improvement (delta > 0) found?
                # continue searching for the *best* improvement
            if best_pair is None:
                break                     # local optimum reached
            i, j = best_pair
            vec[i], vec[j] = vec[j], vec[i]
            cur_c5 = best_new
        return vec, cur_c5

    # ----------------------------------------------------------------------
    # Random escape helper (allow occasional worsening)
    # ----------------------------------------------------------------------
    def escape_and_optimize(start_vec, max_tries=5):
        """Try a few random swaps, each followed by local optimisation."""
        best_vec, best_val = start_vec, c5_of(start_vec)
        for _ in range(max_tries):
            # make a random 1↔0 swap (even if it worsens)
            ones = np.where(start_vec > 0.5)[0]
            zeros = np.where(start_vec < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = random.choice(ones.tolist())
            j = random.choice(zeros.tolist())
            pert = start_vec.copy()
            pert[i], pert[j] = pert[j], pert[i]
            # local optimise the perturbed configuration
            pert_opt, pert_val = local_optimize(pert)
            if pert_val < best_val - 1e-12:
                best_vec, best_val = pert_opt, pert_val
        return best_vec, best_val

    # ----------------------------------------------------------------------
    # Main search loop
    # ----------------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")

    # first evaluate any predetermined candidates
    for cand in candidates:
        opt_vec, opt_val = local_optimize(cand.copy())
        if opt_val < best_c5:
            best_c5, best_h = opt_val, opt_vec.copy()

    # random restarts
    while time.time() < deadline:
        h0 = random_binary()
        # deterministic improvement
        h_opt, val_opt = local_optimize(h0)
        if val_opt < best_c5:
            best_c5, best_h = val_opt, h_opt.copy()
        # occasional escape attempts from this local optimum
        if random.random() < 0.15:          # 15 % of restarts try a perturb‑plus‑opt.
            h_esc, val_esc = escape_and_optimize(h_opt, max_tries=3)
            if val_esc < best_c5:
                best_c5, best_h = val_esc, h_esc.copy()
        # tiny sanity check to keep the loop fast
        if time.time() - start_time > budget_s * 0.94:
            break

    # ----------------------------------------------------------------------
    # Final safety – enforce exact sum (numerical drift may appear)
    # ----------------------------------------------------------------------
    if best_h is not None:
        best_h = project(best_h)
        best_c5 = c5_of(best_h)

    # ----------------------------------------------------------------------
    # Return result
    # ----------------------------------------------------------------------
    # best_h is a NumPy array; convert to plain Python list as required.
    return (best_h.tolist(), float(best_c5), n)
