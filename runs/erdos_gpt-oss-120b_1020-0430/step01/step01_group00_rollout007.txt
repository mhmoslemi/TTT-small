import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h)=n/2 using a water‑filling scheme.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random feasible vector (uniform in the hyper‑cube, projected)."""
    rng = np.random.default_rng()
    a = rng.random(n)
    return _project_to_feasible(a, n)


# --------------------------------------------------------------
# Core optimisation routine (projected gradient on smooth max)
# --------------------------------------------------------------

def _refine_h(
    h0: np.ndarray,
    budget: float,
    target_bound: float = 0.3808,
) -> tuple:
    """
    Refine a feasible h using a smoothed max objective.
    Returns (best_h, best_c5).
    """
    n = h0.size
    h = _project_to_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    cur_c5 = best_c5

    start = time.time()
    max_iter = 300_000                     # generous upper bound
    alpha_start, alpha_end = 1.0, 30.0     # smooth‑max temperature schedule
    lr_start, lr_end = 0.5, 0.001          # learning‑rate schedule

    stagnation = 0
    for it in range(max_iter):
        if time.time() - start > budget - 0.05:
            break

        # linear schedules
        t = it / max_iter
        alpha = alpha_start + t * (alpha_end - alpha_start)
        lr = lr_start + t * (lr_end - lr_start)

        # ----- gradient of smoothed max -----
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = corr.max()
        # softmax weights (stable)
        exp_term = np.exp(alpha * (corr - m))
        w = exp_term / exp_term.sum()

        # convolutions give Σ w_s h_{p±s}
        conv_fwd = np.convolve(w, h, mode="full")
        conv_rev = np.convolve(w[::-1], h, mode="full")
        center = n - 1
        part1 = conv_fwd[center:center + n]
        part2 = conv_rev[center:center + n]
        grad = 1.0 - (part1 + part2)   # gradient of smooth max

        # ----- gradient step + projection -----
        h_new = _project_to_feasible(h - lr * grad, n)
        new_c5 = _c5_from_h(h_new)

        # accept if improvement (monotone descent)
        if new_c5 + 1e-12 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            stagnation = 0
            if new_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5
                if best_c5 <= target_bound:
                    break
        else:
            stagnation += 1
            # occasional random jump if stuck
            if stagnation >= 4000:
                h = _project_to_feasible(best_h + 0.02 * np.random.randn(n), n)
                cur_c5 = _c5_from_h(h)
                stagnation = 0

    return best_h, best_c5

    except NameError:
        # fall back to uniform half‑half on a modest grid
        h_initial = np.full(128, 0.5, dtype=float)

    h_initial = _project_to_feasible(h_initial, h_initial.size)

    best_h = h_initial
    best_c5 = _c5_from_h(best_h)
    best_n = best_h.size

    target = 0.3808

    # ---- refinement stages (coarse → fine) ----
    stages = [128, 256, 300]   # ordered list of grid sizes (even numbers)
    # allocate time proportionally (more for the coarse stage)
    allocated = {
        128: 0.5 * budget_s,
        256: 0.3 * budget_s,
        300: 0.2 * budget_s,
    }

    for n in stages:
        if time_left() < 2.0:
            break

        # Build starting point for this resolution
        if best_n != n:
            h_start = _upscale_h(best_h, n)
        else:
            h_start = best_h.copy()

        # Give a quick random‑restart budget as well
        restarts = 3
        remaining = min(time_left(), allocated.get(n, time_left()))
        per_restart = remaining / (restarts + 1)

        # first, refine the deterministic upscale
        h_opt, c5_opt = _refine_h(h_start, per_restart, target_bound=target)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

        # then a few random starts
        for _ in range(restarts):
            if time_left() < 2.0:
                break
            h_rand = _random_feasible(n)
            h_opt, c5_opt = _refine_h(h_rand, per_restart, target_bound=target)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 <= target:
                break
        if best_c5 <= target:
            break

    # ---- final safety fallback (uniform half‑half) ----
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
