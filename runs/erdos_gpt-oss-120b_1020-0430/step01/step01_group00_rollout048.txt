def local_search(h, max_iter, temp_start, temp_end):
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    n = len(h)
    # schedule temperature decays linearly
    for it in range(max_iter):
        t_frac = it / max_iter
        temp = temp_start * (1 - t_frac) + temp_end * t_frac
        # choose random i,j dist
        i, j = np.random.choice(n, 2, replace=False)
        # compute feasible delta
        max_up_i = 1.0 - current_h[i]
        max_down_j = current_h[j]
        max_up_j = 1.0 - current_h[j]
        max_down_i = current_h[i]
        # random direction
        if np.random.rand() < 0.5:
            # move from j to i
            max_delta = min(max_up_i, max_down_j)
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # move from i to j
            max_delta = min(max_up_j, max_down_i)
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / max(temp, 1e-12)):
            current_h = new_h
            current_c5 = new_c5
    return current_h, current_c5

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 500) -> np.ndarray:
    """Project h onto the feasible set: 0≤h≤1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value (max overlap integral) for a given h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """Inequality constraints: t - correlation_k >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Equality constraint: sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    """Run SLSQP starting from initial_h, return feasible h and its C5."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _local_search(h: np.ndarray, max_iter: int, time_left_func) -> tuple:
    """Simple simulated‑annealing style pairwise move search."""
    n = h.size
    cur_h = h.copy()
    cur_c5 = _c5_from_h(cur_h)
    # temperature schedule (starts warm, ends near zero)
    T0, T1 = 1e-2, 1e-6
    for it in range(max_iter):
        if time_left_func() < 0.5:
            break
        # linear temperature decay
        T = T0 * (1 - it / max_iter) + T1 * (it / max_iter)
        i, j = np.random.choice(n, 2, replace=False)
        # random direction
        if np.random.rand() < 0.5:
            # move mass from j to i
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # move mass from i to j
            max_delta = min(1.0 - cur_h[j], cur_h[i])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_c5 = _c5_from_h(new_h)
        d = new_c5 - cur_c5
        if d < 0 or np.random.rand() < np.exp(-d / max(T, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
    return cur_h, cur_c5


# -------------------------------------------------
# Main entry point
# -------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (list_of_h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # -----------------------------------------------------------------
    # Retrieve optional parent construction
    # -----------------------------------------------------------------
    try:
        init_h_global = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h_global = None

    # -----------------------------------------------------------------
    # Configuration
    # -----------------------------------------------------------------
    target_bound = 0.3808
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Determine a set of grid sizes to explore
    n0 = init_h_global.size if init_h_global is not None else 128
    n_candidates = set()
    n_candidates.add(n0)
    # multiples of the base size
    for mul in range(2, 7):
        n = n0 * mul
        if n <= 512 and n % 2 == 0:
            n_candidates.add(n)
    # some handcrafted even sizes up to 512
    for n in (150, 170, 200, 240, 280, 320, 360, 400, 440, 480, 512):
        if n % 2 == 0:
            n_candidates.add(n)
    n_candidates = sorted(n_candidates)

    # -----------------------------------------------------------------
    # Main optimisation loop over grid sizes
    # -----------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 8.0:
            break

        # prepare a base function for this resolution
        if init_h_global is not None and init_h_global.size != n:
            base_h = _upscale_h(init_h_global, n)
        elif init_h_global is not None:
            base_h = init_h_global.copy()
        elif best_h is not None and best_n != n:
            base_h = _upscale_h(best_h, n)
        else:
            base_h = np.full(n, 0.5)

        # number of stochastic restarts, grows when we have time
        max_restarts = 12
        sigma = 0.3  # initial perturbation scale

        for restart in range(max_restarts):
            if time_left() < 6.0:
                break

            # produce a perturbed start
            start_h = base_h + sigma * np.random.randn(n)
            start_h = _project_to_feasible(start_h, n)

            # run SLSQP
            h_opt, c5_val = _solve_h(start_h, max_iter=350)

            # local simulated annealing refinement
            if time_left() > 2.0:
                h_opt, c5_val = _local_search(
                    h_opt,
                    max_iter=800,
                    time_left_func=time_left,
                )

            if c5_val < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_val, n
                # early stop if we passed the target bound
                if best_c5 < target_bound:
                    break

            # shrink perturbation radius gradually
            sigma *= 0.84

        if best_c5 < target_bound:
            break

    # -----------------------------------------------------------------
    # Fallback (should never be needed)
    # -----------------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    # Return as plain Python list for the harness
    return best_h.tolist(), float(best_c5), int(best_n)
