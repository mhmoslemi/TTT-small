def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Problem size and target sum
    # --------------------------------------------------------------
    try:
        init_vals = np.array(initial_h_values, dtype=float)
        n = int(init_vals.shape[0])
        target = n // 2
        # binary start from initial values (threshold 0.5)
        h0 = (init_vals >= 0.5).astype(np.int8)
        # repair to exact target count
        while h0.sum() != target:
            if h0.sum() > target:
                i = random.choice(np.where(h0 == 1)[0].tolist())
                h0[i] = 0
            else:
                i = random.choice(np.where(h0 == 0)[0].tolist())
                h0[i] = 1
    except NameError:
        n = 70
        target = n // 2
        h0 = np.zeros(n, dtype=np.int8)
        h0[random.sample(range(n), target)] = 1

    dx = 2.0 / n

    # --------------------------------------------------------------
    # Objective evaluation
    # --------------------------------------------------------------
    def c5_of(v):
        """max correlation * dx"""
        # v may be integer or float
        return float(np.correlate(v, 1 - v, mode="full").max() * dx)

    # --------------------------------------------------------------
    # Initial best
    # --------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = c5_of(best_h)

    # --------------------------------------------------------------
    # Genetic algorithm parameters
    # --------------------------------------------------------------
    pop_size = 200
    if pop_size % 2:
        pop_size += 1
    tournament_k = 5
    mutation_prob = 0.07
    elite_num = 2

    # --------------------------------------------------------------
    # Population initialization
    # --------------------------------------------------------------
    population = [best_h]
    while len(population) < pop_size:
        ind = np.zeros(n, dtype=np.int8)
        ind[random.sample(range(n), target)] = 1
        population.append(ind)
    population = np.array(population, dtype=np.int8)

    # --------------------------------------------------------------
    # Main GA loop
    # --------------------------------------------------------------
    while time.time() - start < budget_s * 0.90:
        # evaluate
        scores = np.array([c5_of(ind) for ind in population])
        # keep elites
        new_pop = []
        elite_idx = np.argsort(scores)[:elite_num]
        for i in elite_idx:
            new_pop.append(population[i].copy())

        # produce offspring
        while len(new_pop) < pop_size:
            # tournament selection
            cand = np.random.choice(pop_size, tournament_k, replace=False)
            p1 = population[cand[np.argmin(scores[cand])]]
            cand = np.random.choice(pop_size, tournament_k, replace=False)
            p2 = population[cand[np.argmin(scores[cand])]]

            # uniform crossover
            mask = np.random.randint(0, 2, size=n).astype(bool)
            child = np.where(mask, p1, p2).astype(np.int8)

            # repair to keep exact target count
            cur = int(child.sum())
            if cur > target:
                ones = np.where(child == 1)[0]
                for i in random.sample(ones.tolist(), cur - target):
                    child[i] = 0
            elif cur < target:
                zeros = np.where(child == 0)[0]
                for i in random.sample(zeros.tolist(), target - cur):
                    child[i] = 1

            # mutation
            if mutation_prob > 0:
                mut = np.random.rand(n) < mutation_prob
                child[mut] = 1 - child[mut]
                cur = int(child.sum())
                if cur > target:
                    ones = np.where(child == 1)[0]
                    for i in random.sample(ones.tolist(), cur - target):
                        child[i] = 0
                elif cur < target:
                    zeros = np.where(child == 0)[0]
                    for i in random.sample(zeros.tolist(), target - cur):
                        child[i] = 1

            # short local greedy improve (targeted swap)
            for _ in range(1):
                corr = np.correlate(child, 1 - child, mode="full")
                s_idx = int(np.argmax(corr))
                s = s_idx - (n - 1)
                if s >= 0:
                    i_idx = np.arange(n - s)
                    p_idx = i_idx + s
                else:
                    i_idx = np.arange(-s, n)
                    p_idx = i_idx + s
                # find a pair with contribution 1
                mask_pair = (child[i_idx] == 1) & (child[p_idx] == 0)
                if mask_pair.any():
                    k = random.choice(np.where(mask_pair)[0].tolist())
                    i_swap, p_swap = i_idx[k], p_idx[k]
                    child[i_swap] = 0
                    child[p_swap] = 1
                else:
                    # random swap as fallback
                    ones = np.where(child == 1)[0]
                    zeros = np.where(child == 0)[0]
                    if ones.size and zeros.size:
                        i_swap = random.choice(ones.tolist())
                        p_swap = random.choice(zeros.tolist())
                        child[i_swap], child[p_swap] = 0, 1

            new_pop.append(child)

        population = np.array(new_pop[:pop_size], dtype=np.int8)

        # Update global best
        scores = np.array([c5_of(ind) for ind in population])
        idx = int(scores.argmin())
        if scores[idx] < best_c5:
            best_c5 = float(scores[idx])
            best_h = population[idx].copy()

    # --------------------------------------------------------------
    # Post‑GA local refinement (swap hill‑climbing)
    # --------------------------------------------------------------
    h = best_h.astype(float)  # work in float for safety
    cur_c5 = best_c5

    while time.time() - start < budget_s * 0.99:
        corr = np.correlate(h, 1 - h, mode="full")
        s_idx = int(np.argmax(corr))
        s = s_idx - (n - 1)
        if s >= 0:
            i_idx = np.arange(n - s)
            p_idx = i_idx + s
        else:
            i_idx = np.arange(-s, n)
            p_idx = i_idx + s
        # look for a profitable 1→0, 0→1 swap on the worst shift
        mask_pair = (h[i_idx] > 0.5) & (h[p_idx] < 0.5)
        made_swap = False
        if mask_pair.any():
            k = random.choice(np.where(mask_pair)[0].tolist())
            i_sw, p_sw = i_idx[k], p_idx[k]
            h[i_sw] = 0.0
            h[p_sw] = 1.0
            made_swap = True
        else:
            # random swap of a 1 and a 0 elsewhere
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if ones.size and zeros.size:
                i_sw = random.choice(ones.tolist())
                p_sw = random.choice(zeros.tolist())
                h[i_sw] = 0.0
                h[p_sw] = 1.0
                made_swap = True

        if not made_swap:
            break

        new_c5 = c5_of(h)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i_sw] = 1.0
            h[p_sw] = 0.0

    # --------------------------------------------------------------
    # Return result
    # --------------------------------------------------------------
    final_c5 = c5_of(best_h)
    return (best_h.astype(float).tolist(), final_c5, n)
