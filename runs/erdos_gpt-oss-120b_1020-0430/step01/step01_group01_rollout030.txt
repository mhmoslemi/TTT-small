def _grad_smooth_max(h, alpha, dx):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = np.max(corr)
    w = np.exp(alpha * (corr - max_c))
    sum_w = np.sum(w)
    probs = w / sum_w
    grad = np.zeros(n)
    offset = n - 1
    for idx, p in enumerate(probs):
        if p == 0:
            continue
        s = idx - offset
        # compute gradient contributions
        # base ones
        g = np.full(n, 1.0)
        if s >= 0:
            if s < n:
                # subtract h[i+s] for i < n-s
                g[:n - s] -= h[s:]
                # subtract h[i-s] for i >= s
                g[s:] -= h[:n - s]
        else:

            if s_abs < n:
                # subtract h[i-s] = h[i+s_abs] for i < n - s_abs
                g[:n - s_abs] -= h[s_abs:]
                # subtract h[i+s] = h[i - s_abs] for i >= s_abs

    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x_i<=1, sum x_i = target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Overlap evaluation
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Derivative of the (unscaled) correlation at a given shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _grad_smooth_max(h: np.ndarray, alpha: float, dx: float) -> np.ndarray:
    """Gradient of the smooth max of correlations."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = np.max(corr)
    w = np.exp(alpha * (corr - max_c))
    sum_w = np.sum(w)
    probs = w / sum_w
    grad = np.zeros(n)

    offset = n - 1
    for idx, p in enumerate(probs):
        if p == 0.0:
            continue
        s = idx - offset
        # start from vector of ones (the constant part of derivative)
        g = np.full(n, 1.0)
        if s >= 0:
            if s < n:
                g[: n - s] -= h[s:]
                g[s:] -= h[: n - s]
        else:
            s_abs = -s
            if s_abs < n:
                g[: n - s_abs] -= h[s_abs:]
                g[s_abs:] -= h[: n - s_abs]
        grad += p * g * dx
    return grad


# ----------------------------------------------------------------------
# Equality constraint helpers (passed to scipy via args)
# ----------------------------------------------------------------------
def _eq_fun(x, alpha, dx, target_sum):
    return np.sum(x) - target_sum


def _eq_jac(x, alpha, dx, target_sum):
    return np.ones_like(x)


# ----------------------------------------------------------------------
# Smooth‑max objective and Jacobian (extra args allow a single args tuple)
# ----------------------------------------------------------------------
def _smooth_max_obj(x, alpha, dx, target_sum):
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(w)) / alpha


def _smooth_max_jac(x, alpha, dx, target_sum):
    return _grad_smooth_max(x, alpha, dx)


# ----------------------------------------------------------------------
# SLSQP refinement using the smooth‑max surrogate
# ----------------------------------------------------------------------
def _slsqp_optimize(x0: np.ndarray, target_sum: float,
                    alpha: float, max_iter: int = 200) -> np.ndarray:
    n = x0.shape[0]
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': _eq_fun,
             'jac': _eq_jac}]
    args = (alpha, dx, target_sum)
    res = minimize(_smooth_max_obj, x0, args=args, jac=_smooth_max_jac,
                   method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target_sum)


# ----------------------------------------------------------------------
# Exact‑max gradient descent (simple projected GD)
# ----------------------------------------------------------------------
def _gd_optimize(x0: np.ndarray, target_sum: float,
                 max_iter: int = 300, step0: float = 0.5,
                 decay: float = 0.99) -> np.ndarray:
    h = x0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    step = step0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift) * dx
        h = h - step * g
        h = _project_onto_capped_simplex(h, target_sum)
        step *= decay
    return h
