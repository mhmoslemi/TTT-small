def _fast_sa(start_h, max_seconds):
    h = start_h.copy()
    n = h.shape[0]
    off = n-1
    L = 2*n - 1
    padded = np.concatenate([np.zeros(off), h, np.zeros(off)])
    C = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cur_c = np.max(C) * dx
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    accepted = 0
    total_iter = 0
    while time.time() - start < max_seconds:
        total_iter +=1
        # probability to sample based on weight maybe:
        i = random.randrange(n)
        j = random.randrange(n-1)
        if j >= i:
            j += 1
        hi = h[i]; hj = h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        max_step = 0.03
        low_cap = max(low, -max_step)
        high_cap = min(high, max_step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)
        # compute delta vectors
        forward_i = padded[i : i+L]
        back_i = padded[i+2*off : i+2*off - L : -1]
        vec_i = (1.0 - forward_i) - back_i
        forward_j = padded[j : j+L]
        back_j = padded[j+2*off : j+2*off - L : -1]
        vec_j = (1.0 - forward_j) - back_j
        dC_vec = delta * vec_i - delta * vec_j
        C_new = C + dC_vec
        est_c = np.max(C_new) * dx

        # temperature schedule
        prog = (time.time() - start) / max_seconds
        T0 = 0.02
        T_end = 1e-6
        T = T0 * (T_end / T0) ** prog
        accept = False
        if est_c < cur_c:
            accept = True
        else:
            if T > 0 and random.random() < math.exp(-(est_c - cur_c) / T):
                accept = True
        if accept:
            h[i] += delta
            h[j] -= delta
            padded[off + i] = h[i]
            padded[off + j] = h[j]
            C = C_new
            cur_c = est_c
            accepted += 1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            # occasional exact recompute
            if accepted % 200 == 0:
                C = np.correlate(h, 1.0 - h, mode='full')
                cur_c = np.max(C) * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
    return best_h, best_c

def _initial_vector(n):

        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0., 2., n0, endpoint=False)
        x = np.linspace(0., 2., n, endpoint=False)
        h = np.interp(x, x0, h0)

            lo, hi = 0.0, 1.0
            if lo <= h[idx] + deficit <= hi:
                h[idx] += deficit

import time, math, random, numpy as np

def _c5_metric(h):
    """Exact C5 value for sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _initial_vector(n):
    """Create a feasible start vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + deficit <= hi:
                h[idx] += deficit
                break
    return h


def _fast_sa(start_h, max_seconds):
    """Fast simulated‑annealing using linear delta updates."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    off = n - 1
    L = 2 * n - 1

    # padded array: zeros on both sides for out‑of‑range accesses
    padded = np.concatenate([np.zeros(off), h, np.zeros(off)])
    C = np.correlate(h, 1.0 - h, mode="full")
    cur_c = np.max(C) * dx
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    accepted = 0
    T0, T_end = 0.05, 1e-6
    max_step = 0.04

    while time.time() - start < max_seconds:
        # pick a pair (i,j)
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        lowc = max(low, -max_step)
        highc = min(high, max_step)
        delta = random.uniform(lowc, highc) if lowc < highc else random.uniform(low, high)

        # delta vectors (length 2n‑1)
        f_i = padded[i : i + L]
        b_i = padded[i + 2 * off : i + 2 * off - L : -1]
        vec_i = (1.0 - f_i) - b_i

        f_j = padded[j : j + L]
        b_j = padded[j + 2 * off : j + 2 * off - L : -1]
        vec_j = (1.0 - f_j) - b_j

        dC = delta * vec_i - delta * vec_j
        Cnew = C + dC
        est_c = np.max(Cnew) * dx

        # temperature schedule
        prog = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** prog
        accept = False
        if est_c < cur_c:
            accept = True
        elif T > 0 and random.random() < math.exp(-(est_c - cur_c) / T):
            accept = True

        if accept:
            h[i] = hi + delta
            h[j] = hj - delta
            padded[off + i] = h[i]
            padded[off + j] = h[j]
            C = Cnew
            cur_c = est_c
            accepted += 1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            # occasional exact recompute to avoid drift
            if accepted % 200 == 0:
                C = np.correlate(h, 1.0 - h, mode="full")
                cur_c = np.max(C) * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
    return best_h, best_c


def _hill_climb(h, cur_c, time_limit):
    """Small deterministic hill‑climbing passes."""
    n = h.size
    dx = 2.0 / n
    start = time.time()
    while time.time() - start < time_limit:
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.05
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
        else:
            h[i] = hi
            h[j] = hj
    return h, cur_c

    random.seed(seed)
