def local_optimize(h, rng, max_iters=5000):
    # Hill climbing with occasional resets
    best = h.copy()
    best_c = compute_c5(best)
    step = 0.5
    for it in range(max_iters):
        # propose move:
        # either transfer epsilon between two random indices (same as perturb_h)
        # or random reset some fraction of variables
        if rng.random() < 0.1:
            # big perturb: randomly assign new values to a fraction of indices, then repair sum
            new = best.copy()
            idxs = rng.choice(len(new), size=int(len(new)*0.2), replace=False)
            new[idxs] = rng.random(len(idxs))
            # repair sum and clip and rescale
            new = np.clip(new, 0., 1.)
            # adjust sum to target
            target = len(new) / 2.0
            diff = target - new.sum()
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, len(new))
                add = min(diff, 1.0 - new[i]) if diff > 0 else max(diff, -new[i])
                new[i] += add
                diff = target - new[i].sum()

            new = perturb_h(best, rng, scale=step)
        c_new = compute_c5(new)
        if c_new < best_c:
            best, best_c = new, c_new
            # maybe increase step after improvement?
        # reduce step gradually
        if (it+1) % (max_iters//10) == 0:
            step *= 0.8
    return best, best_c

def softmax_obj(x):
    corr = np.correlate(x, 1-x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

CURRENT_TARGET = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step

            h = new_h
        # decay step

                global_best_h, global_best_c = h_opt.copy(), c_opt

        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

    # -------------------------------------------------------------
    # 5. Optional up‑sampling to a finer grid (if time permits)
    # -------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.96 and n * 2 <= 200:
        h_up = np.repeat(global_best_h, 2)
        n_up = h_up.size
        # sum is already correct (each entry duplicated), but enforce anyway
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # refine on the finer grid
        h_up_opt, c_up_opt = hill_climb(h_up, rng, max_iters=4000, start_step=0.3)

        for beta in beta_list:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up_opt = slsqp_optimize(h_up_opt, beta, max_iter=400)
            except Exception:
                pass
            c_up_opt = compute_c5(h_up_opt)
            if c_up_opt < global_best_c:
                global_best_h, global_best_c, n = h_up_opt.copy(), c_up_opt, n_up

    # -------------------------------------------------------------
    # 6. Final clean‑up (clip + exact sum)
    # -------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
