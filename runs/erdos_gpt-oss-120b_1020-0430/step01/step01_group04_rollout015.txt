import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2−x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def generate_cosine(n: int) -> np.ndarray:
    """Cosine bump h(x)=0.5+0.5*cos(pi*x), symmetrised and normalised."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1] with exact integral constraint."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def symmetric_binary(n: int) -> np.ndarray:
    """Binary symmetric block: half ones at each end, exactly n/2 total mass."""
    half = n // 2
    left = np.concatenate([np.ones(half // 2), np.zeros(half - half // 2)])
    return np.concatenate([left, left[::-1]]).astype(float)


# ------------------------------------------------------------
# Objective helpers for optimisation (penalty form)
# ------------------------------------------------------------
def obj_softmax_penalty(x, target_sum, alpha, lam):
    """softmax C5 + quadratic penalty for the integral constraint."""
    return softmax_c5(x, alpha) + lam * (np.sum(x) - target_sum) ** 2


def obj_de_penalty(x, target_sum, lam):
    """max‑overlap + quadratic penalty (used by DE)."""
    maxc = np.max(np.correlate(x, 1.0 - x, mode="full"))
    return maxc + lam * (np.sum(x) - target_sum) ** 2


# ------------------------------------------------------------
# Local refinement (two‑stage)
# ------------------------------------------------------------
def local_refine(start_h, n, target_sum, time_deadline):
    """Two‑stage L‑BFGS‑B refinement with growing softness."""
    lam = 300.0
    bounds = [(0.0, 1.0)] * n
    # stage 1 – soft
    res1 = opt.minimize(
        obj_softmax_penalty,
        start_h,
        args=(target_sum, 5.0, lam),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 250, "ftol": 1e-9},
    )
    if not res1.success or time.time() > time_deadline:
        return None
    # stage 2 – sharper
    res2 = opt.minimize(
        obj_softmax_penalty,
        res1.x,
        args=(target_sum, 30.0, lam),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 300, "ftol": 1e-9},
    )
    if not res2.success or time.time() > time_deadline:
        return None
    h_opt = np.clip(res2.x, 0.0, 1.0)
    # final exact projection onto the integral constraint
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-12:
        h_opt += diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # grid sizes – even numbers give an exact binary feasible point
    candidate_ns = [80, 100, 120, 150, 180, 200, 240]

    # incorporate parent if present
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    for n in candidate_ns:
        if time.time() > time_limit:
            break
        target_sum = n / 2.0
        # -----------------------------------------------------------------
        # Build a diverse set of starting vectors
        # -----------------------------------------------------------------
        starts = []

        # 0. parent (upsampled or exact)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # 1. symmetric binary block
        starts.append(symmetric_binary(n))

        # 2. classic triangular shape
        starts.append(generate_triangular(n))

        # 3. cosine bump
        starts.append(generate_cosine(n))

        # 4. several random feasible vectors
        for _ in range(5):
            starts.append(random_feasible(n, rng))

        # 5. piecewise‑constant random (K=10)
        K = 10
        blk_sz = n // K
        for _ in range(3):
            w = rng.random(K)
            w = w * ((K / 2.0) / w.sum())
            w = np.clip(w, 0.0, 1.0)
            diff = (K / 2.0) - w.sum()
            if abs(diff) > 1e-12:
                w += diff / K
                w = np.clip(w, 0.0, 1.0)
            h = np.repeat(w, blk_sz)
            if h.shape[0] < n:                     # handle non‑multiple case
                h = np.concatenate([h, np.zeros(n - h.shape[0])])
            starts.append(h)

        # -----------------------------------------------------------------
        # Evaluate candidates and keep a few best for local refinement
        # -----------------------------------------------------------------
        evaluated = []
        for s in starts:
            c5val = compute_c5(s)
            evaluated.append((c5val, s))
        evaluated.sort(key=lambda x: x[0])
        top_candidates = [cand for _, cand in evaluated[:3]]

        # -----------------------------------------------------------------
        # Local refinement of the elite candidates
        # -----------------------------------------------------------------
        for cand in top_candidates:
            if time.time() > time_limit:
                break
            refined = local_refine(cand, n, target_sum, time_limit)
            if refined is None:
                continue
            c5ref = compute_c5(refined)
            if c5ref < best_c5 - 1e-9:
                best_c5 = c5ref
                best_h = refined.copy()
                best_n = n

        # -----------------------------------------------------------------
        # Global search with Differential Evolution (if time permits)
        # -----------------------------------------------------------------
        remaining = time_limit - time.time()
        if remaining > 20.0 and n <= 200:   # keep the expensive stage modest
            lam_de = 200.0
            bounds = [(0.0, 1.0)] * n
            # DE parameters – modest population & generations to stay fast
            de_maxiter = 15
            de_popsize = 5
            # callback to abort when we run out of time
            def de_callback(xk, convergence):
                return time.time() > time_limit

            de_res = opt.differential_evolution(
                obj_de_penalty,
                bounds,
                args=(target_sum, lam_de),
                strategy="best1bin",
                maxiter=de_maxiter,
                popsize=de_popsize,
                polish=False,
                updating="deferred",
                seed=seed,
                callback=de_callback,
                disp=False,
                tol=1e-6,
            )
            if de_res.success and time.time() <= time_limit:
                de_best = de_res.x
                refined = local_refine(de_best, n, target_sum, time_limit)
                if refined is not None:
                    c5ref = compute_c5(refined)
                    if c5ref < best_c5 - 1e-9:
                        best_c5 = c5ref
                        best_h = refined.copy()
                        best_n = n

    # -----------------------------------------------------------------
    # Fallback in the unlikely event that nothing succeeded
    # -----------------------------------------------------------------
    if best_h is None:
        n = 100
        best_h = generate_triangular(n)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final sanity check – enforce exact integral constraint
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
