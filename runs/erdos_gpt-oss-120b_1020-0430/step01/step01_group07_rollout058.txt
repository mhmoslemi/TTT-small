    # Convert to shift from - (n - 1) to (n - 1)

    # compute arrays h_shift_left = shift left (i+shift) maybe

    if shift >=0:

         h_shift_minus = np.concatenate((h[shift:], np.zeros(shift)))  # for i-shift
    else:

         h_shift_plus = np.concatenate((h[s:], np.zeros(s)))  # i+shift
         h_shift_minus = np.concatenate((np.zeros(s), h[:n-s]))  # i-shift

import time, math, random
import numpy as np

def _c5_metric(h):
    """Integral overlap C5 for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _repair(h, target):
    """Clamp to [0,1] and enforce exact sum = target."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # try to distribute the tiny remainder in one place
        for idx in np.random.permutation(h.size):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + diff <= hi:
                h[idx] += diff
                break
    return h


def _interpolate(h, new_n):
    """Linear interpolation of a feasible h onto a finer grid."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    cur = h_new.sum()
    if cur != 0:
        h_new *= target / cur
    return _repair(h_new, target)


def _anneal(start_h, max_seconds):
    """Simulated‑annealing with moves biased to the current worst shift."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    target = n / 2.0

    # initial metric
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_c = np.max(corr) * dx
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    T0, T_end = 0.2, 1e-6          # temperature schedule
    max_step0 = 0.2                # initial step bound
    p_target = 0.7                 # probability to use a shift‑biased move

    while time.time() - start < max_seconds:
        # temperature and step size decay (log‑linear)
        prog = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** prog
        max_step = max_step0 * (T / T0)

        # correlation and the shift giving the current maximum C5
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # choose a pair of indices
        if random.random() < p_target and shift != 0:
            # indices that actually overlap for this shift
            if shift > 0:
                i_low, i_high = 0, n - shift - 1
            else:
                i_low, i_high = -shift, n - 1
            if i_low <= i_high:
                i = random.randint(i_low, i_high)
                j = i + shift
            else:
                i, j = random.sample(range(n), 2)
        else:
            i, j = random.sample(range(n), 2)

        hi, hj = h[i], h[j]
        # feasible signed delta for the pair
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # cap the move size to keep steps modest
        low_cap = max(low, -max_step)
        high_cap = min(high, max_step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)

        # apply the move
        h[i] = hi + delta
        h[j] = hj - delta

        # new metric
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_c = np.max(new_corr) * dx
        dC = new_c - cur_c

        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

    return best_h, best_c
