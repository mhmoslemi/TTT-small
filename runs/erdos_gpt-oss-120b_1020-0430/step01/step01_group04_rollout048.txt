def gradient_max_corr(h):
    n = len(h)
    conv = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(conv)
    k = idx - (n-1)
    # compute forward term: shift = +k
    if k >= 0:
        forward = np.concatenate([np.zeros(k), h[:n-k]])  # length n
        backward = np.concatenate([h[k:], np.zeros(k)])
    else:
        k_abs = -k
        forward = np.concatenate([h[k_abs:], np.zeros(k_abs)])
        backward = np.concatenate([np.zeros(k_abs), h[:n-k_abs]])
    grad = -(forward + backward)
    return grad, conv[idx]*dx

def project(v, target, eps=1e-12):
    low = -np.max(v)  # ensure all v+low <=0
    high = 1 - np.min(v)  # ensure all v+high >=1
    for _ in range(60):
        mid = (low+high)/2
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if s > target:
            high = mid
        else:
            low = mid
    w = np.clip(v + (low+high)/2,0,1)
    # adjust in case of rounding errors
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # add diff uniformly to non-saturated entries
        free = (w>0+1e-12) & (w<1-1e-12)
        num = np.sum(free)
        if num>0:
            w[free] += diff/num
            w = np.clip(w,0,1)
    return w

def project_to_feasible(v, target):
    # v may be outside [0,1] bounds; we need to find lambda s.t.
    # w_i = min(1, max(0, v_i + lambda)) has sum target.
    # Since sum is monotonic increasing in lambda, we can binary search.
    lo = -1.0
    hi = 1.0
    # make sure lo yields sum <= target; hi yields sum >= target.
    for _ in range(50):
        mid = (lo + hi) / 2
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo+hi)/2, 0.0, 1.0)
    # after binary search may be slightly off due to floating; adjust uniform distribution
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute diff onto non-bound elements
        free = (w > 0) & (w < 1)
        cnt = free.sum()
        if cnt > 0:
            w[free] += diff / cnt
            w = np.clip(w, 0, 1)
    return w

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * (2.0 / n))


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def project_to_feasible(v: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """
    Project vector v onto {0 ≤ w ≤ 1, sum(w) = target}.
    Uses a simple bisection on a shift λ so that w_i = clip(v_i+λ,0,1).
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    # tiny correction to eliminate numerical drift
    diff = target - w.sum()
    if abs(diff) > eps:
        free = (w > 0.0) & (w < 1.0)
        cnt = free.sum()
        if cnt > 0:
            w[free] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w


def generate_triangular(n: int) -> np.ndarray:
    """Classic symmetric triangle (height 1, base 2), exact integral = 1."""
    x = np.arange(n) * (2.0 / n) + (1.0 / n)  # mid‑point sample
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    return np.clip(tri, 0.0, 1.0)


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Continuous random feasible vector (values in [0,1], sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    return project_to_feasible(a, n / 2.0)


def random_binary_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary feasible vector (exactly n/2 entries equal to 1)."""
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new resolution and renormalisation."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    return np.clip(up, 0.0, 1.0)


def generate_trapezoid(n: int, L: float) -> np.ndarray:
    """
    Symmetric trapezoid:
       plateau of width L (0≤L≤1) at height 1,
       linear slopes of length a = 1-L on each side.
    Integral is forced to 1.
    """
    L = np.clip(L, 0.0, 1.0)
    a = 1.0 - L
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    d = np.abs(x - 1.0)
    h = np.where(d <= L / 2,
                 1.0,
                 np.where(d <= L / 2 + a,
                          1.0 - (d - L / 2) / a,
                          0.0))
    target = n / 2.0
    h = h * (target / h.sum())
    return np.clip(h, 0.0, 1.0)


def generate_power_shape(n: int, p: float) -> np.ndarray:
    """
    Shape h(x)=max(0,1-(|x-1|/a)^p) with a chosen for integral =1.
    p≥1 ensures support ≤2.
    """
    p = max(1.0, float(p))
    a = (p + 1.0) / (2.0 * p)
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    d = np.abs(x - 1.0)
    h = np.maximum(0.0, 1.0 - (d / a) ** p)
    target = n / 2.0
    h = h * (target / h.sum())
    return np.clip(h, 0.0, 1.0)


# ------------------------------------------------------------
# Gradient‑descent optimiser on the exact max‑overlap objective
# ------------------------------------------------------------
def max_overlap_and_shift(h: np.ndarray):
    """Return current C5 and the shift (as integer lag) attaining it."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    k = idx - (n - 1)          # signed shift
    c5 = conv[idx] * (2.0 / n) # undo the missing dx factor
    return c5, k


def shifted_arrays(h: np.ndarray, k: int):
    """Return forward = h shifted by +k, backward = h shifted by -k (zero‑padded)."""
    n = h.shape[0]
    fwd = np.zeros_like(h)
    bwd = np.zeros_like(h)
    if k >= 0:
        fwd[k:] = h[: n - k]
        bwd[: n - k] = h[k:]
    else:
        kk = -k
        fwd[: n - kk] = h[kk:]
        bwd[kk:] = h[: n - kk]
    return fwd, bwd


def gd_optimize(start: np.ndarray, target_sum: float, deadline: float):
    """
    Simple projected gradient descent on the exact max‑overlap.
    Returns (h_opt, c5_opt). Exits early if time runs out.
    """
    h = start.copy()
    n = h.shape[0]
    c5 = compute_c5(h)
    lr = 0.1                      # initial step size
    it = 0
    while time.time() < deadline and it < 2000:
        it += 1
        c5, k = max_overlap_and_shift(h)
        fwd, bwd = shifted_arrays(h, k)
        delta = fwd + bwd                # gradient of C5 is -(fwd+bwd)
        # Try a few step halving attempts
        improved = False
        lr_try = lr
        for _ in range(6):
            h_new = h + lr_try * delta
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-9:
                h, c5 = h_new, c5_new
                lr = lr_try * 1.05
                improved = True
                break
            lr_try *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                break
    return h, c5
