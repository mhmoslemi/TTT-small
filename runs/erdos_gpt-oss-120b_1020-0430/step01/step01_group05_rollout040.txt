def smooth_gradient(h, alpha):
    n = len(h)
    # compute corr (full)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # weights
    # use log-sum-exp stable
    maxc = np.max(alpha * corr)
    exp_vals = np.exp(alpha * corr - maxc)
    w = exp_vals / exp_vals.sum()
    g = np.zeros_like(h)
    # For shift indices:
    for shift in range(-n+1, n):
        weight = w[shift + n - 1]
        if weight == 0.0:
            continue
        if shift >= 0:
            s = shift
            # first term
            g[:n - s] += weight * (1.0 - h[s:])
            # second term
            g[s:] -= weight * h[:n - s]
        else:
            d = -shift
            g[d:] += weight * (1.0 - h[:n - d])
            g[:n - d] -= weight * h[d:]
    return g

import time, random, math
import numpy as np

# -------------------------------------------------
# Helper utilities
# -------------------------------------------------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n with Σx = target using bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised objective C₅ = max_k ∫h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_gradient(h: np.ndarray, alpha: float = 20.0) -> np.ndarray:
    """
    Gradient of the smooth (log‑sum‑exp) approximation of max_k corr_k.
    Returns ∇_h smooth_max(corr).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    # stable softmax weights
    a_corr = alpha * corr
    a_max = np.max(a_corr)
    exp_vals = np.exp(a_corr - a_max)
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)
    g = np.zeros_like(h)

    # positive shifts (including 0)
    for s in range(0, n):
        weight = w[n - 1 + s]
        if weight == 0.0:
            continue
        # (1 - h_{i+s}) term
        g[: n - s] += weight * (1.0 - h[s:])
        # -h_{i-s} term (skip s==0 to avoid double‑subtract)
        if s > 0:
            g[s:] -= weight * h[: n - s]

    # negative shifts
    for d in range(1, n):
        weight = w[n - 1 - d]
        if weight == 0.0:
            continue
        # (1 - h_{i-d}) term
        g[d:] += weight * (1.0 - h[: n - d])
        # -h_{i+d} term
        g[: n - d] -= weight * h[d:]

    return g


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape (peak 1 at centre, linear to 0 at edges)."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2]."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)
