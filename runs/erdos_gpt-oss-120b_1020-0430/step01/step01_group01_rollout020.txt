import time
import numpy as np

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project vector v onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap for a given shift (the shift that attains the max)."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return corr


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for the current discretisation."""
    return float(np.max(_corr_fft(h)) * dx)

    n_base = len(base) if base is not None else 0
    n = max(300, n_base)                 # use a finer grid than the parent
    target_sum = n / 2.0                  # sum of samples (integral = 1)
    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # Upscale parent solution if present, otherwise start random
    # ----------------------------------------------------------------------
    def _upscale(v):
        if len(v) == n:
            return v.copy()
        x_old = np.linspace(0.0, 2.0, len(v), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, v)

    if base is not None:
        h = _upscale(base)
        h = _project_onto_capped_simplex(h, target_sum)
    else:
        v = rng.rand(n)
        h = _project_onto_capped_simplex(v, target_sum)

    cur_c5 = _compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # Simulated annealing on the feasible simplex (pairwise transfers)
    # ----------------------------------------------------------------------
    T0 = 0.02                     # initial temperature
    cooling = 0.9998              # exponential cooling factor
    T = T0
    it = 0
    check_interval = 1500

    while True:
        # time guard
        if time.time() - start > budget_s * 0.94:
            break

        # pick two distinct indices
        i, j = rng.randint(n, size=2)
        while i == j:
            j = rng.randint(n)

        # feasible delta range (transfer from i to j)
        max_pos = min(h[i], 1.0 - h[j])          # max we can move i -> j
        max_neg = min(h[j], 1.0 - h[i])          # max we can move j -> i
        lo, hi = -max_neg, max_pos
        if hi <= lo:
            continue
        delta = rng.uniform(lo, hi)              # random feasible transfer
        if delta == 0.0:
            continue

        # propose move
        h[i] -= delta
        h[j] += delta

        new_c5 = _compute_c5(h, dx)
        diff = new_c5 - cur_c5

        # Metropolis acceptance
        if diff <= 0.0 or rng.rand() < np.exp(-diff / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta

        # temperature update
        T *= cooling
        it += 1

        # occasional full‑gradient refinement (cheap, every few hundred steps)
        if it % check_interval == 0:
            # compute correlation array once (used for gradient)
            corr = np.correlate(h, 1.0 - h, mode='full')
            shift = int(np.argmax(corr)) - (n - 1)
            g = _gradient_for_shift(h, shift)
            step = 0.2
            h = h - step * g
            h = _project_onto_capped_simplex(h, target_sum)
            cur_c5 = _compute_c5(h, dx)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # ----------------------------------------------------------------------
    # Final local gradient descent from the best solution found
    # ----------------------------------------------------------------------
    h = best_h.copy()
    step = 0.5
    for _ in range(300):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target_sum)
        step *= 0.995
    best_h = h
    best_c5 = _compute_c5(best_h, dx)

    # final safety projection (ensures feasibility within numeric noise)
    best_h = _project_onto_capped_simplex(best_h, target_sum)

    return (best_h, float(best_c5), n)
