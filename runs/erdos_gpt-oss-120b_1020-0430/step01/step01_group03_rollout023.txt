def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    # random transfers
    for _ in range(n*5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i==j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    # ensure sum exactly n/2
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters=2000, start_step=0.5, p_targeted=0.8):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    step = start_step
    for it in range(max_iters):
        # Determine move type
        if rng.random() < p_targeted:
            # targeted based on worst shift
            corr = np.correlate(h, 1 - h, mode='full')
            corr_pos = corr[n-1:]  # nonnegative shifts

            # collect all max shift indices
            ks = np.flatnonzero(corr_pos == max_corr)
            k = rng.choice(ks)  # shift steps
            # contributions for this shift
            n_shift = n - k
            if n_shift <= 0:

                    cur_c = new_c

# SLSQP optimizer using softmax surrogate (same as in previous code)
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

    return np.repeat(h, factor)

CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters=2000, start_step=0.5, p_targeted=0.8):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    step = start_step
    for it in range(max_iters):
        if rng.random() < p_targeted:
            # targeted move using worst shift
            corr = np.correlate(h, 1 - h, mode='full')
            corr_pos = corr[n-1:]  # shifts >=0

            ks = np.flatnonzero(corr_pos == max_corr)
            k = rng.choice(ks)
            n_shift = n - k
            if n_shift <= 0:
                new_h = perturb_h(h, rng, scale=step)
                new_c = compute_c5(new_h)
            else:
                h_i = h[:n_shift]
                h_j = h[k:]
                contrib = h_i * (1.0 - h_j)
                if np.sum(contrib) > 0:
                    prob = contrib / np.sum(contrib)
                    i = rng.choice(n_shift, p=prob)
                else:
                    i = rng.integers(0, n_shift)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    new_h = perturb_h(h, rng, scale=step)
                    new_c = compute_c5(new_h)
                else:
                    eps = (rng.random() - 0.5) * 2.0 * max_eps * step
                    new_h = h.copy()
                    new_h[i] -= eps
                    new_h[j] += eps
                    new_c = compute_c5(new_h)
        else:
            new_h = perturb_h(h, rng, scale=step)
            new_c = compute_c5(new_h)
        if new_c <= cur_c + 1e-12:
            h = new_h
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = new_h.copy()
        # step decay
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

        h0,
        method="SLSQP",

        constraints=cons,

    )

                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref
        except Exception:
            continue

    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    final_c = compute_c5(best_h)
    return best_h, final_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = base_h.size
        base_h = np.clip(base_h, 0.0, 1.0)
        target = base_n / 2.0
        diff = target - base_h.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, base_n)
                if diff > 0:
                    add = min(diff, 1.0 - base_h[i])
                else:
                    add = max(diff, -base_h[i])
                base_h[i] += add
                diff = target - base_h.sum()
    else:
        base_n = 120
        base_h = np.full(base_n, 0.5)

    overall_best_h = base_h.copy()
    overall_best_c = compute_c5(overall_best_h)
    overall_best_n = base_n

    candidates = [base_n]
    if base_n * 2 <= 360:
        candidates.append(base_n * 2)
    if base_n * 3 <= 360 and (base_n * 3) not in candidates:
        candidates.append(base_n * 3)

    for n in candidates:
        now = time.time()
        remaining = budget_s - (now - start)
        if remaining < 5.0:
            break
        if n == base_n:
            init_h = base_h.copy()
        else:
            factor = n // base_n
            init_h = upsample_repeat(base_h, factor)
        deadline = now + remaining * 0.5
        h_opt, c_opt = search_n(n, init_h, rng, deadline)
        if c_opt < overall_best_c:
            overall_best_c = c_opt
            overall_best_h = h_opt.copy()
            overall_best_n = n

    overall_best_h = np.clip(overall_best_h, 0.0, 1.0)
    target = overall_best_n / 2.0
    diff = target - overall_best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, overall_best_n)
            if diff > 0:
                add = min(diff, 1.0 - overall_best_h[i])
            else:
                add = max(diff, -overall_best_h[i])
            overall_best_h[i] += add
            diff = target - overall_best_h.sum()
    final_c = compute_c5(overall_best_h)
    return overall_best_h.tolist(), float(final_c), int(overall_best_n)
