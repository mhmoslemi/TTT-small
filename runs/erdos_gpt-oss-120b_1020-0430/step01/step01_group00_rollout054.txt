def simulated_anneal_binary(b, time_left, n):
    # returns best pattern and best C5
    max_iter = 50000  # may be limited by time
    cur = b.copy()
    cur_c5 = compute_c5(cur.astype(float))
    best = cur.copy()
    best_c5 = cur_c5
    T0 = 0.01
    T_end = 1e-5
    # Precompute indices of 1 and 0 positions
    for it in range(max_iter):
        if time_left() < 2: break
        T = T0 * (T_end / T0) ** (it / max_iter)
        # pick random indices
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        new_c5 = compute_c5(cur.astype(float))
        delta = new_c5 - cur_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            # reject: swap back
            cur[i] = 1
            cur[j] = 0
        # Occasionally shuffle zeros and ones eq to update index view, but we recompute each loop; okay.
    return best, best_c5

def local_refine_continuous(h, time_left):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_val = cur_c5
    step = 0.1
    while time_left()>2 and step>1e-4:
        improved=False
        for _ in range(1000):
            i, j = np.random.choice(n,2,replace=False)
            delta = step * (np.random.rand()*2-1)
            # adjust i by -delta, j by +delta, respecting bounds
            if cur[i] - delta <0 or cur[i] - delta >1: continue
            if cur[j] + delta <0 or cur[j] + delta >1: continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            new_c5 = compute_c5(new)
            if new_c5 < cur_c5:
                cur = new
                cur_c5 = new_c5
                improved=True
                if cur_c5 < best_val:
                    best = cur.copy()
                    best_val = cur_c5
        if not improved:
            step *= 0.5
    return best, best_val
```

def _project_to_feasible(h, n, tol=1e-12, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to reduce sum
            free = np.where(h > 0)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            h[free] -= delta
            h = np.clip(h, 0.0, 1.0)
        else:
            # need to increase sum
            free = np.where(h < 1)[0]
            if free.size == 0:
                break
            delta = diff / free.size  # diff negative, delta negative => add magnitude
            h[free] -= delta  # subtract negative => add
            h = np.clip(h, 0.0, 1.0)
    # final small correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

            if new_val < cur_val:
                cur, cur_val = new, new_val
                improved = True
                if cur_val < best_val:
                    best, best_val = cur.copy(), cur_val
        if not improved:
            step *= 0.5
    return best, best_val


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1 that yields
    a low value of C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial construction, if provided by the harness
    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        pass

    best_h = None
    best_c5 = float("inf")
    best_n = None

    if init_h is not None:
        n0 = init_h.size
        init_h = _project_to_feasible(init_h, n0)
        c5 = _c5_fft(init_h)
        best_h, best_c5, best_n = init_h, c5, n0

    # candidate grid sizes (favor the supplied size, then a few powers of two)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (128, 256):
        if n not in n_candidates:
            n_candidates.append(n)
    for n in (160, 200, 224):
        if n not in n_candidates:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    # main optimisation loop
    for n in n_candidates:
        if time_left() < 2:
            break

        # build a list of starting binary patterns
        patterns = []
        # Thue–Morse pattern for powers of two
        if (n & (n - 1)) == 0:
            tm = np.array([i.bit_count() & 1 for i in range(n)], dtype=np.int8)
            if tm.sum() == n // 2:
                patterns.append(tm)
        # a few random balanced patterns
        for _ in range(2):
            pat = np.zeros(n, dtype=np.int8)
            pat[: n // 2] = 1
            np.random.shuffle(pat)
            patterns.append(pat)

        for base_bin in patterns:
            if time_left() < 1:
                break
            # binary simulated annealing
            best_bin, _ = _simulated_anneal_binary(base_bin, time_left, n)
            # continuous refinement
            h_float = best_bin.astype(float)
            h_ref, _ = _local_refine_continuous(h_float, time_left)
            h_ref = _project_to_feasible(h_ref, n)
            c5_val = _c5_fft(h_ref)

            if c5_val < best_c5:
                best_h, best_c5, best_n = h_ref, c5_val, n
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # fallback – should never trigger
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_fft(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
