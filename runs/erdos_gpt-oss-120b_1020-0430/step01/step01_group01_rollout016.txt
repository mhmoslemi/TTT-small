def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    # pick n_points = 400
    n = 400
    target_sum = n/2
    dx = 2.0/n

    # Get initial h: if initial_h_values exist, upscale to n using interp; else random.
    try: base = np.array(initial_h_values, dtype=float)
    except Exception: base = None
    if base is not None:
        # upscale
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h0_initial = np.interp(x_new, x_old, base)
        h0_initial = _project_onto_capped_simplex(h0_initial, target_sum)
    else:
        h0_initial = _random_feasible(rng, n, target_sum)

    best_h = h0_initial.copy()
    best_val = _compute_c5(best_h, dx)

    # GD function with configurable steps
    def gd_optimize(h_start, max_iter, step0, decay):
        h = h_start.copy()
        step = step0
        for i in range(max_iter):
            # compute correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            grad = _gradient_for_shift(h, shift)
            h = h - step * grad
            # projection
            h = _project_onto_capped_simplex(h, target_sum)
            step *= decay
        return h

    # Main loop
    iter_count = 0
    while time.time() - start < budget_s * 0.9:
        # choose new starting point: perturb best or random
        if best_h is not None and rng.rand() < 0.3:
            # perturb best
            h_start = best_h + rng.normal(scale=0.01, size=n)
        else:
            h_start = _random_feasible(rng, n, target_sum)

        # GD optimization

import time
import numpy as np

# -------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# -------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# -------------------------------------------------------------
# Random feasible point (0<=h<=1, sum = target)
# -------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# -------------------------------------------------------------
# C5 evaluation (integral form)
# -------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# -------------------------------------------------------------
# Gradient of the correlation for a given shift
# -------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# -------------------------------------------------------------
# Projected sub‑gradient descent on the max‑correlation objective
# -------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float, max_iter: int,
                 step0: float, decay: float, dx: float, time_limit: float,
                 start_time: float) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        # early exit if out of budget
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

    dx = 2.0 / n

    # ---------------------------------------------------------

    # ---------------------------------------------------------
    try:
