def _refine_slsqp(h0, target, alpha, max_iter):
    n = h0.shape[0]
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        # smooth max
        max_corr = np.max(corr)
        shifted = corr - max_corr
        return max_corr + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    cons = ({'type':'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto the capped simplex ----------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ---------- random feasible vectors ----------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _random_binary_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """binary 0/1 vector with exact sum = target (target should be integer)."""
    k = int(round(target))
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:k]] = 1.0
    return h


# ---------- evaluation ----------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ---------- gradient for a single worst shift ----------
def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    if shift >= 0:
        s = shift
        g = np.empty_like(h)
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g = np.empty_like(h)
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 800, step0: float = 0.5, decay: float = 0.995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_worst_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ---------- smoothed gradient via softmax ----------
def _soft_weights(corr: np.ndarray, alpha: float) -> np.ndarray:
    m = np.max(corr)
    exp = np.exp(alpha * (corr - m))
    return exp / exp.sum()


def _smooth_gradient(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max objective."""
    n = h.shape[0]
    # w length = 2n-1
    g1_full = np.correlate(w, 1.0 - h, mode='full')
    g1 = g1_full[n - 1: 2 * n - 1]
    w_rev = w[::-1]
    g2_full = np.correlate(w_rev, h, mode='full')
    g2 = g2_full[n - 1: 2 * n - 1]
    return g1 - g2


def _smooth_gd_optimize(h0: np.ndarray, target: float,
                        max_iter: int = 4000,
                        step0: float = 0.8,
                        alpha_start: float = 2.0,
                        alpha_end: float = 80.0) -> np.ndarray:
    h = h0.copy()
    n = h.shape[0]
    for t in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        w = _soft_weights(corr, alpha)
        grad = _smooth_gradient(h, w)
        step = step0 / np.sqrt(t + 1.0)
        h -= step * grad
        h = _project_onto_capped_simplex(h, target)
    return h


# ---------- SLSQP refinement (smooth max) ----------
def _refine_slsqp(h0: np.ndarray, target: float,
                  alpha: float = 100.0, max_iter: int = 600) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        shifted = corr - m
        return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
