def gradient_of_lag(h, lag):
    n = h.size
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if lag >= 0:
        # term1 defined for i< n-lag
        term1[:n-lag] = 1 - h[lag:]
        # term2 defined for i>=lag
        term2[lag:] = h[:n-lag]
    else:
        # lag negative: shift left
        # term1 defined for i < n+lag (because i+lag < n)
        # since lag negative, n+lag < n
        # Example: lag = -k where k>0
        # i+lag in [0,n-1] => i >= -lag (i>=k)?? Actually i+(-k) >=0 => i>=k. So term1 defined for i>=k.
        # Let's compute using symmetrical formula: treat -lag as positive shift in opposite direction.
        k = -lag
        # term1 for i >= k
        term1[k:] = 1 - h[:n-k]
        # term2 for i< n-k
        term2[:n-k] = h[k:]
    # gradient g = term1 - term2
    return term1 - term2

    g = np.zeros(n)
    # term1:
    i_start = max(0, -lag) # i such that i+lag >=0

    # For those i, term1 = 1 - h[i+lag]

import time
import math
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Helper functions – feasibility projection, evaluation, etc.
# ------------------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # Tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given discretised h (integral constraint already satisfied)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ------------------------------------------------------------
# Gradient / subgradient utilities
# ------------------------------------------------------------

def _max_corr_and_lag(h: np.ndarray):
    """Return (max correlation value, lag) for h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_corr = float(corr[idx])
    lag = idx - (n - 1)
    return max_corr, lag


def _gradient_from_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Subgradient of the active correlation term at given lag."""
    n = h.size
    g = np.zeros(n, dtype=float)

    # term (1 - h[i+lag])
    i_start = max(0, -lag)
    i_end = min(n, n - lag)
    if i_start < i_end:
        g[i_start:i_end] = 1.0 - h[i_start + lag:i_end + lag]

    # term -h[i-lag]
    i_start2 = max(0, lag)
    i_end2 = min(n, n + lag)
    if i_start2 < i_end2:
        g[i_start2:i_end2] -= h[i_start2 - lag:i_end2 - lag]

    return g


def _subgradient_optimize(start_h: np.ndarray,
                          max_iter: int = 2000,
                          alpha0: float = 0.3,
                          time_left=lambda: 1e9):
    """Projected subgradient descent on the max‑correlation objective."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    for t in range(max_iter):
        if time_left() < 2.0:
            break
        max_corr, lag = _max_corr_and_lag(h)
        g = _gradient_from_lag(h, lag)
        step = alpha0 / math.sqrt(t + 1.0)
        h_new = h - step * g
        h_new = _project_to_feasible(h_new, n)
        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 250):
    """SLSQP refinement starting from a feasible point."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP",
                            bounds=bounds, constraints=constraints,
                            options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound using a projected subgradient method
    with occasional SLSQP refinements.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – list of floats representing the step function on the
                    chosen grid,
        c5_bound – the achieved upper bound,
        n_points – number of discretisation points used.
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Load any provided initial construction.
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # Target bound for early termination.
    target_bound = 0.3808

    best_h = None
    best_c5 = math.inf
    best_n = None

    # Candidate grid sizes (prefer the supplied size, then upscale).
    cand_ns = []
    if init_h is not None:
        cand_ns.append(init_h.size)
    for mul in (1, 2, 4):
        if init_h is not None:
            n = init_h.size * mul
        else:
            n = 128 * mul
        if n <= 512:
            cand_ns.append(int(n))
    # Some handcrafted even sizes.
    cand_ns.extend([80, 96, 128, 160, 192, 256, 320, 384, 512])
    cand_ns = sorted(set(cand_ns))

    for n in cand_ns:
        if time_left() < 5.0:
            break

        # Build a base h for this grid size.
        if init_h is not None:
            if init_h.size != n:
                base_h = _upscale_h(init_h, n)
            else:
                base_h = init_h.copy()
        else:
            base_h = np.full(n, 0.5)

        base_h = _project_to_feasible(base_h, n)

        # Determine iteration budget depending on n.
        max_iters = 3000 if n <= 256 else 2500
        alpha0 = 0.35

        # First deterministic run.
        h_opt, c5_opt = _subgradient_optimize(base_h,
                                               max_iter=max_iters,
                                               alpha0=alpha0,
                                               time_left=time_left)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < target_bound:
                break

        # Random restarts.
        restarts = 5
        sigma = 0.22
        for _ in range(restarts):
            if time_left() < 5.0:
                break
            start_h = base_h + sigma * np.random.randn(n)
            start_h = _project_to_feasible(start_h, n)
            h_opt, c5_opt = _subgradient_optimize(start_h,
                                                   max_iter=max_iters,
                                                   alpha0=alpha0,
                                                   time_left=time_left)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                if best_c5 < target_bound:
                    break
            sigma *= 0.85  # gradually reduce perturbation size

        if best_c5 < target_bound:
            break

    # Optional final refinement with SLSQP if time permits.
    if best_h is not None and time_left() > 10.0:
        try:
            refined_h, refined_c5 = _solve_h(best_h, max_iter=500)
            if refined_c5 < best_c5:
                best_h, best_c5 = refined_h, refined_c5
                best_n = best_h.size
        except Exception:
            pass

    # Fallback – uniform function if nothing succeeded.
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
