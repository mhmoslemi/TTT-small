    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    c = conv  # shape 2n-1
    expo = np.exp(alpha * c)
    sum_expo = expo.sum()
    w = expo / sum_expo  # shape 2n-1
    
    g = np.ones_like(h)  # constant term = sum_k w_k * 1 = 1
    # subtract contributions
    # For each shift k:
    for idx, wk in enumerate(w):
        k = idx - (n - 1)
        if wk == 0: continue
        if k >= 0:
            # h[i+k] term: i from 0..n-1-k
            g[:n - k] -= wk * h[k:]  # h[i+k] for i from 0..n-1-k
            # h[i-k] term: i from k..n-1 -> h[i-k] = h[:n - k]
            g[k:] -= wk * h[:n - k]

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5_and_grad(h: np.ndarray, alpha: float):
    """Softmax approximation of max overlap and its gradient."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")          # length 2n-1
    lse = logsumexp(alpha * conv)                        # log-sum-exp
    f = lse / alpha                                      # softmax value

    # normalized weights w_k = exp(alpha*c_k) / sumexp
    w = np.exp(alpha * conv - lse)

    g = np.ones_like(h)  # from the constant term Σ w_k * 1 = 1
    off = n - 1
    # subtract Σ w_k * (h[i+k] + h[i-k])
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - off
        if k >= 0:
            # right shift
            g[: n - k] -= wk * h[k:]
            # left shift
            g[k:] -= wk * h[: n - k]
        else:
            d = -k
            g[d:] -= wk * h[: n - d]
            g[: n - d] -= wk * h[d:]
    return f, g


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1], sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, keeping exact integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def generate_random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones (n should be even)."""
    half = n // 2
    arr = np.zeros(n)
    arr[:half] = 1.0
    rng.shuffle(arr)
    return arr

# Optimisation helpers

def _proj_sum_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """Simple projection onto [0,1] and the affine sum constraint."""
    diff = target - h.sum()
    h = h + diff / h.shape[0]
    s = h.sum()
    if s != 0:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)


def optimise_with_alpha(start: np.ndarray, alpha: float, target: float,
                        deadline: float, rng: np.random.Generator):
    """L‑BFGS‑B optimisation of the softmax objective with quadratic penalty."""
    n = start.shape[0]
    lam = 500.0  # penalty weight for the sum constraint
    bounds = opt.Bounds(0.0, 1.0)

    def fun_and_grad(x):
        f, g = softmax_c5_and_grad(x, alpha)
        err = x.sum() - target
        f_total = f + lam * err * err
        g_total = g + 2.0 * lam * err * np.ones_like(x)
        return f_total, g_total

    # limit iterations to keep within the global budget
    maxiter = 200
    res = opt.minimize(fun_and_grad, start, jac=True, method="L-BFGS-B",
                       bounds=bounds, options={"maxiter": maxiter, "ftol": 1e-12, "disp": False})
    if time.time() > deadline:
        return None
    return res


def hill_climb_random(h: np.ndarray, target: float, max_steps: int,
                      step_scale: float, rng: np.random.Generator,
                      deadline: float):
    """Simple pairwise‑swap hill climbing on the exact C5."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    n = cur.shape[0]
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        low = max(-step_scale, -cur[i], cur[j] - 1.0)
        high = min(step_scale, 1.0 - cur[i], cur[j])
        if low >= high:
            continue
        delta = rng.uniform(low, high)
        nxt = cur.copy()
        nxt[i] += delta
        nxt[j] -= delta
        nxt_c5 = compute_c5(nxt)
        if nxt_c5 < cur_c5 - 1e-12:
            cur, cur_c5 = nxt, nxt_c5
    return cur, cur_c5


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even lengths are convenient)
    candidate_ns = [80, 100, 120, 150, 180, 200, 240]

    # parent construction, if supplied
    parent_h = globals().get("initial_h_values")
    if parent_h is not None:
        parent_h = np.asarray(parent_h, dtype=float)

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # ------------------------------------------------------------------
        # build a diverse pool of starting points
        # ------------------------------------------------------------------
        starts = []

        if parent_h is not None:
            if parent_h.shape[0] == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        starts.append(generate_triangular(n))
        starts.append(generate_random_binary(n, rng))
        for _ in range(3):
            starts.append(random_start(n, rng))
        # jittered triangular
        for _ in range(2):
            tri = generate_triangular(n)
            pert = tri + rng.uniform(-0.2, 0.2, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = pert * (target / pert.sum())
            starts.append(pert)

        # ------------------------------------------------------------------
        # optimise each start
        # ------------------------------------------------------------------
        for s in starts:
            if time.time() > deadline:
                break
            cur = _proj_sum_bounds(s, target)

            # staged softmax optimisation with increasing sharpness
            for alpha in (5.0, 30.0, 120.0):
                if time.time() > deadline:
                    break
                res = optimise_with_alpha(cur, alpha, target, deadline, rng)
                if res is None or not res.success:
                    continue
                cur = np.clip(res.x, 0.0, 1.0)
                cur = _proj_sum_bounds(cur, target)

            # final hill‑climbing with the true objective
            if time.time() < deadline:
                cur, cur_c5 = hill_climb_random(cur, target, max_steps=2000,
                                                step_scale=0.05, rng=rng,
                                                deadline=deadline)
            else:
                cur_c5 = compute_c5(cur)

            # keep global best
            if cur_c5 < best_c5 - 1e-9:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n

    # ------------------------------------------------------------------
    # fallback if nothing succeeded
    # ------------------------------------------------------------------
    if best_h is None:
        n = 100
        if parent_h is not None and parent_h.shape[0] == n:
            best_h = parent_h.copy()
        else:
            best_h = generate_triangular(n)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final projection to guarantee the integral exactly
    target = best_n / 2.0
    best_h = _proj_sum_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
