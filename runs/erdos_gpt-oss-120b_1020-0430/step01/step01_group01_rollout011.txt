    pbest_positions = positions.copy()

    gbest_idx = int(np.argmin(pbest_values))

        positions = positions + velocities

                    gbest_position = positions[i].copy()
                    if val < best_val:

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto {0 <= x_i <= 1, sum x_i = target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    return _project_onto_capped_simplex(rng.rand(n), target)


# ----------------------------------------------------------------------
# Objective evaluation
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Gradient for the maximizing shift (subgradient of the max)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s:] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Simple projected gradient descent on the max‑correlation objective
# ----------------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 2000, step0: float = 0.2,
                 decay: float = 0.9995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# Smooth approximation of max for SLSQP refinement
# ----------------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 300) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9,
                            'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
