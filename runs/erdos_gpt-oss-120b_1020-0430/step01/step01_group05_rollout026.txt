def de_optimize(m, target_sum, n, blocksize, max_gens=500, popsize=50, F=0.8, Cr=0.9, time_limit):
   start = time.time()
   pop = np.random.rand(popsize, m)
   for i in range(popsize):
       pop[i] = project_onto_bounds_and_sum(pop[i], target_sum)
   fitness = np.array([compute_c5_expand(pop[i]) for i in range(popsize)])
   best_idx = np.argmin(fitness)
   best = pop[best_idx].copy()
   best_f = fitness[best_idx]

   gen = 0
   while gen < max_gens and time.time() - start < time_limit:
       for i in range(popsize):
           idxs = list(range(popsize))
           idxs.remove(i)
           a,b,c = np.random.choice(idxs, 3, replace=False)
           mutant = pop[a] + F * (pop[b] - pop[c])
           # clip before projection to keep them within bounds somewhat
           mutant = np.clip(mutant, 0.0, 1.0)
           mutant = project_onto_bounds_and_sum(mutant, target_sum)
           
           crossover_mask = np.random.rand(m) < Cr
           # ensure at least one True
           if not np.any(crossover_mask):
               crossover_mask[np.random.randint(0, m)] = True
           trial = np.where(crossover_mask, mutant, pop[i])
           trial = project_onto_bounds_and_sum(trial, target_sum)
           
           trial_f = compute_c5_expand(trial)
           
           if trial_f <= fitness[i]:
               pop[i] = trial
               fitness[i] = trial_f
               if trial_f < best_f:
                   best_f = trial_f
                   best = trial.copy()
       gen += 1
   return best, best_f
```

import time, random, math
import numpy as np

def project_onto_bounds_and_sum(x, target):
    # Bisection
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - (low + high) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def upsample_h(init, n):
    old_n = len(init)
    if old_n == n:
        return np.array(init, float)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return h

def make_triangle(n):
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    return project_onto_bounds_and_sum(h, target)

def make_sin2(n):
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    target = n / 2.0
    return project_onto_bounds_and_sum(h, target)

import time, random, math
import numpy as np

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto the box [0,1]^n intersected with sum(x)=target via bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function h on the discretised grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def upsample_h(init, n):
    """Linear interpolation of a parent vector to length n."""
    old_n = len(init)
    if old_n == n:
        return np.array(init, float)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, init)


def make_triangle(n):
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n):
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)
