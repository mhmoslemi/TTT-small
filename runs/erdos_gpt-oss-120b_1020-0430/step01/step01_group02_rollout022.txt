def hill_climb(h, steps, rng):
    best_h = h.copy()
    best_c5 = _evaluate_c5(h)
    n = len(h)
    target = n / 2.0
    for t in range(steps):
        # Choose two indices i, j at random
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # pick random direction: move from i to j or j to i
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        # max delta limited by src value and dst capacity
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        # step size scaling with current best maybe
        delta = rng.random() * max_move  # uniform

        # evaluate

    return best_h, best_c5

def gradient_logsumexp(h, alpha):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1 - h, mode='full')
    # compute weights
    max_ar = np.max(alpha * r)
    exp_ = np.exp(alpha * r - max_ar)
    w = exp_ / exp_.sum()
    # compute gradient
    grad = np.zeros_like(h)
    # We can loop over shifts
    shifts = np.arange(-n+1, n)
    for idx, s in enumerate(shifts):
        w_i = w[idx]
        if w_i == 0.0:
            continue
        if s >= 0:
            # h_{k+s} for k from 0..n-1-s ; for k >= n-s, missing
            h_shift_pos = np.concatenate((np.zeros(s), h[:n-s]))
        else:
            h_shift_pos = np.concatenate((h[-s:], np.zeros(-s)))
        term1 = 1 - h_shift_pos
        # h_{k-s}
        if s <= 0:
            h_shift_neg = np.concatenate((np.zeros(-s), h[:n+s]))
        else:
            h_shift_neg = np.concatenate((h[s:], np.zeros(s)))

    ar = alpha * r

    exp_ar = np.exp(ar - max_ar)

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1.
    Uses a simple water‑filling / draining routine.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # give / take mass iteratively
    while abs(diff) > 1e-12:
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
            if not np.any(mask):
                break
            caps = 1.0 - x[mask]
            total_cap = caps.sum()
            if diff <= total_cap:
                add = diff * caps / total_cap
                x[mask] += add
                diff = 0.0
            else:
                x[mask] = 1.0
                diff -= total_cap
        else:
            mask = x > 1e-12
            if not np.any(mask):
                break
            caps = x[mask]
            total_cap = caps.sum()
            if -diff <= total_cap:
                sub = (-diff) * caps / total_cap
                x[mask] -= sub
                diff = 0.0
            else:
                x[mask] = 0.0
                diff += total_cap
    return np.clip(x, 0.0, 1.0)


def _gradient_logsumexp(h: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of ϕ_α(h)=log∑_k exp(α·corr_k) with respect to h.
    Uses correlation to avoid explicit loops.
    """
    n = h.size
    dx = 2.0 / n
    # raw correlations
    corr = np.correlate(h, 1.0 - h, mode="full")
    # soft‑max weights
    ar = alpha * corr
    max_ar = np.max(ar)
    w = np.exp(ar - max_ar)
    w /= w.sum()

    # B = ∑_s w_s (1‑h_{k+s})  (zero‑padded)
    B_full = np.correlate(w, 1.0 - h, mode="full")
    start = n - 1
    B = B_full[start:start + n]

    # D = ∑_s w_s h_{k‑s}
    w_rev = w[::-1]
    D_full = np.correlate(w_rev, h, mode="full")
    D = D_full[start:start + n]

    grad = dx * (B - D)
    return grad


def _hill_climb_steps(h: np.ndarray, steps: int, rng: np.random.Generator) -> tuple:
    """
    Simple pairwise mass‑exchange hill climbing.
    Returns the best (h, C5) encountered in the given number of steps.
    """
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    for _ in range(steps):
        i, j = rng.choice(n, size=2, replace=False)
        # direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move

        cand = cur_h.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_c5 = _evaluate_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
    return best_h, best_c5

                h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
            else:
                h = rng.random(n)
                h = _adjust_sum(h, target)

            # ---- gradient‑descent phase (log‑sum‑exp smoothing) ------------------------
            alpha = 1.0
            gd_steps = 350 if n <= 60 else 450
            for t in range(gd_steps):
                grad = _gradient_logsumexp(h, alpha)
                step = 0.25 / (1.0 + t * 0.02)
                h_new = h - step * grad
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = _adjust_sum(h_new, target)

                # simple improvement test
                if _evaluate_c5(h_new) < _evaluate_c5(h):
                    h = h_new

                # increase smoothing strength periodically
                if (t + 1) % 100 == 0:
                    alpha *= 2.0

            # ---- fine hill‑climbing ----------------------------------------------------
            hc_steps = 6000
            h_opt, c5_opt = _hill_climb_steps(h, hc_steps, rng)

            # ---- record global best ----------------------------------------------------
            if c5_opt < best_overall_c5 - 1e-12:
                best_overall_h = h_opt.copy()
                best_overall_c5 = c5_opt
                best_overall_n = n

            restarts_done += 1

        # early exit if time is nearly exhausted
        if time.time() - start_time > budget_s - 4.0:
            break

    # ------------------------------------------------------------------
    # Final polishing on the overall best candidate
    # ------------------------------------------------------------------
    if best_overall_h is None:
        # fallback – should never happen
        n = 45
        target = n / 2.0
        best_overall_h = _adjust_sum(rng.random(n), target)
        best_overall_c5 = _evaluate_c5(best_overall_h)
        best_overall_n = n
    else:
        # a short extra hill‑climb to squeeze any remaining improvement
        extra_h, extra_c5 = _hill_climb_steps(best_overall_h, 8000, rng)
        if extra_c5 < best_overall_c5:
            best_overall_h, best_overall_c5 = extra_h, extra_c5

    # final feasibility enforcement
    best_overall_h = _adjust_sum(np.clip(best_overall_h, 0.0, 1.0), best_overall_n / 2.0)
    best_overall_c5 = float(_evaluate_c5(best_overall_h))

    return best_overall_h, best_overall_c5, best_overall_n
