def full_corr(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    # FFT of h and reversed (1-h)
    a = np.fft.rfft(h, L)
    b = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(a * b, L)[:2*n-1]
    return conv

import numpy as np
import random
import time
import math

def _proj_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | sum(x)=target, 0<=x<=1}."""
    lo, hi = -1.0, 2.0  # bounds wide enough to force clipping to 0 or 1
    for _ in range(30):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = w.sum() - target
    if abs(diff) > 1e-12:
        idx = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if idx.size:
            w[idx[0]] -= diff
            w[idx[0]] = np.clip(w[idx[0]], 0.0, 1.0)
    return w

def _full_corr(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1‑h) (no reversal), same as np.correlate(h,1-h,'full')."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    a = np.fft.rfft(h, L)
    b = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(a * b, L)[: 2 * n - 1]
    return conv

def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of f_shift(h)=sum h_i(1-h_{i+shift}) w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n, dtype=float)
    if shift >= 0:
        # forward part (i+shift)
        if shift < n:
            g[: n - shift] -= h[shift:]
        # backward part (i-shift)
        if shift > 0:
            g[shift:] -= h[: n - shift]
    else:
        k = -shift
        # forward part (i+shift)   => i-k
        if k < n:
            g[k:] -= h[: n - k]
        # backward part (i-shift)  => i+k
        if k > 0:
            g[: n - k] -= h[k:]
    return g

def _c5_from_corr_max(cmax: float, n: int) -> float:
    """Convert raw correlation max to the C5 value (integral scale)."""
    return cmax * (2.0 / n)
