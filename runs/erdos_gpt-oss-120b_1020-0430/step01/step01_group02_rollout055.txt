import time
import numpy as np
from scipy.optimize import minimize

def _adjust_sum(x, target):
    """Project x onto the hyperplane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple water‑filling redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not np.any(mask):
            break
        free = np.where(mask)[0]
        per = diff / len(free)
        for i in free:
            if diff > 0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _evaluate_c5(h):
    """Return the overlap integral C₅ for a step vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _max_corr(h):
    """Maximum of the raw correlation (integer for binary vectors)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))

def _binary_annealing(n, rng, start, deadline):
    """
    Simulated‑annealing search on binary vectors of length n with exactly n//2 ones.
    Returns (best_vector, best_max_corr).
    """
    target = n // 2
    # start from a random balanced binary vector
    h = np.zeros(n, dtype=np.float64)
    h[:target] = 1.0
    rng.shuffle(h)

    cur = _max_corr(h)
    best = cur
    best_h = h.copy()

    # annealing schedule
    T0, T1 = 5.0, 0.001
    steps_without_improve = 0
    max_stale = 8000

    while time.time() < deadline:
        # temperature based on relative progress
        elapsed = time.time() - start
        total = deadline - start
        prog = elapsed / total if total > 0 else 1.0
        T = T0 * (1 - prog) + T1 * prog

        # pick a random 1 and a random 0 and swap them
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0

        new = _max_corr(h)
        diff = new - cur
        accept = False
        if diff <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-diff / T):
                accept = True

        if accept:
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
                steps_without_improve = 0
            else:
                steps_without_improve += 1
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            steps_without_improve += 1

        if steps_without_improve > max_stale:
            # restart from a fresh random binary configuration
            h = np.zeros(n, dtype=np.float64)
            h[:target] = 1.0
            rng.shuffle(h)
            cur = _max_corr(h)
            steps_without_improve = 0

    # final greedy improvement (only accepting strict improvement)
    improved = True
    while improved:
        improved = False
        for _ in range(200):
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            best_h[i] = 0.0
            best_h[j] = 1.0
            cand = _max_corr(best_h)
            if cand < best:
                best = cand
                improved = True
            else:
                # revert
                best_h[i] = 1.0
                best_h[j] = 0.0
    return best_h, best

def _fractional_hill_climb(h, cur_c5, deadline, rng):
    """
    Simple greedy hill‑climbing that moves a small amount of mass
    between two coordinates while preserving the total integral.
    """
    n = len(h)
    target = n / 2.0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # direction: move from i to j or opposite
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        # projection not needed – we kept within bounds and sum
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Binary simulated‑annealing (primary search)
    # ------------------------------------------------------------------
    # Choose an even number of points – 60 gives a comfortable
    # margin (target sum = 30) and a reasonably fine discretisation.
    n = 60
    search_deadline = start_time + budget_s * 0.85
    h_bin, max_corr = _binary_annealing(n, rng, start_time, search_deadline)

    c5_bin = max_corr * 2.0 / n
    best_h = h_bin.astype(np.float64)
    best_c5 = c5_bin

    # ------------------------------------------------------------------
    # 2.  Optional refinement using continuous moves (if time permits)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 8.0:
        # Small SLSQP warm‑start passes
        target_sum = n / 2.0
        eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target_sum}
        bounds = [(0.0, 1.0)] * n
        # a few quick SLSQP rounds
        for _ in range(6):
            if time.time() - start_time > budget_s - 5.0:
                break
            x0 = best_h + rng.normal(scale=0.05, size=n)
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = _adjust_sum(x0, target_sum)
            res = minimize(
                _evaluate_c5,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[eq_cons],
                options={"maxiter": 150, "ftol": 1e-9, "disp": False},
            )
            cand_h = np.clip(res.x, 0.0, 1.0) if res.success else x0
            cand_h = _adjust_sum(cand_h, target_sum)
            cand_c5 = _evaluate_c5(cand_h)
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand_h, cand_c5

        # final greedy fractional hill‑climb
        deadline_fc = start_time + budget_s - 0.5
        best_h, best_c5 = _fractional_hill_climb(best_h, best_c5, deadline_fc, rng)

    # ------------------------------------------------------------------
    # 3.  Return the best found solution
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
