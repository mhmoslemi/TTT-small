import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def softmax_c5(h, alpha):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(optimize.logsumexp(alpha * conv) / alpha)

def monotonic_constraints(n):
    cons = []
    half = n // 2
    for i in range(half - 1):
        cons.append({"type": "ineq", "fun": lambda h, i=i: h[i+1] - h[i]})
    for i in range(half, n-1):
        cons.append({"type": "ineq", "fun": lambda h, i=i: h[i] - h[i+1]})
    return cons

def sum_constraint(target):
    return {"type":"eq","fun":lambda h: np.sum(h) - target}

def solve_one_start(start, n, target_sum, time_deadline):
    bounds = [(0.0,1.0)]*n
    cons = [sum_constraint(target_sum)]
    cons += monotonic_constraints(n)  # optional
    # Stage 1
    res1 = opt.minimize(lambda h: softmax_c5(h, alpha=5.0),
                        start, method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter':400,'ftol':1e-8,'disp':False})
    if not res1.success or time.time()>time_deadline:
        return None, None

    # Stage 2
    res2 = opt.minimize(lambda h: softmax_c5(h, alpha=30.0),
                        res1.x, method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter':400,'ftol':1e-9,'disp':False})
    if not res2.success or time.time()>time_deadline:
        return None,None

    # Stage 3
    res3 = opt.minimize(lambda h: softmax_c5(h, alpha=80.0),
                        res2.x, method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter':400,'ftol':1e-10,'disp':False})
    if not res3.success or time.time()>time_deadline:
        return None,None

    h_opt = np.clip(res3.x,0.0,1.0)
    # enforce sum exactly
    diff = target_sum - h_opt.sum()
    if abs(diff)>1e-9:
        h_opt += diff/n
        h_opt = np.clip(h_opt,0.0,1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def symmetrize_and_normalize(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Make array symmetric (v[i]=v[-i-1]), clip to [0,1] and enforce exact sum."""
    n = v.shape[0]
    sym = (v + v[::-1]) / 2.0
    sym = np.clip(sym, 0.0, 1.0)

    cur = sym.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        sym += diff / n
        sym = np.clip(sym, 0.0, 1.0)

    # a second tiny correction in case clipping changed the sum
    cur = sym.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        sym += diff / n
        sym = np.clip(sym, 0.0, 1.0)

    return sym


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def generate_triangle(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def monotonic_constraints_half(m: int):
    """Inequality constraints a[i+1] - a[i] >= 0 for i=0..m-2."""
    cons = []
    for i in range(m - 1):
        cons.append({"type": "ineq", "fun": lambda a, i=i: a[i + 1] - a[i]})
    return cons


# ----------------------------------------------------------------------
# Optimisation core (half‑vector representation, symmetric h)
# ----------------------------------------------------------------------
def solve_one_start_half(start_a: np.ndarray, n: int, target_sum: float,
                         time_deadline: float):
    """Optimise a symmetric step function starting from half‑vector `start_a`."""
    m = start_a.shape[0]
    bounds = [(0.0, 1.0)] * m
    cons = [{"type": "eq",
             "fun": lambda a: np.sum(a) - target_sum / 2.0}]
    cons += monotonic_constraints_half(m)

    def obj(a, alpha):
        h = np.concatenate([a, a[::-1]])
        conv = np.correlate(h, 1.0 - h, mode="full")
        return float(logsumexp(alpha * conv) / alpha)

    # stage 1 – gentle smoothing
    res1 = minimize(lambda a: obj(a, 5.0), start_a, method="SLSQP",
                    bounds=bounds, constraints=cons,
                    options={"maxiter": 400, "ftol": 1e-8, "disp": False})
    if not res1.success or time.time() > time_deadline:
        return None, None

    # stage 2 – sharper smoothing
    res2 = minimize(lambda a: obj(a, 30.0), res1.x, method="SLSQP",
                    bounds=bounds, constraints=cons,
                    options={"maxiter": 400, "ftol": 1e-9, "disp": False})
    if not res2.success or time.time() > time_deadline:
        return None, None

    # stage 3 – very sharp smoothing
    res3 = minimize(lambda a: obj(a, 80.0), res2.x, method="SLSQP",
                    bounds=bounds, constraints=cons,
                    options={"maxiter": 400, "ftol": 1e-10, "disp": False})
    if not res3.success or time.time() > time_deadline:
        return None, None

    a_opt = np.clip(res3.x, 0.0, 1.0)
    # enforce exact half‑sum
    diff = (target_sum / 2.0) - a_opt.sum()
    if abs(diff) > 1e-9:
        a_opt += diff / m
        a_opt = np.clip(a_opt, 0.0, 1.0)

    h_opt = np.concatenate([a_opt, a_opt[::-1]])
    c5_opt = compute_c5(h_opt)

    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n % 2 == 1:
            # enforce even length by dropping last element
            parent_h = parent_h[:-1]
            parent_n -= 1
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    for n in candidate_ns:
        if time.time() > time_limit:
            break
        target_sum = n / 2.0
        m = n // 2

        starts_a = []

        # 1. Parent (upsampled / symmetrised)
        if parent_h is not None:
            if parent_n == n:
                p = parent_h.copy()
            else:
                p = upsample(parent_h, n)
            p = symmetrize_and_normalize(p, target_sum)
            starts_a.append(p[:m])

        # 2. Triangular baseline (already symmetric)
        tri = generate_triangle(n)
        tri = symmetrize_and_normalize(tri, target_sum)
        starts_a.append(tri[:m])

        # 3. Random monotone half‑vectors (5 variants)
        for _ in range(5):
            a = rng.random(m)
            a.sort()
            a = a * ((m / 2.0) / a.sum())
            a = np.clip(a, 0.0, 1.0)
            diff = (m / 2.0) - a.sum()
            if abs(diff) > 1e-12:
                a += diff / m
                a = np.clip(a, 0.0, 1.0)
            starts_a.append(a)

        # 4. Random binary full vectors, symmetrised (3 variants)
        for _ in range(3):
            full = np.zeros(n)
            ones = n // 2
            idx = rng.choice(n, size=ones, replace=False)
            full[idx] = 1.0
            full = symmetrize_and_normalize(full, target_sum)
            starts_a.append(full[:m])

        # 5. Small perturbations of the triangular shape (2 variants)
        for _ in range(2):
            pert = tri + rng.uniform(-0.15, 0.15, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = symmetrize_and_normalize(pert, target_sum)
            starts_a.append(pert[:m])

        # -----------------------------------------------------------------
        # Optimise each start for this resolution
        # -----------------------------------------------------------------
        for start_a in starts_a:
            if time.time() > time_limit:
                break
            h_opt, c5_opt = solve_one_start_half(start_a, n, target_sum, time_limit)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                # early exit if we already beat the target record
                if best_c5 < 0.3808:
                    break

        if best_c5 < 0.3808:
            break

    # -----------------------------------------------------------------
    # Post‑processing: gentle random perturbations of current best
    # -----------------------------------------------------------------
    if best_h is not None:
        for _ in range(5):
            if time.time() > time_limit:
                break
            pert = best_h + rng.normal(scale=0.02, size=best_n)
            pert = np.clip(pert, 0.0, 1.0)
            # enforce integral
            diff = (best_n / 2.0) - pert.sum()
            pert += diff / best_n
            pert = np.clip(pert, 0.0, 1.0)
            # symmetrise again to keep monotone shape (helps the optimiser)
            pert = symmetrize_and_normalize(pert, best_n / 2.0)
            h_opt, c5_opt = solve_one_start_half(pert[:best_n // 2],
                                                 best_n,
                                                 best_n / 2.0,
                                                 time_limit)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # -----------------------------------------------------------------
    # Fallback: simple half‑filled block if nothing succeeded
    # -----------------------------------------------------------------
    if best_h is None:
        n = 100
        half = np.concatenate([np.ones(n // 2), np.zeros(n // 2)])
        best_h = half.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final sanity check – enforce exact integral (tiny numerical drift)
    target = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h = best_h + (target - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
