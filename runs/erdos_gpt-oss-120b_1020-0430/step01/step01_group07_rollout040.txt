def anneal(start_h, time_budget):
    h = start_h.copy()
    n = len(h)
    best_h = h.copy()
    best_c = calc_c5(h)
    cur_c = best_c
    start = time.time()
    T0 = 1.0
    T_end = 1e-4
    max_step = 0.4
    while time.time() - start < time_budget:
        # progress, temperature
        progress = (time.time() - start)/time_budget
        T = T0 * (T_end / T0) ** progress
        # compute gradient for worst shift
        corr = np.correlate(h, h, mode='full')
        min_idx = np.argmin(corr)
        shift = min_idx - (n-1)
        # compute gradient vector
        g = np.zeros(n)
        if shift >= 0:
            g[:n-shift] += h[shift:]  # h_i+k
            g[shift:] += h[:n-shift]  # h_i-k
        else:
            s = -shift
            g[s:] += h[:n-s]  # forward since shift negative means k = -s; treat similarly
            g[:n-s] += h[s:]  # other side
        # choose i from high g, j from low g
        per = max(1, n // 10)
        high_candidates = np.argpartition(g, -per)[-per:]
        low_candidates = np.argpartition(g, per)[:per]
        i = int(np.random.choice(high_candidates))
        j = int(np.random.choice(low_candidates))
        hi, hj = h[i], h[j]
        low_delta = max(-hi, hj - 1.0)
        high_delta = min(1.0 - hi, hj)
        if low_delta >= high_delta:
            continue
        # scale delta by temperature
        step_cap = max_step * (T / T0)
        delta_low = max(low_delta, -step_cap)
        delta_high = min(high_delta, step_cap)
        if delta_low >= delta_high:
            delta = random.uniform(low_delta, high_delta)
        else:
            delta = random.uniform(delta_low, delta_high)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = calc_c5(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c

import time, math, random, numpy as np


def _calc_c5(h):
    """C5 = 1 - (2/n) * min_k sum_i h_i h_{i+k}."""
    n = h.shape[0]
    corr = np.correlate(h, h, mode="full")
    return 1.0 - (2.0 / n) * corr.min()


def _initial_vector():

            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function, its C5 bound and sample count."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    h0 = _initial_vector()
    n = h0.size

    # primary annealing (≈70 % of the budget)
    main_time = total_budget * 0.7
    best_h, best_c = _anneal(h0, main_time)

    # random binary‑fractional restarts with the remaining time
    remaining = total_budget - (time.time() - start_all)
    while remaining > 5.0:
        target = n / 2.0
        whole = int(np.floor(target))
        frac = target - whole
        h_start = np.zeros(n, dtype=np.float64)
        idx = np.random.permutation(n)
        h_start[idx[:whole]] = 1.0
        if frac > 1e-12:
            h_start[idx[whole]] = frac

        slice_t = min(remaining * 0.2, 30.0)
        cand_h, cand_c = _anneal(h_start, slice_t)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        remaining = total_budget - (time.time() - start_all)

    # deterministic gradient‑hill refinement with whatever time is left
    hill_deadline = time.time() + max(0.0, total_budget - (time.time() - start_all))
    per = max(1, n // 10)
    while time.time() < hill_deadline:
        corr = np.correlate(best_h, best_h, mode="full")
        min_idx = np.argmin(corr)
        shift = min_idx - (n - 1)

        g = np.zeros(n)
        if shift >= 0:
            k = shift
            g[: n - k] += best_h[k:]
            g[k:] += best_h[: n - k]
        else:
            k = -shift
            g[k:] += best_h[: n - k]
            g[: n - k] += best_h[k:]

        high_idx = np.argpartition(g, -per)[-per:]
        low_idx = np.argpartition(g, per)[:per]
        i = int(np.random.choice(high_idx))
        j = int(np.random.choice(low_idx))

        hi, hj = best_h[i], best_h[j]
        lo_delta = max(-hi, hj - 1.0)
        hi_delta = min(1.0 - hi, hj)
        if lo_delta >= hi_delta:
            continue
        delta = random.uniform(lo_delta, hi_delta) * 0.1
        best_h[i] = hi + delta
        best_h[j] = hj - delta

        cur_c = _calc_c5(best_h)
        if cur_c < best_c:
            best_c = cur_c
        else:
            best_h[i] = hi
            best_h[j] = hj

    # enforce exact sum (tiny rounding fix)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # final evaluation (use provided helper if present)
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _calc_c5(best_h)

    return best_h, float(final_c), int(n)
