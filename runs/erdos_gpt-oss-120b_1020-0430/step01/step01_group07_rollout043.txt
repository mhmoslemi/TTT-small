def compute_conv(h):
    return np.convolve(h, h, mode='full').astype(np.int64)

def compute_c5_from_conv(conv, n):
    min_conv = conv.min()
    return (n/2 - min_conv) * (2.0 / n)

def binary_swap_update(conv, h, ones_set, zeros_set, offset, i, j):
    # swap values at i and j (i in ones_set, j in zeros_set)
    # we can assume h[i] == 1, h[j] == 0.
    # Remove contributions of i:
    for x in ones_set:
        if x == i:
            continue
        conv[i - x + offset] -= 1
        conv[x - i + offset] -= 1
    conv[offset] -= 1
    # update set for removal
    ones_set.remove(i)
    zeros_set.add(i)
    # Add contributions of j (now becomes 1)
    for x in ones_set:
        conv[j - x + offset] += 1
        conv[x - j + offset] += 1
    conv[offset] += 1
    # update sets for addition
    zeros_set.remove(j)
    ones_set.add(j)
    # toggle h
    h[i] = 0.0
    h[j] = 1.0
```

Implementation details of `binary_anneal`:

```
def binary_anneal(start_h, max_seconds):
    h = start_h.copy()
    n = h.shape[0]
    offset = n-1
    conv = np.convolve(h, h, mode='full')
    cur_c = compute_c5_from_conv(conv, n)
    best_c = cur_c
    best_h = h.copy()
    ones_set = set(int(i) for i, v in enumerate(h) if v > 0.5)
    zeros_set = set(range(n)) - ones_set
    start = time.time()
    T0 = 0.02
    T_end = 1e-7
    while True:
        now = time.time()
        if now - start >= max_seconds:
            break
        progress = (now - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        i = random.choice(tuple(ones_set))
        j = random.choice(tuple(zeros_set))
        # apply swap
        binary_swap_update(conv, h, ones_set, zeros_set, offset, i, j)
        new_c = compute_c5_from_conv(conv, n)
        delta = new_c - cur_c
        if delta < 0 or random.random() < math.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            binary_swap_update(conv, h, ones_set, zeros_set, offset, i, j)
    return best_h, best_c
```

def hill_climb(h, conv, ones_set, zeros_set, offset, n, iters):
    cur_c = compute_c5_from_conv(conv, n)

        else:
            # revert
            binary_swap_update(conv, h, ones_set, zeros_set, offset, i, j)

def swap_update(corr, h, ones_list, zeros_list, ones_pos, zeros_pos, offset, i, j):
    # Determine which is one
    if h[i] == 1:
        one = i
        zero = j
    else:
        one = j
        zero = i

    # removal contributions from 'one'
    for x in ones_list:
        if x == one:
            continue
        l1 = one - x
        corr[l1 + offset] -= 1
        corr[-l1 + offset] -= 1
    corr[offset] -= 1  # self term

    # Update list positions: replace 'one' with 'zero' in ones_list and vice versa
    pos_one = ones_pos[one]
    pos_zero = zeros_pos[zero]
    ones_list[pos_one] = zero
    ones_pos[zero] = pos_one
    del ones_pos[one]

    zeros_list[pos_zero] = one
    zeros_pos[one] = pos_zero
    del zeros_pos[zero]

    # Update h
    h[one] = 0.0
    h[zero] = 1.0

    # addition contributions from new one 'zero'
    for x in ones_list:
        if x == zero:
            continue
        l1 = zero - x
        corr[l1 + offset] += 1
        corr[-l1 + offset] += 1
    corr[offset] += 1

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    total_budget = float(budget_s)
    start_all = time.time()
    # Choose base resolution
    n_base = 200
    # ensure even
    if n_base % 2: n_base += 1
    # Build initial vector
    if 'initial_h_values' in globals():
        # use interpolation and threshold to binary
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        # resize to n_base
        x0 = np.linspace(0, 2, n0, endpoint=False)
        xb = np.linspace(0, 2, n_base, endpoint=False)
        interp = np.interp(xb, x0, h0)
        # select top n//2 entries
        idx = np.argpartition(interp, -n_base//2)[-n_base//2:]
        h_start = np.zeros(n_base, dtype=np.float64)
        h_start[idx] = 1.0
    else:
        h_start = np.zeros(n_base, dtype=np.float64)
        ones_idx = np.random.choice(n_base, n_base//2, replace=False)
        h_start[ones_idx] = 1.0

    # primary binary annealing
    primary_time = total_budget * 0.75
    best_h, best_c = _binary_anneal(h_start, primary_time)

    # random restarts
    remaining = total_budget - (time.time() - start_all)
    while remaining > 10.0:
        # random binary start
        h_rand = np.zeros(n_base, dtype=np.float64)
        ones_idx = np.random.choice(n_base, n_base//2, replace=False)
        h_rand[ones_idx] = 1.0
        slice_time = min(remaining * 0.3, 30.0)
        cand_h, cand_c = _binary_anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        remaining = total_budget - (time.time() - start_all)

    # optional upscale to double resolution if time left (>20 sec)
    if remaining > 20.0 and n_base * 2 <= 800:
        # upscale by repeating each entry twice
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # ensure integer ones count = n_up // 2 (should hold)
        # Run anneal on upscale
        up_time = remaining
        cand_h_up, cand_c_up = _binary_anneal(h_up, up_time)
        if cand_c_up < best_c:
            best_c, best_h = cand_c_up, cand_h_up

    # final evaluation
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        # as fallback compute directly
        n = best_h.size
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        final_c = np.max(corr) * (2.0 / n)

    return best_h, float(final_c), int(best_h.size)

def _binary_anneal(start_h, max_seconds):
    h = start_h.copy()
    n = h.shape[0]
    offset = n - 1
    # compute correlation (self)
    corr = np.correlate(h.astype(np.int64), h.astype(np.int64), mode='full')
    # Create lists and dict pos
    ones_list = [int(i) for i, v in enumerate(h) if v > 0.5]
    zeros_list = [int(i) for i, v in enumerate(h) if v <= 0.5]
    ones_pos = {idx: i for i, idx in enumerate(ones_list)}
    zeros_pos = {idx: i for i, idx in enumerate(zeros_list)}
    # compute current C5
    cur_c = (n/2 - corr.min()) * (2.0 / n)
    best_c = cur_c
    best_h = h.copy()
    start = time.time()
    T0 = 0.02
    T_end = 1e-7
    while True:
        now = time.time()
        if now - start >= max_seconds:
            break
        progress = (now - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        # sample random pair
        i = random.choice(ones_list)
        j = random.choice(zeros_list)
        # apply swap
        _swap_pair(corr, h, ones_list, zeros_list, ones_pos, zeros_pos, offset, i, j)
        new_c = (n/2 - corr.min()) * (2.0 / n)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c - 1e-12:

    h = start_h.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(h)

    # choose initial alpha, final alpha
    alpha_start = 0.5
    alpha_end = 8.0
    eta_start = 0.5
    eta_end = 0.01
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        eta = eta_start * (1 - progress) + eta_end * progress

        cur_c = _c5_metric(h)

import time, random, math
import numpy as np

def _c5_metric(h):
    """Maximum overlap C5 for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_bounds_sum(x, target, iters=50):
    """Project x onto 0≤x≤1 and sum(x)=target using a simple shift‑and‑clip."""
    lo, hi = -1.0, 1.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _grad_descent_shift(start_h, max_seconds):
    """Gradient descent on the worst shift (max‑overlap)."""
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(h)

    start = time.time()
    while time.time() - start < max_seconds:
        # correlation with complement, find worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_c = corr[idx] * (2.0 / n)

        # shift value (negative indices allowed)
        k = idx - (n - 1)

        # gradient of J = max_corr = Σ h_i (1‑h_{i+k})
        # dJ/dh_i = 1 - h_{i+k} - h_{i‑k} (missing terms treated as 0)
        if k >= 0:
            shift_r = np.concatenate((h[k:], np.zeros(k)))
            shift_l = np.concatenate((np.zeros(k), h[:n - k]))
        else:
            kk = -k
            shift_r = np.concatenate((np.zeros(kk), h[:n - kk]))
            shift_l = np.concatenate((h[kk:], np.zeros(kk)))
        grad = 1.0 - shift_r - shift_l

        # step size schedule (larger early, smaller later)
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        eta = 0.5 * (1.0 - progress) + 0.01 * progress

        # gradient step + projection
        h = _project_bounds_sum(h - eta * grad, target)

        # record best
        new_c = _c5_metric(h)
        if new_c < best_c:
            best_c, best_h = new_c, h.copy()
        # occasionally recompute worst shift (already done at top of loop)
    return best_h, best_c


def _final_tweak(start_h, max_seconds):
    """Small random mass‑transfer hill‑climb to polish a solution."""
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(h)

    start = time.time()
    while time.time() - start < max_seconds:
        i, j = random.sample(range(n), 2)
        # feasible delta moving mass from i to j
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue
        delta = random.uniform(lo, hi) * 0.1  # keep steps modest
        h[i] += delta
        h[j] -= delta
        new_c = _c5_metric(h)
        if new_c < best_c:
            best_c, best_h = new_c, h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_c
