def run(seed=42, budget_s=1100):
    random.seed(seed); np.random.seed(seed)
    n = 400  # maybe 300? Use 400 for better resolution
    # initial vector: interpolate parent if exists else uniform.
    if "initial_h_values" in globals():
        # initial_h_values may be size 100. Interpolate to n.
        h0 = np.interp(np.linspace(0,2, n, endpoint=False),
                       np.linspace(0,2, len(initial_h_values), endpoint=False),
                       initial_h_values)
    else:
        h0 = np.full(n, 0.5)
    # enforce sum by scaling then clip adjust.
    target = n/2.0
    h0 = np.clip(h0, 0.0, 1.0)
    cur = h0.sum()
    if cur != 0:
        h0 *= target / cur
    h0 = np.clip(h0,0,1)
    # adjust sum exactly if small diff
    diff = target - h0.sum()
    if abs(diff) > 1e-12:
        # add diff to a random index respecting bounds
        for idx in np.random.permutation(n):
            if 0 <= h0[idx] + diff <= 1:
                h0[idx] += diff
                break
    best_h = h0.copy()
    best_c = c5_metric(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    start = time.time()
    total_budget = float(budget_s)

    while time.time() - start < total_budget * 0.9:
        elapsed = time.time() - start
        progress = elapsed / (total_budget * 0.9)
        # temperature schedule
        T0 = 0.2
        T_end = 1e-4
        T = T0 * (T_end/T0) ** progress
        max_step = 0.5 * (1 - progress) + 0.01  # gradually shrink
        # propose move
        i = random.randrange(n)
        j = random.randrange(n-1)
        if j >= i:
            j += 1
        hi, hj = cur_h[i], cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        step_lo = max(low, -max_step)
        step_hi = min(high, max_step)
        if step_lo < step_hi:
            delta = random.uniform(step_lo, step_hi)
        else:
            delta = random.uniform(low, high)
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_c = c5_metric(cur_h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = hi
            cur_h[j] = hj

def c5_metric(h):
    n = h.shape[0]
    L = 2*n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)
    # take first 2*n-1 entries (full correlation)
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

def _anneal(start_h, max_seconds):
    h = start_h.copy()
    best_h = h.copy()
    best_c = c5_metric(h)
    cur_c = best_c
    n = h.size
    start = time.time()
    # parameters
    T0 = 0.3
    T_end = 1e-5
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        # temperature schedule exponential
        T = T0 * (T_end / T0) ** progress
        # max step size shrink
        max_step = 0.5 * (1 - progress) + 0.01  # between 0.5 and 0.01
        # pick i,j random distinct
        i = random.randint(0, n-1)
        j = random.randint(0, n-2)
        if j >= i:
            j += 1
        hi = h[i]; hj = h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        lo = max(low, -max_step)
        hi_step = min(high, max_step)
        if lo < hi_step:
            delta = random.uniform(lo, hi_step)
        else:
            delta = random.uniform(low, high)
        # apply
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_c
```

def guided_step(h):
    # compute correlation O_s via c5_metric but also want the shift index of max.
    n = len(h)
    L = 2*n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)
    # compute slice full correlation length 2*n-1
    corr_full = corr[:2*n-1]
    idx_max = np.argmax(corr_full)
    # shift s corresponds: The full correlation index offset = idx
    # For np.correlate with mode='full', shift index - (n-1) corresponds to actual shift.
    s_star = idx_max - (n - 1)
    # compute gradient
    grad = np.zeros(n)
    if s_star >= 0:
        grad[:n - s_star] += h[s_star:]   # h_{i+s}
        grad[s_star:] += h[:n - s_star]   # h_{i-s}
    else:
        s = -s_star
        grad[s:] += h[:n - s]   # h_{i+s} where i>=s => i-s index
        grad[:n - s] += h[s:]   # h_{i-s}
    # Clip gradient? not needed.
    # Choose i_max from indices where grad high and h[i] < 1.
    # Choose i_min where grad low and h[i] > 0.
    candidates_max = np.where(h < 1 - 1e-12)[0]
    candidates_min = np.where(h > 1e-12)[0]
    if len(candidates_max) == 0 or len(candidates_min) == 0:
        return False, None
    i_max = candidates_max[np.argmax(grad[candidates_max])]
    i_min = candidates_min[np.argmin(grad[candidates_min])]
    # Transfer delta up to min(1-h[i_max], h[i_min]) and also limited by step size
    max_delta = min(1.0 - h[i_max], h[i_min])
    if max_delta <= 0:
        return False, None
    # Choose delta maybe fraction of max_delta (e.g., 0.1*max_delta)
    delta = max_delta * 0.5  # adjust step
    # apply
    h[i_max] += delta
    h[i_min] -= delta
    return True, (i_min,i_max,delta)

import time, math, random
import numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """Fast C5 evaluation using FFT (full linear correlation)."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[: 2 * n - 1]
    return float(np.max(corr) * (2.0 / n))


def _initial_vector(n: int) -> np.ndarray:
    """Create a feasible starting vector of length n."""
    if "initial_h_values" in globals():
        src = np.asarray(initial_h_values, dtype=np.float64)
        m = src.size
        xs = np.linspace(0.0, 2.0, m, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, xs, src)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
    return h


def _anneal(start_h: np.ndarray, max_seconds: float) -> tuple:
    """Simulated‑annealing improvement (FFT‑based metric)."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0, T_end = 0.3, 1e-5

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        T = T0 * (T_end / T0) ** prog
        max_step = 0.5 * (1.0 - prog) + 0.01

        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)        # keep both in [0,1]
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        lo = max(low, -max_step)
        hi_step = min(high, max_step)
        if lo < hi_step:
            delta = random.uniform(lo, hi_step)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c

        if dC < 0.0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    return best_h, best_c

    """Deterministic refinement targeting the worst shift."""
    best_h = h.copy()
    best_c = cur_best
    n = best_h.size
    start = time.time()
    while time.time() - start < max_seconds:
        # correlation and worst shift
        L = 2 * n
        H = np.fft.rfft(best_h, L)
        G = np.fft.rfft(1.0 - best_h, L)

            grad[: n - ss] += best_h[ss:]

        low_candidates = np.where(best_h > 1e-12)[0]
        high_candidates = np.where(best_h < 1.0 - 1e-12)[0]
        if low_candidates.size == 0 or high_candidates.size == 0:
            break

        i_min = low_candidates[np.argmin(grad[low_candidates])]
        i_max = high_candidates[np.argmax(grad[high_candidates])]
        max_delta = min(best_h[i_min], 1.0 - best_h[i_max])
        if max_delta <= 1e-12:
            break

        delta = max_delta * 0.5
        best_h[i_min] -= delta
        best_h[i_max] += delta
        new_c = _c5_metric(best_h)

        if new_c + 1e-12 < best_c:
            best_c = new_c
        else:
            # revert and try a smaller transfer
            best_h[i_min] += delta
            best_h[i_max] -= delta
            delta2 = max_delta * 0.1
            if delta2 > 1e-12:
                best_h[i_min] -= delta2
                best_h[i_max] += delta2
                new_c2 = _c5_metric(best_h)
                if new_c2 + 1e-12 < best_c:
                    best_c = new_c2
                else:
                    best_h[i_min] += delta2
                    best_h[i_max] -= delta2
                    break
            else:
                break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C₅ bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    n = 400                         # resolution (≲1000 as required)
    h0 = _initial_vector(n)

    total = float(budget_s)
    start_all = time.time()

    # primary annealing (≈70 % of budget)
    primary_time = total * 0.70
    best_h, best_c = _anneal(h0, primary_time)

    # random‑restart phase (remaining budget)
    while True:
        elapsed = time.time() - start_all
        remaining = total - elapsed
        if remaining < 6.0:
            break

        # binary start: exactly n/2 ones (plus an optional fractional entry)
        target = n / 2.0
        ones = int(target)
        frac = target - ones
        h_rand = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        h_rand[perm[:ones]] = 1.0
        if frac > 0:
            h_rand[perm[ones]] = frac

        slice_time = min(remaining * 0.30, 25.0)
        cand_h, cand_c = _anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h

    # final guided refinement (≈ remaining 30 %)
    elapsed = time.time() - start_all
    remaining = total - elapsed
    if remaining > 2.0:
        best_h, best_c = _guided_refine(best_h, remaining, best_c)

    # final safeguard for exact mass
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    try:
        final_c = evaluate_erdos_solution(best_h)  # pre‑imported helper, if present
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
