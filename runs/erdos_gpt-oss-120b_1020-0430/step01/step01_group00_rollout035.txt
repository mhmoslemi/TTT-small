def _gradient_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        a = 0
        b = n - shift
        g[a:b] += 1.0
        g[a:b] -= h[shift:]  # first term

        g[:n - k] -= h[k:]   # -h[i - shift] where i - shift = i + k
    return g

def _project_to_feasible(h, n, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # simple iterative adjustment
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # adjust all free variables equally
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[mask] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # apply to any variable not at bound
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _solve_h(initial_h, max_iter=250):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # define constraints functions referencing n
    def ineq_func(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode='full')
    def eq_func(x):
        return np.sum(x[:n]) - (n / 2.0)

    constraints = [{'type':'ineq','fun':ineq_func},
                   {'type':'eq','fun':eq_func}]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol':1e-9, 'disp': False}
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val

import time
import numpy as np
from scipy import optimize


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 via a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0.0
        else:
            free = h < 1.0
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1.0 - 1e-12:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given step function."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _corr_compute(h: np.ndarray):
    """Return max correlation, its shift and the full correlation array."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (n - 1)          # integer shift (positive → right)
    return max_val, shift, corr


def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the correlation at the maximizing shift."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        a = 0
        b = n - shift
        g[a:b] += 1.0
        g[a:b] -= h[shift:]         # -(1‑h[i+shift]) part
        g[shift:] -= h[:n - shift]   # -h[i‑shift] part
    else:
        k = -shift
        a = k
        b = n
        g[a:b] += 1.0
        g[a:b] -= h[:n - k]          # -(1‑h[i+shift]) = -h[i‑k]
        g[:n - k] -= h[k:]           # -h[i‑shift]   = -h[i+k]
    return g


def _optimize_h(
    h0: np.ndarray,
    n: int,
    time_left,
    max_iter: int = 800,
    init_lr: float = 0.9,
    lr_decay: float = 0.997,
    target: float = 0.3808,
):
    """Projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    lr = init_lr

    for it in range(max_iter):
        if time_left() < 0.8:
            break

        max_val, shift, _ = _corr_compute(h)
        c5 = max_val * (2.0 / n)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            if best_c5 < target:
                break

        g = _gradient_shift(h, shift)

        # trial step
        h_new = h - lr * g
        h_new = _project_to_feasible(h_new, n)

        new_c5 = _c5_from_h(h_new)

        # simple Armijo‑like acceptance
        if new_c5 <= c5:
            h = h_new
            lr = min(lr * 1.07, 1.5)
        else:
            lr *= 0.65

        lr *= lr_decay
    return best_h, best_c5


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _solve_h(initial_h: np.ndarray, max_iter: int = 300):
    """SLSQP refinement of a feasible h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)

    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)
