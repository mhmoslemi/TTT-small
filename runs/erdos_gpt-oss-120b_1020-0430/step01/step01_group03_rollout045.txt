def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n/2.0
    # define objective
    def obj(x, beta):
        # correlation
        corr = np.correlate(x, 1 - x, mode='full')
        # safe log-sum-exp
        bc = beta * corr
        max_bc = np.max(bc)
        return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / beta
    # eq constraint
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0,1.0)]*n
    res = opt.minimize(obj, h0, args=(beta,), method='SLSQP', bounds=bounds,
        constraints=cons, options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # enforce sum exactly
    diff = target - x.sum()
    # adjust as before
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_feasible_vector(n, rng):
    # Start uniform 0.5
    h = np.full(n, 0.5)
    # random transfers
    for _ in range(n * 12):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        h[i] -= eps
        h[j] += eps
    # enforce sum exactly n/2
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(3000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_time_limited(h_start, rng, max_time):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < max_time:
        elapsed = time.time() - t0
        fraction = elapsed / max_time
        # scale decays linearly from 0.5 to 1e-4
        scale = max(0.5 * (1 - fraction), 1e-4)
        new_h = perturb_h(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
    return best_h, best_c

def simulated_annealing(h0, rng, time_budget):
    start = time.time()
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    # temperature schedule
    T0 = 0.008  # initial temperature
    Tf = 1e-9   # final temperature
    while time.time() - start < time_budget:
        frac = (time.time() - start) / time_budget
        T = T0 * (1 - frac) + Tf * frac
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        # epsilon proportional to max_eps, adapt with temperature maybe
        eps = rng.random() * max_eps * 0.5
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def upsample_h(h_small, n_new, rng):
    n_small = h_small.size
    x_small = np.linspace(0, 2, n_small, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_small, h_small)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new / 2.0
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n_new)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()
            else:
                sub = min(-diff, h_new[i])
                h_new[i] -= sub
                diff = target - h_new.sum()
    return h_new

def slsqp_optimize(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n / 2.0
    # objective
    def obj(x, beta):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # stable log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    # equality constraint
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, args=(beta,), method='SLSQP', bounds=bounds,
                    constraints=cons,
                    options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum adjustment
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff = target - x.sum()
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff = target - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # target resolution
    n = 120
    # Initialise h
    if 'initial_h_values' in globals():
        h_small = np.array(initial_h_values, dtype=float)
        h0 = upsample_h(h_small, n, rng)
    else:
        h0 = random_feasible_vector(n, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Simulated annealing (≈50% of budget)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    sa_time = min(remaining * 0.5, max(5.0, remaining * 0.5))
    if sa_time > 5.0:
        h_sa, c_sa = simulated_annealing(best_h, rng, sa_time)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # Phase 2: Deterministic hill climb (≈30% budget)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    hill_time = remaining * 0.4
    if hill_time > 2.0:
        h_hc, c_hc = hill_climb_time_limited(best_h, rng, hill_time)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # Phase 3: SLSQP refinement with increasing beta
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 1.0:
        betas = [10.0, 50.0, 200.0, 800.0]
        for beta in betas:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                trial = slsqp_optimize(best_h, beta, rng, max_iter=400)
                c_trial = compute_c5(trial)
                if c_trial < best_c:
                    best_h, best_c = trial, c_trial
            except Exception:
                continue

    # Phase 4: Optional quick random restarts if time left
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 4.0:
            break
        # random restart
        h_start = random_feasible_vector(n, rng)
        # short SA
        sa_short = min(remaining * 0.3, 8.0)
        h_sa, c_sa = simulated_annealing(h_start, rng, sa_short)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa
        # quick hill climb
        hc_short = remaining * 0.2
        h_hc, c_hc = hill_climb_time_limited(best_h, rng, hc_short)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
                best_h[i] += add
                diff = target - best_h.sum()
            else:
                sub = min(-diff, best_h[i])
                best_h[i] -= sub
                diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core evaluation
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Feasibility utilities
# ----------------------------------------------------------------------
def random_feasible_vector(n, rng):
    """Create a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 12):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        h[i] -= eps
        h[j] += eps
    # enforce the exact sum
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(3000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff = target - h.sum()
    return h


def upsample_h(h_small, n_new, rng):
    """Linear interpolation of a smaller step function to a larger grid."""
    n_small = h_small.size
    x_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_small, h_small)
    h_new = np.clip(h_new, 0.0, 1.0)

    target = n_new / 2.0
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n_new)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()
            else:
                sub = min(-diff, h_new[i])
                h_new[i] -= sub
                diff = target - h_new.sum()
    return h_new

def perturb_h(h, rng, scale=0.2):
    """Transfer a random amount of mass from i to j (keeps total unchanged)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_time_limited(h_start, rng, max_time):
    """Deterministic hill‑climb (accept only improvements) with a time budget."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < max_time:
        elapsed = time.time() - t0
        frac = elapsed / max_time
        scale = max(0.5 * (1.0 - frac), 1e-4)
        new_h = perturb_h(h, rng, scale=scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
    return best_h, best_c

# Simulated annealing
# ----------------------------------------------------------------------
def simulated_annealing(h0, rng, time_budget):
    """Simple SA that respects the sum constraint by moving mass between two points."""
    start = time.time()
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    T0 = 0.02          # initial temperature
    Tf = 1e-10         # final temperature

    while time.time() - start < time_budget:
        frac = (time.time() - start) / time_budget
        T = T0 * (1.0 - frac) + Tf * frac

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps                     # allow full transfer
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)

        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Run one SLSQP pass on the log‑sum‑exp surrogate of max_{k} Σ h_i (1−h_{i+k})."""
    n = h0.size
    target = n / 2.0

    def obj(x, beta):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, h0, args=(beta,), method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})

    x = np.clip(res.x, 0.0, 1.0)

    # exact sum repair (tiny drift may remain)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff = target - x.sum()
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff = target - x.sum()
    return x
