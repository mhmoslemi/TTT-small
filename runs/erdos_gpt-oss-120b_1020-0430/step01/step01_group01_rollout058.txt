def _project_onto_capped_simplex(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= 1e-9:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)
```

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[:n - s] = 1.0 - h[s:]
        g[n - s:] = 1.0
        g[s:] += -h[:n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[:n - s]
        g[:s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _adam_optimize(h0, target, max_iter=800, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = h.shape[0]
    for t in range(1, max_iter + 1):
        # correlation compute shift with max
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)  # gradient of raw corr
        # scale by dx? Not needed if we treat objective scaled accordingly
        # But we might multiply by 2.0/n (dx)
        # Compute Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # Projection
        h = _project_onto_capped_simplex(h, target)
    return h

def _slsqp_optimize(h0, target, alpha=30.0, max_iter=1000):
    n = len(h0)
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        # smooth max
        m = np.max(corr)
        shifted = corr - m
        return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
```

def _local_search(h, target, dx, rng, steps=100):
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)
    n = len(h)
    for _ in range(steps):

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Projection onto the capped simplex {0≤x≤1, sum(x)=target}
# ------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ------------------------------------------------------------
# Objective utilities
# ------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ------------------------------------------------------------
# Adam optimizer for the non‑smooth max objective
# ------------------------------------------------------------
def _adam_optimize(h0: np.ndarray, target: float,
                   max_iter: int = 800, lr: float = 0.5,
                   beta1: float = 0.9, beta2: float = 0.999,
                   eps: float = 1e-8) -> np.ndarray:
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = h.shape[0]
    for t in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)  # raw gradient of the maximal shift
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * (1.0 - (t - 1) / max_iter) * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = _project_onto_capped_simplex(h, target)
    return h


# ------------------------------------------------------------
# Optional smooth‑max refinement via SLSQP
# ------------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 1000) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        shifted = corr - m
        return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ------------------------------------------------------------
# Simple local pairwise transfer search
# ------------------------------------------------------------
def _local_search(h: np.ndarray, target: float, dx: float,
                  rng: np.random.RandomState, steps: int = 500) -> tuple:
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)
    n = h.shape[0]
    for _ in range(steps):
        i, j = rng.choice(n, 2, replace=False)
        if best_h[i] <= 1e-12 or best_h[j] >= 1.0 - 1e-12:
            continue
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.rand() * 0.5
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _compute_c5(cand, dx)
        if val + 1e-9 < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val
