def hill_climb(h, max_iters, step_max, time_limit, rng):
    n = len(h)
    dx = 2.0 / n
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    it = 0
    while it < max_iters and time.time() < time_limit:
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # ensure we can move mass
        max_delta = min(step_max, best_h[i], 1 - best_h[j])
        if max_delta <= 0:
            # try other direction
            max_delta = min(step_max, best_h[j], 1 - best_h[i])
            if max_delta <= 0:
                it += 1
                continue
            # swap i j and delta direction
            i, j = j, i
        delta = rng.random() * max_delta
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_c5(new_h)
        # Accept if better
        if new_c5 + 1e-12 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        it += 1
    return best_h, best_c5
```

def hill_climb(h, n, max_iters, step_max, deadline, rng):
    best = h.copy()
    best_val = compute_c5(best)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # adapt step size shrinking linearly
        cur_step = step_max * (1.0 - it / max_iters)
        if cur_step < 1e-4:
            cur_step = 1e-4
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # compute max delta we can transfer respecting bounds
        delta = rng.random() * cur_step
        max_delta = min(delta, best[i], 1.0 - best[j])
        if max_delta <= 0:
            # try opposite direction
            max_delta = min(delta, best[j], 1.0 - best[i])
            if max_delta <= 0:
                continue
            i, j = j, i
        new = best.copy()
        new[i] -= max_delta
        new[j] += max_delta
        new_val = compute_c5(new)
        if new_val + 1e-12 < best_val:
            best = new
            best_val = new_val
    return best, best_val

def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

            inits.append(random_binary(n, rng))
        for _ in range(3):
            inits.append(random_continuous(n, rng))

        for _ in range(2):
            base = generate_triangular(n)
            pert = base + rng.uniform(-0.1, 0.1, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            diff = target_sum - pert.sum()
            if abs(diff) > 1e-12:
                pert = pert + diff / n
                pert = np.clip(pert, 0.0, 1.0)
            inits.append(pert)

        for start in inits:
            if time.time() > deadline:
                break
            start = np.clip(start, 0.0, 1.0)
            diff = target_sum - start.sum()
            if abs(diff) > 1e-12:
                start = start + diff / n
                start = np.clip(start, 0.0, 1.0)

            h_opt = slsqp_optimize(start, n, deadline)
            if time.time() > deadline:
                break

            remaining = deadline - time.time()
            max_iters = int(min(20000, max(0.0, remaining * 2000)))
            if max_iters > 0:
                h_opt, _ = hill_climb(h_opt, n, max_iters, 0.05, deadline, rng)

            c5_val = compute_c5(h_opt)
            if c5_val < best_c5 - 1e-9:
                best_h = h_opt.copy()
                best_c5 = c5_val
                best_n = n

    if best_h is None:
        n = 120
        best_h = random_binary(n, rng)
        best_c5 = compute_c5(best_h)
        best_n = n

    # optional upsampling refinement
    while time.time() < deadline:
        if best_n * 2 > 500:
            break
        new_n = best_n * 2
        h_up = upsample(best_h, new_n)
        remaining = deadline - time.time()
        max_iters = int(min(40000, max(0.0, remaining * 2000)))
        if max_iters <= 0:
            break
        h_up, _ = hill_climb(h_up, new_n, max_iters, 0.04, deadline, rng)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 - 1e-9:
            best_h = h_up
            best_c5 = c5_up
            best_n = new_n
        else:
            break

    # final feasibility adjustment
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
