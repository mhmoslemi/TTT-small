import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9,
                         max_iter: int = 100) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)

    # tiny final correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _weighted_shift_sum(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute for each index j: sum_{k} w_k * h_{j+k},
    where w is given for shifts k = -(n-1)..(n-1).
    Out‑of‑range indices are ignored (treated as 0).
    """
    n = h.size
    offset = n - 1
    out = np.zeros_like(h)
    for idx, w_val in enumerate(w):
        if w_val == 0.0:
            continue
        k = idx - offset  # shift amount
        if k >= 0:
            out[: n - k] += w_val * h[k:]
        else:
            out[-k :] += w_val * h[: n + k]
    return out


def _smooth_max_weights(corr: np.ndarray, alpha: float) -> np.ndarray:
    """Softmax weights that approximate the max over corr."""
    cmax = np.max(corr)
    ex = np.exp(alpha * (corr - cmax))
    return ex / np.sum(ex)


def _optimize_h_pg(h0: np.ndarray, time_left_func, max_iters: int = 3000) -> tuple:
    """
    Projected gradient descent on a smooth approximation of max_{k}
    ∫ h(1-h(·+k)). Returns (best_h, best_c5) found.
    """
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_from_h(h)

    step0 = 1.0
    alpha_start, alpha_end = 5.0, 40.0

    for it in range(max_iters):
        if time_left_func() < 0.1:
            break

        # correlation vector (unscaled)
        corr = np.correlate(h, 1.0 - h, mode="full")

        # smooth max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _smooth_max_weights(corr, alpha)

        # gradient of the smooth max objective
        conv1 = _weighted_shift_sum(h, w)
        conv2 = _weighted_shift_sum(h, w[::-1])
        grad = 1.0 - conv1 - conv2

        # step size schedule
        step = step0 / np.sqrt(it + 1)

        h = h - step * grad
        h = _project_to_feasible(h, n)

        val = _c5_from_h(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            # early stop if already very good
            if best_val < 0.3808:
                break

    return best_h, best_val


def _binarize_h(h: np.ndarray) -> np.ndarray:
    """Convert a continuous h into a balanced binary vector (exactly n/2 ones)."""
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h


def _hill_climb_binary(h0: np.ndarray, time_left_func,
                       max_iters: int = 2000) -> tuple:
    """
    Simple random pair‑swap hill climbing for binary vectors.
    Returns (best_h, best_c5).
    """
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(h)

    for _ in range(max_iters):
        if time_left_func() < 0.05:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        val = _c5_from_h(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

        # early exit if we beat the target
        if best_val < 0.3808:
            break

    return best_h, best_val


def _thue_morse_binary(n: int) -> np.ndarray:
    """Generate a balanced Thue‑Morse binary sequence of length n (n even)."""
    # parity of bits gives Thue-Morse pattern
    seq = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # ensure exact half ones by taking most‑significant indices
    # (for powers of two the sequence is already balanced)
    if seq.sum() != n // 2:
        # replace the smallest values with 1 until balanced
        idx_sorted = np.argsort(seq)  # zeros first
        needed = n // 2 - int(seq.sum())
        seq[idx_sorted[:needed]] = 1.0
    return seq


def _random_binary_h(n: int) -> np.ndarray:
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h
