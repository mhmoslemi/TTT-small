def _anneal_targeted(start_h, max_seconds):

    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _c5_metric_and_corr(h: np.ndarray):
    """Return (C5, full correlation array) for a sampled step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n), corr


def _shifted(h: np.ndarray, lag: int) -> np.ndarray:
    """Return h shifted by lag (out‑of‑range entries become 0)."""
    n = h.shape[0]
    if lag >= 0:
        out = np.empty(n, dtype=h.dtype)
        out[: n - lag] = h[lag:]
        out[n - lag :] = 0.0
    else:
        out = np.empty(n, dtype=h.dtype)
        out[-lag :] = h[: n + lag]
        out[: -lag] = 0.0
    return out


def _initial_vector(target_n: int) -> np.ndarray:
    """Create a feasible starting vector of length target_n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(target_n, 0.5, dtype=np.float64)

    target_sum = target_n / 2.0
    cur_sum = h.sum()
    if cur_sum > 0:
        h *= target_sum / cur_sum
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target_sum - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(target_n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _anneal_targeted(start_h: np.ndarray, max_seconds: float):
    """Simulated‑annealing with occasional moves guided by the worst lag."""
    h = start_h.copy()
    n = h.size
    cur_c, corr = _c5_metric_and_corr(h)
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    T0 = 0.07               # initial temperature
    T_end = 1e-6
    max_step_initial = 0.30   # maximal step fraction of a feasible delta

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        step_scale = max_step_initial * (T / T0) ** 0.5  # decays with temperature

        # 35 % of the moves are biased toward the lag that currently dominates
        if random.random() < 0.35:
            idx_max = int(np.argmax(corr))
            lag = idx_max - (n - 1)
            shifted = _shifted(h, lag)

            K = 5
            K = min(K, n)
            donor_candidates = np.argpartition(-shifted, K - 1)[:K]
            receiver_candidates = np.argpartition(shifted, K - 1)[:K]

            donor = None
            receiver = None
            for d in donor_candidates:
                if h[d] > 1e-12:
                    donor = d
                    break
            for r in receiver_candidates:
                if h[r] < 1.0 - 1e-12:
                    receiver = r
                    break
            if donor is None or receiver is None or donor == receiver:
                donor, receiver = random.sample(range(n), 2)
        else:
            donor, receiver = random.sample(range(n), 2)

        # ensure we move mass from donor to receiver
        max_delta = min(h[donor], 1.0 - h[receiver])
        if max_delta <= 0.0:
            # try opposite direction
            max_delta = min(h[receiver], 1.0 - h[donor])
            if max_delta <= 0.0:
                continue
            donor, receiver = receiver, donor

        # pick a delta (scaled)
        delta = random.uniform(0.05, 0.5) * max_delta if random.random() < 0.35 else random.uniform(0.0, max_delta) * step_scale
        if delta <= 0.0:
            continue

        # propose move
        h[donor] -= delta
        h[receiver] += delta

        new_c, new_corr = _c5_metric_and_corr(h)
        dC = new_c - cur_c
        if dC < 0.0 or random.random() < math.exp(-dC / (T + 1e-12)):
            cur_c = new_c
            corr = new_corr
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[donor] += delta
            h[receiver] -= delta

    return best_h, best_c


def _local_refine(h: np.ndarray, max_iter: int = 5000):
    """Deterministic hill‑climbing with very small redistributions."""
    n = h.size
    cur_c = _c5_metric(h)
    for _ in range(max_iter):
        i, j = random.sample(range(n), 2)
        # direction that respects bounds
        if h[i] > 0.0 and h[j] < 1.0:
            donor, receiver = i, j
        elif h[j] > 0.0 and h[i] < 1.0:
            donor, receiver = j, i
        else:
            continue
        max_delta = min(h[donor], 1.0 - h[receiver])
        if max_delta <= 0.0:
            continue
        delta = random.uniform(0.0, max_delta) * 0.05
        if delta == 0.0:
            continue
        h[donor] -= delta
        h[receiver] += delta
        new_c = _c5_metric(h)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
        else:
            h[donor] += delta
            h[receiver] -= delta
    return h, cur_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C5 bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # use a finer grid than the parent construction
    n = 200
    if n % 2:
        n += 1
    h0 = _initial_vector(n)

    # primary simulated‑annealing phase
    primary_time = total_budget * 0.80
    best_h, best_c = _anneal_targeted(h0, primary_time)

    # random‑restart phase
    elapsed = time.time() - start_all
    while total_budget - elapsed > 12.0:          # keep a few seconds for polishing
        # binary (0/1) start respecting the sum constraint
        target = n / 2.0
        ones = int(target)
        frac = target - ones
        h_rand = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        h_rand[perm[:ones]] = 1.0
        if frac > 0:
            h_rand[perm[ones]] = frac
        slice_time = min((total_budget - elapsed) * 0.30, 35.0)
        cand_h, cand_c = _anneal_targeted(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        elapsed = time.time() - start_all

    # final deterministic polishing
    best_h, best_c = _local_refine(best_h, max_iter=8000)

    # enforce exact sum (tiny rounding errors)
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    np.clip(best_h, 0.0, 1.0, out=best_h)

    try:
        final_c = evaluate_erdos_solution(best_h)  # provided helper, if any
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
