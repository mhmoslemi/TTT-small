def random_start(n):
    # start with random uniform values.
    h = np.random.rand(n)
    # Clip to [0,1] already.
    # Adjust sum to target using scaling then adjustment.
    target = n / 2.0
    # scaling factor to bring sum close to target.
    cur_sum = h.sum()
    if cur_sum > 0:
        h *= target / cur_sum
    # after scaling, some may exceed 1 due to scaling up? Actually scaling down. If cur_sum > target, scaling factor <1; stays below 1. If cur_sum < target, scaling factor >1 may cause some >1. Clip then adjust iteratively.
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    # iterative correction similar to earlier code.
    while abs(cur_sum - target) > 1e-12:
        diff = cur_sum - target
        if diff > 0:
            cand = np.where(h > 1e-12)[0]
            i = random.choice(cand.tolist())
            delta = min(h[i], diff)
            h[i] -= delta
        else:
            cand = np.where(h < 1 - 1e-12)[0]
            i = random.choice(cand.tolist())
            delta = min(1 - h[i], -diff)
            h[i] += delta
        cur_sum = h.sum()
    return h

def hill_climb(h, time_limit):
    best_h = h.copy()
    best_c5 = c5_of(h)
    start_time = time.time()
    T = 0.01
    decay = 0.999
    iter_count = 0
    while time.time() - start_time < time_limit:
        # Random move: choose i and j
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1 - hi, hj)
        max_down = min(hi, 1 - hj)
        if max_up + max_down <= 0:
            continue
        delta = (random.random() * (max_up + max_down)) - max_down
        if abs(delta) < 1e-12:
            continue
        nh = h.copy()
        nh[i] = hi + delta
        nh[j] = hj - delta
        new_c5 = c5_of(nh)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = nh.copy()
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            h = nh
            cur_c5 = new_c5
        iter_count += 1
        if iter_count % 2000 == 0:
            T *= decay
    return best_h, best_c5

def greedy_descent_step(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx_best = np.argmax(corr)
    max_corr = corr[idx_best]
    s = idx_best - (n - 1)
    # define slices
    if s >= 0:
        i_inds = np.arange(n - s)
        p_inds = i_inds + s
    else:
        i_inds = np.arange(-s, n)
        p_inds = i_inds + s
    hi = h[i_inds]
    hp = h[p_inds]
    d = 1.0 + (hi - hp)            # coefficient
    max_move = np.minimum(hi, 1 - hp)
    improvement = max_move * d
    # We will select index with largest improvement > epsilon
    if improvement.size == 0:
        return h, max_corr
    idx = np.argmax(improvement)
    best_impr = improvement[idx]
    if best_impr < 1e-12:

                h[i] -= delta
            else:
                cand = np.where(h < 1 - 1e-12)[0]
                i = random.choice(cand.tolist())
                delta = min(1 - h[i], -diff)
                h[i] += delta
            cur_sum = h.sum()

    # --------------------------------------------------------------
    # Helper utilities
    # --------------------------------------------------------------
    dx = 2.0 / n

    def c5_of(v):
        corr = np.correlate(v, 1 - v, mode="full")
        return float(corr.max() * dx)

    # --------------------------------------------------------------
    # Optional deterministic starter (triangular shape)
    # --------------------------------------------------------------
    x = (np.arange(n) + 0.5) * dx
    tri = np.minimum(x, 2 - x)          # exact sum == target for this discretisation
    tri_c5 = c5_of(tri)
    cur_c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5
    if tri_c5 < best_c5:
        best_h = tri.copy()
        best_c5 = tri_c5
        cur_c5 = tri_c5
        h = tri.copy()

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    T = max(1e-5, cur_c5 * 0.05)        # initial temperature
    decay = 0.9995
    it = 0
    no_improve = 0
    max_no_improve = 3000               # trigger restart
    L = 3                               # number of worst shifts to inspect

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # ----- analyse current worst shifts -----
        corr = np.correlate(h, 1 - h, mode="full")
        if L < len(corr):
            worst_idxs = np.argpartition(corr, -L)[-L:]
        else:
            worst_idxs = np.arange(len(corr))

        best_candidate = None
        best_candidate_c5 = cur_c5

        # ----- try targeted moves for each of the L worst shifts -----
        for idx_max in worst_idxs:
            s = int(idx_max) - (n - 1)                 # shift attaining the max
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:
                s_abs = -s
                i_inds = np.arange(s_abs, n)
                p_inds = i_inds + s

            hi = h[i_inds]
            hp = h[p_inds]
            d = 1.0 + (hi - hp)                        # linear improvement coefficient
            max_move = np.minimum(hi, 1.0 - hp)        # feasible ε
            improve = max_move * d
            if improve.size == 0:
                continue
            idx_best = int(np.argmax(improve))
            if improve[idx_best] <= 1e-12:
                continue
            eps = max_move[idx_best]                  # full feasible move
            if eps <= 1e-12:
                continue
            i0 = i_inds[idx_best]
            p0 = p_inds[idx_best]
            nh = h.copy()
            nh[i0] -= eps
            nh[p0] += eps
            new_c5 = c5_of(nh)
            if new_c5 < best_candidate_c5:
                best_candidate_c5 = new_c5
                best_candidate = nh

        # ----- accept the best targeted move if any -----
        if best_candidate is not None:
            if (best_candidate_c5 < cur_c5 or
                random.random() < math.exp(-(best_candidate_c5 - cur_c5) / max(T, 1e-12))):
                h = best_candidate
                cur_c5 = best_candidate_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            no_improve += 1

        # ----- random two‑coordinate perturbation -----
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)          # increase i, decrease j
        max_down = min(hi, 1.0 - hj)        # decrease i, increase j
        if max_up + max_down > 1e-12:
            delta = (random.random() * (max_up + max_down)) - max_down
            if abs(delta) > 1e-14:
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_c5 = c5_of(nh)
                if (new_c5 < cur_c5 or
                    random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12))):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        # ----- restart if stuck for a while -----
        if no_improve >= max_no_improve:
            # new random feasible point
            h = np.random.rand(n)
            cur_sum = h.sum()
            if cur_sum > 0:
                h *= target / cur_sum
            h = np.clip(h, 0.0, 1.0)
            cur_sum = h.sum()
            while abs(cur_sum - target) > 1e-9:
                diff = cur_sum - target
                if diff > 0:
                    cand = np.where(h > 1e-12)[0]
                    k = random.choice(cand.tolist())
                    delta = min(h[k], diff)
                    h[k] -= delta
                else:
                    cand = np.where(h < 1 - 1e-12)[0]
                    k = random.choice(cand.tolist())
                    delta = min(1 - h[k], -diff)
                    h[k] += delta
                cur_sum = h.sum()
            cur_c5 = c5_of(h)
            no_improve = 0
            T = max(1e-5, cur_c5 * 0.05)

        # ----- temperature schedule -----
        it += 1
        if it % 500 == 0:
            T *= decay

    # --------------------------------------------------------------
    # Final correction of sum (tiny numerical drift)
    # --------------------------------------------------------------
    err = best_h.sum() - target
    if abs(err) > 1e-8:
        for _ in range(500):
            if abs(err) < 1e-12:
                break
            if err > 0:
                cand = np.where(best_h > 1e-12)[0]
                if cand.size == 0:
                    break
                k = random.choice(cand.tolist())
                delta = min(best_h[k], err)
                best_h[k] -= delta
                err -= delta
            else:
                cand = np.where(best_h < 1 - 1e-12)[0]
                if cand.size == 0:
                    break
                k = random.choice(cand.tolist())
                delta = min(1 - best_h[k], -err)
                best_h[k] += delta
                err += delta

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
