import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ------------------------------------------------------------
# Feasible random vector (continuous)
# ------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ------------------------------------------------------------
# C5 evaluation (max overlap)
# ------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ------------------------------------------------------------
# Gradient of the correlation for the shift attaining the max
# ------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ------------------------------------------------------------
# Projected (sub)gradient descent on the max‑correlation objective
# ------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 1200, step0: float = 0.6,
                 decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ------------------------------------------------------------
# Smooth max (log‑sum‑exp) used by SLSQP
# ------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


# ------------------------------------------------------------
# SLSQP refinement on the smooth max objective
# ------------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 40.0, max_iter: int = 800) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ------------------------------------------------------------
# Binary hill‑climbing (swap 1↔0) to locate a good discrete pattern
# ------------------------------------------------------------
def _binary_hill_climb(rng: np.random.RandomState, n: int,
                       target: float, dx: float,
                       time_budget: float) -> tuple[np.ndarray, float]:
    """Return best binary (0/1) vector and its C5 value found within time_budget."""
    best_val = np.inf
    best_vec = None
    start = time.time()
    target_int = int(round(target))

    # inner helper: compute C5 for a binary vector
    def eval_bin(vec):
        return _compute_c5(vec.astype(float), dx)

    while time.time() - start < time_budget:
        # random binary start
        h_bin = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=target_int, replace=False)
        h_bin[ones] = 1.0
        cur_val = eval_bin(h_bin)

        # local hill climbing – limited number of swaps
        for _ in range(600):
            idx1 = rng.choice(np.where(h_bin == 1)[0])
            idx2 = rng.choice(np.where(h_bin == 0)[0])
            # swap
            h_bin[idx1] = 0.0
            h_bin[idx2] = 1.0
            new_val = eval_bin(h_bin)
            if new_val < cur_val:
                cur_val = new_val
                # keep swap
            else:
                # revert
                h_bin[idx1] = 1.0
                h_bin[idx2] = 0.0

        if cur_val < best_val:
            best_val = cur_val
            best_vec = h_bin.copy()
            # early exit if we already beat the target record
            if best_val < 0.3808:
                break
    return best_vec, best_val
