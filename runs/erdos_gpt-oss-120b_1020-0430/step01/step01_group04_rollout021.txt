def project_onto_feasible(y, target):
    # Find lambda such that sum(clip(y+lambda, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    # Expand bounds to be safe:
    lo = -np.max(y) - 1.0
    hi = 1 - np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo+hi)/2.0, 0.0, 1.0)
    # due to numeric may not be exactly target; adjust slight residual
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute small diff to non-bound entries
        mask = (x > 0) & (x < 1)
        if mask.any():
            adjust = diff / mask.sum()
            x[mask] += adjust
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust uniformly (will violate bound but minimal)
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x
```

def upsample_vector(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 0.5 * (2.0/old_n)
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 0.5 * (2.0/new_n)
    up = np.interp(new_x, old_x, arr, left=arr[0], right=arr[-1])
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()

import time
import numpy as np

# ------------------------------------------------------------
# Core evaluation helpers
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 using FFT‑based full correlation."""
    n = h.shape[0]
    size = 2 * n - 1
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    conv = np.fft.ifft(H * np.conj(G)).real
    return float(conv.max() * (2.0 / n))


def correlation_full(h: np.ndarray) -> np.ndarray:
    """Return the un‑scaled full correlation of h with (1‑h)."""
    n = h.shape[0]
    size = 2 * n - 1
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    return np.fft.ifft(H * np.conj(G)).real


# ------------------------------------------------------------
# Projection onto the feasible set {0≤h≤1, sum(h)=n/2}
# ------------------------------------------------------------
def project_onto_feasible(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection onto box [0,1] and a single linear equality."""
    # binary search for λ in h = clip(y+λ,0,1) with sum = target_sum
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction (rare)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x


# ------------------------------------------------------------
# Gradient of the correlation at a given shift
# ------------------------------------------------------------
def gradient_at_shift(h: np.ndarray, k: int) -> np.ndarray:
    """∇_h  Σ_i h_i (1‑h_{i+k})  for a fixed integer shift k."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        if k < n:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
    else:
        s = -k
        if s < n:
            g[s:] += 1.0 - h[: n - s]
            g[: n - s] -= h[s:]
    return g


# ------------------------------------------------------------
# Simple symmetric shapes
# ------------------------------------------------------------
def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangle of height 1 on [0,2] (integral =1)."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    h = np.minimum(x, 2.0 - x)
    target = n / 2.0
    # enforce the exact integral constraint
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_vector(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old grid to new_n points, re‑normalised."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 0.5 * (2.0 / old_n)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 0.5 * (2.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

# Local optimisation based on the exact max objective
# ------------------------------------------------------------
def local_opt_max(h0: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator):
    """Projected gradient descent on the true max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.2
    min_step = 1e-6
    top_k = 5                     # number of shifts whose gradients we average

    while time.time() < deadline and step > min_step:
        conv = correlation_full(h)
        # pick the *top_k* shifts (may include ties)
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            grad += gradient_at_shift(h, k)
        grad /= len(idxs)

        if np.linalg.norm(grad) < 1e-12:
            break

        cur_c5 = conv.max() * (2.0 / n)
        improved = False

        for _ in range(20):                     # back‑track line search
            y = h - step * grad
            y = project_onto_feasible(y, target_sum)
            new_c5 = compute_c5(y)

            if new_c5 < cur_c5 - 1e-9:
                h = y
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = y.copy()
                step *= 1.1
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            # small random shake to escape possible plateaux
            eps = rng.uniform(-0.02, 0.02, size=n)
            y = h + eps
            y = project_onto_feasible(y, target_sum)
            y_c5 = compute_c5(y)
            if y_c5 < best_c5:
                h = y
                best_h = y.copy()
                best_c5 = y_c5
            else:
                # if shake does not help, terminate this run
                break

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    start_time = time.time()
    time_limit = start_time + budget_s * 0.95   # safety margin

    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    # candidate discretisations (even numbers are convenient)

    if parent_h is not None:

        if pn not in candidate_ns:
            candidate_ns = [pn] + candidate_ns

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in candidate_ns:
        if time.time() > time_limit:
            break

        target_sum = n / 2.0

        # initialise base shape
        if parent_h is not None:

            else:
                base = upsample_vector(parent_h, n)
        else:

        attempts = 4
        for _ in range(attempts):
