import time, math, random
import numpy as np

def _c5_metric(h):
    """C₅ value for a sampled step‑function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _initial_vector(target_n=None):
    """Construct a feasible start vector."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        n = target_n or max(200, ((n0 + 9) // 10) * 10)
        if n % 2:
            n += 1
        # linear interpolation onto the new grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        # simple triangular “tent’’ shape, already has integral 1
        n = target_n or 200
        if n % 2:
            n += 1
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.minimum(x, 2.0 - x)
    # enforce exact integral (=1)  →  sum(h) = n/2
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _gradient_overlap(h, offset):
    """∂/∂h_i of the overlap term at the worst shift."""
    n = h.shape[0]
    grad = np.zeros(n, dtype=h.dtype)
    if offset >= 0:
        # i‑offset term
        grad[offset:] = 1.0 - h[:n - offset]
        # i+offset term
        grad[:n - offset] -= h[offset:]
    else:
        s = -offset
        grad[:n - s] = 1.0 - h[s:]
        grad[s:] -= h[:n - s]
    return grad


def _anneal(start_h, max_seconds):
    """Simulated‑annealing with a dedicated move that attacks the worst lag."""
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0, T_end = 0.04, 1e-6          # temperature schedule
    max_step = 0.18                 # max absolute mass moved in one step
    prob_targeted = 0.35            # how often we use the “worst‑lag’’ move

    while time.time() - start < max_seconds:
        progress = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** progress

        if random.random() < prob_targeted:
            # ---- targeted move ------------------------------------------------
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_idx = int(np.argmax(corr))
            offset = max_idx - (n - 1)

            grad = _gradient_overlap(h, offset)

            # indices that can be increased (grad < 0) / decreased (grad > 0)
            inc_cands = np.where((grad < -1e-12) & (h < 1.0))[0]
            dec_cands = np.where((grad >  1e-12) & (h > 0.0))[0]

            if inc_cands.size == 0 or dec_cands.size == 0:
                # fall back to a random pair move
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                lo = max(-hi, hj - 1.0)
                hi_ = min(1.0 - hi, hj)
                if lo >= hi_:
                    continue
                delta = random.uniform(lo, hi_)
                h[i] = hi + delta
                h[j] = hj - delta
                new_c = _c5_metric(h)
                dC = new_c - cur_c
                if dC < 0 or random.random() < math.exp(-dC / T):
                    cur_c = new_c
                    if new_c < best_c:
                        best_c = new_c
                        best_h = h.copy()
                else:
                    h[i] = hi
                    h[j] = hj
                continue

            i = inc_cands[np.argmin(grad[inc_cands])]   # most negative grad
            j = dec_cands[np.argmax(grad[dec_cands])]   # most positive grad

            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_ = min(1.0 - hi, hj)
            if lo >= hi_:
                continue
            # move mass from j to i (delta > 0)
            delta = min(hi_, max_step)
            if delta <= 0:
                continue
            h[i] = hi + delta
            h[j] = hj - delta
            new_c = _c5_metric(h)
            # accept only if it helps; otherwise reject
            if new_c < cur_c:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] = hi
                h[j] = hj
        else:
            # ---- simple random pair move --------------------------------------
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_ = min(1.0 - hi, hj)
            if lo >= hi_:
                continue
            # modest step most of the time
            lo_cap = max(lo, -max_step)
            hi_cap = min(hi_, max_step)
            if lo_cap < hi_cap:
                delta = random.uniform(lo_cap, hi_cap)
            else:
                delta = random.uniform(lo, hi_)
            h[i] = hi + delta
            h[j] = hj - delta
            new_c = _c5_metric(h)
            dC = new_c - cur_c
            if dC < 0 or random.random() < math.exp(-dC / T):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] = hi
                h[j] = hj

    return best_h, best_c
