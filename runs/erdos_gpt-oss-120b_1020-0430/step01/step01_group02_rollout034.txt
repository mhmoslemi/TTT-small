import numpy as np

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """Project x onto {0<=x<=1, sum(x)=target}."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _shift_array(a, offset):
    """Return array b where b[i] = a[i+offset] if inside range, else 0."""
    n = a.shape[0]
    b = np.empty_like(a)
    if offset >= 0:
        b[: n - offset] = a[offset:]
        b[n - offset :] = 0.0
    else:
        off = -offset
        b[off:] = a[: n - off]
        b[:off] = 0.0
    return b

            break

        start_vecs.append(sin2)

        # if external vector matches this resolution, use it

            start_vecs.append(external_init)

        # a few random starts
        for _ in range(3):
            rand = rng.random(n)
            rand = rand * (target / rand.sum())
            start_vecs.append(rand)

        local_best_c5 = np.inf

            c0 = _evaluate_c5(h0)

import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a discrete step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x, target):
    """Project onto {0≤x≤1, sum(x)=target} (simple redistribution)."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution among free variables
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _shift_array(a, offset):
    """Return a shifted copy: b[i]=a[i+offset] if inside, else 0."""
    n = a.shape[0]
    b = np.empty_like(a)
    if offset >= 0:
        b[: n - offset] = a[offset:]
        b[n - offset :] = 0.0
    else:
        off = -offset
        b[off:] = a[: n - off]
        b[:off] = 0.0
    return b
