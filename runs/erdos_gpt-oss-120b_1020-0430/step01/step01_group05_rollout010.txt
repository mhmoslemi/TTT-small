    n = h.shape[0]
    N = 2 * n

    H = np.fft.rfft(h, N)

    B = np.fft.rfft((1.0 - h)[::-1], N)

    corr = conv[:2*n - 1]

import time, math, random
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n intersected with the hyperplane sum(x)=target."""
    lo, hi = -2.0, 2.0  # wide enough to bracket the correct lambda
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5_fft(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx using FFT (discrete version)."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, N)
    B = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * B, N)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape (peak 1 at centre, linear to 0 at edges)."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def resample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of old_h (defined on [0,2)) to new_n points, then project."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, old_h)
    return project_onto_bounds_and_sum(h_new, new_n / 2.0)

# Simulated annealing core
# ------------------------------------------------------------
def anneal(start_h: np.ndarray, n: int, target_sum: float, time_limit: float):
    """Return (best_h, best_val) after annealing for at most time_limit seconds."""
    cur_h = project_onto_bounds_and_sum(start_h, target_sum)
    cur_val = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    # schedule parameters
    T_start, T_end = 0.02, 1e-6
    max_step, min_step = 0.2, 0.001
    large_move_prob = 0.02

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        # exponential cooling
        T = T_start * (T_end / T_start) ** frac
        step = max_step * (1.0 - frac) + min_step * frac

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # feasible interval for delta (mass transferred from j to i)
        lo = max(-cur_h[i], cur_h[j] - 1.0)
        hi = min(1.0 - cur_h[i], cur_h[j])
        if lo >= hi:
            continue

        if np.random.rand() < large_move_prob:
            delta = np.random.uniform(lo, hi)          # unrestricted move
        else:
            lo = max(lo, -step)
            hi = min(hi, step)
            if lo >= hi:
                continue
            delta = np.random.uniform(lo, hi)          # small step

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = compute_c5_fft(new_h)
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val
