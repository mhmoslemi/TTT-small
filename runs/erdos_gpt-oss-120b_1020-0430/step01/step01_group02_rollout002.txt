        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)

        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            continue

        delta = max_move if rng.random() < 0.1 else rng.random() * max_move

        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        new_val = _evaluate_c5(new_h)

        # Metropolis accept/reject
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                last_improve = iter_cnt

        iter_cnt += 1

        # restart if no improvement for a long stretch
        if iter_cnt - last_improve > 200_000:
            cur_h = _random_start(n, target, rng)
            cur_val = _evaluate_c5(cur_h)
            last_improve = iter_cnt

    # ------------------------------------------------------------------
    # 2. Local refinement using SLSQP (few restarts)
    # ------------------------------------------------------------------
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n

    remain = budget_s - (time.time() - start_time) - 5.0
    attempts = min(max(1, int(remain // 1.0)), 8)

    refined_h = best_h.copy()
    refined_val = best_val

    for _ in range(attempts):
        x0 = refined_h + rng.normal(scale=0.05, size=n)
        x0 = np.clip(x0, 0.0, 1.0)
        x0 = _project_sum(x0, target)

        res = minimize(
            obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 250, "ftol": 1e-9, "disp": False},
        )
        cand = res.x if res.success else x0
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_sum(cand, target)
        cand_val = _evaluate_c5(cand)

        if cand_val < refined_val:
            refined_h, refined_val = cand, cand_val

    # ------------------------------------------------------------------
    # 3. Deterministic greedy hill climbing (fine‑grained)
    # ------------------------------------------------------------------
    h = refined_h.copy()
    c5 = refined_val
    deadline = start_time + budget_s - 0.5
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)

        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue

        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)

        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5

    # ------------------------------------------------------------------
    # 4. Final projection and return
    # ------------------------------------------------------------------
    h = _project_sum(np.clip(h, 0.0, 1.0), target)
    final_c5 = float(_evaluate_c5(h))

    return h, final_c5, n
