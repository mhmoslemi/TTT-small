import cvxpy as cp
def solve_mip(n, time_limit):
    x = cp.Variable(n, boolean=True)
    y = cp.Variable((n,n), boolean=True)
    constraints = []
    # sum constraint
    constraints.append(cp.sum(x) == n//2)
    # y[i,i] == x[i]
    for i in range(n):
        constraints.append(y[i,i] == x[i])
    # product constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(y[i,j] <= x[i])
            constraints.append(y[i,j] <= x[j])
            constraints.append(y[i,j] >= x[i] + x[j] - 1)
    # shift constraints
    t = cp.Variable(integer=True) # maybe integer
    constraints.append(t >= 0)
    for s in range(n):
        # overlapping length n-s
        indices = range(n - s)
        expr = cp.sum([y[i, i+s] for i in indices])
        constraints.append(expr >= t)
    # objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='CBC', maxSeconds=time_limit, verbose=False)
    # after solve, extract x, t
    if x.value is None:
        return None
    # round x to binary
    x_bin = np.rint(x.value).astype(int)
    t_val = int(round(t.value))
    return x_bin, t_val
```

import numpy as np, cvxpy as cp, time, math

def solve_mip(n, time_limit):
    # Variables
    x = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = []
    constraints.append(cp.sum(x) == n/2)
    # product constraints
    for i in range(n):
        constraints.append(y[i,i] == x[i])
        for j in range(i+1, n):
            constraints.append(y[i,j] <= x[i])
            constraints.append(y[i,j] <= x[j])
            constraints.append(y[i,j] >= x[i] + x[j] - 1)
    # O_s constraints
    for shift in range(-n+1, n):
        if shift >= 0:
            length = n - shift
            xs = x[:length]
            # sum x_i over overlapping region
            x_sum = cp.sum(xs)
            # y diagonal sum
            y_sum = cp.sum(cp.hstack([y[i, i+shift] for i in range(length)]))
        else:
            shift_abs = -shift
            length = n - shift_abs
            xs = x[shift_abs:]  # length same
            x_sum = cp.sum(xs)
            y_sum = cp.sum(cp.hstack([y[i, i+shift_abs] for i in range(length)]))
        O = x_sum - y_sum
        constraints.append(O <= t)
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver='CBC', maxSeconds=time_limit, verbose=False)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
    h = np.rint(x.value).astype(float)  # convert binary
    t_val = float(t.value)
    return h, t_val

def subgradient_descent(h, target_sum, max_iter, time_budget):
    start = time.time()
    n = len(h)
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
        # compute O_s
        corr = np.correlate(h, 1-h, mode='full')
        O_vals = corr  # not scaled
        s_idx = np.argmax(O_vals)
        s = s_idx - (n - 1)   # shift
        # compute gradient for this shift
        grad = np.zeros_like(h)
        if s >= 0:
            # indices 0..n-s-1
            i_range = np.arange(n - s)
            grad[i_range] = 1.0 - h[i_range + s]
            # indices s..n-1
            j_range = np.arange(s, n)
            grad[j_range] -= h[j_range - s]
        else:
            s_abs = -s
            i_range = np.arange(n - s_abs)
            grad[i_range + s_abs] = 1.0 - h[i_range]
            grad[i_range] -= h[i_range + s_abs]
        # step size schedule
        alpha = 0.1 / (1 + it * 0.001)  # decreasing
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, target_sum)
        # compute new max O
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_max = new_corr.max()
        # accept if improvement
        if new_max < O_vals.max():
            h = h_new
        # else maybe still accept small probability? Not needed.
    return h

import random
import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all top‑level as required)
# ------------------------------------------------------------
def c5_of(v: np.ndarray) -> float:
    """Maximum overlap integral (unscaled) * dx."""
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(corr.max() * (2.0 / len(v)))   # dx = 2 / n


def project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the intersection of the box [0,1]^n and the hyperplane sum(v)=target.
    The solution has the form clip(v + λ,0,1) for a suitable λ found by binary search.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def gradient_step(h: np.ndarray, target_sum: float, step: float) -> np.ndarray:
    """
    One sub‑gradient step on the worst shift.
    Returns a new feasible vector (still respects bounds and sum).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    s_idx = int(np.argmax(corr))
    shift = s_idx - (n - 1)

    grad = np.zeros_like(h)

    if shift >= 0:
        L = n - shift
        grad[:L] += 1.0
        grad[:L] -= h[shift:]          # -h_{i+shift}
        grad[shift:] -= h[:L]          # -h_{i-shift}
    else:
        s = -shift
        L = n - s
        grad[s:] += 1.0
        grad[s:] -= h[:L]              # -h_{i-s}
        grad[:L] -= h[s:]              # -h_{i+s}

    h_new = h - step * grad
    return project_sum_box(h_new, target_sum)
