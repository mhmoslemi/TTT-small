import time
import numpy as np

# ------------------------------------------------------------
#  Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 80) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    return _project_onto_capped_simplex(rng.rand(n), target)


# ------------------------------------------------------------
#  Objective and gradient (smooth max of correlations)
# ------------------------------------------------------------
def _smooth_max_and_grad(v: np.ndarray, dx: float, alpha: float):
    n = v.shape[0]
    corr = np.correlate(v, 1.0 - v, mode='full')
    c = corr * dx                       # true correlations
    maxc = c.max()
    # softmax weights (normalize)
    w = np.exp(alpha * (c - maxc))
    w_sum = w.sum()
    w /= w_sum

    # convolution based gradient:
    # conv1 = sum_s w_s * v_ext[j+s]   (v_ext padded with zeros)
    # conv2 = sum_s w_s * v[j-s]
    w_rev = w[::-1]
    v_ext = np.concatenate([v, np.zeros(n - 1, dtype=v.dtype)])

    conv1_full = np.convolve(w_rev, v_ext, mode='full')
    start = n - 1
    conv1 = conv1_full[start:start + n]

    conv2_full = np.convolve(w, v, mode='full')
    conv2 = conv2_full[start:start + n]

    grad = (1.0 - conv1 - conv2) * dx
    smooth_val = maxc + np.log(w_sum) / alpha
    return smooth_val, grad


# ------------------------------------------------------------
#  Projected (sub)gradient descent on the *true* max correlation
# ------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 600, step0: float = 0.5,
                 decay: float = 0.997) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ------------------------------------------------------------
#  Projected gradient descent on the *smooth* max objective
# ------------------------------------------------------------
def _gd_smooth(v0: np.ndarray, target: float, dx: float,
               alphas, steps_per_alpha,
               init_step: float = 0.8, decay: float = 0.997) -> np.ndarray:
    v = v0.copy()
    step = init_step
    for alpha in alphas:
        for _ in range(steps_per_alpha):
            _, grad = _smooth_max_and_grad(v, dx, alpha)
            v = v - step * grad
            v = _project_onto_capped_simplex(v, target)
            step *= decay
    return v


# ------------------------------------------------------------
#  True C5 value for a given discretisation
# ------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


# ------------------------------------------------------------
#  Upscale a vector to a desired length (linear interpolation)
# ------------------------------------------------------------
def _upscale(vec: np.ndarray, n: int) -> np.ndarray:
    m = len(vec)
    if m == n:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, vec)

    # Load parent solution if present
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    # Choose a moderately fine grid (larger grids give more freedom)
    n = max(500, len(base_h) if base_h is not None else 0)
    dx = 2.0 / n
    target_sum = n / 2.0

    # Initial candidate
    if base_h is not None:
        h0 = _upscale(base_h, n)
        h0 = _project_onto_capped_simplex(h0, target_sum)
        best_h = h0.copy()
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = None
        best_val = float('inf')

    # Parameters for the smooth‑max schedule
    smooth_alphas = [5.0, 15.0, 40.0, 80.0, 160.0]
    steps_per_alpha = 120   # total ≈ 600 GD steps per restart

    restart = 0
    # Main optimisation loop – stop a bit before the budget ends
    while time.time() - start_time < budget_s * 0.92:
        # ----------------------------------------------------
        #  Initialise a starting point
        # ----------------------------------------------------
        if best_h is not None and rng.rand() < 0.2:
            h_start = best_h + rng.normal(scale=0.03, size=n)
        else:
            h_start = _random_feasible(rng, n, target_sum)

        # ----------------------------------------------------
        #  Smooth‑max gradient descent
        # ----------------------------------------------------
        h_opt = _gd_smooth(h_start, target_sum, dx,
                           alphas=smooth_alphas,
                           steps_per_alpha=steps_per_alpha,
                           init_step=0.9,
                           decay=0.998)

        # ----------------------------------------------------
        #  Refine against the true max correlation
        # ----------------------------------------------------
        h_opt = _gd_optimize(h_opt, target_sum,
                             max_iter=400,
                             step0=0.35,
                             decay=0.998)

        # ----------------------------------------------------
        #  Final projection and evaluation
        # ----------------------------------------------------
        h_opt = _project_onto_capped_simplex(h_opt, target_sum)
        val = _compute_c5(h_opt, dx)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            # early exit if we already beat the target record
            if best_val < 0.3808:
                break

        restart += 1

    # Fallback (should never be needed)
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
