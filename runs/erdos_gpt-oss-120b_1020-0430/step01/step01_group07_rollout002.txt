def neighbor_sum(h, shift):
    n = len(h)
    s = np.zeros(n)
    if shift >=0:
        s[:n-shift] += h[shift:]
        s[shift:] += h[:n-shift]
    else:
        s[-shift:] += h[:n+shift]  # because -shift > 0
        s[:n+shift] += h[-shift:]
    return s

        else:   # shift < 0
            d = -shift
            s[d:] += vec[:nlen - d]
            s[:nlen - d] += vec[d:]
        return s

    # --------------------------------------------------------------
    # 4. Record the best solution seen so far
    best_h = h.copy()
    best_c = compute_c(best_h)

    # --------------------------------------------------------------
    # 5. Greedy improvement phase (≈60 % of the budget)
    greedy_budget = total_budget * 0.6
    while time.time() - start_all < greedy_budget:
        q = np.correlate(h, h, mode='full')
        min_q = q.min()
        cur_c = 1.0 - min_q * dx
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

        # worst lag (the one giving the largest overlap)
        idx_min = np.argmin(q)
        shift = idx_min - (n - 1)          # shift ∈ [-(n-1), n-1]

        # neighbour‑sum vector for that lag
        s = neighbor_sum(h, shift)

        # candidate indices – a short list keeps the loop cheap
        desc = np.argsort(-s)[:30]
        asc  = np.argsort(s)[:30]

        moved = False
        for i in desc:
            if h[i] >= 1.0 - 1e-12:
                continue
            for j in asc:
                if i == j or h[j] <= 1e-12:
                    continue
                diff_s = s[i] - s[j]
                if diff_s <= 1e-12:
                    continue
                max_delta = min(1.0 - h[i], h[j])
                # a modest step keeps the search stable
                delta = min(max_delta, 0.5 * diff_s)
                if delta <= 1e-12:
                    continue
                h[i] += delta
                h[j] -= delta
                moved = True
                break
            if moved:
                break

        if not moved:
            # fallback tiny random redistribution
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_ = min(1.0 - hi, hj)
            if lo < hi_:
                delta = random.uniform(lo, hi_) * 0.1
                h[i] = hi + delta
                h[j] = hj - delta

    # --------------------------------------------------------------
    # 6. Simulated‑annealing refinement (use whatever time remains)
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 2.0:
        T0, T_end = 0.02, 1e-6
        sa_start = time.time()
        cur_c = compute_c(h)

        while time.time() - start_all < total_budget:
            # temperature schedule (log‑linear)
            prog = (time.time() - sa_start) / remaining
            T = T0 * (T_end / T0) ** prog if remaining > 0 else T_end

            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_ = min(1.0 - hi, hj)
            if lo >= hi_:
                continue
            # random step (biased towards larger moves)
            delta = random.uniform(lo, hi_) * (0.5 + 0.5 * random.random())
            h[i] = hi + delta
            h[j] = hj - delta

            new_c = compute_c(h)
            dC = new_c - cur_c
            if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
                cur_c = new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h.copy()
            else:
                h[i], h[j] = hi, hj  # reject

    # --------------------------------------------------------------
    # 7. Final tiny hill‑climb (few seconds)
    h = best_h.copy()
    cur_c = compute_c(h)
    hill_start = time.time()
    while time.time() - hill_start < 5.0:
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        lo = max(-hi, hj - 1.0)
        hi_ = min(1.0 - hi, hj)
        if lo >= hi_:
            continue
        delta = random.uniform(lo, hi_) * 0.05
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = compute_c(h)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = hi, hj

    # --------------------------------------------------------------
    # 8. Enforce the exact sum constraint (tiny rounding error correction)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # --------------------------------------------------------------
    # 9. Evaluate with the provided helper if available
    try:
        final_c = evaluate_erdos_solution(best_h)   # pre‑imported
    except Exception:
        final_c = compute_c(best_h)

    return best_h, float(final_c), int(n)
