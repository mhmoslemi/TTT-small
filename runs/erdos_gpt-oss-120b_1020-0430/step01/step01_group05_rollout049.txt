def max_correlation(h, dx):
    n=len(h)
    L = 1 << (2*n - 1).bit_length()  # next power of two >= 2n-1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(H * G, L)  # length L
    # linear convolution first 2n-1 entries
    conv = conv[:2*n-1]
    return np.max(conv) * dx

import time
import random
import numpy as np

# ---------- helper utilities ----------
def project_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n intersect sum(x)=target via bisection on Lagrange multiplier."""
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    """Linear increase to the centre, then decrease (peak =1)."""
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(xs - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_bounds_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(pi*x/2) on [0,2] (integral =1)."""
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * xs / 2.0) ** 2
    return project_bounds_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Block of height 1 on [0.5,1.5] (mass =1)."""
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx * 0.5
    h = np.where((xs >= 0.5) & (xs <= 1.5), 1.0, 0.0)
    return project_bounds_sum(h, n / 2.0)


def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation from an older grid to size n."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_bounds_sum(h, n / 2.0)

    # ---- hill‑climbing routine (single move = transfer delta between two points) ----
    def hill_climb(initial_h: np.ndarray, time_limit: float):
        nonlocal best_h, best_val, start
        h = initial_h.copy()
        f = 1.0 - h
        C = np.correlate(h, f, mode="full")
