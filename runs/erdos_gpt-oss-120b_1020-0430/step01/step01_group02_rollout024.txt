   steps_without_improve = 0

        max_move = min(best_h[src], 1.0 - best_h[dst])

        delta = rng.random() * max_move

        new_h[dst] += delta

        new_c5 = _evaluate_c5(new_h)

        # accept rule
        if new_c5 < best_c5 - 1e-12:

            best_c5 = new_c5
            steps_without_improve = 0
        else:

                best_c5 = new_c5
                steps_without_improve = 0

                steps_without_improve += 1

        # temperature decay
        T *= T_decay
        if T < 1e-8:

def _evaluate_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # water-filling style
    # repeated redistribution until diff negligible
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        per = diff / idx.size
        # disallow overshoot
        if diff > 0:
            add = np.minimum(per, 1.0 - x[idx])
            x[idx] += add
            diff -= add.sum()
        else:
            sub = np.minimum(-per, x[idx])
            x[idx] -= sub
            diff += sub.sum()
    return np.clip(x, 0.0, 1.0)

            sub = np.minimum(-per, x[idx])
            x[idx] -= sub
            diff += sub.sum()
    return np.clip(x, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise (use the provided construction if available)
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_arr = np.asarray(init_vec, dtype=np.float64)
        n = init_arr.size               # keep the original resolution
        target = n / 2.0
        h0 = init_arr.copy()
    else:
        n = 81                          # a modestly larger resolution
        target = n / 2.0
        h0 = rng.random(n)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _adjust_sum(h0, target)

    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing / targeted hill‑climb loop
    # ------------------------------------------------------------------
    T = 0.01                # initial temperature
    T_decay = 0.9996
    no_improve = 0
    it = 0
    deadline = start + budget_s - 0.5

    while time.time() < deadline:
        # ----- random pairwise mass transfer ---------------------------------
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_move = min(best_h[src], 1.0 - best_h[dst])
        if max_move > 0.0:
            delta = rng.random() * max_move
            new_h = best_h.copy()
            new_h[src] -= delta
            new_h[dst] += delta
            new_c5 = _evaluate_c5(new_h)

            if new_c5 < best_c5 - 1e-12:
                best_h, best_c5 = new_h, new_c5
                no_improve = 0
            else:
                # Metropolis acceptance
                prob = np.exp(-(new_c5 - best_c5) / T) if T > 0 else 0.0
                if rng.random() < prob:
                    best_h, best_c5 = new_h, new_c5
                    no_improve = 0
                else:
                    no_improve += 1

        # ----- temperature decay --------------------------------------------
        T = max(T * T_decay, 1e-9)

        # ----- periodic global jitter ----------------------------------------
        if it % 2000 == 0 and it > 0:
            scale = max(0.02, T * 5)
            pert = best_h + rng.normal(scale=scale, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _adjust_sum(pert, target)
            pert_c5 = _evaluate_c5(pert)
            if pert_c5 < best_c5:
                best_h, best_c5 = pert, pert_c5
                no_improve = 0

        # ----- targeted move based on the current worst lag ------------------
        if it % 1000 == 0 and it > 0:
            full_corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_idx = int(np.argmax(full_corr))
            lag = k_idx - (n - 1)                     # signed shift
            if lag >= 0:
                idxs = np.arange(n - lag)
                contrib = best_h[idxs] * (1.0 - best_h[idxs + lag])
            else:
                p = -lag
                idxs = np.arange(p, n)
                contrib = best_h[idxs] * (1.0 - best_h[idxs - p])
            src = idxs[int(np.argmax(contrib))]
            dst = rng.choice(np.delete(np.arange(n), src))
            max_move = min(best_h[src], 1.0 - best_h[dst])
            if max_move > 0.0:
                delta = rng.random() * max_move
                new_h = best_h.copy()
                new_h[src] -= delta
                new_h[dst] += delta
                new_c5 = _evaluate_c5(new_h)
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h, new_c5
                    no_improve = 0

        # ----- occasional SLSQP refinement ------------------------------------
        if it % 5000 == 0 and it > 0 and time.time() < deadline - 2:
            x0 = best_h + rng.normal(scale=0.03, size=n)
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = _adjust_sum(x0, target)

            def obj(x):
                corr = np.correlate(x, 1.0 - x, mode="full")
                return np.max(corr) * (2.0 / n)

            cons = {"type": "eq", "fun": lambda x: x.sum() - target}
            bounds = [(0.0, 1.0)] * n
            res = minimize(
                obj,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                cand = np.clip(res.x, 0.0, 1.0)
                cand = _adjust_sum(cand, target)
                cand_c5 = _evaluate_c5(cand)
                if cand_c5 < best_c5:
                    best_h, best_c5 = cand, cand_c5
                    no_improve = 0

        # ----- restart if stagnating ----------------------------------------
        if no_improve > 50000:
            new_h = rng.random(n)
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = _adjust_sum(new_h, target)
            new_c5 = _evaluate_c5(new_h)
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5
            no_improve = 0

        it += 1

    # ------------------------------------------------------------------
    # 3. Final projection and return
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
