import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x: np.ndarray, target: float, max_iter: int = 200) -> np.ndarray:
    """Project x onto the hyper‑plane sum(x)=target while staying in [0,1]."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        per = diff / idx.size
        # limited by the remaining capacity of each coordinate
        if diff > 0.0:
            add = np.minimum(per, 1.0 - x[idx])
            x[idx] += add
            diff -= add.sum()
        else:
            sub = np.minimum(-per, x[idx])
            x[idx] -= sub
            diff += sub.sum()
        if abs(diff) < 1e-12:
            break
    return np.clip(x, 0.0, 1.0)


def _pairwise_hill_climb(h, cur_c5, deadline, rng):
    """Random pairwise mass‑exchange hill climbing (accept only improvements)."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # direction: move from i to j or j to i
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
    return h, cur_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size and initialise population
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        n = 45
        h0 = rng.random(n)
    target = n / 2.0

    # ensure feasibility of the initial vector
    h0 = _adjust_sum(np.clip(h0, 0.0, 1.0), target)

    # population for a simple evolutionary search
    pop_size = 8
    population = [h0]
    scores = [_evaluate_c5(h0)]
    while len(population) < pop_size:
        cand = rng.random(n)
        cand = _adjust_sum(cand, target)
        population.append(cand)
        scores.append(_evaluate_c5(cand))

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    # ------------------------------------------------------------------
    # 2. Evolutionary search with crossover + mutation + elitism
    # ------------------------------------------------------------------
    deadline = start + budget_s - 5.0  # reserve a few seconds for final refinement
    iteration = 0
    while time.time() < deadline:
        # parent selection
        p1, p2 = rng.choice(pop_size, size=2, replace=False)
        parent1, parent2 = population[p1], population[p2]

        # uniform crossover
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent1, parent2).astype(np.float64)

        # mutation with a decaying scale
        frac = (time.time() - start) / (budget_s - 5.0)
        sigma = 0.12 * (1.0 - frac)  # linearly decaying stddev
        child += rng.normal(scale=sigma, size=n)

        # feasibility projection
        child = np.clip(child, 0.0, 1.0)
        child = _adjust_sum(child, target)

        child_score = _evaluate_c5(child)

        # replace the worst individual if we improve
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_c5:
                best_h, best_c5 = child.copy(), child_score

        # occasional random injection to maintain diversity
        if iteration % 250 == 0:
            rand = rng.random(n)
            rand = _adjust_sum(rand, target)
            rand_score = _evaluate_c5(rand)
            worst_idx = int(np.argmax(scores))
            if rand_score < scores[worst_idx]:
                population[worst_idx] = rand
                scores[worst_idx] = rand_score
                if rand_score < best_c5:
                    best_h, best_c5 = rand.copy(), rand_score

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Focused local optimisation (pairwise hill climbing + SLSQP)
    # ------------------------------------------------------------------
    # a) intensive hill climbing
    deadline = start + budget_s - 1.0
    best_h, best_c5 = _pairwise_hill_climb(best_h, best_c5, deadline, rng)

    # b) occasional short SLSQP refinement
    def obj(x):
        return _evaluate_c5(x)

    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n

    # give SLSQP a few quick tries (they are cheap for n≈45)
    for _ in range(5):
        perturb = best_h + rng.normal(scale=0.03, size=n)
        perturb = np.clip(perturb, 0.0, 1.0)
        perturb = _adjust_sum(perturb, target)
        res = minimize(
            obj,
            perturb,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 150, "ftol": 1e-9, "disp": False},
        )
        cand = _adjust_sum(np.clip(res.x, 0.0, 1.0), target)
        cand_score = _evaluate_c5(cand)
        if cand_score + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_score

    # ------------------------------------------------------------------
    # 4. Final sanity check and return
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
