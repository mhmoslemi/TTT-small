import time, numpy as np, math
from scipy.optimize import minimize

# functions: compute_c5, sum_eq_constraint, softmax_obj, slsqp_optimize

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def sum_eq_constraint(x):
    # will set global TARGET before use
    return np.sum(x) - GLOBAL_TARGET

def softmax_obj(x):
    # will set global BETA before use
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = GLOBAL_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + math.log(np.sum(np.exp(bc - m)))) / GLOBAL_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global GLOBAL_TARGET, GLOBAL_BETA
    n = h0.size
    GLOBAL_TARGET = n / 2.0
    GLOBAL_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq','fun':sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = GLOBAL_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = GLOBAL_TARGET - x.sum()
    return x

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def random_feasible_vector(n, rng):
    # start with uniform 0.5 and random transfers
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:

        max_eps = min(h[i], 1.0 - h[j])

        h[j] += eps

    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

    return h

def gradient_wrt_worst_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        forward = np.empty(n)
        forward[:n-shift] = 1.0 - h[shift:]
        forward[n-shift:] = 1.0
        backward = np.empty(n)
        backward[shift:] = h[:n-shift]
        backward[:shift] = 0.0
        g = forward - backward
    else:

def guided_move(h, shift, step, rng):
    g = gradient_wrt_worst_shift(h, shift)
    # largest positive gradient (decrease h) and most negative (increase h)
    pos = np.argmax(g)
    neg = np.argmin(g)
    if g[pos] <= 0 or g[neg] >= 0:
        return None
    max_transfer = min(h[pos], 1.0 - h[neg])
    if max_transfer <= 0:
        return None
    eps = step * max_transfer
    if eps <= 0:
        return None
    eps = min(eps, max_transfer)
    new_h = h.copy()
    new_h[pos] -= eps
    new_h[neg] += eps
    # sum unchanged, bounds respected by eps definition
    return new_h

def hill_climb_guided(h0, rng, max_time, initial_step=0.5):
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    step = initial_step
    start = time.time()
    it = 0
    last_improve_it = 0
    while time.time() - start < max_time:
        # compute worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        new_h = guided_move(cur_h, shift, step, rng)
        if new_h is None:

            new_h = perturb_h(cur_h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c - 1e-12:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:

        # shrink step periodically
        it += 1
        if it % (n * 10) == 0:

            step = max(step, 1e-4)
        # if stagnated for many iterations, re‑initialize near best
        if it - last_improve_it > n * 200:
            cur_h = perturb_h(best_h, rng, scale=step * 2.0)
            cur_c = compute_c5(cur_h)
            last_improve_it = it
    return best_h, best_c

def enforce_feasibility(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    rng = np.random.default_rng()
    for _ in range(5000):

            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

import time, math
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
GLOBAL_TARGET = 0.0
GLOBAL_BETA = 0.0


def compute_c5(h):
    """exact C5 for a discrete step function (dx = 2 / len(h))."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def sum_eq_constraint(x):
    """equality Σ x_i = GLOBAL_TARGET, used by SLSQP."""
    return np.sum(x) - GLOBAL_TARGET


def softmax_obj(x):
    """smooth log‑sum‑exp surrogate of max_k Σ_i h_i(1‑h_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = GLOBAL_BETA * corr
    m = np.max(bc)
    return (m + math.log(np.sum(np.exp(bc - m)))) / GLOBAL_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """single SLSQP run with a given beta."""
    global GLOBAL_TARGET, GLOBAL_BETA
    n = h0.size
    GLOBAL_TARGET = n / 2.0
    GLOBAL_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny correction of the integral (numerical drift)
    diff = GLOBAL_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = GLOBAL_TARGET - x.sum()
    return x


def perturb_h(h, rng, scale=0.2):
    """transfer a small amount ε from i to j (keeps the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def random_feasible_vector(n, rng):
    """simple random feasible point (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def gradient_wrt_worst_shift(h, shift):
    """gradient of Σ_i h_i(1‑h_{i+shift}) w.r.t. h (treat out‑of‑range as 0)."""
    n = h.shape[0]
    if shift >= 0:
        forward = np.empty(n)
        forward[: n - shift] = 1.0 - h[shift:]
        forward[n - shift :] = 1.0
        backward = np.empty(n)
        backward[shift:] = h[: n - shift]
        backward[:shift] = 0.0
    else:
        s = -shift
        forward = np.empty(n)
        forward[s:] = 1.0 - h[: n - s]
        forward[:s] = 1.0
        backward = np.empty(n)
        backward[: n - s] = h[s:]
        backward[n - s :] = 0.0
    return forward - backward


def guided_move(h, shift, step):
    """move ε from the index with largest positive gradient to the one with most negative."""
    g = gradient_wrt_worst_shift(h, shift)
    pos = int(np.argmax(g))
    neg = int(np.argmin(g))
    if g[pos] <= 0.0 or g[neg] >= 0.0:
        return None
    max_transfer = min(h[pos], 1.0 - h[neg])
    if max_transfer <= 0.0:
        return None
    eps = step * max_transfer
    eps = min(eps, max_transfer)
    if eps <= 0.0:
        return None
    new_h = h.copy()
    new_h[pos] -= eps
    new_h[neg] += eps
    return new_h


def hill_climb_guided(h0, rng, max_time, initial_step=0.5):
    """guided hill‑climb using the worst‑shift gradient."""
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    step = initial_step
    start = time.time()
    it = 0
    last_improve = 0

    while time.time() - start < max_time:
        # worst shift for the current iterate
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        new_h = guided_move(cur_h, shift, step)
        if new_h is None:
            new_h = perturb_h(cur_h, rng, scale=step)

        new_c = compute_c5(new_h)

        if new_c < cur_c - 1e-12:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
                last_improve = it
        it += 1

        # periodic step decay
        if it % (n * 10) == 0:
            step = max(step * 0.9, 1e-4)

        # restart from the current best if stagnation lasts too long
        if it - last_improve > n * 200:
            cur_h = perturb_h(best_h, rng, scale=step * 2.0)
            cur_c = compute_c5(cur_h)
            last_improve = it

    return best_h, best_c


def enforce_feasibility(h):
    """clip to [0,1] and enforce Σh = len(h)/2 exactly."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    rng = np.random.default_rng()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
