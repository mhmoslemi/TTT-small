def compute_gradient(h, shift):

def _project_to_feasible(h, n, tol=1e-9, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # iterative adjust
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # Determine indices that can be adjusted in direction needed
        if diff > 0:
            # need to decrease sum: reduce values > 0
            idx = np.where(h > 0)[0]
        else:
            # need to increase sum: raise values < 1
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        h = np.clip(h, 0.0, 1.0)
    # Tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residues greedily
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- C5 evaluation ----------
def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- SLSQP constraints ----------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum(h) == n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


# ---------- local SLSQP optimisation ----------
def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """Run SLSQP on the auxiliary variable formulation."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- up‑sampling ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- worst shift and gradient ----------
def _worst_shift(h: np.ndarray) -> tuple:
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    val = corr[idx]
    shift = idx - (h.size - 1)
    return shift, val


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """∂C_k/∂h_i for the shift attaining the current maximum."""
    n = h.size
    grad = np.empty(n, dtype=float)
    if shift >= 0:
        # term1 = 1 - h[i+shift] (or 1 if out of range)
        term1 = np.ones(n)
        idx1 = np.arange(n - shift)
        term1[idx1] = 1.0 - h[idx1 + shift]
        # term2 = h[i-shift] (or 0 if out of range)
        term2 = np.zeros(n)
        idx2 = np.arange(shift, n)
        term2[idx2] = h[idx2 - shift]
        grad = term1 - term2
    else:
        k = -shift
        term1 = np.ones(n)
        idx1 = np.arange(k, n)
        term1[idx1] = 1.0 - h[idx1 - k]
        term2 = np.zeros(n)
        idx2 = np.arange(0, n - k)
        term2[idx2] = h[idx2 + k]
        grad = term1 - term2
    return grad


# ---------- gradient‑descent refinement ----------
def _refine_gradient(h: np.ndarray, time_budget: float) -> tuple:
    """Iterative projected gradient descent on the worst‑shift objective."""
    start = time.time()
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    n = h.size
    step = 0.12

    while time.time() - start < time_budget:
        shift, _ = _worst_shift(best_h)
        grad = _gradient_worst_shift(best_h, shift)

        if np.all(np.abs(grad) < 1e-12):
            # random jitter to escape flat region
            cand = best_h + 0.02 * np.random.randn(n)
        else:
            cand = best_h - step * grad

        cand = _project_to_feasible(cand, n)
        cand_c5 = _c5_from_h(cand)

        if cand_c5 < best_c5 - 1e-7:
            best_h, best_c5 = cand, cand_c5
            step = min(step * 1.08, 0.5)
        else:
            step *= 0.74
            if step < 2e-5:
                step = 2e-5
    return best_h, best_c5


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    target = 0.3808

    # initial construction if supplied
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate grid sizes
    cand = []
    if init_h is not None:
        cand.append(init_h.size)
    for mul in (2, 3, 4):
        n = (init_h.size if init_h is not None else 128) * mul
        if n <= 380:
            cand.append(int(n))
    for n in (160, 200, 256, 300, 320):
        if n <= 380:
            cand.append(n)
    n_candidates = sorted(set(cand))

    for n in n_candidates:
        if time.time() - start > budget_s - 5.0:
            break

        # base h for this resolution
        if init_h is not None:
            base_h = _upscale_h(init_h, n) if init_h.size != n else init_h.copy()
        else:
            base_h = np.full(n, 0.5)

        # number of stochastic restarts adapted to remaining time
        remaining = budget_s - (time.time() - start)
        max_attempts = max(1, int(min(12, remaining // 5)))
        sigma = 0.25

        for att in range(max_attempts):
            if time.time() - start > budget_s - 4.0:
                break
            if att == 0:
                start_h = base_h
            else:
                pert = sigma * np.random.randn(n)
                start_h = base_h + pert
                start_h = _project_to_feasible(start_h, n)
                sigma *= 0.85

            h_opt, c5_val = _solve_h(start_h, max_iter=400)
            if c5_val < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 <= target:
                    return best_h.tolist(), float(best_c5), int(best_n)

        # gradient refinement if this size currently holds the best
        if best_n == n and best_h is not None:
            time_left = budget_s - (time.time() - start)
            if time_left > 6.0:
                ref_time = min(30.0, time_left * 0.5)
                refined_h, refined_c5 = _refine_gradient(best_h, ref_time)
                if refined_c5 < best_c5 - 1e-9:
                    best_h, best_c5 = refined_h, refined_c5
                    if best_c5 <= target:
                        return best_h.tolist(), float(best_c5), int(best_n)

    # fallback construction
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
