       s = shift
       g[: n - s] = 1.0 - h[s:]
       g[n - s :] = 1.0
       g[s:] += -h[: n - s]
   else:

       g[s:] = 1.0 - h[: n - s]
       g[: s] = 1.0
       g[: n - s] += -h[s:]

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ------------------------------------------------------------
# Objective evaluation
# ------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


# ------------------------------------------------------------
# Gradient for a specific shift (subgradient of max shift)
# ------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient (softmax weighting)
# ------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _smooth_gradient(h: np.ndarray, alpha: float = 30.0) -> np.ndarray:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exp_term = np.exp(alpha * shifted)
    weights = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    # iterate over all possible shifts
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            g[: n - s] += w * (1.0 - h[s:])
            g[n - s :] += w * 1.0
            g[s:] += w * (-h[: n - s])
        else:
            s = -shift
            g[s:] += w * (1.0 - h[: n - s])
            g[: s] += w * 1.0
            g[: n - s] += w * (-h[s:])
    return g


# ------------------------------------------------------------
# Projected (sub)gradient descent on the max‑correlation objective
# ------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 2000, step0: float = 0.8,
                 decay: float = 0.997) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ------------------------------------------------------------
# SLSQP on a smooth approximation of the max objective
# ------------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 1200) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    def jac(v):
        return _smooth_gradient(v, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ------------------------------------------------------------
# Simple two‑block search (coarse) – it often gives a decent start
# ------------------------------------------------------------
def _two_block_search(n: int, target: float, dx: float):
    best_h = None
    best_val = float('inf')
    p_vals = np.linspace(0.05, 0.95, 19)          # block‑size fraction
    a_vals = np.linspace(0.0, 1.0, 21)           # value on the first block
    for p in p_vals:
        L_a = int(round(p * n))
        if L_a <= 0 or L_a >= n:
            continue
        L_b = n - L_a
        for a in a_vals:
            b = (target - L_a * a) / L_b
            if not (0.0 <= b <= 1.0):
                continue
            h = np.empty(n, dtype=float)
            h[:L_a] = a
            h[L_a:] = b
            # projection only as a safety net
            h = _project_onto_capped_simplex(h, target)
            val = _compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h on [0,2] that (hopefully) improves the Erdős C5 bound."""
    rng = np.random.RandomState(seed)
    start = time.time()

    # --------------------------------------------------------
    # Load parent solution if available
    # --------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    # --------------------------------------------------------
    # Discretisation size – at least 400 points
    # --------------------------------------------------------
    n0 = len(base_h) if base_h is not None else 0
    n = max(400, n0)
    target_sum = n / 2.0
    dx = 2.0 / n

    # --------------------------------------------------------
    # Upscale helper (linear interpolation)
    # --------------------------------------------------------
    def _upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # --------------------------------------------------------
    # Initialise best solution (from parent if possible)
    # --------------------------------------------------------
    best_h = None
    best_val = float('inf')
    if base_h is not None:
        best_h = _upscale(base_h)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)

    # --------------------------------------------------------
    # Run a cheap two‑block search to possibly improve the start
    # --------------------------------------------------------
    tb_h, tb_val = _two_block_search(n, target_sum, dx)
    if tb_h is not None and tb_val < best_val:
        best_h, best_val = tb_h, tb_val

    # --------------------------------------------------------
    # Main optimisation loop – random restarts with GD + occasional SLSQP
    # --------------------------------------------------------
    restart = 0
    # we reserve ~8% of the budget for the final safety projection / evaluation
    while time.time() - start < budget_s * 0.92:
        # starting point
        if best_h is not None and rng.rand() < 0.25:
            h0 = best_h + rng.normal(scale=0.02, size=n)
        else:
            h0 = _random_feasible(rng, n, target_sum)

        # Gradient‑descent phase (max‑shift subgradient)
        h_opt = _gd_optimize(h0, target_sum,
                             max_iter=1500,
                             step0=0.8,
                             decay=0.997)

        # Occasionally polish with SLSQP on a smooth surrogate
        if restart % 4 == 0:
            h_opt = _slsqp_optimize(h_opt, target_sum,
                                    alpha=35.0, max_iter=1000)

        # Evaluate
        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        restart += 1

    # --------------------------------------------------------
    # Final safety projection and evaluation
    # --------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
