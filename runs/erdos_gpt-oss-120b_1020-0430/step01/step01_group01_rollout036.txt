import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9,
                                 max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ------------------------------------------------------------
# Objective evaluation
# ------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ------------------------------------------------------------
# Gradient for a single shift (sub‑gradient of the max‑correlation)
# ------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 800,
                 step0: float = 0.5,
                 decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ------------------------------------------------------------
# Weighted gradient for the smooth‑max objective
# ------------------------------------------------------------
def _weighted_gradient(h: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """weights length = 2*n-1, sum to 1."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    off = n - 1
    for i, w in enumerate(weights):
        if w == 0.0:
            continue
        s = i - off
        if s >= 0:
            ss = s
            # term1: 1 - h[ss:] over first n-ss positions
            grad[: n - ss] += w * (1.0 - h[ss:])
            # term1 for the tail (no partner)
            if ss > 0:
                grad[n - ss :] += w * 1.0
            # term2: -h[: n-ss] added starting at ss
            grad[ss:] += w * (-h[: n - ss])
        else:
            ss = -s
            # term1 over i >= ss
            grad[ss:] += w * (1.0 - h[: n - ss])
            # term1 over i < ss
            grad[: ss] += w * 1.0
            # term2 over i < n-ss
            grad[: n - ss] += w * (-h[ss:])
    return grad


# ------------------------------------------------------------
# Adam optimisation on the smooth‑max objective
# ------------------------------------------------------------
def _adam_optimize(h0: np.ndarray, target: float, dx: float,
                   max_iter: int = 800,
                   lr: float = 0.05,
                   alpha_start: float = 5.0,
                   alpha_end: float = 150.0) -> np.ndarray:
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    n = h.shape[0]
    for it in range(1, max_iter + 1):
        # correlation (without dx – scaling absorbed in lr)
        corr = np.correlate(h, 1.0 - h, mode="full")
        # smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iter - 1)
        max_corr = np.max(corr)
        shifted = corr - max_corr
        exp_alpha = np.exp(alpha * shifted)
        sum_exp = np.sum(exp_alpha)
        weights = exp_alpha / sum_exp
        # weighted gradient of the smooth max
        g = _weighted_gradient(h, weights)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = _project_onto_capped_simplex(h, target)

                if len(best_h) != n:
                    x_old_best = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
                    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                    h0 = np.interp(x_new, x_old_best, best_h)
                else:
                    h0 = best_h.copy()
                h0 += rng.normal(scale=0.02, size=n)
                h0 = _project_onto_capped_simplex(h0, target)
            elif base_vec is not None and rng.rand() < 0.3:
                # start from the (interpolated) parent with a tiny jitter
                h0 = base_vec + rng.normal(scale=0.01, size=n)
                h0 = _project_onto_capped_simplex(h0, target)
            else:
                h0 = _random_feasible(rng, n, target)

            # -------------------------------------------------------------
            # Adam phase (smooth max)
            # -------------------------------------------------------------
            # allocate iterations based on remaining time (at least 200)
            remain = budget_s - (time.time() - start_time)
            if remain < 5.0:
                break
            max_iter_adam = min(800, int(remain * 12))  # ~12 it/sec budget
            h_mid = _adam_optimize(h0, target, dx,
                                   max_iter=max_iter_adam,
                                   lr=0.06,
                                   alpha_start=5.0,
                                   alpha_end=180.0)

            # -------------------------------------------------------------
            # Sub‑gradient refinement (focus on the worst shift)
            # -------------------------------------------------------------
            h_ref = _gd_optimize(h_mid, target,
                                 max_iter=600,
                                 step0=0.45,
                                 decay=0.997)

            val = _c5(h_ref)
            if val < best_val:
                best_val = val
                best_h = h_ref.copy()
                best_n = n
                # early exit if we already beat the target bound
                if best_val <= 0.3808:
                    break
            restart += 1
        if best_val <= 0.3808:
            break

    # -----------------------------------------------------------------
    # Fallback (should never happen)
    # -----------------------------------------------------------------
    if best_h is None:
        best_n = 200
        best_h = np.full(best_n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, best_n / 2.0)
        best_val = _compute_c5(best_h, 2.0 / best_n)

    # final safety projection for the winning vector
    best_h = _project_onto_capped_simplex(best_h, best_n / 2.0)
    best_val = _compute_c5(best_h, 2.0 / best_n)

    return (best_h, float(best_val), best_n)
