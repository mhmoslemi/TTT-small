def binary_sa(n, time_limit):
    start = time.time()
    # initial random binary vector with n/2 ones
    b = np.zeros(n, dtype=np.int8)
    idx = np.random.choice(n, n//2, replace=False)
    b[idx] = 1
    d = 1 - b
    corr = np.correlate(b.astype(float), d.astype(float), mode='full')
    max_corr = corr.max()
    best_corr = max_corr
    best_b = b.copy()
    # temperature schedule:
    T0 = max_corr  # maybe
    T = T0
    decay = 0.9995  # per iteration
    it = 0
    while time.time() - start < time_limit:
        # pick random swap
        i = np.random.randint(n)
        if b[i] == 0:
            continue
        # choose random zero
        j = np.random.randint(n)
        if b[j] == 1:
            continue
        # compute delta
        delta = np.zeros_like(corr, dtype=int)
        # iterate lags
        for lag in range(-n+1, n):
            idx_corr = lag + n - 1
            # for i
            ii = i + lag
            if 0 <= ii < n:
                delta[idx_corr] -= d[ii]   # d[i+lag]
            ii2 = i - lag
            if 0 <= ii2 < n:
                delta[idx_corr] += b[ii2]  # b[i-lag]
            # for j
            jj = j + lag
            if 0 <= jj < n:
                delta[idx_corr] += d[jj]
            jj2 = j - lag
            if 0 <= jj2 < n:
                delta[idx_corr] -= b[jj2]
        new_corr = corr + delta
        new_max = new_corr.max()
        # acceptance
        if new_max < max_corr or np.random.rand() < np.exp(-(new_max - max_corr) / T):
            # accept
            corr = new_corr
            max_corr = new_max
            # swap b and d
            b[i] = 0
            b[j] = 1
            d[i] = 1
            d[j] = 0
            if max_corr < best_corr:
                best_corr = max_corr
                best_b = b.copy()
        # update temperature
        T *= decay
        it += 1
    # compute c5 bound from best_b
    h_best = best_b.astype(float)  # binary
    c5 = (2.0 / n) * best_corr
    return h_best, c5

def binary_sa(n, time_limit):
    start = time.time()
    # initial binary with exactly n//2 ones
    b = np.zeros(n, dtype=np.int8)
    ones = np.random.choice(n, n // 2, replace=False)
    b[ones] = 1
    d = 1 - b
    corr = np.correlate(b.astype(np.float32), d.astype(np.float32), mode='full')
    max_corr = int(corr.max())
    best_corr = max_corr
    best_b = b.copy()
    lags = np.arange(-n+1, n, dtype=int)
    shift = n - 1
    # initial temperature
    T = max_corr * 0.5 + 1.0
    decay = 0.9995
    while time.time() - start < time_limit:
        # pick random indices i (1) and j (0)
        # random i among indices of ones, random j among zeros for speed
        ones_idx = np.flatnonzero(b)
        zeros_idx = np.flatnonzero(d)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = int(np.random.choice(ones_idx))
        j = int(np.random.choice(zeros_idx))
        # compute delta
        delta = np.zeros(2*n-1, dtype=int)
        # -d[i+lag]
        idx = i + lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] -= d[idx[mask]]
        # +b[i - lag]
        idx = i - lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] += b[idx[mask]]
        # +d[j + lag]
        idx = j + lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] += d[idx[mask]]
        # -b[j - lag]
        idx = j - lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] -= b[idx[mask]]
        new_corr = corr + delta
        new_max = new_corr.max()
        # acceptance
        accept = False
        if new_max <= max_corr:
            accept = True
        else:
            # Probabilistic acceptance
            prob = np.exp(-(new_max - max_corr) / T)
            if np.random.rand() < prob:
                accept = True
        if accept:

            b[i] = 0
            b[j] = 1
            d[i] = 1
            d[j] = 0

        T *= decay
        # optional early exit if max_corr already low
        if best_corr <= int(0.38 * n / 2 * 1.01):  # rough threshold?
            pass
    # compute c5 bound
    c5 = (2.0 / n) * best_corr
    return best_b.astype(float), c5

import time
import numpy as np
import scipy.optimize as opt

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        free = np.where(h > 0)[0] if diff > 0 else np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 from a (feasible) vector."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return (2.0 / n) * max_corr


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all lags."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """Sum constraint: Σh = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """Refine a feasible h with SLSQP."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(_objective, x0, method="SLSQP",
                       bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def binary_sa(n: int, time_limit: float) -> tuple:
    """
    Simulated‑annealing for binary (0/1) vectors with exactly n/2 ones.
    Returns (best_h, best_c5).
    """
    start = time.time()
    # initialise a balanced binary vector
    b = np.zeros(n, dtype=np.int8)
    ones = np.random.choice(n, n // 2, replace=False)
    b[ones] = 1
    d = 1 - b
    corr = np.correlate(b.astype(np.float32), d.astype(np.float32), mode="full")
    max_corr = int(corr.max())
    best_corr = max_corr
    best_b = b.copy()

    lags = np.arange(-n + 1, n, dtype=int)
    shift = n - 1
    T = max_corr * 0.5 + 1.0
    decay = 0.9995

    while time.time() - start < time_limit:
        # pick a random 1 and a random 0
        ones_idx = np.flatnonzero(b)
        zeros_idx = np.flatnonzero(d)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = int(np.random.choice(ones_idx))
        j = int(np.random.choice(zeros_idx))

        delta = np.zeros(2 * n - 1, dtype=int)

        # -d[i+lag]
        idx = i + lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] -= d[idx[mask]]

        # +b[i-lag]
        idx = i - lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] += b[idx[mask]]

        # +d[j+lag]
        idx = j + lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] += d[idx[mask]]

        # -b[j-lag]
        idx = j - lags
        mask = (idx >= 0) & (idx < n)
        pos = np.nonzero(mask)[0] + shift
        delta[pos] -= b[idx[mask]]

        new_corr = corr + delta
        new_max = int(new_corr.max())

        accept = False
        if new_max <= max_corr:
            accept = True
        else:
            prob = np.exp(-(new_max - max_corr) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            corr = new_corr
            max_corr = new_max
            b[i] = 0
            b[j] = 1
            d[i] = 1
            d[j] = 0
            if max_corr < best_corr:
                best_corr = max_corr
                best_b = b.copy()
        T *= decay

    best_c5 = (2.0 / n) * best_corr
    return best_b.astype(float), best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    target_c5 = 0.3808

    # candidate grid sizes
    n_candidates = [128, 160, 200, 256, 300]
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_candidates = [init_h.size] + n_candidates
    except NameError:
        init_h = None

    n_candidates = sorted(set(n_candidates), reverse=True)

    best_h, best_c5, best_n = None, float("inf"), None

    for n in n_candidates:
        if time.time() - start_time > budget_s - 5.0:
            break
        time_left = budget_s - (time.time() - start_time)
        # allocate most of the remaining budget to this size
        sa_budget = min(time_left * 0.75, 300.0)

        # ----- binary simulated annealing -----
        h_bin, c5_bin = binary_sa(n, sa_budget)
        if c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin, c5_bin, n
            if best_c5 < target_c5:
                break

        # ----- local continuous refinement -----
        # add a tiny random perturbation before SLSQP
        h_start = h_bin.copy()
        if np.random.rand() < 0.5:
            h_start += 0.02 * (np.random.rand(n) - 0.5)
            h_start = _project_to_feasible(h_start, n)
        h_opt, c5_opt = _solve_h(h_start, max_iter=500)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < target_c5:
                break

    # fallback if nothing succeeded
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
