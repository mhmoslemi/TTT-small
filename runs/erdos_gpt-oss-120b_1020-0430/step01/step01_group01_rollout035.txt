import time
import numpy as np

# ----------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 70) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

# ----------------------------------------------------------
# Random feasible point (uniform then project)
# ----------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

# ----------------------------------------------------------
# Compute the true C5 value (max overlap) for a vector
# ----------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

# ----------------------------------------------------------
# Gradient of a single shift (used for refinement)
# ----------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        end = n - s
        g[:end] = 1.0 - h[s:]
        g[end:] = 1.0
        g[s:] += -h[:end]
    else:
        s = -shift
        end = n - s
        g[s:] = 1.0 - h[:end]
        g[:s] = 1.0
        g[:end] += -h[s:]
    return g

# ----------------------------------------------------------
# Gradient of the smooth‑max objective
# ----------------------------------------------------------
def _smooth_max_gradient(h: np.ndarray, corr: np.ndarray, alpha: float) -> np.ndarray:
    n = h.shape[0]
    # stable softmax weights over the correlation entries
    scaled = alpha * corr
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()
    g = np.zeros_like(h)
    # iterate over all possible shifts
    for shift in range(-n + 1, n):
        ws = w[shift + n - 1]
        if ws == 0.0:
            continue
        if shift >= 0:
            s = shift
            end = n - s
            g[:end] += ws * (1.0 - h[s:])
            g[s:]   += ws * (-h[:end])
        else:
            t = -shift
            end = n - t
            g[t:]   += ws * (1.0 - h[:end])
            g[:end] += ws * (-h[t:])
    return g

# ----------------------------------------------------------
# Adam optimizer on the smooth‑max objective
# ----------------------------------------------------------
def _adam_optimize(h0: np.ndarray, target: float,
                   alphas: list, steps_per_alpha: list,
                   lr: float = 0.05,
                   beta1: float = 0.9, beta2: float = 0.999,
                   eps: float = 1e-8) -> np.ndarray:
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for alpha, steps in zip(alphas, steps_per_alpha):
        for _ in range(steps):
            t += 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            grad = _smooth_max_gradient(h, corr, alpha)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = _project_onto_capped_simplex(h, target)
    return h

# ----------------------------------------------------------
# Simple projected GD focusing on the worst shift
# ----------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 300,
                 step0: float = 0.4,
                 decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h
