def simulated_annealing(h, n_iter, time_limit):
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    T = init_temp  # e.g., 0.01? but we need to tune.
    for it in range(n_iter):
         if time.time() > time_limit: break
         # random move
         i, j = rng.choice(m, size=2, replace=False)  # m = n//2
         # propose delta uniformly in [-step, step]
         # limit delta by bounds
         hi = cur[i]; hj = cur[j]
         max_up_i = 1.0 - hi
         max_down_i = hi
         max_up_j = 1.0 - hj
         max_down_j = hj
         # we will move mass from j to i: delta positive means increase i, decrease j.
         # Determine max allowed delta: can increase i up to max_up_i, can decrease j up to max_down_j.
         delta_max = min(max_up_i, max_down_j, max_step)
         delta_min = -min(max_up_j, max_down_i, max_step)
         delta = rng.uniform(delta_min, delta_max)
         if delta == 0: continue
         # propose new h
         new = cur.copy()
         new[i] += delta
         new[j] -= delta
         new[n-1-i] = new[i]
         new[n-1-j] = new[j]
         new_c5 = compute_c5(new)
         # accept if better or by prob
         delta_c5 = new_c5 - cur_c5
         if delta_c5 < 0 or rng.random() < math.exp(-delta_c5 / T):
               cur = new
               cur_c5 = new_c5
               if cur_c5 < best_c5:
                    best = cur.copy()
                    best_c5 = cur_c5
         # temperature schedule
         T = max(T * decay, min_temp)

def random_symmetric_feasible(n, rng):
    half = n // 2
    target_half = n / 4.0
    a = rng.random(half)
    a = a * (target_half / a.sum())
    # Clip values >1
    a = np.clip(a, 0.0, 1.0)
    # adjust sum again
    diff = target_half - a.sum()
    if abs(diff) > 1e-12:
        a += diff / half
        a = np.clip(a, 0.0, 1.0)
    # Build full vector
    h = np.empty(n)
    h[:half] = a
    h[half:] = a[::-1]  # reflect
    # ensure total sum
    cur_sum = h.sum()
    target = n / 2.0
    if abs(cur_sum - target) > 1e-9:
        h = h + (target - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb_sa(start, n_iter, time_limit, rng):
    h = start.copy()
    n = h.shape[0]
    half = n // 2
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # Temperature schedule: start with some T.
    T = 0.001  # maybe.
    decay = 0.9995
    min_T = 1e-7
    step_factor = 1.0  # delta will be full allowed range
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        # select two distinct indices in first half
        i, j = rng.choice(half, size=2, replace=False)
        hi = h[i]; hj = h[j]
        # Compute allowed delta
        max_up_i = 1.0 - hi
        max_down_i = hi
        max_up_j = 1.0 - hj
        max_down_j = hj
        lo = max(-max_down_i, -max_up_j)
        hi_bound = min(max_up_i, max_down_j)
        if lo >= hi_bound:
            continue
        delta = rng.uniform(lo, hi_bound)
        # If delta close to zero, skip maybe
        if abs(delta) < 1e-12:
            continue
        # propose new
        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        # mirror
        new_h[n-1-i] = new_h[i]
        new_h[n-1-j] = new_h[j]
        # evaluate
        new_c5 = compute_c5(new_h)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # uphill acceptance
            if rng.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):

        if accept:
            h = new_h
            cur_c5 = new_c5

                best_c5 = cur_c5
        # temperature decay
        T = max(T * decay, min_T)

        if it % 5000 == 0 and best_c5 < cur_c5:

import time, math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular hat h(x)=min(x,2−x) on n samples, already normalized."""
    x = (np.arange(n) + 0.5) * (2.0 / n)          # centre of each bin
    tri = np.minimum(x, 2.0 - x)
    # The integral of this hat is exactly 1, but enforce numerically:
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def random_symmetric_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible symmetric vector (h[i]=h[n-1-i]), sum=n/2."""
    half = n // 2
    target_half = n / 4.0
    a = rng.random(half)
    a = a * (target_half / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target_half - a.sum()
    if abs(diff) > 1e-12:
        a += diff / half
        a = np.clip(a, 0.0, 1.0)
    h = np.empty(n)
    h[:half] = a
    h[half:] = a[::-1]
    # final global fix (rare)
    cur = h.sum()
    target = n / 2.0
    if abs(cur - target) > 1e-12:
        h = h + (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h


# ------------------------------------------------------------
# Local stochastic optimisation (simple SA / hill‑climb)
# ------------------------------------------------------------
def hill_climb_sa(start: np.ndarray, rng: np.random.Generator,
                  n_iter: int, time_deadline: float) -> tuple:
    """
    Small simulated‑annealing style hill‑climber.
    Works on symmetric vectors (assumes start already symmetric).
    Returns best vector found and its C5.
    """
    h = start.copy()
    n = h.shape[0]
    half = n // 2
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    T = 2e-3          # initial temperature
    decay = 0.9995
    min_T = 1e-9
    # occasional big jump probability
    big_jump_prob = 0.001

    for it in range(n_iter):
        if time.time() > time_deadline:
            break

        # occasionally propose a completely new random symmetric point
        if rng.random() < big_jump_prob:
            cand = random_symmetric_feasible(n, rng)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5
                h, cur_c5 = cand, cand_c5
            continue

        i, j = rng.choice(half, size=2, replace=False)
        hi, hj = h[i], h[j]

        # allowed delta so that both stay in [0,1]
        max_up_i = 1.0 - hi
        max_down_i = hi
        max_up_j = 1.0 - hj
        max_down_j = hj
        lo = max(-max_down_i, -max_up_j)
        hi_bound = min(max_up_i, max_down_j)
        if lo >= hi_bound:
            continue
        delta = rng.uniform(lo, hi_bound)

        # propose new symmetric vector
        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_h[n - 1 - i] = new_h[i]
        new_h[n - 1 - j] = new_h[j]

        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5

        T = max(T * decay, min_T)

    return best_h, best_c5


# ------------------------------------------------------------
# Precise refinement with SLSQP (soft‑max surrogate)
# ------------------------------------------------------------
def refine_slsqp(start: np.ndarray, n: int, time_deadline: float) -> tuple:
    """Two‑stage SLSQP refinement; returns (h_opt, c5_opt) or (None,None)."""
    if time.time() > time_deadline:
        return None, None
    target_sum = n / 2.0

    def eq_cons(x):
        return np.sum(x) - target_sum

    cons = {"type": "eq", "fun": eq_cons}
    bounds = [(0.0, 1.0)] * n

    # Stage 1 – smooth surrogate, small alpha
    try:
        res1 = minimize(lambda x: softmax_c5(x, alpha=5.0),
                        start, method="SLSQP",
                        bounds=bounds, constraints=cons,
                        options={"maxiter": 200, "ftol": 1e-9, "disp": False})
        if not res1.success or time.time() > time_deadline:
            return None, None
        # Stage 2 – sharper surrogate
        res2 = minimize(lambda x: softmax_c5(x, alpha=30.0),
                        res1.x, method="SLSQP",
                        bounds=bounds, constraints=cons,
                        options={"maxiter": 200, "ftol": 1e-9, "disp": False})
        if not res2.success:
            return None, None
        h_opt = np.clip(res2.x, 0.0, 1.0)
        # enforce exact integral (tiny numerical drift)
        cur = h_opt.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            h_opt = h_opt + diff / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return None, None

    start_time = time.time()

    # candidate grid sizes (even for perfect symmetry)
