Function gradient_descent(h, max_iter, alpha, step_size, target_sum, dx):

      max_corr = np.max(corr)

      w = np.exp(alpha * (corr - max_corr))
      w /= w.sum()

def targeted_move(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    if k >= 0:
        i_indices = np.arange(n - k)
        # contributions
        contrib = h[i_indices] * (1 - h[i_indices + k])
        i_max = i_indices[np.argmax(contrib)]
        # compute delta

def project_onto_bounds_and_sum(x, target):
    # clip to [0,1] then adjust to meet sum via adding constant lambda and reclip.
    # We'll implement bisection like prior code.
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high)/2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high)/2.0
    return np.clip(x - lam, 0.0, 1.0)

import time, random, math
import numpy as np

# ------------------------------------------------------------
# Helper functions

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector x onto the intersection of the box [0,1]^n
    and the hyperplane sum(x)=target by a simple bisection on the
    Lagrange multiplier.
    """
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretised."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    """Peak‑1 triangle centred at 1.0."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2] (already integral 1)."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Indicator of the interval [0.5,1.5] (length 1)."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.where((pts >= 0.5) & (pts <= 1.5), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

def attempt_targeted_move(h, cur_val, dx):
    """
    Move mass from a point that contributes most to the worst shift
    towards its shifted partner. Returns (new_h, new_val, improved_flag).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)                         # shift corresponding to max overlap

    if k >= 0:
        i_range = np.arange(n - k)
        contrib = h[i_range] * (1.0 - h[i_range + k])
        i_max = i_range[np.argmax(contrib)]
        max_delta = min(h[i_max], 1.0 - h[i_max + k])
        if max_delta <= 1e-12:
            return h, cur_val, False
        delta = max_delta * 0.8                     # take a large fraction
        cand = h.copy()
        cand[i_max] -= delta
        cand[i_max + k] += delta
    else:
        t = -k
        i_range = np.arange(t, n)
        contrib = h[i_range] * (1.0 - h[i_range - t])
        i_max = i_range[np.argmax(contrib)]
        max_delta = min(h[i_max], 1.0 - h[i_max - t])
        if max_delta <= 1e-12:
            return h, cur_val, False
        delta = max_delta * 0.8
        cand = h.copy()
        cand[i_max] -= delta
        cand[i_max - t] += delta

    # compute new objective
    new_val = np.max(np.correlate(cand, 1 - cand, mode='full')) * dx
    if new_val < cur_val - 1e-12:
        return cand, new_val, True
    return h, cur_val, False


def attempt_random_move(h, cur_val, dx):
    """
    Randomly transfer a feasible amount of mass between two positions.
    Returns (new_h, new_val, improved_flag).
    """
    n = h.shape[0]
    for _ in range(10):  # a few attempts before giving up
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * (0.5 + 0.5 * np.random.rand())   # between 0.5 and 1 of max
        cand = h.copy()
        cand[i] -= delta
        cand[j] += delta
        new_val = np.max(np.correlate(cand, 1 - cand, mode='full')) * dx
        if new_val < cur_val - 1e-12:
            return cand, new_val, True
    return h, cur_val, False
