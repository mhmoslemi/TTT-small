def anneal_binary(n, time_deadline, rng):
    h = np.zeros(n, dtype=np.float64)
    # random choose n/2 ones
    indices = rng.choice(n, size=n//2, replace=False)
    h[indices] = 1.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # parameters
    T = cur_c5  # initial temperature roughly current value
    iteration = 0
    while time.time() < time_deadline and iteration < 50000:
        # swap
        i = rng.integers(n)
        while h[i] != 1.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0.0:
            j = rng.integers(n)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # cooling
        if iteration % 1000 == 0 and iteration > 0:
            T *= 0.9  # exponential
        iteration += 1
    return best_h, best_c5

import time, numpy as np, scipy.optimize as opt, scipy.special as sp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(sp.logsumexp(alpha * conv) / alpha)

def generate_triangular(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

def balanced_binary(n):
    # golden ratio conjugate (phi = (sqrt(5)-1)/2)
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    keys = ((np.arange(n, dtype=np.float64) * phi) % 1.0)
    idx = np.argsort(keys)
    h = np.zeros(n, dtype=np.float64)
    half = n // 2
    h[idx[:half]] = 1.0
    return h

def anneal_binary(n, time_deadline, rng, start_h=None, max_iters=None):
    if start_h is None:
        # random binary with exact half ones
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1.0
    else:
        h = np.asarray(start_h, dtype=np.float64).copy()
        # make sure it's binary and exactly half ones
        # if not binary, threshold at 0.5
        h = (h >= 0.5).astype(np.float64)
        current = int(h.sum())
        target = n // 2
        if current > target:
            # turn some ones to 0
            ones = np.where(h == 1.0)[0]
            turn_off = rng.choice(ones, size=current - target, replace=False)
            h[turn_off] = 0.0
        elif current < target:
            zeros = np.where(h == 0.0)[0]
            turn_on = rng.choice(zeros, size=target - current, replace=False)
            h[turn_on] = 1.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = max(cur_c5, 1.0)
    iteration = 0
    # prepare list of indices for fast random selection
    # We'll use while loops to find random 1 and 0.
    while time.time() < time_deadline:
        if max_iters is not None and iteration >= max_iters:
            break
        # pick a random 1
        i = rng.integers(n)
        while h[i] != 1.0:
            i = rng.integers(n)
        # pick a random 0
        j = rng.integers(n)
        while h[j] != 0.0:
            j = rng.integers(n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0.0:
            accept = True
        else:
            # prob scaling
            prob = np.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        # cooling schedule
        if iteration % 1000 == 0 and iteration > 0:
            T *= 0.9
        iteration += 1
    return best_h, best_c5

import time
import numpy as np
import scipy.optimize as opt
import scipy.special as sp

# ----------------------------------------------------------------------
# Core evaluation utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth surrogate for max‑overlap (log‑sum‑exp)."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(sp.logsumexp(alpha * conv) / alpha)


# ----------------------------------------------------------------------
# Simple constructions
# ----------------------------------------------------------------------
def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2−x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old grid to a new length."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def balanced_binary(n: int) -> np.ndarray:
    """0/1 word from the Beatty (Sturmian) construction, exactly n/2 ones."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0          # ≈0.618…
    keys = ((np.arange(n, dtype=np.float64) * phi) % 1.0)
    idx = np.argsort(keys)
    h = np.zeros(n, dtype=np.float64)
    h[idx[: n // 2]] = 1.0
    return h

# Binary simulated‑annealing optimiser
# ----------------------------------------------------------------------
def anneal_binary(
    n: int,
    time_deadline: float,
    rng: np.random.Generator,
    start_h: np.ndarray | None = None,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple annealing on the space of binary vectors with |h|_1 = n/2.
    Returns (best_h, best_c5) found before the deadline.
    """
    if start_h is None:
        # random feasible binary vector
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n // 2, replace=False)
        h[ones] = 1.0
    else:
        h = np.asarray(start_h, dtype=np.float64).copy()
        # enforce binary and exact half‑weight
        h = (h >= 0.5).astype(np.float64)
        cur = int(h.sum())
        target = n // 2
        if cur > target:
            idx = np.where(h == 1.0)[0]
            drop = rng.choice(idx, size=cur - target, replace=False)
            h[drop] = 0.0
        elif cur < target:
            idx = np.where(h == 0.0)[0]
            add = rng.choice(idx, size=target - cur, replace=False)
            h[add] = 1.0

    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    T = max(cur_c5, 1.0)            # initial temperature
    it = 0

    while time.time() < time_deadline:
        if max_iters is not None and it >= max_iters:
            break
        # pick a random 1 and a random 0 and swap them
        i = rng.integers(n)
        while h[i] != 1.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0.0:
            j = rng.integers(n)

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0.0:
            accept = True
        else:
            prob = np.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

        # cooling
        if it % 1000 == 0 and it > 0:
            T *= 0.9
        it += 1

    return best_h, best_c5

# Continuous refinement (SLSQP on soft‑max surrogate)

def solve_one_start(
    start: np.ndarray,
    n: int,
    rng: np.random.Generator,
    time_deadline: float,
) -> tuple[np.ndarray | None, float | None]:
    """Two‑stage SLSQP optimisation, returns (h_opt, C5) or (None, None) on failure."""
    target = n / 2.0

    def eq_sum(h):
        return np.sum(h) - target

    con = {"type": "eq", "fun": eq_sum}
    bounds = [(0.0, 1.0)] * n

    # stage 1 – mild smoothing
    res1 = opt.minimize(
        lambda h: softmax_c5(h, alpha=10.0),
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=[con],
        options={"maxiter": 300, "ftol": 1e-9, "disp": False},
    )
    if not res1.success or time.time() > time_deadline:
        return None, None

    # stage 2 – sharper smoothing
    res2 = opt.minimize(
        lambda h: softmax_c5(h, alpha=30.0),
        res1.x,
        method="SLSQP",
        bounds=bounds,
        constraints=[con],
        options={"maxiter": 300, "ftol": 1e-9, "disp": False},
    )
    if not res2.success or time.time() > time_deadline:
        return None, None

    h_opt = np.clip(res2.x, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)
