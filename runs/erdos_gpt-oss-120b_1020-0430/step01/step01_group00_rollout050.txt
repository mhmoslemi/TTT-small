import time
import numpy as np
import scipy.optimize as opt

# ---------- helper functions (top level) ----------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Project onto {0≤h≤1, sum(h)=n/2} using a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction (should be negligible)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _smooth_max_obj(x: np.ndarray, tau: float, mu: float):
    """
    Smooth approximation of max correlation + quadratic penalty for the sum constraint.
    Returns (f, grad) where grad is the gradient of f.
    """
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    maxv = np.max(corr)
    # stable log‑sum‑exp
    exps = np.exp((corr - maxv) / tau)
    sum_exps = np.sum(exps)
    smooth = maxv + tau * np.log(sum_exps)
    s = np.sum(x) - n / 2.0
    f = smooth + mu * (s ** 2)

    # gradient of the smooth max
    weights = exps / sum_exps
    grad = np.zeros_like(x)
    # iterate over all integer shifts
    for idx, k in enumerate(range(-n + 1, n)):
        w = weights[idx]
        if k >= 0:
            grad[: n - k] += w * (1.0 - x[k:])
            grad[k:] -= w * x[: n - k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - x[: n - kk])
            grad[: n - kk] -= w * x[kk:]
    grad += 2.0 * mu * s
    return f, grad


def _subgradient_max_corr(h: np.ndarray):
    """Sub‑gradient of the max correlation (the worst shift)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    idxs = np.where(corr == m)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            grad[: n - k] += (1.0 - h[k:])
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[kk:] += (1.0 - h[: n - kk])
            grad[: n - kk] -= h[kk:]
    grad /= len(idxs)
    return grad, m


def _subgradient_descent(h0: np.ndarray, max_iter: int = 500, step0: float = 0.5):
    """Simple sub‑gradient descent on the worst‑shift objective."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    for it in range(max_iter):
        grad, _ = _subgradient_max_corr(h)
        alpha = step0 / np.sqrt(it + 1)
        h_new = h - alpha * grad
        h_new = _project_to_feasible(h_new, n)
        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        # accept if improvement or with tiny probability to escape plateaus
        if c5_new < _c5_from_h(h) or np.random.rand() < 0.01:
            h = h_new
    return best_h, best_c5
