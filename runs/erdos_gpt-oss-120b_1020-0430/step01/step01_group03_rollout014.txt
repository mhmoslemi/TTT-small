def project_onto_box_sum(v, target, tol=1e-12, max_iter=100):
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        u = np.clip(v - mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2.0, 0.0, 1.0)

    n = h.size
    dx = 2.0 / n

def project_onto_box_sum(v, target, tol=1e-10, max_iter=60):
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        u = np.clip(v - mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            low = mid
        else:
            high = mid
    u = np.clip(v - (low+high)/2.0, 0.0, 1.0)
    # small numerical correction to meet exactly target
    diff = target - u.sum()

                u[i] += add
                diff -= add
            if abs(diff) < 1e-14:

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        u = np.clip(v - mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            low = mid
        else:
            high = mid
    u = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
