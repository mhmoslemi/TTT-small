import time
import numpy as np
from scipy.optimize import minimize

# ----- globals used by the optimizer -----
CURRENT_TARGET = None  # will be set before each SLSQP run
CURRENT_BETA = None    # soft‑max temperature


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i h_i (1‑h_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(vec, target, rng):
    """Adjust vec in‑place so that Σ vec = target while 0≤vec≤1."""
    diff = target - vec.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, vec.size)
        if diff > 0:
            add = min(diff, 1.0 - vec[i])
        else:
            add = max(diff, -vec[i])
        vec[i] += add
        diff = target - vec.sum()
    return vec


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    target = n / 2.0
    # start with a Dirichlet‑like distribution
    x = rng.random(n)
    x = x / x.sum() * target
    # clip >1 and redistribute excess
    for _ in range(50):
        over = x > 1.0
        if not over.any():
            break
        excess = x[over] - 1.0
        total_excess = excess.sum()
        x[over] = 1.0
        under = x < 1.0
        if not under.any():
            break
        # spread uniformly
        x[under] += total_excess / under.sum()
    # final correction of the total sum
    enforce_sum(x, target, rng)
    return x


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # gradually shrink step
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.8, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run one SLSQP optimisation using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    try:
        res = minimize(
            softmax_obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h0
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

            best_h, best_c = trial, c_trial

    # ---- random‑restart search -------------------------------------
    while time.time() - start_time < budget_s * 0.97:
        # generate a fresh start at the current resolution
        h_start = random_feasible_vector(best_h.size, rng)
        # short hill‑climb
        h_tmp, c_tmp = hill_climb(h_start, rng, max_iters=2000, start_step=0.4)
        # few quick SLSQP passes
        for beta in beta_sequence[:2]:
            if time.time() - start_time > budget_s * 0.98:
                break
            h_tmp = slsqp_optimize(h_tmp, beta=beta, max_iter=200)
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---- optional up‑sampling (if time permits) -------------------
    while time.time() - start_time < budget_s * 0.99:
        cur_n = best_h.size
        if cur_n >= 360:
            break
        # double the resolution by repeating each point
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        enforce_sum(h_up, n_up / 2.0, rng)
        # local optimisation at the finer grid
        h_up, c_up = hill_climb(h_up, rng, max_iters=2500, start_step=0.4)
        for beta in beta_sequence:
            if time.time() - start_time > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta=beta, max_iter=300)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up
        else:
            # if no improvement, stop further up‑sampling
            break

    # ---- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
