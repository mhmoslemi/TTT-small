import time, math, random, numpy as np

def _c5_metric(h, fft_len=None):
    """Return (C5 bound, full correlation array) for a sampled h."""
    n = h.shape[0]
    dx = 2.0 / n
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, fft_len)
    G = np.fft.rfft(1.0 - h, fft_len)
    corr = np.fft.irfft(H * np.conj(G), fft_len)[: 2 * n - 1]
    return np.max(corr) * dx, corr


def _project_onto_feasible(v, target, lo=0.0, hi=1.0):
    """Project vector v onto {x: lo<=x<=hi, sum x = target}."""
    # Solve x_i = clip(v_i - λ, lo, hi) with Σx_i = target.
    # λ is found by bisection; the sum is decreasing in λ.
    lam_low, lam_high = -10.0, 10.0
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)


def _gradient_shift(h, k):
    """Gradient of f_k = ∫ h(x)(1-h(x+k)) dx w.r.t. h for integer lag k."""
    n = h.size
    g = np.ones(n, dtype=np.float64)          # plus part = 1
    if k >= 0:
        if k < n:
            g[: n - k] = 1.0 - h[k:]          # plus where i+k in range
            g[k:] -= h[: n - k]              # minus where i-k in range
    else:
        kk = -k
        if kk < n:
            g[kk:] = 1.0 - h[: n - kk]       # plus for i >= kk
            g[: n - kk] -= h[kk:]            # minus for i < n-kk
    return g

        if low_c < high_c:
            delta = random.uniform(low_c, high_c)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta

        new_c, _ = _c5_metric(h, fft_len)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    # final projection to enforce exact sum (tiny rounding)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = _project_onto_feasible(best_h, target)
    best_c, _ = _c5_metric(best_h)
    return best_h, best_c


# ------------------------------------------------------------
# Gradient‑based refinement
# ------------------------------------------------------------
def _gradient_refine(start_h, max_seconds):
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    fft_len = 1 << ((2 * n - 1).bit_length())

    cur_c, corr = _c5_metric(h, fft_len)
    start = time.time()
    base_step = 0.25

    while time.time() - start < max_seconds:
        cur_c, corr = _c5_metric(h, fft_len)
        max_val = np.max(corr)
        # indices achieving (or nearly) the maximum
        idxs = np.flatnonzero(corr >= max_val - 1e-9)
        shift_idx = int(np.random.choice(idxs))
        k = shift_idx - (n - 1)                 # lag corresponding to the max

        g = _gradient_shift(h, k)
        g -= g.mean()                           # zero‑sum direction

        # decreasing step size
        elapsed = time.time() - start
        t_frac = elapsed / max_seconds
        alpha = base_step * (1.0 - t_frac)

        improved = False
        for _ in range(5):                     # back‑track attempts
            cand = h - alpha * g
            cand = _project_onto_feasible(cand, target)
            new_c, _ = _c5_metric(cand, fft_len)
            if new_c <= cur_c:
                h = cand
                cur_c = new_c
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # small random jitter to escape flat spots
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low < high:
                delta = random.uniform(low, high) * 0.03
                h[i] = hi + delta
                h[j] = hj - delta
                h = _project_onto_feasible(h, target)
                cur_c, _ = _c5_metric(h, fft_len)
    return h, cur_c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------
    # Choose a modestly large grid (500 points → 0.004 Δx)
    # --------------------------------------------------------
    n = 500
    if n % 2:
        n += 1

    # initial feasible vector
    h0 = _initial_vector(n)

    total_budget = float(budget_s)
    start_all = time.time()

    # --------------------------------------------------------
    # Primary simulated‑annealing phase (≈55 % of budget)
    # --------------------------------------------------------
    primary_time = total_budget * 0.55
    best_h, best_c = _anneal(h0, primary_time)

    # --------------------------------------------------------
    # Random‑restart annealing (use remaining budget while
    #   at least 30 s are left for each restart)
    # --------------------------------------------------------
    elapsed = time.time() - start_all
    while total_budget - elapsed > 40.0:
        # binary start: exactly n/2 ones (fractional if needed)
        target = n / 2.0
        ones = int(target)
        frac = target - ones
        h_rand = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        h_rand[perm[:ones]] = 1.0
        if frac > 0:
            h_rand[perm[ones]] = frac
        slice_time = min((total_budget - elapsed) * 0.25, 30.0)
        cand_h, cand_c = _anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        elapsed = time.time() - start_all

    # --------------------------------------------------------
    # Gradient‑based local refinement (use rest of the budget)
    # --------------------------------------------------------
    remaining = total_budget - (time.time() - start_all)
    if remaining > 8.0:
        best_h, best_c = _gradient_refine(best_h, remaining)

    # final safeguard: exact sum
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = _project_onto_feasible(best_h, target_sum)
        best_c, _ = _c5_metric(best_h)

    # --------------------------------------------------------
    # Return result (use helper if available)
    # --------------------------------------------------------
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = best_c

    return best_h, float(final_c), int(n)
