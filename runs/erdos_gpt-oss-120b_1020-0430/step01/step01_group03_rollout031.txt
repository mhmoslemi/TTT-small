    best_h = cur_h.copy()
    best_c = cur_c
    step = init_step
    temp = init_temp

        if i==j: continue

        eps = (rng.random() - 0.5) * 2 * max_eps * step

        new_h = cur_h.copy()

        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):

            if cur_c < best_c:

            step = max(step * 0.7, 1e-5)
            temp = max(temp * 0.9, 1e-8)

import time
import numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target, rng):
    """Clip to [0,1] and adjust the sum to `target` by random redistribution."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # simple random redistribution
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, v.size)
        if diff > 0.0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    # random pairwise transfers to mix the vector
    for _ in range(n * 8):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return project_to_feasible(h, n / 2.0, rng)


def perturb_pair(h, rng, scale):
    """Transfer a small amount ε from index i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_pair(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # schedule step reduction
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def anneal(h_start, rng, max_iters=8000, init_step=0.4, init_temp=0.02):
    """Simulated‑annealing style search."""
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    step = init_step
    temp = init_temp
    n = cur_h.size
    for it in range(max_iters):
        # propose a transfer move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        eps = max(-max_eps, min(eps, max_eps))
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / temp):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        # schedule
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-6)
            temp = max(temp * 0.85, 1e-9)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=500):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce the integral exactly (tiny drift may appear)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(3000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
