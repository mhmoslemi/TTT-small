def _targeted_anneal(start_h, max_seconds):
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0
    # ensure sum exact (scale if needed)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        h *= target_sum / cur
        # clip again
        np.clip(h, 0, 1, out=h)

    best_h = h.copy()
    # compute initial objective
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = corr.max() * dx

    start = time.time()
    # parameters
    T0 = 0.05
    T_end = 1e-6
    step0 = 0.5
    step_end = 1e-3

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        max_step = step0 * (step_end / step0) ** progress

        # compute correlation and find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        c_max = corr.max()
        current_obj = c_max * dx
        s_max = corr.argmax()
        offset = s_max - (n - 1)

        # overlapping indices
        if offset >= 0:
            i_range = np.arange(0, n - offset, dtype=int)
        else:
            i_range = np.arange(-offset, n, dtype=int)
        p_range = i_range + offset

        # compute terms
        term = h[i_range] * (1.0 - h[p_range])
        # index i with max term
        i_bad = i_range[np.argmax(term)]

        # Compute term for all indices for candidate selection
        # We'll compute term_all vector for all indices (where partner exists)
        term_all = np.empty(n, dtype=np.float64)
        term_all.fill(np.inf)  # default large if cannot increase due to no partner? Actually we want minimal term; assign high for those not increase? We'll compute mask.
        # Valid mask for all indices where partner exists
        if offset >= 0:
            valid_mask = np.arange(n) < n - offset
        else:
            valid_mask = np.arange(n) >= -offset

        # compute term_all for valid mask only
        term_all[valid_mask] = h[valid_mask] * (1.0 - h[valid_mask + offset])
        # For indices where partner out of range, term = 0 (no penalty)
        term_all[~valid_mask] = 0.0

        # Candidate indices where we can increase (h < 1 - eps) and not same i_bad
        increase_mask = (h < 1.0 - 1e-12) & (np.arange(n) != i_bad)
        candidates = np.where(increase_mask)[0]
        if candidates.size == 0:
            # No possible increase; break
            break
        # choose j with minimal term_all among candidates
        cand_terms = term_all[candidates]
        j = candidates[np.argmin(cand_terms)]

        # compute delta
        delta_limit = min(h[i_bad], 1.0 - h[j], max_step)
        if delta_limit <= 0:
            continue
        delta = random.random() * delta_limit
        if delta < 1e-12:
            continue

        # apply move
        h[i_bad] -= delta
        h[j] += delta

        # recompute objective
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_obj = corr_new.max() * dx

        # acceptance
        if new_obj < current_obj - 1e-12 or random.random() < math.exp(-(new_obj - current_obj) / max(T, 1e-12)):
            # accept
            current_obj = new_obj
            if new_obj < best_c:
                best_c = new_obj
                best_h = h.copy()
        else:
            # revert
            h[i_bad] += delta
            h[j] -= delta

    return best_h, best_c

import time
import math
import random
import numpy as np

def _c5_metric(h):
    """Exact C5 value for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _initial_vector():
    """Build a feasible starting vector (interpolated if a parent exists)."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        # at least 200 samples for enough freedom
        n = max(200, ((n0 + 9) // 10) * 10)
        if n % 2:
            n += 1
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    # enforce the exact integral constraint
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _targeted_anneal(start_h, max_seconds):
    """Simulated‑annealing that focuses on the current worst shift."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    target = n / 2.0

    # initial evaluation
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c = np.max(corr) * dx
    best_h = h.copy()

    start = time.time()
    T0, T_end = 0.08, 1e-6          # temperature schedule
    step0, step_end = 0.5, 1e-3     # maximal delta schedule

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        max_step = step0 * (step_end / step0) ** progress

        # current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_obj = np.max(corr) * dx
        s_max = corr.argmax()
        offset = s_max - (n - 1)

        # indices participating in this shift
        if offset >= 0:
            i_range = np.arange(0, n - offset, dtype=int)
        else:
            i_range = np.arange(-offset, n, dtype=int)
        p_range = i_range + offset

        # contribution of each pair to the current worst overlap
        term = h[i_range] * (1.0 - h[p_range])
        i_bad = i_range[np.argmax(term)]

        # term for every index (0 if partner missing)
        term_all = np.empty(n, dtype=np.float64)
        term_all.fill(0.0)
        if offset >= 0:
            valid = np.arange(n) < n - offset
        else:
            valid = np.arange(n) >= -offset
        term_all[valid] = h[valid] * (1.0 - h[valid + offset])

        # pick a place where increasing h hurts the overlap the least
        inc_mask = (h < 1.0 - 1e-12) & (np.arange(n) != i_bad)
        cand = np.where(inc_mask)[0]
        if cand.size == 0:
            break
        j = cand[np.argmin(term_all[cand])]

        # feasible delta (keep within [0,1] and respect max_step)
        delta = min(h[i_bad], 1.0 - h[j], max_step)
        if delta <= 1e-12:
            continue
        delta *= random.random()

        # trial move
        h[i_bad] -= delta
        h[j] += delta

        # evaluate
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        new_obj = np.max(corr_new) * dx

        # Metropolis acceptance
        if new_obj < cur_obj or random.random() < math.exp(-(new_obj - cur_obj) / max(T, 1e-12)):
            cur_obj = new_obj
            if new_obj < best_c:
                best_c = new_obj
                best_h = h.copy()
        else:
            # revert
            h[i_bad] += delta
            h[j] -= delta

    # final safety: exact integral
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function, its C5 bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    h0 = _initial_vector()
    n = h0.size
    total_budget = float(budget_s)
    start_all = time.time()

    # primary phase – focused annealing
    primary_time = total_budget * 0.75
    best_h, best_c = _targeted_anneal(h0, primary_time)

    # secondary phase – quick random restarts
    remaining = total_budget - (time.time() - start_all)
    while remaining > 8.0:
        # random binary start (exactly n/2 ones)
        m = n // 2

        h_rand[perm[:m]] = 1.0
        slice_time = min(remaining * 0.3, 25.0)
        cand_h, cand_c = _targeted_anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        remaining = total_budget - (time.time() - start_all)

    # final deterministic polishing – tiny swaps
    for _ in range(3):
        improved = False
        for _ in range(1500):
            i, j = random.sample(range(n), 2)
            hi, hj = best_h[i], best_h[j]
            low = max(-hi, hj - 1.0)

            best_h[j] = hj - delta
            new_c = _c5_metric(best_h)

                best_c = new_c
                improved = True
            else:
                best_h[i] = hi
                best_h[j] = hj
        if not improved:
            break

    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
