def v_to_h(v, n):
    # v: array shape (m,) for half interval [0,1]
    assert n % 2 == 0.
    half_n = n // 2 + 1   # include midpoint? Actually we need n points uniformly across [0,2]; we can generate half points of length half_n (including center) and mirror excluding center.
    # Create x values on [0,1] for interpolation: using half_n points from 0 to 1 inclusive.
    xp = np.linspace(0.0, 1.0, len(v))   # length m
    x_half = np.linspace(0.0, 1.0, half_n)   # half_n points.
    half_vals = np.interp(x_half, xp, v)
    # Mirror to full interval: for left part (0-1) we already have; for right part (1-2) use reversed (except endpoint maybe duplicate).
    right_vals = half_vals[-2::-1]  # exclude the middle point to avoid duplication at x=1.
    h = np.concatenate([half_vals, right_vals])
    # Now ensure length n
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        # pad at end with last value
        pad = np.full(n - len(h), h[-1])
        h = np.concatenate([h, pad])
    # Normalize integral
    target = n / 2.0
    cur_sum = h.sum()
    h = h * (target / cur_sum)
    # Clip to [0,1] after scaling; may cause small bound violation; re-adjust via small diff:
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target) > 1e-12:
        h = h + (target - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

# ------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def v_to_h(v: np.ndarray, n: int) -> np.ndarray:
    """
    Build a symmetric step function on n points from a half‑interval
    vector v (values on [0,1] for x∈[0,1]).  The function satisfies
    the integral constraint by a simple renormalisation.
    """
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx            # centre of each sub‑interval
    ts = np.minimum(xs, 2.0 - xs)            # fold onto [0,1]
    # linear interpolation of v on [0,1]
    h = np.interp(ts, np.linspace(0.0, 1.0, len(v)), v, left=0.0, right=1.0)

    target = n / 2.0
    cur = h.sum()
    if cur > 0:
        h = h * (target / cur)
    else:
        h = np.full_like(h, target / n)

    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h = h + (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def softmax_c5_v(v: np.ndarray, n: int, alpha: float = 30.0) -> float:
    """Smooth surrogate of max overlap; used for polishing."""
    h = v_to_h(v, n)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(special.logsumexp(alpha * conv) / alpha)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ----------------------------------------------------------------
    # If a parent construction is supplied, use it as a benchmark.
    # ----------------------------------------------------------------
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_c5 = compute_c5(parent_h)
        best_h, best_c5, best_n = parent_h.copy(), parent_c5, parent_h.shape[0]
    except NameError:
        parent_h = None

    # ----------------------------------------------------------------
    # Candidate grid sizes – larger values give finer discretisation.
    # ----------------------------------------------------------------
    n_candidates = [120, 150, 180, 200, 240]

    for n in n_candidates:
        if time.time() > deadline:
            break
        # dimension of the parametric representation (half‑interval)
        m = max(6, int(round(n / 12)))          # ~10‑20 for the chosen n
        bounds = [(0.0, 1.0)] * m

        # Triangular baseline (linear rise from 0 to 1)
        v_tri = np.linspace(0.0, 1.0, m)

        # Build a modest initial population for DE
        popsize = 12
        init_pop = rng.random((popsize, m))
        init_pop[0] = v_tri
        # a few perturbed copies of the triangle
        for i in range(1, min(4, popsize)):
            pert = v_tri + rng.normal(scale=0.12, size=m)
            init_pop[i] = np.clip(pert, 0.0, 1.0)

        # ------------------------------------------------------------
        # Objective for DE: the exact C5 (non‑smooth but cheap)
        # ------------------------------------------------------------
        def obj(v):
            return compute_c5(v_to_h(v, n))

        # Time‑sensitive limit on DE iterations
        iter_budget = 70
        time_left = deadline - time.time()
        # Rough guard: if not enough time for a full run, skip.
        if time_left < 5.0:
            break

        result = optimize.differential_evolution(
            obj,
            bounds,
            seed=seed + n,
            maxiter=iter_budget,
            popsize=popsize,
            init=init_pop,
            polish=False,
            updating='deferred',
            workers=1,
        )

        v_best = result.x

        # ------------------------------------------------------------
        # Local polishing with a smooth surrogate (log‑sum‑exp)
        # ------------------------------------------------------------
        if time.time() < deadline - 1.0:
            def soft_obj(v):
                return softmax_c5_v(v, n, alpha=30.0)

            loc_res = optimize.minimize(
                soft_obj,
                v_best,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 250, 'ftol': 1e-12},
            )
            if loc_res.success:
                v_best = loc_res.x

        # ------------------------------------------------------------
        # Evaluate final candidate for this n
        # ------------------------------------------------------------
        h_candidate = v_to_h(v_best, n)
        c5_candidate = compute_c5(h_candidate)

        if c5_candidate < best_c5 - 1e-9:
            best_h, best_c5, best_n = h_candidate.copy(), c5_candidate, n

    # ----------------------------------------------------------------
    # Fallback: simple binary block (guaranteed feasible) if nothing else.
    # ----------------------------------------------------------------
    if best_h is None:
        n = 80
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # ----------------------------------------------------------------
    # Final sanity check – enforce the integral exactly.
    # ----------------------------------------------------------------
    target = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-12:
        best_h = best_h + (target - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        # one more correction if needed
        cur = best_h.sum()
        if abs(cur - target) > 1e-12:
            best_h = best_h * (target / cur)
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), int(best_n)
