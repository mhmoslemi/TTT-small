def compute_c5(h):
    n = h.shape[0]
    n2 = 2*n
    fft_h = np.fft.fft(h, n2)
    fft_one_minus_h = np.fft.fft(1.0 - h, n2)
    corr = np.fft.ifft(fft_h * np.conj(fft_one_minus_h)).real
    corr = corr[:2*n-1]  # keep first 2*n-1 entries
    return np.max(corr) * (2.0 / n)   # dx
```

import time, random, numpy as np
from scipy.optimize import minimize

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Projection onto [0,1]^n ∩ {sum = target} via bisection on lambda."""
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(arr: np.ndarray, alpha: float) -> float:
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx

def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_rect(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.where((pts >= 0.5) & (pts < 1.5), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_random_binary(n: int) -> np.ndarray:
    target = int(n / 2)
    h = np.zeros(n)
    idx = np.random.choice(n, target, replace=False)
    h[idx] = 1.0
    return h

def make_trapezoid(n: int, s: float) -> np.ndarray:
    """Symmetric trapezoid: side length s (0<s<=1)."""
    if s <= 0.0:
        s = 1e-6
    if s > 1.0:
        s = 1.0
    h0 = 1.0 / (2.0 - s)          # plateau height
    w = 2.0 - 2.0 * s             # plateau width
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    left = pts < s
    right = pts >= (s + w)
    middle = ~(left | right)
    h = np.empty(n)
    h[left] = h0 * (pts[left] / s)
    h[right] = h0 * ((2.0 - pts[right]) / s)
    h[middle] = h0
    return project_onto_bounds_and_sum(h, n / 2.0)

def upsample_initial(h_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation from old grid to new grid."""
    m = h_old.shape[0]
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h = np.interp(x_new, x_old, h_old)
    return project_onto_bounds_and_sum(h, n_new / 2.0)
