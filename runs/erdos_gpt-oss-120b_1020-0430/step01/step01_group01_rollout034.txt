def anneal_binary(n, target_ones, rng, budget_seconds, start_time):
    best_h = None, best_score large.
    while time left > some:
        h = np.zeros(n, dtype=int)
        h[:target_ones] = 1
        rng.shuffle(h)
        cur_h = h.copy()
        cur_score = compute_max_overlap(cur_h)
        best_local_h = cur_h.copy()
        best_local_score = cur_score
        # annealing schedule
        max_iter = 5000  # adjust to budget
        T0 = 5.0
        Tf = 0.01
        for it in range(max_iter):
            if time.time() - start_time > budget_seconds*0.9: break
            # degrade T
            T = T0 * (Tf/T0)**(it / max_iter)
            i = rng.randint(n)
            # choose j opposite
            j = rng.randint(n)
            while cur_h[i] == cur_h[j]:
                j = rng.randint(n)
            # swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_score = compute_max_overlap(cur_h)
            delta = new_score - cur_score
            if delta <= 0 or rng.rand() < np.exp(-delta / T):
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local_h = cur_h.copy()
            else:
                # revert
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # update global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_h = best_local_h.copy()

def gradient_for_shift(h, shift):
    n = len(h)
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        # term1: for i < n-s: 1 - h[i+s]; else 1
        g[: n - s] = 1.0 - h[s:]

def project_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

import time
import math
import numpy as np

def _project_onto_capped_simplex(v: np.ndarray,
                                 target: float,
                                 tol: float = 1e-9,
                                 max_iter: int = 60) -> np.ndarray:
    """
    Project vector v onto the set {0 ≤ x ≤ 1, sum(x) = target}
    using a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Generate a random vector and project it onto the feasible set."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Exact C5 value for a discretised step function."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the correlation for the shift that currently attains the maximum.
    """
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray,
                 target: float,
                 max_iter: int = 200,
                 step0: float = 0.8,
                 decay: float = 0.99) -> np.ndarray:
    """
    Simple projected (sub)gradient descent on the max‑correlation objective.
    """
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that yields a low Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Discretisation
    # ----------------------------------------------------------------------
    n = 200                     # number of samples on [0,2]
    target_sum = n / 2.0        # ensures integral equals 1
    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # Helper: evaluate a candidate and possibly record it as the best one
    # ----------------------------------------------------------------------
    best_h = None
    best_val = float('inf')

    # ----------------------------------------------------------------------
    # If a parent solution exists, evaluate it
    # ----------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        if init.shape[0] != n:
            # upscale / downscale linearly to required length
            x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = np.clip(init, 0.0, 1.0)
        init = _project_onto_capped_simplex(init, target_sum)
        val = _compute_c5(init, dx)
        best_h = init.copy()
        best_val = val
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Main stochastic optimisation loop (projected GD with many restarts)
    # ----------------------------------------------------------------------
    # reserve ~90% of the budget for this phase
    time_limit = budget_s * 0.90
    while time.time() - start < time_limit:
        # ---- generate a start point ---------------------------------------
        if best_h is not None and rng.rand() < 0.2:
            h0 = best_h + rng.normal(scale=0.02, size=n)
        else:
            h0 = _random_feasible(rng, n, target_sum)

        # ---- two‑stage gradient descent ------------------------------------
        h_mid = _gd_optimize(h0, target_sum,
                             max_iter=120, step0=1.0, decay=0.99)
        h_opt = _gd_optimize(h_mid, target_sum,
                             max_iter=250, step0=0.5, decay=0.996)

        # ---- evaluate ------------------------------------------------------
        val = _compute_c5(h_opt, dx)
        if val < best_val - 1e-7:
            best_val = val
            best_h = h_opt.copy()
            # early exit if we beat the target benchmark
            if best_val < 0.38085:
                break

    # ----------------------------------------------------------------------
    # Final refinement from the best vector found
    # ----------------------------------------------------------------------
    if best_h is None:
        best_h = _random_feasible(rng, n, target_sum)

    # a lengthier descent to polish the solution
    best_h = _gd_optimize(best_h, target_sum,

    # final projection to guarantee exact feasibility
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)
