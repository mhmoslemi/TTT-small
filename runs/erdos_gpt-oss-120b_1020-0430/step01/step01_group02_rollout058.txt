def gradient_max_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        # source indices
        src = np.arange(0, n - shift)
        tgt = src + shift
        grad[src] = (1 - h[tgt])
        grad[tgt] = -h[src]
    else:
        s = -shift
        src = np.arange(s, n)
        tgt = src - s
        grad[src] = (1 - h[tgt])
        grad[tgt] = -h[src]
    return grad, shift

import time, numpy as np, math

def _evaluate_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _adjust_sum(x, target):
    # Clip then adjust sum to target while staying within [0,1] using simple water-filling algorithm
    # Variation: use while loop to reallocate diff.
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Use water-filling: if diff > 0, add to entries with room; if diff < 0, remove from entries >0
    # We'll do iterative proportional distribution until diff small.
    while abs(diff) > 1e-12:
        if diff > 0:
            free = 1.0 - x
        else:
            free = x
        mask = free > 1e-12
        if not np.any(mask):
            break
        avail = free[mask]
        # distribute equally across free entries
        per = diff / np.count_nonzero(mask)
        # limit per to not overshoot per entry capacity
        delta = np.minimum(per, avail) if diff > 0 else np.maximum(per, -avail)
        # apply
        x[mask] += delta
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

def _gradient_max_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        src = np.arange(0, n - shift)
        tgt = src + shift
        grad[src] = 1 - h[tgt]
        grad[tgt] = -h[src]
    else:
        s = -shift
        src = np.arange(s, n)
        tgt = src - s
        grad[src] = 1 - h[tgt]
        grad[tgt] = -h[src]
    return grad, shift

import time, math
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto the hyper‑plane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Simple water‑filling redistribution
    while abs(diff) > 1e-12:
        if diff > 0.0:          # need to add mass
            mask = x < 1.0 - 1e-12
        else:                    # need to remove mass
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0.0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _gradient_max_shift(h: np.ndarray):
    """Sub‑gradient of the maximal correlation term."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)          # integer shift that attains the max
    grad = np.zeros_like(h)
    if shift >= 0:
        src = np.arange(0, n - shift)
        tgt = src + shift
        grad[src] = 1.0 - h[tgt]
        grad[tgt] = -h[src]
    else:
        s = -shift
        src = np.arange(s, n)
        tgt = src - s
        grad[src] = 1.0 - h[tgt]
        grad[tgt] = -h[src]
    return grad, shift

def _hill_climb(h: np.ndarray, cur_c5: float, deadline: float, rng: np.random.Generator):
    """Random pairwise mass‑exchange hill climbing."""
    n = h.size
    target = n / 2.0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # direction random
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
        # keep best feasible vector
        if new_c5 + 1e-12 < cur_c5:
            cur_c5 = new_c5
            h = h_new
    # final projection for safety
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
    return h, _evaluate_c5(h)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise – we use a moderately fine discretisation.
    # ------------------------------------------------------------------
    n = 100                     # number of samples (feel free to adjust)
    target = n / 2.0
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        # start from the supplied vector, but embed it into a larger array
        old = np.asarray(init_vec, dtype=np.float64)
        old = np.clip(old, 0.0, 1.0)
        old = _adjust_sum(old, len(old) / 2.0)
        # up‑sample / pad to the chosen resolution
        h = np.interp(
            np.linspace(0.0, 2.0, n, endpoint=False),
            np.linspace(0.0, 2.0, len(old), endpoint=False),
            old,
        )
    else:
        h = rng.random(n)
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)

    cur_h = h.copy()
    cur_c5 = _evaluate_c5(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5

    # ------------------------------------------------------------------
    # 2. Gradient‑descent style optimisation with occasional noise.
    # ------------------------------------------------------------------
    step = 0.35                 # initial step size (acts as a temperature)
    decay = 0.9997
    min_step = 1e-4
    stagnation = 0

    while True:
        if time.time() - start_time > budget_s - 1.0:
            break

        grad, _ = _gradient_max_shift(cur_h)
        # move opposite the gradient (descending)
        new_h = cur_h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = _adjust_sum(new_h, target)
        new_c5 = _evaluate_c5(new_h)

        # accept if improvement, otherwise Metropolis‑type acceptance
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            stagnation = 0
        else:
            temp = max(step, 1e-8)
            if rng.random() < math.exp((cur_c5 - new_c5) / temp):
                cur_h, cur_c5 = new_h, new_c5
                stagnation = 0
            else:
                stagnation += 1

        # update global best
        if cur_c5 < best_c5:
            best_h, best_c5 = cur_h.copy(), cur_c5

        # step‑size schedule
        step = max(min_step, step * decay)

        # occasional random perturbation to escape flat regions
        if rng.random() < 0.002:
            pert = rng.normal(scale=0.07, size=n)
            cur_h = np.clip(cur_h + pert, 0.0, 1.0)
            cur_h = _adjust_sum(cur_h, target)
            cur_c5 = _evaluate_c5(cur_h)

        # if we have been stagnant for a while, re‑seed around the best found
        if stagnation > 8000:
            cur_h = best_h + rng.normal(scale=0.04, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _adjust_sum(cur_h, target)
            cur_c5 = _evaluate_c5(cur_h)
            stagnation = 0

    # ------------------------------------------------------------------
    # 3. Final fine‑grained hill climbing from the best candidate.
    # ------------------------------------------------------------------
    deadline = start_time + budget_s - 0.2
    best_h, best_c5 = _hill_climb(best_h, best_c5, deadline, rng)

    # ------------------------------------------------------------------
    # 4. Return the solution.
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
