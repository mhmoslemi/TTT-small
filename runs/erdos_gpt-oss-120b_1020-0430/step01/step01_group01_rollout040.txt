def _subgradient_opt(h0, target, max_iter=200, step0=0.2, decay=0.995):
    h = h0.copy()
    step = step0
    n = len(h)
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        # get list of shifts attaining near max
        # we can just pick first argmax
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient for that shift
        g = _gradient_for_shift(h, shift)
        # (If many, average? Not needed)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

def _simulated_annealing(h_start, iterations, temp_start, temp_end, dx, rng):
    h = h_start.copy()
    cur_val = _compute_c5(h, dx)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    for t in range(iterations):
        # linear temperature schedule
        T = temp_start + (temp_end - temp_start) * t / iterations
        i = rng.randint(n)
        j = rng.randint(n-1)
        if j >= i:
            j += 1
        # compute max possible delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # sample delta
        delta = max_delta * rng.rand() * 0.5  # up to 0.5 of max
        if delta == 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = _compute_c5(h_new, dx)
        dyn = new_val - cur_val
        if dyn < 0 or rng.rand() < np.exp(-dyn / max(T, 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        # optional restart after stagnation
    return best_h, best_val

def _slsqp_optimize(h0, target, alpha=30.0, max_iter=500):
    n = len(h0)
    # objective
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        # smooth max
        m = np.max(corr)
        shifted = corr - m
        # using logsumexp
        return (m + np.log(np.sum(np.exp(alpha * shifted))) / alpha) * (2.0/n)
    # constraint: sum(v) - target = 0
    cons = ({'type':'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0,1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': max_iter, 'ftol':1e-10, 'disp':False})
    # ensure feasible
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ---------------------------------------------------------

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    return _project_onto_capped_simplex(rng.rand(n), target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    return float((np.correlate(h, 1.0 - h, mode="full") * dx).max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.size
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 400,
                 step0: float = 0.5, decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.size
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float, alpha: float = 30.0,
                    max_iter: int = 500) -> np.ndarray:
    n = h0.size
    dx = 2.0 / n

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha) * dx

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                             'ftol': 1e-10,
                                             'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
