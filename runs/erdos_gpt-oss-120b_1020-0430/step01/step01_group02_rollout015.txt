import time
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyper‑plane sum(x)=target under 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        per = diff / idx.size
        if diff > 0:
            cap = 1.0 - x[idx]
            add = np.minimum(per, cap)
            x[idx] += add
            diff -= add.sum()
        else:
            cap = x[idx]
            sub = np.minimum(-per, cap)
            x[idx] -= sub
            diff += sub.sum()
    return np.clip(x, 0.0, 1.0)


def _gradient_of_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the correlation term at the worst lag."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        i = np.arange(0, n - L)
        g[i] += 1.0 - h[i + L]
        g[i + L] -= h[i]
    else:
        L = -lag
        i = np.arange(L, n)
        g[i] += 1.0 - h[i - L]
        g[i - L] -= h[i]
    return g

    start = time.time()
    iter_cnt = 0
    base_step = 0.3

    while time.time() - start < time_budget:
        # worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag = int(np.argmax(corr) - (n - 1))
        # gradient
        grad = _gradient_of_lag(h, lag)

        # decreasing step size
        elapsed = time.time() - start
        frac = 1.0 - elapsed / time_budget
        step = base_step * frac

        # propose move
        h_new = _adjust_sum(h - step * grad, target)
        c5_new = _evaluate_c5(h_new)

        # accept if improvement, else occasional uphill move
        if c5_new + 1e-12 < cur_c5:
            h, cur_c5 = h_new, c5_new
        elif rng.random() < 0.005 * frac:
            h, cur_c5 = h_new, c5_new

        if cur_c5 < best_c5:
            best_h, best_c5 = h.copy(), cur_c5

        # occasional jitter to escape plateaus
        if iter_cnt % 25 == 0:
            jitter = rng.normal(scale=0.04 * frac, size=n)
            h = _adjust_sum(np.clip(h + jitter, 0.0, 1.0), target)
            cur_c5 = _evaluate_c5(h)
            if cur_c5 < best_c5:
                best_h, best_c5 = h.copy(), cur_c5

        iter_cnt += 1

    return best_h, best_c5


def _random_redistribute(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Select a small random subset and reshuffle its mass."""
    n = h.size
    m = int(rng.integers(2, min(6, n) + 1))
    idx = rng.choice(n, size=m, replace=False)
    total = h[idx].sum()
    # draw a Dirichlet‑like random split
    new_vals = rng.random(m)
    new_vals = new_vals / new_vals.sum() * total
    h2 = h.copy()
    h2[idx] = new_vals
    # enforce box and sum constraints
    return _adjust_sum(h2, n / 2.0)


def _hill_climb(h: np.ndarray, cur_c5: float, deadline: float, rng: np.random.Generator):
    """Simple pairwise mass‑exchange hill climbing with occasional reshuffles."""
    n = h.size
    best_h, best_c5 = h.copy(), cur_c5
    while time.time() < deadline:
        # pairwise exchange
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(best_h[src], 1.0 - best_h[dst])
        if max_move > 1e-12:
            delta = rng.uniform(0.0, max_move)
            cand = best_h.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand_c5 = _evaluate_c5(cand)
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand, cand_c5
                continue

        # occasional small reshuffle of a subset
        if rng.random() < 0.02:
            cand = _random_redistribute(best_h, rng)
            cand_c5 = _evaluate_c5(cand)
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand, cand_c5

    return best_h, best_c5


def _upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear up‑sampling of a step function."""
    n = h.size
    old_x = np.linspace(0.0, 2.0, n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    return _adjust_sum(np.clip(new_h, 0.0, 1.0), new_n / 2.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of samples."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initial vector
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
    else:
        n = 45
        h0 = rng.random(n)
        target = n / 2.0
        h0 = h0 * (target / h0.sum())
        h0 = _adjust_sum(h0, target)

    n = h0.size
    target = n / 2.0
    h0 = _adjust_sum(np.clip(h0, 0.0, 1.0), target)

    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Main stochastic optimisation phase
    # ------------------------------------------------------------------
    total_budget = budget_s
    # reserve a small margin for the final hill climb
    reserve = 7.0
    while time.time() - start_time < total_budget - reserve:
        remaining = total_budget - (time.time() - start_time) - reserve
        # each local run gets at most 12 s or 15 % of what is left
        chunk = min(12.0, remaining * 0.15)
        if chunk < 1.0:
            break
        # sometimes start from current best, sometimes from a fresh random point
        if rng.random() < 0.33:
            start_point = best_h.copy()
        else:
            start_point = rng.random(n)
            start_point = _adjust_sum(start_point, target)
        cand_h, cand_c5 = _optimize_local(start_point, chunk, rng)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5

    # ------------------------------------------------------------------
    # 3. Optional up‑sampling if time allows
    # ------------------------------------------------------------------
    time_used = time.time() - start_time
    time_left = total_budget - time_used - reserve
    if time_left > 20.0 and n < 80:
        # try a finer discretisation (≈ double)
        new_n = min(120, n * 2)
        up_h = _upsample(best_h, new_n)
        # devote most of the remaining time to refining the larger problem
        cand_h, cand_c5 = _optimize_local(up_h, time_left * 0.85, rng)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5
            n = best_h.size
            target = n / 2.0

    # ------------------------------------------------------------------
    # 4. Final hill‑climbing refinement
    # ------------------------------------------------------------------
    deadline = start_time + total_budget - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, deadline, rng)

    # ------------------------------------------------------------------
    # 5. Final feasibility & return
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
