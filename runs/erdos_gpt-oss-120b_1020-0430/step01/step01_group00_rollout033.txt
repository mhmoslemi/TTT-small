def hillclimb_binary(n, time_budget, init_h=None):
    # ensure we have w=n//2 ones
    w = n // 2
    best_h = None
    best_val = np.inf
    # start from init_h if plausible (binary) else random.
    if init_h is not None:
        # convert to binary by rounding
        binary = np.zeros(n, dtype=int)
        # get top w indices by value
        idx = np.argsort(-init_h)[:w]
        binary[idx] = 1
        cur_h = binary
    else:
        cur_h = np.zeros(n, dtype=int)
        cur_h[:w] = 1
        np.random.shuffle(cur_h)
    cur_val = get_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    while time.time() - start < time_budget:
        # random swap
        # pick random 1 index i, random 0 index j
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_val = get_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            # maybe update best
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                # early exit if great
                if best_val < target_bound:
                    break
        else:
            # revert with probability (simulated annealing)
            # simple acceptance probability: exp(-(new-cur)/temp)
            # define decreasing temperature
            # For simplicity, revert always.
            cur_h[i] = 1
            cur_h[j] = 0
        # iterate
    return best_h.astype(float), best_val

def _binary_optimize(h_init, n, budget, target_bound=0.3808):
    start = time.time()
    h = h_init.copy()
    # compute initial
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    best_h = h.copy()
    best_corr = max_corr

    # compute target max_corr int
    target_max_corr = int(np.floor(target_bound * n / 2))

    # deterministic swap loop
    while time.time() - start < budget * 0.5:
        # find swap
        i_swap, j_swap, cur_max = _find_swap_for_worst_shift(h)
        if i_swap is None:
            break
        # perform swap
        h[i_swap] = 1
        h[j_swap] = 0
        # recompute max corr
        max_corr = np.max(np.correlate(h, 1 - h, mode='full'))
        if max_corr < best_corr:
            best_corr = max_corr
            best_h = h.copy()
        # if reached target, break
        if max_corr <= target_max_corr:
            best_h = h.copy()
            best_corr = max_corr
            break

    # random hillclimb
    while time.time() - start < budget:
        # random swap
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_corr = np.max(np.correlate(h, 1 - h, mode='full'))
        if new_corr < best_corr:
            best_corr = new_corr
            best_h = h.copy()
            # keep new state as base for further search
        else:
            # revert
            h[i] = 1
            h[j] = 0
        # early exit if target reached
        if best_corr <= target_max_corr:
            break
    return best_h.astype(float), best_corr

import time
import numpy as np
from scipy import optimize

# -------------------- feasibility utilities --------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water-filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# -------------------- SLSQP optimizer --------------------
def _ineq_constraint(x: np.ndarray):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """Local SLSQP refinement of a feasible h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# -------------------- binary deterministic improvement --------------------
def _find_swap_for_worst_shift(h: np.ndarray):
    """Find a pair (i zero, j one) that reduces the worst correlation."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr == max_val)[0]
    for idx in idxs:
        k = idx - (n - 1)  # shift value
        if k == 0:
            continue
        if k > 0:
            start_i = 0
            end_i = n - k
        else:
            start_i = -k
            end_i = n
        i_range = np.arange(start_i, end_i)
        hi = h[i_range]
        hik = h[i_range + k]
        zero_mask = (hi == 0) & (hik == 1)
        one_mask = (hi == 1) & (hik == 0)
        zero_idxs = i_range[zero_mask]
        one_idxs = i_range[one_mask]
        if zero_idxs.size and one_idxs.size:
            return int(zero_idxs[0]), int(one_idxs[0]), max_val
    return None, None, max_val


def _binary_optimize(base_bin: np.ndarray, n: int, budget: float, target_bound: float):
    """Start from a binary vector and improve it via deterministic swaps + random hill‑climb."""
    target_max_corr = int(np.floor(target_bound * n / 2.0))
    start = time.time()
    h = base_bin.copy()
    best_h = h.astype(float).copy()
    best_corr = np.max(np.correlate(h, 1 - h, mode="full"))

    # deterministic phase (≈40% of budget)
    while time.time() - start < budget * 0.4:
        i_swap, j_swap, cur_max = _find_swap_for_worst_shift(h)
        if i_swap is None:
            break
        h[i_swap] = 1
        h[j_swap] = 0
        new_corr = np.max(np.correlate(h, 1 - h, mode="full"))
        if new_corr < best_corr:
            best_corr = new_corr
            best_h = h.astype(float).copy()
        if new_corr <= target_max_corr:
            return best_h, new_corr
    # random hill‑climb phase (remaining budget)
    while time.time() - start < budget:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_corr = np.max(np.correlate(h, 1 - h, mode="full"))
        if new_corr < best_corr:
            best_corr = new_corr
            best_h = h.astype(float).copy()
            if new_corr <= target_max_corr:
                return best_h, new_corr
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_corr


def _refine_continuous(h_start: np.ndarray, limit_seconds: float, max_iter: int = 400):
    """Run a few SLSQP refinements from perturbed starts within the given time budget."""
    best_h = h_start
    best_val = _c5_from_h(h_start)
    start = time.time()
    attempts = 4
    while attempts > 0 and time.time() - start < limit_seconds:
        sigma = 0.12
        pert = sigma * np.random.randn(*h_start.shape)
        cand = _project_to_feasible(h_start + pert, h_start.size)
        h_opt, val = _solve_h(cand, max_iter=max_iter)
        if val < best_val:
            best_val = val
            best_h = h_opt
        attempts -= 1
    return best_h, best_val


# -------------------- main entry point --------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # grab parent construction if provided
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # candidate grid sizes
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for n in (128, 160, 200):
        if n not in n_candidates:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 8.0:
            break

        # build a binary starter for this grid
        w = n // 2
        if init_h is not None and init_h.size == n:
            idx = np.argsort(-init_h)[:w]
            base_bin = np.zeros(n, dtype=int)
            base_bin[idx] = 1
        else:
            base_bin = np.zeros(n, dtype=int)
            base_bin[:w] = 1
            np.random.shuffle(base_bin)

        # allocate time slice for this n
        slice_time = min( max(30.0, time_left() / (len(n_candidates) * 1.5)), time_left())
        # deterministic + random binary improvement
        bin_h, bin_max_corr = _binary_optimize(base_bin, n, slice_time * 0.5, target_bound)
        bin_c5 = bin_max_corr * (2.0 / n)

        if bin_c5 < best_c5:
            best_c5, best_h, best_n = bin_c5, bin_h, n
            if best_c5 <= target_bound:
                break

        # lots of SLSQP restarts (continuous refinement)
        remaining = time_left()
        if remaining < 5.0:
            continue
        # give the continuous phase a modest share of remaining budget
        cont_time = min(remaining * 0.5, 60.0)

        # first refinement from binary result
        refined_h, refined_c5 = _refine_continuous(bin_h, cont_time, max_iter=500)
        if refined_c5 < best_c5:
            best_c5, best_h, best_n = refined_c5, refined_h, n
            if best_c5 <= target_bound:
                break

        # additional random SLSQP starts (if time permits)
        extra_start = time_left()
        while extra_start > 6.0:
            # create a random feasible start
            rand_vec = np.random.rand(n)
            rand_vec = _project_to_feasible(rand_vec, n)
            opt_h, opt_c5 = _solve_h(rand_vec, max_iter=500)
            if opt_c5 < best_c5:
                best_c5, best_h, best_n = opt_c5, opt_h, n
                if best_c5 <= target_bound:
                    break
            extra_start = time_left()
        if best_c5 <= target_bound:
            break

    # fallback if nothing found
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
