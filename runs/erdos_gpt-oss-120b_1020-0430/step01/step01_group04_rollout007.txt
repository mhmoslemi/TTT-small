import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5_full(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5_full(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular bump, normalized to integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 0.5 * (2.0 / n)
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, preserving the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 0.5 * (2.0 / old_n)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 0.5 * (2.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def expand_sym(left: np.ndarray) -> np.ndarray:
    """Create a symmetric full vector from its left half."""
    return np.concatenate([left, left[::-1]])


def maxiter_for_n(n: int) -> int:
    """Heuristic iteration budget for SLSQP."""
    if n <= 120:
        return 400
    if n <= 200:
        return 300
    return 200


# ------------------------------------------------------------
# Optimisation of a symmetric half‑vector
# ------------------------------------------------------------
def optimise_half(z0: np.ndarray, n: int, time_deadline: float,
                  alphas=(5.0, 20.0, 50.0)) -> tuple:
    """Two‑stage SLSQP on the left half; returns (h_opt, c5_opt)."""
    m = n // 2
    target_left = n / 4.0
    z = np.copy(z0)

    for alpha in alphas:
        if time.time() > time_deadline:
            break

        def obj(v):
            h = expand_sym(v)
            return softmax_c5_full(h, alpha=alpha)

        cons = {"type": "eq", "fun": lambda v: np.sum(v) - target_left}
        bounds = [(0.0, 1.0)] * m
        res = minimize(
            obj,
            z,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter_for_n(n), "ftol": 1e-10, "disp": False},
        )
        if not res.success:
            # keep current iterate anyway
            pass
        z = np.clip(res.x, 0.0, 1.0)
        # tiny projection onto the hyperplane (numerical drift)
        diff = target_left - np.sum(z)
        if abs(diff) > 1e-9:
            z += diff / m
            z = np.clip(z, 0.0, 1.0)

    h_opt = expand_sym(z)
    return h_opt, compute_c5_full(h_opt)

                r = np.clip(r, 0.0, 1.0)
            starts.append(r)

        # 5. Small perturbations of the current best of this size (if available)
        if best_h is not None and best_n == n:
            best_left = best_h[:m]
            for eps in (0.03, 0.07):
                pert = best_left + rng.uniform(-eps, eps, size=m)
                pert = np.clip(pert, 0.0, 1.0)
                diff = target_left - pert.sum()
                if abs(diff) > 1e-12:
                    pert += diff / m
                    pert = np.clip(pert, 0.0, 1.0)
                starts.append(pert)

        # Limit the number of starts for large n to keep within budget
        max_starts = 8 if n > 240 else 12
        starts = starts[:max_starts]

        # ----- optimise each start -----
        for z0 in starts:
            if time.time() > time_limit:
                break
            # quick pre‑check with the exact C5 (cheap) – skip hopeless starts
            cand_h = expand_sym(z0)
            cand_c5 = compute_c5_full(cand_h)
            if cand_c5 >= best_c5:
                # still try optimisation – it may improve
                pass

            h_opt, c5_opt = optimise_half(z0, n, time_limit)
            if c5_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_opt, n

    # -----------------------------------------------------------------
    # Final refinement of the overall best (if time permits)
    # -----------------------------------------------------------------
    if best_h is not None and time.time() < time_limit - 5.0:
        n = best_n
        m = n // 2
        left = best_h[:m]
        for alpha in (100.0, 150.0):
            if time.time() > time_limit:
                break
            def obj(v):
                return softmax_c5_full(expand_sym(v), alpha=alpha)
            cons = {"type": "eq", "fun": lambda v: np.sum(v) - n / 4.0}
            bounds = [(0.0, 1.0)] * m
            res = minimize(
                obj,
                left,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 250, "ftol": 1e-12, "disp": False},
            )
            if res.success:
                left = np.clip(res.x, 0.0, 1.0)
                diff = (n / 4.0) - np.sum(left)
                if abs(diff) > 1e-10:
                    left += diff / m
                    left = np.clip(left, 0.0, 1.0)
        best_h = expand_sym(left)
        best_c5 = compute_c5_full(best_h)

    # -----------------------------------------------------------------
    # Safeguard: enforce the integral exactly and final recompute
    # -----------------------------------------------------------------
    if best_h is None:
        # fallback to a simple symmetric block (guaranteed feasible)
        n = 80
        blk = np.concatenate([np.ones(n // 2), np.zeros(n // 2)]).astype(float)
        best_h = blk
        best_c5 = compute_c5_full(best_h)
        best_n = n

    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5_full(best_h)
    return best_h, final_c5, best_n
