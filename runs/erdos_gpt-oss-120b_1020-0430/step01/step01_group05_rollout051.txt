def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    g = 1 - h
    # reversed h
    h_rev = h[::-1]
    L = 2 * n
    H = np.fft.rfft(np.concatenate([h_rev, np.zeros(n)], dtype=np.float64))
    G = np.fft.rfft(np.concatenate([g, np.zeros(n)], dtype=np.float64))
    conv = np.fft.irfft(H * G, n=L)
    corr = conv[:2 * n - 1]
    return np.max(corr) * dx

import time, random, math
import numpy as np
from scipy.optimize import minimize

# Global for equality constraint
TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto [0,1]^n intersect {sum = target} using bisection on the Lagrange multiplier.
    """
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def compute_c5_fft(h: np.ndarray) -> float:
    """Exact objective using FFT for speed."""
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    h_rev = h[::-1]
    L = 2 * n
    # pad to length L
    a = np.concatenate((h_rev, np.zeros(n, dtype=np.float64)))
    b = np.concatenate((g, np.zeros(n, dtype=np.float64)))
    A = np.fft.rfft(a)
    B = np.fft.rfft(b)
    conv = np.fft.irfft(A * B, n=L)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Stable log‑sum‑exp approximation of max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth proxy for the max objective (used by SLSQP)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def make_triangle(n: int) -> np.ndarray:
    """Triangular profile, peak 1 at center."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2] (already integrates to 1)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Indicator of a unit‑length interval centred at 1."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.where((pts >= 0.5) & (pts <= 1.5), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new length."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

        except Exception:
            continue
        h_opt = project_onto_bounds_and_sum(res.x, target_sum)
        val = compute_c5_fft(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # If nothing succeeded, start from a random feasible point
    if best_h is None:
        rnd = np.random.rand(n)
        best_h = project_onto_bounds_and_sum(rnd, target_sum)
        best_val = compute_c5_fft(best_h)

    # ------------------------------------------------------------------
    # Second stage: extensive stochastic refinement (SA / hill climbing)
    cur_h = best_h.copy()
    cur_val = best_val
    temperature = 0.001
    decay = 0.9999
    min_temp = 1e-6

    while time.time() - start < budget_s - 1.0:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue

        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-14:
            continue

        cand = cur_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta

        cand_val = compute_c5_fft(cand)
        diff = cand_val - cur_val

        if diff < -1e-12 or np.random.rand() < math.exp(-diff / max(temperature, 1e-12)):
            cur_h = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

        temperature = max(min_temp, temperature * decay)

        # occasional re‑heat
        if np.random.rand() < 0.0002:
            temperature = 0.001

        # occasional random perturbation around the best found so far
        if np.random.rand() < 0.0001:
            noise = np.random.uniform(-0.05, 0.05, size=n)
            cand = best_h + noise
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_bounds_and_sum(cand, target_sum)
            cur_h = cand
            cur_val = compute_c5_fft(cur_h)

    # ------------------------------------------------------------------
    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5_fft(best_h)

    return (best_h.tolist(), float(best_val), int(n))
