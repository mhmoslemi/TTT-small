def simulated_annealing(h, target_sum, time_limit):
    start=time.time()
    cur=h.copy()
    cur_c5=c5_of_vec(cur)
    best=cur.copy()
    best_c5=cur_c5
    T = max(cur_c5 * 0.05, 1e-4)  # initial temperature
    decay=0.9995
    iter=0
    while time.time() < time_limit:
        # random swap
        i,j = random.sample(range(len(cur)),2)
        hi, hj = cur[i], cur[j]
        # compute max possible delta in values (ensuring binary)
        # ensure we swap values (just swap if hi != hj)
        if hi != hj:
            # swap
            cur[i],cur[j] = hj, hi
            new_c5 = c5_of_vec(cur)
            # accept
            if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best = cur.copy()
            else:
                # revert
                cur[i],cur[j] = hi, hj
        # target move
        if iter % 200 == 0:
            # determine worst shift and best beneficial swap within that shift
            corr = np.correlate(cur, 1 - cur, mode='full')
            idx_max=int(np.argmax(corr))
            s = idx_max - (len(cur) - 1)
            # get pairs
            if s>=0:
                i_idx = np.arange(len(cur)-s)
                p_idx = i_idx + s
            else:
                sabs = -s
                i_idx = np.arange(sabs, len(cur))
                p_idx = i_idx + s
            # compute contrib of each pair where cur[i]=1 and cur[p]=0
            cand = np.where((cur[i_idx]==1)&(cur[p_idx]==0))[0]
            if cand.size>0:
                k = random.choice(cand)
                i_best = i_idx[k]; p_best = p_idx[k]
                # swap
                cur[i_best],cur[p_best] = 0,1
                new_c5 = c5_of_vec(cur)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5)/T):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best = cur.copy()

                    # revert

import time, math, random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def c5_of_vec(v: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = v.shape[0]
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(corr.max() * (2.0 / n))


def repair_binary(v: np.ndarray, target_sum: int) -> np.ndarray:
    """Force a 0/1 vector to have exactly target_sum ones."""
    v = v.astype(float)
    cur = int(v.sum())
    if cur > target_sum:
        ones = np.where(v > 0.5)[0]
        np.random.shuffle(ones)
        for i in ones:
            if cur <= target_sum:
                break
            v[i] = 0.0
            cur -= 1
    elif cur < target_sum:
        zeros = np.where(v <= 0.5)[0]
        np.random.shuffle(zeros)
        for i in zeros:
            if cur >= target_sum:
                break
            v[i] = 1.0
            cur += 1
    return v


def targeted_improve(v: np.ndarray, max_iters: int = 20) -> np.ndarray:
    """
    Greedy local improvement that repeatedly fixes the worst shift.
    Keeps the vector binary and preserves the number of ones.
    """
    n = v.shape[0]
    for _ in range(max_iters):
        cur_c5 = c5_of_vec(v)
        corr = np.correlate(v, 1.0 - v, mode="full")
        idx_max = int(np.argmax(corr))
        s = idx_max - (n - 1)
        if s >= 0:
            i_inds = np.arange(n - s)
            p_inds = i_inds + s
        else:
            s_abs = -s
            i_inds = np.arange(s_abs, n)
            p_inds = i_inds + s
        mask = (v[i_inds] == 1) & (v[p_inds] == 0)
        cand = np.where(mask)[0]
        if cand.size == 0:
            break
        k = random.choice(cand.tolist())
        i = i_inds[k]
        p = p_inds[k]
        v[i], v[p] = 0.0, 1.0
        new_c5 = c5_of_vec(v)
        if new_c5 < cur_c5:
            continue  # keep the swap
        v[i], v[p] = 1.0, 0.0  # revert
        break
    return v


def simulated_annealing(
    h: np.ndarray, target_sum: int, deadline: float
) -> tuple[np.ndarray, float]:
    """Binary simulated annealing with occasional targeted moves."""
    n = h.shape[0]
    cur = h.copy()
    cur_c5 = c5_of_vec(cur)
    best = cur.copy()
    best_c5 = cur_c5

    T = max(cur_c5 * 0.05, 1e-4)
    decay = 0.9997
    it = 0

    while time.time() < deadline:
        # ---- random swap of two positions (preserves count) ----
        i, j = random.sample(range(n), 2)
        if cur[i] != cur[j]:
            cur[i], cur[j] = cur[j], cur[i]
            new_c5 = c5_of_vec(cur)
            delta = new_c5 - cur_c5
            if delta < 0 or random.random() < math.exp(-delta / T):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best = cur.copy()
            else:
                cur[i], cur[j] = cur[j], cur[i]

        # ---- occasional targeted move (every 200 iterations) ----
        if it % 200 == 0:
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:
                s_abs = -s
                i_inds = np.arange(s_abs, n)
                p_inds = i_inds + s
            mask = (cur[i_inds] == 1) & (cur[p_inds] == 0)
            cand = np.where(mask)[0]
            if cand.size:
                k = random.choice(cand.tolist())
                i_best = i_inds[k]
                p_best = p_inds[k]
                cur[i_best], cur[p_best] = 0.0, 1.0
                new_c5 = c5_of_vec(cur)
                delta = new_c5 - cur_c5
                if delta < 0 or random.random() < math.exp(-delta / T):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best = cur.copy()
                else:
                    cur[i_best], cur[p_best] = 1.0, 0.0

        it += 1
        if it % 1000 == 0:
            T *= decay

    return best, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()

    # --------------------------------------------------------------
    # Problem parameters (fixed length 70 – works with the provided
    # initial_h_values if present)
    # --------------------------------------------------------------
    n = 70
    target_sum = n // 2  # exactly half the entries are 1

    # --------------------------------------------------------------
    # Initial population for a simple genetic algorithm
    # --------------------------------------------------------------
    pop_size = 120
    population = []
    scores = []

    # include the supplied construction if available
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        bin_vec = (init_vec > 0.5).astype(float)
        bin_vec = repair_binary(bin_vec, target_sum)
        population.append(bin_vec)
        scores.append(c5_of_vec(bin_vec))
    except NameError:
        pass

    while len(population) < pop_size:
        v = np.zeros(n, dtype=float)
        ones = np.random.choice(n, target_sum, replace=False)
        v[ones] = 1.0
        population.append(v)
        scores.append(c5_of_vec(v))

    # keep the best seen so far
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    # --------------------------------------------------------------
    # Genetic algorithm (global search)
    # --------------------------------------------------------------
    ga_deadline = start + budget_s * 0.85
    tourn_size = 3
    cx_prob = 0.8
    mut_prob = 0.5

    while time.time() < ga_deadline:
        new_pop = []
        new_scores = []

        for _ in range(pop_size):
            # --- tournament selection for parent 1 ---
            idxs = np.random.choice(pop_size, tourn_size, replace=False)
            p1 = population[int(idxs[np.argmin([scores[i] for i in idxs])])]

            # --- tournament selection for parent 2 ---
            idxs = np.random.choice(pop_size, tourn_size, replace=False)
            p2 = population[int(idxs[np.argmin([scores[i] for i in idxs])])]

            # --- crossover ---
            if random.random() < cx_prob:
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, p1, p2).astype(float)
            else:
                child = p1.copy()

            # --- mutation (swap a 1 and a 0) ---
            if random.random() < mut_prob:
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                if ones.size and zeros.size:
                    i = np.random.choice(ones)
                    j = np.random.choice(zeros)
                    child[i], child[j] = 0.0, 1.0

            # --- repair (crossover can break the sum) ---
            if child.sum() != target_sum:
                child = repair_binary(child, target_sum)

            # --- evaluate ---
            sc = c5_of_vec(child)
            new_pop.append(child)
            new_scores.append(sc)

        # combine old and new, keep the elite
        combined_pop = population + new_pop
        combined_scores = scores + new_scores
        elite_idx = np.argsort(combined_scores)[:pop_size]
        population = [combined_pop[i] for i in elite_idx]
        scores = [combined_scores[i] for i in elite_idx]

        # improve the current best with a few targeted steps
        cur_best_idx = int(np.argmin(scores))
        cur_best = population[cur_best_idx]
        cur_best = targeted_improve(cur_best, max_iters=8)
        cur_best_score = c5_of_vec(cur_best)
        if cur_best_score < scores[cur_best_idx]:
            population[cur_best_idx] = cur_best
            scores[cur_best_idx] = cur_best_score

        # update global best
        if scores[cur_best_idx] < best_c5:
            best_c5 = scores[cur_best_idx]
            best_h = population[cur_best_idx].copy()
            if best_c5 < 0.3808:
                break

    # --------------------------------------------------------------
    # Fine‑grained local search (simulated annealing) on the best
    # --------------------------------------------------------------
    remaining = start + budget_s - time.time()
    if remaining > 5.0:
        best_h, best_c5 = simulated_annealing(best_h, target_sum, time.time() + remaining)

    # --------------------------------------------------------------
    # Final sanity check (exact half‑mass)
    # --------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        # fix any tiny drift by flipping bits
        while err > 0:
            i = random.choice(np.where(best_h == 1)[0].tolist())
            best_h[i] = 0.0
            err -= 1
        while err < 0:
            i = random.choice(np.where(best_h == 0)[0].tolist())
            best_h[i] = 1.0
            err += 1

    final_c5 = c5_of_vec(best_h)

    return (best_h.tolist(), final_c5, n)
