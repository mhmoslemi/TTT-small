    rng = np.random.RandomState(seed)

        x_new = np.linspace(0,2, n, endpoint=False)
        return np.interp(x_new, x_old, arr)

        h0 = _project_onto_capped_simplex(h0, target)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto {0<=x<=1, sum(x)=target} ----------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ---------- C5 evaluation using FFT ----------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum of h * (1-h) correlation on the interval."""
    n = h.shape[0]
    g = 1.0 - h
    N = 1 << ((2 * n - 1).bit_length())        # next power of two
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(g, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[: 2 * n - 1]
    return float(np.max(corr) * dx)


# ---------- sub‑gradient for a given shift ----------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ---------- projected (sub)gradient descent ----------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 800, step0: float = 0.5,
                 decay: float = 0.995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        grad = _gradient_for_shift(h, shift)
        h -= step * grad
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ---------- smooth max (log‑sum‑exp) ----------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


# ---------- SLSQP refinement ----------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 1000) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                             'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)

    # ----- initial feasible point -----

    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
