    grad = np.ones(n)
    if s>=0:
        # i from 0 to n-1-s: h_i+s exists
        grad[:n-s] -= h[s:]  # h_{i+s}

def project_to_feasible(h, n, tol=1e-9):
    # clip 0-1
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Water-filling iterative adjust
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute across any indices with capacity
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h_start, max_iter=200):
    n = h_start.size
    # Ensure feasible
    h0 = project_to_feasible(h_start, n)
    corr0 = np.convolve(h0, 1 - h0, mode='full')
    t0 = corr0.max()
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.convolve(h, 1 - h, mode='full')
        return t - corr

    def eq(x):
        h = x[:n]
        return np.sum(h) - (n / 2.0)

    def obj(x):
        return x[-1]

    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    # Use SLSQP from scipy.optimize
    from scipy.optimize import minimize
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = project_to_feasible(h_opt, n)
    c5_val = compute_c5(h_opt)
    return h_opt, c5_val

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_of_h(h: np.ndarray) -> float:
    """C5 = max correlation * dx (dx = 2/n)."""
    n = h.size
    dx = 2.0 / n
    corr = np.convolve(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of max‑correlation term for a given shift."""
    n = h.size
    g = np.ones(n)
    if shift >= 0:
        s = shift
        g[:n - s] -= h[s:]
        g[s:] -= h[:n - s]
    else:
        t = -shift
        g[t:] -= h[:n - t]
        g[:n - t] -= h[t:]
    return g


def _local_descent(h0: np.ndarray,
                   max_iter: int = 5000,
                   alpha0: float = 0.2,
                   decay: float = 0.995) -> tuple:
    """Projected sub‑gradient descent on max‑correlation."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_val = _c5_of_h(h)
    cur_val = best_val
    for it in range(max_iter):
        # correlation and worst shift
        conv = np.convolve(h, 1.0 - h, mode="full")
        max_val = conv.max()
        shift = int(conv.argmax() - (n - 1))

        # gradient of the active shift
        grad = _gradient_of_shift(h, shift)

        # step size
        alpha = alpha0 * (decay ** it)
        h_new = h - alpha * grad
        h_new = _project_to_feasible(h_new, n)

        new_val = _c5_of_h(h_new)

        # keep record of the best ever seen
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        # accept if not worse (simple descent)
        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
        # otherwise stay where we are

    return best_h, best_val


def _slsqp_optimize(h_start: np.ndarray, max_iter: int = 200) -> tuple:
    """Brief SLSQP refinement (together with auxiliary variable)."""
    n = h_start.size
    h0 = _project_to_feasible(h_start, n)
    t0 = np.convolve(h0, 1.0 - h0, mode="full").max()
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.convolve(h, 1.0 - h, mode="full")
        return t - corr

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def obj(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = minimize(obj, x0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_of_h(h_opt)


# ------------------------------------------------------------
# Main optimisation driver
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # obtain the (optional) parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # --------------------------------------------------------
    # 1️⃣  Choose grid sizes to explore
    # --------------------------------------------------------
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
        # Multiples of the parent size up to ~500
        for mul in (2, 3, 4):
            n = init_h.size * mul
            if n <= 500:
                n_candidates.append(n)
    # generic even sizes
    for n in (80, 100, 128, 160, 192, 224, 256, 300, 384, 400):
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    # --------------------------------------------------------
    # 2️⃣  Coarse pass: try each grid, a few restarts, short descent
    # --------------------------------------------------------
    for n in n_candidates:
        if time_left() < 5.0:
            break

        # build starting function for this n
        if init_h is not None:
            if init_h.size == n:
                base_h = init_h.copy()
            else:
                base_h = _upscale_h(init_h, n)
        else:
            base_h = np.full(n, 0.5)

        # quick descent parameters (shorter runs for coarse scan)
        max_iter = max(800, int(2000 * 200 / n))
        restarts = 3

        for r in range(restarts):
            if time_left() < 2.0:
                break
            if r == 0:
                h0 = base_h.copy()
            else:
                # noisy perturbation
                h0 = base_h + 0.25 * np.random.randn(n)
                h0 = _project_to_feasible(h0, n)

            h_opt, c5_val = _local_descent(h0, max_iter=max_iter)

            # brief SLSQP polishing if enough budget
            if time_left() > 20.0:
                h_opt, c5_val = _slsqp_optimize(h_opt, max_iter=120)

            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 < target:
                    break
        if best_c5 < target:
            break

    # --------------------------------------------------------
    # 3️⃣  Intensive refinement on the best grid found
    # --------------------------------------------------------
    if best_h is not None and time_left() > 5.0:
        n = best_n
        # aggressive descent: many iterations, many restarts
        while time_left() > 5.0:
            # start from current best with a slight random shake
            h0 = best_h + 0.1 * np.random.randn(n)
            h0 = _project_to_feasible(h0, n)

            max_iter = int(6000 * 200 / n)  # around 6000 for n~200
            h_opt, c5_val = _local_descent(h0, max_iter=max_iter, alpha0=0.15)

            # optional polishing
            if time_left() > 15.0:
                h_opt, c5_val = _slsqp_optimize(h_opt, max_iter=150)

            if c5_val < best_c5:
                best_h, best_c5 = h_opt, c5_val
                if best_c5 < target:
                    break

    # --------------------------------------------------------
    # 4️⃣  Fallback (unlikely)
    # --------------------------------------------------------
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_of_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
