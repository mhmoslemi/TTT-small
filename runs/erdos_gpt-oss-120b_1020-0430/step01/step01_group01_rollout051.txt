import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0 <= x_i <= 1, sum x_i = target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ----------------------------------------------------------------------
# Gradient of the correlation for a particular shift
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# C5 evaluation (max correlation * dx)
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


# ----------------------------------------------------------------------
# Simple symmetric up‑sampling / interpolation
# ----------------------------------------------------------------------
def _upscale(vec: np.ndarray, n_target: int) -> np.ndarray:
    if vec.shape[0] == n_target:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, vec.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_new, x_old, vec)


# ----------------------------------------------------------------------
# Symmetrisation (h(x)=h(2−x))
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Random feasible point (followed by symmetrisation)
# ----------------------------------------------------------------------
def _random_symmetric_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    v = _project_onto_capped_simplex(v, target)
    v = _symmetrize(v)
    v = _project_onto_capped_simplex(v, target)
    return v


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) used by the SLSQP refinement
# ----------------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


# ----------------------------------------------------------------------
# SLSQP refinement on a smoothed objective
# ----------------------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 40.0, max_iter: int = 800) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({"type": "eq", "fun": lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (h.shape[0] - 1)
        g = _gradient_for_shift(h, shift)

        vel = beta * vel + (1.0 - beta) * g
        h_new = h - step * vel
        h_new = _project_onto_capped_simplex(h_new, target)
        h_new = _symmetrize(h_new)
        h_new = _project_onto_capped_simplex(h_new, target)

        val_new = _compute_c5(h_new, dx)

        if val_new <= best_val:
            h = h_new
            best_val = val_new
            step = min(step * 1.02, step_max)
        else:
            step = max(step * 0.5, step_min)
            vel.fill(0.0)  # discard momentum on a rejected step

    return h, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Base solution, if provided
    # ------------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    # ------------------------------------------------------------------
    # Low‑resolution phase (fast, many restarts)
    # ------------------------------------------------------------------
    low_n = max(150, len(base_h) if base_h is not None else 0)
    low_n = min(low_n, 200)  # keep it modest
    target_low = low_n / 2.0
    dx_low = 2.0 / low_n

    if base_h is not None:
        h0_low = _upscale(base_h, low_n)
        h0_low = _project_onto_capped_simplex(h0_low, target_low)
        h0_low = _symmetrize(h0_low)
        h0_low = _project_onto_capped_simplex(h0_low, target_low)
        best_h = h0_low
        best_val = _compute_c5(best_h, dx_low)
    else:
        best_h = _random_symmetric_feasible(rng, low_n, target_low)
        best_val = _compute_c5(best_h, dx_low)

    # ------------------------------------------------------------------
    # Random‑restart GD on the low grid
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.45:
        if rng.rand() < 0.2:
            h_start = best_h + rng.normal(scale=0.02, size=low_n)
        else:
            h_start = _random_symmetric_feasible(rng, low_n, target_low)

        h_start = _project_onto_capped_simplex(h_start, target_low)
        h_start = _symmetrize(h_start)
        h_start = _project_onto_capped_simplex(h_start, target_low)

        h_opt, val_opt = _gd_optimize(
            h_start,
            target_low,
            dx_low,
            max_iter=1500,
            step0=0.5,
            beta=0.9,
            step_min=1e-6,
            step_max=1.0,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # ------------------------------------------------------------------
    # Upscale to a finer grid for final polishing
    # ------------------------------------------------------------------
    high_n = max(300, low_n * 2)
    target_high = high_n / 2.0
    dx_high = 2.0 / high_n

    best_h = _upscale(best_h, high_n)
    best_h = _project_onto_capped_simplex(best_h, target_high)
    best_h = _symmetrize(best_h)
    best_h = _project_onto_capped_simplex(best_h, target_high)
    best_val = _compute_c5(best_h, dx_high)

    # ------------------------------------------------------------------
    # High‑resolution optimisation (fewer restarts, longer runs)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # perturb current best
        h_start = best_h + rng.normal(scale=0.01, size=high_n)
        h_start = _project_onto_capped_simplex(h_start, target_high)
        h_start = _symmetrize(h_start)
        h_start = _project_onto_capped_simplex(h_start, target_high)

        h_opt, val_opt = _gd_optimize(
            h_start,
            target_high,
            dx_high,
            max_iter=2000,
            step0=0.4,
            beta=0.9,
            step_min=1e-6,
            step_max=1.0,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

        # occasional SLSQP refinement (smooth‑max objective)
        if rng.rand() < 0.1:
            refined = _slsqp_optimize(best_h, target_high, alpha=80.0, max_iter=1200)
            refined = _symmetrize(refined)
            refined = _project_onto_capped_simplex(refined, target_high)
            val_ref = _compute_c5(refined, dx_high)
            if val_ref < best_val:
                best_val = val_ref
                best_h = refined.copy()

    # final safety projection
    best_h = _project_onto_capped_simplex(best_h, target_high)
    best_h = _symmetrize(best_h)
    best_h = _project_onto_capped_simplex(best_h, target_high)
    best_val = _compute_c5(best_h, dx_high)

    return (best_h, float(best_val), high_n)
