def improve_binary(h):
    while True:
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        max_idx = np.argmax(corr)
        if max_val == 0: break
        L = max_idx - (len(h) - 1)
        # compute overlap mask for shift L
        w = np.zeros(len(h), dtype=np.int8)
        if L >= 0:
            w[:len(h)-L] = 1 - h[L:]  # vector of 1 - h[i+L]
        else:
            w[-L:] = 1 - h[:len(h)+L]  # when L negative
        #  find any i with h[i]==1 and w[i]==1
        over_i = np.where((h == 1) & (w == 1))[0]
        # find any j with h[j]==0 and w[j]==0
        nonover_j = np.where((h == 0) & (w == 0))[0]
        if over_i.size == 0 or nonover_j.size == 0:
            # cannot improve using this shift, maybe try alternative move?
            # break greedy loop.
            break
        i = over_i[0]  # pick first
        j = nonover_j[0]
        # swap
        h[i] = 0
        h[j] = 1
        # continue loop
    return h, max_val

def _greedy_reduce_binary(h, max_iter=2000):
    # h assumed binary
    n = h.shape[0]
    best_c_raw = _raw_c5_metric(h)
    iteration = 0
    while iteration < max_iter:
        iteration += 1
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = int(np.max(corr))
        if max_val == 0:
            break
        max_idx = int(np.argmax(corr))
        L = max_idx - (n - 1)
        # compute overlap mask w (1 if h[i]=1 overlaps zero at i+L)
        w = np.zeros(n, dtype=np.int8)
        if L >= 0:
            # For i from 0 to n-1-L
            # w[i] = 1 - h[i+L]
            w[:n-L] = 1 - h[L:]
        else:
            # L negative: shift left; w[i] = 1 - h[i+L] where i+L >=0 => i >= -L
            w[-L:] = 1 - h[:n+L]
        # overlapping ones indices
        over_i = np.where((h == 1) & (w == 1))[0]
        # good zeros (zeros aligning with ones) indices
        good_j = np.where((h == 0) & (w == 0))[0]
        if over_i.size == 0 or good_j.size == 0:
            # can't improve this worst lag; try look at second worst maybe.
            # break out because we can't reduce max.
            break
        i = over_i[0]
        j = good_j[0]
        # perform swap
        h[i] = 0
        h[j] = 1
        new_max = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
        if new_max < max_val:
            best_c_raw = new_max
            # continue with new configuration
            continue
        else:
            # revert
            h[i] = 1
            h[j] = 0
            # try swapping other candidates up to limited attempt
            improved = False
            # limit attempts to maybe 10 pairs
            for ii in over_i[:min(5, over_i.size)]:
                for jj in good_j[:min(5, good_j.size)]:
                    h[ii] = 0
                    h[jj] = 1
                    new_max2 = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
                    if new_max2 < max_val:
                        best_c_raw = new_max2
                        improved = True
                        break
                    else:
                        h[ii] = 1
                        h[jj] = 0
                if improved:
                    break
            if not improved:
                break

    final_c_raw = _raw_c5_metric(h)

def _anneal_general(start_h, time_limit):
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    start = time.time()
    T0 = 0.06
    T_end = 1e-6
    max_step = 0.12
    while time.time() - start < time_limit:
        # compute progress and T
        progress = (time.time() - start) / time_limit
        T = T0 * (T_end / T0) ** progress
        # decide where to select indices
        # compute worst shift occasionally
        if random.random() < 0.2:
            # compute once to get weighting
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            L = max_idx - (n - 1)
            # compute w array
            w = np.zeros(n)
            if L >= 0:
                w[:n-L] = 1.0 - h[L:]
            else:
                w[-L:] = 1.0 - h[:n+L]
            # compute probabilities
            prob_i = w + 0.01  # ensure positive
            prob_i /= prob_i.sum()
            prob_j = (1.0 - w) + 0.01
            prob_j /= prob_j.sum()
            # sample i, j using np.random.choice with p
            i = np.random.choice(n, p=prob_i)
            j = np.random.choice(n, p=prob_j)
            # ensure i != j; if equal resample
            while j == i:
                j = np.random.choice(n, p=prob_j)
        else:
            i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        # feasible delta range based on box constraints and sum preservation
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # restrict to modest step size: consider cap
        low_cap = max(low, -max_step)

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic metric helpers
# ----------------------------------------------------------------------
def _c5_metric(h: np.ndarray) -> float:
    """Scaled C5 value (max correlation * dx)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * (2.0 / n)


def _raw_c5_metric(h: np.ndarray) -> int:
    """Raw (unscaled) max correlation – integer for binary h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return int(corr.max())


# ----------------------------------------------------------------------
# Binary‑only simulated annealing
# ----------------------------------------------------------------------
def _binary_anneal(start_h: np.ndarray, time_limit: float):
    """Anneal a binary vector (0/1) while preserving the exact sum."""
    h = start_h.copy()
    n = h.size
    cur = _raw_c5_metric(h)
    best_h = h.copy()
    best = cur
    start = time.time()
    T0, T_end = 0.08, 1e-5

    while time.time() - start < time_limit:
        prog = (time.time() - start) / time_limit
        T = T0 * (T_end / T0) ** prog

        # pick a random 1 and a random 0 to swap
        i = np.random.randint(n)
        if h[i] == 1:
            j = np.random.randint(n)
            while h[j] != 0:
                j = np.random.randint(n)
        else:
            j = np.random.randint(n)
            while h[j] != 1:
                j = np.random.randint(n)

        # swap
        h[i] = 0.0
        h[j] = 1.0
        new = _raw_c5_metric(h)
        d = new - cur
        if d < 0 or random.random() < math.exp(-d / T):
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best


# ----------------------------------------------------------------------
# Greedy reduction for binary vectors
# ----------------------------------------------------------------------
def _greedy_reduce_binary(h: np.ndarray, max_iter: int = 2000):
    """Attempt to lower the maximal overlap by deterministic swaps."""
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = int(corr.max())
        if max_val == 0:
            break
        max_idx = int(corr.argmax())
        L = max_idx - (n - 1)

        # build overlap mask w[i] = 1 - h[i+L] where defined
        w = np.zeros(n, dtype=np.int8)
        if L >= 0:
            w[: n - L] = 1 - h[L:]
        else:
            w[-L :] = 1 - h[: n + L]

        over_i = np.where((h == 1) & (w == 1))[0]
        good_j = np.where((h == 0) & (w == 0))[0]
        if over_i.size == 0 or good_j.size == 0:
            break

        i, j = over_i[0], good_j[0]
        h[i] = 0.0
        h[j] = 1.0
        new_max = int(np.correlate(h, 1.0 - h, mode="full").max())
        if new_max < max_val:
            continue  # improvement accepted
        # revert and try a few alternatives
        h[i] = 1.0
        h[j] = 0.0
        improved = False
        for ii in over_i[: min(5, over_i.size)]:
            for jj in good_j[: min(5, good_j.size)]:
                h[ii] = 0.0
                h[jj] = 1.0
                if int(np.correlate(h, 1.0 - h, mode="full").max()) < max_val:
                    improved = True
                    break
                h[ii] = 1.0
                h[jj] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, _raw_c5_metric(h)


# ----------------------------------------------------------------------
# General (fractional) simulated annealing with bias toward worst lag
# ----------------------------------------------------------------------
def _anneal_general(start_h: np.ndarray, time_limit: float):
    """Continuous‑valued annealing; moves preserve the total sum."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    start = time.time()
    T0, T_end = 0.06, 1e-6
    max_step = 0.12

    while time.time() - start < time_limit:
        prog = (time.time() - start) / time_limit
        T = T0 * (T_end / T0) ** prog

        # 20% of the time bias the choice of indices using the worst lag
        if random.random() < 0.2:
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_idx = int(corr.argmax())
            L = max_idx - (n - 1)

            w = np.zeros(n)
            if L >= 0:
                w[: n - L] = 1.0 - h[L:]
            else:
                w[-L :] = 1.0 - h[: n + L]

            prob_i = w + 0.01
            prob_i /= prob_i.sum()
            prob_j = (1.0 - w) + 0.01
            prob_j /= prob_j.sum()
            i = np.random.choice(n, p=prob_i)
            j = np.random.choice(n, p=prob_j)
            while j == i:
                j = np.random.choice(n, p=prob_j)
        else:
            i, j = random.sample(range(n), 2)

        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # modest step cap
        lowc = max(low, -max_step)
        highc = min(high, max_step)
        if lowc < highc:
            delta = random.uniform(lowc, highc)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    return best_h, best_c

    return h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C₅ bound and sample count."""

    random.seed(seed)
    np.random.seed(seed)

    start_all = time.time()
    total_budget = float(budget_s)

    # ------------------------------------------------------------------
    # Choose a working resolution (even for exact half‑ones)
    # ------------------------------------------------------------------
    n = 400
    if n % 2:
        n += 1
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Binary search phase – many short annealing runs with greedy polishing
    # ------------------------------------------------------------------
    best_h = None
    best_raw = np.inf

    while True:
        elapsed = time.time() - start_all
        remaining = total_budget - elapsed
        if remaining < 15.0:  # not enough time for another full binary run
            break

        # allocate a slice of time for this run
        slice_time = min(60.0, max(10.0, remaining * 0.25))

        # random binary start (exactly n/2 ones)
        h0 = np.zeros(n, dtype=np.float64)
        idx = np.random.permutation(n)
        h0[idx[: n // 2]] = 1.0

        # anneal binary
        h_bin, raw_val = _binary_anneal(h0, slice_time)

        # greedy reduction
        h_bin, raw_val = _greedy_reduce_binary(h_bin, max_iter=2000)

        if raw_val < best_raw:
            best_raw = raw_val
            best_h = h_bin.copy()
            # early stop if already below the target bound
            if raw_val * (2.0 / n) <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Fallback: if binary phase never produced a candidate, use interpolation
    # ------------------------------------------------------------------
    if best_h is None:
        if "initial_h_values" in globals():
            base = np.asarray(initial_h_values, dtype=np.float64)
            n0 = base.size
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            best_h = np.interp(x, x0, base)
        else:
            best_h = np.full(n, 0.5, dtype=np.float64)

        # enforce the integral constraint exactly
        cur = best_h.sum()
        if cur != 0:
            best_h *= target_sum / cur
        np.clip(best_h, 0.0, 1.0, out=best_h)
        diff = target_sum - best_h.sum()
        if abs(diff) > 1e-12:
            for ii in np.random.permutation(n):
                if 0.0 <= best_h[ii] + diff <= 1.0:
                    best_h[ii] += diff
                    break

    # ------------------------------------------------------------------
    # Refinement with continuous values (fractional moves)
    # ------------------------------------------------------------------
    remaining = total_budget - (time.time() - start_all)
    if remaining > 5.0:
        # spend most of the rest on a focused anneal
        refine_time = remaining * 0.7
        h_refined, c_refined = _anneal_general(best_h, refine_time)
        # tiny hill‑climb
        h_refined, c_refined = _final_hill_climb(h_refined, steps=3000)
    else:
        h_refined = best_h
        c_refined = _c5_metric(h_refined)

    # Final repair of the sum (tiny rounding errors)
    # ------------------------------------------------------------------
    diff = target_sum - h_refined.sum()
    if abs(diff) > 1e-12:
        for ii in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h_refined[ii] + diff <= hi:
                h_refined[ii] += diff
                break

    # Compute the final bound (use the provided helper if available)
    # ------------------------------------------------------------------
    try:
        final_c = evaluate_erdos_solution(h_refined)
    except Exception:
        final_c = _c5_metric(h_refined)

    return h_refined, float(final_c), int(n)
