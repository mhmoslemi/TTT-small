        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.05
        h[i], h[j] = hi + delta, hj - delta
        new_c = _c5_metric(h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
        else:
            h[i], h[j] = hi, hj
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C₅ bound and the number of samples."""
    random.seed(seed)
    np.random.seed(seed)
    total_budget = float(budget_s)
    start_all = time.time()

    # ---- coarse optimisation -------------------------------------------------
    n_coarse = 200
    h0 = _initial_vector(n_coarse)
    time_coarse = total_budget * 0.45
    best_h, best_c = _anneal(h0, time_coarse)

    # ---- up‑sample to a finer grid -----------------------------------------
    n_fine = 400
    x_old = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    h_fine = np.interp(x_new, x_old, best_h)
    h_fine = _project_to_feasible(h_fine, n_fine / 2.0)

    # ---- fine‑grained annealing ---------------------------------------------
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 10.0:
        fine_h, fine_c = _anneal(h_fine, remaining * 0.8)
        if fine_c < best_c:
            best_h, best_c = fine_h, fine_c
        # small deterministic polishing
        remaining = total_budget - (time.time() - start_all)
        if remaining > 5.0:
            best_h = _hill_climb(best_h, max_iter=5000, max_time=remaining)

    # ---- final feasibility check --------------------------------------------
    target_sum = best_h.size / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(best_h.size):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # ---- compute final bound -------------------------------------------------
    try:
        final_c = evaluate_erdos_solution(best_h)      # pre‑imported helper
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(best_h.size)
