import time, random, math
import numpy as np
from scipy.optimize import minimize

# Global for the equality constraint
TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n ∩ {sum = target} by bisection on the Lagrange multiplier."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(arr: np.ndarray, alpha: float) -> float:
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx

def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - TARGET_SUM_GLOBAL

def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_rect(n: int) -> np.ndarray:
    half = n // 2
    start = (n - half) // 2
    h = np.zeros(n)
    h[start:start + half] = 1.0
    return h

def make_sinp(n: int, p: float) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    raw = np.sin(np.pi * pts / 2.0) ** p
    scaling = (n / 2.0) / raw.sum()
    h = raw * scaling
    return project_onto_bounds_and_sum(h, n / 2.0)

def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

def targeted_move(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Transfer a small amount from a high-contributing index to a low one w.r.t. the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # positive means i+shift in range
    if shift >= 0:
        contrib = h[: n - shift] * (1.0 - h[shift:])
        i = int(np.argmax(contrib))
        j = int(np.argmin(contrib))
    else:
        s = -shift
        contrib = h[s:] * (1.0 - h[: n - s])
        i = int(np.argmax(contrib)) + s
        j = int(np.argmin(contrib))
    hi, hj = h[i], h[j]
    delta_max = min(1.0 - hi, hj)
    delta_min = -min(hi, 1.0 - hj)
    if delta_max < delta_min:
        return None
    delta = np.random.uniform(delta_min, delta_max) * 0.5
    if abs(delta) < 1e-12:
        return None
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    return project_onto_bounds_and_sum(cand, target_sum)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    n = 250                     # grid size (even)
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # Build diverse initial seeds
    seeds = []

    if "initial_h_values" in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rect(n))
    for p in (0.5, 1.0, 1.5):
        try:
            seeds.append(make_sinp(n, p))
        except Exception:
            pass

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Multi‑stage SLSQP optimisation for each seed
    alphas = (20.0, 100.0, 500.0)
    maxiter_stage = 250

    for h0 in seeds:
        if time.time() - start > budget_s - 1.0:
            break
        h = project_onto_bounds_and_sum(h0, target_sum)
        for alpha in alphas:
            try:
                res = minimize(
                    objective_smooth,
                    h,
                    args=(dx, alpha),
                    method="SLSQP",
                    bounds=bounds,
                    constraints={"type": "eq", "fun": sum_eq_constraint},
                    options={"maxiter": maxiter_stage, "ftol": 1e-9, "disp": False},
                )
                h = project_onto_bounds_and_sum(res.x, target_sum)
            except Exception:
                continue
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
            if best_val <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Random feasible restarts with a single high‑alpha optimisation
    while time.time() - start < budget_s - 2.0:
        rnd = np.random.rand(n)
        h0 = project_onto_bounds_and_sum(rnd, target_sum)
        try:
            res = minimize(
                objective_smooth,
                h0,
                args=(dx, 500.0),
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": 150, "ftol": 1e-9, "disp": False},
            )
            h = project_onto_bounds_and_sum(res.x, target_sum)
            val = compute_c5(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
                if best_val <= 0.3808:
                    break
        except Exception:
            continue

    # ------------------------------------------------------------------
    # Local refinement: targeted moves + occasional random pair swaps
    while time.time() - start < budget_s - 0.5:
        # try a move aimed at the current worst shift
        cand = targeted_move(best_h, target_sum)
        if cand is not None:
            cval = compute_c5(cand)
            if cval < best_val:
                best_val, best_h = cval, cand
                if best_val <= 0.3808:
                    break
                continue
        # fallback random pairwise change
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max) * 0.5
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i], cand[j] = hi + delta, hj - delta
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cval = compute_c5(cand)
        if cval < best_val:
            best_val, best_h = cval, cand
            if best_val <= 0.3808:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
