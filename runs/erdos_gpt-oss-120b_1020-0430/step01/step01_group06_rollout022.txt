            if err > 0:
                cand = np.where(best_h > 1e-12)[0]
                if not len(cand):
                    break
                i = random.choice(cand.tolist())
                delta = min(best_h[i], err)
                best_h[i] -= delta
                err -= delta
            else:
                cand = np.where(best_h < 1.0 - 1e-12)[0]
                if not len(cand):
                    break
                i = random.choice(cand.tolist())
                delta = min(1.0 - best_h[i], -err)
                best_h[i] += delta
                err += delta
            if abs(err) < 1e-12:
                break

    final_c5 = _c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
