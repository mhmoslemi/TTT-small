def compute_c5_fft(h):

    N = 1 << ((2*n-1).bit_length())

    a = np.fft.rfft(h, N)

def make_triangle(n):
    dx=2.0/n
    pts = np.arange(n)*dx + dx/2
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n/2)

def make_sin2(n):
    dx=2.0/n

def project_onto_bounds_and_sum(x, target):
    # considering x some array.
    # We want to find lambda such that proj = np.clip(x - lam, 0, 1) sums to target.
    # Use bisection with bounds lam low, high.
    # Lower bound lam_low = -1 (makes x - lam > x+1; sum likely > target) and lam_high=1 (makes x - lam < x-1; sum < target). Might not guarantee for any x? typical approach.
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - ((low+high)*0.5), 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    a = np.fft.rfft(h, N)
    b = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(a * np.conj(b), N)[:2*n-1]
    # correlation max over all shifts corresponds to max of corr

import time
import numpy as np

# ---------- utilities ----------
def _project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n ∩ {sum = target} via bisection on the Lagrange multiplier."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def _correlation_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (full, zero‑padded)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    a = np.fft.rfft(h, N)
    b = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(a * np.conj(b), N)[: 2 * n - 1]
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Exact objective for a discretised step function."""
    n = h.shape[0]
    return float(np.max(_correlation_full(h)) * (2.0 / n))


def _make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return _project_onto_bounds_and_sum(h, n / 2.0)


def _make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return _project_onto_bounds_and_sum(h, n / 2.0)


def _make_rectangle(n: int) -> np.ndarray:
    """Width‑1 rectangle centred at 1.0."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = ((pts >= 0.5) & (pts <= 1.5)).astype(float)
    return _project_onto_bounds_and_sum(h, n / 2.0)


def _upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing vector to a new length."""
    n0 = init.shape[0]
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False) + (2.0 / n0) * 0.5
    x_new = np.linspace(0.0, 2.0, n, endpoint=False) + (2.0 / n) * 0.5
    h = np.interp(x_new, x_old, init)
    return _project_onto_bounds_and_sum(h, n / 2.0)


def _gradient_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Subgradient of the correlation at a given integer shift s."""
    n = h.shape[0]
    g = np.ones_like(h)
    if s >= 0:
        g[: n - s] -= h[s:]
        g[s:] -= h[: n - s]
    else:
        t = -s
        g[t:] -= h[: n - t]
        g[: n - t] -= h[t:]
    return g


# ---------- main optimisation ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low Erdős C₅ value."""
    np.random.seed(seed)
    start = time.perf_counter()

    # ----- evaluate seeds and pick the best as a start -----
    best_h = None
    best_val = np.inf
    for h0 in seeds:

        if val < best_val:
            best_val = val
            best_h = h0.copy()

    cur = best_h.copy()
    cur_val = best_val

    # ----- stochastic local optimisation -----
    # temperature schedule (simulated annealing)
    TEMP_START = 0.05
    TEMP_END = 1e-4

    while time.perf_counter() - start < budget_s - 0.3:
        elapsed = time.perf_counter() - start
        remaining = max(0.0, budget_s - elapsed)
        # linear cooling based on remaining fraction
        temp = TEMP_END + (TEMP_START - TEMP_END) * (remaining / budget_s)
