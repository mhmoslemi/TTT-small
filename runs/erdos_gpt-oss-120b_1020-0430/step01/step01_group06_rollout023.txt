import numpy as np, time, random, math, cvxpy as cp

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max() * (2.0 / len(h)))

def solve_lp(n, max_time_per_n):
    # Setup LP variables and constraints.
    x = cp.Variable(n)
    M = cp.Variable()
    constraints = [cp.sum(x) == n/2,
                   x >= 0,
                   x <= 1,
                   M >= 0]
    for s in range(n):
        L = n - s
        if L < 1:
            continue
        a = x[:L]
        b = 1 - x[s:]
        w = cp.Variable(L)
        constraints += [w >= 0,
                        w >= a + b - 1,
                        w <= a,
                        w <= b,
                        cp.sum(w) <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    # Solve with ECOS or other solver
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000, verbose=False)
    except Exception:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback: maybe not solved
        return None, None
    x_val = np.clip(x.value, 0, 1)
    M_val = M.value
    return x_val, M_val

def local_search(h, target_time, start_temp=0.1, decay=0.9995):
    n=len(h)
    cur_h = h.copy()
    cur_c5 = c5_of(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    T = start_temp
    iteration=0
    start = time.time()
    while time.time() - start < target_time:
        # random move: pick i,j, adjust delta as before
        i,j = random.sample(range(n),2)
        # propose new values ensuring bounds & sum constraint.
        # We'll move some epsilon from i to j:
        hi, hj = cur_h[i], cur_h[j]
        max_up = min(1.0 - hi, hj)    # increase i, decrease j
        max_down = min(hi, 1 - hj)    # decrease i, increase j
        if max_up + max_down == 0:
            continue
        delta = (random.random() * (max_up + max_down)) - max_down
        if abs(delta) < 1e-12:
            continue
        nh = cur_h.copy()
        nh[i] = hi + delta
        nh[j] = hj - delta
        new_c5 = c5_of(nh)
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = nh.copy()
        iteration += 1
        if iteration % 1000 == 0:
            T *= decay
    return best_h, best_c5

            hi, hj = cur_h[i], cur_h[j]
            max_up = min(1.0 - hi, hj)
            max_down = min(hi, 1.0 - hj)
            if max_up + max_down > 0:

                if abs(delta) > 1e-14:

                    nh[j] = hj - delta

                        cur_h = nh
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = nh.copy()
            it += 1
            if it % 1000 == 0:
                T *= decay

    # Final projection to ensure sum exactly n/2
    # else adjust
    final_h = best_h.copy()
    target_sum = best_n / 2.0
    err = final_h.sum() - target_sum
    if abs(err) > 1e-8:
        # adjust by distributing error across random indices
        adjust_indices = np.where((err > 0) & (final_h > 0) | (err < 0) & (final_h < 1))[0]

                err = 0
                break
    final_c5 = compute_c5(final_h)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math, sys

    # optional cvxpy import (fallback to pure heuristic if unavailable)
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        cp = None

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ---------- utilities ----------
    def c5_of(v):
        """max overlap * dx for a vector v."""
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(corr.max() * (2.0 / len(v)))

    def project_sum_one(v, target):
        """Project v onto the hyperplane sum(v)=target while staying in [0,1]."""
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        if abs(cur - target) < 1e-12:
            return v
        # simple water‑filling correction
        for _ in range(100):
            diff = cur - target
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = v > 1e-12
                if not mask.any():
                    break
                delta = diff / mask.sum()
                v[mask] -= np.minimum(v[mask], delta)
            else:
                mask = v < 1 - 1e-12
                if not mask.any():
                    break
                delta = (-diff) / mask.sum()
                v[mask] += np.minimum(1.0 - v[mask], delta)
            cur = v.sum()
        return v

    # ---------- problem parameters ----------
    # try to use the provided initial vector, otherwise start from a random binary one
    try:
        init = np.array(initial_h_values, dtype=float).copy()
        n0 = init.shape[0]
    except Exception:  # pragma: no cover
        n0 = 70
        init = np.zeros(n0)
        idx = np.arange(n0)
        np.random.shuffle(idx)
        init[idx[: n0 // 2]] = 1.0

    # we will explore a small set of lengths; keep them even so half‑sum is integer
    n_candidates = [n0, 80, 90, 100, 120]
    best_h, best_c5, best_n = None, float('inf'), None

    # ---------- linear‑programming relaxation (if possible) ----------
    def lp_relaxation(n, max_time):
        """Return a feasible (possibly fractional) vector with a low max overlap."""
        if cp is None:
            return None, None
        half = n // 2
        x = cp.Variable(n)
        M = cp.Variable()
        cons = [cp.sum(x) == half,
                x >= 0,
                x <= 1,
                M >= 0]
        # McCormick envelopes for w_{i,s}=x_i*(1-x_{i+s})
        for s in range(n):
            L = n - s
            if L == 0:
                continue
            a = x[:L]
            b = 1 - x[s:]
            w = cp.Variable(L)
            cons += [w >= 0,
                     w >= a + b - 1,
                     w <= a,
                     w <= b,
                     cp.sum(w) <= M]
        prob = cp.Problem(cp.Minimize(M), cons)
        # ECOS is usually fast for this size; give it a modest iteration cap
        try:
            prob.solve(solver=cp.ECOS, max_iters=20000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=25000, verbose=False)
            except Exception:
                return None, None
        if prob.status not in ("optimal", "optimal_inaccurate"):
            return None, None
        xv = np.clip(x.value, 0.0, 1.0)
        xv = project_sum_one(xv, half)
        return xv, prob.value

    # ---------- main search over candidate lengths ----------
    for n in n_candidates:
        if time.time() - start > budget_s * 0.9:
            break
        half = n // 2
        # try LP relaxation first
        remaining = budget_s - (time.time() - start)
        lp_vec, lp_M = lp_relaxation(n, remaining * 0.5)
        if lp_vec is not None:
            cur_c5 = c5_of(lp_vec)
            if cur_c5 < best_c5:
                best_c5, best_h, best_n = cur_c5, lp_vec.copy(), n

        # if we already have a good bound, we can stop early
        if best_c5 <= 0.38085:
            break

    # ---------- fallback initialisation ----------
    if best_h is None:
        # use the supplied initial vector (or a fresh binary one) as start
        best_h = init.copy()
        best_n = n0
        target_sum = best_n / 2.0
        best_h = project_sum_one(best_h, target_sum)
        best_c5 = c5_of(best_h)

    # ---------- local refinement (simulated annealing) ----------
    def local_search(h, time_limit):
        n = h.shape[0]
        target_sum = n / 2.0
        cur = h.copy()
        cur_c5 = c5_of(cur)
        best_loc = cur.copy()
        best_loc_c5 = cur_c5

        T = max(1e-4, cur_c5 * 0.05)
        decay = 0.9995
        it = 0
        t0 = time.time()
        while time.time() - t0 < time_limit:
            # --- targeted move (every 400 it) ---
            if it % 400 == 0:
                corr = np.correlate(cur, 1 - cur, mode='full')
                idx = int(np.argmax(corr))
                s = idx - (n - 1)
                if s >= 0:
                    i_inds = np.arange(n - s)
                    p_inds = i_inds + s
                else:
                    s_abs = -s
                    i_inds = np.arange(s_abs, n)
                    p_inds = i_inds + s
                contrib = cur[i_inds] * (1.0 - cur[p_inds])
                if contrib.size:
                    k = int(np.argmax(contrib))
                    i0, p0 = i_inds[k], p_inds[k]
                    eps_max = min(cur[i0], 1.0 - cur[p0])
                    if eps_max > 1e-12:
                        eps = eps_max * 0.5
                        nh = cur.copy()
                        nh[i0] -= eps
                        nh[p0] += eps
                        new_c5 = c5_of(nh)
                        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                            cur, cur_c5 = nh, new_c5
                            if new_c5 < best_loc_c5:
                                best_loc_c5, best_loc = new_c5, nh.copy()

            # --- random two‑coordinate exchange preserving the sum ---
            i, j = random.sample(range(n), 2)
            hi, hj = cur[i], cur[j]
            max_up = min(1.0 - hi, hj)
            max_down = min(hi, 1.0 - hj)
            if max_up + max_down > 0:
                delta = (random.random() * (max_up + max_down)) - max_down
                if abs(delta) > 1e-14:
                    nh = cur.copy()
                    nh[i] = hi + delta
                    nh[j] = hj - delta
                    new_c5 = c5_of(nh)
                    if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                        cur, cur_c5 = nh, new_c5
                        if new_c5 < best_loc_c5:
                            best_loc_c5, best_loc = new_c5, nh.copy()

            it += 1
            if it % 1000 == 0:
                T *= decay
        return best_loc, best_loc_c5

    time_left = budget_s - (time.time() - start)
    if time_left > 5:
        refined_h, refined_c5 = local_search(best_h, time_left * 0.9)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5
            best_n = best_h.shape[0]

    # ---------- final projection to eliminate numerical drift ----------
    target_sum = best_n / 2.0
    err = best_h.sum() - target_sum
    if abs(err) > 1e-8:
        # simple correction distributing the error uniformly
        step = err / best_n
        best_h -= step
        best_h = np.clip(best_h, 0.0, 1.0)
        # final re‑project if needed
        best_h = project_sum_one(best_h, target_sum)

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, int(best_n))
