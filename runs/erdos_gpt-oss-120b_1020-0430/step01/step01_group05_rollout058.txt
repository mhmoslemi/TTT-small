import time, random, math
import numpy as np

# ----- helper functions ----------------------------------------------------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Projection of x onto [0,1]^n intersect sum(x)=target via bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)  # peak 1 at centre, linear to 0 at edges
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rect(n: int) -> np.ndarray:
    """Indicator of length 1 centred at 1 (height 1)."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.where((xs >= 0.5) & (xs <= 1.5), 1.0, 0.0)
    # exact mass may be off by at most one sample; fix with projection
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, n / 2.0)


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Subgradient of the max‑correlation term for a given integer shift."""
    n = h.shape[0]
    g = np.ones(n)
    s = shift
    if s >= 0:
        if s < n:
            g[: n - s] -= h[s:]
            g[s:] -= h[: n - s]
    else:
        s = -s
        if s < n:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
    return g


def smooth_gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the smooth max (log‑sum‑exp) objective."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_term = np.exp(alpha * (corr - m))
    w = exp_term / exp_term.sum()               # softmax weights, sum to 1
    w_rev = w[::-1]

    # convolution to get Σ w_s h_{i+s} and Σ w_s h_{i‑s}
    conv2 = np.convolve(w_rev, h, mode="full")[n - 1 : n - 1 + n]
    conv3 = np.convolve(w, h, mode="full")[n - 1 : n - 1 + n]

    grad = (1.0 - conv2 - conv3) * dx
    return grad

            step = 0.5 / (inner + 1.0)
            h_new = h - step * grad
            h_new = project_onto_bounds_and_sum(h_new, target_sum)
            val = compute_c5(h_new)
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
                h = h_new
            else:
                h = h_new
            if best_val <= 0.3808:
                break
        if best_val <= 0.3808:
            break

    # ----- final random pairwise tweaks ------------------------------------
    h = best_h.copy()
    while time.time() - start < budget_s - 2.0:
        i, j = random.randint(0, n - 1), random.randint(0, n - 1)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = random.uniform(delta_min, delta_max)
        cand = h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cval = compute_c5(cand)
        if cval < best_val:
            best_val = cval
            best_h = cand.copy()
            h = cand
        if best_val <= 0.3808:
            break

    # final safety projection and evaluation
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
