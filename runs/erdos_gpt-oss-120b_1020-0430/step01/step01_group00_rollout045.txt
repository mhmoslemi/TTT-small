import time
import numpy as np
from scipy import optimize

def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discrete step function h on [0,2]."""
    n = h.size
    # linear correlation (full) between h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_to_feasible_via_offset(raw0: np.ndarray, target_sum: float) -> np.ndarray:
    """Find offset a0 so that tanh(a0+raw0) scaled to [0,1] has the required sum."""
    # initial bracket
    lo, hi = -20.0, 20.0
    # ensure the bracket contains the root
    def _sum_at(offset):
        return ((np.tanh(offset + raw0) + 1.0) * 0.5).sum()
    s_lo = _sum_at(lo)
    s_hi = _sum_at(hi)
    # expand if needed
    while s_lo > target_sum:
        lo -= 20.0
        s_lo = _sum_at(lo)
    while s_hi < target_sum:
        hi += 20.0
        s_hi = _sum_at(hi)
    # binary search
    for _ in range(30):
        mid = (lo + hi) * 0.5
        s_mid = _sum_at(mid)
        if s_mid > target_sum:
            hi = mid
        else:
            lo = mid
    a0 = (lo + hi) * 0.5
    return ((np.tanh(a0 + raw0) + 1.0) * 0.5)
