import time, numpy as np

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_sum(x, target):
    # Clip to [0,1] and adjust sum via iterative redistribution
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Use water-filling like algorithm: distribute diff uniformly among free variables
    # We'll loop a few times.
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # Determine indices that can receive (+) or give (-)
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        candidates = np.where(mask)[0]
        per = diff / len(candidates)
        # Update
        if diff > 0:
            add = np.minimum(per, 1.0 - x[candidates])
            x[candidates] += add
            diff -= add.sum()
        else:
            sub = np.minimum(-per, x[candidates])
            x[candidates] -= sub
            diff += sub.sum()
    return np.clip(x, 0.0, 1.0)

def _triangular_init(n):
    x = np.arange(n)
    center = (n-1)/2.0
    h = 1.0 - np.abs(x - center) / center
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    h = _project_sum(h, target)
    return h

def _upsample(h_old, n_new):
    n_old = len(h_old)
    # interpolate using linear interpolation
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new/2.0
    h_new = _project_sum(h_new, target)
    return h_new

def _greedy_reduce(h, max_iters, rng):
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_score = _evaluate_c5(best_h)
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)  # shift amount (can be negative)
        if lag == 0:
            # try to push h towards 0 or 1
            # pick i with middle value and j with extreme opposite
            middle = np.abs(h - 0.5).argmin()
            # choose j far from middle
            # if h[middle] > 0.5, we move mass to an index with smallest h (near 0)
            if h[middle] > 0.5:
                j = int(np.argmin(h))
            else:
                j = int(np.argmax(h))
            i = middle
        elif lag > 0:
            # positive shift
            k = lag
            max_range = n - k
            contrib = h[:max_range] * (1.0 - h[k:])
            i_rel = int(np.argmax(contrib))
            i = i_rel
            j = i + k
        else:
            # negative shift
            k = -lag
            max_range = n - k
            contrib = h[k:] * (1.0 - h[:max_range])
            i_rel = int(np.argmax(contrib))
            i = i_rel + k
            j = i_rel
        # move some mass from i to j
        if i == j:
            # fallback random
            i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 1e-12:
            # try opposite direction move
            max_move = min(h[j], 1.0 - h[i])
            if max_move <= 1e-12:
                # cannot move, break
                break
            # swap i,j
            i, j = j, i
        delta = max_move * 0.5  # move half of possible
        h[i] -= delta
        h[j] += delta
        # ensure bounds (should hold)
        cur_score = _evaluate_c5(h)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta
            # maybe try a random small move instead
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            h[i] -= delta
            h[j] += delta
            cur_score = _evaluate_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def _simulated_annealing(h, deadline, rng):
    n = len(h)
    target = n/2.0
    cur_h = h.copy()
    cur_score = _evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    while time.time() < deadline:
        # temperature schedule: linear in remaining time
        remaining = deadline - time.time()
        total = deadline - start
        frac = remaining / total
        T = 0.02 * frac + 1e-6
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 1e-12:
            continue
        delta = max_move * rng.random()
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = _evaluate_c5(cur_h)
        improve = new_score < cur_score - 1e-12
        if improve or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
        # occasional restart if stuck
        if rng.random() < 0.0005:
            # perturb best with small noise
            cur_h = best_h + rng.normal(scale=0.03, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_sum(cur_h, target)
            cur_score = _evaluate_c5(cur_h)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # Choose resolution
    n = 100  # even, allows half integral sum integer
    target = n / 2.0

    # Build initial candidates
    seeds = []

    # 1. Triangular shape
    seeds.append(_triangular_init(n))

    # 2. Constant 0.5
    seeds.append(np.full(n, 0.5))

    # 3. Random uniform
    rand_vec = rng.random(n)
    rand_vec = _project_sum(rand_vec, target)
    seeds.append(rand_vec)

    # 4. Upsample previous if exists
    init_vec = globals().get('initial_h_values')
    if init_vec is not None:
        old = np.asarray(init_vec, dtype=np.float64)
        up = _upsample(old, n)
        seeds.append(up)

    best_h = None
    best_score = float('inf')
    # Allocate time roughly equally among seeds for greedy phase
    time_per_seed = (budget_s - 5.0) / max(1, len(seeds))
    for s in seeds:
        s = np.clip(s, 0.0, 1.0)
        s = _project_sum(s, target)
        # Greedy reduction
        deadline = time.time() + time_per_seed * 0.4
        h_g, sc_g = _greedy_reduce(s, max_iters=2000, rng=rng)
        # Simulated annealing using remaining time slice
        deadline = time.time() + time_per_seed * 0.6
        h_a, sc_a = _simulated_annealing(h_g, deadline, rng)
        if sc_a < best_score:
            best_score = sc_a
            best_h = h_a
        # quick check if time exhausted
        if time.time() - start > budget_s - 1.0:
            break

    # Final polishing hill climb (deterministic)
    if best_h is not None:
        # simple hill climbing pass
        deadline = start + budget_s - 0.5
        # reuse same hill climb logic
        n = len(best_h)
        cur_h = best_h.copy()
        cur_score = _evaluate_c5(cur_h)
        while time.time() < deadline:
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 1e-12:
                continue
            delta = rng.random() * max_move
            cur_h[i] -= delta
            cur_h[j] += delta
            new_score = _evaluate_c5(cur_h)
            if new_score + 1e-12 < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = cur_h.copy()
            else:
                cur_h[i] += delta
                cur_h[j] -= delta
        best_h = _project_sum(np.clip(best_h, 0.0, 1.0), target)
        best_score = float(_evaluate_c5(best_h))

    return best_h, best_score, n

import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_sum(x, target):
    """Project x onto the hyperplane sum(x)=target while staying in [0,1]."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple water‑filling iteration
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        if diff > 0:
            add = np.minimum(per, 1.0 - x[idx])
            x[idx] += add
            diff -= add.sum()
        else:
            sub = np.minimum(-per, x[idx])
            x[idx] -= sub
            diff += sub.sum()
    return np.clip(x, 0.0, 1.0)


def _triangular_init(n):
    """Linear “tent” shape, then projected to the feasible set."""
    x = np.arange(n)
    c = (n - 1) / 2.0
    h = 1.0 - np.abs(x - c) / c
    h = np.clip(h, 0.0, 1.0)
    return _project_sum(h, n / 2.0)


def _upsample(old_h, new_n):
    """Linear interpolation of a coarser vector to a finer grid."""
    old_n = len(old_h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h = np.interp(x_new, x_old, old_h)
    h = np.clip(h, 0.0, 1.0)
    return _project_sum(h, new_n / 2.0)


def _greedy_reduce(h, max_iters, rng):
    """Targeted hill‑climbing that always reduces the current worst shift."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = _evaluate_c5(best_h)

    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)      # shift amount (negative => left)

        if lag == 0:
            # push a middle value toward an extreme
            i = int(np.abs(h - 0.5).argmin())
            if h[i] > 0.5:
                j = int(np.argmin(h))
            else:
                j = int(np.argmax(h))
        elif lag > 0:
            k = lag
            max_range = n - k
            contrib = h[:max_range] * (1.0 - h[k:])
            i_rel = int(np.argmax(contrib))
            i = i_rel
            j = i + k
        else:                     # lag < 0
            k = -lag
            max_range = n - k
            contrib = h[k:] * (1.0 - h[:max_range])
            i_rel = int(np.argmax(contrib))
            i = i_rel + k
            j = i_rel

        if i == j:
            i, j = rng.choice(n, size=2, replace=False)

        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 1e-12:
            # try the opposite direction
            max_move = min(h[j], 1.0 - h[i])
            if max_move <= 1e-12:
                continue
            i, j = j, i

        delta = max_move * 0.5
        h[i] -= delta
        h[j] += delta

        cur_score = _evaluate_c5(h)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_h = h.copy()
        else:
            # revert; try a tiny random move instead
            h[i] += delta
            h[j] -= delta
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            h[i] -= delta
            h[j] += delta
            cur_score = _evaluate_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def _simulated_annealing(h, deadline, rng):
    """Classic annealing that respects the sum and box constraints."""
    n = h.shape[0]
    target = n / 2.0
    cur_h = h.copy()
    cur_score = _evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()

    while time.time() < deadline:
        # linear temperature schedule
        remaining = deadline - time.time()
        total = deadline - start
        frac = remaining / total
        T = 0.02 * frac + 1e-6

        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        cur_h[i] -= delta
        cur_h[j] += delta

        new_score = _evaluate_c5(cur_h)
        improve = new_score < cur_score - 1e-12
        if improve or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

        # occasional stochastic restart
        if rng.random() < 0.0005:
            cur_h = best_h + rng.normal(scale=0.03, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_sum(cur_h, target)
            cur_score = _evaluate_c5(cur_h)

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):

    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).

    start = time.time()
    rng = np.random.default_rng(seed)

    # up‑sample a parent construction if it exists
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        try:

    best_h = None
    best_score = float("inf")

    for idx, s in enumerate(seeds):

        # a) quick deterministic greedy phase
        s_g, sc_g = _greedy_reduce(s, max_iters=3000, rng=rng)

        seeds_left = num_seeds - idx

        deadline = time.time() + time_left / seeds_left
        s_a, sc_a = _simulated_annealing(s_g, deadline, rng)

        if sc_a < best_score:
            best_score = sc_a
            best_h = s_a

        if time.time() - start > budget_s - 1.0:
            break

    # ------------------------------------------------------------------
    # 3. Final fine‑grained hill climbing (pure improvement moves)
    # ------------------------------------------------------------------
    if best_h is None:
        # fall back to a simple random feasible vector
        best_h = _project_sum(rng.random(n), target)
        best_score = _evaluate_c5(best_h)

    deadline = start + budget_s - 0.3
    cur_h = best_h.copy()
    cur_score = best_score
    while time.time() < deadline:

        if max_move <= 1e-12:

        cur_h[i] -= delta
        cur_h[j] += delta

    best_score = float(_evaluate_c5(best_h))
