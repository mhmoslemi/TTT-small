def make_trapezoid(n, a): # a in [0,0.5]
    dx = 2.0 / n
    # compute coordinates of shape with continuous formula.
    # generate positions at midpoints:
    pts = (np.arange(n) + 0.5) * dx

    # compute left_boundary = 0.5 - a/2
    left_boundary = 0.5 - a / 2.0
    ramp_end = left_boundary + a
    plateau_end = left_boundary + 1.0  # left_boundary + a + b = left_boundary + a + (1 - a) = left_boundary +1

    # piecewise:
    h = np.zeros_like(pts)
    # ramp up region
    idx = (pts >= left_boundary) & (pts < ramp_end)
    h[idx] = (pts[idx] - left_boundary) / a  # a may be 0 => handle division
    # plateau region
    idx = (pts >= ramp_end) & (pts < plateau_end)
    h[idx] = 1.0
    # ramp down region
    idx = (pts >= plateau_end) & (pts <= plateau_end + a)
    # right_boundary = plateau_end + a = 1.5 + a/2
    # linear descending:
    h[idx] = (plateau_end + a - pts[idx]) / a
    # ensure bounds
    h = np.clip(h, 0.0, 1.0)

    # If a==0, the ramps vanish and shape becomes rectangular plateau width=1? In that case we compute accordingly. Handle division by zero: If a==0, skip ramp parts; only plateau width=1 spanning from left_boundary=0.5 to plateau_end=1.5. So h[i]=1 for pts in [0.5,1.5]; else 0.

    return project_onto_bounds_and_sum(h, n/2)

import numpy as np, random, time, math
from scipy.optimize import minimize

TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x, target):
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low+high)/2
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low+high)/2
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(arr, alpha=20.0):
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def objective_smooth(h, dx, alpha=20.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha) * dx

def sum_eq_constraint(x):
    return np.sum(x) - TARGET_SUM_GLOBAL

def upsample_initial(init, n):
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n/2.0)

def make_trapezoid(n, a):
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    left_boundary = 0.5 - a / 2.0
    if a < 1e-12:

import time, random, numpy as np
from scipy.optimize import minimize

TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n intersect sum(x)=target by bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth objective used by SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of the provided initial vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_trapezoid(n: int, a: float) -> np.ndarray:
    """
    Symmetric trapezoid with plateau height 1.
    a = length of each sloping side (0 ≤ a ≤ 0.5).
    """
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    left_boundary = 0.5 - a / 2.0
    if a < 1e-12:
        # pure rectangle of height 1, width 1 centered at 1
        h = np.where((pts >= left_boundary) & (pts < left_boundary + 1.0), 1.0, 0.0)
        return project_onto_bounds_and_sum(h, n / 2.0)

    ramp_end = left_boundary + a
    plateau_end = left_boundary + 1.0
    right_end = plateau_end + a

    h = np.zeros_like(pts)
    idx = (pts >= left_boundary) & (pts < ramp_end)
    h[idx] = (pts[idx] - left_boundary) / a
    idx = (pts >= ramp_end) & (pts < plateau_end)
    h[idx] = 1.0
    idx = (pts >= plateau_end) & (pts <= right_end)
    h[idx] = (right_end - pts[idx]) / a
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_triangle(n: int) -> np.ndarray:
    """Special case a = 0.5 (triangular shape)."""
    return make_trapezoid(n, 0.5)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin²(πx/2) on [0,2] (already integral 1)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_plateau(n: int, c: float, a: float) -> np.ndarray:
    """
    Symmetric shape with plateau height `c` (0<c≤1) and side length `a`.
    The width of the flat region is w = 1/c - a (guaranteed ≥0).
    """
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    if c <= 0:
        return np.zeros(n)

    w = 1.0 / c - a
    left_boundary = 1.0 - (w / 2.0 + a)

    h = np.zeros_like(pts)

    if a > 1e-12:
        # rising ramp
        idx = (pts >= left_boundary) & (pts < left_boundary + a)
        h[idx] = c * (pts[idx] - left_boundary) / a
        # falling ramp
        idx = (pts >= left_boundary + a + w) & (pts <= left_boundary + a + w + a)
        h[idx] = c * (left_boundary + a + w + a - pts[idx]) / a

    if w > 1e-12:
        idx = (pts >= left_boundary + a) & (pts < left_boundary + a + w)
        h[idx] = c

    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    n = 500                               # moderate size for speed & accuracy
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Gather a few deterministic seeds
    seeds = []

    if "initial_h_values" in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_trapezoid(n, 0.0))   # rectangle of height 1

    # ------------------------------------------------------------------
    # Scan the 2‑parameter family (c, a)
    c_grid = np.linspace(0.5, 1.0, 11)      # height of the plateau
    for c in c_grid:
        # maximal side length allowed by the support constraint
        a_max = 2.0 - 1.0 / c if c < 1.0 else 1.0
        if a_max < 0:
            continue
        a_vals = np.linspace(0.0, a_max, 31)   # coarse sweep for each c
        for a in a_vals:
            h = make_plateau(n, c, a)
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
            seeds.append(h)

    # ------------------------------------------------------------------
    # Evaluate all gathered seeds (some may be duplicates)
    for h in seeds:
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Refine the current best with a few SLSQP runs (smooth max approx.)
    def slsqp_refine(start_h, maxiter=400, alpha=200.0):
        try:
            res = minimize(
                objective_smooth,
                start_h,
                args=(dx, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": False},
            )
        except Exception:
            return None
        h_opt = project_onto_bounds_and_sum(res.x, target_sum)
        return h_opt

    for _ in range(5):
        if time.time() - start > budget_s - 5.0:
            break
        pert = best_h + np.random.uniform(-0.02, 0.02, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = project_onto_bounds_and_sum(pert, target_sum)
        refined = slsqp_refine(pert)
        if refined is None:
            continue
        val = compute_c5(refined)
        if val < best_val - 1e-9:
            best_val = val
            best_h = refined.copy()
            if best_val <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Deterministic pairwise improvement directed by the worst shift.
    no_improve = 0
    while time.time() - start < budget_s - 1.0 and no_improve < 2000:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)                     # integer shift (in samples)

        if shift >= 0:
            g = np.concatenate((np.zeros(shift), best_h[: n - shift]))
        else:
            g = np.concatenate((best_h[-shift:], np.zeros(-shift)))

        inc_idx = np.where(best_h < 1.0 - 1e-12)[0]
        dec_idx = np.where(best_h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break

        i = inc_idx[np.argmax(g[inc_idx])]
        j = dec_idx[np.argmin(g[dec_idx])]

        if i == j:
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue

        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        if delta_max <= 1e-14:
            # fallback random move
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            hi, hj = best_h[i], best_h[j]
            delta_max = min(1.0 - hi, hj)
            if delta_max <= 1e-14:
                no_improve += 1
                continue
        delta = 0.5 * delta_max
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cval = compute_c5(cand)
        if cval < best_val - 1e-12:
            best_val = cval
            best_h = cand
            no_improve = 0
            if best_val <= 0.3808:
                break
        else:
            # small random swap as secondary attempt
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            hi, hj = best_h[i], best_h[j]
            delta_max = min(1.0 - hi, hj)
            delta_min = -min(hi, 1.0 - hj)
            if delta_max <= delta_min:
                no_improve += 1
                continue
            delta = np.random.uniform(delta_min, delta_max) * 0.5
            cand = best_h.copy()
            cand[i] += delta
            cand[j] -= delta
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_bounds_and_sum(cand, target_sum)
            cval = compute_c5(cand)
            if cval < best_val - 1e-12:
                best_val = cval
                best_h = cand
                no_improve = 0
                if best_val <= 0.3808:
                    break
            else:
                no_improve += 1

    # ------------------------------------------------------------------
    # Final safety projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
