import time
import math
import itertools

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 80) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1] with prescribed sum."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Evaluation of the true C5 bound (max overlap)
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Cross‑correlation via FFT (p ⋆ q)[i] = sum_s p_s * q_{i+s}
# ----------------------------------------------------------------------
def _cross_corr_fft(p: np.ndarray, q: np.ndarray) -> np.ndarray:
    """Return p ⋆ q evaluated on indices 0..len(q)-1."""
    n = len(q)
    m = len(p)                     # = 2*n-1
    L = 1 << ((m + n - 1).bit_length())    # next power of two
    # convolution of reversed p with q gives the desired cross‑correlation
    p_rev = p[::-1]
    conv = np.fft.irfft(np.fft.rfft(p_rev, L) * np.fft.rfft(q, L), L)
    offset = m - 1
    return conv[offset: offset + n]


# ----------------------------------------------------------------------
# Gradient of the smooth maximum objective
# ----------------------------------------------------------------------
def _gradient_smooth_max(h: np.ndarray, alpha: float, dx: float) -> np.ndarray:
    """∇_h  (1/α) log Σ exp(α·corr) , where corr = ∫ h·(1‑h(·+k))."""
    n = len(h)
    # raw correlation (without dx)
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx

    # soft‑max weights
    max_c = corr.max()
    exp_alpha = np.exp(alpha * (corr - max_c))
    p = exp_alpha / exp_alpha.sum()          # shape (2n‑1,)

    # cross‑correlations needed for gradient
    g_one = _cross_corr_fft(p, np.ones(n, dtype=np.float64))
    g_h   = _cross_corr_fft(p, h)

    # gradient (dx factor because corr = dx·corr_raw)
    grad = dx * (g_one - 2.0 * g_h)
    return grad


# ----------------------------------------------------------------------
# Adam optimiser with projection onto the feasible set
# ----------------------------------------------------------------------
def _adam_optimize(h0: np.ndarray, target_sum: float, dx: float,
                   phases, rng: np.random.RandomState):
    """Run Adam across several phases (different α and learning‑rates)."""
    h = h0.copy()
    n = len(h)
    m = np.zeros_like(h)          # first moment
    v = np.zeros_like(h)          # second moment
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for alpha, lr, iters in phases:
        for _ in range(iters):
            t += 1
            g = _gradient_smooth_max(h, alpha, dx)
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = _project_onto_capped_simplex(h, target_sum)
    return h

    Return a step function discretisation h on [0,2] together with an

    start = time.time()

    # ------------------------------------------------------------------

                return m + np.log(np.sum(np.exp(a * shifted))) / a

        best_val = _compute_c5(best_h, dx)
