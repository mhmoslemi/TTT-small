        if k >= 0:

            # grad[i] includes h[i+k] for i from 0..n-1-k
            # grad[i+k] includes h[i] for same i
            # Equivalent: grad[:n-k] += h[k:] ; grad[k:] += h[:n - k]
            grad[:n - k] += h[k:]
            grad[k:] += h[:n - k]
        else:
            d = -k
            # shift left
            # grad[i] includes h[i - d] for i from d..n-1 => grad[d:] += h[:n - d]
            # grad[i - d] includes h[i] for i from d..n-1 => grad[:n - d] += h[d:]
            grad[d:] += h[:n - d]
            grad[:n - d] += h[d:]
    return grad

Reason check: For negative k = -d, the shift is left. The sum_i h_i h_{i+k} = sum_i h_i h_{i-d} (i>=d). That term's gradient w.r.t h_i appears from both positions: derivative w.r.t h_i is h_{i-d} (if i>=d) + h_{i+d} (if i+d < n). So same as above. So gradient code is correct.

def project_to_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # water-filling
    for _ in range(1000):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust arbitrary index
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

# ---------- helper functions ----------

def project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-10) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h) = n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def c5_from_h(h: np.ndarray) -> float:
    """Compute the C5 value for a feasible step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def random_feasible(n: int) -> np.ndarray:
    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    h = np.random.rand(n)
    return project_to_feasible(h, n)


def gradient_for_shifts(h: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Gradient of sum_i h_i h_{i+k} for each shift k in idxs (full correlation indices)."""
    n = h.size
    offset = n - 1
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - offset
        if k >= 0:
            # shift right by k
            grad[: n - k] += h[k:]
            grad[k:] += h[: n - k]
        else:
            d = -k
            grad[d:] += h[: n - d]
            grad[: n - d] += h[d:]
    return grad


def gradient_ascent(
    h0: np.ndarray, max_iter: int, end_time: float, target: float = 0.3808
) -> tuple:
    """Projected gradient ascent on min autocorrelation (equivalently minimise C5)."""
    h = project_to_feasible(h0, h0.size)
    best_h = h.copy()
    best_c5 = c5_from_h(best_h)
    n = h.size

    for _ in range(max_iter):
        if time.time() > end_time:
            break

        # autocorrelation of h with itself
        ac = np.correlate(h, h, mode="full")
        min_val = np.min(ac)
        eps = 1e-12
        min_idxs = np.where(np.abs(ac - min_val) <= eps)[0]

        grad = gradient_for_shifts(h, min_idxs)
        norm = np.abs(grad).sum()
        if norm == 0:
            break

        improved = False
        # try decreasing step sizes (L1‑scaled)
        for mul in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            step = mul / norm
            cand = h + step * grad
            cand = project_to_feasible(cand, n)
            c5cand = c5_from_h(cand)
            if c5cand + 1e-12 < best_c5:
                best_c5 = c5cand
                best_h = cand.copy()
                h = cand
                improved = True
                break

        if not improved:
            # random jitter to escape plateaus
            sigma = 0.05
            pert = sigma * (np.random.rand(n) - 0.5) * 2.0
            cand = h + pert
            cand = project_to_feasible(cand, n)
            c5cand = c5_from_h(cand)
            if c5cand + 1e-12 < best_c5:
                best_c5 = c5cand
                best_h = cand.copy()
                h = cand

        if best_c5 < target:
            break

    return best_h, best_c5
