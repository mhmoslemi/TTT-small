def compute_worst_shift_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    # compute autocorrelation as correlation of h with h
    autoc = np.correlate(h, h, mode='full') * dx
    idx = autoc.argmin()  # minimal
    shift = idx - (n - 1)  # integer shift
    # compute gradient contributions
    # create vector g of zeros
    g = np.zeros_like(h)
    k = shift
    if k >= 0:
        # for i from 0 to n-1-k: both i and i+k within bounds
        g[:n-k] += h[k:]   # h[i+k] term
        g[k:] += h[:n-k]   # h[i-k] term (i>=k)
    else:
        # k negative: shift left
        kk = -k
        # g[kk:] for i+(-kk) = i-kk where i-kk >=0
        g[kk:] += h[:n-kk]
        # g[:n-kk] for i-kk in range
        g[:n-kk] += h[kk:]
    return shift, g, autoc.min()

def gradient_worst_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    k = shift
    if k >= 0:
        # i from 0 to n-1-k: both i, i+k in bounds
        g[:n - k] += h[k:]   # h_{i+k}
        g[k:] += h[:n - k]   # h_{i-k}
    else:
        kk = -k
        g[kk:] += h[:n - kk]   # i >= kk, i-k = i-kk
        g[:n - kk] += h[kk:]   # i <= n-kk-1, i+k = i+kk
    return g

    rng = np.random.default_rng(seed)
    start = time.time()

        # choose upscale factor

    # Ensure n even

    dx = 2.0 / n

        h = np.interp(x_new, x_old, np.asarray(init_vec))
        h = np.clip(h, 0.0, 1.0)
        h = _adjust_sum(h, target_sum)

        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx

    best_c5 = _c5_of(best_h)

import time
import numpy as np

def _adjust_sum(x, target):
    """Project vector x onto the hyperplane sum(x)=target while respecting 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    # simple redistribution loop
    while abs(diff) > 1e-14:
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0.0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return np.clip(x, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimizer for the Erdős minimum overlap problem (C5).
    Returns a tuple (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose a discretisation size.
    # Prefer a finer grid than the parent (45 points) but stay well below 1k.
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        n0 = len(init_vec)
        # upscale a bit – 4× the original size gives a good trade‑off.
        n = max(120, n0 * 4)
    else:
        n = 180
    if n % 2:
        n += 1                     # even size simplifies shift handling

    dx = 2.0 / n
    target_sum = n / 2.0           # because integral must be 1

    # ------------------------------------------------------------------
    # Build an initial feasible h.
    # ------------------------------------------------------------------
    if init_vec is not None:
        # linear interpolation of the parent solution onto the new grid
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False) + 1.0 / n0
        x_new = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        h = np.interp(x_new, x_old, np.asarray(init_vec, dtype=np.float64))
        h = _adjust_sum(np.clip(h, 0.0, 1.0), target_sum)
    else:
        h = rng.random(n)
        h = _adjust_sum(np.clip(h, 0.0, 1.0), target_sum)

    # ------------------------------------------------------------------
    # Helper to evaluate C5 for a vector.
    # ------------------------------------------------------------------
    def _c5_of(vec):
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    # ------------------------------------------------------------------
    # Main guided hill‑climbing loop.
    # ------------------------------------------------------------------
    no_improve = 0
    NO_IMPROVE_LIMIT = 2500       # after this many failures we randomise a bit

    while time.time() - start < budget_s - 0.5:
        # correlation and worst (maximal) shift
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = int(np.argmax(corr_full))
        cur_c5 = corr_full[max_idx] * dx
        # safety – recompute best_c5 if something got out of sync
        if cur_c5 > best_c5 + 1e-12:
            best_c5 = cur_c5

        shift = max_idx - (n - 1)          # signed integer shift producing the max

        # gradient of the objective for this shift:
        #   d/dh_i (h_i*(1-h_{i+shift})) = 1 - h_{i+shift}   (if i overlaps)
        grad = np.zeros_like(best_h)
        if shift >= 0:
            grad[: n - shift] = 1.0 - best_h[shift:]
        else:
            k = -shift
            grad[k:] = 1.0 - best_h[: n - k]

        # pick a source (large gradient, can decrease) and a destination (small gradient, can increase)
        src_candidates = np.where((grad > 0.0) & (best_h > 1e-12))[0]
        dst_candidates = np.where((grad < 1.0) & (best_h < 1.0 - 1e-12))[0]

        moved = False
        if src_candidates.size and dst_candidates.size:
            src = src_candidates[np.argmax(grad[src_candidates])]
            dst = dst_candidates[np.argmin(grad[dst_candidates])]
            max_move = min(best_h[src], 1.0 - best_h[dst])
            if max_move > 1e-12:
                # random fraction of the allowable move (up to half of it)
                delta = max_move * rng.random() * 0.5
                new_h = best_h.copy()
                new_h[src] -= delta
                new_h[dst] += delta
                new_c5 = _c5_of(new_h)
                if new_c5 + 1e-12 < best_c5:
                    best_h, best_c5 = new_h, new_c5
                    no_improve = 0
                    moved = True
        if not moved:
            # either no useful pair or move didn’t improve – try a tiny random shake
            shake = rng.normal(scale=0.02, size=n)
            cand = best_h + shake
            cand = np.clip(cand, 0.0, 1.0)
            cand = _adjust_sum(cand, target_sum)
            cand_c5 = _c5_of(cand)
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand, cand_c5
                no_improve = 0
                continue
            no_improve += 1

        # Periodic diversification when stuck
        if no_improve >= NO_IMPROVE_LIMIT:
            no_improve = 0
            # perturb the current best with a larger noise burst
            pert = best_h + rng.normal(scale=0.15, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _adjust_sum(pert, target_sum)
            pert_c5 = _c5_of(pert)
            if pert_c5 < best_c5:
                best_h, best_c5 = pert, pert_c5

    # ------------------------------------------------------------------
    # Final sanity checks and return.
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = float(_c5_of(best_h))
    return best_h, best_c5, n
