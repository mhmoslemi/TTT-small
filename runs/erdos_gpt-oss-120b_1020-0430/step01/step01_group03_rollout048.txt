def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)
```

def anneal_binary(n, rng, time_limit, init_vec=None):
    # start with init_vec if provided else random binary with sum n/2
    if init_vec is not None:
        h = init_vec.copy()
    else:
        h = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    current_h = h
    current_c = best_c
    # Simulated annealing schedule
    start = time.time()
    # initial temperature maybe based on best_c variance
    temp = 0.1 * best_c  # small; we will adapt
    # choose iteration count high
    it = 0
    while time.time() - start < time_limit:
        # Propose a swap
        # pick indices i where h[i]=1, j where h[j]=0
        one_idx = rng.choice(np.where(current_h == 1)[0])
        zero_idx = rng.choice(np.where(current_h == 0)[0])
        # swap
        new_h = current_h.copy()
        new_h[one_idx] = 0
        new_h[zero_idx] = 1
        new_c = compute_c5(new_h)
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            # accept
            current_h = new_h
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # schedule temperature decay
        it += 1
        # linear or exponential decay
        if it % 1000 == 0:
            # reduce temp gradually
            temp *= 0.99
            # avoid temp too low
            if temp < 1e-6:
                temp = 1e-6
    return best_h, best_c
```

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun":sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h≤1, Σh=n/2)."""
    h = np.zeros(n, dtype=np.float64)
    # start from a binary pattern with exactly n/2 ones
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # random pairwise transfers to make it fractional
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    np.clip(h, 0.0, 1.0, out=h)
    # exact sum correction
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # shrink step periodically
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction (numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def upsample(h, factor):
    """Simple nearest‑neighbour up‑sampling (repeat each entry)."""
    return np.repeat(h, factor)
