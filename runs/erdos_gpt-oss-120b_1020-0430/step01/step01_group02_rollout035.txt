def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        # compute slack
        slack = 1.0 - x
        total_slack = slack.sum()
        if total_slack <= 0:
            return x  # can't add
        # add proportionally
        add = diff * (slack / total_slack)
        x = x + add
        # after addition some may exceed 1 due to rounding, clip and repeat if needed
        x = np.clip(x, 0.0, 1.0)
        # recompute diff recursively if necessary (if diff not fully allocated)
        new_diff = target - x.sum()
        if abs(new_diff) > 1e-12 and abs(new_diff) > 1e-9:
            # recursion (avoid infinite)
            return _adjust_sum(x, target)
        else:
            return x
    else:
        # diff negative: need to remove mass
        diff_abs = -diff
        available = x
        total_avail = available.sum()
        if total_avail <= 0:
            return x
        sub = diff_abs * (available / total_avail)
        x = x - sub
        x = np.clip(x, 0.0, 1.0)
        new_diff = target - x.sum()
        if abs(new_diff) > 1e-12 and abs(new_diff) > 1e-9:
            return _adjust_sum(x, target)
        else:
            return x

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # bounds zero tolerance.
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - x
            # indices where slack > 0
            mask = slack > 1e-12
            if not np.any(mask):
                break
            total_slack = slack[mask].sum()
            if total_slack <= 0:
                break
            add = diff * (slack / total_slack)
            # Ensure not exceeding bound
            add = np.where(mask, np.minimum(add, slack), 0.0)
            x += add

    return x

import time
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a given step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:                     # need to add mass
            slack = 1.0 - x
            mask = slack > 1e-12
            if not np.any(mask):
                break
            total = slack[mask].sum()
            add = diff * (slack / total)
            add = np.where(mask, np.minimum(add, slack), 0.0)
            x += add
        else:                               # need to remove mass
            avail = x
            mask = avail > 1e-12
            if not np.any(mask):
                break
            total = avail[mask].sum()
            sub = (-diff) * (avail / total)
            sub = np.where(mask, np.minimum(sub, avail), 0.0)
            x -= sub
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)


def _gradient_for_shift(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the maximal correlation term for a given lag."""
    n = h.size
    if lag >= 0:
        plus = np.concatenate((h[lag:], np.zeros(lag, dtype=h.dtype)))
        minus = np.concatenate((np.zeros(lag, dtype=h.dtype), h[: n - lag]))
    else:
        s = -lag
        plus = np.concatenate((np.zeros(s, dtype=h.dtype), h[: n - s]))
        minus = np.concatenate((h[s:], np.zeros(s, dtype=h.dtype)))
    return 1.0 - plus - minus


def _subgradient_descent(
    h0: np.ndarray,
    target: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    cur_c5 = _evaluate_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    step = 0.2
    it = 0
    while time.time() < deadline:
        # compute the lag achieving the maximal correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (h.size - 1)

        # sub‑gradient for this lag
        g = _gradient_for_shift(h, lag)

        # gradient step
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _adjust_sum(h_new, target)

        new_c5 = _evaluate_c5(h_new)

        # simple acceptance rule
        if new_c5 < cur_c5 - 1e-12:
            h, cur_c5 = h_new, new_c5
            step = min(step * 1.02, 0.5)          # slight step increase on improvement
        else:
            step *= 0.9995                       # decay when no progress

        # keep global best
        if cur_c5 < best_c5 - 1e-12:
            best_h, best_c5 = h.copy(), cur_c5

        # occasional jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.uniform(-0.01, 0.01, size=h.size)
            h = np.clip(h + jitter, 0.0, 1.0)
            h = _adjust_sum(h, target)
            cur_c5 = _evaluate_c5(h)
        it += 1
        if step < 1e-6:
            step = 1e-6
    return best_h, best_c5


def _hill_climb(
    h: np.ndarray,
    c5: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass‑exchange hill climbing."""
    n = h.size
    while time.time() < deadline:

        if rng.random() < 0.5:

    # --------------------------------------------------------------
    best_h = _adjust_sum(np.clip(h0, 0.0, 1.0), target)
    best_c5 = _evaluate_c5(best_h)

    # --------------------------------------------------------------
    # Main optimisation: projected sub‑gradient descent + final hill‑climb
    # --------------------------------------------------------------
    # allocate 80% of the time budget to the descent phase
    descent_deadline = start_time + budget_s * 0.8
    best_h, best_c5 = _subgradient_descent(best_h, target, descent_deadline, rng)

    # final fine‑grained random hill climbing with the remaining time
    final_deadline = start_time + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, final_deadline, rng)

    # final projection (just in case)
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))

    return best_h, best_c5, n
