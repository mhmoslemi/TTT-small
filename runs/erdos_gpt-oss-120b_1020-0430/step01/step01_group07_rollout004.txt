def adjust_sum(vec, target):
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # Use greedy fill
    # Choose random order
    indices = np.random.permutation(vec.size)
    for idx in indices:
        if diff > 0:
            cap = 1.0 - vec[idx]
            if cap <= 0: continue
            inc = min(cap, diff)
            vec[idx] += inc
            diff -= inc
        else:
            cap = vec[idx]  # amount can be removed
            if cap <= 0: continue
            dec = min(cap, -diff)
            vec[idx] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    # If still diff not zero due to insufficient capacity, fallback to scaling
    if abs(diff) > 1e-9:
        # Scale all entries (should not happen) but ensure sum=target
        vec = vec * (target / vec.sum())
        # Clip again
        np.clip(vec, 0.0, 1.0, out=vec)
        # If sum off due to clipping, adjust iteratively maybe call recursively
        cur = vec.sum()
        if abs(cur - target) > 1e-9:
            vec[-1] += target - cur  # small adjustment
    return vec

def adjust_sum(vec, target):
    # Ensure sum equals target while keeping in [0,1].
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # greedy adjust
    indices = np.random.permutation(vec.size)
    for idx in indices:
        if diff > 0:
            cap = 1.0 - vec[idx]
            if cap <= 0:
                continue
            inc = cap if cap < diff else diff
            vec[idx] += inc
            diff -= inc
        else:
            cap = vec[idx]  # amount we can subtract
            if cap <= 0:
                continue
            dec = cap if cap < -diff else -diff
            vec[idx] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    # If still not exact adjust by small epsilon on first element
    if abs(diff) > 1e-12:
        vec[0] += diff
        # clip again ensuring within bounds
        np.clip(vec, 0.0, 1.0, out=vec)
        # final scaling to meet target (very small)
        cur = vec.sum()
        if abs(cur - target) > 1e-9:
            vec *= target / cur
            np.clip(vec, 0.0, 1.0, out=vec)
    return vec

                pop[i] = trial
                c_vals[i] = trial_c
                if trial_c < best_c:
                    best_c = trial_c
                    best_h = trial.copy()

        # occasional random refresh of the worst individual
        if random.random() < 0.05:
            worst = int(np.argmax(c_vals))
            v = np.random.rand(n)
            v *= target_sum / v.sum()
            np.clip(v, 0.0, 1.0, out=v)
            v = _adjust_sum(v, target_sum)
            pop[worst] = v
            c_vals[worst] = c5_metric(v)
            if c_vals[worst] < best_c:
                best_c = c_vals[worst]
                best_h = v.copy()

    # --------------------------------------------------------------
    # 4.  gradient‑guided hill‑climbing (remaining time)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        hill_start = time.time()
        max_step_initial = 0.2
        min_step = 0.001
        while time.time() - hill_start < remaining:
            # compute gradient for the current worst shift
            g, shift, idx_max = _worst_shift_gradient(best_h)

            pos = np.where(g > 1e-8)[0]
            neg = np.where(g < -1e-8)[0]
            if pos.size == 0 or neg.size == 0:
                break

            i = int(np.random.choice(pos))
            j = int(np.random.choice(neg))

            hi, hj = best_h[i], best_h[j]
            delta_max = min(hi, 1.0 - hj)          # keep within [0,1] and preserve sum
            if delta_max <= 0.0:
                continue

            # step size decays linearly with elapsed hill‑climbing time
            elapsed = time.time() - hill_start
            frac = elapsed / remaining
            step = max_step_initial * (1.0 - frac) + min_step * frac
            delta = random.uniform(0.0, min(delta_max, step))
            if delta <= 0.0:
                continue

            best_h[i] = hi - delta
            best_h[j] = hj + delta

            new_c = c5_metric(best_h)
            if new_c + 1e-12 < best_c:
                best_c = new_c
            else:
                # revert
                best_h[i] = hi
                best_h[j] = hj

    # --------------------------------------------------------------
    # 5.  final sanity check (exact sum)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = _adjust_sum(best_h, target_sum)

    # --------------------------------------------------------------
    # 6.  final evaluation (use the harness helper if present)
    try:
        final_c = evaluate_erdos_solution(best_h)   # type: ignore
    except Exception:
        final_c = c5_metric(best_h)

    return best_h, float(final_c), int(n)
