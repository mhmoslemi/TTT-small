def compute_grad_smooth(h, alpha):
    # n = len(h)
    # compute correlation c
    n = h.shape[0]
    # compute conv_rev = convolution of h with reverse h (auto-correlation)
    # approach: use FFT for c.
    # We'll compute zero-padded length m = 1 << (int(np.ceil(np.log2(2*n-1)))) maybe just use 2*n for convenience.
    m = 1
    while m < 2*n:
        m <<= 1
    # compute Fourier transforms
    H = np.fft.rfft(h, n=m)
    H_rev = np.fft.rfft(h[::-1], n=m)  # reverse
    # autocorrelation conv_rev = ifft(H * H_rev)
    conv_rev_full = np.fft.irfft(H * H_rev, n=m)[:2*n-1]   # length 2n-1
    # So c = target_sum - conv_rev_full  (since sum h = target)
    S = np.sum(h)
    c = S - conv_rev_full
    # compute weight w
    maxc = np.max(c)
    expa = np.exp(alpha * (c - maxc))
    w = expa / np.sum(expa)
    # compute w_rev for gradient.
    w_rev = w[::-1]

    # convolution of w_rev with h
    # compute via FFT
    m2 = 1
    while m2 < len(w_rev) + n:
        m2 <<= 1
    # compute FFT of w_rev and h
    W_rev = np.fft.rfft(w_rev, n=m2)
    H2 = np.fft.rfft(h, n=m2)
    conv_wr = np.fft.irfft(W_rev * H2, n=m2)
    # length len(w_rev)+n-1 = 3n-2
    start = len(w_rev) - 1  # = 2n-2
    term1 = conv_wr[start:start+n]
    # conv of w with h
    W = np.fft.rfft(w, n=m2)
    conv_w = np.fft.irfft(W * H2, n=m2)
    term2 = conv_w[start:start+n]

    grad = np.ones_like(h) - (term1 + term2)
    return grad, c

    for i in range(max_iter):
        # schedule alpha linearly increase
        alpha = alpha_start + (alpha_end - alpha_start) * i / max_iter
        grad, _ = compute_grad_smooth(h, alpha)
        # step size maybe adapt: step = step0 * (decay**i)
        step = step0 * (decay**i)
        h -= step * grad
        h = project_onto_capped_simplex(h, target)
    return h

def compute_grad_smooth(h, alpha):

    n = len(h)

    # compute autocorrelation to get c
    # Use FFT conv as earlier.
    m = 1
    while m < 2*n:
        m <<= 1
    H = np.fft.rfft(h, n=m)
    H_rev = np.fft.rfft(h[::-1], n=m)
    conv_rev_full = np.fft.irfft(H * H_rev, n=m)[:2*n-1]
    S = np.sum(h)
    c = S - conv_rev_full

    # weight
    maxc = np.max(c)
    expa = np.exp(alpha * (c - maxc))
    w = expa / np.sum(expa)

    # convolution for term1 and term2:

    # Precompute proper length for convolution of w and h (len w + n -1)
    L = 2*n-1
    conv_len = L + n - 1  # = 3n-2
    m2 = 1
    while m2 < conv_len:
        m2 <<= 1

    # do FFT for h and w
    H2 = np.fft.rfft(h, n=m2)
    W = np.fft.rfft(w, n=m2)
    W_rev = np.fft.rfft(w[::-1], n=m2)

    conv_w_h = np.fft.irfft(W * H2, n=m2)[:conv_len]
    conv_wr_h = np.fft.irfft(W_rev * H2, n=m2)[:conv_len]

    start = L - 1  # index offset for getting length n out
    term1 = conv_wr_h[start:start+n]   # shift +s
    term2 = conv_w_h[start:start+n]    # shift -s
    grad = np.ones_like(h) - (term1 + term2)
    return grad, c

def _local_search(h, target, dx, rng, max_moves=2000, time_deadline=None):
    best = h.copy()
    best_val = _compute_c5(best, dx)
    n = len(h)
    for i_move in range(max_moves):
        if time_deadline is not None and time.time() > time_deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n-1)
        if j >= i:
            j += 1
        # max delta limited by bounds
        max_delta = min(0.02, 1 - best[i], best[j]) # move from j to i
        if max_delta <= 1e-12:
            continue
        delta = (rng.rand() - 0.5) * max_delta * 2  # within [-max_delta, max_delta]
        # ensure bounds
        new_i = best[i] + delta
        new_j = best[j] - delta
        if not (0 <= new_i <= 1 and 0 <= new_j <= 1):
            continue
        h2 = best.copy()
        h2[i] = new_i
        h2[j] = new_j
        val = _compute_c5(h2, dx)
        if val < best_val - 1e-9:
            best = h2

    return best

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9,
                                 max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# --------------------------------------------------------------
# Evaluation of the Erdős C5 functional (exact, via FFT)
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    n = h.shape[0]
    # linear convolution h * reverse(h)
    m = 1
    while m < 2 * n:
        m <<= 1
    H = np.fft.rfft(h, n=m)
    H_rev = np.fft.rfft(h[::-1], n=m)
    conv = np.fft.irfft(H * H_rev, n=m)[: 2 * n - 1]   # Σ_j h_j h_{j+shift}
    S = h.sum()
    # max_{k} ∫ h(x)(1-h(x+k))dx = dx * (S - min(conv))
    maxc = S - conv.min()
    return float(maxc * dx)


# --------------------------------------------------------------
# Gradient of a single shift (used for max‑correlation descent)
# --------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# --------------------------------------------------------------
# Smooth‑max gradient (softmax weighting) via FFT
# --------------------------------------------------------------
def _compute_grad_smooth(h: np.ndarray, alpha: float):
    n = h.shape[0]

    # ---- auto‑correlation (linear) ---------------------------------
    m = 1
    while m < 2 * n:
        m <<= 1
    H = np.fft.rfft(h, n=m)
    H_rev = np.fft.rfft(h[::-1], n=m)
    conv = np.fft.irfft(H * H_rev, n=m)[: 2 * n - 1]      # Σ_j h_j h_{j+shift}
    S = h.sum()
    c = S - conv                                         # c_i = Σ h (1‑h shifted)

    # ---- soft‑max weights -----------------------------------------
    maxc = np.max(c)
    expa = np.exp(alpha * (c - maxc))
    w = expa / np.sum(expa)                              # Σ w_i = 1

    # ---- convolutions needed for the gradient ---------------------
    L = 2 * n - 1
    conv_len = L + n - 1                                 # = 3n‑2
    m2 = 1
    while m2 < conv_len:
        m2 <<= 1

    H2 = np.fft.rfft(h, n=m2)
    W = np.fft.rfft(w, n=m2)
    W_rev = np.fft.rfft(w[::-1], n=m2)

    conv_w_h = np.fft.irfft(W * H2, n=m2)[:conv_len]
    conv_wr_h = np.fft.irfft(W_rev * H2, n=m2)[:conv_len]

    start = L - 1
    term1 = conv_wr_h[start:start + n]   # Σ w_i shift_{+i} h
    term2 = conv_w_h[start:start + n]    # Σ w_i shift_{‑i} h

    grad = np.ones_like(h) - (term1 + term2)
    return grad, c


def _smooth_descent_optimize(h0: np.ndarray,
                             target: float,
                             max_iter: int = 1200,
                             step0: float = 0.5,
                             decay: float = 0.9995,
                             alpha_start: float = 10.0,
                             alpha_end: float = 80.0) -> np.ndarray:
    h = h0.copy()
    step = step0
    for it in range(max_iter):
        alpha = alpha_start + (alpha_end - alpha_start) * it / max_iter
        grad, _ = _compute_grad_smooth(h, alpha)
        h -= step * grad
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# --------------------------------------------------------------
# Max‑correlation (nonsmooth) projected gradient descent
# --------------------------------------------------------------
def _gd_optimize_max(h0: np.ndarray,
                     target: float,
                     max_iter: int = 500,
                     step0: float = 0.5,
                     decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    for _ in range(max_iter):
        n = h.shape[0]
        # correlation via FFT (linear)
        m = 1
        while m < 2 * n:
            m <<= 1
        H = np.fft.rfft(h, n=m)
        H_rev = np.fft.rfft(h[::-1], n=m)
        conv = np.fft.irfft(H * H_rev, n=m)[: 2 * n - 1]
        S = h.sum()
        corr = S - conv
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

# Small local random search (keeps the exact sum)
# --------------------------------------------------------------
def _local_search(h: np.ndarray,
                  target: float,
                  dx: float,
                  rng: np.random.RandomState,
                  max_moves: int = 5000,
                  deadline: float = None) -> np.ndarray:
    best = h.copy()
    best_val = _compute_c5(best, dx)
    n = len(h)
    for _ in range(max_moves):
        if deadline is not None and time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        # maximal admissible delta keeping both entries inside [0,1]
        max_delta = min(0.02,
                        1.0 - best[i],
                        best[j])
        if max_delta <= 1e-12:
            continue
        delta = (rng.rand() - 0.5) * 2 * max_delta
        new_i = best[i] + delta
        new_j = best[j] - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        h2 = best.copy()
        h2[i] = new_i
        h2[j] = new_j
        val = _compute_c5(h2, dx)
        if val + 1e-12 < best_val:
            best = h2
            best_val = val
    return best
