def gradient_descent(h, budget, target_sum):
    # use start time
    start = time.time()
    n = h.shape[0]
    dx = 2.0 / n
    total_iters_est = int(budget // 0.01) # maybe approximate but dynamic.
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    # schedule parameters
    beta0 = 1.0
    beta_max = 200.0
    lr0 = 0.1
    lr_min = 0.001
    iter = 0
    while time.time() - start < budget:
        t = time.time() - start
        progress = t / budget
        beta = beta0 * ((beta_max / beta0) ** progress)  # exponential growth
        lr = lr0 * (1 - progress) + lr_min * progress   # linear interpolation
        # compute c and weights
        conv = np.correlate(h, h, mode='full') * dx
        c_vec = 1 - conv  # length 2n-1
        # compute softmax weights
        # using stable softmax
        c_max = np.max(c_vec)
        w = np.exp(beta * (c_vec - c_max))
        w_sum = w.sum()
        w /= w_sum
        # compute convolution w*h using FFT
        L = 2 * n - 1
        N = n + L - 1
        # compute conv_w_h quickly
        fft_w = np.fft.rfft(w, n=N)
        fft_h = np.fft.rfft(h, n=N)
        conv_wh = np.fft.irfft(fft_w * fft_h, n=N)
        # extract slice i from n-1 to n-1 + n
        conv_slice = conv_wh[(n-1):(n-1 + n)]
        # gradient
        g = dx * (1 - 2 * conv_slice)
        # update
        h_new = h - lr * g
        # project onto box+sum
        h_new = proj_bounds_sum(h_new, target_sum)
        # compute new objective (actual max)
        new_c = np.max(1 - np.correlate(h_new, h_new, mode='full') * dx)  # same as max(c_vec_new)
        if new_c < best_c:
            best_c = new_c
            best_h = h_new.copy()
        h = h_new
        iter += 1
    return best_h, best_c

def anneal(start_h, max_seconds):
    h = start_h.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    start = time.time()
    # schedule parameters
    T0 = 0.5
    T_end = 1e-6
    max_step = 0.5
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # propose delta
        # allow large steps with probability
        if random.random() < 0.7:
            delta = random.uniform(-max_step, max_step)
            delta = max(low, min(high, delta))
        else:
            delta = random.uniform(low, high)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    # final projection ensure sum exact:
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break
    return best_h, best_c

    dx = 2.0 / n

    # compute total time for schedule

         progress = (time.time() - start) / max_seconds
         # schedule
         beta = 1.0 * (200.0 ** progress)  # using exponent; starting 1, ends 200.
         lr = 0.2 * (1 - progress) + 0.001 * progress
         # compute conv and c
         conv = np.correlate(h, h, mode='full') * dx
         c_vec = 1 - conv
         # weights
         maxc = np.max(c_vec)
         w = np.exp(beta * (c_vec - maxc))
         w /= w.sum()
         # FFT conv of w and h
         L = 2*n - 1
         N = n + L - 1
         fft_w = np.fft.rfft(w, n=N)
         fft_h = np.fft.rfft(h, n=N)
         conv_wh = np.fft.irfft(fft_w * fft_h, n=N)
         conv_slice = conv_wh[(n-1):(n-1 + n)]

         # evaluate

         if c < best_c:

            best_h = h.copy()
    return best_h, best_c

import time
import random
import math
import numpy as np

def _c5_metric(h):
    """Discrete C5 metric used when the helper is unavailable."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, h, mode="full") * dx
    return np.max(1 - corr)

def _proj_bounds_sum(v, target):
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum=v_target.
    Uses a simple bisection on the Lagrange multiplier λ for the problem
        min ||x‑v||²  s.t. 0≤x≤1 , sum(x)=target.
    """
    lo, hi = -2.0, 2.0                # wide enough for all feasible λ
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual fix
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(len(x)):
            if 0.0 <= x[idx] + diff <= 1.0:
                x[idx] += diff
                break
    return x

def _initial_vector(n):
    """Create a feasible start vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0, left=h0[0], right=h0[-1])
    else:
        h = np.full(n, 0.5, dtype=np.float64)
    target = n / 2.0
    h = _proj_bounds_sum(h, target)
    return h

def _random_feasible(n):
    """Return a random feasible vector (uniform in the box, then projected)."""
    v = np.random.rand(n)
    target = n / 2.0
    return _proj_bounds_sum(v, target)

def _gradient_descent(start_h, max_seconds, target):
    """Smooth‑max gradient descent with a softmax surrogate."""
    n = start_h.shape[0]
    dx = 2.0 / n
    h = start_h.copy()
    best_h = h.copy()
    best_c = _c5_metric(h)

    start = time.time()
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds

        # schedules
        beta = 1.0 * (200.0 ** prog)               # grows from 1 to 200
        lr = 0.2 * (1 - prog) + 0.001 * prog       # decays from 0.2 to 0.001

        # correlation and surrogate weights
        conv = np.correlate(h, h, mode="full") * dx
        c_vec = 1.0 - conv                         # length 2n‑1
        c_max = np.max(c_vec)
        w = np.exp(beta * (c_vec - c_max))
        w /= w.sum()                               # softmax weights, sum to 1

        # gradient via convolution w * h (zero‑padded)
        L = 2 * n - 1
        N = n + L - 1
        fft_w = np.fft.rfft(w, n=N)
        fft_h = np.fft.rfft(h, n=N)
        conv_wh = np.fft.irfft(fft_w * fft_h, n=N)
        slice_conv = conv_wh[(n - 1):(n - 1 + n)]   # sum_s w_s * h[i‑s] == sum_s w_s * h[i+s]

        g = dx * (1.0 - 2.0 * slice_conv)          # gradient of the surrogate

        # update and project
        h = _proj_bounds_sum(h - lr * g, target)

        # evaluate true objective
        cur_c = _c5_metric(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def _anneal(start_h, max_seconds):
    """Simple simulated‑annealing refinement."""
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c

    start = time.time()
    T0 = 0.5
    T_end = 1e-6
    max_step = 0.5

    while time.time() - start < max_seconds:
        prog = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** prog

        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        if random.random() < 0.7:
            delta = random.uniform(-max_step, max_step)
            delta = max(low, min(high, delta))
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    # final exact‑sum correction
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C₅ bound and the number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    n = 400                                 # resolution
    target_sum = n / 2.0

    # start from the parent construction (interpolated) and keep it as a baseline
    best_h = _initial_vector(n)
    best_c = _c5_metric(best_h)

    # ---- gradient‑descent phase (≈60 % of the budget) ----
    grad_budget = total_budget * 0.6
    # use a few random starts within the allocated time
    slice_time = 70.0                       # seconds per gradient run
    while time.time() - start_all < grad_budget:
        remaining = grad_budget - (time.time() - start_all)
        if remaining < 20.0:
            break
        cur_slice = min(slice_time, remaining)
        start_vec = _random_feasible(n)
        cand_h, cand_c = _gradient_descent(start_vec, cur_slice, target_sum)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h

    # ---- annealing refinement (remaining time) ----
    remaining = total_budget - (time.time() - start_all)
    if remaining > 5.0:
        best_h, best_c = _anneal(best_h, remaining)

    # final exact‑sum safeguard (tiny rounding errors)
    best_h = _proj_bounds_sum(best_h, target_sum)

    # compute final bound using the supplied helper if available
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
