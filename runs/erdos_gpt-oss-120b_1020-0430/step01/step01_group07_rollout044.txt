def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    random.seed(seed)
    # set n = 500 (maybe adjust if initial_h_values length low)
    n0 = len(initial_h_values) if 'initial_h_values' in globals() else 0
    n = max(200, ((n0*5)//1))  # maybe 500 if initial_h_values present
    # Ensure n even
    if n % 2:
        n += 1
    # Interpolate given initial_h_values to n (like earlier code)
    if n0:
        x0 = np.linspace(0., 2., n0, endpoint=False)
        x = np.linspace(0., 2., n, endpoint=False)
        h = np.interp(x, x0, np.asarray(initial_h_values))
    else:
        h = np.full(n, 0.5)
    # enforce sum
    target = n / 2.
    cur = h.sum()
    if cur != 0:
        h = h * (target / cur)
    np.clip(h, 0., 1., out=h)
    # adjust small deficit
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        # add to random index that can accept
        for idx in np.random.permutation(n):
            if 0 <= h[idx] + deficit <= 1:
                h[idx] += deficit
                break

    best_h = h.copy()
    best_c = _c5_metric(best_h)  # we will define _c5_metric as earlier

    start = time.time()
    total_budget = float(budget_s)

    # parameters
    max_step = 0.2                     # base step size
    # schedule for step size decreasing (linear)
    while time.time() - start < total_budget:
        # compute correlation and shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c = corr[max_idx] * (2.0 / n)   # dx
        shift = max_idx - (n - 1)

        # compute gradient for this shift
        g = np.zeros_like(h)
        if shift >= 0:
            s = shift
            g[:n - s] += 1 - h[s:]
            g[s:] -= h[:n - s]
        else:
            s = -shift
            g[s:] += 1 - h[:n - s]
            g[:n - s] -= h[s:]

        # choose source and sink indices
        # source: positive gradient highest (decrease h)
        # sink: negative gradient lowest (increase h)
        # But we should ensure source h > 0, sink h < 1.
        # Use mask
        src_candidates = np.where((g > 1e-8) & (h > 1e-8))[0]
        sink_candidates = np.where((g < -1e-8) & (h < 1 - 1e-8))[0]
        if src_candidates.size > 0 and sink_candidates.size > 0:
            i = src_candidates[np.argmax(g[src_candidates])]
            j = sink_candidates[np.argmin(g[sink_candidates])]
            # amount to move
            delta_max = min(h[i], 1 - h[j], max_step)
            if delta_max > 0:
                # let delta be random portion up to delta_max
                delta = random.random() * delta_max
                h[i] -= delta
                h[j] += delta
                # compute new C
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_c = new_corr.max() * (2.0 / n)
                if new_c + 1e-12 < cur_c:
                    cur_c = new_c
                    if new_c < best_c:
                        best_c = new_c
                        best_h = h.copy()
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
        else:
            # fallback random pair move
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low < high:
                delta = random.uniform(-max_step, max_step)
                # Clip delta to feasible range
                delta = max(min(delta, high), low)
                if delta != 0:
                    h[i] = hi + delta
                    h[j] = hj - delta

        # occasionally random perturbation
        if random.random() < 0.01:
            # reinitialize some proportion
            perm = np.random.permutation(n)
            for idx in perm[:int(0.05 * n)]:
                h[idx] = random.random()
            # project to sum
            s = h.sum()
            if s != 0:
                h *= target / s
            np.clip(h, 0., 1., out=h)
            # adjust small deficit
            deficit = target - h.sum()
            if abs(deficit) > 1e-12:
                for idx in np.random.permutation(n):
                    if 0 <= h[idx] + deficit <= 1:
                        h[idx] += deficit
                        break

    # final best after budget
    final_c = best_c
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)

import time, math, random, numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """C5 value for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _make_initial(n: int) -> np.ndarray:
    """Create a feasible start vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst‑overlap term with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] += 1.0 - h[s:]
        g[s:] -= h[: n - s]
    else:
        s = -shift
        g[s:] += 1.0 - h[: n - s]
        g[: n - s] -= h[s:]
    return g
