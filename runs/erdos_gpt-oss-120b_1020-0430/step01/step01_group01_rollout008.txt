def _gradient_for_shift(h, shift):
    n = len(h)
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        # compute term1 (1 - h[i+s]) for i < n-s else 1
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        # term2 -h[i-s] for i >= s
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[:s] = 1.0
        g[: n - s] += -h[s:]   # careful with indexes
    return g
```

def _grad_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    exp_term = np.exp(alpha * (corr - max_corr))
    w = exp_term / exp_term.sum()
    # compute weighted gradient
    grad = np.zeros_like(h)
    for shift_idx, weight in enumerate(w):
        if weight == 0:
            continue
        shift = shift_idx - (n - 1)
        grad += weight * _gradient_for_shift(h, shift)
    return grad, max_corr  # return value for step if needed

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x_i<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Objective evaluation
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Gradient for a specific shift (subgradient of the max‑correlation)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Subgradient descent on the true max (non‑smooth) objective
# ----------------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 800, step0: float = 0.5,
                 decay: float = 0.995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_max(corr: np.ndarray, alpha: float) -> float:
    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _grad_smooth_max(h: np.ndarray, alpha: float):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    exp_term = np.exp(alpha * (corr - max_corr))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    for i, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = i - (n - 1)
        grad += wi * _gradient_for_shift(h, shift)
    return grad, max_corr


def _smooth_gd_optimize(h0: np.ndarray, target: float,
                        alpha: float = 30.0, step0: float = 0.2,
                        max_iter: int = 1200, decay: float = 0.9995) -> np.ndarray:
    h = h0.copy()
    step = step0
    for _ in range(max_iter):
        grad, _ = _grad_smooth_max(h, alpha)
        h -= step * grad
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# SLSQP refinement using the smooth max (as in the reference solution)
# ----------------------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 1000) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ----------------------------------------------------------------------
# Local pairwise refinement (move mass between two coordinates)
# ----------------------------------------------------------------------
def _local_pairwise_refine(h: np.ndarray, target: float, dx: float,
                          cur_val: float, rng: np.random.RandomState,
                          max_iter: int = 2000,
                          eps_start: float = 0.05,
                          eps_end: float = 0.001) -> (np.ndarray, float):
    n = h.shape[0]
    best_h = h.copy()
    best_val = cur_val
    eps = eps_start
    decay = (eps_end / eps_start) ** (1.0 / max_iter) if max_iter > 0 else 1.0

    for it in range(max_iter):
        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue
        if rng.rand() < 0.5:
            # transfer from j to i
            max_delta = min(1.0 - best_h[i], best_h[j])
            if max_delta <= 0:
                continue
            delta = rng.uniform(0.0, max_delta) * eps
        else:
            # transfer from i to j
            max_delta = min(best_h[i], 1.0 - best_h[j])
            if max_delta <= 0:
                continue
            delta = -rng.uniform(0.0, max_delta) * eps
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # evaluate
        val = _compute_c5(new_h, dx)
        if val < best_val:
            best_h = new_h
            best_val = val
        eps *= decay  # gradually shrink step size
    return best_h, best_val
