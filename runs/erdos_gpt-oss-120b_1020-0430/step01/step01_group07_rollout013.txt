def _anneal(start_h, max_seconds):
    h = start_h.copy()
    n = len(h)
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    start = time.time()
    while time.time() - start < max_seconds:
        t_elapsed = time.time() - start
        progress = t_elapsed / max_seconds
        # temperature schedule: exponential
        T = T_start * (T_end / T_start) ** progress
        # sample two indices
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        hi, hj = h[i], h[j]
        # feasible delta range (increase hi, decrease hj)
        low = max(-hi, hj - 1.0)      # can't go below 0 or above 1 for hj
        high = min(1.0 - hi, hj)      # can't exceed 1 for hi or go below 0 for hj
        if low >= high: continue
        # step size scale with temperature
        scale = max_step * T / T_start   # reduces as T falls, but keep > min
        # limit scale to max_step
        # produce delta uniformly within [low, high] but with chance to stay small.
        # For exploration, sometimes random across whole feasible.
        if np.random.rand() < 0.7:
            # small move
            low_s = max(low, -scale)
            high_s = min(high, scale)
            if low_s < high_s:
                delta = np.random.uniform(low_s, high_s)
            else:
                delta = np.random.uniform(low, high)
        else:
            delta = np.random.uniform(low, high)
        # apply
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or np.random.rand() < math.exp(-dC / (T+1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_c

import time
import math
import random
import numpy as np

def _c5_metric(h):
    """C5 value for a sampled step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _interpolate_initial(target_n):
    """Interpolate the provided initial_h_values onto a new grid."""
    h0 = np.asarray(initial_h_values, dtype=np.float64)
    n0 = h0.size
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h = np.interp(x, x0, h0)
    target = target_n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(target_n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _binary_seed(n, target):
    """Generate a 0/1 (with at most one fractional) seed with exact sum."""
    whole = int(target)
    frac = target - whole
    h = np.zeros(n, dtype=np.float64)
    if whole:
        h[:whole] = 1.0
    if frac > 1e-12:
        h[whole] = frac
    np.random.shuffle(h)
    return h


def _continuous_seed(n, target):
    """Generate a random continuous seed with exact sum."""
    h = np.random.rand(n)
    h = h / h.sum() * target
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _anneal(start_h, max_seconds):
    """Simulated‑annealing improvement of a step function."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    T0 = 0.30          # initial temperature
    T_min = 1e-5
    max_step = 0.5     # maximal delta at high temperature

    start = time.time()
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        # exponential cooling
        T = T0 * (T_min / T0) ** prog

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # feasible interval for delta (add to i, subtract from j)
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # occasionally allow a full‑range move
        if random.random() < 0.08:
            delta = random.uniform(low, high)
        else:
            # bound magnitude by temperature‑scaled step size
            bound = max_step * (T / T0)
            bound = max(bound, (high - low) * 0.01)  # ensure non‑zero
            low_b = max(low, -bound)
            high_b = min(high, bound)
            if low_b < high_b:
                delta = random.uniform(low_b, high_b)
            else:
                delta = random.uniform(low, high)

        # apply move
        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0.0 or random.random() < math.exp(-dC / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

    return best_h, best_c


def _hill_climb(start_h, max_seconds):
    """Tiny deterministic improvements after annealing."""
    h = start_h.copy()
    n = h.size
    best_c = _c5_metric(h)
    start = time.time()
    while time.time() - start < max_seconds:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.02
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
        else:
            h[i] = hi
            h[j] = hj
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C5 bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    # decide resolution
    base_n = len(initial_h_values) if "initial_h_values" in globals() else 100
    n = max(200, int(base_n * 5))
    if n % 2:
        n += 1
    target_sum = n / 2.0

    best_h = None
    best_c = float("inf")

    start_all = time.time()
    total_budget = float(budget_s)

    # ---- primary seed -----------------------------------------------------------------
    if "initial_h_values" in globals():
        seed_vec = _interpolate_initial(n)
        primary_time = min(total_budget * 0.25, 90.0)
        cand_h, cand_c = _anneal(seed_vec, primary_time)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ---- iterative random restarts ----------------------------------------------------
    while time.time() - start_all < total_budget - 8.0:
        # choose seed type
        if random.random() < 0.5:
            seed_vec = _binary_seed(n, target_sum)
        else:
            seed_vec = _continuous_seed(n, target_sum)

        slice_time = min((total_budget - (time.time() - start_all)) * 0.2, 30.0)
        cand_h, cand_c = _anneal(seed_vec, slice_time)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ---- final local polishing ---------------------------------------------------------
    remaining = total_budget - (time.time() - start_all)
    if remaining > 2.0 and best_h is not None:
        best_h, best_c = _hill_climb(best_h, remaining)

    # ---- enforce exact sum -------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_c = _c5_metric(best_h)

    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break
    np.clip(best_h, 0.0, 1.0, out=best_h)

    # final evaluation (use helper if present)
    try:
        final_c = evaluate_erdos_solution(best_h)  # pre‑imported helper
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
