import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Projection onto the capped simplex {0 ≤ x ≤ 1, sum(x) = target}
# ------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ------------------------------------------------------------
# Random feasible vector (uniform then project)
# ------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ------------------------------------------------------------
# Correlation based C5 evaluation
# ------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ------------------------------------------------------------
# Subgradient for a given shift (the shift attaining the max)
# ------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ------------------------------------------------------------
# Projected (sub)gradient descent on the max‑correlation objective
# ------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 2000,
                 step0: float = 0.8, decay: float = 0.995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ------------------------------------------------------------
# Smooth‑max objective (log‑sum‑exp) used by SLSQP
# ------------------------------------------------------------
def _smooth_max(corr: np.ndarray, alpha: float) -> float:
    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_refine(h0: np.ndarray, target: float,
                  alphas=(80.0, 40.0, 20.0, 10.0), max_iter: int = 200) -> np.ndarray:
    h = h0.copy()
    n = h.shape[0]

    for alpha in alphas:
        def obj(v):
            corr = np.correlate(v, 1.0 - v, mode='full')
            return _smooth_max(corr, alpha)

        cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
        h = np.clip(res.x, 0.0, 1.0)
        h = _project_onto_capped_simplex(h, target)
    return h


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # ----- Determine discretisation size ---------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    n0 = len(base_h) if base_h is not None else 0
    n = max(400, n0)                     # use at least 400 points for accuracy
    target_sum = n / 2.0                 # because ∫h = 1 and dx = 2/n
    dx = 2.0 / n

    # ----- Helper to upscale a low‑resolution vector ----------------------
    def _upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ----- Initialise best known solution ---------------------------------
    if base_h is not None:
        h_initial = _upscale(base_h)
        h_initial = _project_onto_capped_simplex(h_initial, target_sum)
        best_h = h_initial.copy()
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = None
        best_val = float('inf')

    restart = 0
    # ----- Main optimisation loop ----------------------------------------
    while time.time() - start < budget_s * 0.92:
        # ----- Starting point ---------------------------------------------
        if best_h is not None and rng.rand() < 0.4:
            h0 = best_h + rng.normal(scale=0.03, size=n)
        else:

        # ----- Gradient‑descent phase --------------------------------------
        h_opt = _gd_optimize(h0, target_sum, max_iter=2500,
                             step0=0.8, decay=0.996)

        # ----- Occasionally polish with SLSQP (smooth max) -----------------
        if restart % 7 == 0:
            h_opt = _slsqp_refine(h_opt, target_sum,
                                  alphas=(80.0, 40.0, 20.0, 10.0),
                                  max_iter=150)

        # ----- Evaluate ----------------------------------------------------
        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        restart += 1

    # ----- Final safety projection ----------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
