def compute_c5_fft(h):
    n = h.size
    # pad to size m = next power of 2 > 2*n
    m = 1 << ((2*n-1).bit_length())
    # compute convolution via FFT
    f1 = np.fft.rfft(h, m)
    f2 = np.fft.rfft(np.flip(1 - h), m)
    conv = np.fft.irfft(f1 * f2, m)
    # conv length m; take the first (2*n-1) values (valid linear conv)
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)

def propose_move(h, rng, step_scale):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # decide direction
    if rng.random() < 0.5:
        donor, receiver = i, j
    else:
        donor, receiver = j, i
    max_eps = min(h[donor], 1.0 - h[receiver])
    if max_eps <= 0:
        return None
    eps = rng.random() * max_eps * step_scale
    new_h = h.copy()
    new_h[donor] -= eps
    new_h[receiver] += eps
    return new_h

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

import time
import numpy as np
from scipy.optimize import minimize

# ----- Global variables for SLSQP soft‑max objective -----------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement using a soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def compute_c5_fft(h):
    """Exact discrete C5 using FFT (faster than np.correlate)."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    f1 = np.fft.rfft(h, m)
    f2 = np.fft.rfft(np.flip(1 - h), m)
    conv = np.fft.irfft(f1 * f2, m)
    corr = conv[: 2 * n - 1]
    return float(np.max(corr) * (2.0 / n))

def propose_move(h, rng, step_scale):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n - 1)

def simulated_annealing(h0, rng, time_limit):
