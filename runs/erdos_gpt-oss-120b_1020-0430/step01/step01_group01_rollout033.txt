import time
import math
import numpy as np

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} using bisection on a dual variable."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Uniform random point on the capped simplex."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    return float((np.correlate(h, 1.0 - h, mode="full") * dx).max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation for the shift attaining the current maximum."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

    """Search for a step function giving a better Erdős C5 upper bound."""

    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise a feasible vector

        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, base)
    else:

    h0 = _project_onto_capped_simplex(h0, target_sum)

    # ----------------------------------------------------------------------
    # A short deterministic gradient descent to get a decent baseline
    # ----------------------------------------------------------------------
    h = h0.copy()
    step = 0.6
