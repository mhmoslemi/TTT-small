def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode="full")
    # using logsumexp on α*conv
    from scipy.special import logsumexp
    return float(logsumexp(alpha * conv) / alpha)

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global helpers (no closures, no lambdas)
# ----------------------------------------------------------------------
global_target_sum = None
global_alpha = None


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def softmax_c5_obj(h: np.ndarray) -> float:
    """Wrapper for use by scipy.minimize – reads global_alpha."""
    return softmax_c5(h, alpha=global_alpha)


def constraint_eq_fun(h: np.ndarray) -> float:
    """Equality constraint: sum(h) == target (global_target_sum)."""
    return np.sum(h) - global_target_sum


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, preserving the integral exactly."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    # renormalise to exact integral
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def binary_from_continuous(vec: np.ndarray) -> np.ndarray:
    """Turn a continuous vector into a feasible binary one (half 1s)."""
    n = vec.shape[0]
    idx = np.argsort(-vec)[: n // 2]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def simulated_annealing_binary(
    init_bin: np.ndarray,
    budget_sec: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple:
    """Simple SA on binary vectors preserving exact half‑fill."""
    n = init_bin.shape[0]
    cur = init_bin.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    # temperature schedule parameters
    T0 = 0.02
    max_iter = 200_000
    it = 0
    while time.time() - start_time < budget_sec and it < max_iter:
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = cur.copy()
        new[i] = 0.0
        new[j] = 1.0
        new_c5 = compute_c5(new)

        if new_c5 < cur_c5:
            accept = True
        else:
            elapsed = time.time() - start_time
            # linear cooling; reaches near zero at the end of the allocated budget
            T = T0 * max(0.0, 1.0 - elapsed / budget_sec)
            if T <= 0.0:
                accept = False
            else:
                prob = math.exp(-(new_c5 - cur_c5) / T)
                accept = rng.random() < prob

        if accept:
            cur = new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        it += 1

    return best, best_c5


def local_refine_with_slsqp(start_vec: np.ndarray, max_seconds: float, start_time: float):
    """Two‑stage SLSQP using a smooth max‑overlap surrogate."""
    n = start_vec.shape[0]
    global global_target_sum, global_alpha
    global_target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": constraint_eq_fun}

    # stage 1
    global_alpha = 5.0
    res1 = minimize(
        softmax_c5_obj,
        start_vec,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 400, "ftol": 1e-10, "disp": False},
    )
    if not res1.success or time.time() - start_time > max_seconds:
        return start_vec

    # stage 2
    global_alpha = 30.0
    res2 = minimize(
        softmax_c5_obj,
        np.clip(res1.x, 0.0, 1.0),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 400, "ftol": 1e-10, "disp": False},
    )
    if not res2.success:
        return np.clip(res1.x, 0.0, 1.0)
    return np.clip(res2.x, 0.0, 1.0)
