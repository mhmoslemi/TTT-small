def worst_shift_and_contrib(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)  # can be negative.
    # contributions array length n, zero for indices where out-of-range.
    contrib = np.zeros_like(h)
    if shift >= 0:
        # overlap region i in [0, n-shift-1]
        i = np.arange(n - shift)
        contrib[i] = h[i] * (1 - h[i + shift])
    else:
        # shift negative
        s = -shift
        # i in [s, n-1]
        i = np.arange(s, n)
        contrib[i] = h[i] * (1 - h[i - s])
    return shift, contrib, corr[max_idx]

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x, target):
    """
    Project vector x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1.
    Simple iterative redistribution that respects bounds.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    # iterative redistribution
    while abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return np.clip(x, 0.0, 1.0)


def _worst_shift_and_contrib(h):
    """Return shift index (relative to centre), contribution per point, and max correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)          # shift in index units, can be negative
    contrib = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        contrib[i] = h[i] * (1.0 - h[i + shift])
    else:
        s = -shift
        i = np.arange(s, n)
        contrib[i] = h[i] * (1.0 - h[i - s])
    return shift, contrib, corr[max_idx]


def _interpolate(vec, new_n):
    """Linear interpolation of a periodic vector to a new length."""
    old_n = len(vec)
    if old_n == new_n:
        return np.array(vec, dtype=np.float64)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, vec)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # a little safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")   # may be None
    if init_vec is not None:
        init_vec = np.asarray(init_vec, dtype=np.float64)
        n0 = init_vec.size
        n = max(90, n0 * 2)               # work at a finer resolution
        h = _interpolate(init_vec, n)
    else:
        n = 90
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        # start from the smooth sine‑squared shape (integral = 1)
        h = np.sin(np.pi * x / 2.0) ** 2

    target = n / 2.0
    h = _adjust_sum(h, target)

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Stochastic local search with guided moves
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # parameters
    max_iter_without_improve = 2000
    iter_without_improve = 0

    while time.time() < deadline:
        # occasional full restart to diversify search
        if rng.random() < 0.001:
            cur_h = rng.random(n)
            cur_h = _adjust_sum(cur_h * (target / cur_h.sum()), target)
            cur_c5 = _evaluate_c5(cur_h)
            continue

        # decide whether to perform a guided move (focus on worst shift)
        if rng.random() < 0.6:
            shift, contrib, _ = _worst_shift_and_contrib(cur_h)
            # source: point with largest contribution to the current worst shift
            src = int(np.argmax(contrib))
            # overlapping region for the shift
            if shift >= 0:
                overlap = np.arange(n - shift)
                partner_vals = 1.0 - cur_h[overlap + shift]
                overlap_src = src if src < n - shift else None
            else:
                s = -shift
                overlap = np.arange(s, n)
                partner_vals = 1.0 - cur_h[overlap - s]
                overlap_src = src if src >= s else None

            # choose a destination where the partner is already large (i.e. 1‑h small)
            mask = (cur_h[overlap] < 1.0 - 1e-12) & (partner_vals < 0.5)
            candidates = overlap[mask]
            if candidates.size == 0:
                # fall back to any free spot in the overlapping region
                mask = cur_h[overlap] < 1.0 - 1e-12
                candidates = overlap[mask]
            if candidates.size == 0:
                # as a last resort choose any index with room
                free = np.where(cur_h < 1.0 - 1e-12)[0]
                if free.size == 0:
                    continue
                dst = rng.choice(free)
            else:
                dst = int(rng.choice(candidates))

            # amount to transfer
            max_move = min(cur_h[src], 1.0 - cur_h[dst])
            if max_move <= 1e-12:
                continue
            delta = max_move * rng.random() * 0.5   # limit to modest chunk
            new_h = cur_h.copy()
            new_h[src] -= delta
            new_h[dst] += delta
            new_c5 = _evaluate_c5(new_h)

            # Accept if it improves the global best
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = new_h, new_c5
                cur_h, cur_c5 = new_h, new_c5
                iter_without_improve = 0
                continue
            # Accept as a downhill step for the current state
            if new_c5 + 1e-12 < cur_c5:
                cur_h, cur_c5 = new_h, new_c5
                iter_without_improve = 0
                continue
        else:
            # plain random pairwise mass exchange
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 1e-12:
                continue
            delta = max_move * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_c5 = _evaluate_c5(new_h)

            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = new_h, new_c5
                cur_h, cur_c5 = new_h, new_c5
                iter_without_improve = 0
                continue
            if new_c5 + 1e-12 < cur_c5:
                cur_h, cur_c5 = new_h, new_c5
                iter_without_improve = 0
                continue

        # no improvement this round
        iter_without_improve += 1
        if iter_without_improve >= max_iter_without_improve:
            # perturb current solution a little to escape stagnation
            perturb = rng.normal(scale=0.03, size=n)
            cur_h = np.clip(cur_h + perturb, 0.0, 1.0)
            cur_h = _adjust_sum(cur_h, target)
            cur_c5 = _evaluate_c5(cur_h)
            iter_without_improve = 0

    # ------------------------------------------------------------------
    # 3. Optional final refinement with a small SLSQP run
    # ------------------------------------------------------------------
    def obj(x):
        dx = 2.0 / len(x)
        return np.max(np.correlate(x, 1.0 - x, mode="full")) * dx

    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(
            obj,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 120, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            cand = np.clip(res.x, 0.0, 1.0)
            cand = _adjust_sum(cand, target)
            cand_c5 = _evaluate_c5(cand)
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand, cand_c5
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Return the best discovered solution
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
