def project_to_feasible(x, target):
    x = np.clip(x, 0.0, 1.0)
    # check if current sum already equals target within tolerance
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    # bisection for lambda
    # f(lambda) decreasing.
    lo = -1.0
    hi = 1.0
    # adjust bounds to ensure f(lo) >= target >= f(hi)
    while np.clip(x - lo, 0.0, 1.0).sum() < target:
        lo *= 2  # more negative, increase f
    while np.clip(x - hi, 0.0, 1.0).sum() > target:
        hi *= 2
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = np.clip(x - mid, 0.0, 1.0).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
```

import time
import numpy as np


def _evaluate_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_feasible(v, target):
    """
    Euclidean projection of v onto the box [0,1]^n intersected with the hyper‑plane sum = target.
    Uses a simple bisection on the shift λ in the formula  clip(v-λ,0,1).
    """
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v

    # function f(λ) = sum_i clip(v_i-λ,0,1)
    lo, hi = -1.0, 1.0
    while np.clip(v - lo, 0.0, 1.0).sum() < target:
        lo *= 2.0
    while np.clip(v - hi, 0.0, 1.0).sum() > target:
        hi *= 2.0

    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _random_feasible(n, target, rng):
    """Generate a random vector in [0,1]^n with sum = target."""
    v = rng.random(n)
    return _project_to_feasible(v, target)


def _gradient_descent(h, target, rng, deadline):
    """Smooth‑max gradient descent with increasing β and step decay."""
    n = h.shape[0]
    best_h, best_c5 = h.copy(), _evaluate_c5(h)
    cur_h, cur_c5 = best_h.copy(), best_c5

    beta, beta_max = 1.0, 30.0
    step = 0.12            # learning rate
    no_improve = 0

    while time.time() < deadline:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_corr = corr.max()

        # soft‑max weights
        w = np.exp(beta * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = Σ_s w_s * (1 - (h_{i-s}+h_{i+s}))
        grad = np.zeros_like(cur_h)
        pad = np.pad(cur_h, (n - 1, n - 1), mode="constant")
        for idx, ws in enumerate(w):
            if ws == 0.0:
                continue
            s = idx - (n - 1)                 # shift value
            start_minus = (n - 1) - s
            start_plus = (n - 1) + s
            hm = pad[start_minus : start_minus + n]
            hp = pad[start_plus : start_plus + n]
            grad += ws * (1.0 - (hm + hp))

        # update and project
        new_h = cur_h - step * grad
        new_h = _project_to_feasible(new_h, target)
        new_c5 = _evaluate_c5(new_h)

        # accept if improvement; occasional uphill moves to escape plateaus
        if new_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            no_improve = 0
        else:
            if rng.random() < 0.001:
                cur_h, cur_c5 = new_h, new_c5
            no_improve += 1

        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, cur_h.copy()
            no_improve = 0

        # schedule
        if beta < beta_max:
            beta = min(beta_max, beta + (beta_max - 1.0) / 2000.0)
        step *= 0.9997

        # diversification after stagnation
        if no_improve > 5000:
            pert = rng.normal(scale=0.06, size=n)
            cur_h = _project_to_feasible(best_h + pert, target)
            cur_c5 = _evaluate_c5(cur_h)
            no_improve = 0

    return best_h, best_c5


def _hill_climb(h, c5, rng, deadline):
    """Simple pairwise mass‑exchange hill climbing."""
    n = h.shape[0]
    cur_h, cur_c5 = h.copy(), c5
    best_h, best_c5 = cur_h.copy(), cur_c5

    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        new_c5 = _evaluate_c5(new_h)
        if new_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
        else:
            # tiny chance to accept uphill (temperature‑free)
            if rng.random() < 0.001:
                cur_h, cur_c5 = new_h, new_c5
    return best_h, best_c5


def _optimize_for_n(n, init_vec, rng, deadline):
    """Run gradient descent + hill climbing for a given resolution."""
    target = n / 2.0

    # initialise feasible vector (interpolate if possible)
    if init_vec is not None:
        src_n = np.asarray(init_vec, dtype=np.float64).size
        if src_n == n:
            h0 = np.asarray(init_vec, dtype=np.float64).copy()
        else:
            x_src = np.linspace(0.0, 2.0, src_n, endpoint=False)
            x_dst = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_dst, x_src, init_vec)
    else:
        h0 = _random_feasible(n, target, rng)

    h0 = _project_to_feasible(h0, target)

    # split remaining time: 80 % gradient descent, 20 % hill climb
    time_left = deadline - time.time()
    if time_left <= 0.0:
        return h0, _evaluate_c5(h0)
    gd_deadline = time.time() + time_left * 0.80
    hc_deadline = deadline

    best_h, best_c5 = _gradient_descent(h0, target, rng, gd_deadline)
    best_h, best_c5 = _hill_climb(best_h, best_c5, rng, hc_deadline)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    init_vec = globals().get("initial_h_values")
    # list of resolutions to explore; 45 uses the provided start directly
    resolutions = [45, 60, 80]
    best_h, best_c5, best_n = None, float("inf"), None

    # allocate equal time slices (leaving a small safety buffer)
    slice_sec = budget_s / len(resolutions)
    for idx, n in enumerate(resolutions):
        # ensure we never exceed the global budget
        remaining = budget_s - (time.time() - start_time)
        if remaining < 2.0:
            break
        # give a little less than the ideal slice to keep buffer
        this_budget = min(slice_sec, remaining - 0.5)
        deadline = time.time() + this_budget

        # initialise: for the first resolution we can use the given vector
        init_for_this = init_vec if (init_vec is not None and idx == 0) else None

        h_opt, c5_opt = _optimize_for_n(n, init_for_this, rng, deadline)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

    # final feasibility enforcement
    if best_h is None:
        # fallback: simple uniform feasible vector
        best_n = 45
        best_h = _random_feasible(best_n, best_n / 2.0, rng)
        best_c5 = _evaluate_c5(best_h)

    best_h = _project_to_feasible(best_h, best_n / 2.0)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, best_n
