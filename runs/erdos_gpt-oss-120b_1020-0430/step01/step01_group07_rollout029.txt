def compute_I_k(h):
    n = len(h)
    I = np.empty(2*n-1)
    for k in range(n):
        prod = np.dot(h[:n-k], h[k:])
        I[n-1 + k] = prod
        I[n-1 - k] = prod  # symmetry
    return I

def proj_box_sum(x, target, low=0.0, high=1.0, max_iter=50):
    # binary search λ such that sum(clip(x+λ, low, high)) = target
    lo = -1e3
    hi = 1e3
    for _ in range(max_iter):
        mid = (lo+hi)/2
        y = np.clip(x + mid, low, high)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2, low, high)

def project_onto_box_sum(x, target):
    """Project onto [0,1] box and sum constraint."""
    # Clip to [0,1] first. then adjust.
    y = np.clip(x, 0.0, 1.0)
    s = y.sum()
    if abs(s - target) < 1e-10:
        return y
    # For binary search we modify all entries uniformly.
    # Find λ such that sum(clip(x + λ, 0,1)) = target.
    lo = -1.0
    hi = 1.0
    # expand bounds
    while np.clip(x + lo, 0, 1).sum() > target:
        lo *= 2
    while np.clip(x + hi, 0, 1).sum() < target:
        hi *= 2
    for _ in range(50):
        mid = (lo + hi) / 2.0
        ymid = np.clip(x + mid, 0.0, 1.0)
        smid = ymid.sum()
        if smid < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

def greedy_balancing(h, time_budget, start_time):

        max_c = np.max(c_vals)
        worst_idx = np.argmax(c_vals)

        # compute coefficient per index for improving this lag's product (minimizing c_k)
        abs_k = abs(k)
        # If abs_k == 0: coefficient = 2*h (since product shift zero)
        coeff = np.zeros(n)

            coeff = 2 * h

            coeff[abs_k:]   += h[:n-abs_k] # neighbor -k

            break

def _project_box_sum(x, target):
    lo = -2.0
    hi = 2.0
    # adjust bounds
    # ensure sum(lo) >= target? Actually sum after clip at lo.
    # function to compute sum after clipping x+mid.
    # Expand boundaries as needed
    while True:
        y = np.clip(x + lo, 0.0, 1.0)
        s = y.sum()
        if s <= target: # Lo is low enough that sum less or equal target
            break
        lo *= 2
    while True:
        y = np.clip(x + hi, 0.0, 1.0)
        s = y.sum()
        if s >= target:
            break
        hi *= 2
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid

def _auto_corr(h):

def _c5_metric_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    I = _auto_corr(h)
    c_vals = 1.0 - I * dx
    return np.max(c_vals)

def _subgradient_optimize(h_start, total_budget, start_time):
    h = h_start.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_metric_fft(h)
    cur_c = best_c
    it = 0
    # Some parameters
    step0 = 0.5   # initial step size; adjust after experiments
    # max iterations maybe large
    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_budget - 0.1:  # leave small margin
            break
        # compute autocorrelation
        I = _auto_corr(h)
        c_vals = 1.0 - I * dx
        max_idx = int(np.argmax(c_vals))
        max_c = c_vals[max_idx]
        # update best
        if max_c < best_c - 1e-12:
            best_c = max_c
            best_h = h.copy()
        # worst lag
        k = max_idx - (n - 1)
        # compute gradient
        g = np.ones(n)
        if k >= 0:
            if k != 0:
                g[:n - k] -= h[k:]
                g[k:]     -= h[:n - k]
        else:
            kk = -k
            g[kk:]   -= h[:n - kk]
            g[:n - kk] -= h[kk:]

        # step size
        step = step0 / math.sqrt(it + 1)
        # update
        h_new = h - step * g
        # projection
        h_new = _project_box_sum(h_new, target)
        # compute new c
        I_new = _auto_corr(h_new)
        new_c = np.max(1.0 - I_new * dx)

        # Accept new point always (gradient descent)
        h = h_new
        cur_c = new_c
        it += 1
        # occasional random perturbation if stuck
        if it % 5000 == 0:
            # small random noise
            noise = np.random.uniform(-0.02, 0.02, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h = _project_box_sum(h, target)
    return best_h, best_c

import time, math, random
import numpy as np

# ------------------------------------------------------------
# basic utilities

def _c5_metric_fft(h):
    """C5 value using FFT based autocorrelation (full, non‑circular)."""
    n = h.shape[0]
    dx = 2.0 / n
    I = _auto_corr(h)                     # length 2*n-1, lag –(n‑1)…+(n‑1)
    c_vals = 1.0 - I * dx                  # c_k = ∫ h(x)(1‑h(x+k))dx
    return float(np.max(c_vals))


def _auto_corr(h):
    """Linear autocorrelation I_k = ∫ h(x)h(x+k)dx, returns full array."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    H = np.fft.rfft(h, size)
    conv = np.fft.irfft(H * np.conj(H), size)    # circular autocorr, zero‑padded
    I_pos = conv[:n]                             # k = 0,…,n‑1
    # make symmetric array for negative lags
    return np.concatenate((I_pos[1:][::-1], I_pos))


def _project_box_sum(x, target):
    """
    Project vector x onto the box [0,1]^n while forcing sum(x) = target.
    Uses a simple binary search on a uniform shift λ.
    """
    lo, hi = -2.0, 2.0
    # enlarge bounds until the sum condition is bracketed
    while np.sum(np.clip(x + lo, 0.0, 1.0)) > target:
        lo *= 2.0
    while np.sum(np.clip(x + hi, 0.0, 1.0)) < target:
        hi *= 2.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _initial_vector(n):

        h0 = np.asarray(initial_h_values, dtype=np.float64)

        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder to a random interior entry
        idx = np.random.choice(np.where((h > 0) & (h < 1))[0])
        h[idx] += diff
    return h

def _subgradient_optimize(start_h, max_seconds):
    """Projected sub‑gradient descent on max_k ∫h(1‑h_k)."""
    n = start_h.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    h = start_h.copy()
    best_h = h.copy()
    best_c = _c5_metric_fft(h)
    step0 = 0.5
    it = 0
    start = time.time()
    while time.time() - start < max_seconds:
        I = _auto_corr(h)
        c_vals = 1.0 - I * dx
        max_idx = int(np.argmax(c_vals))
        max_c = c_vals[max_idx]
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()
        k = max_idx - (n - 1)                # signed lag of current worst term

        # sub‑gradient of the active term
        g = np.ones(n)
        if k >= 0:
            if k:
                g[:n - k] -= h[k:]
                g[k:]     -= h[:n - k]
        else:
            kk = -k
            g[kk:]   -= h[:n - kk]
            g[:n - kk] -= h[kk:]

        step = step0 / math.sqrt(it + 1.0)
        h_new = h - step * g
        h_new = _project_box_sum(h_new, target)

        h = h_new
        it += 1

        # occasional small random jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = np.random.uniform(-0.02, 0.02, size=n)
            h = np.clip(h + jitter, 0.0, 1.0)
            h = _project_box_sum(h, target)

    # final check
    final_c = _c5_metric_fft(h)
    if final_c < best_c:
        best_c, best_h = final_c, h.copy()
    return best_h, best_c


def _hill_climb(start_h, max_seconds):
    """Simple pair‑wise hill climbing (few thousand moves)."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric_fft(h)
    cur_c = best_c
    max_step = 0.12
    start = time.time()
    while time.time() - start < max_seconds:
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        low  = max(-hi,     hj - 1.0)
        high = min( 1.0 - hi, hj)
        if low >= high:
            continue
        # keep steps modest most of the time
        low_cap  = max(low,  -max_step)
        high_cap = min(high,  max_step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric_fft(h)
        if new_c + 1e-12 < cur_c:          # improvement
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = hi, hj            # revert
    return best_h, best_c

# main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # choose a moderate resolution (even number)
    n_target = 200
    if n_target % 2:
        n_target += 1

    # primary start – interpolated parent or uniform half‑density
    h0 = _initial_vector(n_target)

    # --------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (≈70 % of budget)
    # --------------------------------------------------------
    phase1_time = total_budget * 0.70
    best_h, best_c = _subgradient_optimize(h0, phase1_time)

    # --------------------------------------------------------
    # Phase 2 – local hill‑climbing on the remainder
    # --------------------------------------------------------
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 5.0:
        hc_h, hc_c = _hill_climb(best_h, remaining)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

    # --------------------------------------------------------
    # Optional quick random restarts if time permits
    # --------------------------------------------------------
    while True:
        elapsed = time.time() - start_all
        if elapsed >= total_budget - 5.0:
            break
        # tiny random feasible start
        h_rand = np.random.rand(n_target)
        h_rand = _project_box_sum(h_rand, n_target / 2.0)

        # split remaining budget proportionally
        slice_time = min((total_budget - elapsed) * 0.25, 30.0)
        cand_h, cand_c = _subgradient_optimize(h_rand, slice_time)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        # a short hill‑climb after each restart
        elapsed = time.time() - start_all
        if total_budget - elapsed > 5.0:
            hc_h, hc_c = _hill_climb(cand_h, 5.0)
            if hc_c < best_c:
                best_h, best_c = hc_h, hc_c

    # final safeguard – enforce exact sum
    target_sum = n_target / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = _project_box_sum(best_h, target_sum)

    # evaluate with the provided helper if available
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric_fft(best_h)

    return best_h, float(final_c), int(n_target)
