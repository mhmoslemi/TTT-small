import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _evaluate_c5(h):
    """Maximum overlap integral for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x, target):
    """
    Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1.
    Simple water‑filling loop that respects the bounds.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _worst_shift_and_gradient(h):
    """Return worst shift, its value and gradient of the corresponding overlap."""
    n = h.size
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)          # lag (can be negative)
    # gradient of the overlap (without the final dx factor)
    grad = np.zeros_like(h)
    if s >= 0:
        grad[: n - s] = a[s:]
        grad[s:] -= h[: n - s]
    else:
        t = -s
        grad[t:] = a[: n - t]
        grad[: n - t] -= h[t:]
    # incorporate dx (objective = overlap*dx)
    grad *= dx
    val = corr[idx] * dx
    return s, val, grad


def _gradient_descent(h0, deadline, rng, max_iters=2000):
    """Simple projected gradient descent on the worst shift."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _evaluate_c5(h)
    step = 0.2                     # initial step size (in objective units)

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        s, cur_val, grad = _worst_shift_and_gradient(h)
        # try a step opposite to the gradient
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _adjust_sum(h_new, target)
        new_val = _evaluate_c5(h_new)

        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.2, 1.0)      # accelerate when improving
        else:
            step *= 0.5                     # back‑off on failure
            if step < 1e-6:                 # restart from current best if stuck
                h = best_h.copy()
                step = 0.2
    return best_h, best_val


def _targeted_move(h, s, rng):
    """Move mass from a high‑impact index to a low‑impact index for shift s."""
    n = h.size
    # impact of increasing h[i] on the worst overlap (larger = worse)
    inc_cost = np.empty(n, dtype=h.dtype)
    if s >= 0:
        inc_cost[: n - s] = 1.0 - h[s:]
        inc_cost[n - s :] = 1.0
    else:
        t = -s
        inc_cost[t:] = 1.0 - h[: n - t]
        inc_cost[: t] = 1.0

    src_candidates = np.where(h > 1e-12)[0]
    dst_candidates = np.where(h < 1.0 - 1e-12)[0]
    if src_candidates.size == 0 or dst_candidates.size == 0:
        return None, False

    src = src_candidates[np.argmax(inc_cost[src_candidates])]
    dst = dst_candidates[np.argmin(inc_cost[dst_candidates])]
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-12:
        return None, False
    delta = max_move * rng.random()
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new, True


def _hill_climb_mix(h0, val0, deadline, rng, max_iters=5000):
    """
    Random + targeted hill climbing. Returns the best vector seen.
    """
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = val0

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        # decide move type
        if rng.random() < 0.3:          # 30% targeted
            s, _, _ = _worst_shift_and_gradient(h)
            h_new, ok = _targeted_move(h, s, rng)
            if not ok:
                continue
        else:                           # generic random mass exchange

            max_move = min(h[src], 1.0 - h[dst])

        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_h = h_new.copy()
            h = h_new
        else:
            # occasional uphill acceptance (tiny probability) to escape plateaus
            if rng.random() < 0.001:

    return best_h, best_val


def _upsample(vec, n):
    """Linear interpolation of a shorter vector to length n."""
    n0 = vec.size
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x, x0, vec)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = _adjust_sum(h, target)
    return h
