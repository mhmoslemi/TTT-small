def hill_climb(h, best_c5, time_deadline):
    n = len(h)
    target_sum = n / 2.0
    best = h.copy()
    best_val = best_c5
    cur = h.copy()
    cur_val = best_c5
    while time.time() < time_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose delta
        max_delta = min(cur[i], 1 - cur[j])
        if max_delta <= 1e-9:
            continue
        # random fraction
        delta = rng.random() * max_delta

import time, numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode='full')
    # Use log-sum-exp to approximate max
    return float(logsumexp(alpha * conv) / alpha)

def generate_triangular(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    # scale to have sum n/2
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def random_start(n, rng):
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

def solve_one_start(start, n, rng, deadline):
    target = n / 2.0
    def eq_sum(h):
        return np.sum(h) - target
    cons = {"type": "eq", "fun": eq_sum}
    bounds = [(0.0, 1.0)] * n
    # progressive smoothing schedule
    alphas = [5.0, 15.0, 35.0, 70.0]
    cur = start.copy()
    for a in alphas:
        if time.time() > deadline:
            return None, None
        res = minimize(lambda h: softmax_c5(h, alpha=a),
                       cur,
                       method="SLSQP",
                       bounds=bounds,
                       constraints=cons,
                       options={"maxiter": 400, "ftol": 1e-9, "disp": False})
        if not res.success:
            # keep current best even if failure
            cur = res.x
        else:
            cur = res.x
    h_opt = np.clip(cur, 0.0, 1.0)
    # final fix sum
    cur_sum = h_opt.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        h_opt = h_opt + diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_val = compute_c5(h_opt)
    return h_opt, c5_val

def hill_climb(h0, c5_init, deadline):

    h = h0.copy()

    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1] and sum=n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

# Optimisation primitives
# ------------------------------------------------------------
def solve_one_start(start: np.ndarray, n: int, rng: np.random.Generator,
                    deadline: float) -> tuple:
    """Two‑stage SLSQP optimisation with progressive smoothing."""
    target = n / 2.0

    def eq_sum(h):
        return np.sum(h) - target

    cons = {"type": "eq", "fun": eq_sum}
    bounds = [(0.0, 1.0)] * n

    # smoothing schedule (alpha – low is smooth, high is sharp)
    alphas = [5.0, 15.0, 35.0, 70.0]
    cur = start.copy()
    for a in alphas:
        if time.time() > deadline:
            return None, None
        res = minimize(
            lambda h: softmax_c5(h, alpha=a),
            cur,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 400, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            # keep the last iterate even on failure
            cur = res.x
        else:
            cur = res.x

    h_opt = np.clip(cur, 0.0, 1.0)
    # final sum‑correction (rarely needed)
    diff = target - h_opt.sum()
    if abs(diff) > 1e-12:
        h_opt = h_opt + diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)

    c5_val = compute_c5(h_opt)
    return h_opt, c5_val


def hill_climb(h0: np.ndarray, c5_init: float, deadline: float) -> tuple:
    """Greedy (with occasional uphill) local search respecting the constraints."""
    rng = np.random.default_rng(12345)
    h = h0.copy()
    cur_val = c5_init
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    target = n / 2.0
    start_time = time.time()
    T0 = 0.002  # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        if total <= 0.0:
            break
        T = T0 * (1.0 - elapsed / total)  # linear cooling

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        # step size grows with temperature
        frac = 0.01 + 0.99 * (T / T0)  # 1 % → 100 %
        delta = rng.random() * max_delta * frac

        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        diff = new_val - cur_val
        accept = diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12))
        if accept:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta

    # final sum‑fix
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = best_h + diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val

# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function for the Erdős C₅ problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Grid sizes to explore (larger gives more flexibility)
    candidate_ns = [150, 180, 200, 240, 300]

    # If a parent construction is available, use it
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target_sum = n / 2.0

        # ----------------- build a diverse start pool -----------------
        starts = []

        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        starts.append(generate_triangular(n))

        for _ in range(4):
            starts.append(random_start(n, rng))

        tri = generate_triangular(n)
        for _ in range(3):
            pert = tri + rng.uniform(-0.2, 0.2, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            diff = target_sum - pert.sum()
            if abs(diff) > 1e-12:
                pert += diff / n
                pert = np.clip(pert, 0.0, 1.0)
            starts.append(pert)

        # ----------------- optimise each start -----------------
        for s in starts:
            if time.time() > deadline:
                break

            # tiny feasibility cleanup before optimisation
            s = np.clip(s, 0.0, 1.0)
            diff = target_sum - s.sum()
            if abs(diff) > 1e-12:
                s = s + diff / n
                s = np.clip(s, 0.0, 1.0)

            h_opt, c5_opt = solve_one_start(s, n, rng, deadline)
            if h_opt is not None and c5_opt < best_c5 - 1e-10:
                best_h, best_c5, best_n = h_opt, c5_opt, n

                # local refinement if time permits
                if time.time() < deadline:
                    refined_h, refined_c5 = hill_climb(best_h, best_c5, deadline)
                    if refined_c5 < best_c5 - 1e-12:
                        best_h, best_c5 = refined_h, refined_c5

                # early stop when target is achieved
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # ----------------- fallback (should never happen) -----------------
    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final feasibility enforcement
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
