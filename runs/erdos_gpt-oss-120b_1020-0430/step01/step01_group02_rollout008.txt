def targeted_move(h):
    # find shift with max C(s)
    corr = np.correlate(h, 1 - h, mode='full')
    s_idx = np.argmax(corr) - (n-1)  # index offset
    # shift s corresponds to lag = s_idx (can be negative)
    s = s_idx
    # find i such that both i and i+s in range and both < 1
    # generate list of such i
    indices = np.arange(n)
    i_valid = indices[(indices + s >= 0) & (indices + s < n) & (h < 1-1e-8) & (h[indices + s] < 1-1e-8)]
    if len(i_valid) == 0:
        return False
    i = rng.choice(i_valid)
    # need to add delta to both i and i+s
    max_delta = min(1 - h[i], 1 - h[i + s])
    # we need to subtract same total mass from other positions
    # choose two sources j,k (or one source if enough)
    # For simplicity, pick source j (different) with h[j] > delta
    # We'll move delta from j to i, and delta from k to i+s
    # pick j and k distinct and not equal to i or i+s
    candidates = np.arange(n)
    mask = (candidates != i) & (candidates != i + s)
    cand_src = candidates[mask]
    # pick src1 and src2
    src1 = rng.choice(cand_src)
    src2 = rng.choice(cand_src)
    # compute possible deltas
    delta1 = min(h[src1], max_delta)
    delta2 = min(h[src2], max_delta)
    # we want both deltas equal, so delta = min(delta1, delta2, max_delta)
    delta = min(delta1, delta2, max_delta)
    if delta <= 1e-12:
        return False
    # apply moves
    h[src1] -= delta
    h[i] += delta
    h[src2] -= delta
    h[i + s] += delta
    return True

def _evaluate_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # iterative redistribution
    # tolerance
    eps = 1e-12
    while abs(diff) > eps:
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        indices = np.where(mask)[0]
        per = diff / len(indices)
        # vectorized update:
        add = np.where(per > 0, np.minimum(per, 1.0 - x[indices]), np.maximum(per, -x[indices]))
        x[indices] += add
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize


def _evaluate_c5(h: np.ndarray) -> float:
    """
    Compute the Erdős C5 value for a step vector h.
    C5 = max_k int h(x) (1 - h(x+k)) dx  ≈  max_corr * dx
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the box [0,1]^n while forcing sum(x) = target.
    Simple water‑filling style iteration – fast enough for n ≤ 1000.
    """
    x = np.clip(x, 0.0, 1.0)
    eps = 1e-12
    diff = target - x.sum()
    while abs(diff) > eps:
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        # vectorised bounded redistribution
        if per > 0:
            add = np.minimum(per, 1.0 - x[idx])
        else:
            add = np.maximum(per, -x[idx])
        x[idx] += add
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)
