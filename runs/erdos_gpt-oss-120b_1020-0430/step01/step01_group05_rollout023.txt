def compute_c5_fft(h):
    n = h.shape[0];
    L = int(2**np.ceil(np.log2(2*n-1)))
    a = np.fft.rfft(h, n=L)
    b = np.fft.rfft((1-h)[::-1], n=L) # reversed
    conv = np.fft.irfft(a * b, n=L)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())  # next pow of 2
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)

def targeted_hill_climb(h, target_sum, deadline):
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5_fft(best_h)
    while time.time() < deadline:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = int(np.argmax(corr))
        # shift sign: correlation array indexes from -(n-1) to n-1; index offset = n-1
        lag = k_max - (n-1)
        if lag == 0:
            # skip if zero shift (should be constant 0.5 maybe)
            # just random move
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            # same as previous random pairwise move
        else:
            # use i where i+lag in range
            if lag > 0:
                valid_i = np.arange(n - lag)
            else:
                valid_i = np.arange(-lag, n)  # i+lag >=0
            # compute contributions hi*(1 - h[i+lag])
            partner = valid_i + lag
            contributions = best_h[valid_i] * (1 - best_h[partner])

            idx = np.argmax(contributions)
            i = valid_i[idx]
            j = partner[idx]  # its partner

            delta = min(best_h[i], 1 - best_h[j]) * np.random.rand()

            cand = np.clip(cand, 0.0, 1.0)

            if lag == 0:
                # fallback random
                i = random.randint(block_len, n - block_len - 1)
                j = random.randint(block_len, n - block_len - 1)
                # swap as before random pairwise
                # ...
                continue
            # compute valid i range
            if lag > 0:
                i_vals = np.arange(n - lag)
            else:
                i_vals = np.arange(-lag, n)
            # contributions
            partners = i_vals + lag
            contributions = best_h[i_vals] * (1 - best_h[partners])
            # choose i with max contribution
            max_idx = int(np.argmax(contributions))
            i = int(i_vals[max_idx])
            j = int(partners[max_idx])

            delta_max = min(best_h[i], 1 - best_h[j])
            if delta_max <= 0:

            cand[j] += delta

            j = random.randint(block_len, n - block_len - 1)

import time, random, numpy as np
from scipy.optimize import differential_evolution


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while enforcing sum(x)=target (assumes feasible)."""
    low, high = -2.0, 2.0
    for _ in range(80):
        lam = (low + high) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    """Triangular bump that peaks at 1 in the centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new length."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def targeted_hill_climb(
    h: np.ndarray, target_sum: float, deadline: float, block_len: int
) -> tuple[np.ndarray, float]:
    """
    Local optimisation.  Moves mass from a point that contributes heavily to the
    current worst shift towards its partner, and also performs occasional random swaps.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    interior_start = block_len
    interior_end = n - block_len

    while time.time() < deadline:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)  # shift corresponding to the maximum

        if lag != 0:
            # indices where the term h[i]*(1-h[i+lag]) actually appears
            if lag > 0:
                i_vals = np.arange(interior_start, interior_end - lag)
                partner = i_vals + lag
            else:
                i_vals = np.arange(interior_start - lag, interior_end)
                partner = i_vals + lag

            if i_vals.size == 0:
                # fall back to a random pairwise move
                i = random.randint(interior_start, interior_end - 1)
                j = random.randint(interior_start, interior_end - 1)
                hi, hj = best_h[i], best_h[j]
                delta_max = min(1.0 - hi, hj)
                delta_min = -min(hi, 1.0 - hj)
                if delta_max < delta_min:
                    continue
                delta = random.uniform(delta_min, delta_max)
                if abs(delta) < 1e-12:
                    continue
                cand = best_h.copy()
                cand[i] = hi + delta
                cand[j] = hj - delta
            else:
                # choose the pair with the largest contribution to the worst shift
                contrib = best_h[i_vals] * (1.0 - best_h[partner])
                idx = int(np.argmax(contrib))
                i = int(i_vals[idx])
                j = int(partner[idx])
                delta_max = min(best_h[i], 1.0 - best_h[j])
                if delta_max <= 0:
                    continue
                delta = delta_max * random.random()
                cand = best_h.copy()
                cand[i] -= delta
                cand[j] += delta
        else:
            # pure random feasible transfer
            i = random.randint(interior_start, interior_end - 1)
            j = random.randint(interior_start, interior_end - 1)
            if i == j:
                continue
            hi, hj = best_h[i], best_h[j]
            delta_max = min(1.0 - hi, hj)
            delta_min = -min(hi, 1.0 - hj)
            if delta_max < delta_min:
                continue
            delta = random.uniform(delta_min, delta_max)
            if abs(delta) < 1e-12:
                continue
            cand = best_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

        # clip to bounds (sum is already preserved)
        cand = np.clip(cand, 0.0, 1.0)
        cval = compute_c5(cand)

        if cval < best_val:
            best_val, best_h = cval, cand
            if best_val <= 0.3808:
                break

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise a step‑function for the Erdős C₅ constant."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # problem parameters ----------------------------------------------------
    M = 20                     # number of constant blocks
    block_len = 25             # points per block, 20*25 = 500 total samples
    n = M * block_len
    dx = 2.0 / n
    target_sum = n / 2.0
    target_block_sum = target_sum / block_len  # sum of block values

    # initial feasible vector -----------------------------------------------
    if "initial_h_values" in globals():
        try:
            h0 = upsample_initial(np.array(initial_h_values, dtype=float), n)
        except Exception:
            h0 = make_triangle(n)
    else:
        h0 = make_triangle(n)

    # interior block variables (first and last block forced to 0)
    v0 = h0.reshape(M, block_len).mean(axis=1)   # block averages
    w0 = v0[1:-1]                                # interior blocks
    w0 = project_onto_bounds_and_sum(w0, target_block_sum)

    # differential evolution on interior blocks -----------------------------
    bounds = [(0.0, 1.0)] * (M - 2)
    popsize = 15

    def de_obj(w):
        """Penalty objective for DE (max C₅ + quadratic sum penalty)."""
        v = np.empty(M)
        v[0] = 0.0
        v[-1] = 0.0
        v[1:-1] = w
        s = v.sum()
        diff = s - target_block_sum
        penalty = 1e5 * diff * diff if abs(diff) > 1e-9 else 0.0
        h = np.repeat(v, block_len)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5 + penalty

    # initialise a population (first individual = w0)
    init_pop = np.random.rand(popsize, M - 2)
    for i in range(popsize):
        row = init_pop[i]
        row = row - (row.sum() - target_block_sum) / (M - 2)
        init_pop[i] = np.clip(row, 0.0, 1.0)
    init_pop[0] = w0.copy()

    # run DE (time‑bounded)
    time_left = budget_s - (time.time() - start) - 30.0  # reserve time for local search
    maxiter_de = max(1, int(time_left // (popsize * 0.001)))  # crude estimate
    maxiter_de = min(maxiter_de, 400)                      # safeguard
    de_res = differential_evolution(
        de_obj,
        bounds,
        maxiter=maxiter_de,
        popsize=popsize,
        init=init_pop,
        polish=False,
        seed=seed,
        updating="deferred",
        disp=False,
        workers=1,
    )
    best_w = de_res.x
    best_w = project_onto_bounds_and_sum(best_w, target_block_sum)

    # reconstruct full step function from block values
    v_best = np.concatenate(([0.0], best_w, [0.0]))
    h_best = np.repeat(v_best, block_len)
    h_best = project_onto_bounds_and_sum(h_best, target_sum)

    best_val = compute_c5(h_best)

    # if early success, return immediately
    if best_val <= 0.3808:
        return (h_best.tolist(), float(best_val), int(n))

    # local refinement -------------------------------------------------------
    deadline = start + budget_s - 0.5
    h_final, val_final = targeted_hill_climb(h_best, target_sum, deadline, block_len)

    # final safety projection and evaluation
    h_final = np.clip(h_final, 0.0, 1.0)
    h_final = project_onto_bounds_and_sum(h_final, target_sum)
    best_val = compute_c5(h_final)

    return (h_final.tolist(), float(best_val), int(n))
