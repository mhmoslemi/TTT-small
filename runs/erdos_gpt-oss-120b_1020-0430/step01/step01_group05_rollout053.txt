def compute_c5_fft(h, L=None):
    n = h.shape[0]
    L = L or (1 << (2*n-1).bit_length())
    # compute rfft for zero-padded length L
    fh = np.fft.rfft(h, n=L)
    f1h = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fh * np.conj(f1h), n=L)[:2*n-1]
    return np.max(corr) * (2.0 / n)

    dx = 2.0/n

    # compute initial value and set current
    cur_h = h.copy()
    cur_val = best_val
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = 0.05  # temperature
    # schedule: T = T0 * (1 - (time_elapsed / budget)) maybe linear decrease

    while time.time() - start_time < budget - 0.5:
        # maybe set iterations per check?
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

def compute_c5_fft(h):
    n = h.shape[0]
    L = (1 << ((2*n-1).bit_length()))  # next pow of two >= 2n-1

import time, math
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Global for equality constraint (set inside run)
TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n intersect sum(x)=target via bisection on lambda."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def sum_eq_constraint(x: np.ndarray) -> float:  # used by SLSQP
    return np.sum(x) - TARGET_SUM_GLOBAL


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 using FFT based correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, n=L)
    f1h = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fh * np.conj(f1h), n=L)[: 2 * n - 1]
    return np.max(corr) * dx


def objective_smooth_fft(h: np.ndarray, alpha: float = 100.0) -> float:
    """Smooth approximation of max C5, used by SLSQP."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, n=L)
    f1h = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fh * np.conj(f1h), n=L)[: 2 * n - 1]
    return smooth_max(corr, alpha) * dx


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Width 1 centred rectangle (integral = 1)."""
    h = np.zeros(n)
    dx = 2.0 / n
    length = int(round(1.0 / dx))
    start = (n - length) // 2
    h[start : start + length] = 1.0
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_double_rectangle(n: int) -> np.ndarray:
    """Two disjoint rectangles each of width 0.5."""
    h = np.zeros(n)
    dx = 2.0 / n
    length = int(round(0.5 / dx))
    h[:length] = 1.0
    h[-length:] = 1.0
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h that lowers the Erdős C5 constant."""
    global TARGET_SUM_GLOBAL
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # problem size
    n = 400                     # even, < 1000
    target_sum = n / 2.0
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # --------------------------------------------------------
    # build a diverse set of initial seeds
    seeds = []

    if "initial_h_values" in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rectangle(n))
    seeds.append(make_double_rectangle(n))

    # --------------------------------------------------------
    best_h = None
    best_val = np.inf

    # ----- local SLSQP refinement on each seed ----------------
    for seed_vec in seeds:
        if time.time() - start_time > budget_s - 5.0:
            break
        try:
            res = minimize(
                objective_smooth_fft,
                seed_vec,
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
                args=(120.0,),
            )
        except Exception:
            continue

        h_opt = project_onto_bounds_and_sum(res.x, target_sum)
        val = compute_c5_fft(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        if best_val <= 0.3808:
            break

    # if nothing succeeded, fall back to a deterministic shape
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5_fft(best_h)

    # --------------------------------------------------------
    # Simulated annealing starting from the best solution found
    cur_h = best_h.copy()
    cur_val = best_val
    global_best_h = best_h.copy()
    global_best_val = best_val

    # temperature schedule parameters
    T0 = 0.04  # initial temperature
    step = 0

    while time.time() - start_time < budget_s - 1.0:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = T0 * max(0.0, 1.0 - frac)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        hi, hj = cur_h[i], cur_h[j]

        # propose either a simple mass transfer or a random redistribution
        if rng.random() < 0.15:  # random redistribution
            s = hi + hj
            lo = max(0.0, s - 1.0)
            hi_bound = min(1.0, s)
            a = lo + rng.random() * (hi_bound - lo)
            b = s - a
            cand = cur_h.copy()
            cand[i], cand[j] = a, b
        else:  # transfer from j to i
            max_inc = 1.0 - hi
            max_dec = hj
            dmax = min(max_inc, max_dec)
            if dmax <= 0:
                continue
            delta = rng.random() * dmax
            if delta <= 0:
                continue
            cand = cur_h.copy()
            cand[i] += delta
            cand[j] -= delta

        cand_val = compute_c5_fft(cand)

        # acceptance
        if cand_val < cur_val:
            cur_h, cur_val = cand, cand_val
            if cand_val < global_best_val:
                global_best_val, global_best_h = cand_val, cand.copy()
        else:
            if T > 0 and rng.random() < math.exp((cur_val - cand_val) / T):
                cur_h, cur_val = cand, cand_val

        step += 1
        # occasional re‑projection to kill numerical drift
        if step % 2000 == 0:
            cur_h = project_onto_bounds_and_sum(cur_h, target_sum)

        if global_best_val <= 0.3808:
            break

    # final projection and evaluation
    best_h = project_onto_bounds_and_sum(global_best_h, target_sum)
    best_val = compute_c5_fft(best_h)

    return (best_h.tolist(), float(best_val), int(n))
