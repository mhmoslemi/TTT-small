import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Core evaluation
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 (max overlap integral)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


# ----------------------------------------------------------------------
# Simple constructions
# ----------------------------------------------------------------------
def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum n/2."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def generate_triangular(n: int) -> np.ndarray:
    """Triangular hat, renormalised to integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def initialize_cosine(n: int) -> np.ndarray:
    """Raised‑cosine bump (integral already 1)."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = 0.5 * (1.0 + np.cos(np.pi * (x - 1.0)))
    target = n / 2.0
    s = h.sum()
    if s != 0.0:
        h = h * (target / s)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(arr) to new_n, then renormalise."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    np.clip(up, 0.0, 1.0, out=up)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        np.clip(up, 0.0, 1.0, out=up)
    return up


def symmetrize(h: np.ndarray) -> np.ndarray:
    """Enforce reflection symmetry without changing the total sum."""
    return (h + h[::-1]) * 0.5

# Simulated annealing (pairwise mass exchange)
# ----------------------------------------------------------------------
def simulated_annealing(start: np.ndarray, rng: np.random.Generator,
                       deadline: float, max_iter: int = 25000) -> tuple:
    """Greedy/SA improvement preserving the sum exactly."""
    h = start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    step = 0.2          # maximal transferred amount per move
    temp = 0.02         # annealing temperature

    for it in range(max_iter):
        if time.time() > deadline:
            break
        # schedule
        if it and it % 5000 == 0:
            step *= 0.5
            temp *= 0.5
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # feasible delta range respecting bounds and step size
        lo = max(-h[i], -(1.0 - h[j]), -step)
        hi = min(1.0 - h[i], h[j], step)
        if lo >= hi:
            continue
        delta = rng.uniform(lo, hi)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # numeric safety
        np.clip(h_new, 0.0, 1.0, out=h_new)
        new_c5 = compute_c5(h_new)
        # acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    return best_h, best_c5

def refine_with_slsqp(start: np.ndarray, n: int, deadline: float) -> tuple:
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    h_curr = start.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)

    for alpha in (5.0, 20.0, 60.0, 200.0):
        if time.time() > deadline:
            break
        res = minimize(
            lambda h: softmax_c5(h, alpha=alpha),
            h_curr,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            continue
        h_curr = np.clip(res.x, 0.0, 1.0)
        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()
    return best_h, best_c5

    # parent construction, if supplied by the harness
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    # candidate grid sizes – give priority to the parent size
    candidate_ns = [80, 100, 120, 150, 180, 200, 240]
    if parent_n is not None and parent_n not in candidate_ns:
        candidate_ns.insert(0, parent_n)

    # ------------------------------------------------------------------
    # Sweep over resolutions
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > deadline:
            break
        target_sum = n / 2.0

        # ----------------- build a diverse start set -----------------
        starts = []

        # parent (upsampled if needed)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # analytic baselines
        starts.append(initialize_cosine(n))
        starts.append(generate_triangular(n))

        # random seeds
        for _ in range(4):
            starts.append(random_feasible(n, rng))

        # perturbed cosine
        base_cos = initialize_cosine(n)
        for _ in range(3):
            p = base_cos + rng.uniform(-0.2, 0.2, size=n)
            np.clip(p, 0.0, 1.0, out=p)
            diff = target_sum - p.sum()
            if abs(diff) > 1e-12:
                p += diff / n
                np.clip(p, 0.0, 1.0, out=p)
            starts.append(p)

        # -----------------------------------------------------------------
        # Optimize each start
        # -----------------------------------------------------------------
        for s in starts:
            if time.time() > deadline:
                break
            s = symmetrize(s)                     # enforce reflection symmetry
            # quick SA phase
            sa_h, sa_c5 = simulated_annealing(s, rng, deadline, max_iter=20000)
            # final local refinement
            fin_h, fin_c5 = refine_with_slsqp(sa_h, n, deadline)
            fin_h = symmetrize(fin_h)             # maintain symmetry
            if fin_c5 < best_c5 - 1e-12:
                best_h, best_c5, best_n = fin_h, fin_c5, n
                # early exit if we already beat the target bound
                if best_c5 <= 0.3808:
                    return best_h, best_c5, best_n

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing succeeded
    # ------------------------------------------------------------------
    if best_h is None:
        n = 120
        best_h = random_feasible(n, rng)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final sanity correction of the integral
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        np.clip(best_h, 0.0, 1.0, out=best_h)
        diff = target - best_h.sum()
        if abs(diff) > 1e-12:
            best_h += diff / best_n
            np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
