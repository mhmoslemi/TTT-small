def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    n = 260? 250 maybe.
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    seeds = []
    if 'initial_h_values' in globals():
        try: seeds.append(upsample_initial(np.array(initial_h_values,float), n))
        except: pass
    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rectangle(n))
    # add few random seeds later

    best_h = None
    best_val = np.inf

    # Phase 1: SLSQP with moderate alpha
    for alpha in [50, 150]:
        for seed_vec in seeds:
            if time.time() - start > budget_s - 5: break
            # run SLSQP
            res = minimize(objective_smooth, seed_vec, args=(dx, alpha),
                          method='SLSQP', bounds=bounds,
                          constraints={'type':'eq','fun':sum_eq_constraint},
                          options={'maxiter':400, 'ftol':1e-12, 'disp':False})
            h_opt = project_onto_bounds_and_sum(res.x, target_sum)
            val = compute_c5(h_opt)
            if val < best_val:
                best_val = val
                best_h = h_opt.copy()
            if best_val <= 0.3808:
                break

    # Phase 2: Random restarts if time remains
    while time.time() - start < budget_s - 5:
        # random feasible start
        rnd = np.random.rand(n)
        h0 = project_onto_bounds_and_sum(rnd, target_sum)
        # try with high alpha
        alpha = 200
        res = minimize(objective_smooth, h0,
                       args=(dx, alpha),
                       method='SLSQP', bounds=bounds,
                       constraints={'type':'eq','fun':sum_eq_constraint},
                       options={'maxiter':300, 'ftol':1e-10, 'disp':False})
        h_opt = project_onto_bounds_and_sum(res.x, target_sum)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        if best_val <= 0.3808:
            break
        # if time low break
        if time.time()-start > budget_s - 20: break

    # Phase 3: Local refinement hill climbing
    # while time left, try many random pairwise adjustments
    while time.time() - start < budget_s - 0.5:
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        hi, hj = best_h[i], best_h[j]
        # max allowed delta (increase hi, decrease hj)
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        # random delta within allowed range, weighted by step size decreasing over time
        # Step size decreasing as time passes
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        max_step = 0.1 * (remaining / budget_s) + 0.01  # adapt
        delta_range = max(delta_min, -max_step), min(delta_max, max_step)
        if delta_range[0] >= delta_range[1]:
            continue
        delta = np.random.uniform(delta_range[0], delta_range[1])
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        # project to bounds and sum? Already ensure within [0,1] due to delta selection; sum preserved.
        # Optionally small numeric clip
        cand = np.clip(cand, 0.0, 1.0)
        val = compute_c5(cand)
        if val < best_val:

            best_h = cand.copy()
            if best_val <= 0.3808:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while forcing sum(x)=target via bisection on λ."""
    low, high = -1.0, 1.0

        s = proj.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth version of C5 used for SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint ∑h = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Symmetric triangle, height 1 at centre."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2). Already integrates to 1."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Indicator of an interval of length 1 centred at 1 (half the domain)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = ((pts >= 0.5) & (pts <= 1.5)).astype(float)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to the current resolution."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Find a step function h on [0,2] with a low Erdős C5 bound."""
    global TARGET_SUM_GLOBAL
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # problem size – 300 points gives a good trade‑off between flexibility and speed
    n = 300
    dx = 2.0 / n
    target_sum = n / 2.0
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # build a collection of diverse starting vectors
    seeds = []

    if "initial_h_values" in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rectangle(n))

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – deterministic SLSQP runs with increasing smoothness
    smooth_alphas = [50.0, 150.0, 300.0]
    maxiter_slsqp = 400

    for alpha in smooth_alphas:
        for h0 in seeds:
            if time.time() - start > budget_s - 5.0:
                break
            try:
                res = minimize(
                    objective_smooth,
                    h0,
                    args=(dx, alpha),
                    method="SLSQP",
                    bounds=bounds,
                    constraints=[{"type": "eq", "fun": sum_eq_constraint}],
                    options={"maxiter": maxiter_slsqp, "ftol": 1e-12, "disp": False},
                )
            except Exception:
                continue
            h_opt = project_onto_bounds_and_sum(res.x, target_sum)
            val = compute_c5(h_opt)
            if val < best_val:
                best_val, best_h = val, h_opt.copy()
            if best_val <= 0.3808:
                break
        if best_val <= 0.3808 or time.time() - start > budget_s - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase 2 – many random hill‑climbing runs
    def hill_climb(start_vec, steps, max_step):
        """Greedy pairwise adjustments preserving the sum."""
        h = start_vec.copy()
        cur = compute_c5(h)
        for _ in range(steps):
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            delta_max = min(1.0 - hi, hj)
            delta_min = -min(hi, 1.0 - hj)
            if delta_max < delta_min:
                continue
            # restrict the magnitude to avoid huge jumps
            bound = min(max_step, delta_max, -delta_min)
            if bound <= 0:
                continue
            delta = np.random.uniform(-bound, bound)
            # ensure we stay in the feasible interval
            if delta > delta_max:
                delta = delta_max
            if delta < delta_min:
                delta = delta_min
            cand = h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            cand_val = compute_c5(cand)
            if cand_val < cur:
                h, cur = cand, cand_val
                if cur < best_val:
                    nonlocal best_h, best_val
                    best_h, best_val = h.copy(), cur
                if cur <= 0.3808:
                    return h, cur, True
        return h, cur, False

    # allocate remaining budget to hill climbing
    while time.time() - start < budget_s - 5.0:
        # random feasible start
        rnd = np.random.rand(n)
        h0 = project_onto_bounds_and_sum(rnd, target_sum)
        # a modest number of steps; step size shrinks as time runs out
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        # gradually reduce step size (max 0.1 → 0.02)
        max_step = 0.1 * (remaining / budget_s) + 0.02
        h_res, val_res, done = hill_climb(h0, steps=2000, max_step=max_step)
        if done or best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Phase 3 – fine‑tune the current best with a focused greedy search
    if best_h is not None:
        max_step = 0.05
        _, _, _ = hill_climb(best_h, steps=5000, max_step=max_step)

    # ------------------------------------------------------------------
    # final safety projection and exact evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
