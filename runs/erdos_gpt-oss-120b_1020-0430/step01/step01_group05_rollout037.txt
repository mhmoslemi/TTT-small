def gradient_worst_shift(h, k):
    n = h.shape[0]

import time, random, math
import numpy as np

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect sum(x)=target via bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5: max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """∂/∂h of the correlation term for a given shift k."""
    n = h.shape[0]
    if k >= 0:
        forward = np.empty_like(h)
        forward[: n - k] = h[k:]
        forward[n - k :] = 0.0
        backward = np.empty_like(h)
        backward[:k] = 0.0
        backward[k:] = h[: n - k]
    else:
        kp = -k
        forward = np.empty_like(h)
        forward[kp:] = h[: n - kp]
        forward[:kp] = 0.0
        backward = np.empty_like(h)
        backward[: n - kp] = h[kp:]
        backward[n - kp :] = 0.0
    return (1.0 - forward) - backward


def make_triangle(n: int) -> np.ndarray:
    """Symmetric triangle of height 1 covering [0,2]."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation to a new size."""
    n_old = old.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, old)
    return h_new


def random_feasible_h(n: int, target: float) -> np.ndarray:
    """Uniform random start then projection onto constraints."""
    x = np.random.rand(n)
    return project_onto_bounds_and_sum(x, target)


def subgradient_optimize(h_start: np.ndarray,
                         target_sum: float,
                         time_limit: float,
                         goal: float = 0.3808) -> tuple:
    """Projected sub‑gradient descent on max‑correlation objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    step0 = 0.30                     # initial step size
    top_m = 3                         # number of top shifts used for gradient
    h_cur = h_start.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    best_h = h_cur.copy()
    best_val = corr.max() * dx
    iter_cnt = 0
    stagnated = 0
    start = time.time()

    while time.time() - start < time_limit - 0.2:
        # top‑m shifts (might include the global max)
        idxs = np.argpartition(corr, -top_m)[-top_m:]
        shifts = idxs - (n - 1)

        # averaged gradient over the selected shifts
        grad = np.zeros_like(h_cur)
        for k in shifts:
            grad += gradient_for_shift(h_cur, k)
        grad /= top_m

        # diminishing step size
        step = step0 / math.sqrt(iter_cnt + 1.0)
        h_new = h_cur - step * grad
        h_new = project_onto_bounds_and_sum(h_new, target_sum)

        # evaluate new point
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        val_new = corr_new.max() * dx

        # keep the best ever seen
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
            stagnated = 0
        else:
            stagnated += 1

        # early exit if we already beat the target
        if best_val <= goal:
            break

        # accept the new point for the next iteration
        h_cur = h_new
        corr = corr_new
        iter_cnt += 1

        # occasional random restart if stuck
        if stagnated >= 2000:
            h_cur = random_feasible_h(n, target_sum)
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            stagnated = 0

    # ----- final fine‑grained pairwise tweaking -----
    while time.time() - start < time_limit - 0.2:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand
            if best_val <= goal:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Main entry point – returns (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    random.seed(seed)
    start_total = time.time()

    # ---- Phase 1 : modest resolution ----
    n1 = 256
    target1 = n1 / 2.0
    if "initial_h_values" in globals():
        try:
            h0 = upsample_initial(np.array(initial_h_values, dtype=float), n1)
            h0 = project_onto_bounds_and_sum(h0, target1)
        except Exception:
            h0 = make_triangle(n1)
    else:
        h0 = make_triangle(n1)

    # give half of the budget to the first phase
    phase1_time = budget_s * 0.5
    h_best, val_best = subgradient_optimize(h0, target1, phase1_time)

    # ---- Phase 2 : finer resolution (if time permits) ----
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > 5.0:
        n2 = 512
        target2 = n2 / 2.0
        # upscale the best solution from phase 1
        h_up = upsample_initial(h_best, n2)
        h_up = project_onto_bounds_and_sum(h_up, target2)
        # use the rest of the budget (keep a small buffer)
        h_best, val_best = subgradient_optimize(h_up, target2, remaining * 0.95)

    # final sanity check
    val_best = compute_c5(h_best)

    return (h_best.tolist(), float(val_best), int(h_best.shape[0]))
