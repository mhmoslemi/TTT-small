def greedy_refine(h):
    while True:
        corr = np.correlate(h, 1-h, mode='full')
        s = np.argmax(corr) - (len(h)-1)
        # compute indices and contributions
        if s >= 0:
            i_inds = np.arange(len(h)-s)
            p_inds = i_inds + s
        else:
            i_inds = np.arange(-s, len(h))
            p_inds = i_inds + s  # negative shift
        contrib = h[i_inds] * (1 - h[p_inds])
        k = np.argmax(contrib)
        i_best = i_inds[k]
        p_best = p_inds[k]
        eps_max = min(h[i_best], 1 - h[p_best])
        if eps_max < 1e-12:
            break
        # compute new C5 after using full eps_max
        h_new = h.copy()
        h_new[i_best] -= eps_max
        h_new[p_best] += eps_max
        new_c5 = c5_of(h_new)
        if new_c5 < c5_of(h) - 1e-12:
            h = h_new
        else:
            break
    return h
```

def random_h(n, target_sum):

    x = np.random.rand(n)
    x = np.clip(x, 0.0, 1.0)
    # scale to target sum
    s = x.sum()
    if s == 0:

    else:
        x = x * (target_sum / s)
    # ensure within [0,1]

def random_h(n, target_sum):
    x = np.random.rand(n)
    # ensure sum, if sum zero (unlikely), set uniform
    s = x.sum()
    if s == 0:
        x.fill(target_sum / n)
    else:
        x = x * (target_sum / s)
    np.clip(x, 0.0, 1.0, out=x)
    err = x.sum() - target_sum
    if abs(err) > 1e-12:
        # adjust a random position
        for _ in range(10):
            i = random.randrange(n)
            if err > 0 and x[i] >= err:
                x[i] -= err
                break
            if err < 0 and x[i] <= 1 + err:
                x[i] -= err
                break
    return x
