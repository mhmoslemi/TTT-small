def smooth_max_and_grad(h, alpha):
    corr = np.correlate(h, 1-h, mode='full')
    # compute weights
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_alpha = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_alpha)
    weights = exp_alpha / sum_exp  # shape (2n-1)
    # compute gradient: sum_i weights[i] * gradient_of_corr_i
    # pre-allocate g = np.zeros_like(h)
    g = np.zeros_like(h)
    n = h.shape[0]
    for idx, w in enumerate(weights):
        shift = idx - (n - 1)
        # gradient vector for shift
        g_shift = _gradient_for_shift(h, shift)
        g += w * g_shift
    return max_corr + np.log(sum_exp)/alpha, g

def optimize_h(h0, target_sum, max_iters=2000, alpha_schedule=None, step0=0.5, decay=0.995):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1=0.9; beta2=0.999; eps=1e-8
    step = step0
    num_alpha = len(alpha_schedule) if alpha_schedule else 1
    iter_per_alpha = max_iters // num_alpha
    for a_idx, alpha in enumerate(alpha_schedule):
        for i in range(iter_per_alpha):
            # compute smooth max and gradient
            _, grad = _smooth_max_and_grad(h, alpha)
            # Adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad**2)
            m_hat = m/(1-beta1**(i+1))
            v_hat = v/(1-beta2**(i+1))
            h = h - step * m_hat / (np.sqrt(v_hat) + eps)
            # projection
            h = _project_onto_capped_simplex(h, target_sum)
        # maybe reduce step after each alpha
        step *= decay

    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ---------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ---------------------------------------------------------
# Gradient of a single correlation shift
# ---------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ---------------------------------------------------------
# Projected (sub)gradient descent on the true max objective
# ---------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 600, step0: float = 0.6,
                 decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

# Smooth max (log-sum-exp) and its gradient (vectorized)
# ---------------------------------------------------------
def _smooth_max_and_grad(h: np.ndarray, alpha: float):
    # correlation without dx
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = corr - max_corr          # max_corr becomes 0 after shift
    exp_alpha = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_alpha)
    weights = exp_alpha / sum_exp     # sum_i w_i = 1

    n = h.shape[0]
    grad = np.ones(n)                 # base contribution from every shift
    w0 = weights[n - 1]               # shift = 0
    grad -= 2.0 * w0 * h

    # contributions from symmetric pairs of shifts
    for d in range(1, n):
        wd = weights[n - 1 + d] + weights[n - 1 - d]
        if wd == 0.0:
            continue
        # subtract h shifted left by d
        grad[: n - d] -= wd * h[d:]
        # subtract h shifted right by d
        grad[d:] -= wd * h[: n - d]

    smooth_val = max_corr + np.log(sum_exp) / alpha
    return smooth_val, grad

def _smooth_optimize(h0: np.ndarray, target: float,
                     alpha_schedule, n_iter_per_alpha: int = 120,
                     step0: float = 0.5, decay: float = 0.998):
    h = h0.copy()
    step = step0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    for alpha in alpha_schedule:
        for _ in range(n_iter_per_alpha):
            t += 1
            _, grad = _smooth_max_and_grad(h, alpha)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h -= step * m_hat / (np.sqrt(v_hat) + eps)
            h = _project_onto_capped_simplex(h, target)
            step *= decay
    return h


# ---------------------------------------------------------
# Optional SLSQP refinement (smooth max as surrogate)
# ---------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 800) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exp_alpha = np.exp(alpha * shifted)
        sum_exp = np.exp(alpha * m) * np.sum(exp_alpha)   # re‑scale for stability
        return m + np.log(sum_exp) / alpha

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ---------------------------------------------------------
# Main entry point
# ---------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # Determine discretisation size
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None
    n = len(base_h) if base_h is not None else 200
    target_sum = n / 2.0
    dx = 2.0 / n

    # Evaluation of the exact C5 value
    def eval_c5(h):
        return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

    # Initial best solution
    if base_h is not None:
        best_h = _project_onto_capped_simplex(base_h, target_sum)
        best_val = eval_c5(best_h)
    else:
        best_h = None
        best_val = float('inf')

    # Optimisation hyper‑parameters
    alpha_sched = [5, 12, 25, 50, 100, 150]      # increasing smoothness
    iters_alpha = 110                           # per‑alpha iterations
    step0 = 0.5
    decay = 0.998

    restart = 0
    while time.time() - start < budget_s * 0.92:
        # ---------- generate starting point ----------
        if best_h is not None and rng.rand() < 0.22:
            h0 = best_h + rng.normal(scale=0.025, size=n)
        else:
            if rng.rand() < 0.35:
                # binary feasible vector (exactly target_sum ones)
                h0 = np.zeros(n)
                idx = rng.choice(n, int(target_sum), replace=False)
                h0[idx] = 1.0
            else:
                h0 = _random_feasible(rng, n, target_sum)
        h0 = _project_onto_capped_simplex(h0, target_sum)

        # ---------- smooth‑max optimisation ----------
        h_opt = _smooth_optimize(h0, target_sum,
                                 alpha_schedule=alpha_sched,
                                 n_iter_per_alpha=iters_alpha,
                                 step0=step0, decay=decay)

        # ---------- final refinement ----------
        h_opt = _gd_optimize(h_opt, target_sum,
                             max_iter=300, step0=0.2, decay=0.996)
        if restart % 4 == 0:
            try:
                h_opt = _slsqp_optimize(h_opt, target_sum,
                                        alpha=80.0, max_iter=500)
            except Exception:
                pass

        val = eval_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        restart += 1

    # Safety fallback
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = eval_c5(best_h)
    else:
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = eval_c5(best_h)

    return (best_h, float(best_val), n)
