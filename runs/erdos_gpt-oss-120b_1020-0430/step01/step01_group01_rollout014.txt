def _neighbor_move(h, rng, max_transfer=0.05):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # Determine max amount we can transfer from i to j while staying in [0,1]
    max_plus = min(h[i], 1 - h[j])  # amount we can move from i to j (negative delta for i)
    max_minus = min(h[j], 1 - h[i])  # amount we can move from j to i (positive delta for i)
    # choose direction randomly:
    if rng.rand() < 0.5:
        max_transfer_dir = max_plus
        if max_transfer_dir > 0:
            delta = rng.uniform(0, min(max_transfer_dir, max_transfer))
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            return h_new
    else:
        max_transfer_dir = max_minus
        if max_transfer_dir > 0:
            delta = rng.uniform(0, min(max_transfer_dir, max_transfer))
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            return h_new
    # fallback: nothing
    return h.copy()

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# --------------------------------------------------------------
# Random feasible vector (uniform then project)
# --------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# --------------------------------------------------------------
# Correlation based C5 evaluation
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


# --------------------------------------------------------------
# Gradient of single‑shift correlation (subgradient of max)
# --------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# --------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient
# --------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _smooth_max_gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    # stable softmax weights
    max_val = np.max(alpha * corr)
    w = np.exp(alpha * corr - max_val)
    w /= w.sum()
    grad = np.zeros_like(h)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx - (n - 1)
        grad += weight * _gradient_for_shift(h, shift)
    return grad


# --------------------------------------------------------------
# Simple projected gradient descent on the (non‑smooth) max
# --------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 1500, step0: float = 0.8,
                 decay: float = 0.997) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# --------------------------------------------------------------
# Momentum projected gradient descent
# --------------------------------------------------------------
def _momentum_gd_optimize(h0: np.ndarray, target: float,
                          max_iter: int = 1500, step0: float = 0.6,
                          decay: float = 0.997, mu: float = 0.8) -> np.ndarray:
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        v = mu * v - step * g
        h = h + v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# --------------------------------------------------------------
# Gradient descent on a smoothed max (log‑sum‑exp)
# --------------------------------------------------------------
def _smooth_gd_optimize(h0: np.ndarray, target: float,
                        alpha: float = 30.0,
                        max_iter: int = 400,
                        step0: float = 0.5,
                        decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    for _ in range(max_iter):
        g = _smooth_max_gradient(h, alpha)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# --------------------------------------------------------------
# Local refinement with SLSQP on the smooth max
# --------------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 40.0,
                    max_iter: int = 500) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10,
                            'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# --------------------------------------------------------------
# Simple simulated‑annealing neighbourhood move (pairwise transfer)
# --------------------------------------------------------------
def _neighbor_move(h: np.ndarray, rng: np.random.RandomState,
                   max_transfer: float = 0.12) -> np.ndarray:
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # direction: move from i to j or opposite
    if rng.rand() < 0.5:
        max_put = min(h[i], 1.0 - h[j])
        delta = rng.uniform(0.0, min(max_put, max_transfer))
        h2 = h.copy()
        h2[i] -= delta
        h2[j] += delta
    else:
        max_put = min(h[j], 1.0 - h[i])
        delta = rng.uniform(0.0, min(max_put, max_transfer))
        h2 = h.copy()
        h2[i] += delta
        h2[j] -= delta
    return h2


# --------------------------------------------------------------
# Simulated annealing using the neighbourhood above
# --------------------------------------------------------------
def _simulated_annealing(start_h: np.ndarray, target: float,
                         budget_s: float, start_time: float):
    rng = np.random.RandomState(123456)
    h_cur = start_h.copy()
    cur_val = _compute_c5(h_cur, 2.0 / len(start_h))
    best_h, best_val = h_cur, cur_val
    T = 0.05  # initial temperature
    decay = 0.9997
    while time.time() - start_time < budget_s * 0.98:
        h_new = _neighbor_move(h_cur, rng)
        val_new = _compute_c5(h_new, 2.0 / len(start_h))
        if val_new < cur_val or rng.rand() < np.exp(-(val_new - cur_val) / max(T, 1e-12)):
            h_cur, cur_val = h_new, val_new
            if val_new < best_val:
                best_h, best_val = h_new, val_new
        T *= decay
    return best_h, best_val
