def project_bounds_sum(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Compute current sum
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    # If s < target, need to distribute additional mass to variables not at 1
    if s < target:
        # water-filling: compute capacity to fill for each entry
        # fill proportionally until sum = target or all at 1
        deficit = target - s
        idx = np.where(x < 1.0)[0]

            x[idx] += addition

def _gradient_step(h, target_sum, beta, eta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    w = np.exp(beta * (corr - cmax))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # compute left and right contributions
    left = np.zeros(n)
    right = np.zeros(n)
    # lags: d = index - (n-1)
    for idx, d in enumerate(range(-(n-1), n)):
        wd = w[idx]
        if wd == 0.0:
            continue
        if d >= 0:
            # i from 0 to n-d-1: i+d in range
            left[:n-d] += wd * h[d:]
            # i from d to n-1: i-d in range
            right[d:] += wd * h[:n-d]
        else:
            dd = -d
            # i from dd to n-1: i+d = i-dd in range
            left[dd:] += wd * h[:n-dd]
            # i from 0 to n-dd-1: i-d = i+dd in range
            right[:n-dd] += wd * h[dd:]
    grad = np.ones(n) - left - right
    h_new = h - eta * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _adjust_sum(h_new, target_sum)
    return h_new

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the hyper‑plane sum(x)=target while respecting
    the box constraints 0 ≤ x ≤ 1.
    Simple water‑filling redistribution.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x

    # iterative redistribution among free variables
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        candidates = np.where(mask)[0]
        per = diff / len(candidates)
        for i in candidates:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _gradient_step(h: np.ndarray, target: float,
                   beta: float, eta: float) -> np.ndarray:
    """
    One smoothed gradient descent step.
    - beta controls the focus on the current maximum lag.
    - eta is the step size.
    """
    n = h.size
    # raw correlation (without dx)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()
    # smooth weights, larger for larger correlation entries
    w = np.exp(beta * (corr - cmax))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # weighted sums of shifted values
    left = np.zeros(n)
    right = np.zeros(n)
    # d runs from -(n-1) .. (n-1)
    for idx, d in enumerate(range(-(n - 1), n)):
        wd = w[idx]
        if wd == 0.0:
            continue
        if d >= 0:
            # i+d is inside for i = 0 .. n-d-1
            left[: n - d] += wd * h[d:]
            # i-d is inside for i = d .. n-1
            right[d:] += wd * h[: n - d]
        else:
            dd = -d
            # i+d (i - dd) inside for i = dd .. n-1
            left[dd:] += wd * h[: n - dd]
            # i-d (i + dd) inside for i = 0 .. n-dd-1
            right[: n - dd] += wd * h[dd:]

    grad = np.ones(n) - left - right
    h_new = h - eta * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _adjust_sum(h_new, target)
    return h_new

# Main optimisation routine
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function that gives a low upper bound on C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # a few resolutions; larger n gives finer control but costs a bit more time
    candidate_ns = [45, 63, 81, 99, 117, 135]

    for n in candidate_ns:
        target = n / 2.0
        # initialise
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h = np.asarray(initial_h_values, dtype=np.float64).copy()
            h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
        else:
            # random start with correct total mass
            h = rng.random(n)
            h = h * (target / h.sum())
            h = _adjust_sum(h, target)

        current_c5 = _evaluate_c5(h)

        # keep the best seen so far
        if current_c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), current_c5, n

        # ----------------------------------------------------------------
        # optimisation loop for this resolution
        # ----------------------------------------------------------------
        # parameters for the inner loop
        max_iter = 10_000_000  # effectively limited by time
        iter_cnt = 0
        # temperature‑like schedule for occasional uphill moves
        temp_start = 5e-3
        temp_end = 1e-6

        while time.time() - start_time < budget_s - 0.4:
            # linear temperature decay
            elapsed = time.time() - start_time
            T = temp_start + (temp_end - temp_start) * (elapsed / budget_s)

            # ---- gradient descent step ----
            beta = 5.0 + rng.random() * 20.0          # 5 … 25
            eta = 1e-3 + rng.random() * 0.02          # 0.001 … 0.021
            h_new = _gradient_step(h, target, beta, eta)
            new_c5 = _evaluate_c5(h_new)

            Δ = new_c5 - current_c5
            if Δ < -1e-12 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
                h, current_c5 = h_new, new_c5
                if new_c5 < best_c5:
                    best_h, best_c5, best_n = h_new.copy(), new_c5, n

            # ---- random pairwise mass exchange (hill climbing) ----
            i, j = rng.integers(0, n, size=2)
            max_move = min(h[i], 1.0 - h[j])
            if max_move > 1e-12:
                delta = max_move * rng.random()
                h2 = h.copy()
                h2[i] -= delta
                h2[j] += delta
                c5_2 = _evaluate_c5(h2)
                if c5_2 < current_c5 - 1e-12 or rng.random() < np.exp(-(c5_2 - current_c5) / max(T, 1e-12)):
                    h, current_c5 = h2, c5_2
                    if c5_2 < best_c5:
                        best_h, best_c5, best_n = h2.copy(), c5_2, n

            # ---- small random jitter to keep exploration alive ----
            if rng.random() < 0.02:
                h += (rng.random(n) - 0.5) * 0.01
                h = np.clip(h, 0.0, 1.0)
                h = _adjust_sum(h, target)
                current_c5 = _evaluate_c5(h)
                if current_c5 < best_c5:
                    best_h, best_c5, best_n = h.copy(), current_c5, n

            iter_cnt += 1
            # safety break if too many pure iterations (unlikely)
            if iter_cnt > max_iter:
                break

        # ----------------------------------------------------------------
        # early exit if time is almost exhausted
        # ----------------------------------------------------------------
        if time.time() - start_time > budget_s - 0.4:
            break

    # final feasibility sanitisation
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    best_c5 = float(_evaluate_c5(best_h))

    return best_h, best_c5, best_n
