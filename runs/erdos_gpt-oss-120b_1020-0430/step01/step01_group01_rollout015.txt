def _gradient_for_shift(h, shift):
   n = len(h)
   g = np.empty_like(h)
   if shift >= 0:
       s = shift
       g[: n - s] = 1.0 - h[s:]
       g[n - s :] = 1.0
       g[s:] += -h[: n - s]
   else:
       s = -shift
       g[s:] = 1.0 - h[: n - s]
       g[: s] = 1.0
       g[: n - s] += -h[s:]
   return g

import time, numpy as np, scipy.optimize as opt

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _corr_raw(h):
    n = h.shape[0]
    N = 2 * n - 1
    fh = np.fft.rfft(h, n=N)
    fg = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(fh * np.conj(fg), n=N)
    return corr

def _compute_c5(h, dx):
    corr = _corr_raw(h) * dx
    return float(corr.max())

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _adam_optimize(h0, target, max_iter=1200, step=0.05,
                   beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = h.shape[0]
    for t in range(1, max_iter + 1):
        corr = _corr_raw(h)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        grad = _gradient_for_shift(h, shift)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h = _project_onto_capped_simplex(h, target)
    return h

    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
