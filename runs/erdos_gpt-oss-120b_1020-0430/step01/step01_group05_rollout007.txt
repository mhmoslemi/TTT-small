    # generate random delta modifications
    i, j = np.random.randint(0, n, size=2)
    if i == j: continue
    delta = (np.random.rand() - 0.5) * 0.1  # small +/-0.05
    # adjust hi and hj to keep sum: hi' = hi + delta, hj' = hj - delta
    hi, hj = best_h[i], best_h[j]

import time, random, math
import numpy as np
from scipy.optimize import minimize

# ----- global variables for constraints and objective -----
TARGET_SUM_GLOBAL = None
DX_GLOBAL = None
ALPHA_GLOBAL = None

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect sum(x)=target using bisection on lambda."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * DX_GLOBAL

def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Smooth max via log-sum-exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx

def objective_global(x: np.ndarray) -> float:
    """Wrapper using globals for minimize."""
    return objective_smooth(x, DX_GLOBAL, ALPHA_GLOBAL)

def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - TARGET_SUM_GLOBAL

def make_triangle(n: int) -> np.ndarray:
    """Triangular peak at centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(pi x/2)."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_inv_triangle(n: int) -> np.ndarray:
    """Inverted triangular shape (valley)."""
    tri = make_triangle(n)
    inv = 1.0 - tri
    return project_onto_bounds_and_sum(inv, n / 2.0)

def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear up‑sampling of a parent vector."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with low C5."""
    global TARGET_SUM_GLOBAL, DX_GLOBAL, ALPHA_GLOBAL
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # problem size (feel free to adjust)
    n = 300                     # even for exact half‑mass
    dx = 2.0 / n
    target_sum = n / 2.0
    DX_GLOBAL = dx
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ----- build initial seeds -----
    seed_vectors = []
    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))
    seed_vectors.append(make_inv_triangle(n))
    # a few random feasible starts
    for _ in range(4):
        rnd = np.random.rand(n)
        seed_vectors.append(project_onto_bounds_and_sum(rnd, target_sum))

    best_h = None
    best_val = math.inf

    # ----- progressive smooth‑max SLSQP optimisation -----
    alphas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    constraints = {"type": "eq", "fun": sum_eq_constraint}
    for h0 in seed_vectors:
        if time.time() - start > budget_s - 8.0:
            break
        h_cur = h0.copy()
        for a in alphas:
            if time.time() - start > budget_s - 8.0:
                break
            ALPHA_GLOBAL = a
            try:
                res = minimize(
                    objective_global,
                    h_cur,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": 200, "ftol": 1e-9, "disp": False},
                )
                h_cur = project_onto_bounds_and_sum(res.x, target_sum)
            except Exception:
                continue
        val = compute_c5(h_cur)
        if val < best_val:
            best_val = val
            best_h = h_cur.copy()
        if best_val <= 0.3808:
            break

    # ----- local pairwise hill‑climbing refinement -----
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    while time.time() - start < budget_s - 2.0:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        # maximum admissible delta preserving bounds
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cval = compute_c5(cand)
        if cval < best_val:
            best_val = cval
            best_h = cand.copy()
            if best_val <= 0.3808:
                break

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
