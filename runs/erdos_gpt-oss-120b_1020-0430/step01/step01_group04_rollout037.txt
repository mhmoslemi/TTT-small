def compute_c5_fft(h):
    n = h.shape[0]
    nfull = 2*n - 1
    a = h
    b = 1 - h
    # reversed b
    br = b[::-1]
    A = np.fft.rfft(a, nfull)
    B = np.fft.rfft(br, nfull)
    conv = np.fft.irfft(A * B, nfull)
    # conv corresponds to correlation values k = -(n-1) .. (n-1) (i.e., index offset n-1)
    # compute max
    max_val = np.max(conv)
    dx = 2.0 / n
    return max_val * dx
```

def compute_c5(h):
    n = h.shape[0]
    nfull = 2*n - 1
    a = h.astype(np.float64)
    b = 1.0 - a
    br = b[::-1]
    A = np.fft.rfft(a, nfull)
    B = np.fft.rfft(br, nfull)
    conv = np.fft.irfft(A * B, nfull)
    max_val = conv.max()
    return max_val * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]
    nfull = 2*n - 1
    a = h.astype(np.float64)
    b = 1.0 - a
    br = b[::-1]
    A = np.fft.rfft(a, nfull)
    B = np.fft.rfft(br, nfull)
    conv = np.fft.irfft(A * B, nfull)
    c5 = float(np.max(conv) * (2.0 / n))
    return c5
```

def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(alpha * conv) / alpha)

def generate_triangular(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0/n
    tri = np.minimum(x, 2.0 - x)
    # Normalize area to 1
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0/old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# Core utilities

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function using full correlation."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / n))


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth max via log‑sum‑exp (used for local refinement)."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular h(x)=min(x,2‑x) sampled on n points and normalized."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1] and exact integral)."""
    a = rng.random(n)

    old_n = arr.shape[0]

def simulated_annealing(h0: np.ndarray, deadline: float, rng: np.random.Generator):
    n = h0.shape[0]
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    log_T0 = np.log(0.05)   # start temperature
    log_Tend = np.log(1e-6)  # final temperature

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        T = np.exp(log_T0 + (log_Tend - log_T0) * frac)
        eps = 0.2 * (1.0 - frac) + 0.001 * frac  # max delta size

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_pos = min(eps, 1.0 - h[i], h[j])
        max_neg = min(eps, h[i], 1.0 - h[j])
        if max_pos + max_neg <= 0:
