def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx
```

```
def simulated_annealing(h_init, time_deadline):
    n = len(h_init)
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng()
    # parameters
    max_iter = 1000000  # we break by time
    iter = 0
    # cooling schedule: T = T0 * (0.999**iter) maybe
    T0 = 0.01  # starting temperature for objective difference (C5 scale)
    # step scaling
    step_scale = 0.1    # max delta weight proportion of 1 ( = 0.1)
    while time.time() < time_deadline and iter < max_iter:
        iter += 1
        # temperature schedule
        T = T0 * (0.999 ** iter)  # decays exponentially
        # also step size decays: step = step_scale * max(0.01, T/T0)
        step = step_scale * max(0.01, T / T0)  # as T decays, step decreases
        # Randomly choose i where cur[i] < 1; j where cur[j] > 0
        i_candidates = np.where(cur < 1 - 1e-12)[0]
        j_candidates = np.where(cur > 1e-12)[0]
        if len(i_candidates) == 0 or len(j_candidates) == 0:
            break
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)
        # compute max delta allowed
        max_delta = min(step, 1 - cur[i], cur[j])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0, max_delta)
        # apply swap
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        # evaluate
        new_c5 = compute_c5(new)
        # accept
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur = new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
    return best, best_c5
```

def binaryize(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(-h)  # descending
    bin_vec = np.zeros(n)
    bin_vec[idx[:k]] = 1.0
    return bin_vec

    best = cur.copy()

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 40.0) -> float:
    """Smooth approximation of the max overlap (log‑sum‑exp)."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1] and sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def project_onto_box_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0≤x≤1, sum(x)=target}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0  # enough to push all entries to 0 or 1
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # final tiny correction if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x


def worst_shift_and_gradient(h: np.ndarray):
    """Return the shift producing the maximal overlap and its gradient."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)                     # signed shift
    # gradient of the worst term f_k(h) = Σ h_i (1‑h_{i+shift})
    g = np.ones(n)
    if shift > 0:
        i = np.arange(n - shift)
        g[i] -= h[i + shift]                # (1‑h_{i+shift}) part
        i2 = np.arange(shift, n)
        g[i2] -= h[i2 - shift]              # -h_{i‑shift} part
    elif shift < 0:
        s = -shift
        i = np.arange(s, n)
        g[i] -= h[i - s]
        i2 = np.arange(0, n - s)
        g[i2] -= h[i2 + s]
    else:
        g = 1.0 - 2.0 * h                     # shift = 0
    return shift, g, conv.max() * (2.0 / n)


def gradient_descent_start(h0: np.ndarray, target: float, time_deadline: float):
    """
    Fast projected gradient descent on the worst‑overlap term.
    Returns the best vector discovered and its C5 value.
    """
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.5
    it = 0
    while time.time() < time_deadline:
        it += 1
        shift, g, cur_c5 = worst_shift_and_gradient(h)
        # simple step decay
        if it % 2000 == 0:
            step *= 0.8
        h_new = h - step * g
        h_new = project_onto_box_and_sum(h_new, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-9:
            best_h, best_c5 = h_new.copy(), new_c5
        # accept if not worse (greedy); occasional random walk could be added
        if new_c5 <= cur_c5:
            h = h_new
        else:
            # occasional small random move to escape plateaus
            if np.random.rand() < 0.001:
                eps = np.random.uniform(-0.01, 0.01, size=n)
                h = project_onto_box_and_sum(h + eps, target)
    return best_h, best_c5
