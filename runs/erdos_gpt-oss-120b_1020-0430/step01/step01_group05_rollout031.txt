import time, random, math
import numpy as np

# ----- helper functions -----------------------------------------------------

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect {sum = target} using bisection."""
    lo, hi = -2.0, 2.0               # wide enough to bracket the solution
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def correlation_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) using FFT (full length)."""
    n = h.shape[0]
    size = np.fft.next_fast_len(2 * n - 1)
    A = np.fft.rfft(h, n=size)
    B = np.fft.rfft((1.0 - h)[::-1], n=size)
    conv = np.fft.irfft(A * B, n=size)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx  (discretised)."""
    conv = correlation_fft(h)
    dx = 2.0 / h.shape[0]
    return conv.max() * dx


def shift_array(arr: np.ndarray, offset: int) -> np.ndarray:
    """Return arr shifted by offset (zero‑filled, non‑periodic)."""
    if offset == 0:
        return arr
    n = arr.shape[0]
    out = np.empty_like(arr)
    if offset > 0:
        out[: n - offset] = arr[offset:]
        out[n - offset :] = 0.0
    else:
        off = -offset
        out[off:] = arr[: n - off]
        out[:off] = 0.0
    return out


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n, then project."""
    old_n = init.shape[0]
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin_pow(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """h(x)=sin(π(x+shift)/2)^p on [0,2] (clipped to [0,1] and projected)."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * (x + shift) / 2.0) ** p
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_plateau(n: int, b: float) -> np.ndarray:
    """
    Piecewise shape:
        a = 1-b   (half‑width of the top plateau)
        for |x-1| ≤ a          : h = 1
        for a < |x-1| < b      : linear drop to 0
        otherwise               : h = 0
    Then projected to satisfy the integral constraint.
    """
    a = 1.0 - b
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    d = np.abs(x - 1.0)
    h = np.where(d <= a, 1.0,
                 np.where(d < b, (b - d) / (b - a), 0.0))
    return project_onto_bounds_and_sum(h, n / 2.0)

                    break

    # final safety projection (ensures exact sum and bounds)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
