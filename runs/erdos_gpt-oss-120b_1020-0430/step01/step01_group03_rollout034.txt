def targeted_descent(h, target, max_iters, init_step, rng, start_time, time_limit):
    n = h.size
    best_h = h.copy()
    corr, max_corr, _ = compute_corr_max(h)
    best_val = max_corr * (2.0 / n)
    cur_h = h.copy()
    cur_val = best_val
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # recompute corr each iteration
        corr, max_corr, max_idx = compute_corr_max(cur_h)
        cur_val = max_corr * (2.0 / n)
        shift = max_idx - (n - 1)
        grad = gradient_shift(cur_h, shift)
        # find positive and negative indices
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            # random perturb if gradient not informative
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_transfer = min(cur_h[i], 1 - cur_h[j])
            if max_transfer <= 0:
                continue
            eps = min(init_step, max_transfer) * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_transfer = min(cur_h[i], 1 - cur_h[j])
            if max_transfer <= 0:
                continue
            eps = min(init_step, max_transfer) * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        # Ensure feasibility: already preserved sum and bound through eps
        # Compute new value
        new_corr, new_max_corr, _ = compute_corr_max(new_h)
        new_val = new_max_corr * (2.0 / n)
        # Simple accept if improvement or with probability based on temperature
        # temperature schedule linear
        T = max(0.0, 0.5 * (1 - it / max_iters))
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # accept with prob exp(-(new-val)/T) if T>0
            if T > 1e-12:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
        # optionally decrease step size over time (cooling)
        # we can gradually reduce init_step
        if (it + 1) % (max_iters // 10 + 1) == 0:
            init_step = max(init_step * 0.9, 1e-4)
    return best_h, best_val

def targeted_descent(h, target, max_iters, step, rng, deadline):
    n = h.size
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_h = h.copy()
    cur_val = best_val
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute overlap and shift
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        cur_val = max_corr * (2.0 / n)
        # optionally update best
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
        shift = max_idx - (n - 1)
        grad = gradient_shift(cur_h, shift)
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size > 0 and neg.size > 0:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            eps = min(step, max_transfer) * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # random transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            eps = min(step, max_transfer) * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        # evaluation
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_max_corr = np.max(new_corr)
        new_val = new_max_corr * (2.0 / n)
        # acceptance: deterministic improvement or SA
        # temperature schedule decreasing linearly
        T = max(0.0, 0.1 * (1 - it / max_iters))  # small temperature
        accept = (new_val < cur_val) or (T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T))

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap for a single shift k."""
    n = h.size
    if shift >= 0:
        # term1 = (1 - h_{i+shift})  (zero where i+shift out of range)
        term1 = np.concatenate([1 - h[shift:], np.zeros(shift)])
        # term2 = h_{i-shift}        (zero where i-shift out of range)
        term2 = np.concatenate([np.zeros(shift), h[: n - shift]])
    else:
        d = -shift
        term1 = np.concatenate([np.zeros(d), 1 - h[: n - d]])
        term2 = np.concatenate([h[d:], np.zeros(d)])
    return term1 - term2


def project_onto_feasible(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to exactly `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary feasible vector with exactly n/2 entries equal to 1."""
    h = np.zeros(n, dtype=float)
    indices = rng.choice(n, size=n // 2, replace=False)
    h[indices] = 1.0
    return h


def targeted_descent(
    h: np.ndarray,
    target: float,
    max_iters: int,
    step: float,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent focused on the currently worst shift.
    Returns the best vector encountered and its C5 value.
    """
    n = h.size
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # compute current worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        cur_val = max_corr * (2.0 / n)

        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()

        shift = max_idx - (n - 1)
        grad = gradient_shift(cur_h, shift)

        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]

        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            eps = min(step, max_transfer) * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # random transfer when gradient gives no direction
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            eps = min(step, max_transfer) * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        # evaluate the proposal
        new_corr = np.correlate(new_h, 1 - new_h, mode="full")
        new_val = np.max(new_corr) * (2.0 / n)

        # simple simulated‑annealing acceptance
        T = max(0.0, 0.1 * (1.0 - it / max_iters))
        accept = (new_val < cur_val) or (T > 0.0 and rng.random() < np.exp(-(new_val - cur_val) / T))
        if accept:
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

        # gradual step‑size decay
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.9, 1e-5)

    best_h = project_onto_feasible(best_h, target, rng)
    best_val = compute_c5(best_h)
    return best_h, best_val

    """

    start_time = time.time()
    total_deadline = start_time + budget_s

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    target = n / 2.0

    best_h = project_onto_feasible(best_h, target, rng)
    final_c = compute_c5(best_h)
