def shape_from_p(p, n):

def project_onto_bounds_and_sum(x, target):
    # lambda bisection to adjust.
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)
```

import time
import random
import numpy as np

# ---------- helper functions ----------

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the box [0,1]^n while enforcing sum(x)=target.
    Uses bisection on the Lagrange multiplier for the equality constraint.
    """
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised objective: max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def shape_from_params(a: float, p: float, n: int) -> np.ndarray:
    """
    Piecewise‑linear shape:
        - plateau of height a on |x‑1| ≤ p
        - linear drop to 0 on p < |x‑1| ≤ p+q,
          where q = 1/a – 2p (so that total area = 1)
    """
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    t = np.abs(xs - 1.0)

    s = 1.0 / a                     # total weighted length (2p+q)
    q = s - 2.0 * p                 # tail length per side

    h = np.zeros_like(xs)

    # plateau
    mask_plateau = t <= p
    h[mask_plateau] = a

    # tails (if any)
    if q > 1e-12:
        mask_tail = (t > p) & (t <= p + q)
        # linear interpolation from a at t=p to 0 at t=p+q
        h[mask_tail] = a * (p + q - t[mask_tail]) / q
    # else q≈0 → pure plateau (already set)

    return np.clip(h, 0.0, 1.0)

            h = shape_from_params(a, p, n)
            h = project_onto_bounds_and_sum(h, target_sum)
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        # early stop if record already beaten
        if best_val <= 0.3808:
            break
        if time.time() - start > budget_s - 5.0:
            break

    # fall back to simple triangle if nothing found
    if best_h is None:
        best_h = shape_from_params(1.0, 0.0, n)   # pure triangle
        best_h = project_onto_bounds_and_sum(best_h, target_sum)
        best_val = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    # ---- targeted simulated‑annealing style refinement ----
    T = 0.02
    decay = 0.9998
    it = 0
    while time.time() - start < budget_s - 0.5:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)                     # shift producing the current max
        # range of indices that actually overlap for this shift
        if shift >= 0:
            i_range = np.arange(0, n - shift)
        else:
            i_range = np.arange(-shift, n)

        i_vals = cur_h[i_range]
        j_vals = cur_h[i_range + shift]
        contrib = i_vals * (1.0 - j_vals)              # contribution to the max term

        moved = False
        if contrib.size > 0:
            idx = int(np.argmax(contrib))
            i = int(i_range[idx])
            j = i + shift
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max > 1e-12:
                delta = delta_max * random.random()
                cand = cur_h.copy()
                cand[i] -= delta
                cand[j] += delta
                moved = True

        if not moved:
            # generic random transfer as a fallback
            i, j = random.sample(range(n), 2)
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max > 1e-12:
                delta = delta_max * random.random()
                cand = cur_h.copy()
                cand[i] -= delta
                cand[j] += delta
                moved = True

        if not moved:
            break

        cand = np.clip(cand, 0.0, 1.0)               # stay in bounds
        cand_val = compute_c5(cand)

        # Metropolis acceptance
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = np.exp((cur_val - cand_val) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                it = 0   # reset stagnation counter
        else:
            it += 1

        # temperature schedule and occasional re‑projection
        T *= decay
        if it % 2000 == 0:
            cur_h = project_onto_bounds_and_sum(cur_h, target_sum)
            cur_val = compute_c5(cur_h)

        if best_val <= 0.3808:
            break

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
