def propose(h):
    choice = random.random()
    if choice < 0.7:  # simple two-coordinate move (same as previous)
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        # pick delta within allowed range
        max_up = min(1.0 - hi, hj)
        max_down = min(hi, 1.0 - hj)
        if max_up + max_down == 0: return h
        delta = random.uniform(-max_down, max_up)
        nh = h.copy()
        nh[i] = hi + delta
        nh[j] = hj - delta
        return nh
    else:  # gradient move
        # compute worst shift and gradient
        corr_full = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr_full)
        shift = idx_max - (n - 1)
        # compute gradient
        g = np.empty(n)
        # compute contributions
        # vectorized:
        # components from (1 - h[i+shift]) where i+shift < n else 1.
        # Use np.arange to compute indices
        i = np.arange(n)
        i_plus = i + shift
        g = np.where(i_plus < n, 1.0 - h[i_plus], 1.0)
        i_minus = i - shift
        g -= np.where(i_minus >= 0, h[i_minus], 0.0)
        # step magnitude = epsilon * T
        eps = (random.random() * 0.5 + 0.5) * T  # maybe variable
        nh = h - eps * g
        # project onto bounds and sum
        nh = np.clip(nh, 0.0, 1.0)
        # adjust sum might violate target
        nh = project_sum_and_bounds(nh, target_sum)
        return nh

    it = 0
    last_improve_iter = 0

    while time.time() - start < budget_s * 0.95:

        if random.random() < 0.6:
            # random swap move
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]

            delta = random.uniform(-max_down, max_up)
            if abs(delta) < 1e-15:

            nh[i] = hi + delta
            nh[j] = hj - delta
        else:

            corr_full = np.correlate(h, 1 - h, mode='full')
            idx_max = np.argmax(corr_full)

            ip = i + shift

            im = i - shift

            g = inc

            nh = h - eps * g

            nh = project_sum_and_bounds(nh, target_sum)

        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):

                best_h = h.copy()

def project_sum_and_bounds(x, target):
    # Clip first
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    diff = s - target
    if abs(diff) < 1e-12:
        return x
    # Determine free indices (not at bound)
    # We iteratively adjust
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # get indices where we can adjust in needed direction
        if diff > 0:
            # need to reduce sum: can decrease values > 0
            candidates = np.where(x > 1e-12)[0]
            if len(candidates) == 0:
                break
            # amount to subtract proportionally (or equally)
            # subtract min(diff/len(cand), x[i])
            # We'll loop maybe random to avoid overflow
            # Simple: for each i in candidates:
            #   dec = min(x[i], diff / len(candidates))
            # more efficient:
            dec = diff / len(candidates)
            # ensure not exceed each x[i]
            max_dec = x[candidates].min()
            dec = min(dec, max_dec)
            x[candidates] -= dec
            diff -= dec * len(candidates)
        else:
            # need to increase sum
            candidates = np.where(x < 1 - 1e-12)[0]
            if len(candidates) == 0:
                break
            inc = -diff / len(candidates)
            max_inc = (1 - x[candidates]).min()
            inc = min(inc, max_inc)
            x[candidates] += inc
            diff += inc * len(candidates)
    # Final tweak if still diff small
    if abs(x.sum() - target) > 1e-9:
        # distribute leftover to first free index
        diff = x.sum() - target
        if diff > 0:
            idx = np.where(x > 1e-12)[0][0]
            x[idx] -= diff
        else:
            idx = np.where(x < 1 - 1e-12)[0][0]
            x[idx] -= diff  # diff negative => addition
    return x

import numpy as np, random, time, math

def c5_of(v):
    """Maximum overlap integral for a discretised step function."""
    # full correlation of h with (1-h)
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(corr.max() * (2.0 / len(v)))

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and then adjust to hit the exact sum `target`."""
    x = np.clip(x, 0.0, 1.0)
    diff = x.sum() - target
    # Simple random walk correction – fast enough for tiny residuals.
    max_iter = 20000
    while abs(diff) > 1e-12 and max_iter:
        if diff > 0.0:
            cand = np.where(x > 1e-12)[0]
            if cand.size == 0:
                break
            i = random.choice(cand.tolist())
            delta = min(x[i], diff)
            x[i] -= delta
            diff -= delta
        else:
            cand = np.where(x < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = random.choice(cand.tolist())
            delta = min(1.0 - x[i], -diff)
            x[i] += delta
            diff += delta
        max_iter -= 1
    return x

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise vector size – use initial_h_values if available, else 100.
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    except Exception:
        n = 100
        init = None

    target_sum = n / 2.0

    if init is not None and init.shape[0] == n:
        h = init.copy()
    else:
        # random binary start
        h = np.zeros(n)
        idx = np.arange(n)
        random.shuffle(idx.tolist())
        h[idx[: n // 2]] = 1.0

    h = project_sum_and_bounds(h, target_sum)
    cur_c5 = c5_of(h)

    best_h = h.copy()
    best_c5 = cur_c5

    # annealing parameters
    T = max(1e-5, cur_c5 * 0.08)
    decay = 0.9997
    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        move_type = random.random()
        if move_type < 0.6:
            # ----- simple two‑coordinate exchange -----
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            max_up = min(1.0 - hi, hj)
            max_down = min(hi, 1.0 - hj)
            if max_up + max_down == 0.0:
                continue
            delta = random.uniform(-max_down, max_up)
            if abs(delta) < 1e-15:
                continue
            nh = h.copy()
            nh[i] = hi + delta
            nh[j] = hj - delta
        elif move_type < 0.85:
            # ----- targeted move on the worst shift -----
            corr_full = np.correlate(h, 1.0 - h, mode="full")
            idx_max = int(np.argmax(corr_full))
            shift = idx_max - (n - 1)

            if shift >= 0:
                i_inds = np.arange(n - shift)
                p_inds = i_inds + shift
            else:
                s = -shift
                i_inds = np.arange(s, n)
                p_inds = i_inds - s

            contrib = h[i_inds] * (1.0 - h[p_inds])
            if contrib.size == 0:
                continue
            k = int(np.argmax(contrib))
            i_best = i_inds[k]
            p_best = p_inds[k]
            eps_max = min(h[i_best], 1.0 - h[p_best])
            if eps_max <= 1e-12:
                continue
            eps = eps_max * 0.5
            nh = h.copy()
            nh[i_best] -= eps
            nh[p_best] += eps
        else:
            # ----- gradient step on the worst shift -----
            corr_full = np.correlate(h, 1.0 - h, mode="full")
            idx_max = int(np.argmax(corr_full))
            shift = idx_max - (n - 1)

            i = np.arange(n)
            ip = i + shift
            term1 = np.where(ip < n, 1.0 - h[ip], 1.0)
            im = i - shift
            term2 = np.where(im >= 0, h[im], 0.0)
            g = term1 - term2  # gradient of the max‑shift correlation

            eps = random.random() * T
            nh = h - eps * g
            nh = np.clip(nh, 0.0, 1.0)
            nh = project_sum_and_bounds(nh, target_sum)

        new_c5 = c5_of(nh)

        # Metropolis acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improve = it

        it += 1
        if it % 3000 == 0:
            T *= decay

        # occasional restart if stagnated
        if it - last_improve > 150000:
            h = np.zeros(n)
            idx = np.arange(n)
            random.shuffle(idx.tolist())
            h[idx[: n // 2]] = 1.0
            h = project_sum_and_bounds(h, target_sum)
            cur_c5 = c5_of(h)
            last_improve = it

    # ------------------------------------------------------------------
    # final local refinement with targeted moves only (deterministic)
    # ------------------------------------------------------------------
    for _ in range(5000):
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx_max = int(np.argmax(corr_full))
        shift = idx_max - (n - 1)

        if shift >= 0:
            i_inds = np.arange(n - shift)
            p_inds = i_inds + shift
        else:
            s = -shift
            i_inds = np.arange(s, n)
            p_inds = i_inds - s

        contrib = best_h[i_inds] * (1.0 - best_h[p_inds])
        if contrib.size == 0:
            break
        k = int(np.argmax(contrib))
        i_best = i_inds[k]
        p_best = p_inds[k]
        eps_max = min(best_h[i_best], 1.0 - best_h[p_best])
        if eps_max <= 1e-12:
            break
        eps = eps_max * 0.6
        nh = best_h.copy()
        nh[i_best] -= eps
        nh[p_best] += eps
        new_c5 = c5_of(nh)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = nh
        else:
            break

    # final exactness of sum constraint
    final_h = best_h.copy()
    err = final_h.sum() - target_sum
    if abs(err) > 1e-10:
        # adjust a non‑boundary element
        for i in range(n):
            if err > 0 and final_h[i] >= err:
                final_h[i] -= err
                break
            if err < 0 and final_h[i] <= 1.0 + err:
                final_h[i] -= err
                break

    final_c5 = c5_of(final_h)
    return (final_h.tolist(), final_c5, n)
