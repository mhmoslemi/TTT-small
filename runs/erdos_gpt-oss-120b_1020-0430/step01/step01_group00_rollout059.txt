def simulated_anneal(n, init_h, time_left):
    best_h = init_h.copy()
    best_val = evaluate(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    # schedule
    T0 = 1.0
    Tmin = 1e-4
    # number of steps maybe 200000, but break based on time.
    steps = 0
    while time.time() - start < time_left:
        steps += 1
        # compute temperature schedule using exponential decay
        frac = (time.time() - start) / time_left
        T = T0 * (1 - frac) + Tmin * frac
        # random swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        new_val = evaluate(cur_h)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_val = new_val
            if new_val < best_val:
                best_val, best_h = new_val, cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_val

def project_continuous(h):
    n = h.size
    h = np.clip(h,0,1)
    target = n/2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    # Adjust by adding/subtracting evenly while respecting bounds
    # Use iterative approach.
    for _ in range(1000):
        diff = s - target
        if abs(diff) < 1e-9:
            break
        if diff > 0:
            # need decrease sum
            free = np.where(h > 0)[0]
            delta = diff / free.size
            h[free] = np.maximum(0, h[free] - delta)
        else:
            free = np.where(h < 1)[0]
            delta = -diff / free.size
            h[free] = np.minimum(1, h[free] + delta)
        s = h.sum()
    # final adjustment
    residual = target - s
    if abs(residual) > 1e-12:
        # adjust arbitrary index
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1-h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h
```

def slsqp_refine(h0, max_iter=300):
    n = h0.size
    # Ensure feasible start:
    h0 = project_continuous(h0)
    # initial t
    corr = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    constraints = [{'type':'ineq', 'fun': lambda x: x[-1] - np.correlate(x[:n], 1 - x[:n], mode='full')},

    # Use scipy.optimize.minimize with SLSQP

def hill_climb_step(h, cur_val, sample_size=30):
    n = h.size
    # quickly find indices
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return False, cur_val
    best_delta = 0
    best_swap = None
    best_new_val = cur_val
    for _ in range(sample_size):
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Perform swap
        h[i] = 0
        h[j] = 1

        h[j] = 0

        h[i] = 0
        h[j] = 1
        return True, best_new_val

def slsqp_refine(h0, max_iter=200):
    n = h0.size
    # project to feasible (continuous)
    h0 = project_continuous(h0)
    corr = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {'type':'ineq', 'fun': lambda x: x[-1] - np.correlate(x[:n], 1 - x[:n], mode='full')},
        {'type':'eq', 'fun': lambda x: np.sum(x[:n]) - n/2.0}
    ]
    res = optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                            constraints=constraints,
                            options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_continuous(h_opt)
    return h_opt, evaluate_c5(h_opt)

import time, numpy as np
from scipy import optimize

def _evaluate_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _project_continuous(h):
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # simple iterative water-filling
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
            if free.size == 0: break
            delta = diff / free.size
            h[free] -= delta
            h = np.clip(h, 0.0, 1.0)
        else:
            free = np.where(h < 1)[0]
            if free.size == 0: break
            delta = -diff / free.size
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _binarize(arr):
    n = arr.size
    idx = np.argsort(arr)[::-1]
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def _resize_h(h, n_new):
    n_old = h.size
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def _make_constraints(n):
    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1 - h, mode='full')
    def eq(x):
        return np.sum(x[:n]) - n / 2.0
    return ineq, eq

def _slsqp_refine(h0, max_iter=200):
    n = h0.size
    h0 = _project_continuous(h0)
    corr = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    ineq, eq = _make_constraints(n)
    cons = [{'type':'ineq','fun':ineq},
            {'type':'eq','fun':eq}]
    res = optimize.minimize(lambda x: x[-1], x0,
                            method='SLSQP',
                            bounds=bounds,
                            constraints=cons,
                            options={'maxiter':max_iter,'ftol':1e-9,'disp':False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_continuous(h_opt)
    return h_opt, _evaluate_c5(h_opt)

def _sim_anneal_binary(h0, time_budget):
    n = h0.size
    cur = h0.copy()
    cur_val = _evaluate_c5(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_budget:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_val = _evaluate_c5(cur)
        elapsed = time.time() - start

            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_val

import time
import numpy as np
from scipy import optimize

def _evaluate_c5(h: np.ndarray) -> float:
    """C5 for a step function h on [0,2] with grid spacing 2/len(h)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _project_continuous(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 (water‑filling)."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            h[free] -= delta
            h = np.clip(h, 0.0, 1.0)
        else:
            free = np.where(h < 1)[0]
            if free.size == 0:
                break
            delta = -diff / free.size
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _binarize(arr: np.ndarray) -> np.ndarray:
    """Return a 0/1 version with exactly half ones (ties broken arbitrarily)."""
    n = arr.size
    idx = np.argsort(arr)[::-1]
    b = np.zeros(n)
    b[idx[: n // 2]] = 1.0
    return b

def _resize_h(h: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of h onto a new grid of length n_new."""
    n_old = h.size
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def _make_constraints(n: int):
    """Return SLSQP inequality (t ≥ correlation) and equality (sum= n/2)."""
    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1 - h, mode="full")
    def eq(x):
        return np.sum(x[:n]) - n / 2.0
    return ineq, eq

def _slsqp_refine(h0: np.ndarray, max_iter: int = 200):
    """Run a short SLSQP from h0; returns refined h and its C5."""
    n = h0.size
    h0 = _project_continuous(h0)
    corr = np.correlate(h0, 1 - h0, mode="full")
    t0 = np.max(corr)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    ineq, eq = _make_constraints(n)
    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    res = optimize.minimize(lambda x: x[-1], x0,
                            method="SLSQP",
                            bounds=bounds,
                            constraints=cons,
                            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_continuous(h_opt)
    return h_opt, _evaluate_c5(h_opt)

def _sim_anneal_binary(h0: np.ndarray, time_budget: float):
    """Simulated annealing on a binary vector (fixed number of ones)."""
    n = h0.size
    cur = h0.copy()
    cur_val = _evaluate_c5(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_budget:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_val = _evaluate_c5(cur)
        elapsed = time.time() - start
        T = max(1e-4, (time_budget - elapsed) / time_budget * 0.5)  # linear cooling
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_val
