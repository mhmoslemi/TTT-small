def de_optimize(n, budget, start_time, rng):
    popsize = max(10, n // 15) maybe 30.
    # initialize pop: random_feasible_vector for each
    pop = [random_feasible_vector(n,rng) for _ in range(popsize)]
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best = pop[best_idx]
    best_f = fitness[best_idx]

    # parameters: F=0.7, crossover CR=0.9
    while time.time() - start_time < remaining budget:
        for i in range(popsize):
            # choose three distinct indices different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global helpers for SLSQP (soft‑max surrogate) ----------
CURRENT_TARGET = None   # sum constraint target = n/2
CURRENT_BETA = None     # soft‑max temperature

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    """smooth approximation of max_k Σ_i x_i (1-x_{i+k})"""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def _slsqp_optimize(h0, beta, max_iter=300):
    """one SLSQP run on the soft‑max surrogate, preserving bounds and sum."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift fix
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- Problem‑specific utilities ----------
def compute_c5(h):
    """Exact discrete C5 for a step vector h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)

def _make_feasible(h):
    """Clip to [0,1] and adjust the sum to exactly n/2."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Start from the uniform 0.5 vector and perform random pairwise transfers."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return _make_feasible(h)

def perturb_h(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5(best)
    step = start_step
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best = c, cand.copy()
            h = cand
        # periodic step decay
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best, best_c

# ---------- Differential Evolution for the constrained problem ----------
def _de_optimize(n, rng, time_limit):
    """Simple DE that respects box and sum constraints."""
    target = n / 2.0
    popsize = max(15, n // 12)        # modest population
    # initialise population
    pop = [random_feasible_vector(n, rng) for _ in range(popsize)]
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = int(np.argmin(fitness))
    best = pop[best_idx].copy()
    best_f = float(fitness[best_idx])

    F = 0.7
    CR = 0.9
    while time.time() < time_limit:
        for i in range(popsize):
            # mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)

            # enforce sum via random redistribution
            diff = target - mutant.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    if abs(diff) < 1e-12:
                        break
                    j = rng.integers(0, n)
                    if diff > 0:
                        add = min(diff, 1.0 - mutant[j])
                    else:
                        add = max(diff, -mutant[j])
                    mutant[j] += add
                    diff = target - mutant.sum()

            # crossover
            trial = pop[i].copy()
            jrand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = mutant[mask]

            # ensure feasibility again
            trial = np.clip(trial, 0.0, 1.0)
            diff = target - trial.sum()
            if abs(diff) > 1e-12:
                for _ in range(2000):
                    if abs(diff) < 1e-12:
                        break
                    j = rng.integers(0, n)
                    if diff > 0:
                        add = min(diff, 1.0 - trial[j])
                    else:
                        add = max(diff, -trial[j])
                    trial[j] += add
                    diff = target - trial.sum()

            f_trial = compute_c5(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_f:
                    best_f, best = f_trial, trial.copy()
        # occasional early exit if time is almost up
        if time.time() > time_limit - 0.5:
            break
    return best, best_f

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- Determine discretisation size -----
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n_parent = parent.size
        # aim for roughly double resolution, but stay ≤ 300 points
        n = min(max(150, n_parent * 2), 300)
        # linear interpolation to the new grid
        x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
    else:
        n = 180
        h = np.full(n, 0.5)

    h = _make_feasible(h)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- Phase 1: quick local improvement of the parent -----
    if time.time() - start < budget_s * 0.10:
        h1, c1 = hill_climb(best_h, rng, max_iters=2500, start_step=0.6)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # ----- Phase 2: Differential Evolution (global exploration) -----
    de_deadline = start + budget_s * 0.65
    if time.time() < de_deadline:
        h_de, c_de = _de_optimize(n, rng, de_deadline)
        if c_de < best_c:
            best_h, best_c = h_de, c_de

    # ----- Phase 3: SLSQP refinements with increasing beta -----
    beta_schedule = [20.0, 80.0, 200.0, 500.0]
    for beta in beta_schedule:
        if time.time() - start > budget_s * 0.90:
            break
        try:
            refined = _slsqp_optimize(best_h, beta=beta, max_iter=400)
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_h, best_c = refined, c_ref
        except Exception:
            continue

    # ----- Phase 4: final deterministic hill‑climb -----
    if time.time() - start < budget_s * 0.98:
        h_final, c_final = hill_climb(best_h, rng, max_iters=5000, start_step=0.2)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    # ----- Ensure exact feasibility before returning -----
    best_h = _make_feasible(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
