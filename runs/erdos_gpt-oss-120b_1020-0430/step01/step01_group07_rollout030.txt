    # compute correlation
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    ns = _neighbor_sum(h, lag)
    # Want to move from low ns to high ns
    # We'll pick i_low as index with smallest ns among those with h[i] > 0 (source)
    # i_high as index with largest ns among those with h[i] < 1 (sink)
    # Might add small tolerance in case can't move.
    candidates_low = np.where(h > 0)[0]
    if candidates_low.size == 0:
        return None, None, None
    low = candidates_low[np.argmin(ns[candidates_low])]
    candidates_high = np.where(h < 1)[0]

        return None, None, None
    delta = random.uniform(0, delta_max)
    # create new h via copy
    new_h = h.copy()
    new_h[low] -= delta
    new_h[high] += delta
    return new_h, lag, (low, high, delta)

def _interpolate_h(old_h, n):
    n0 = old_h.shape[0]
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x, x0, old_h)
    # enforce sum
    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    # adjust any small rounding diff
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + diff <= hi:
                h[idx] += diff
                break
    return h

def _anneal_directed(start_h, max_seconds):
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    start = time.time()
    # scheduling parameters
    T0 = 0.02
    T_end = 1e-6
    max_step = 0.2
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        if random.random() < 0.7:
            new_h, _lag, _info = _directed_move_once(h, max_step)
            if new_h is None:
                # fallback random swap
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low >= high:
                    continue
                delta = random.uniform(max(low, -max_step), min(high, max_step))
                new_h = h.copy()
                new_h[i] = hi + delta
                new_h[j] = hj - delta
        else:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = random.uniform(max(low, -max_step), min(high, max_step))
            new_h = h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
        new_c = _c5_metric(new_h)
        dC = new_c - cur_c
        if dC < 0.0 or random.random() < math.exp(-dC / T):
            h = new_h
            cur_c = new_c

                best_c = new_c

import time

def _c5_metric(h):
    """Compute C_5 for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _interpolate_h(old_h, n):
    """Resize old_h to length n while keeping the integral exactly 1."""
    n0 = old_h.shape[0]
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x, x0, old_h)
    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
    return h

def _directed_move_once(h, max_step):
    """Generate a move that tries to improve the worst lag."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)

    ns = np.zeros(n, dtype=h.dtype)
    if lag >= 0:
        ns[: n - lag] += h[lag:]
        ns[lag:] += h[: n - lag]
    else:
        pos = -lag
        ns[pos:] += h[: n - pos]
        ns[: n - pos] += h[pos:]

    src = np.where(h > 0)[0]
    if src.size == 0:
        return None, None, None
    low = src[np.argmin(ns[src])]

    tgt = np.where(h < 1)[0]
    if tgt.size == 0:
        return None, None, None
    high = tgt[np.argmax(ns[tgt])]

    if low == high:
        return None, None, None
    delta_max = min(max_step, h[low], 1.0 - h[high])
    if delta_max <= 0.0:
        return None, None, None
    delta = random.uniform(0.0, delta_max)
    new_h = h.copy()
    new_h[low] -= delta
    new_h[high] += delta
    return new_h, lag, (low, high, delta)

def _anneal_directed(start_h, max_seconds):
    """Simulated‑annealing with a bias toward the worst lag."""
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c

    start = time.time()
    T0 = 0.02
    T_end = 1e-6
    max_step = 0.2
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress

        if random.random() < 0.7:
            move = _directed_move_once(h, max_step)
            if move[0] is None:
                # fall back to a random pair move
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                lo = max(-hi, hj - 1.0)
                hi_ = min(1.0 - hi, hj)
                if lo >= hi_:
                    continue
                delta = random.uniform(max(lo, -max_step), min(hi_, max_step))
                new_h = h.copy()
                new_h[i] = hi + delta
                new_h[j] = hj - delta
            else:
                new_h = move[0]
        else:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_ = min(1.0 - hi, hj)
            if lo >= hi_:
                continue

            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def _refine_gradient(start_h, max_seconds):
    """Deterministic gradient ascent on the worst lag."""
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(h)
    cur_c = best_c
    start = time.time()
    while time.time() - start < max_seconds:
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag = int(np.argmax(corr) - (n - 1))

        ns = np.zeros(n, dtype=h.dtype)
        if lag >= 0:
            ns[: n - lag] += h[lag:]
            ns[lag:] += h[: n - lag]
        else:
            pos = -lag
            ns[pos:] += h[: n - pos]
            ns[: n - pos] += h[pos:]

        direction = ns - ns.mean()
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        alpha = 0.05 * (0.5 ** progress)
        delta = alpha * direction

        h_new = h + delta
        np.clip(h_new, 0.0, 1.0, out=h_new)

        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for idx in np.random.permutation(n):
                if 0.0 <= h_new[idx] + diff <= 1.0:
                    h_new[idx] += diff
                    break

        new_c = _c5_metric(h_new)
        if new_c < best_c:
            best_c = new_c
            best_h = h_new.copy()
        if new_c <= cur_c:
            h = h_new
            cur_c = new_c
    return best_h, best_c
