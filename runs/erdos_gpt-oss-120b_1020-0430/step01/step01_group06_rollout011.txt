    v = np.clip(v, 0., 1.)
    cur = v.sum()
    diff = cur - target
    if abs(diff) < 1e-12:
        return v
    # Simple proportional scaling then fixing with small adjustments.
    if cur > 0:

    v = np.clip(v, 0., 1.)

    cur = v.sum()

def _project_sum(v, target):
    """Clip to [0,1] and adjust to meet sum=target."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    if cur > 0:
        v = v * (target / cur)
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    # simple iterative correction
    for _ in range(1000):
        diff = cur - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to decrease sum
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            i = random.choice(cand.tolist())
            delta = min(v[i], diff)
            v[i] -= delta
        else:
            # need to increase sum
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = random.choice(cand.tolist())
            delta = min(1.0 - v[i], -diff)
            v[i] += delta
        cur = v.sum()
    return v

def _c5(v):
    """Return the C5 value for vector v using the required discretization."""
    n = v.shape[0]
    corr = np.correlate(v, 1.0 - v, mode='full')
    return float(corr.max() * (2.0 / n))

def _self_corr(v):
    """Return self‐correlation array used to locate the worst shift."""
    return np.correlate(v, v, mode='full')

def _worst_shift(v):
    """Return (shift, min_corr) where shift gives the smallest self‑correlation."""
    corr = _self_corr(v)
    min_index = int(np.argmin(corr))
    shift = min_index - (len(v) - 1)
    return shift, float(corr[min_index])

def _gradient(v, shift):
    """Gradient of S_shift = sum_i v_i v_{i+shift} with respect to v."""
    n = len(v)
    g = np.zeros(n, dtype=float)
    d = abs(shift)
    if d == 0:
        g = 2.0 * v
    else:
        # term v[i+shift]
        g[:n - d] += v[d:]
        # term v[i-shift]
        g[d:] += v[:n - d]
    return g

def _improve(v, target_sum, time_budget):
    """Local search that tries to raise the worst self‑correlation."""
    n = len(v)
    cur = v.copy()
    cur_c5 = _c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    # temperature schedule
    T = max(1e-5, cur_c5 * 0.02)
    decay = 0.9995
    it = 0
    start = time.time()

    while time.time() - start < time_budget:
        # locate worst shift
        shift, _ = _worst_shift(cur)
        g = _gradient(cur, shift)

        # choose candidates for transfer
        eps = 1e-12
        up_mask = cur < (1.0 - eps)
        down_mask = cur > eps
        up_idx = None
        down_idx = None
        if up_mask.any() and down_mask.any():
            up_candidates = np.where(up_mask)[0]
            down_candidates = np.where(down_mask)[0]
            # pick highest gradient among up‑candidates
            up_idx = up_candidates[np.argmax(g[up_candidates])]
            down_idx = down_candidates[np.argmin(g[down_candidates])]
            # ensure improvement direction
            if g[up_idx] <= g[down_idx] + 1e-12:
                up_idx = None  # fallback to random move
        if up_idx is None:

            i, j = random.sample(range(n), 2)
            hi, hj = cur[i], cur[j]
            max_up = min(1.0 - hi, hj)        # increase i, decrease j
            max_down = min(hi, 1.0 - hj)      # decrease i, increase j

            nh[j] = hj - delta

            if max_delta <= 0:

            nh = cur.copy()
            nh[up_idx] += delta

def _project_sum(v, target):

    v = np.clip(v, 0.0, 1.0)
