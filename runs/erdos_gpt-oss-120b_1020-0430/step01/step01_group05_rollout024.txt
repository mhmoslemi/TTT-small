def project_onto_bounds_and_sum(x, target):
    # Clip to bounds after subtract lambda; bisection to find lambda such that sum clipped = target.
    lo = -10.0  # big negative ensure sum>target
    hi = 10.0   # big positive
    for _ in range(50):
        mid = (lo+hi)/2
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo+hi)/2, 0.0, 1.0)
    return proj

import time
import random
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project vector x onto the box [0,1]^n while enforcing sum(x)=target.
    Uses a simple bisection on the Lagrange multiplier."""
    lo, hi = -10.0, 10.0  # guarantees sum at lo > target, at hi < target
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # autocorrelation (full) of h with itself
    conv = np.correlate(h, h, mode="full")
    min_conv = conv.min()
    # ∫h = 1 by construction, so C5 = 1 - min_conv * dx
    return 1.0 - min_conv * dx


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape with peak 1 at the centre."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)  # linear from centre to edges
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x) = sin^2(πx/2) on [0,2] (already integrates to 1)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new length."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def grad_worst_shift(h: np.ndarray) -> np.ndarray:
    """Gradient (w.r.t. h) of the worst‑shift autocorrelation term."""
    n = h.shape[0]
    conv = np.correlate(h, h, mode="full")
    idx = int(conv.argmin())
    d = idx - (n - 1)                     # offset = shift index (can be negative)
    g = np.zeros_like(h)

    if d >= 0:
        if d < n:
            g[: n - d] += h[d:]
            g[d:] += h[: n - d]
    else:  # d < 0
        off = -d
        if off < n:
            g[off:] += h[: n - off]
            g[: n - off] += h[off:]
    return g


# ------------------------------------------------------------
# Core optimisation routine
# ------------------------------------------------------------
def gradient_ascent(h0: np.ndarray, target: float, max_iters: int,
                    step0: float, start: float, budget: float) -> tuple:
    """Projected gradient ascent on the worst‑shift autocorrelation."""
    h = h0.copy()
    best_val = compute_c5(h)
    best_h = h.copy()
    n = h.shape[0]

    for t in range(max_iters):
        if time.time() - start > budget - 0.2:
            break
        g = grad_worst_shift(h)
        norm = np.linalg.norm(g)
        if norm < 1e-12:
            break
        # step length that yields a roughly constant step in h‑space
        step = step0 / np.sqrt(t + 1) / (norm + 1e-12)
        h_new = h + step * g
        h_new = project_onto_bounds_and_sum(h_new, target)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            h = h_new
            # early exit if we have already beaten the target record
            if best_val <= 0.3808:
                break
        else:
            # keep current h (no move)
            pass
    return best_h, best_val


def random_refine(h: np.ndarray, target: float, best_val: float,
                  start: float, budget: float) -> tuple:
    """Simple Monte‑Carlo pairwise mass transfers."""
    n = h.shape[0]
    h_cur = h.copy()
    val_cur = best_val

    while time.time() - start < budget - 0.2:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        # explore a few candidate deltas, keep the best
        best_local = val_cur
        best_candidate = None
        for _ in range(3):
            delta = np.random.uniform(delta_min, delta_max)
            if abs(delta) < 1e-12:
                continue
            cand = h_cur.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            cand = np.clip(cand, 0.0, 1.0)
            # sum unchanged (delta transferred), but clipping may have altered it;
            # project to be safe
            cand = project_onto_bounds_and_sum(cand, target)
            val = compute_c5(cand)
            if val < best_local:
                best_local = val
                best_candidate = cand
        if best_candidate is not None and best_local < val_cur:
            h_cur = best_candidate
            val_cur = best_local
            if val_cur <= 0.3808:
                break
    return h_cur, val_cur

    np.random.seed(seed)

    start_time = time.time()
    # Choose a moderate size (still fast O(n^2) for correlation)

    seeds = []

    seeds.append(make_sin2(n))

    for _ in range(3):
        rnd = np.random.rand(n)
        seeds.append(project_onto_bounds_and_sum(rnd, target_sum))

    best_h = None
    best_val = np.inf

    # --------------------------------------------------------
    # Gradient ascent on each seed
    # --------------------------------------------------------
    for idx, seed_h in enumerate(seeds):
        if time.time() - start_time > budget_s - 5.0:
            break
        # number of gradient steps adapts to remaining time
        remaining = budget_s - (time.time() - start_time)
        max_iters = int(min(3000, remaining * 5))  # rough budget
        h_opt, val = gradient_ascent(
            seed_h,
            target_sum,
            max_iters=max_iters,
            step0=2.0,
            start=start_time,
            budget=budget_s,
        )
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        if best_val <= 0.3808:
            break

    # Final stochastic refinement
    # --------------------------------------------------------
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    # Random local search while time permits
    best_h, best_val = random_refine(best_h, target_sum, best_val,
                                     start_time, budget_s)

    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
