def simulated_annealing_binary(initial_h, rng, deadline):
    n = len(initial_h)
    target_sum = n / 2
    # ensure integer sum target
    # initial_h expected binary (0/1)
    h = initial_h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    # lists of indices
    # convert to python lists for speed
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] <= 0.5]
    # Ensure lengths correct
    assert len(ones) == target_sum and len(zeros) == n - target_sum
    
    temp = 0.01  # initial temperature
    decay = 0.9999
    
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # pick swap
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new C5
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            # acceptance probability
            if temp > 0:
                prob = np.exp(-delta / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            # accept swap
            c5 = new_c5
            # update lists
            ones[i_pos] = j
            zeros[j_pos] = i
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                # early exit if below target threshold maybe
                # we might break if best_c5 <= 0.3808
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # decay temperature
        temp *= decay
        # optional: reset temperature if too low? Not needed.
        # break if temp below small threshold? Not needed.
        # Possibly break if no improvement for many iter? Could skip.

    return best_h, best_c5

def run(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    best_h = None
    best_c5 = np.inf
    best_n = None

    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    # candidate n sizes
    candidate_ns = [80,100,120,150,180]  # ensure even
    if parent_n is not None and parent_n not in candidate_ns:
        candidate_ns.insert(0,parent_n)

    # For each n, will have multiple start draws
    for n in candidate_ns:
        if time.time() > deadline:
            break
        target_sum = n // 2
        # Generate start list
        starts = []
        # Upsampled parent if exists
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))
        # Triangular shape
        starts.append(generate_triangular(n))
        # Random continuous starts
        for _ in range(6):
            starts.append(random_start(n, rng))
        # Random binary start (half ones)
        for _ in range(3):
            # binary vector of half ones
            idx = rng.permutation(n)[:target_sum]
            hb = np.zeros(n, dtype=float)
            hb[idx] = 1.0
            starts.append(hb)
        # Perturbations of parent or triangular
        for _ in range(3):
            base = generate_triangular(n) if rng.random()<0.5 else (starts[0] if len(starts)>0 else random_start(n,rng))
            pert = base + rng.uniform(-0.15,0.15,size=n)
            pert = np.clip(pert,0.0,1.0)
            # Enforce sum
            diff = target_sum - pert.sum()
            if abs(diff) > 1e-12:
                pert += diff / n
                pert = np.clip(pert,0.0,1.0)
            starts.append(pert)

        # Optimize each start
        for s in starts:
            if time.time() > deadline:
                break
            h_opt, c5_opt = solve_one_start(s, n, rng, deadline)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = n
                # early break if very good
                if best_c5 <= 0.3808:
                    break
        if best_c5 <= 0.3808:
            break

    # If we have a best solution and still time left, attempt binary SA refinement.
    if best_h is not None and time.time() < deadline:
        # convert to binary by thresholding to appropriate number of ones.
        n = best_h.shape[0]
        target_one = n // 2
        # choose indices with largest values
        idx_sorted = np.argsort(-best_h)  # descending
        binary_h = np.zeros(n, dtype=float)
        binary_h[idx_sorted[:target_one]] = 1.0
        # run SA on binary version
        sa_h, sa_c5 = simulated_annealing_binary(binary_h, rng, deadline)
        if sa_c5 < best_c5 - 1e-12:
            best_h = sa_h
            best_c5 = sa_c5
            best_n = n

    # Fallback if no best (rare)
    if best_h is None:
        n = 80
        best_h = np.concatenate([np.ones(n//2), np.zeros(n - n//2)]).astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Ensure sum constraint precise
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        # simple projection: shift by constant then clip
        correction = (target - cur_sum) / best_n
        best_h = best_h + correction
        best_h = np.clip(best_h, 0.0, 1.0)
        # Rebalance if still off
        cur_sum = best_h.sum()
        if abs(cur_sum - target) > 1e-9:
            # scaling approach
            best_h = best_h * (target / cur_sum)
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n

def solve_one_start(start, n, rng, deadline):
    target = n / 2.0
    # equality constraint: sum(h) - target = 0
    def constraint_eq(h):
        return np.sum(h) - target
    cons = {'type':'eq', 'fun': constraint_eq}
    bounds = [(0.0,1.0)] * n

    # Stage 1
    # Use softmax objective
    def objective_stage(h, alpha, lam):
        return softmax_c5(h, alpha) + lam * np.sum(h * (1 - h))

    # stage schedule list of (alpha, lam, maxiter)
    stage_params = [(5.0, 0.0, 200), (15.0, 0.05, 200), (30.0, 0.1, 200)]
    h_cur = start.copy()
    for alpha, lam, maxiter in stage_params:
        if time.time() > deadline:
            break
        res = minimize(
            lambda h: objective_stage(h, alpha, lam),
            h_cur,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            # continue anyway with result maybe
            pass
        h_cur = np.clip(res.x, 0.0, 1.0)
    # final evaluation
    c5_val = compute_c5(h_cur)
    # Verify sum close:
    sum_val = h_cur.sum()
    if abs(sum_val - target) > 1e-6:
        # simple projection adjustment
        correction = (target - sum_val) / n
        h_cur = h_cur + correction
        h_cur = np.clip(h_cur, 0.0, 1.0)
    return h_cur, c5_val

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core evaluation utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap via log-sum-exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


# ------------------------------------------------------------
# Simple feasible constructions
# ------------------------------------------------------------
def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    # final exactness
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible start (values in [0,1] and sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, then renormalise."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


# ------------------------------------------------------------
# Optimisation of a single start
# ------------------------------------------------------------
def solve_one_start(start: np.ndarray, n: int, rng: np.random.Generator,
                    deadline: float) -> tuple:
    """Two‑stage SLSQP optimisation with a softmax proxy and a binary‑penalty."""
    target = n / 2.0

    def eq_sum(h):
        return np.sum(h) - target

    cons = {"type": "eq", "fun": eq_sum}
    bounds = [(0.0, 1.0)] * n

    # objective = softmax_c5 + λ·∑h(1−h)  (the penalty encourages 0/1 values)
    def objective(h, alpha, lam):
        return softmax_c5(h, alpha) + lam * np.sum(h * (1.0 - h))

    # schedule of (α, λ, maxiter)
    stages = [(5.0, 0.0, 200), (15.0, 0.05, 200), (30.0, 0.1, 200)]

    h_cur = start.copy()
    for alpha, lam, maxit in stages:
        if time.time() > deadline:
            break
        res = minimize(
            lambda h: objective(h, alpha, lam),
            h_cur,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxit, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            # keep whatever we have
            pass
        h_cur = np.clip(res.x, 0.0, 1.0)

    # final clean‑up of the sum (tiny numerical drift)
    cur_sum = h_cur.sum()
    if abs(cur_sum - target) > 1e-9:
        correction = (target - cur_sum) / n
        h_cur = h_cur + correction
        h_cur = np.clip(h_cur, 0.0, 1.0)

    return h_cur, compute_c5(h_cur)


# ------------------------------------------------------------
# Simple binary simulated‑annealing refinement
# ------------------------------------------------------------
def simulated_annealing_binary(start_h: np.ndarray,
                               rng: np.random.Generator,
                               deadline: float) -> tuple:
    """Hill‑climbing with occasional uphill moves on a fixed‑weight binary vector."""
    n = start_h.shape[0]
    target_one = n // 2
    h = start_h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # maintain index lists for fast swaps
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] <= 0.5]

    temp = 0.01
    decay = 0.9999
    while time.time() < deadline:
        # propose a swap
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5

        accept = False
        if new_c5 <= best_c5:
            accept = True
        else:
            if temp > 0 and rng.random() < np.exp(-delta / temp):
                accept = True

        if accept:
            # keep the swap
            best_c5 = new_c5
            best_h = h.copy()
            # update index lists
            ones[i_pos] = j
            zeros[j_pos] = i
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        temp *= decay
        # stop early if we already beat the target record
        if best_c5 <= 0.3808:
            break

    return best_h, best_c5
