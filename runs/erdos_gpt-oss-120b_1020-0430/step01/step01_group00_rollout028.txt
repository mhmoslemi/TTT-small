def gradient_descent(h0, n, max_iters, sigma_list, step0, time_check_func):
   h = h0.copy()
   best_h = h.copy()
   best_val = c5_from_h(h)
   iter_total = 0
   for sigma in sigma_list:
       for it in range(max_iters):
           if not time_check_func():
               break

           raw_max = np.max(corr)

def gradient_weighted(h, w):
    n = h.shape[0]
    grad = np.ones(n)
    L = 2*n - 1
    # w has length L, offsets from -(n-1) to (n-1)
    for offset in range(-(n-1), n):
        weight = w[offset + n - 1]
        if weight == 0.0:
            continue
        if offset == 0:
            grad -= 2.0 * weight * h
        elif offset > 0:
            # subtract weight * h[i+offset] for i=0..n-offset-1
            grad[:n - offset] -= weight * h[offset:]
            # subtract weight * h[i-offset] for i=offset..n-1
            grad[offset:] -= weight * h[:n - offset]
        else:
            s = -offset
            # offset negative: i + offset = i - s
            # subtract weight * h[i-s] for i = s..n-1
            grad[s:] -= weight * h[:n - s]  # i-s index = (i - s) = 0..n-s-1 corresponds to h[0..n-s-1] for i=s..n-1
            # subtract weight * h[i - offset] = h[i + s] for i=0..n-s-1
            grad[:n - s] -= weight * h[s:]
    return grad

def project_to_feasible(h, n, target=None, tol=1e-9, max_iter=500):
    if target is None: target = n / 2.0
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    for _ in range(max_iter):
        total = h.sum()
        diff = total - target
        if abs(diff) <= tol:
            break
        # free indices where we can move
        if diff > 0:
            # need to reduce sum, reduce elements >0
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        # Clip again to keep within [0,1]
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual
        if residual > 0:
            # add to indices <1
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        for i in idx:
            if residual > 0:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            else:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

# ----------------------------------------------------------------------
# Feasibility projection: clip to [0,1] and enforce sum(h)=n/2
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9,
                         max_iter: int = 500) -> np.ndarray:
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0.0:
            free = np.where(h > 0.0)[0]
        else:
            free = np.where(h < 1.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction (should be negligible)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0.0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        for i in idx:
            if residual > 0.0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            else:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# Compute C5 from a discretised h (including dx factor)
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ----------------------------------------------------------------------
# Linear interpolation to a finer grid
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑overlap objective.
# w is a probability vector (sum 1) over all shifts (length 2n‑1)
def _gradient_weighted(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = h.size
    grad = np.ones(n, dtype=float)          # Σ w_s = 1 gives the constant term
    # offsets run from -(n-1) .. (n-1)
    for offset in range(-(n - 1), n):
        wgt = w[offset + n - 1]
        if wgt == 0.0:
            continue
        if offset == 0:
            grad -= 2.0 * wgt * h
        elif offset > 0:
            # term h[i+offset]
            grad[: n - offset] -= wgt * h[offset:]
            # term h[i-offset]
            grad[offset:] -= wgt * h[: n - offset]
        else:  # offset < 0
            s = -offset
            # term h[i-s] (i >= s)
            grad[s:] -= wgt * h[: n - s]
            # term h[i+s] (i <= n-s-1)
            grad[: n - s] -= wgt * h[s:]
    return grad


# ----------------------------------------------------------------------
# Projected (smoothed) sub‑gradient descent on the max‑overlap.
def _gradient_descent(start_h: np.ndarray, n: int,
                      max_iters_per_sigma: int,
                      sigma_list,
                      step0: float,
                      start_time: float,
                      budget_s: float):
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    it_counter = 0

    for sigma in sigma_list:
        for _ in range(max_iters_per_sigma):
            if time.time() - start_time > budget_s - 5.0:
                return best_h, best_val
            # correlation (raw, without dx)
            corr = np.correlate(h, 1.0 - h, mode="full")
            raw_max = np.max(corr)
            cur_val = raw_max * (2.0 / n)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            # smoothed weights
            if sigma > 1e-12:
                shift = corr - raw_max                 # ≤0
                w = np.exp(shift / sigma)
                w_sum = w.sum()
                if w_sum == 0.0 or not np.isfinite(w_sum):
                    w = np.zeros_like(corr)
                    w[np.argmax(corr)] = 1.0
                else:
                    w /= w_sum
            else:
                w = np.zeros_like(corr)
                w[np.argmax(corr)] = 1.0
            # gradient
            grad = _gradient_weighted(h, w)
            # step size (decreasing)
            step = step0 / np.sqrt(it_counter + 1.0)
            h = h - step * grad
            h = _project_to_feasible(h, n)
            it_counter += 1
    # final check
    final_val = _c5_from_h(h)
    if final_val < best_val:
        best_h, best_val = h, final_val
    return best_h, best_val

# Simple local random improvement (pairwise mass transfer)
def _local_search(start_h: np.ndarray, n: int, rng: np.random.Generator,
                  max_iter: int,
                  start_time: float,
                  budget_s: float):
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_from_h(best_h)
    # step magnitude, will shrink a little with the iteration count
    base_delta = 0.05

    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5.0:
            break
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = (rng.random() * 2.0 - 1.0) * base_delta
        hi, hj = h[i], h[j]
        new_i = hi - delta
        new_j = hj + delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        h_candidate = h.copy()
        h_candidate[i] = new_i
        h_candidate[j] = new_j
        val = _c5_from_h(h_candidate)
        if val + 1e-12 < best_val:
            best_val = val
            best_h = h_candidate
            h = h_candidate      # continue from the improved point
    return best_h, best_val

            base_h = _upscale_h(init_h, n) if init_h.size != n else init_h.copy()
        else:
            base_h = np.full(n, 0.5)

        # Number of random restarts (reduced if we are running out of time)
        remaining = budget_s - (time.time() - start_time)
        restarts = 6 if remaining > budget_s * 0.6 else 3

        for _ in range(restarts):
            if time.time() - start_time > budget_s - 5.0:
                break

            # Small random perturbation of the base
            start_h = base_h + 0.12 * rng.standard_normal(n)
            start_h = _project_to_feasible(start_h, n)

            # Gradient descent phase
            sigma_seq = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            h_opt, val_opt = _gradient_descent(
                start_h, n,
                max_iters_per_sigma=40,
                sigma_list=sigma_seq,
                step0=0.5,
                start_time=start_time,
                budget_s=budget_s,
            )

            # Local random improvement
            h_opt, val_opt = _local_search(
                h_opt, n, rng,
                max_iter=800,
                start_time=start_time,
                budget_s=budget_s,
            )

            # Update global best
            if val_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), val_opt, n
                if best_val <= target_bound:
                    break

        if best_val <= target_bound:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform half function if nothing else succeeded
    if best_h is None:
        best_n = 128
        best_h = np.full(best_n, 0.5)
        best_val = _c5_from_h(best_h)

    # Ensure feasibility one last time
    best_h = _project_to_feasible(best_h, best_n)

    return best_h.tolist(), float(best_val), int(best_n)
