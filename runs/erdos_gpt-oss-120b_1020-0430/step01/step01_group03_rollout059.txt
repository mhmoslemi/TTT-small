    grad_h = term1 - term2

def softmax_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    obj = (m + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    g = 1 - h
    conv1 = np.convolve(w, g, mode='full')
    term1 = conv1[n-1 : n-1 + n]
    w_rev = w[::-1]
    conv2 = np.convolve(w_rev, h, mode='full')
    term2 = conv2[n-1 : n-1 + n]
    grad_h = term1 - term2
    return obj, grad_h

def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0
    # set global values needed for constraints; they could be captured but we avoid closure.
    # We'll define inner functions that reference target and beta via args.

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        exp_bc = np.exp(bc - m)
        sum_exp = np.exp(0) # not needed
        obj = (m + np.log(np.sum(exp_bc))) / beta
        return obj

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = ensure_feasible(x, target, np.random.default_rng())  # maybe using rng for redistribution
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global helpers for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth max Σ_i x_i (1-x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def ensure_feasible(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # simple random redistribution
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def symmetrize(h: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry h[i] = h[n-1-i]."""
    return (h + h[::-1]) * 0.5


# ---------- stochastic moves ----------
def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return ensure_feasible(h, n / 2.0, rng)


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.5):
    """Greedy hill‑climb: transfer mass between two coordinates."""
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # shrink step periodically
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    best_h = ensure_feasible(best_h, target, rng)
    best_h = symmetrize(best_h)
    return best_h, best_c


# ---------- softmax gradient ----------
def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Return (softmax objective, gradient w.r.t. h)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    obj = (m + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp                     # weights for each lag
    g = 1 - h
    # term1 = w ⋆ g   (convolution, slice to length n)
    conv1 = np.convolve(w, g, mode="full")
    term1 = conv1[n - 1 : n - 1 + n]
    # term2 = (w reversed) ⋆ h
    w_rev = w[::-1]
    conv2 = np.convolve(w_rev, h, mode="full")
    term2 = conv2[n - 1 : n - 1 + n]
    grad_h = term1 - term2
    return obj, grad_h


# ---------- gradient descent on logit space ----------
def gradient_descent_opt(h0: np.ndarray, rng: np.random.Generator,
                        time_budget: float, max_iters: int = 20000,
                        beta: float = 200.0, base_step: float = 0.5,
                        min_step: float = 0.01):
    """Projected gradient descent on sigmoid‑parameterised variables."""
    n = h0.size
    target = n / 2.0
    eps = 1e-8
    h0 = np.clip(h0, eps, 1 - eps)
    y = np.log(h0 / (1 - h0))                # logit representation
    best_h = 1.0 / (1.0 + np.exp(-y))
    best_h = ensure_feasible(best_h, target, rng)
    best_c = compute_c5(best_h)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        h = 1.0 / (1.0 + np.exp(-y))
        h = symmetrize(h)
        h = ensure_feasible(h, target, rng)
        _, grad_h = softmax_obj_and_grad(h, beta)
        dh_dy = h * (1.0 - h)
        grad_y = grad_h * dh_dy
        step = base_step * (1.0 - it / max_iters) + min_step
        y -= step * grad_y
        if (it & 31) == 0:                  # occasional exact check
            cur_h = 1.0 / (1.0 + np.exp(-y))
            cur_h = symmetrize(cur_h)
            cur_h = ensure_feasible(cur_h, target, rng)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c


# ---------- SLSQP wrapper ----------
def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    try:
        res = minimize(softmax_obj, h0, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12,
                                "disp": False})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = np.clip(h0, 0.0, 1.0)
    return ensure_feasible(x, n / 2.0, rng)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(120, 0.5)                # fallback size
    n = h0.size
    target = n / 2.0
    h0 = ensure_feasible(h0, target, rng)
    h0 = symmetrize(h0)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- phase 1: hill‑climb from the parent -----
    if time.time() - start_time < budget_s * 0.70:
        h1, c1 = hill_climb(best_h, rng, max_iters=3000, start_step=0.6)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # ----- phase 2: gradient descent on several random starts -----
    while time.time() - start_time < budget_s * 0.85:
        # allocate a slice of the remaining budget to this run
        remaining = budget_s - (time.time() - start_time)
        # each GD run gets at most 30 s
        run_time = min(30.0, remaining * 0.25)
        # start from a fresh random feasible vector
        h_start = random_feasible_vector(n, rng)
        h_start = symmetrize(h_start)
        # short hill‑climb to give a decent foothold
        h_start, _ = hill_climb(h_start, rng, max_iters=800, start_step=0.5)
        h_gd, c_gd = gradient_descent_opt(h_start, rng, time_budget=run_time,
                                          max_iters=15000, beta=300.0)
        if c_gd < best_c:
            best_h, best_c = h_gd, c_gd

    # ----- phase 3: SLSQP refinements with increasing β -----
    for beta in (20.0, 80.0, 200.0, 500.0):
        if time.time() - start_time > budget_s * 0.95:
            break
        trial = slsqp_optimize(best_h, beta, rng, max_iter=400)
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial

    # ----- phase 4: optional up‑sampling & fine‑tuning -----
    if time.time() - start_time < budget_s * 0.98 and n * 2 <= 300:
        # linear interpolation to double resolution
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n * 2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        target_up = (n * 2) / 2.0
        h_up = ensure_feasible(h_up, target_up, rng)
        h_up = symmetrize(h_up)

        # a quick hill‑climb
        h_up, _ = hill_climb(h_up, rng, max_iters=2000, start_step=0.5)

        # gradient descent at higher resolution
        remaining = budget_s - (time.time() - start_time)
        gd_time = max(0.0, min(remaining * 0.6, 120.0))
        h_up, c_up = gradient_descent_opt(h_up, rng,
                                          time_budget=gd_time,
                                          max_iters=20000,
                                          beta=400.0)
        # final SLSQP passes
        for beta in (80.0, 200.0, 500.0):
            if time.time() - start_time > budget_s:
                break
            trial = slsqp_optimize(h_up, beta, rng, max_iter=300)
            c_trial = compute_c5(trial)
            if c_trial < c_up:
                h_up, c_up = trial, c_trial
        if c_up < best_c:
            best_h, best_c = h_up, c_up
            n = best_h.size

    # ----- final clean‑up -----
    best_h = ensure_feasible(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
