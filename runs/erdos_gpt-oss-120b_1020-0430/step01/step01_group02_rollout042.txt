def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # Choose even n
    n = 48
    target_ones = n // 2
    # Initialize binary vector with random selection of ones
    h = np.zeros(n, dtype=np.float64)
    ones_idx = rng.choice(n, size=target_ones, replace=False)
    h[ones_idx] = 1.0

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # Precompute dx
    dx = 2.0 / n
    # Determine iteration budget: let's run as many as possible but check time
    while time.time() - start < budget_s - 0.5:
        # compute autocorrelation to find minimal shift
        corr = np.correlate(h, h, mode='full')
        # positive shifts start at index n (center+1)
        # skip shift 0 at center
        pos_corr = corr[n:]  # includes shift0? Actually corr[n-1] = zero shift; So corr[n:] start at shift+1.
        # find min_corr among shifts 1..n-1
        min_idx = int(np.argmin(pos_corr))  # idx in pos_corr corresponding to shift = idx+1
        shift = min_idx + 1  # shift s
        # Determine candidate source and destination for swap
        # source positions: indices i where h[i]==1 and i+shift < n and h[i+shift]==0
        src_candidates = np.where((h == 1) & (np.arange(n) + shift < n) & (h[np.arange(n) + shift] == 0))[0]
        # destination positions: indices j where h[j]==0 and j+shift < n and h[j+shift]==1
        dst_candidates = np.where((h == 0) & (np.arange(n) + shift < n) & (h[np.arange(n) + shift] == 1))[0]

        if src_candidates.size > 0 and dst_candidates.size > 0:
            i = rng.choice(src_candidates)
            j = rng.choice(dst_candidates)
            # perform swap
            h[i], h[j] = h[j], h[i]
            # evaluate C5
            new_c5 = _evaluate_c5(h)
            if new_c5 + 1e-12 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            # continue (keep swap)
        else:
            # fallback random swap
            # pick random 1 index and random 0 index
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            # swap
            h[i], h[j] = h[j], h[i]
            new_c5 = _evaluate_c5(h)
            if new_c5 + 1e-12 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
    # final refinement: small random hill climbs using our earlier _hill_climb maybe generic
    # We'll implement a simple greedy improvement similar to earlier hill climb but tailored for small deltas
    # Use final greedy moves limiting to maybe 10000 attempts or time left
    deadline = start + budget_s - 0.1
    cur_h = best_h.copy()
    cur_c5 = best_c5
    while time.time() < deadline:
        # hill climb small exchange as before
        i, j = rng.choice(n, size=2, replace=False)
        # small fractional move? But binary we only swap
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = _evaluate_c5(cur_h)
        if new_c5 + 1e-12 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), new_c5
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    # return best
    final_h = best_h
    final_c5 = float(_evaluate_c5(final_h))
    return final_h, final_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Init h
    init_vec = globals().get('initial_h_values')
    if init_vec is not None:
        h = np.asarray(init_vec, dtype=np.float64)
        n = h.size
    else:
        n = 45
        h = rng.random(n)
        target = n / 2.0
        h = h * (target / h.sum())
        h = _adjust_sum(h, target)

    target = n / 2.0
    # Ensure feasibility
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    dx = 2.0 / n

    # Main loop
    deadline = start + budget_s - 0.5
    while time.time() < deadline:
        # compute autocorrelation of h with itself
        autoc = np.correlate(h, h, mode='full')
        # index for shift s: autoc[n-1 + s] = sum h_i h_{i+s}
        # exclude shift 0 (s=0) since it gives sum of squares.
        # we need s from 1..n-1
        # find minimal autocorrelation among positive shifts
        pos_autoc = autoc[n:]  # from shift+1 onward
        s_idx = int(np.argmin(pos_autoc))  # 0-indexed
        s = s_idx + 1  # shift amount

        # Build partner array p of length n such that p[i] = h[i+s] if i+s < n else 0.
        p = np.empty(n, dtype=np.float64)
        if s < n:
            p[:n-s] = h[s:]
            p[n-s:] = 0.0
        else:
            p[:] = 0.0

        # compute source/destination scores
        src_scores = h * (1.0 - p)   # high where h=1 and partner low
        dst_scores = (1.0 - h) * p   # high where h=0 and partner high

        # Determine if we have promising move
        src_idx = int(np.argmax(src_scores))
        dst_idx = int(np.argmax(dst_scores))
        max_move = min(h[src_idx], 1.0 - h[dst_idx])
        if max_move > 1e-12 and dst_scores[dst_idx] > src_scores[src_idx] + 1e-12:
            # make a directed move
            delta = rng.uniform(0.0, max_move)
            h_new = h.copy()
            h_new[src_idx] -= delta
            h_new[dst_idx] += delta
            new_c5 = _evaluate_c5(h_new)
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5
            # accept if improvement else maybe accept prob
            if new_c5 < best_c5 or rng.random() < 0.01:
                h = h_new
                continue  # go next iteration

        # fallback random move (pairwise exchange)
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5 = _evaluate_c5(h_new)
        # Accept if improvement or SA
        # Use temperature schedule
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        # Simple linear cooling
        T0 = 0.005
        temp = T0 * (remaining / budget_s)
        accept = False
        if new_c5 < best_c5 - 1e-12:
            accept = True
        else:
            if temp > 1e-12 and rng.random() < np.exp(-(new_c5 - best_c5) / temp):
                accept = True
        if accept:

def _adjust_sum(x, target, eps=1e-12):
    """Project x onto the hyperplane sum(x)=target while keeping 0<=x<=1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < eps:
        return x
    # water-filling style redistribution
    while abs(diff) > eps:
        if diff > 0:
            # need to add mass
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < eps:
                break

    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. main search – directed + random moves
    # ------------------------------------------------------------------
    deadline = start + budget_s - 0.8  # reserve a little for final polish
    while time.time() < deadline:
        # autocorrelation (used to find the worst shift)
        autoc = np.correlate(h, h, mode="full")
        pos_autoc = autoc[n:]                 # shifts 1 … n‑1
        s_min = int(np.argmin(pos_autoc)) + 1  # shift with minimal overlap

        # partner array for this shift
        if s_min < n:
            p = np.empty(n, dtype=np.float64)
            p[:n - s_min] = h[s_min:]
            p[n - s_min:] = 0.0
        else:
            p = np.zeros(n, dtype=np.float64)

        # scores that encourage moving mass from a low‑partner position to a high‑partner one
        src_score = h * (1.0 - p)
        dst_score = (1.0 - h) * p
        src_idx = int(np.argmax(src_score))
        dst_idx = int(np.argmax(dst_score))

        moved = False
        if src_score[src_idx] > 1e-12 and dst_score[dst_idx] > 1e-12:
            max_move = min(h[src_idx], 1.0 - h[dst_idx])
            if max_move > 1e-12 and p[dst_idx] > p[src_idx] + 1e-12:
                delta = rng.uniform(0.0, max_move)
                h_new = h.copy()
                h_new[src_idx] -= delta
                h_new[dst_idx] += delta
                new_c5 = _evaluate_c5(h_new)
                if new_c5 + 1e-12 < best_c5:
                    best_h, best_c5 = h_new.copy(), new_c5
                if new_c5 < _evaluate_c5(h) - 1e-12:
                    h = h_new
                moved = True

        if not moved:
            # fallback random pairwise exchange
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 1e-12:
                continue
            delta = rng.uniform(0.0, max_move)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

            cur_c5 = _evaluate_c5(h)
            new_c5 = _evaluate_c5(h_new)

            # simulated‑annealing acceptance
            elapsed = time.time() - start
            remain = max(budget_s - elapsed, 1e-6)
            temp = 0.003 * (remain / budget_s)  # linear cooling
            accept = False
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5
                accept = True
            elif temp > 1e-12 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
                accept = True

            if accept:
                h = h_new

        # early stop if target achieved
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # 3. short SLSQP refinement (time permitting)
    # ------------------------------------------------------------------
    if time.time() < start + budget_s - 0.3:
        refined_h, refined_c5 = _slsqp_refine(best_h, budget_s - (time.time() - start))
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # ------------------------------------------------------------------
    # 4. final projection and return
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
