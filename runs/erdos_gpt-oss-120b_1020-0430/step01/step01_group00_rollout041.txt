import time
import numpy as np
from scipy import optimize


def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project a vector onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target.
    """
    lo = np.min(v) - 1.0   # ensures all v - lo >= 1 → clipped to 1 → sum = n
    hi = np.max(v)         # ensures all v - hi <= 0 → clipped to 0 → sum = 0
    for _ in range(60):    # binary search for λ
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny residual fix
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-9) & (proj < 1 - 1e-9))[0]
        if free.size:
            proj[free] -= diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C₅ for a discrete step function h on [0,2]."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every shift."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    h = x[:n]
    return np.sum(h) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]

        {"type": "eq", "fun": _eq_constraint},

        method="SLSQP",
        bounds=bounds,
        constraints=constraints,

            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = _tent_h(n)

        # Deterministic refinement.
        h_opt, c5_val = _solve_h(base_h, max_iter=400)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target_bound:
                break

        # Stochastic restarts.
        attempts = 12
        sigma = 0.15
        for _ in range(attempts):
            if time_left() < 5.0:
                break
            # Choose a random starting point.
            if np.random.rand() < 0.6:
                start = base_h + sigma * np.random.randn(n)
            else:
                start = np.random.rand(n)  # uniform in [0,1)
            start = _project_to_simplex_box(start, n / 2.0)

            h_opt, c5_val = _solve_h(start, max_iter=350)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 < target_bound:
                    break
            sigma *= 0.88  # gradual cooling

        if best_c5 < target_bound:
            break

    # Fallback in the highly unlikely event that everything failed.
    if best_h is None:
        n = 128
        best_h = _tent_h(n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
