def _gradient_step(h):
    n = len(h)
    ac = np.correlate(h, h, mode='full')
    min_idx = int(np.argmin(ac))
    shift = min_idx - (n - 1)
    # compute gradient g
    if shift == 0:
        g = 2.0 * h
    else:
        s = abs(shift)
        if shift > 0:
            g = np.empty_like(h)
            g[:] = 0.0
            g[:n-s] += h[s:]
            g[s:] += h[:n-s]
        else:
            g = np.empty_like(h)
            g[:] = 0.0
            g[:n-s] += h[s:]
            g[s:] += h[:n-s]
    # direction

    # compute max feasible step

    pos = d > 1e-12

def _project_to_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _gradient_descent_step(h, target, max_alpha_factor):
    n = h.shape[0]
    # Compute correlation with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # find worst shift (max correlation)
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # Compute gradient of correlation w.r.t h (the gradient of f = corr)
    # grad_i = 1 - (h_{i+shift} + h_{i-shift}) (if within bounds)
    grad = np.zeros_like(h)
    # base 1
    grad.fill(1.0)
    if shift != 0:
        s = abs(shift)
        # compute neighbor sum
        neighbor = np.zeros_like(h)
        neighbor[:n - s] += h[s:]
        neighbor[s:] += h[:n - s]
        grad -= neighbor
    else:
        grad -= 2.0 * h
    # Move opposite gradient to reduce corr
    # Determine feasible step size to keep within [0,1]
    pos = grad > 0  # because we subtract grad, so if grad>0, h decreases
    # Actually new_h = h - alpha * grad
    # For each i:
    # if grad_i > 0: new_h_i = h_i - alpha * grad_i >= 0 => alpha <= h_i/grad_i
    # if grad_i < 0: new_h_i = h_i - alpha * grad_i <= 1 => alpha <= (1 - h_i)/(-grad_i)
    # So compute bounds accordingly.
    pos_mask = grad > 1e-12
    neg_mask = grad < -1e-12
    if np.any(pos_mask):
        max_alpha_pos = np.min(h[pos_mask] / grad[pos_mask])
    else:
        max_alpha_pos = np.inf
    if np.any(neg_mask):
        max_alpha_neg = np.min((1.0 - h[neg_mask]) / (-grad[neg_mask]))
    else:
        max_alpha_neg = np.inf
    max_alpha = min(max_alpha_pos, max_alpha_neg)
    if max_alpha <= 0:
        return h, corr.max() * (2.0 / n)  # no move possible
    # step size factor
    alpha = max_alpha * max_alpha_factor
    h_new = h - alpha * grad
    # projection to sum constraint
    h_new = _project_to_box_sum(h_new, target)
    return h_new, None # we will compute C5 later outside

def _optimize_start(h0, time_limit):
    start = time.time()
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_metric(h)
    iter = 0
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        # alpha factor decays: maybe factor = 0.5 * (1 - progress) + 0.01 small.
        factor = max(0.01, 0.5 * (1 - progress))

        if cur_c < best_c:

            best_h = h.copy()
        iter += 1

            perturb = np.random.normal(scale=0.02, size=n)
            h = np.clip(h + perturb, 0, 1)
            # fix sum
            h = _project_to_box_sum(h, target)
    return best_h, best_c

def _c5_metric(h: np.ndarray) -> float:
    """C5 value for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(v: np.ndarray, target_sum: float, rng: np.random.Generator) -> None:
    """Add a tiny correction so that sum(v) == target_sum while staying in [0,1]."""
    diff = target_sum - v.sum()
    if abs(diff) < 1e-13:
        return
    # try to add the whole diff to a single admissible entry
    idxs = rng.permutation(v.shape[0])
    for i in idxs:
        lo, hi = 0.0, 1.0
        if lo <= v[i] + diff <= hi:
            v[i] += diff
            return
    # fallback: distribute uniformly (very small) and clip
    v += diff / v.shape[0]
    np.clip(v, 0.0, 1.0, out=v)
    # final tidy‑up
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        for i in idxs:
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                break


def _gradient_step(h: np.ndarray, target_sum: float, step_factor: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    One sub‑gradient descent step aimed at reducing the worst
    overlap (the maximal correlation with the complement).
    """
    n = h.shape[0]
    # correlation with complement – the array of overlap values (without dx)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # worst shift (largest overlap)
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # signed integer shift
    # gradient of the worst overlap w.r.t. h
    if shift == 0:
        grad = 1.0 - 2.0 * h                      # ∂/∂h_i (h_i(1‑h_i)) = 1‑2h_i
    else:
        s = abs(shift)
        neighbor = np.zeros_like(h)
        neighbor[:n - s] += h[s:]                 # h_{i+shift}
        neighbor[s:] += h[:n - s]                 # h_{i‑shift}
        grad = 1.0 - neighbor                     # 1 – (h_{i+shift}+h_{i‑shift})
    # feasible step size so that 0 ≤ h – α·grad ≤ 1
    pos = grad > 1e-12
    neg = grad < -1e-12
    if np.any(pos):
        max_up = np.min(h[pos] / grad[pos])
    else:
        max_up = np.inf
    if np.any(neg):
        max_down = np.min((1.0 - h[neg]) / (-grad[neg]))
    else:
        max_down = np.inf
    max_alpha = min(max_up, max_down)
    if max_alpha <= 0.0:
        return h                     # nothing feasible
    alpha = max_alpha * step_factor   # shrink a bit for safety
    h_new = h - alpha * grad
    np.clip(h_new, 0.0, 1.0, out=h_new)   # numerical safety
    # restore the exact sum constraint
    _adjust_sum(h_new, target_sum, rng)
    return h_new


def _optimize_start(h0: np.ndarray, budget: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform many gradient steps starting from h0 until `budget` seconds elapse.
    Returns the best h seen and its C5 value.
    """
    start = time.time()
    n = h0.shape[0]
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_metric(h)
    it = 0
    while time.time() - start < budget:
        elapsed = time.time() - start
        progress = elapsed / budget               # 0 → 1
        # step factor decays from 0.5 to ~0.02
        step_factor = max(0.02, 0.5 * (1.0 - progress))
        h = _gradient_step(h, target_sum, step_factor, rng)
        cur_c = _c5_metric(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        it += 1
        # occasional tiny random jitter to escape plateaus
        if (it & 1023) == 0:                     # every 1024 iterations
            jitter = rng.normal(scale=0.01, size=n)
            np.add(h, jitter, out=h)
            np.clip(h, 0.0, 1.0, out=h)
            _adjust_sum(h, target_sum, rng)
    return best_h, best_c
