import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x_i<=1, sum x_i = target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Objective: max correlation = C5 bound for a given step vector
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Simple sub‑gradient descent on the max‑correlation objective
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 800,
                 step0: float = 0.5, decay: float = 0.995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# Smooth‑max + SLSQP refinement
# ----------------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float, alpha: float = 30.0,
                    max_iter: int = 800) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ----------------------------------------------------------------------
# Particle Swarm Optimisation (global search)
# ----------------------------------------------------------------------
def _pso_optimize(rng: np.random.RandomState, n: int, target: float, dx: float,
                  base_vec: np.ndarray | None, time_limit: float) -> np.ndarray:
    # PSO hyper‑parameters
    n_particles = 70
    w = 0.7298
    c1 = c2 = 1.496
    v_max = 0.2

    # initialise particles
    particles = np.empty((n_particles, n))
    velocities = np.zeros_like(particles)

    # first particle = base (if provided)
    if base_vec is not None:
        particles[0] = base_vec.copy()
    else:
        particles[0] = _random_feasible(rng, n, target)

    # remaining particles = random (with occasional bias towards base)
    for i in range(1, n_particles):
        if base_vec is not None and rng.rand() < 0.3:
            # perturb base
            cand = base_vec + rng.normal(scale=0.05, size=n)
            particles[i] = _project_onto_capped_simplex(cand, target)
        else:
            particles[i] = _random_feasible(rng, n, target)

    # evaluate initial particles
    scores = np.array([_compute_c5(p, dx) for p in particles])
    pbest = particles.copy()
    pbest_scores = scores.copy()
    best_idx = int(np.argmin(pbest_scores))
    gbest = pbest[best_idx].copy()
    gbest_score = float(pbest_scores[best_idx])

    start = time.time()
    iteration = 0
    # simple iteration limit – we break early when out of time
    max_iters = 3000
    while iteration < max_iters:
        now = time.time()
        if now - start > time_limit:
            break

        r1 = rng.rand(n_particles, n)
        r2 = rng.rand(n_particles, n)

        # velocity update
        velocities = (w * velocities
                      + c1 * r1 * (pbest - particles)
                      + c2 * r2 * (gbest - particles))
        velocities = np.clip(velocities, -v_max, v_max)

        # position update + projection
        particles += velocities
        for i in range(n_particles):
            particles[i] = _project_onto_capped_simplex(particles[i], target)

        # evaluate
        scores = np.array([_compute_c5(p, dx) for p in particles])

        # personal best update
        improved = scores < pbest_scores
        if np.any(improved):
            pbest[improved] = particles[improved].copy()
            pbest_scores[improved] = scores[improved]

        # global best update
        cur_best_idx = int(np.argmin(pbest_scores))
        cur_best_score = float(pbest_scores[cur_best_idx])
        if cur_best_score < gbest_score:
            gbest_score = cur_best_score
            gbest = pbest[cur_best_idx].copy()

        # occasional re‑initialisation of worst particles to escape stagnation
        if iteration % 50 == 0 and iteration > 0:
            worst_idx = np.argsort(pbest_scores)[-5:]  # ten worst
            for idx in worst_idx:
                particles[idx] = _random_feasible(rng, n, target)
                velocities[idx] = np.zeros(n)

        iteration += 1

    return gbest

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine discretisation size and base vector (if any)
    # ------------------------------------------------------------------
    try:
        base = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base = None

    n0 = len(base) if base is not None else 0
    n = max(200, n0)                 # reasonable expressive grid
    target_sum = n / 2.0
    dx = 2.0 / n

    def upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    base_vec = None
    if base is not None:
        base_vec = upscale(base)
        base_vec = _project_onto_capped_simplex(base_vec, target_sum)

    # ------------------------------------------------------------------
    # Global search via PSO (time‑limited)
    # ------------------------------------------------------------------
    time_for_pso = budget_s * 0.70
    gbest = _pso_optimize(rng, n, target_sum, dx, base_vec, time_for_pso)

    # ------------------------------------------------------------------
    # Local refinement: gradient descent then SLSQP
    # ------------------------------------------------------------------
    time_elapsed = time.time() - start_time
    time_left = budget_s - time_elapsed
    # allocate remaining time to local refinements
    if time_left > 5.0:
        gd_steps = 800
        gbest = _gd_optimize(gbest, target_sum, max_iter=gd_steps,
                             step0=0.6, decay=0.996)
        time_elapsed = time.time() - start_time
        time_left = budget_s - time_elapsed
        if time_left > 5.0:
            # a modest number of SLSQP iterations
            gbest = _slsqp_optimize(gbest, target_sum, alpha=40.0, max_iter=600)

    # ------------------------------------------------------------------
    # Final safety projection and evaluation
    # ------------------------------------------------------------------
    best_h = _project_onto_capped_simplex(gbest, target_sum)
    best_val = _compute_c5(best_h, dx)

    # In the (unlikely) case we failed to produce a vector, fall back
    if best_h is None or np.isnan(best_val):
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
