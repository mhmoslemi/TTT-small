def delta_corr_swap(h, comp, i, j):
    n = len(h)
    delta = np.zeros(2*n-1, dtype=int)  # maybe int16
    offset = n-1
    # loop L
    # Precomputations: For each L, compute comp_iL = comp[i+L] if in range else 0, etc.
    # We'll loop over L index idx from 0 to 2n-2, with L = idx - offset.
    for idx in range(2*n-1):
        L = idx - offset
        val = 0
        p = i + L
        if 0 <= p < n:
            val -= comp[p]
        p = j + L
        if 0 <= p < n:
            val += comp[p]
        p = i - L
        if 0 <= p < n:

        elif L == (j - i):
            val += 1
        elif L == (i - j):
            val += 1
        delta[idx] = val
    return delta

def delta_corr_swap(h, comp, i, j):
    n = len(h)
    offset = n - 1
    delta = np.zeros(2*n-1, dtype=int)
    # compute contributions from d*comp:
    # Use loops over L
    for idx in range(2*n-1):
        L = idx - offset
        val = 0
        p = i + L
        if 0 <= p < n:
            val -= comp[p]
        p = j + L
        if 0 <= p < n:
            val += comp[p]
        p = i - L
        if 0 <= p < n:
            val += h[p]
        p = j - L
        if 0 <= p < n:
            val -= h[p]
        if L == 0:
            val -= 2
        elif L == j - i:
            val += 1
        elif L == i - j:
            val += 1
        delta[idx] = val
    return delta

def binary_local_search(h0, time_budget):
    n = len(h0)
    h = h0.copy()
    comp = 1 - h
    corr = np.correlate(h, comp, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    # while time left within budget:
    while time.time() - start < time_budget:
        # pick swap
        # choose random i where h[i]==1, random j where h[j]==0
        i = int(np.random.choice(np.where(h==1)[0]))
        j = int(np.random.choice(np.where(h==0)[0]))
        delta = delta_corr_swap(h, comp, i, j)
        new_corr = corr + delta
        new_max = new_corr.max()
        # Accept if improved or with small probability
        if new_max < cur_max or np.random.rand() < 0.001:
            # accept
            h[i] = 0
            h[j] = 1
            comp = 1 - h
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        # maybe after many iterations we could stop
    return best_h, best_max

import time
import numpy as np
from scipy import optimize

# ------------------ helper utilities ------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ------------------ SLSQP optimisation ------------------

def _solve_h(initial_h: np.ndarray, max_iter: int = 800) -> tuple:
    """Run one SLSQP optimisation starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ------------------ Binary local search ------------------

def _delta_corr_swap(h: np.ndarray, comp: np.ndarray, i: int, j: int) -> np.ndarray:
    """Delta of full correlation when swapping a 1 at i with a 0 at j (binary case)."""
    n = h.size
    offset = n - 1
    delta = np.zeros(2 * n - 1, dtype=int)
    # iterate over all lags
    for idx in range(2 * n - 1):
        L = idx - offset
        val = 0
        p = i + L
        if 0 <= p < n:
            val -= comp[p]
        p = j + L
        if 0 <= p < n:
            val += comp[p]
        p = i - L
        if 0 <= p < n:
            val += h[p]
        p = j - L
        if 0 <= p < n:
            val -= h[p]
        if L == 0:
            val -= 2
        elif L == j - i:
            val += 1
        elif L == i - j:
            val += 1
        delta[idx] = val
    return delta


def _binary_hill_climb(h0: np.ndarray, steps: int, target_corr: int = None) -> tuple:
    """Simple hill‑climbing on a binary vector (sum = n/2)."""
    n = h0.size
    h = h0.copy()
    comp = 1 - h
    corr = np.correlate(h, comp, mode="full")
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    for _ in range(steps):
        # pick a random 1 and a random 0
        i = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)
        delta = _delta_corr_swap(h, comp, i, j)
        new_corr = corr + delta
        new_max = new_corr.max()
        if new_max < cur_max:
            # accept the swap
            h[i] = 0
            h[j] = 1
            comp = 1 - h
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                if target_corr is not None and best_max <= target_corr:
                    return best_h, best_max
    return best_h, best_max


def _binary_search_best(n: int, time_budget: float, target_corr: int = None) -> tuple:
    """Multiple random restarts of binary hill climbing within a time budget."""
    n_half = n // 2
    start = time.time()
    best_h = None
    best_max = np.iinfo(np.int32).max
    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=int)
        h[:n_half] = 1
        np.random.shuffle(h)
        # a modest hill‑climb
        steps = 3000
        h_opt, max_corr = _binary_hill_climb(h, steps, target_corr)
        if max_corr < best_max:
            best_max = max_corr
            best_h = h_opt.copy()
            if target_corr is not None and best_max <= target_corr:
                break
    return best_h, best_max

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    TARGET_C5 = 0.3808

    # acquire optional parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    # candidate grid sizes
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    # add multiples of the base size (up to about 256)
    base_sizes = [128, 160, 200, 240, 256]
    for n in base_sizes:
        if n <= 300:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # main loop over grid sizes
    for n in n_candidates:
        if time.time() - start_time > budget_s - 5:
            break  # keep a safety margin

        # build a starting continuous function for this size
        if init_h is not None and init_h.size != n:
            base_h = _upscale_h(init_h, n)
        elif init_h is not None:
            base_h = init_h.copy()
        else:
            base_h = np.full(n, 0.5)

        # ensure feasibility
        base_h = _project_to_feasible(base_h, n)

        # -------------------------------------------------
        # 1) binary search (takes roughly 30% of the time left for this n)
        # -------------------------------------------------
        time_left = budget_s - (time.time() - start_time)
        bin_time = min(time_left * 0.3, 120.0)   # at most 2 minutes per size
        target_corr = int((TARGET_C5 * n) / 2 + 0.5)   # integer target for early exit
        bin_h, bin_corr = _binary_search_best(n, bin_time, target_corr)
        if bin_h is not None:
            c5_bin = bin_corr * (2.0 / n)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = bin_h.astype(float)
                best_n = n
                if best_c5 < TARGET_C5:
                    return best_h.tolist(), float(best_c5), int(best_n)

        # -------------------------------------------------
        # 2) continuous optimisation seeded from the best binary (or from base)
        # -------------------------------------------------
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5:
            break
        cont_time = min(time_left * 0.6, 300.0)   # at most 5 minutes per size
        end_cont = time.time() + cont_time
        attempts = 0
        while time.time() < end_cont:
            attempts += 1
            # start from binary best if we have one, otherwise randomised base
            if bin_h is not None:
                start_h = bin_h.astype(float) + 0.2 * np.random.randn(n)
            else:
                start_h = base_h + 0.3 * np.random.randn(n)
            start_h = _project_to_feasible(start_h, n)
            h_opt, c5_val = _solve_h(start_h, max_iter=1200)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n
                if best_c5 < TARGET_C5:
                    return best_h.tolist(), float(best_c5), int(best_n)
        # end of this size’s search

    # fallback: simple half‑one pattern
    if best_h is None:
        n = 128
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)], dtype=float)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
