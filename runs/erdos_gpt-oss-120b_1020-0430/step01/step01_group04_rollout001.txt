def project_feasible(v, target):
    # v is a 1d numpy array
    # find lambda such that sum(clip(v - lam, 0,1)) = target
    lo = -2.0

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.special import logsumexp
from functools import partial

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2] with spacing dx=2/n."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5_alpha(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max_k overlap using log-sum-exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def sum_constraint_func(h: np.ndarray, target: float) -> float:
    """Equality constraint: sum(h) = target."""
    return np.sum(h) - target


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 ≤ h_i ≤ 1  and  sum(h) = target
    Simple binary search for a Lagrange multiplier λ such that
        sum(clip(v_i - λ, 0, 1)) = target.
    """
    lo, hi = -10.0, 10.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(v - lam, 0.0, 1.0)
    # tiny numerical adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.shape[0]
        h = np.clip(h, 0.0, 1.0)
    return h


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape (min(x,2-x)) sampled on n points, renormalised exactly."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (box & sum constraints)."""
    v = rng.random(n)
    v = v * ((n / 2.0) / v.sum())
    v = np.clip(v, 0.0, 1.0)
    diff = (n / 2.0) - v.sum()
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v


def de_obj(v: np.ndarray, target: float) -> float:
    """Objective for differential evolution: C5 after projection."""
    h = project_to_feasible(v, target)
    return compute_c5(h)


def local_refine(h0: np.ndarray) -> np.ndarray | None:
    """Two‑stage SLSQP refinement starting from feasible h0."""
    n = h0.shape[0]
    target = n / 2.0
    cons = {"type": "eq", "fun": partial(sum_constraint_func, target=target)}
    bounds = [(0.0, 1.0)] * n

    # Stage 1 – soft objective, low α
    obj1 = partial(softmax_c5_alpha, alpha=5.0)
    res1 = minimize(
        obj1,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res1.success:
        return None

    # Stage 2 – sharper α
    obj2 = partial(softmax_c5_alpha, alpha=30.0)
    res2 = minimize(
        obj2,
        res1.x,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res2.success:
        return None

    h_opt = np.clip(res2.x, 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - target) > 1e-9:
        h_opt = h_opt + (target - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # retrieve parent if available
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    # candidate grid sizes – modest but enough resolution
    candidate_ns = [80, 100, 120, 150, 180]

    for n in candidate_ns:
        if time.time() > time_limit:
            break

        target_sum = n / 2.0

        # Prepare a few deterministic starts for quick comparison
        tri = generate_triangular(n)
        c5_tri = compute_c5(tri)
        if c5_tri < best_c5:
            best_h, best_c5, best_n = tri.copy(), c5_tri, n

        if parent_h is not None:
            if parent_n == n:
                parent_start = parent_h.copy()
            else:
                parent_start = upsample(parent_h, n)
            c5_parent = compute_c5(parent_start)
            if c5_parent < best_c5:
                best_h, best_c5, best_n = parent_start.copy(), c5_parent, n

        # -----------------------------------------------------------------
        # Global search with Differential Evolution
        # -----------------------------------------------------------------
        bounds = [(0.0, 1.0)] * n
        # adapt population size / iteration count to remaining time
        remaining = time_limit - time.time()
        max_iter = 200
        if remaining < 30.0:
            max_iter = 30
        if n >= 150:
            max_iter = min(max_iter, 150)

        de_fun = partial(de_obj, target=target_sum)
        de_seed = (seed + n) % (2**32 - 1)

        try:
            result = differential_evolution(
                de_fun,
                bounds,
                strategy="best1bin",
                maxiter=max_iter,
                popsize=10,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=de_seed,
                polish=False,
                init="random",
                updating="deferred",
                disp=False,
            )
        except Exception:
            result = None

        if result is not None and result.success:
            h_candidate = project_to_feasible(result.x, target_sum)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h, best_c5, best_n = h_candidate.copy(), c5_candidate, n

            # -------------------------------------------------------------
            # Local refinement of the DE result
            # -------------------------------------------------------------
            if time.time() < time_limit:
                refined = local_refine(h_candidate)
                if refined is not None:
                    c5_ref = compute_c5(refined)
                    if c5_ref < best_c5:
                        best_h, best_c5, best_n = refined.copy(), c5_ref, n

        # -------------------------------------------------------------
        # Small hill‑climbing steps around current best (if any)
        # -------------------------------------------------------------
        if best_h is not None and time.time() < time_limit:
            # 5 quick perturbation–refine cycles
            for _ in range(5):
                if time.time() > time_limit:
                    break
                pert = best_h + rng.normal(scale=0.07, size=best_n)
                pert = np.clip(pert, 0.0, 1.0)
                pert = project_to_feasible(pert, best_n / 2.0)
                refined = local_refine(pert)
                if refined is not None:
                    c5_ref = compute_c5(refined)
                    if c5_ref < best_c5:
                        best_h, best_c5, best_n = refined.copy(), c5_ref, best_n

    # -----------------------------------------------------------------
    # Fallback in the unlikely case nothing succeeded
    # -----------------------------------------------------------------
    if best_h is None:
        n = 80
        best_h = generate_triangular(n)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final sanity: enforce integral exactly
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
