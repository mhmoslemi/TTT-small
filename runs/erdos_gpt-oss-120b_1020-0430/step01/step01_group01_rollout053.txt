def _gd_momentum_optimize(h0, target, max_iter=2000, step0=0.5, beta=0.9, decay=0.998, eps=1e-9):
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    n = h.shape[0]
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # find shifts close to max
        # using binary mask
        # tolerance relative maybe 1e-12 absolute
        tol = 1e-8
        mask = corr >= max_corr - tol
        # note: may include many indices near max due to floating point; restrict to up to, say, 5 indices
        # find indices
        idxs = np.where(mask)[0]
        # If too many, pick first few
        if idxs.size > 5:
            idxs = idxs[:5]
        g = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n - 1)
            g += _gradient_for_shift(h, shift)
        g = g / idxs.size
        # momentum update
        v = beta * v + (1 - beta) * g

        h = _project_onto_capped_simplex(h, target)
        # decay step
        step *= decay
        # optional early stop if step too small
        if step < 1e-6:
            break
    return h

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} by bisection on the dual variable."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation for a specific shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float, alpha: float = 30.0,
                    max_iter: int = 1000) -> np.ndarray:
    """Refine using SLSQP on a smooth approximation of the max."""
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


def _subgradient(h: np.ndarray) -> np.ndarray:
    """A (sub)gradient for the max‑correlation objective (averaged over ties)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    # tolerance to capture ties (absolute)
    eps = 1e-12
    mask = corr >= maxc - eps
    idxs = np.where(mask)[0]
    # keep at most a few ties to limit cost
    if idxs.size > 5:
        idxs = idxs[:5]
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        g += _gradient_for_shift(h, shift)
    return g / max(1, idxs.size)


def _gd_momentum_optimize(h0: np.ndarray, target: float,
                          max_iter: int = 1500, step0: float = 0.7,
                          beta: float = 0.85, decay: float = 0.999) -> np.ndarray:
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        g = _subgradient(h)
        v = beta * v + (1.0 - beta) * g
        h = h - step * v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
        if step < 1e-6:
            break
    return h


def _local_swap_search(h: np.ndarray, target: float, dx: float,
                       rng: np.random.RandomState, steps: int = 500) -> np.ndarray:
    """Simple hill‑climbing that swaps mass between two coordinates while preserving the sum."""
    best = h.copy()
    best_val = _compute_c5(best, dx)
    n = h.shape[0]
    for _ in range(steps):
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = best[i], best[j]
        lo = max(-hi, hj - 1.0)
        hi_delta = min(1.0 - hi, hj)
        if hi_delta <= lo:
            continue
        delta = rng.uniform(lo, hi_delta)
        new = best.copy()
        new[i] = hi + delta
        new[j] = hj - delta
        val = _compute_c5(new, dx)
        if val < best_val:
            best, best_val = new, val
    return best


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h on [0,2] that (hopefully) improves the Erdős C5 bound."""
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Load parent if available
    # ------------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    # ------------------------------------------------------------------
    # Choose discretisation size (larger than the parent to give more freedom)
    # ------------------------------------------------------------------
    n_base = len(base_h) if base_h is not None else 0
    n = max(300, n_base)          # 300 points gives a good trade‑off
    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Utility to upscale a parent vector to the current resolution
    # ------------------------------------------------------------------
    def upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ------------------------------------------------------------------
    # Initialise best solution
    # ------------------------------------------------------------------
    if base_h is not None:
        best_h = upscale(base_h)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = None
        best_val = float('inf')

    # ------------------------------------------------------------------
    # Main optimisation loop (time‑aware)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() - start_time < budget_s * 0.92:
        # ----- generate starting point ---------------------------------
        if best_h is not None and rng.rand() < 0.2:
            h0 = best_h + rng.normal(scale=0.02, size=n)
        else:
            h0 = _random_feasible(rng, n, target_sum)

        # ----- momentum gradient descent --------------------------------
        h_opt = _gd_momentum_optimize(h0, target_sum,
                                      max_iter=1800,
                                      step0=0.8,
                                      beta=0.85,
                                      decay=0.9995)

        # ----- occasional SLSQP polishing -------------------------------
        if restart % 4 == 0:
            h_opt = _slsqp_optimize(h_opt, target_sum,
                                     alpha=50.0, max_iter=1000)

        # ----- local swap hill‑climbing --------------------------------
        h_opt = _local_swap_search(h_opt, target_sum, dx, rng, steps=800)

        # ----- evaluate ------------------------------------------------
        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        restart += 1

    # ------------------------------------------------------------------
    # Final safety projection and evaluation
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
    else:
        best_h = _project_onto_capped_simplex(best_h, target_sum)

    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
