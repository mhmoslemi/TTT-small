    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.shape[0]
    for t in range(max_iter):
        if time.time() - start_time > budget - 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        idxs = np.where(corr == max_corr)[0]
        # compute gradient average
        g = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n - 1)
            # compute (1 - h_{i+shift}) - h_{i-shift}
            h_plus = shift_arr(h, shift)
            h_minus = shift_arr(h, -shift)
            g += (1.0 - h_plus) - h_minus

            best_val = val
            best_h = h.copy()
        # early stop if best_val less than target
        if best_val <= 0.3808:
            break
    return best_h, best_val

import time, random, numpy as np

def project_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while keeping sum(x)=target."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def shift_arr(arr: np.ndarray, shift: int) -> np.ndarray:
    """Return arr shifted by `shift` positions (zero‑filled outside)."""
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if shift >= 0:
        if shift < n:
            out[shift:] = arr[:n - shift]
    else:
        s = -shift
        if s < n:
            out[:n - s] = arr[s:]
    return out


def subgradient_descent(
    h0: np.ndarray,
    max_iter: int,
    step0: float,
    target_sum: float,
    budget: float,
    start_time: float,
    stop_val: float = 0.3808,
):
    """Projected sub‑gradient descent on  max_k <h,1‑h(·+k)>."""
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.shape[0]
    for t in range(max_iter):
        if time.time() - start_time > budget - 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mx = corr.max()
        idxs = np.where(corr == mx)[0]
        g = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            h_plus = shift_arr(h, k)
            h_minus = shift_arr(h, -k)
            g += (1.0 - h_plus) - h_minus
        g /= len(idxs)
        step = step0 / np.sqrt(t + 1.0)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_bounds_sum(h_new, target_sum)
        h = h_new
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
            if best_val <= stop_val:
                break
    return best_h, best_val


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_bounds_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_bounds_sum(h, n / 2.0)


def make_rect(n: int) -> np.ndarray:
    """Block of length 1 centred at 1."""
    start, end = 0.5, 1.5
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = ((pts >= start) & (pts < end)).astype(float)
    return project_bounds_sum(h, n / 2.0)


def upsample_initial(old_h: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of older solution to length n, then project."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return project_bounds_sum(new_h, n / 2.0)
