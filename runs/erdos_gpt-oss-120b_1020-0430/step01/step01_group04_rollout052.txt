def normalize_h(h):
    target = h.shape[0] / 2.0
    total = h.sum()
    if total != 0:
        h = h * (target / total)
    # clip to [0,1] limits and adjust sum if needed
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def normalize_h(h: np.ndarray) -> np.ndarray:
    """Scale and clip to satisfy sum(h)=len(h)/2 and 0≤h≤1."""
    n = h.shape[0]
    target = n / 2.0
    if np.sum(h) == 0:
        h = np.full_like(h, target / n)
    else:
        h = h * (target / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, scaled to the integral constraint."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    return normalize_h(tri)


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    return normalize_h(a)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger resolution, with renormalisation."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    return normalize_h(up)


# ------------------------------------------------------------
# Optimisation phases
# ------------------------------------------------------------
def slsqp_optimize(start: np.ndarray, n: int, rng: np.random.Generator,
                   deadline: float) -> tuple:
    """Two‑stage SLSQP with increasing soft‑max sharpness."""
    target_sum = n / 2.0
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n

    h_cur = start.copy()
    alphas = [5.0, 15.0, 30.0, 60.0]
    for a in alphas:
        if time.time() > deadline:
            break
        res = minimize(
            lambda h: softmax_c5(h, alpha=a),
            h_cur,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-10, "disp": False},
        )
        if not res.success:
            # ignore failure – keep current point
            pass
        h_cur = np.clip(res.x, 0.0, 1.0)
    return h_cur, compute_c5(h_cur)


def hill_climb(initial_h: np.ndarray, cur_best: float,
               deadline: float, rng: np.random.Generator) -> tuple:
    """Stochastic local search with occasional targeted moves."""
    h = initial_h.copy()
    best_h = h.copy()
    best_val = cur_best
    n = h.shape[0]

    step = 0.3          # maximal transfer per move
    temperature = 0.002

    itr = 0
    while time.time() < deadline:
        itr += 1
        # slowly shrink step and temperature
        if itr % 4000 == 0:
            step = max(0.01, step * 0.85)
            temperature = max(1e-6, temperature * 0.85)

        # ----- targeted move on the currently worst shift -----
        if itr % 47 == 0:
            conv = np.correlate(h, 1.0 - h, mode="full")
            k_idx = int(np.argmax(conv))
            k = k_idx - (n - 1)          # signed shift
            if k >= 0:
                i_start, i_end = 0, n - k
            else:
                i_start, i_end = -k, n
            if i_end > i_start:
                i = rng.integers(i_start, i_end)
                j = i + k
                hi, hj = h[i], h[j]
                # move mass from the larger entry to the smaller one
                if hi > hj:
                    delta_max = min(step, hi, 1.0 - hj)
                    if delta_max > 0:
                        delta = rng.uniform(0.0, delta_max)
                        cand = h.copy()
                        cand[i] = hi - delta
                        cand[j] = hj + delta
                    else:
                        continue
                elif hj > hi:
                    delta_max = min(step, hj, 1.0 - hi)
                    if delta_max > 0:
                        delta = rng.uniform(0.0, delta_max)
                        cand = h.copy()
                        cand[i] = hi + delta
                        cand[j] = hj - delta
                    else:
                        continue
                else:
                    continue
        else:
            # ----- generic random two‑index exchange -----
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            hi, hj = h[i], h[j]
            delta_min = max(-hi, hj - 1.0, -step)
            delta_max = min(1.0 - hi, hj, step)
            if delta_min >= delta_max:
                continue
            delta = rng.uniform(delta_min, delta_max)
            cand = h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

        val = compute_c5(cand)

        # Metropolis acceptance
        if val < best_val - 1e-12 or rng.random() < np.exp(-(val - best_val) / temperature):
            h = cand
            if val < best_val:
                best_val = val
                best_h = cand.copy()
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95   # safety margin

    # target resolution (balance quality / speed)
    n = 200

    # collect diverse starting vectors
    starts = []

    # 1. parent if supplied
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n != n:
            starts.append(upsample(parent_h, n))
        else:
            starts.append(parent_h.copy())
    except NameError:
        parent_h = None

    # 2. classic triangular shape
    starts.append(generate_triangular(n))

    # 3. a few random feasible vectors
    for _ in range(3):
        starts.append(random_start(n, rng))

    best_h = None
    best_val = np.inf

    # ---- SLSQP phase on each start ----
    for s in starts:
        if time.time() > time_limit:
            break
        s = normalize_h(s)
        h_opt, val_opt = slsqp_optimize(s, n, rng, time_limit)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # fallback if nothing succeeded
    if best_h is None:
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)], dtype=float)
        best_val = compute_c5(best_h)

    # ---- Local stochastic refinement ----
    if time.time() < time_limit - 2.0:
        refined_h, refined_val = hill_climb(best_h, best_val, time_limit, rng)
        if refined_val < best_val:
            best_h, best_val = refined_h, refined_val

    # final safety normalisation
    best_h = normalize_h(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
