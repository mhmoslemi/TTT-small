def hill_climb_binary(h, rng, max_iters=10000):
    n = h.size
    w = n // 2
    best_h = h.copy()
    best_c = compute_c5_bin(best_h)
    for it in range(max_iters):
        # choose i (1) j (0)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        c = compute_c5_bin(new_h)
        if c < best_c - 1e-12:  # improvement
            best_h, best_c = new_h, c

    best_c = compute_c5_bin(best_h)

        if c2 < best_c:
            best_h, best_c = h2, c2

def subgradient_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    # find shifts where max_corr occurs
    idxs = np.where(np.abs(corr - max_corr) < 1e-12)[0]  # tolerance
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n-1)  # shift
        if k >= 0:
            A = np.concatenate([1 - h[k:], np.zeros(k)])
            B = np.concatenate([np.zeros(k), h[:n-k]])
        else:
            kk = -k
            A = np.concatenate([np.zeros(kk), 1 - h[:n-kk]])
            B = np.concatenate([h[kk:], np.zeros(kk)])
        grad += (A - B)
    grad /= len(idxs)
    return grad

def subgradient_descent_with_budget(h0, target, rng, start_time, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = len(h)
    it = 0
    while time.time() - start_time < time_limit:
        # evaluate and subgradient
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        c = max_corr * (2.0 / n)
        # track best
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # find active shifts
        active = np.where(corr >= max_corr - 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in active:
            k = idx - (n-1)
            if k >= 0:
                # A: for i=0..n-1-k
                grad[:n-k] += 1 - h[k:]
                # B: for i=k..n-1
                grad[k:] -= h[:n-k]

                # A: for i=kk..n-1
                grad[kk:] += 1 - h[:n-kk]
                # B: for i=0..n-kk-1
                grad[:n-kk] -= h[kk:]

        # step size schedule
        step = 0.5 / np.sqrt(it + 1)
        # update
        h = h - step * grad
        # project onto [0,1] and sum
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

            for _ in range(2000):

                diff = target - h.sum()
        it += 1
        # check time
        if time.time() - start_time > time_limit:

import time
import numpy as np

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_sum_bounds(h, target, rng, max_iter=2000):
    """Adjust h (already in [0,1]) so that sum(h)=target by redistributing the difference."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_continuous(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb using the perturbation above."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    # final projection to guarantee exact sum
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, target, rng)
    return best_h, best_c


# ------------------------------------------------------------
# Subgradient descent for the max‑overlap objective
# ------------------------------------------------------------
def subgradient_descent_iter(h0, target, rng, max_iters=3000, step0=0.5):
    """Run a bounded‑subgradient descent for a fixed number of iterations."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        # correlation and maximal value
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = corr.max()
        c = max_corr * (2.0 / n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # active shifts (those attaining the max)
        active = np.where(corr >= max_corr - 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in active:
            k = idx - (n - 1)          # shift value
            if k >= 0:
                # A part
                grad[: n - k] += 1 - h[k:]
                # B part
                grad[k:] -= h[: n - k]
            else:
                kk = -k
                grad[kk:] += 1 - h[: n - kk]
                grad[: n - kk] -= h[kk:]
        grad /= len(active)
        # step size decays like 1/sqrt(t)
        step = step0 / np.sqrt(it + 1)
        h = h - step * grad
        # enforce box constraints
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum via small redistributions
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # only a few correction rounds are needed (diff stays tiny)
            for _ in range(50):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return best_h, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()
