import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto {0≤x≤1, sum(x)=target} via bisection on the dual."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, dim: int, target: float) -> np.ndarray:
    v = rng.rand(dim)
    return _project_onto_capped_simplex(v, target)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the correlation for the shift attaining the max."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

            x = _project_onto_capped_simplex(x, target_half)
            step *= 0.997

            if it % 30 == 0:
                h_cur = _full_from_half(x)
                val = _compute_c5(h_cur, dx)
                if val < best_val:
                    best_val = val
                    best_h = h_cur.copy()
                    best_x = x.copy()
                    if best_val < 0.3808:
                        break

        # ------- occasional simple random local search -------------------------
        for _ in range(200):
            i, j = rng.randint(0, half, size=2)
            eps = rng.uniform(-0.015, 0.015)
            x_new = x.copy()
            x_new[i] = np.clip(x_new[i] + eps, 0.0, 1.0)
            x_new[j] = np.clip(x_new[j] - eps, 0.0, 1.0)
            x_new = _project_onto_capped_simplex(x_new, target_half)
            val = _compute_c5(_full_from_half(x_new), dx)
            if val < best_val:
                best_val = val
                best_h = _full_from_half(x_new).copy()
                best_x = x_new.copy()
                if best_val < 0.3808:
                    break

        # ------- SLSQP fine‑tuning (smooth max) --------------------------------
        if time.time() - start > budget_s * 0.5 and time.time() - start < budget_s * 0.95:
            try:
                alpha = 80.0

                def obj(v):
                    corr = np.correlate(v, 1.0 - v, mode="full")
                    return _smooth_max(corr, alpha)

                cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target_full},)
                bounds = [(0.0, 1.0)] * n
                res = minimize(
                    obj,
                    best_h,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 1200, "ftol": 1e-12, "disp": False},
                )
                if res.success:
                    cand = np.clip(res.x, 0.0, 1.0)
                    cand = _project_onto_capped_simplex(cand, target_full)
                    val = _compute_c5(cand, dx)
                    if val < best_val:
                        best_val = val
                        best_h = cand.copy()
                        best_x = cand[:half].copy()
            except Exception:
                pass

        # early exit if we already beat the target bound
        if best_val < 0.3808:
            break

    # ----------------------------------------------------------------------
    # Final safety projection and evaluation
    # ----------------------------------------------------------------------
    best_h = _project_onto_capped_simplex(best_h, target_full)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
