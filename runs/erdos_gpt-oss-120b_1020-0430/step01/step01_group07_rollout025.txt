import time, math, random, numpy as np

def _c5_metric_periodic(h):
    n = h.shape[0]
    H = np.fft.rfft(h)
    G = np.fft.rfft(1.0 - h)
    corr = np.fft.irfft(H * np.conj(G), n=n)
    return corr.max() * (2.0 / n)

def _bias_move_indices(h, lag, rng):
    n = h.shape[0]
    # contributions for current lag
    idx = np.arange(n)
    targets = (idx + lag) % n
    w = h * (1.0 - h[targets])
    total = w.sum()
    if total > 0:
        donor = rng.choice(n, p=w/total) # will lose mass
    else:
        donor = rng.integers(n)
    # recipient from low contributions
    w_rec = w.max() - w
    rec_total = w_rec.sum()
    if rec_total > 0:
        recipient = rng.choice(n, p=w_rec/rec_total)
    else:
        recipient = rng.integers(n)
    if recipient == donor:
        # pick random different
        other = rng.integers(n-1)
        if other >= donor:
            other += 1
        recipient = other
    return recipient, donor

def _anneal_periodic(start_h, max_seconds, bias_prob=0.5, max_step=0.12, T0=0.04, T_end=1e-6):
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric_periodic(best_h)
    cur_c = best_c
    start = time.time()
    rng = np.random.default_rng()
    while True:
        elapsed = time.time() - start
        if elapsed >= max_seconds:
            break
        # compute correlation and lag
        # use FFT
        H = np.fft.rfft(h)
        G = np.fft.rfft(1.0 - h)
        corr = np.fft.irfft(H * np.conj(G), n=n)
        max_idx = int(np.argmax(corr))
        lag = max_idx  # shift amount (0..n-1)
        # pick indices
        if rng.random() < bias_prob:
            i_recv, j_don = _bias_move_indices(h, lag, rng)
        else:
            i_recv, j_don = rng.integers(n, size=2)
            while i_recv == j_don:
                j_don = rng.integers(n)
        hi = h[i_recv]
        hj = h[j_don]
        low = max(-hi, hj - 1.0)      # i receives delta, j loses delta
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # sample delta, bias towards positive (transfer from donor to receiver)
        # choose a delta within [0, high] with max_step cap
        if bias_prob > 0 and rng.random() < bias_prob:
            max_allowed = min(high, max_step)
            if max_allowed <= 0:
                continue
            delta = rng.random() * max_allowed
        else:
            # generic move: allow full range but cap step size in both directions
            low_cap = max(low, -max_step)
            high_cap = min(high, max_step)
            if low_cap < high_cap:
                delta = low_cap + rng.random() * (high_cap - low_cap)
            else:
                delta = low + rng.random() * (high - low)
        # apply move
        h[i_recv] = hi + delta
        h[j_don] = hj - delta
        new_c = _c5_metric_periodic(h)
        dC = new_c - cur_c
        # temperature schedule
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        if dC < 0 or rng.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i_recv] = hi
            h[j_don] = hj
    return best_h, best_c

def _make_feasible(h, n):
    target = n / 2.0
    # scale sum to target
    cur = h.sum()
    if cur != 0:
        h = h * (target / cur)
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        # adjust randomly
        idx = np.random.randint(n)
        # ensure within bounds
        lo = max(0.0, h[idx] + deficit)
        hi = min(1.0, h[idx] + deficit)
        if lo == hi:
            # find another index
            for idx2 in np.random.permutation(n):
                if 0.0 <= h[idx2] + deficit <= 1.0:
                    idx = idx2
                    break
        h[idx] += deficit
    return h

import time, math, random, numpy as np

def _c5_metric(h):
    """Periodic max overlap metric."""
    n = h.shape[0]
    H = np.fft.rfft(h)
    G = np.fft.rfft(1.0 - h)
    corr = np.fft.irfft(H * np.conj(G), n=n)
    return corr.max() * (2.0 / n)


def _make_feasible(arr, n):
    """Scale/clip to satisfy sum_i arr_i = n/2 and 0≤arr_i≤1."""
    target = n / 2.0
    cur = arr.sum()
    if cur != 0:
        arr = arr * (target / cur)
    np.clip(arr, 0.0, 1.0, out=arr)
    deficit = target - arr.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= arr[idx] + deficit <= 1.0:
                arr[idx] += deficit
                break
    return arr


def _initial_vector():
    """Create a feasible start from the provided construction or a uniform half‑density."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        n = max(100, ((n0 + 9) // 10) * 10)
        if n % 2:
            n += 1
        if n != n0:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x, x0, h0)
        else:
            h = h0.copy()
    else:
        n = 120
        h = np.full(n, 0.5, dtype=np.float64)
    return _make_feasible(h, n)


def _bias_pair(h, lag, rng):
    """Select a receiver (i) and a donor (j) biased by the worst‑lag contributions."""
    n = h.size
    idx = np.arange(n)
    w = h * (1.0 - h[(idx + lag) % n])          # contribution of each index to the worst lag
    total = w.sum()
    if total > 0:
        donor = rng.choice(n, p=w / total)     # will lose mass
    else:
        donor = rng.integers(n)
    w_rec = w.max() - w                         # low‑contribution indices
    rec_total = w_rec.sum()
    if rec_total > 0:
        receiver = rng.choice(n, p=w_rec / rec_total)
    else:
        receiver = rng.integers(n)
    if receiver == donor:
        other = rng.integers(n - 1)
        if other >= donor:
            other += 1
        receiver = other
    return receiver, donor


def _anneal_periodic(start_h, max_seconds, select_bias=0.5,
                    max_step=0.12, T0=0.04, T_end=1e-6):
    """Simulated‑annealing optimisation for the periodic metric."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c
    start = time.time()
    rng = np.random.default_rng()
    while True:
        elapsed = time.time() - start
        if elapsed >= max_seconds:
            break
        # current correlation and worst lag
        H = np.fft.rfft(h)
        G = np.fft.rfft(1.0 - h)
        corr = np.fft.irfft(H * np.conj(G), n=n)
        lag = int(np.argmax(corr))

        # choose pair
        if rng.random() < select_bias:
            i, j = _bias_pair(h, lag, rng)      # i receives, j loses
        else:
            i, j = rng.integers(n, size=2)
            while i == j:
                j = rng.integers(n)
        hi, hj = h[i], h[j]

        # feasible delta interval for i (receiver) and j (donor)
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # propose delta
        if rng.random() < select_bias:          # biased move → positive transfer
            max_allowed = min(high, max_step)
            if max_allowed <= 0:
                continue
            delta = rng.random() * max_allowed
        else:                                   # generic move, limited step size
            low_cap = max(low, -max_step)
            high_cap = min(high, max_step)
            if low_cap < high_cap:
                delta = low_cap + rng.random() * (high_cap - low_cap)
            else:
                delta = low + rng.random() * (high - low)

        # apply
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        if dC < 0 or rng.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c
