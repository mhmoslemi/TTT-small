    n = len(h)
    i, j = rng.choice(n, 2, replace=False)
    # Random direction
    if rng.random() < 0.5:
        # move from i to j
        max_move = min(h[i], 1 - h[j])

        max_move = min(h[j], 1 - h[i])

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        candidates = np.where(mask)[0]
        per = diff / len(candidates)
        for i in candidates:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)
```

import time
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Exact max overlap integral for a step vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the feasible set 0<=x<=1 and sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # redistribute remaining mass iteratively
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _final_hill_climb(h: np.ndarray, c5: float, deadline: float, rng: np.random.Generator):
    """Deterministic pairwise improvement until time runs out."""
    n = h.size
    while time.time() < deadline:
        # pick distinct indices without using choice for speed
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # try moving from i to j
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 1e-12:
            delta = rng.random() * max_move
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_c5 = _evaluate_c5(h_new)
            if new_c5 + 1e-12 < c5:
                h, c5 = h_new, new_c5
                continue
        # try opposite direction
        max_move = min(h[j], 1.0 - h[i])
        if max_move > 1e-12:
            delta = rng.random() * max_move
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            new_c5 = _evaluate_c5(h_new)
            if new_c5 + 1e-12 < c5:
                h, c5 = h_new, new_c5
    return h, c5

def run(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Search for a step function h on [0,2] minimizing the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine discretisation size and initialise h
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_arr = np.asarray(init_vec, dtype=np.float64)
        n0 = init_arr.size
        factor = 4                           # increase resolution (45*4 = 180)
        n = n0 * factor
        h0 = np.repeat(init_arr, factor)     # keeps total mass exact
    else:
        # fallback: 180 points, random start respecting the integral
        n = 180
        h0 = rng.random(n)
    target = n / 2.0
    h0 = _adjust_sum(np.clip(h0, 0.0, 1.0), target)

    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ------------------------------------------------------------------
    # 2. Simulated annealing / diversified local search
    # ------------------------------------------------------------------
    # Reserve a small tail for final deterministic hill climb
    deadline = start_time + budget_s - 0.5
    total_time = deadline - start_time
    # Move type probabilities
    move_probs = np.array([0.45, 0.25, 0.15, 0.15])  # pairwise, guided, swap, global

    while time.time() < deadline:
        time_left = deadline - time.time()
        # linear temperature schedule (scale 0..0.02)
        T = max(1e-9, 0.02 * (time_left / total_time))

        move_type = rng.choice(4, p=move_probs)

        if move_type == 0:
            # ----- pairwise mass transfer -----
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # random direction
            if rng.random() < 0.5:
                max_move = min(cur_h[i], 1.0 - cur_h[j])
                if max_move <= 1e-12:
                    continue
                delta = rng.random() * max_move
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                max_move = min(cur_h[j], 1.0 - cur_h[i])
                if max_move <= 1e-12:
                    continue
                delta = rng.random() * max_move
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta

        elif move_type == 1:
            # ----- guided move: improve the worst shift -----
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            worst_idx = int(np.argmax(corr))
            shift = worst_idx - (n - 1)
            # avoid zero shift
            if shift == 0:
                # find the next worst shift
                sorted_idx = np.argsort(corr)[::-1]
                for idx in sorted_idx:
                    s = idx - (n - 1)
                    if s != 0:
                        worst_idx = idx
                        shift = s
                        break
            if shift > 0:
                i = rng.integers(0, n - shift)
                t = i + shift
            else:
                s = -shift
                i = rng.integers(s, n)
                t = i - s
            source = rng.integers(n)
            max_move = min(cur_h[source], 1.0 - cur_h[t])
            if max_move <= 1e-12:
                continue
            delta = rng.random() * max_move
            new_h = cur_h.copy()
            new_h[source] -= delta
            new_h[t] += delta

        elif move_type == 2:
            # ----- swap two entries (reordering) -----
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if cur_h[i] == cur_h[j]:
                continue
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]

        else:
            # ----- global random perturbation + projection -----
            scale = 0.04
            new_h = cur_h + rng.normal(scale=scale, size=n)
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = _adjust_sum(new_h, target)

        # evaluate candidate
        new_c5 = _evaluate_c5(new_h)

        # accept according to Metropolis criterion
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5

    # ------------------------------------------------------------------
    # 3. Final deterministic hill climbing
    # ------------------------------------------------------------------
    final_deadline = start_time + budget_s - 0.1
    best_h, best_c5 = _final_hill_climb(best_h, best_c5, final_deadline, rng)

    # ------------------------------------------------------------------
    # 4. Return feasible result
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
