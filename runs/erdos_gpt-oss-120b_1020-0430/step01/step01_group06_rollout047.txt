def _next_pow_two(x: int) -> int:
    """return the smallest power of two >= x"""
    return 1 << (x - 1).bit_length()


def compute_c5_max(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function h."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    nfft = _next_pow_two(2 * n - 1)
    prod = np.fft.fft(h, nfft) * np.fft.fft(rev, nfft)
    corr = np.fft.ifft(prod).real[: 2 * n - 1]
    return float(corr.max() * (2.0 / n))


def _project_to_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while keeping 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Simple iterative uniform correction – enough because changes are tiny.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        add = diff / h.size
        h += add
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

        for idx in range(best_n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    final_c5 = compute_c5_max(best_h)
    return (best_h.tolist(), float(final_c5), int(best_n))
