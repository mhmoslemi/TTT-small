def local_refine(h, target_sum, time_budget, start_time):
    best = compute_c5(h)
    best_h = h.copy()
    n = len(h)
    rng = np.random.default_rng()
    while time.time() - start_time < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best:
            best = val
            best_h = cand.copy()
            if best <= 0.3808:
                break
    return best_h, best

import time, random, math
import numpy as np

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto box [0,1]^n intersect sum(x)=target via bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5: max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(xs - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * xs / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    h = np.zeros(n)
    w = n // 2
    start = (n - w) // 2
    h[start:start + w] = 1.0
    return h  # already satisfies sum=n/2


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def get_initial_seeds(n: int):
    seeds = []
    if "initial_h_values" in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rectangle(n))
    return seeds


def gradient_descent(h0: np.ndarray, target_sum: float, time_budget: float, start_time: float):
    """Projected gradient descent on a smooth approximation of C5."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.5
    alphas = [5.0, 20.0, 100.0, 300.0]
    offset = n - 1
    while time.time() - start_time < time_budget:
        for alpha in alphas:
            for _ in range(250):
                if time.time() - start_time >= time_budget:
                    break
                # correlation and smooth weights
                c = np.correlate(h, 1.0 - h, mode="full")
                maxc = np.max(c)
                w = np.exp(alpha * (c - maxc))
                w /= w.sum()
                conv1 = np.convolve(w, h, mode="full")
                conv2 = np.convolve(w[::-1], h, mode="full")
                grad = 1.0 - conv1[offset:offset + n] - conv2[offset:offset + n]
                h_new = h - step * grad
                h_new = project_onto_bounds_and_sum(h_new, target_sum)
                cur = compute_c5(h_new)
                if cur < best_val:
                    best_val, best_h = cur, h_new.copy()
                    h = h_new
                    step = min(step * 1.05, 1.0)
                else:
                    step *= 0.5
                if step < 1e-4:
                    step = 0.5
                if best_val <= 0.3808:
                    return best_h, best_val
            if time.time() - start_time >= time_budget:
                break
    return best_h, best_val


def local_refine(h: np.ndarray, target_sum: float, time_budget: float, start_time: float):
    """Simple pairwise mass swaps to reduce exact C5."""
    best = compute_c5(h)
    best_h = h.copy()
    n = h.shape[0]
    rng = np.random.default_rng()
    while time.time() - start_time < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i], cand[j] = hi + delta, hj - delta
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best:
            best, best_h = val, cand.copy()
            if best <= 0.3808:
                break
    return best_h, best

    start = time.time()

    n = 300  # even, modest size
    target_sum = n / 2.0

    seeds = get_initial_seeds(n)

        if best_val <= 0.3808:
            break

    while True:

        if best_val <= 0.3808:

    # final local refinement with remaining budget

    best_h = project_onto_bounds_and_sum(best_h, target_sum)
