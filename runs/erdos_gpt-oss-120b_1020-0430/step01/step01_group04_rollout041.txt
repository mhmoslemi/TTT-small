def proj_capped_simplex(x, target, tol=1e-9, max_iter=100):
    # Ensure sum target <= n (true) and target >= 0.
    # initial lo, hi:
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(max_iter):
        lam = (lo+hi)/2.0
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam

def softmax_gradient(h, alpha):
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    # compute p = softmax(alpha*conv)
    m = conv.shape[0]
    max_val = np.max(alpha * conv)
    exp_terms = np.exp(alpha * conv - max_val)
    p = exp_terms / exp_terms.sum()  # shape m

    grad = np.ones_like(h)  # start at 1
    offset = n - 1
    for idx in range(m):
        pk = p[idx]
        if pk == 0:
            continue
        k = idx - offset
        if k >= 0:
            # i from 0..n-1-k: i+k in range => subtract pk*h[i+k]
            grad[:n - k] -= pk * h[k:]
        else:
            # k negative: i+k = i - |k| => i from |k|..n-1
            grad[-k:] -= pk * h[:n + k]

        # also subtract pk * h[i - k]
        kk = -k
        if kk >= 0:
            grad[:n - kk] -= pk * h[kk:]
        else:
            grad[-kk:] -= pk * h[:n + kk]
    return grad

def project_capped_simplex(x, target, tol=1e-9, max_iter=100):
    # Project onto {0 <= y_i <= 1, sum yi = target}
    # Clip first? We'll do binary search on lambda.
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) < tol:
            return y
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_multi_stage(start, n, target, alphas, time_deadline, rng):
    h = start.copy()
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    for alpha in alphas:
        if time.time() > time_deadline:
            break
        # define objective using softmax
        obj = lambda h: softmax_c5(h, alpha)
        try:
            res = minimize(obj, h, method="SLSQP", bounds=bounds,
                           constraints=eq_cons,
                           options={"maxiter": 300, "ftol": 1e-9, "disp": False})
        except Exception:
            break
        if not res.success:
            # maybe reduce step? keep current h.
            # optionally try again with fewer iterations?
            pass
        h = np.clip(res.x, 0.0, 1.0)
        # after step, enforce sum exactly
        sum_diff = np.sum(h) - target
        if abs(sum_diff) > 1e-8:
            # project onto capped simplex to correct
            h = project_capped_simplex(h, target)
    c5 = compute_c5(h)
    return h, c5

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of C5 using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular h(x)=min(x,2‑x) sampled on n points and normalised."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    old_x = np.arange(old_n) * (2.0 / old_n) + (1.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n) + (1.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def project_capped_simplex(x: np.ndarray, target: float, tol: float = 1e-9,
                           max_iter: int = 100) -> np.ndarray:
    """Projection onto {0≤y≤1, sum(y)=target}."""
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random feasible vector (0≤h≤1, sum=n/2)."""
    a = rng.random(n)
    a *= (n / 2.0) / a.sum()
    a = np.clip(a, 0.0, 1.0)
    return project_capped_simplex(a, n / 2.0)


def slsqp_multi_stage(start: np.ndarray, n: int, target: float,
                     alphas, deadline: float, rng: np.random.Generator):
    """Run SLSQP with increasing soft‑max temperature."""
    h = start.copy()
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(lambda hh: softmax_c5(hh, alpha), h,
                           method="SLSQP", bounds=bounds,
                           constraints=eq_cons,
                           options={"maxiter": 400, "ftol": 1e-9,
                                    "disp": False})
        except Exception:
            break
        if not res.success:
            # keep current iterate, continue with next alpha
            pass
        h = np.clip(res.x, 0.0, 1.0)
        # enforce the sum exactly (projection is cheap)
        h = project_capped_simplex(h, target)
    return h, compute_c5(h)


def hill_climb(h: np.ndarray, target: float, deadline: float,
               rng: np.random.Generator, max_iters: int = 2000):
    """Greedy mass‑transfer local search."""
    best_h = h.copy()
    best_val = compute_c5(best_h)
    n = len(h)
    step = 0.1                       # maximal transferred mass in a trial
    while time.time() < deadline and max_iters > 0:
        improved = False
        for _ in range(300):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_move = min(best_h[i], 1.0 - best_h[j], step)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move
            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            val = compute_c5(new_h)
            if val < best_val - 1e-12:
                best_h, best_val = new_h, val
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break
        max_iters -= 1
    return best_h, best_val

    """
    Optimises a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """

    start_time = time.time()
    deadline = start_time + 0.95 * budget_s   # safety margin

        parent_h = None
        parent_n = None

    best_h = None
    best_c5 = np.inf
    best_n = None

    # grid sizes to explore – higher resolutions give tighter values
    candidate_ns = [160, 180, 200, 220, 240, 260, 280]
    if parent_n is not None and parent_n not in candidate_ns:

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # --------------------------------------------------------
        starts = []

    best_h = project_capped_simplex(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
