def binary_sa_optimize(rng, n, max_iters):
    # generate random balanced binary vector
    h = np.zeros(n)
    # randomly select n/2 indices to set to 1
    indices = rng.choice(n, n//2, replace=False)
    h[indices] = 1.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    # schedule
    T0 = 1.0
    Tf = 1e-4
    for it in range(max_iters):
        t = T0 * (Tf/T0) ** (it / max_iters)  # exponential cooling
        # pick two indices with different values
        i = rng.integers(0, n)
        # choose j of opposite value
        # we can find list of opposite indices each time; maybe store arrays.
        # Simpler: pick j random until h[j] != h[i]
        j = rng.integers(0, n)
        while h[i] == h[j]:
            j = rng.integers(0, n)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c # improvement negative if lower
        if delta <= 0 or rng.random() < np.exp(-delta / t):
            cur_c = new_c
            # accept
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
```

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()

    best_c = compute_c5(h)
    step = start_step

        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)

            best_h = new_h.copy()
            h = new_h

            step = max(step * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- globals for the constrained optimizer ----------
CURRENT_BETA = None   # soft‑max temperature
SLSQP_TARGET = None   # sum constraint value (n/2)

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = SLSQP_TARGET."""
    return np.sum(x) - SLSQP_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k}) using log‑sum‑exp."""
    global CURRENT_BETA
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_BETA, SLSQP_TARGET
    n = h0.size
    SLSQP_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # repair possible tiny sum drift
    diff = SLSQP_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = SLSQP_TARGET - x.sum()
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random vector in [0,1] with exact sum n/2."""
    # start from the constant 0.5 vector and randomise a bit
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.5):
    """Simple deterministic hill‑climb (only accept improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # shrink step periodically
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def binary_sa_optimize(rng: np.random.Generator, n: int, time_limit: float):
    """Simulated‑annealing on binary (0/1) vectors preserving exactly n/2 ones."""
    start = time.time()
    # initialise a random balanced binary vector
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c

    # exponential cooling schedule
    T0, Tf = 1.0, 1e-4
    it = 0
    while time.time() - start < time_limit:
        it += 1
        t = T0 * (Tf / T0) ** (it / 200000)  # 200k is a rough horizon
        # pick two positions with opposite values
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while cur_h[i] == cur_h[j]:
            j = rng.integers(0, n)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c
