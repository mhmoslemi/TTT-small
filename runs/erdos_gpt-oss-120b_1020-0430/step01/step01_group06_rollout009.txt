import numpy as np, random, time, math

def c5_of(v):
    return float(np.correlate(v, 1 - v, mode='full').max() * (2.0 / v.shape[0]))

def proj_cap(y, target):
    # projection onto [0,1] and sum constraint
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction if needed
    err = x.sum() - target
    if abs(err) > 1e-10:
        # adjust a random non‑boundary element
        idxs = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if len(idxs) == 0:
            idxs = np.arange(x.size)
        k = random.choice(idxs)
        x[k] -= err
        x = np.clip(x, 0.0, 1.0)
    return x

    while time.time() - start < budget_s * 0.95:

def run(seed=42, budget_s=1100, **kwargs):
    import random, time, numpy as np, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # initialise h (use the provided vector if it exists)
    # --------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    except NameError:
        n = 80                     # a modest size that still runs fast
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    target = n / 2.0                # required total mass

    # --------------------------------------------------------------
    # projection onto the capped simplex {0≤x_i≤1, sum x_i = target}
    # --------------------------------------------------------------
    def proj_cap(y, target):
        lo = y.min() - 1.0
        hi = y.max()
        for _ in range(60):
            lam = (lo + hi) / 2.0
            x = np.clip(y - lam, 0.0, 1.0)
            s = x.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
        err = x.sum() - target
        if abs(err) > 1e-10:
            cand = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
            if cand.size == 0:
                cand = np.arange(x.size)
            k = random.choice(cand.tolist())
            x[k] -= err
            x = np.clip(x, 0.0, 1.0)
        return x

    h = np.clip(h, 0.0, 1.0)
    h = proj_cap(h, target)

    # --------------------------------------------------------------
    # helper to compute C5 for a vector
    # --------------------------------------------------------------
    def c5_of(v):
        return float(np.correlate(v, 1.0 - v, mode="full").max() * (2.0 / v.shape[0]))

    best_h = h.copy()
    best_c5 = c5_of(best_h)

    # --------------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------------
    alpha = 10.0            # softness of the max (soft‑max temperature)
    it = 0
    n2 = n
    # -----------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # overlaps for all integer shifts
        O = np.correlate(h, 1.0 - h, mode="full")
        maxO = O.max()
        # soft‑max weights (numerically stable)
        exp_vals = np.exp(alpha * (O - maxO))
        w = exp_vals / exp_vals.sum()

        # gradient (weighted sum of per‑shift gradients)
        grad = np.zeros_like(h)
        for s in range(-(n2 - 1), n2):
            ws = w[s + n2 - 1]
            if ws == 0.0:
                continue
            if s >= 0:
                L = n2 - s
                if L <= 0:
                    continue
                hi = h[:L]
                hj = h[s:s + L]
                grad[:L] += ws - ws * hj
                grad[s:s + L] -= ws * hi
            else:

                L = n2 - sa

                hj = h[:L]
                grad[sa:n2] += ws - ws * hj

        lr = 0.04 / max_abs if max_abs > 0 else 0.0

        h_new = h - lr * grad

        # occasional random perturbation to escape plateaux
        if random.random() < 0.02:
            i, j = random.sample(range(n), 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]

        h_new = proj_cap(h_new, target)

        cur_c5 = c5_of(h_new)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_new.copy()

                h = h_new

            h = h_new
