def guided_hill_climb(h, c5, deadline, rng):
    while time.time() < deadline:
        # compute corr
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        if corr[idx_max] * (2.0/len(h)) >= c5 - 1e-12: # no improvement maybe
            # try random swap as fallback
            i, j = rng.choice(len(h), size=2, replace=False)
            # ensure start and dest
            if rng.random() < 0.5:
                src, dst = i, j
            else:
                src, dst = j, i
            max_move = min(h[src], 1.0 - h[dst])
            if max_move <= 1e-12: continue
            delta = rng.uniform(0.0, max_move)
            h_new = h.copy()
            h_new[src] -= delta
            h_new[dst] += delta
            new_c5 = _evaluate_c5(h_new)
            if new_c5 < c5 - 1e-12:
                h, c5 = h_new, new_c5
            continue
        shift = idx_max - (len(h)-1)
        L = n - abs(shift)
        # compute contributions
        if shift >= 0:
            # i from 0 to n-1-shift
            idx_i = np.arange(0, n - shift)
            contrib = h[idx_i] * (1.0 - h[idx_i + shift])
        else:
            # shift negative; consider i from -shift to n-1
            idx_i = np.arange(-shift, n)
            # contributions = h[idx_i] * (1 - h[idx_i + shift]) where shift negative, i+shift valid
            contrib = h[idx_i] * (1.0 - h[idx_i + shift])  # idx_i + shift = idx_i - abs(shift)
        # choose source and dest 
        src_candidates = np.where(contrib > 0)[0]
        if src_candidates.size == 0:
            # fallback random
            i, j = rng.choice(n, size=2, replace=False)
            src, dst = i, j
        else:
            src_idx = src_candidates[np.argmax(contrib[src_candidates])]
            dst_idx = np.argmin(contrib)  # choose min contribution (maybe zero)
            src = idx_i[src_idx]
            dst = idx_i[dst_idx]  # note may be same? Ensure different
            if src == dst:
                # fallback random
                i, j = rng.choice(n, size=2, replace=False)
                src, dst = i, j
        # Determine max move
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            # random small perturb
            delta = 0.0
        else:
            delta = rng.uniform(0.0, max_move)
        if delta <= 1e-12:
            continue
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5

import time
import numpy as np
from scipy.optimize import differential_evolution

def _evaluate_c5(h):
    """Exact C5 for a step vector."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _adjust_sum(x, target):
    """
    Project x onto {0 ≤ x ≤ 1, sum(x)=target}.
    Simple redistribution loop – fast enough for the tiny vectors we use.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        per = diff / len(idx)
        for i in idx:
            if diff > 0.0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _guided_hill_climb(h, c5, deadline, rng):
    """
    Directed hill‑climbing: always attack the shift that currently yields the
    maximal overlap and transfer a small amount of mass from a high‑contributing
    index to a low‑contributing one.  Falls back to random pairwise moves and
    occasional global jitter when stuck.
    """
    n = h.shape[0]
    target = n / 2.0
    no_improve = 0
    while time.time() < deadline:
        # compute correlation and locate worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # indices that actually participate in this shift
        if shift >= 0:
            i_valid = np.arange(0, n - shift)
            partner = i_valid + shift
        else:
            i_valid = np.arange(-shift, n)
            partner = i_valid + shift  # shift is negative

        contrib = h[i_valid] * (1.0 - h[partner])
        # pick source (large contribution) and destination (small contribution)
        src_local = int(np.argmax(contrib))
        dst_local = int(np.argmin(contrib))
        src = i_valid[src_local]
        dst = i_valid[dst_local]

        # if source and destination coincide or move is impossible, try a random swap
        if src == dst or min(h[src], 1.0 - h[dst]) < 1e-12:
            i, j = rng.choice(n, size=2, replace=False)
            src, dst = (i, j) if rng.random() < 0.5 else (j, i)

        max_move = min(h[src], 1.0 - h[dst])
        if max_move > 0.0:
            delta = rng.uniform(0.0, max_move)
            h_new = h.copy()
            h_new[src] -= delta
            h_new[dst] += delta
            new_c5 = _evaluate_c5(h_new)
            if new_c5 + 1e-12 < c5:
                h, c5 = h_new, new_c5
                no_improve = 0
                continue

        # no improvement – try a cheap random exchange
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_move = min(h[src], 1.0 - h[dst])
        if max_move > 0.0:
            delta = rng.uniform(0.0, max_move)
            h_new = h.copy()
            h_new[src] -= delta
            h_new[dst] += delta
            new_c5 = _evaluate_c5(h_new)
            if new_c5 + 1e-12 < c5:
                h, c5 = h_new, new_c5
                no_improve = 0
                continue

        # still no progress – apply a small global jitter every now and then
        no_improve += 1
        if no_improve >= 200:
            jitter = rng.normal(scale=0.03, size=n)
            h = np.clip(h + jitter, 0.0, 1.0)
            h = _adjust_sum(h, target)
            c5 = _evaluate_c5(h)
            no_improve = 0
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Global optimisation for the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. initialise – use the supplied construction if present
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        n = 60                     # a modest even size gives a convenient simplex
        h0 = rng.random(n)
        h0 = _adjust_sum(np.clip(h0, 0.0, 1.0), n / 2.0)

    target = n / 2.0
    best_h = _adjust_sum(np.clip(h0, 0.0, 1.0), target)
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Global search with differential evolution (smoothly projected)
    # ------------------------------------------------------------------
    de_budget = budget_s * 0.70
    de_deadline = start + de_budget - 5.0

    def de_obj(x):
        # project onto the feasible simplex before evaluation
        h = np.clip(x, 0.0, 1.0)
        h = _adjust_sum(h, target)
        return _evaluate_c5(h)

    bounds = [(0.0, 1.0)] * n

    # use a deterministic seed for reproducibility
    de_seed = int(rng.integers(0, 2 ** 32 - 1))

    def de_callback(xk, convergence):
        if time.time() > de_deadline:
            return True        # stop the DE run
        return False

    try:
        result = differential_evolution(
            de_obj,
            bounds,
            strategy="best1bin",
            maxiter=2000,
            popsize=12,
            init="latinhypercube",
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=de_seed,
            polish=False,
            callback=de_callback,
            disp=False,
            updating="deferred",

            cand_c5 = _evaluate_c5(cand_h)
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand_h, cand_c5
    except Exception:
        # If DE fails for any reason we simply continue with the current best
        pass

    # ------------------------------------------------------------------
    # 3. Fine‑grained directed hill‑climbing
    # ------------------------------------------------------------------
    hill_deadline = start + budget_s - 0.5
    best_h, best_c5 = _guided_hill_climb(best_h, best_c5, hill_deadline, rng)

    # ------------------------------------------------------------------
    # 4. Final feasibility check and return
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
