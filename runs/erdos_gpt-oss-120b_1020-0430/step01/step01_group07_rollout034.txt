def enforce_constraints(h, target):
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    deficit = target - cur
    if abs(deficit) > 1e-12:
        # distribute to feasible entries
        if deficit > 0:
            # need to increase sum
            idx = np.where(h < 1.0)[0]
            # iterate adding as much as possible
            for ii in idx:
                add = min(deficit, 1.0 - h[ii])
                h[ii] += add
                deficit -= add
                if deficit <= 1e-12:
                    break
        else:
            # need to decrease sum
            idx = np.where(h > 0.0)[0]
            for ii in idx:
                sub = min(-deficit, h[ii])
                h[ii] -= sub
                deficit += sub
                if deficit >= -1e-12:
                    break
    return h

import time
import math
import random

# ----------------------------------------------------------------------
def _c5_metric(h):
    """Discrete C5 metric (max overlap) for step function array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _enforce_sum(h, target):
    """Project h onto [0,1] box and adjust to obtain the exact sum = target."""
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:  # need to add mass
        idx = np.where(h < 1.0)[0]
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:          # need to remove mass
        idx = np.where(h > 0.0)[0]
        for i in idx:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h


def _gradient_for_shift(h, k):
    """Gradient of f_k(h)=∑h_i(1-h_{i+k}) w.r.t. h (unscaled, no dx)."""
    n = h.shape[0]
    g = np.ones(n)
    if k >= 0:
        end = n - k
        if end > 0:
            g[:end] -= h[k:]        # -h_{i+k}
            g[k:]   -= h[:end]       # -h_{i-k}
    else:
        kp = -k
        end = n - kp
        if end > 0:
            g[kp:]  -= h[:end]       # -h_{i+k}= -h_{i-kp}
            g[:end] -= h[kp:]        # -h_{i-k}= -h_{i+kp}
    return g


def _random_start_binary(n):
    """Generate a feasible binary (or one‑fractional) start vector."""
    target = n / 2.0
    ones = int(target)
    frac = target - ones
    h = np.zeros(n, dtype=np.float64)
    perm = np.random.permutation(n)
    h[perm[:ones]] = 1.0
    if frac > 1e-12:
        h[perm[ones]] = frac
    return h


def _optimize_until(h0, budget):
    """Guided simulated‑annealing on a single start vector."""
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_corr = np.max(corr)          # unscaled
    start = time.time()

    # annealing / step‑size schedule
    T0, T_end = 0.05, 1e-6
    step0, step_end = 0.5, 0.02

    while time.time() - start < budget:
        elapsed = time.time() - start
        prog = elapsed / budget

        T = T0 * (T_end / T0) ** prog
        max_step = step0 * (step_end / step0) ** prog

        # current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_max = np.max(corr)
        ks = np.where(corr == cur_max)[0]
        k = ks[np.random.randint(len(ks))] - (n - 1)

        # gradient for that shift
        g = _gradient_for_shift(h, k)
        pos = np.where(g > 1e-9)[0]      # decreasing them helps
        neg = np.where(g < -1e-9)[0]     # increasing them helps

        moved = False
        # try a guided redistribution
        if pos.size and neg.size:
            i = np.random.choice(pos)
            j = np.random.choice(neg)
            hi, hj = h[i], h[j]
            delta_max = min(hi, 1.0 - hj, max_step)
            if delta_max > 0:
                delta = random.random() * delta_max
                h[i] = hi - delta
                h[j] = hj + delta
                moved = True
                rec = (i, j, hi, hj, delta, True)

        # fallback: random pair move
        if not moved:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low < high:
                delta = random.uniform(low, high) * (max_step / step0)
                h[i] = hi + delta
                h[j] = hj - delta
                moved = True
                rec = (i, j, hi, hj, delta, False)

        if not moved:
            continue

        # evaluate new state
        new_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
        accept = (new_corr <= cur_max + 1e-12) or (
            random.random() < math.exp(-(new_corr - cur_max) / max(T, 1e-12))
        )
        if accept:
            if new_corr < best_corr - 1e-12:

                best_h = h.copy()
        else:

            h[i] = hi
            h[j] = hj

def _final_polish(h, current_c, time_left):

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a feasible step function achieving a low C5 value.
    Output: (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    start_all = time.time()
    total_budget = float(budget_s)

    # decide on resolution (even number, at least 400)
    base_n = 400
    if "initial_h_values" in globals():
        base_n = max(base_n, len(initial_h_values))
    n = ((base_n + 1) // 2) * 2   # make it even
    dx = 2.0 / n
    target_sum = n / 2.0

    # ---- primary deterministic start (triangular shape) ----
    x = (np.arange(n) + 0.5) * dx               # mid‑point sampling
    h_tri = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    h_tri = _enforce_sum(h_tri, target_sum)

    # ---- optional interpolated start from provided vector ----
    starts = [h_tri]
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_interp = np.interp(x_new, x0, h0)
        h_interp = _enforce_sum(np.clip(h_interp, 0.0, 1.0), target_sum)
        starts.append(h_interp)

    # allocate most of the budget to optimise these starts
    remaining = total_budget - (time.time() - start_all)
    primary_budget = remaining * 0.7
    per_start = primary_budget / len(starts)

    best_h, best_c = None, float("inf")
    for h0 in starts:
        cand_h, cand_c = _optimize_until(h0, per_start)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h

    # ---- random binary restarts for the rest of the time ----
    remaining = total_budget - (time.time() - start_all)
    min_slice = 12.0
    while remaining > min_slice:
        slice_time = min(remaining * 0.4, 120.0)
        h_rand = _random_start_binary(n)
        cand_h, cand_c = _optimize_until(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        remaining = total_budget - (time.time() - start_all)

    # ---- final polishing (tiny moves) ----
    time_left = total_budget - (time.time() - start_all)
    best_h, best_c = _final_polish(best_h, best_c, time_left)

    # ensure exact sum (tiny rounding errors)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # final evaluation (use provided helper if present)
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
