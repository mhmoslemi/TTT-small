function project(x):
    - Clip to [0,1].
    - Compute sum = s.
    - If |s - target| < small: done.
    - If s > target: need to reduce total; we can reduce proportionally from entries not at lower bound (i.e., those >0). Approach: compute sorted indices by value descending and reduce from largest until sum = target, while not dropping below 0. This ensures feasibility.
    - If s < target: need to increase total from entries not at upper bound; fill from smallest until sum = target.

Implementation:

def project_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if s > target + 1e-12:
        # reduce excess
        excess = s - target
        # indices sorted by x descending (large first)
        idx = np.argsort(-x)
        for i in idx:
            if excess <= 1e-12:
                break
            can_reduce = x[i] - 0.0
            red = min(can_reduce, excess)
            x[i] -= red
            excess -= red
    elif s < target - 1e-12:
        deficit = target - s
        idx = np.argsort(x)  # smallest first
        for i in idx:
            if deficit <= 1e-12:
                break
            can_add = 1.0 - x[i]
            add = min(can_add, deficit)
            x[i] += add
            deficit -= add
    return x

This ensures sum exactly target (within floating rounding) after loop.

def get_weights(h, shift):
    n = len(h)
    w = np.zeros(n)

        # w = 2*h (since both directions same)

def directed_hill_climb(h, score, time_deadline, rng):
    while time.time() < time_deadline:
        # compute correlation and find worst shift (max)
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find shift index where corr is max (largest overlap)
        idx = np.argmax(corr)  # index in full array length 2n-1
        # shift m = idx - (n-1)
        shift = idx - (n-1)
        # compute weight vector for moving mass
        w = np.zeros(n)
        m = shift
        if m >= 0:
            # overlapping region length n-m
            w[:n-m] += h[m:]   # h[i+m]
            w[m:] += h[:n-m]   # h[i-m]
        else:
            # negative shift: treat similarly
            m = -m
            w[m:] += h[:n-m]   # h[i-m] (since shift negative)
            w[:n-m] += h[m:]   # h[i+m]
        # choose src among low weight and >0, dst among high weight and <1
        low_indices = np.where((h > 1e-12) & (w == w.min()))[0]
        high_indices = np.where((h < 1 - 1e-12) & (w == w.max()))[0]
        if low_indices.size == 0 or high_indices.size == 0:
            # fallback to random
            i = rng.integers(n)
            j = rng.integers(n)
        else:
            i = rng.choice(low_indices)
            j = rng.choice(high_indices)
        # compute max delta
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 1e-12:
            delta = rng.random() * max_move
            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = evaluate_c5(cand)
            if cand_score + 1e-12 < score:
                h = cand
                score = cand_score
    return h, score

def slsqp_refine(h_guess, max_iter=200):
    # use minimize like before
    n = len(h_guess)
    target = n/2
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    # objective:
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * (2.0 / n)
    res = minimize(obj, h_guess, method='SLSQP', bounds=bounds, constraints=[cons],
                   options={"maxiter": 200, "ftol":1e-9, "disp":False})
    if res.success:
        h_ref = np.clip(res.x,0.0,1.0)
        h_ref = project_sum(h_ref, target)
        val = evaluate_c5(h_ref)
        return h_ref, val
    else:
        return h_guess, evaluate_c5(h_guess)

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if s > target:
        excess = s - target
        # reduce from largest values first
        idx = np.argsort(-x)  # descending
        for i in idx:
            if excess <= 1e-12:
                break
            reducible = x[i]
            dec = min(reducible, excess)
            x[i] -= dec
            excess -= dec
    elif s < target:
        deficit = target - s
        idx = np.argsort(x)  # ascending
        for i in idx:
            if deficit <= 1e-12:
                break
            addable = 1.0 - x[i]
            inc = min(addable, deficit)
            x[i] += inc
            deficit -= inc
    return x

def _random_vector(n, target, rng):
    # generate random values and project
    vec = rng.random(n)
    return _project_sum(vec, target)

def _directed_hill_climb_step(h, shift, rng):
    # compute weight vector for shift
    n = len(h)
    w = np.zeros(n)
    m = shift
    if m >= 0:
        if m < n:
            w[:n-m] += h[m:]
            w[m:] += h[:n-m]
    else:
        m = -m
        if m < n:
            w[m:] += h[:n-m]
            w[:n-m] += h[m:]
    # select src low weight with mass, dst high weight with capacity
    low_candidates = np.where((h > 1e-12) & (w <= np.percentile(w, 30)))[0]
    high_candidates = np.where((h < 1 - 1e-12) & (w >= np.percentile(w, 70)))[0]
    if low_candidates.size == 0 or high_candidates.size == 0:
        return None, None
    src = rng.choice(low_candidates)
    dst = rng.choice(high_candidates)
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-12:
        return None, None
    delta = rng.random() * max_move
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    # sum unchanged, but adjust for numerical errors
    return h_new, _evaluate_c5(h_new)

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures or lambdas)
# ----------------------------------------------------------------------


def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and adjust so that sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if s > target + 1e-12:                 # need to remove excess
        excess = s - target
        idx = np.argsort(-x)               # largest first
        for i in idx:
            if excess <= 1e-12:
                break
            reducible = x[i]
            dec = min(reducible, excess)
            x[i] -= dec
            excess -= dec
    elif s < target - 1e-12:                # need to add deficit
        deficit = target - s
        idx = np.argsort(x)                # smallest first
        for i in idx:
            if deficit <= 1e-12:
                break
            addable = 1.0 - x[i]
            inc = min(addable, deficit)
            x[i] += inc
            deficit -= inc
    return x


def _random_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible step vector."""
    vec = rng.random(n)
    return _project_sum(vec, target)


def _max_corr_shift(h: np.ndarray) -> int:
    """Shift (relative to centre) where the correlation attains its maximum."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (h.shape[0] - 1)


def _directed_hill_climb_step(h: np.ndarray, shift: int,
                              rng: np.random.Generator) -> tuple:
    """
    Propose a move that transfers a small amount of mass from a low‑weight
    position to a high‑weight one, where weight = contribution to the
    product sum for the given shift.
    Returns (new_h, new_score) or (None, None) if no move is possible.
    """
    n = h.shape[0]
    w = np.zeros(n)

    m = shift
    if m >= 0:
        if m < n:
            w[: n - m] += h[m:]
            w[m:] += h[: n - m]
    else:
        m = -m
        if m < n:
            w[m:] += h[: n - m]
            w[: n - m] += h[m:]

    # candidates based on weight percentiles
    low_thr = np.percentile(w, 30)
    high_thr = np.percentile(w, 70)

    low_candidates = np.where((h > 1e-12) & (w <= low_thr))[0]
    high_candidates = np.where((h < 1.0 - 1e-12) & (w >= high_thr))[0]

    if low_candidates.size == 0 or high_candidates.size == 0:
        return None, None

    src = rng.choice(low_candidates)
    dst = rng.choice(high_candidates)
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-12:
        return None, None

    delta = rng.random() * max_move
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    # sum is preserved, but numerical drift may occur
    new_h = _project_sum(new_h, n / 2.0)
    return new_h, _evaluate_c5(new_h)

# Main optimisation routine
# ----------------------------------------------------------------------


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # 1. Initialise – use the provided construction if available
    # --------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        n = 45
        h0 = _random_vector(n, n / 2.0, rng)

    target = n / 2.0
    h0 = _project_sum(np.clip(h0, 0.0, 1.0), target)
    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    # --------------------------------------------------------------
    # 2. Differential‑Evolution style global search
    # --------------------------------------------------------------
    popsize = max(10, min(30, n))
    population = np.empty((popsize, n), dtype=np.float64)
    scores = np.empty(popsize, dtype=np.float64)

    # first member = current best
    population[0] = best_h
    scores[0] = best_c5

    for i in range(1, popsize):
        vec = _random_vector(n, target, rng)
        population[i] = vec
        scores[i] = _evaluate_c5(vec)
        if scores[i] < best_c5:
            best_c5 = scores[i]
            best_h = vec.copy()

    F = 0.6               # mutation factor
    Cr = 0.9              # crossover probability
    de_deadline = start + budget_s - 30.0   # reserve time for final refinements

    while time.time() < de_deadline:
        # one generation
        for i in range(popsize):
            # pick two distinct donors (different from each other)
            b, c = rng.choice(popsize, size=2, replace=False)

            # best/1 mutation (guided by the current global best)
            mutant = best_h + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = _project_sum(mutant, target)

            # binomial crossover
            j_rand = rng.integers(n)
            rand_vals = rng.random(n)
            trial = np.where((rand_vals < Cr) | (np.arange(n) == j_rand),
                             mutant, population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = _project_sum(trial, target)

            trial_score = _evaluate_c5(trial)

            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_c5:
                    best_c5 = trial_score
                    best_h = trial.copy()

        # early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    # --------------------------------------------------------------
    # 3. Focused hill‑climbing (directed + random moves)
    # --------------------------------------------------------------
    hill_deadline = start + budget_s - 5.0   # leave a small buffer
    cur_h = best_h.copy()
    cur_c5 = best_c5

    while time.time() < hill_deadline:
        # try a directed move based on the worst shift
        worst_shift = _max_corr_shift(cur_h)
        cand_h, cand_score = _directed_hill_climb_step(cur_h, worst_shift, rng)

        if cand_h is not None and cand_score + 1e-12 < cur_c5:
            cur_h, cur_c5 = cand_h, cand_score
            if cand_score < best_c5:
                best_c5, best_h = cand_score, cand_h.copy()
            continue

        # fall back to a simple random mass exchange
        src, dst = rng.integers(n, size=2)
        if src == dst:
            continue
        if cur_h[src] < 1e-12 or cur_h[dst] > 1.0 - 1e-12:
            continue
        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        trial = cur_h.copy()
        trial[src] -= delta
        trial[dst] += delta
        trial_score = _evaluate_c5(trial)

        if trial_score + 1e-12 < cur_c5:
            cur_h, cur_c5 = trial, trial_score
            if trial_score < best_c5:
                best_c5, best_h = trial_score, trial.copy()

    # --------------------------------------------------------------
    # 4. Final feasibility check and return
    # --------------------------------------------------------------
    best_h = _project_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))

    return best_h, best_c5, n
