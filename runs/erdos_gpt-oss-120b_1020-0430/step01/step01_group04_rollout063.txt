def project_feasible(v, target):
    # project onto [0,1]ⁿ ∧ sum = target
    # Use algorithm described: clip to [0,1]; then adjust sum.
    # We'll implement iterative water-filling.
    x = np.clip(v, 0.0, 1.0)
    # we need to adjust sum to target.
    # Use iterative method:
    diff = x.sum() - target
    # while |diff| > tolerance:
    # we compute free mask where 0 < x < 1
    # adjust as described.
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        m = free.sum()
        if m == 0:
            # can't adjust any further; distribution with diff by adjusting values within bounds
            # simply distribute diff equally among all and clip again
            # but may cause bounds violation; we break.
            break
        delta = diff / m
        # adjust those free entries
        x[free] -= delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
        diff = x.sum() - target
    # final fallback: if still off, shift uniformly with clipping again
    if abs(x.sum() - target) > 1e-10:
        # adjust by adding constant and clipping
        offset = (target - x.sum()) / len(x)
        x += offset
        x = np.clip(x, 0.0, 1.0)
        # re-fit perhaps iterative again
        # we can just accept small error
    return x

import time
import numpy as np

# ------------------------------------------------------------
# Basic evaluation
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


# ------------------------------------------------------------
# Projection onto the simplex with box constraints [0,1]
# ------------------------------------------------------------
def project_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    # binary search on lambda for x = clip(v - lam, 0, 1)
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lo, 0.0, 1.0)
    # final tiny correction if needed
    diff = x.sum() - target
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        m = free.sum()
        if m > 0:
            x[free] -= diff / m
            x = np.clip(x, 0.0, 1.0)
    return x


# ------------------------------------------------------------
# Random feasible start
# ------------------------------------------------------------
def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


# ------------------------------------------------------------
# Upsample by linear interpolation
# ------------------------------------------------------------
def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    old_n = arr.shape[0]
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


# ------------------------------------------------------------
# Asymmetric ramp–plateau shape generator
# ------------------------------------------------------------
def gen_asym_shape(n: int, r1: float, r2: float) -> np.ndarray:
    """Generate a symmetric piecewise‑linear h on [0,2] with left ramp length r1,
    right ramp length r2 (both in [0,2] and r1+r2≤2)."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    # height a from area constraint
    denom = 2.0 - (r1 + r2) / 2.0
    a = 1.0 / denom if denom > 0 else 0.0
    h = np.full(n, a, dtype=float)
    if r1 > 1e-12:
        left = x < r1
        h[left] = a * x[left] / r1
    if r2 > 1e-12:
        right = x > (2.0 - r2)
        h[right] = a * (2.0 - x[right]) / r2
    h = np.clip(h, 0.0, 1.0)
    return h


# ------------------------------------------------------------
# Projected sub‑gradient descent on max‑overlap objective
# ------------------------------------------------------------
def pgd_refine(h0: np.ndarray, target: float, deadline: float,
               max_iters: int = 3000, step0: float = 0.5) -> tuple:
    """Minimise max_{k} ⟨h,1‑h(·+k)⟩ by sub‑gradient descent."""
    h = h0.copy()
    n = h.shape[0]
    cur = compute_c5(h)
    step = step0
    while max_iters and time.time() < deadline:
        conv = np.correlate(h, 1.0 - h, mode="full")
        i_star = int(np.argmax(conv))
        lag = i_star - (n - 1)
        g = np.ones_like(h)
        if lag >= 0:
            g[: n - lag] -= h[lag:]
            g[lag:] -= h[: n - lag]
        else:
            d = -lag
            g[d:] -= h[: n - d]
            g[: n - d] -= h[d:]
        # trial step
        nh = project_feasible(h - step * g, target)
        nval = compute_c5(nh)
        if nval < cur:
            h, cur, step = nh, nval, min(step * 1.05, 1.0)
        else:
            step *= 0.5
            if step < 1e-8:
                break
        max_iters -= 1
    return h, cur


# ------------------------------------------------------------
# Simple simulated annealing on the feasible set
# ------------------------------------------------------------
def anneal(h0: np.ndarray, target: float, deadline: float,
           max_iter: int = 20000, t0: float = 0.02, tf: float = 1e-6,
           rng: np.random.Generator = None) -> tuple:
    if rng is None:
        rng = np.random.default_rng()
    h = h0.copy()
    cur = compute_c5(h)
    n = len(h)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # temperature schedule
        T = t0 * (tf / t0) ** (it / max_iter)
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # direction: move delta from i to j (i decreases, j increases)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0.0, max_delta) * (T / t0)  # scale with temperature
        if delta < 1e-12:
            continue
        nh = h.copy()
        nh[i] -= delta
        nh[j] += delta
        nval = compute_c5(nh)
        if nval < cur or rng.random() < np.exp(-(nval - cur) / max(T, 1e-12)):
            h, cur = nh, nval
    return h, cur


# ------------------------------------------------------------
# Main optimisation driver
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    # ----- resolution -------------------------------------------------
    candidate_ns = [120, 160, 200]
    best_h, best_c5, best_n = None, np.inf, None

    # if a parent construction is supplied, keep it for reuse
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:  # pragma: no cover
        parent_h = None
        parent_n = None

    for n in candidate_ns:
        if time.time() >= deadline:
            break
        target = n / 2.0

        # ---- 1. Scan the 2‑parameter ramp‑plateau family -------------
        # coarse grid
        step_grid = 0.05
        vals = np.arange(0.0, 2.0 + 1e-12, step_grid)
        for r1 in vals:
            if time.time() >= deadline:
                break
            # r2 limited by 2‑r1
            r2_max = 2.0 - r1
            r2_vals = np.arange(0.0, r2_max + 1e-12, step_grid)
            for r2 in r2_vals:
                h = gen_asym_shape(n, r1, r2)
                h = project_feasible(h, target)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h.copy(), n

        # ---- 2. Local hill‑climbing around the best pair -------------

        # retrieve the parameters that produced best_h (rough estimate)
        # we recompute them by simple search around current best
        # start with a small neighbourhood
        step = 0.1
        while step > 1e-4 and time.time() < deadline:
            improved = False
            # extract parameters from best_h by fitting (approximate)
            # simple heuristic: compute mass centre to guess asymmetry
            # but for speed we just perturb r1,r2 directly around 1.0
            # generate a few trial perturbations
            for dr1 in (-step, 0.0, step):
                for dr2 in (-step, 0.0, step):
                    if dr1 == 0 and dr2 == 0:
                        continue
                    # use current best approximated as near symmetric triangle
                    r1 = 1.0 + dr1
                    r2 = 1.0 + dr2
                    r1 = max(0.0, min(r1, 2.0))
                    r2 = max(0.0, min(r2, 2.0 - r1))
                    h = gen_asym_shape(n, r1, r2)
                    h = project_feasible(h, target)
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-12:
                        best_c5, best_h, best_n = c5, h.copy(), n
                        improved = True
            if not improved:
                step *= 0.5

        # ---- 3. Projected sub‑gradient refinement --------------------
        if best_h is not None and time.time() < deadline:
            h_ref, c5_ref = pgd_refine(best_h, target, deadline,
                                       max_iters=5000, step0=0.5)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref

        # ---- 4. Simulated annealing fine‑tuning ----------------------
        if best_h is not None and time.time() < deadline:
            h_ann, c5_ann = anneal(best_h, target, deadline,
                                   max_iter=15000, t0=0.02, tf=1e-6, rng=rng)
            if c5_ann < best_c5:
                best_h, best_c5 = h_ann, c5_ann

        # ---- 5. Random restarts (PGD) --------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            h0 = random_start(n, rng)
            h_ref, c5_ref = pgd_refine(h0, target, deadline,
                                       max_iters=3000, step0=0.5)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref

        # ---- 6. Incorporate parent solution if available -------------
        if parent_h is not None and time.time() < deadline:
            if parent_n != n:
                hp = upsample(parent_h, n)
            else:
                hp = parent_h.copy()
            hp = project_feasible(hp, target)
            c5p = compute_c5(hp)
            if c5p < best_c5:
                best_h, best_c5 = hp, c5p
            # polish parent via PGD
            hp_ref, c5p_ref = pgd_refine(hp, target, deadline,
                                         max_iters=3000, step0=0.5)
            if c5p_ref < best_c5:
                best_h, best_c5 = hp_ref, c5p_ref

    # -----------------------------------------------------------------
    # Fallback if everything failed
    # -----------------------------------------------------------------
    if best_h is None:
        n = 120
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final sanity projection
    best_h = project_feasible(best_h, best_h.shape[0] / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), best_h.shape[0]
