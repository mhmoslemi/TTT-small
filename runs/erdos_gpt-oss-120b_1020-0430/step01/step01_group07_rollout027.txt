    """Compute C₅ for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (non‑circular) cross‑correlation of h and (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
