def _c5_metric_fft(h):
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]
    N = 1 << ((2*n-1).bit_length())
    ha = np.fft.rfft(h, N)
    hb = np.fft.rfft(b_rev, N)
    conv = np.fft.irfft(ha * hb, N)[:2*n-1]
    return np.max(conv.real) * (2.0 / n)

def _interpolate_to_n(old_h, new_n):
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # enforce sum
    target = new_n / 2.0
    cur = new_h.sum()
    new_h *= target / cur
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # adjust for rounding error
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(new_n):
            if 0.0 <= new_h[idx] + diff <= 1.0:
                new_h[idx] += diff
                break
    return new_h

import time
import math
import random
import numpy as np

def _c5_metric_fft(h: np.ndarray) -> float:
    """Fast C5 evaluation using FFT."""
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    Ha = np.fft.rfft(h, N)
    Hb = np.fft.rfft(b_rev, N)
    conv = np.fft.irfft(Ha * Hb, N)[: 2 * n - 1]
    return float(np.max(conv.real) * (2.0 / n))


def _interpolate_to_n(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function onto a new grid."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    cur = new_h.sum()
    if cur != 0:
        new_h *= target / cur
    np.clip(new_h, 0.0, 1.0, out=new_h)
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(new_n):
            if 0.0 <= new_h[idx] + diff <= 1.0:
                new_h[idx] += diff
                break
    return new_h


def _anneal(start_h: np.ndarray, max_seconds: float) -> tuple:
    """Simulated‑annealing on the C5 objective."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric_fft(best_h)
    cur_c = best_c

    start = time.time()
    T0 = 0.12          # initial temperature
    T_end = 1e-6
    max_step0 = 0.14   # base maximal move size

    while True:
        now = time.time()
        if now - start >= max_seconds:
            break
        progress = (now - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        step = max_step0 * (T / T0)  # shrink with temperature

        # choose two distinct indices
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        low_cap = max(low, -step)
        high_cap = min(high, step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _c5_metric_fft(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / (T + 1e-15)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    return best_h, best_c
