import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for step vector h (full correlation)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """
    Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1.
    Simple redistribution that respects bounds.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterate until the residual is negligible
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        candidates = np.where(mask)[0]
        per = diff / len(candidates)
        for i in candidates:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h, c5, deadline, rng):
    """Simple random pairwise mass exchange hill‑climbing."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        n = 45                      # fallback resolution
        h0 = rng.random(n)
        h0 = _adjust_sum(h0, n / 2.0)

    target = n / 2.0
    h = _adjust_sum(np.clip(h0, 0.0, 1.0), target)
    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Gradient‑descent with soft‑max smoothing of the max‑operator
    # ------------------------------------------------------------------
    iter_cnt = 0
    while time.time() - start_time < budget_s - 5.0:
        # current correlation (raw, without dx)
        corr = np.correlate(h, 1.0 - h, mode="full")
        # soft‑max temperature (increase over time)
        elapsed = time.time() - start_time
        alpha = 5.0 + 45.0 * (elapsed / budget_s)   # from 5 to ~50
        # compute weights
        max_corr = corr.max()
        exp_vals = np.exp(alpha * (corr - max_corr))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective
        g = np.zeros_like(h)
        # lags correspond to indices 0 .. 2n-2, with zero‑lag at n‑1
        for idx, wi in enumerate(w):
            if wi == 0.0:
                continue
            L = idx - (n - 1)          # signed shift
            if L > 0:
                plus = np.concatenate((h[L:], np.zeros(L)))          # h_{j+L}
                minus = np.concatenate((np.zeros(L), h[:n - L]))      # h_{j-L}
            elif L < 0:
                p = -L
                plus = np.concatenate((np.zeros(p), h[:n - p]))       # h_{j+L}
                minus = np.concatenate((h[p:], np.zeros(p)))         # h_{j-L}
            else:  # L == 0
                plus = h
                minus = h
            g += wi * (1.0 - (plus + minus))

        # step size schedule
        eta = 0.5 / (1.0 + iter_cnt * 0.001)
        h_new = h - eta * g
        # projection onto feasible set
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _adjust_sum(h_new, target)

        c5_new = _evaluate_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new

        # accept if improvement, otherwise occasional random perturbation
        cur_c5 = _evaluate_c5(h)
        if c5_new < cur_c5 or rng.random() < 0.01:
            h = h_new
        else:
            # small chance to shake the current state
            if rng.random() < 0.02:
                h = best_h + rng.normal(scale=0.03, size=n)
                h = np.clip(h, 0.0, 1.0)
                h = _adjust_sum(h, target)

        iter_cnt += 1

    # ------------------------------------------------------------------
    # 3. Final fine‑grained hill climbing
    # ------------------------------------------------------------------
    deadline = start_time + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, deadline, rng)

    # ------------------------------------------------------------------
    # 4. Return result
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
