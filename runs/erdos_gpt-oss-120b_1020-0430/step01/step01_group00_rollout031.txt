def _smooth_max_and_grad(h, beta, lam, target):
    n = h.shape[0]
    S = np.sum(h)
    autocorr = np.correlate(h, h, mode='full')
    corr = S - autocorr  # length 2n-1
    max_corr = np.max(corr)
    # For numerical stability, compute shifted exponentials
    exp_shifted = np.exp(beta * (corr - max_corr))
    Z = np.sum(exp_shifted)
    softmax = exp_shifted / Z
    f = max_corr + (np.log(Z) / beta) + lam * (S - target) ** 2

    # smoothmax gradient
    w_arr = softmax
    # compute conv_plus and conv_rev
    conv_plus = np.correlate(w_arr, h, mode='full')[:n]
    conv_rev = np.correlate(w_arr[::-1], h, mode='full')[:n]
    grad = np.ones_like(h) - conv_plus - conv_rev + 2 * lam * (S - target)
    return f, grad
```

def _project_to_feasible(h, n, tol=1e-9, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = np.sum(h)
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]

def _smooth_max_and_grad(h, beta, lam, target):
    n = h.shape[0]
    S = np.sum(h)
    autocorr = np.correlate(h, h, mode='full')
    corr = S - autocorr
    max_corr = np.max(corr)
    # stable exponentials
    exp_shifted = np.exp(beta * (corr - max_corr))
    Z = np.sum(exp_shifted)
    softmax = exp_shifted / Z
    f = max_corr + (np.log(Z) / beta) + lam * (S - target) ** 2

    # gradients
    conv_plus = np.correlate(softmax, h, mode='full')[:n]
    conv_rev = np.correlate(softmax[::-1], h, mode='full')[:n]
    grad = np.ones_like(h) - conv_plus - conv_rev + 2.0 * lam * (S - target)
    return f, grad
```

def _smooth_max_and_grad(h: np.ndarray, beta: float, lam: float, target: float):
    """Smooth approximation of max correlation + quadratic penalty on the sum."""
    n = h.shape[0]
    S = np.sum(h)
    autocorr = np.correlate(h, h, mode="full")
    corr = S - autocorr                     # length 2n‑1
    max_corr = np.max(corr)

    # stable soft‑max
    exp_shifted = np.exp(beta * (corr - max_corr))
    Z = np.sum(exp_shifted)
    softmax = exp_shifted / Z                # weights for each shift

    # smooth objective
    f = max_corr + np.log(Z) / beta + lam * (S - target) ** 2

    # gradient of the smooth part
    conv_plus = np.correlate(softmax, h, mode="full")[:n]
    conv_rev = np.correlate(softmax[::-1], h, mode="full")[:n]
    grad = np.ones_like(h) - conv_plus - conv_rev + 2.0 * lam * (S - target)
    return f, grad


def _objective_with_grad(x: np.ndarray):
    """Wrapper used by scipy.optimize – reads globals for parameters."""
    return _smooth_max_and_grad(x, _current_beta, _current_lambda, _current_target)


def _optimize_h(initial_h: np.ndarray, n: int, time_left):
    """Run several L‑BFGS‑B refinements with increasing smoothness."""
    target = n / 2.0
    best_h_local = None
    best_c5_local = float("inf")
    restarts = 5

    for r in range(restarts):
        if time_left() < 5.0:
            break
        if r == 0:
            h_cur = _project_to_feasible(initial_h, n)
        else:
            h_cur = _project_to_feasible(np.random.rand(n), n)

        lam = 200.0                     # strength of the sum‑penalty
        for beta in (2.0, 5.0, 15.0, 30.0, 60.0):
            if time_left() < 4.0:
                break
            global _current_beta, _current_lambda, _current_target
            _current_beta = beta
            _current_lambda = lam
            _current_target = target

            bounds = [(0.0, 1.0)] * n
            try:
                res = optimize.minimize(
                    _objective_with_grad,
                    h_cur,
                    method="L-BFGS-B",
                    jac=True,
                    bounds=bounds,
                    options={"maxiter": 200, "ftol": 1e-12, "disp": False},
                )
                h_cur = res.x
            except Exception:
                # keep the current iterate on failure
                pass
            # increase penalty a little to tighten the sum constraint
            lam *= 2.0

        h_proj = _project_to_feasible(h_cur, n)
        c5_val = _c5_from_h(h_proj)
        if c5_val < best_c5_local:
            best_c5_local = c5_val
            best_h_local = h_proj.copy()
            if best_c5_local < 0.3808:
                break
    return best_h_local, best_c5_local


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as list), its C5 bound, and the number of samples.
    The algorithm uses a smooth‑max L‑BFGS‑B optimisation with several
    restarts and a final projection onto the feasible set.
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # initial construction, if supplied by the harness
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target_bound = 0.3808

    # candidate grid sizes (prefer the supplied size, then multiples)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for mul in (2, 3, 4):
        if init_h is not None:
            n = init_h.size * mul
        else:
            n = 64 * mul
        if n <= 512:
            n_candidates.append(int(n))
    for n in (80, 96, 128, 160, 192, 224, 256, 300):
        if n <= 512:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # build a base vector for this resolution
        if init_h is not None:
            if init_h.size == n:
                base_h = init_h.copy()
            else:
                # linear interpolation to the finer grid
                x_src = (np.arange(init_h.size) + 0.5) * (2.0 / init_h.size)
                x_tgt = (np.arange(n) + 0.5) * (2.0 / n)
                base_h = np.interp(x_tgt, x_src, init_h)
        else:
            base_h = np.full(n, 0.5)

        h_opt, c5_val = _optimize_h(base_h, n, time_left)
        if h_opt is not None and c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target_bound:
                break

    # fallback – uniform construction (should never be needed)
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
