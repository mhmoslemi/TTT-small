def evaluate_fft(h):
    n = len(h)
    L = 2*n
    H = np.fft.rfft(h, n=L)
    K = np.fft.rfft(1 - h, n=L)
    corr = np.fft.irfft(H * np.conj(K), n=L)
    # result length L, but we want full correlation length 2n-1 (the first 2n-1 elements)
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def _eval_c5(h):
    """C5 value for a step vector h (float array, values in [0,1])."""
    n = h.shape[0]
    dx = 2.0 / n
    # full linear correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _binary_from_continuous(x, target_sum):
    """
    Convert a continuous vector to a binary one with the same sum (or as close as possible).
    The entries with the largest values become 1.
    """
    n = x.shape[0]
    half = int(round(target_sum))
    idx = np.argpartition(x, -half)[-half:]   # indices of the largest `half` entries
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function (binary vector) that lowers the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose discretisation size
    # --------------------------------------------------------------
    n = 200                     # even number → exact half‑ones feasible
    target_sum = n // 2        # required number of ones (integral)

    # --------------------------------------------------------------
    # 2. Initialise current state (use provided construction if any)
    # --------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init = np.asarray(init_vec, dtype=np.float64)
        if init.size != n:
            # resample to the chosen resolution
            old_n = init.size
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(new_x, old_x, init)
        cur_h = _binary_from_continuous(init, target_sum)
    else:
        # pure random binary start
        cur_h = np.zeros(n, dtype=np.float64)
        cur_h[rng.choice(n, target_sum, replace=False)] = 1.0

    cur_c5 = _eval_c5(cur_h)

    # --------------------------------------------------------------
    # 3. Simulated annealing over pairwise swaps
    # --------------------------------------------------------------
    best_h = cur_h.copy()
    best_c5 = cur_c5

    max_temp = 0.5
    min_temp = 0.001
    no_improve = 0          # counter for stagnation
    RESTART_AFTER = 80000   # swaps without improvement -> restart

    while True:
        now = time.time()
        if now >= start_time + budget_s - 0.5:
            break

        # linear cooling based on overall progress
        progress = (now - start_time) / (budget_s - 0.5)
        T = max_temp * (1.0 - progress) + min_temp * progress

        # pick a random 1‑position and a random 0‑position
        i = rng.integers(n)
        while cur_h[i] < 0.5:            # look for a 1
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] > 0.5:            # look for a 0
            j = rng.integers(n)

        # propose a swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = _eval_c5(cur_h)

        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 + 1e-12 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # rollback the swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1

        # occasional restart to escape deep local minima
        if no_improve >= RESTART_AFTER:
            cur_h = np.zeros(n, dtype=np.float64)
            cur_h[rng.choice(n, target_sum, replace=False)] = 1.0
            cur_c5 = _eval_c5(cur_h)
            no_improve = 0

    # --------------------------------------------------------------
    # 4. Final greedy hill‑climbing from the best found state
    # --------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5
    horizon = int((start_time + budget_s - time.time()) / 0.00005)  # rough remaining steps
    for _ in range(max(20000, horizon)):
        i = rng.integers(n)
        while cur_h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] > 0.5:
            j = rng.integers(n)

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = _eval_c5(cur_h)

        if new_c5 + 1e-12 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0

        if time.time() >= start_time + budget_s - 0.05:
            break

    # --------------------------------------------------------------
    # 5. Return the best construction found
    # --------------------------------------------------------------

    final_c5 = float(_eval_c5(best_h))
    return best_h, final_c5, n
