        val = c5_of_h(h)

                best_h = h.copy()
        else:
            # revert swap
            h[i1], h[i0] = h[i0], h[i1]
        # cool
        temp = temp0 * (1 - it / max_iter) # linear cooling maybe

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h)=n/2 by iterative water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # decide where we can move mass
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 = max correlation * dx (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for every lag."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint(x: np.ndarray) -> float:
    """Enforce sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h_slsqp(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    """Run a single SLSQP optimisation, returns (h_opt, c5)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _random_binary_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary vector with exactly n/2 ones."""
    w = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=w, replace=False)
    h[idx] = 1.0
    return h


def _random_cosine_h(n: int, J: int, rng: np.random.Generator) -> np.ndarray:
    """Random cosine series h = 0.5 + sum a_j cos(jπx), clipped and projected."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    cos_mat = np.cos(np.outer(x, np.arange(1, J + 1) * np.pi))
    coeffs = rng.uniform(-0.5, 0.5, size=J)
    h = 0.5 + cos_mat @ coeffs
    return _project_to_feasible(h, n)


def _subgradient_descent(h_start: np.ndarray, max_iter: int, rng: np.random.Generator) -> tuple:
    """Simple projected sub‑gradient descent on max‑correlation objective."""
    n = h_start.size
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    step0 = 0.8
    for it in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        k_abs = abs(k)

        grad = np.ones(n)
        if k_abs < n:
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]

        eta = step0 / np.sqrt(it)
        h_new = h - eta * grad
        h_new = _project_to_feasible(h_new, n)

        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        h = h_new

        # occasional stochastic perturbation to escape flat regions
        if it % 200 == 0:
            if rng.random() < 0.1:
                noise = rng.normal(scale=0.03, size=n)
                h = _project_to_feasible(h + noise, n)

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # retrieve any provided construction
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = None

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    # candidate grid sizes (prefer the given size, then enlargements up to 300)
    n_candidates = []
    if initial_h is not None:
        n_candidates.append(initial_h.size)
    for mul in (2, 3, 4):
        if initial_h is not None:

        else:

        if n <= 300:

    # some extra even sizes
    for n in (80, 100, 128, 160, 200, 240, 256, 300):

    n_candidates = sorted(set(n_candidates))

    rng = np.random.default_rng(seed)

    for n in n_candidates:
        if time_left() < 15.0:

        # build a base h for this grid size
        if initial_h is not None and initial_h.size != n:

            if c5_val < best_c5:

                if best_c5 < target_bound:

                    if best_c5 < target_bound:

        n = 64
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
