import time
import numpy as np
from scipy import optimize

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final adjustment
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 = max correlation * (2/n)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _grad_corr_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of corr(shift) = Σ h[i] (1-h[i+shift]) w.r.t h."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        if end > 0:
            g[:end] += (1.0 - h[shift:])
            g[shift:] += -h[:end]
    else:
        s = -shift
        end = n - s
        if end > 0:
            g[s:] += (1.0 - h[:end])
            g[:end] += -h[s:]
    return g


def _refine_gradient(h0: np.ndarray, time_end: float) -> tuple:
    """Projected sub‑gradient descent on max‑correlation."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    step = 0.5
    eps = 1e-12
    while time.time() < time_end - 0.02:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        idxs = np.where(corr >= max_corr - eps)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            grad += _grad_corr_shift(h, idx - (n - 1))
        grad /= len(idxs)
        d = -grad
        d -= np.mean(d)               # keep sum unchanged before projection
        improved = False
        step_try = step
        for _ in range(10):
            h_new = h + step_try * d
            h_new = _project_to_feasible(h_new, n)
            new_c5 = _c5_from_h(h_new)
            if new_c5 < best_c5 - 1e-12:
                h = h_new
                best_h = h_new.copy()
                best_c5 = new_c5
                step = step_try * 1.05
                improved = True
                break
            step_try *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
    return best_h, best_c5


def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """Single SLSQP run, similar to the parent implementation."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h that minimizes the Erdős C5 bound."""
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial construction, if supplied
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target = 0.3808

    # candidate grid sizes (ascending)
    n_candidates = []
    if init_h is not None:
        n_candidates.append(init_h.size)
    for mul in (2, 3, 4):
        if init_h is not None:
            n = int(init_h.size * mul)
        else:
            n = 64 * mul
        if n <= 300:
            n_candidates.append(n)
    for n in (80, 100, 128, 160, 200, 256, 300):
        if n <= 300:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 4.0:
            break

        # build base h for this size
        if init_h is not None:
            if init_h.size != n:
                base_h = _upscale_h(init_h, n)
            else:
                base_h = init_h.copy()
        else:
            # random binary with exact half ones
            base_h = np.zeros(n)
            idx = np.random.choice(n, n // 2, replace=False)
            base_h[idx] = 1.0

        base_h = _project_to_feasible(base_h, n)

        # gradient refinement (major part of the budget for this n)
        t_end = start + budget_s - 0.5  # leave a small margin
        refined_h, refined_c5 = _refine_gradient(base_h, t_end)

        if refined_c5 < best_c5:
            best_h, best_c5, best_n = refined_h, refined_c5, n
            if best_c5 < target:
                break

        # a few SLSQP restarts to polish the result
        for attempt in range(3):
            if time_left() < 4.0:
                break
            perturb = 0.15 * np.random.randn(n)
            start_h = refined_h + perturb
            start_h = _project_to_feasible(start_h, n)
            h_opt, c5_val = _solve_h(start_h, max_iter=180)
            if c5_val < best_c5:

    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
