    n = h.shape[0]
    shifts = np.arange(-(n-1), n)  # shape (2n-1,)
    idx_a_minus = a - shifts
    idx_a_plus = a + shifts
    idx_b_minus = b - shifts
    idx_b_plus = b + shifts

    # for each idx, need value of h if 0 <= idx < n else 0

        vals = np.zeros_like(idxs, dtype=int)
        vals[in_range] = h[idxs[in_range]]
        return vals

    term = -get_vals(idx_a_minus) + get_vals(idx_b_minus) - get_vals(idx_a_plus) + get_vals(idx_b_plus)

    delta = np.zeros_like(shifts, dtype=int)

    # new auto_corr
    auto_corr = auto_corr + term + delta

```python
import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _compute_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s:] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _gd_optimize(h0, target, max_iter=1500, step0=0.6, decay=0.997):
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

def _smooth_max(x, alpha=30.0):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _slsqp_optimize(h0, target, alpha=30.0, max_iter=1000):
    n = h0.shape[0]
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)
    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)

        return np.array(vec, copy=True)
    x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, vec)

def _binary_anneal(h_start, rng, time_limit):
    n = len(h_start)
    dx = 2.0 / n
    target_sum = n // 2
    # ensure integer sum
    h = h_start.copy()
    # compute auto-correlation
    auto = np.correlate(h, h, mode='full')
    cur_min = auto.min()
    best_h = h.copy()
    best_min = cur_min
    # precompute shift array
    shifts = np.arange(-(n-1), n)
    T = 2.0
    decay = 0.9995
    start = time.time()
    while time.time() - start < time_limit:
        # pick swap
        a = rng.randint(n)
        while h[a] == 0:
            a = rng.randint(n)
        b = rng.randint(n)
        while h[b] == 1:
            b = rng.randint(n)
        # compute delta vector
        # term contributions
        idx_a_minus = a - shifts
        idx_a_plus = a + shifts
        idx_b_minus = b - shifts
        idx_b_plus = b + shifts
        # get values
        # we can compute slices:
        # using function _vals_at
        # but we'll reuse same approach each iteration; maybe slower but okay
        # compute term
        def _vals(idxs):
            valid = (idxs >= 0) & (idxs < n)
            vals = np.zeros_like(idxs, dtype=int)
            vals[valid] = h[idxs[valid]]
            return vals
        term = -_vals(idx_a_minus) + _vals(idx_b_minus) - _vals(idx_a_plus) + _vals(idx_b_plus)
        delta = np.zeros_like(shifts, dtype=int)
        delta[shifts == 0] = 2
        d = b - a
        if d != 0:
            delta[shifts == d] = -1
            delta[shifts == -d] = -1
        delta_auto = term + delta
        candidate_auto = auto + delta_auto
        cand_min = candidate_auto.min()
        accept = False
        if cand_min > cur_min:
            accept = True
        else:
            prob = np.exp((cand_min - cur_min) / T)
            if rng.rand() < prob:
                accept = True
        if accept:
            # apply swap
            h[a] = 0
            h[b] = 1
            auto = candidate_auto
            cur_min = cand_min
            if cur_min > best_min:
                best_min = cur_min
                best_h = h.copy()
        T *= decay
    # return best binary vector and corresponding c5 bound
    c5 = 1.0 - (best_min * 2.0 / n)
    return best_h.astype(float), c5

import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0, target, max_iter=1500, step0=0.6, decay=0.997):
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def _smooth_max(x, alpha=30.0):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def _slsqp_optimize(h0, target, alpha=30.0, max_iter=1000):
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({"type": "eq", "fun": lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


def _upscale(vec, n):
    if len(vec) == n:
        return np.array(vec, copy=True)
    x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, vec)


def _binary_from_continuous(h):
    n = h.shape[0]
    idx = np.argsort(-h)[: n // 2]
    bh = np.zeros(n, dtype=float)
    bh[idx] = 1.0
    return bh


def _binary_anneal(start_h, rng, time_budget):
    n = start_h.shape[0]
    dx = 2.0 / n
    auto = np.correlate(start_h, start_h, mode="full")
    cur_min = auto.min()
    best_h = start_h.copy()
    best_min = cur_min

    shifts = np.arange(-(n - 1), n)
    T = 2.0
    decay = 0.9995
    end_time = time.time() + time_budget

    while time.time() < end_time:
        # pick a 1 and a 0 to swap
        a = rng.randint(n)
        while start_h[a] == 0:
            a = rng.randint(n)
        b = rng.randint(n)
        while start_h[b] == 1:
            b = rng.randint(n)

        # helper for out‑of‑range values
        def vals(idxs):
            ok = (idxs >= 0) & (idxs < n)
            v = np.zeros_like(idxs, dtype=int)
            v[ok] = start_h[idxs[ok]]
            return v

        idx_a_m = a - shifts
        idx_a_p = a + shifts
        idx_b_m = b - shifts
        idx_b_p = b + shifts

        term = -vals(idx_a_m) + vals(idx_b_m) - vals(idx_a_p) + vals(idx_b_p)

        delta = np.zeros_like(shifts, dtype=int)
        delta[shifts == 0] = 2
        d = b - a
        if d != 0:
            delta[shifts == d] = -1
            delta[shifts == -d] = -1

        delta_auto = term + delta
        cand_auto = auto + delta_auto
        cand_min = cand_auto.min()

        accept = False
        if cand_min > cur_min:
            accept = True
        else:
            if rng.rand() < np.exp((cand_min - cur_min) / T):
                accept = True

        if accept:
            start_h[a] = 0.0
            start_h[b] = 1.0
            auto = cand_auto
            cur_min = cand_min
            if cur_min > best_min:
                best_min = cur_min
                best_h = start_h.copy()
        T *= decay

    c5 = 1.0 - best_min * (2.0 / n)
    return best_h, c5
