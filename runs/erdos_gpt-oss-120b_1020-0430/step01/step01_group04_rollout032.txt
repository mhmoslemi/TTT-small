Implement refine_slsqp(start_vec, n, time_deadline):

   target = n/2
   cons = [{'type':'eq','fun': lambda h: np.sum(h) - target }]
   bounds = [(0.0, 1.0)] * n
   # Perform series of minimizations with increasing alpha
   alphas = [5.0, 10.0, 30.0, 60.0]
   x = start_vec.copy()
   for alpha in alphas:
        if time.time() > time_deadline:
            break
        res = minimize(lambda h: softmax_c5(h, alpha=alpha),
                       x, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter':500,'ftol':1e-9,'disp':False})
        if not res.success:
            continue
        x = np.clip(res.x, 0.0, 1.0)
   # final compute
   final_c5 = compute_c5(x)
   return x, final_c5

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core evaluation
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


# ------------------------------------------------------------
# Simple constructions
# ------------------------------------------------------------
def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, re‑normalised."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    return np.clip(up, 0.0, 1.0)


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible start (real values, sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def generate_random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    arr[ones] = 1.0
    return arr


# ------------------------------------------------------------
# Hill‑climbing on binary vectors
# ------------------------------------------------------------
def hill_climb_binary(start: np.ndarray, max_iters: int,
                      rng: np.random.Generator, deadline: float) -> tuple:
    """
    Simple simulated–annealing hill climb on a binary vector.
    Returns (best_vec, best_c5).
    """
    cur = start.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    n = cur.size
    # temperature schedule (high at start, cooling to zero)
    T0 = 0.02

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # choose a random 1 and a random 0 to swap
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = compute_c5(cur)

        # acceptance test
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best = cur.copy()
        else:
            # possibly accept worsening move
            temp = T0 * (1.0 - it / max_iters)
            if temp > 0 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best = cur.copy()
            else:
                # revert swap
                cur[i], cur[j] = cur[j], cur[i]

    return best, best_c5


# ------------------------------------------------------------
# SLSQP refinement for real‑valued vectors
# ------------------------------------------------------------
def refine_slsqp(start: np.ndarray, n: int, deadline: float) -> tuple:
    """
    Run a few SLSQP stages with increasing smoothness.
    Returns (h_opt, c5_opt) or (None, None) if time runs out.
    """
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    bounds = [(0.0, 1.0)] * n

    # schedule of alphas (smooth → sharp)
    alphas = [5.0, 15.0, 30.0, 60.0]
    x = np.clip(start, 0.0, 1.0)

    for a in alphas:
        if time.time() > deadline:
            break
        res = minimize(lambda h: softmax_c5(h, alpha=a),
                       x,
                       method="SLSQP",
                       bounds=bounds,
                       constraints=cons,
                       options={"maxiter": 800, "ftol": 1e-9, "disp": False})
        if not res.success:
            continue
        x = np.clip(res.x, 0.0, 1.0)

    c5 = compute_c5(x)
    return x, c5

# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # --------------------------------------------------------

    for n in candidate_ns:

        # ---- build a small pool of starting vectors ----
        starts = []

        # up‑sampled parent (float)
        if parent_h is not None:
            if parent_n == n:

            else:

        # binary seed obtained from up‑sampled parent (top‑half threshold)
