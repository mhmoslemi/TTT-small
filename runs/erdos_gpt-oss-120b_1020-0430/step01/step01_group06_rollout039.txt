def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, math, time
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # get n
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    except Exception:
        n = 70
        init = None
    target_ones = n // 2
    dx = 2.0 / n

    def c5_of(v):
        # v is numpy array of floats
        return float(np.max(np.correlate(v, 1.0 - v, mode='full')) * dx)

    # helper to project binary vector to correct number of ones
    def make_binary(v):
        v = (v >= 0.5).astype(float)
        cur = int(v.sum())
        if cur > target_ones:
            idx = np.where(v == 1)[0]
            np.random.shuffle(idx)
            v[idx[:cur - target_ones]] = 0.0
        elif cur < target_ones:
            idx = np.where(v == 0)[0]
            np.random.shuffle(idx)
            v[idx[:target_ones - cur]] = 1.0
        return v

    def random_binary():
        a = np.zeros(n, dtype=float)
        a[:target_ones] = 1.0
        np.random.shuffle(a)
        return a

    # ----- hill climbing on binary vectors -----
    def hill_climb(v):
        v = make_binary(v)
        cur_c5 = c5_of(v)
        improved = True
        while improved and (time.time() - start) < budget_s * 0.9:
            improved = False
            ones = np.where(v == 1)[0]
            zeros = np.where(v == 0)[0]
            # random order to avoid deterministic cycles
            np.random.shuffle(ones)
            np.random.shuffle(zeros)
            for i in ones:
                for j in zeros:
                    # quick test: if swapping would change anything
                    # compute new vector
                    v_new = v.copy()
                    v_new[i] = 0.0
                    v_new[j] = 1.0
                    new_c5 = c5_of(v_new)
                    if new_c5 + 1e-12 < cur_c5:
                        v = v_new
                        cur_c5 = new_c5
                        improved = True
                        break
                if improved:
                    break
        return v, cur_c5

    # ----- simulated annealing with fractional moves -----
    def anneal(v, cur_c5, max_iters):
        cur = v.copy()
        best = cur.copy()
        best_c5 = cur_c5
        T = max(1e-5, cur_c5 * 0.02)
        decay = 0.9995
        it = 0
        while it < max_iters and (time.time() - start) < budget_s * 0.97:
            # random transfer move
            i, j = random.sample(range(n), 2)
            hi, hj = cur[i], cur[j]
            max_up = min(1.0 - hi, hj)    # increase i, decrease j
            max_down = min(hi, 1.0 - hj)  # decrease i, increase j
            if max_up + max_down > 0:
                delta = (random.random() * (max_up + max_down)) - max_down
                if abs(delta) > 1e-14:
                    nxt = cur.copy()
                    nxt[i] = hi + delta
                    nxt[j] = hj - delta
                    new_c5 = c5_of(nxt)
                    accept = (new_c5 < best_c5) or (random.random() < math.exp(-(new_c5 - best_c5) / T))
                    if accept:
                        cur = nxt
                        if new_c5 < best_c5:
                            best = nxt.copy()
                            best_c5 = new_c5
            # targeted move every few iterations
            if it % 500 == 0:
                corr = np.correlate(cur, 1.0 - cur, mode='full')
                idx_max = int(np.argmax(corr))
                s = idx_max - (n - 1)
                if s >= 0:
                    i_inds = np.arange(n - s)
                    p_inds = i_inds + s
                else:
                    i_inds = np.arange(-s, n)
                    p_inds = i_inds + s
                contrib = cur[i_inds] * (1.0 - cur[p_inds])
                k = int(np.argmax(contrib))
                i_best, p_best = i_inds[k], p_inds[k]
                eps_max = min(cur[i_best], 1.0 - cur[p_best])
                if eps_max > 1e-12:
                    eps = eps_max * 0.5
                    nxt = cur.copy()
                    nxt[i_best] -= eps
                    nxt[p_best] += eps
                    new_c5 = c5_of(nxt)
                    if new_c5 < best_c5 or random.random() < math.exp(-(new_c5 - best_c5) / T):
                        cur = nxt
                        if new_c5 < best_c5:
                            best = nxt.copy()
                            best_c5 = new_c5
            T *= decay
            it += 1
        return best, best_c5

    best_h = None
    best_val = float('inf')

    # start with initial if available
    if init is not None:
        # try binary hill climb from initial
        v0, val0 = hill_climb(init)
        if val0 < best_val:

        # optionally anneal from this binary solution
        v_ann, val_ann = anneal(v0, val0, max_iters=20000)
        if val_ann < best_val:

    # main random restarts loop
    while (time.time() - start) < budget_s * 0.85:

        v0 = random_binary()
        v1, val1 = hill_climb(v0)
        if val1 < best_val:
            best_h, best_val = v1.copy(), val1
        # occasional anneal to refine
        if random.random() < 0.2:
            v2, val2 = anneal(v1, val1, max_iters=10000)
            if val2 < best_val:
                best_h, best_val = v2.copy(), val2
        # early exit if already good enough
        if best_val <= 0.3808:
            break

    # final safety: ensure exact sum
    err = best_h.sum() - target_ones
    if abs(err) > 1e-9:
        # adjust smallest deviation
        if err > 0:
            # reduce some ones
            idx = np.where(best_h > 0)[0]
            np.random.shuffle(idx)
            for i in idx:
                delta = min(err, best_h[i])
                best_h[i] -= delta
                err -= delta
                if abs(err) < 1e-12:

        else:

            for i in idx:
                delta = min(-err, 1 - best_h[i])
                best_h[i] += delta
                err += delta
                if abs(err) < 1e-12:
                    break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), float(final_c5), n)

def anneal(v, max_iters):
    cur = v.copy()
    cur_c5 = c5_of(cur)
    best = cur.copy()
    best_c5 = cur_c5
    # initial temperature: a fraction of cur_c5 maybe 0.01~0.1
    T = max(1e-5, cur_c5 * 0.02)
    decay = 0.9995
    for it in range(max_iters):
        i, j = random.sample(range(n), 2)
        hi, hj = cur[i], cur[j]
        max_up = min(1.0 - hi, hj)
        max_down = min(hi, 1.0 - hj)
        if max_up + max_down == 0:
            continue
        # Choose delta uniformly in feasible interval
        delta = (random.random() * (max_up + max_down)) - max_down
        if abs(delta) < 1e-14:
            continue
        nxt = cur.copy()
        nxt[i] = hi + delta
        nxt[j] = hj - delta
        new_c5 = c5_of(nxt)
        # acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur = nxt
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best = nxt.copy()
                best_c5 = new_c5
        # targeted move occasionally
        if it % 500 == 0:
            corr = np.correlate(cur, 1.0 - cur, mode='full')
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)
            if shift >= 0:
                i_inds = np.arange(n - shift)
                p_inds = i_inds + shift
            else:
                i_inds = np.arange(-shift, n)
                p_inds = i_inds + shift
            contrib = cur[i_inds] * (1.0 - cur[p_inds])
            k = int(np.argmax(contrib))
            i_best, p_best = i_inds[k], p_inds[k]
            eps_max = min(cur[i_best], 1.0 - cur[p_best])
            if eps_max > 1e-12:
                eps = eps_max * 0.5

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, math, time
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Problem dimensions and helpers
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n = init_vec.shape[0]
    except Exception:          # no initial_h_values supplied
        n = 70
        init_vec = None

    target_ones = n // 2                     # ∫h = 1  ⇒  sum = n/2
    dx = 2.0 / n

    def c5_of(v: np.ndarray) -> float:
        """max overlap integral for a given vector v."""
        return float(np.max(np.correlate(v, 1.0 - v, mode="full")) * dx)

    # ------------------------------------------------------------------
    # Construction utilities
    # ------------------------------------------------------------------
    def make_binary(v: np.ndarray) -> np.ndarray:
        """Project a vector to a binary one with exactly target_ones ones."""
        v = (v >= 0.5).astype(float)
        cur = int(v.sum())
        if cur > target_ones:
            idx = np.where(v == 1)[0]
            np.random.shuffle(idx)
            v[idx[:cur - target_ones]] = 0.0
        elif cur < target_ones:
            idx = np.where(v == 0)[0]
            np.random.shuffle(idx)
            v[idx[:target_ones - cur]] = 1.0
        return v

    def random_binary() -> np.ndarray:
        """Uniform random binary vector with the correct sum."""
        a = np.zeros(n, dtype=float)
        a[:target_ones] = 1.0
        np.random.shuffle(a)
        return a

    def structured_start() -> np.ndarray:
        """A semi‑periodic start: half the vector repeats itself."""
        half = n // 2
        first_ones = target_ones // 2 + (target_ones % 2)   # ceil(target_ones/2)
        first = np.zeros(half, dtype=float)
        first[:first_ones] = 1.0
        np.random.shuffle(first)
        second = first.copy()
        # remove one 1 from the second half to hit the exact total
        ones_idx = np.where(second == 1)[0]
        if ones_idx.size:
            second[np.random.choice(ones_idx)] = 0.0
        return np.concatenate([first, second])

    # ------------------------------------------------------------------
    # Local improvement on binary vectors (steepest descent swaps)
    # ------------------------------------------------------------------
    def hill_climb(v: np.ndarray):
        v = make_binary(v)
        cur = c5_of(v)
        improved = True
        while improved and (time.time() - start_time) < budget_s * 0.90:
            improved = False
            ones = np.where(v == 1)[0]
            zeros = np.where(v == 0)[0]
            np.random.shuffle(ones)
            np.random.shuffle(zeros)
            for i in ones:
                for j in zeros:
                    vn = v.copy()
                    vn[i] = 0.0
                    vn[j] = 1.0
                    new = c5_of(vn)
                    if new + 1e-12 < cur:
                        v = vn
                        cur = new
                        improved = True
                        break
                if improved:
                    break
        return v, cur

    # ------------------------------------------------------------------
    # Simulated annealing on continuous vectors (mass transfers)
    # ------------------------------------------------------------------
    def anneal(start_vec: np.ndarray, max_iters: int):
        cur = start_vec.copy()
        cur_val = c5_of(cur)
        best = cur.copy()
        best_val = cur_val
        T = max(1e-5, cur_val * 0.02)
        decay = 0.9995
        for it in range(max_iters):
            # ----- random transfer move -----
            i, j = random.sample(range(n), 2)
            hi, hj = cur[i], cur[j]
            max_up = min(1.0 - hi, hj)          # increase i, decrease j
            max_down = min(hi, 1.0 - hj)        # decrease i, increase j
            if max_up + max_down > 0:

                    nxt[i] = hi + delta
                    nxt[j] = hj - delta
                    new_val = c5_of(nxt)
                    if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / T):

                        cur_val = new_val
                        if new_val < best_val:
                            best = nxt.copy()
                            best_val = new_val
            # ----- targeted move every 500 iterations -----
            if it % 500 == 0:

        else:

        cand, cand_val = anneal(start.astype(float), max_iters=15000)
        if cand_val < best_val:
            best_h, best_val = cand.copy(), cand_val
        # a quick binary polishing pass (occasionally)
        if random.random() < 0.15:
            bin_cand, bin_val = hill_climb(cand)
            if bin_val < best_val:
                best_h, best_val = bin_cand.copy(), bin_val
        # stop early if we already beat the target
        if best_val <= 0.3808:
            break

    # Final correction of the sum (tiny numerical drift)
    # ------------------------------------------------------------------
    err = best_h.sum() - target_ones
    if abs(err) > 1e-9:

            np.random.shuffle(idx)

            for i in idx:
                d = min(1.0 - best_h[i], -err)
                best_h[i] += d
                err += d
                if abs(err) < 1e-12:
                    break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), float(final_c5), n)
