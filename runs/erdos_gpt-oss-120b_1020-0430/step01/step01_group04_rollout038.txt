def binary_hill_climb(n, rng, deadline, max_iter=30000, temp_start=5.0):
    half = n//2
    h = np.zeros(n)
    indices = rng.choice(n, size=half, replace=False)
    h[indices] = 1.0
    conv = np.correlate(h, 1-h, mode='full')
    cur_max = conv.max()
    best_h = h.copy()
    best_val = cur_max
    temp = temp_start
    iter_count = 0
    while time.time() < deadline and iter_count < max_iter and temp > 1e-4:
        # pick random swap
        # sample random 1 index
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1-h)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # make swap
        h[i], h[j] = 0.0, 1.0  # swapped values
        new_conv = np.correlate(h, 1-h, mode='full')
        new_max = new_conv.max()
        # accept condition
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            conv = new_conv
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        # temperature schedule
        temp *= 0.9995  # maybe degrade slowly
        iter_count += 1
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Core evaluation utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


# ----------------------------------------------------------------------
# Local continuous refinement (SLSQP)
# ----------------------------------------------------------------------
def slsqp_refine(start_h: np.ndarray, n: int, deadline: float) -> tuple:
    """Two‑stage SLSQP optimisation starting from ``start_h``."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}

    # stage 1 – gentle smoothing
    if time.time() > deadline:
        return start_h, compute_c5(start_h)
    res1 = minimize(
        lambda h: softmax_c5(h, alpha=5.0),
        start_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-9, "disp": False},
    )
    h = start_h if not res1.success else np.clip(res1.x, 0.0, 1.0)

    # stage 2 – sharper smoothing
    if time.time() > deadline:
        return h, compute_c5(h)
    res2 = minimize(
        lambda h: softmax_c5(h, alpha=30.0),
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-9, "disp": False},
    )
    if res2.success:
        h = np.clip(res2.x, 0.0, 1.0)

    return h, compute_c5(h)


# ----------------------------------------------------------------------
# Simulated annealing on the exact C5 (continuous moves)
# ----------------------------------------------------------------------
def simulated_annealing(start_h: np.ndarray, n: int, rng: np.random.Generator,
                        deadline: float, max_iter: int = 40000) -> tuple:
    """Neighbour‑search that preserves the integral exactly."""
    target = n / 2.0
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5

    temp = 0.12          # initial temperature
    min_temp = 1e-6
    decay = 0.99993

    for it in range(max_iter):
        if time.time() > deadline:
            break

        i, j = rng.integers(n, size=2, endpoint=False)
        if i == j:
            continue

        up = 1.0 - cur_h[i]    # how much we can increase at i
        down = cur_h[j]        # how much we can decrease at j
        limit = min(up, down)
        if limit <= 0.0:
            continue

        # propose a signed delta scaled by temperature
        delta = rng.uniform(-limit, limit) * temp
        # clip to the admissible interval
        delta = max(-down, min(delta, up))
        if abs(delta) < 1e-12:
            continue

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5

        temp = max(min_temp, temp * decay)

        # occasional re‑projection to fix tiny drift
        if (it + 1) % 5000 == 0:
            diff = target - cur_h.sum()
            if abs(diff) > 1e-12:
                cur_h += diff / n
                cur_h = np.clip(cur_h, 0.0, 1.0)
                cur_c5 = compute_c5(cur_h)
                if cur_c5 < best_c5:
                    best_h, best_c5 = cur_h.copy(), cur_c5

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate grid sizes (even numbers for binary patterns)
    candidate_ns = [200, 180, 160, 140, 120, 100, 80]

    # incorporate parent construction if available
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n % 2 == 0 and parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    # ------------------------------------------------------------------
    # Sweep over resolutions
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > deadline:
            break
        half = n // 2
        target_sum = half

        starts = []

        # 1. parent (upsampled / downsampled if needed)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                # linear interpolation to new size
                old_x = np.linspace(0.0, 2.0, parent_n, endpoint=False) + 1.0 / parent_n
                new_x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
                up = np.interp(new_x, old_x, parent_h)
                up = up * (target_sum / up.sum())
                up = np.clip(up, 0.0, 1.0)
                diff = target_sum - up.sum()
                if abs(diff) > 1e-12:
                    up += diff / n
                    up = np.clip(up, 0.0, 1.0)
                starts.append(up)

        # 2. few random continuous feasible vectors
        for _ in range(3):
            v = rng.random(n)
            v = v * (target_sum / v.sum())
            v = np.clip(v, 0.0, 1.0)
            dif = target_sum - v.sum()
            if abs(dif) > 1e-12:
                v += dif / n
                v = np.clip(v, 0.0, 1.0)
            starts.append(v)

        # 3. binary (balanced) vectors
        for _ in range(3):
            b = np.zeros(n)
            idx = rng.choice(n, size=half, replace=False)
            b[idx] = 1.0
            starts.append(b)

        # ------------------------------------------------------------------
        # Optimise each starting point
        # ------------------------------------------------------------------
        for sv in starts:
            if time.time() > deadline:
                break

            # a) continuous refinement via SLSQP
            h_ref, c5_ref = slsqp_refine(sv, n, deadline)
            if c5_ref < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_ref.copy(), c5_ref, n

            # b) simulated annealing from the refined point
            if time.time() < deadline:
                h_sa, c5_sa = simulated_annealing(h_ref, n, rng, deadline)
                if c5_sa < best_c5 - 1e-9:
                    best_h, best_c5, best_n = h_sa.copy(), c5_sa, n

    # ------------------------------------------------------------------
    # Fallback construction if nothing succeeded
    # ------------------------------------------------------------------
    if best_h is None:
        n = 80
        best_n = n
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Final sanity check (integral constraint)
    # ------------------------------------------------------------------
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
