import time, math, random, numpy as np

def _c5_metric(h, dx):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for a sampled step function."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _scale_to_sum(h, target):
    """Scale/clamp h to satisfy 0≤h≤1 and exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # tiny correction on a feasible entry
        for idx in np.random.permutation(h.size):
            lo, hi = 0.0 - h[idx], 1.0 - h[idx]
            if lo <= diff <= hi:
                h[idx] += diff
                break
    return h

def _optimize(h, budget, start_time):
    """Simulated‑annealing / gradient descent hybrid."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_metric(best_h, dx)
    cur_h = h.copy()
    cur_c = best_c

    # schedule parameters
    T0, T_end = 0.02, 1e-7
    max_step_rand, max_step_tgt = 0.15, 0.2

    while time.time() - start_time < budget:
        # current correlation and lag of the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full") * dx
        cur_c = corr.max()
        lag = int(corr.argmax()) - (n - 1)

        # temperature (log‑linear decay)
        prog = (time.time() - start_time) / budget
        T = T0 * (T_end / T0) ** prog

        # ---- targeted gradient move (probability 0.6) ----
        if random.random() < 0.6:
            g = np.zeros_like(cur_h)
            if lag >= 0:
                g[:n - lag] += 1.0 - cur_h[lag:]
                g[lag:]   -= cur_h[:n - lag]
            else:
                L = -lag
                g[L:]    += 1.0 - cur_h[:n - L]
                g[:n - L] -= cur_h[L:]

            i_up = np.argmax(g)      # positive gradient → increase raises C5
            i_dn = np.argmin(g)      # negative gradient → increase lowers C5
            if g[i_up] > 1e-9 and g[i_dn] < -1e-9:
                dec = cur_h[i_up]
                inc = 1.0 - cur_h[i_dn]
                delta_max = min(dec, inc)
                if delta_max > 1e-12:
                    delta = min(max_step_tgt, delta_max) * random.random()
                    nh = cur_h.copy()
                    nh[i_up] -= delta
                    nh[i_dn] += delta
                    new_c = _c5_metric(nh, dx)
                    dC = new_c - cur_c
                    if dC < 0 or random.random() < math.exp(-dC / T):
                        cur_h, cur_c = nh, new_c
                        if new_c < best_c:
                            best_c, best_h = new_c, nh.copy()
                        continue

        # ---- random pair move ----
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        lo = max(-hi, hj - 1.0)
        hi_cap = min(1.0 - hi, hj)
        if lo >= hi_cap:
            continue
        low_cap = max(lo, -max_step_rand)
        high_cap = min(hi_cap, max_step_rand)
        delta = random.uniform(low_cap, high_cap) if low_cap < high_cap else random.uniform(lo, hi_cap)
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_c = _c5_metric(cur_h, dx)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, cur_h.copy()
        else:
            cur_h[i], cur_h[j] = hi, hj

    return best_h, best_c
