```
def project_onto_capped_simplex(v, target):
    # v: np array, shape (n,)
    # returns projection onto {x: 0<=x<=1, sum x = target}
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        theta = (lo + hi) / 2.0
        x = np.clip(v - theta, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = theta
        else:
            hi = theta
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
```

import time, math, numpy as np

def _max_overlap(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _project_capped_simplex(v, target):
    # Project onto {0<=x<=1, sum(x)=target}
    # Using bisection on theta
    v = np.asarray(v, dtype=np.float64)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        theta = (lo + hi) * 0.5
        x = np.clip(v - theta, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = theta
        else:
            hi = theta
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return x

def _final_hill_climb(h, cur_c5, deadline, rng):
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5 = _max_overlap(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
    return h, cur_c5

def _max_overlap(h):
    """Compute max_{k} ∫ h(x)(1-h(x+k)) dx for a discretized step function."""

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    """
    Euclidean projection of v onto the set {0 ≤ x ≤ 1, Σ x = target}.
    Uses a simple bisection on the threshold theta such that
    x_i = clip(v_i - theta, 0, 1) and Σ x_i = target.
    """

    lo, hi = -1.0, 2.0

        else:

        step = step_max * (1 - frac) + step_min * frac

        if rng.random() < p_grad:
            # ---- gradient proposal on the worst lag ----
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(corr)
            k = idx - (n - 1)                     # lag (may be negative)
            g = np.ones(n)
            if k >= 0:
                g[: n - k] -= h[k:]
                g[k:]      -= h[: n - k]
            else:
                kk = -k
                g[kk:]   -= h[: n - kk]
                g[: n - kk] -= h[kk:]
            h_prop = h - step * g
            h_prop = _project_capped_simplex(h_prop, target)
        else:
            # ---- random mass‑transfer proposal (sum preserving) ----
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move
            h_prop = h.copy()
            h_prop[i] -= delta
            h_prop[j] += delta

        c5_prop = _max_overlap(h_prop)
        delta_c5 = c5_prop - cur_c5
        if delta_c5 < -1e-12:                     # improvement
            h, cur_c5 = h_prop, c5_prop
            if c5_prop < best_c5 - 1e-12:
                best_h, best_c5 = h_prop.copy(), c5_prop
        else:
            # uphill acceptance with temperature
            prob = math.exp(-delta_c5 / (T + 1e-12))
            if rng.random() < prob:
                h, cur_c5 = h_prop, c5_prop

    # ------------------------------------------------------------------
    # 3. Focused hill‑climbing on the best solution found
    # ------------------------------------------------------------------
    final_deadline = start + budget_s - 0.1
    best_h, best_c5 = _targeted_hill_climb(best_h, best_c5, final_deadline, rng)

    # ------------------------------------------------------------------
    # 4. Final feasibility check and return
    # ------------------------------------------------------------------
    best_h = _project_capped_simplex(best_h, target)
    best_c5 = float(_max_overlap(best_h))
    return best_h, best_c5, n
