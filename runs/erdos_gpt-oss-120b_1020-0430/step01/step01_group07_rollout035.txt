    best_h = h.copy()
    best_score = compute_c5(best_h)

    cur_score = best_score

def anneal_targeted_binary(h, time_limit):
    n = len(h)
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.04
    T_end = 1e-6

    while time.time() - start < time_limit:
        progress = (time.time() - start) / time_limit
        T = T0 * (T_end / T0) ** progress

        # compute correlation
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        # decide move
        # compute candidate sets
        if k >= 0:
            i_set = np.where((cur_h[:n-k]==1) & (cur_h[k:]==0))[0]
            j_set = np.where((cur_h[:n-k]==0) & (cur_h[k:]==1))[0]
        else:
            kk = -k
            i_set = np.where((cur_h[kk:]==1) & (cur_h[:n-kk]==0))[0] + kk
            j_set = np.where((cur_h[kk:]==0) & (cur_h[:n-kk]==1))[0] + kk

        # if both sets large, choose from them else random
        if i_set.size and j_set.size and random.random() < 0.9:
            i = int(np.random.choice(i_set))
            j = int(np.random.choice(j_set))
        else:
            # random 1 and 0
            ones = np.where(cur_h==1)[0]
            zeros = np.where(cur_h==0)[0]
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))

        # swap
        cur_h[i], cur_h[j] = 0, 1
        new_score = compute_c5(cur_h)
        dC = new_score - cur_score
        accept = False
        if dC < 0:
            accept = True
        else:
            if random.random() < math.exp(-dC / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1, 0
    return best_h, best_score

import time
import math
import random
import numpy as np

def _c5_metric(h):
    """Return the C_5 value for a step function sampled by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _anneal_targeted_binary(start_h, max_seconds):
    """
    Simulated‑annealing hill‑climb for binary (0/1) vectors.
    Swaps a 1 and a 0, preferring swaps that directly reduce the current
    worst overlap lag.
    """
    h = start_h.astype(np.float64).copy()
    n = h.size
    cur_score = _c5_metric(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    T0 = 0.04
    T_end = 1e-6
    while time.time() - start < max_seconds:
        # temperature schedule
        progress = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** progress

        # compute worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)          # lag (negative allowed)

        # try a targeted swap that helps this lag
        if random.random() < 0.9:
            if k >= 0:
                i_set = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
                j_set = np.where((h[:n - k] == 0) & (h[k:] == 1))[0]
            else:
                kk = -k
                i_set = np.where((h[kk:] == 1) & (h[:n - kk] == 0))[0] + kk
                j_set = np.where((h[kk:] == 0) & (h[:n - kk] == 1))[0] + kk
            if i_set.size and j_set.size:
                i = int(np.random.choice(i_set))
                j = int(np.random.choice(j_set))
            else:
                # fallback random swap
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = int(np.random.choice(ones))
                j = int(np.random.choice(zeros))
        else:
            # pure random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))

        # perform swap
        h[i], h[j] = 0.0, 1.0
        new_score = _c5_metric(h)
        dC = new_score - cur_score
        accept = False
        if dC < 0:
            accept = True
        else:
            if random.random() < math.exp(-dC / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

    return best_h, best_score

    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # ---------- Phase 1 : coarse resolution (n = 200) ----------
    n1 = 200
    if n1 % 2:
        n1 += 1
    start_vectors = []

    # use a binaryised version of the provided construction if present

        n0 = init.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n1, endpoint=False)
        interp = np.interp(x, x0, init)
        # keep the n1/2 largest entries as ones
        idx = np.argsort(-interp)[:n1 // 2]
        h_bin = np.zeros(n1, dtype=np.float64)
        h_bin[idx] = 1.0

    # a few random binary starts
