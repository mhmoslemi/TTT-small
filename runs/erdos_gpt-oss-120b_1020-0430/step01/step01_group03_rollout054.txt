def max_subgradient_descent(h0, max_iter=200, step0=0.1):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    step = step0
    for it in range(max_iter):
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        dmax = np.argmax(corr) - (n-1)  # shift index
        # compute gradient for this shift
        # grad_i = 1 - h_{i+d} - h_{i-d}
        grad = np.ones(n)
        # handle i+d within [0,n)
        if dmax >= 0:
            # for i such that i+dmax < n: subtract h[i+dmax]
            i = np.arange(n - dmax)
            grad[i] -= h[i + dmax]
            # for i such that i - dmax >= 0: subtract h[i - dmax]
            i2 = np.arange(dmax, n)
            grad[i2] -= h[i2 - dmax]
        else:
            d = -dmax
            # i+d < n for all i < n-d
            i = np.arange(n - d)
            grad[i] -= h[i + d]
            # i - d >= 0 for i >= d
            i2 = np.arange(d, n)
            grad[i2] -= h[i2 - d]
        # gradient descent step
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        # compute new max corr
        new_corr = np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))
        old_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if new_corr < old_corr:
            h = h_new
            step = min(step * 1.05, 1.0)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

import time
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 (max overlap) for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact sum using water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    # iterative redistribution (fast because only a few passes are needed)
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size == 0:          # all at bounds – redistribute uniformly
            idx = np.arange(x.size)
        delta = diff / idx.size
        # compute allowable move for each free coordinate
        max_add = 1.0 - x[idx]
        max_sub = -x[idx]
        add = np.where(delta > 0, np.minimum(delta, max_add), np.maximum(delta, max_sub))
        x[idx] += add
        diff = target_sum - x.sum()
    return np.clip(x, 0.0, 1.0)


def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Smooth approximation of max_k  Σ h_i (1‑h_{i+k}) and its gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    obj = (m + np.log(sum_exp)) / beta
    p = exp_bc / sum_exp                     # softmax weights (length 2n‑1)

    # convolution with p and its reversed copy gives Σ_d p_d h_{i±d}
    conv1 = np.convolve(p, h, mode="full")[n - 1 : 2 * n - 1]
    conv2 = np.convolve(p[::-1], h, mode="full")[n - 1 : 2 * n - 1]
    grad = 1.0 - (conv1 + conv2)
    return obj, grad


def gradient_descent(h0: np.ndarray, beta: float, max_iter: int = 400, step0: float = 0.2):
    """Projected gradient descent on the soft‑max surrogate."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    step = step0
    for _ in range(max_iter):
        obj, grad = softmax_obj_and_grad(h, beta)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)

        new_obj, _ = softmax_obj_and_grad(h_new, beta)
        if new_obj < obj:
            h = h_new
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.5
        if step < 1e-8:
            break
    return h


def max_shift_gradient(h: np.ndarray):
    """Gradient of the worst (max) correlation shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    d = idx - (n - 1)                     # actual shift (can be negative)
    grad = np.ones(n)

    if d >= 0:
        if d < n:
            grad[: n - d] -= h[d:]        # term h_{i+d}
            grad[d:] -= h[: n - d]        # term h_{i-d}
    else:
        da = -d
        if da < n:
            grad[da:] -= h[: n - da]      # term h_{i+d}=h_{i-da}
            grad[: n - da] -= h[da:]      # term h_{i-d}=h_{i+da}
    return grad


def max_subgradient_descent(h0: np.ndarray, max_iter: int = 200, step0: float = 0.2):
    """Sub‑gradient descent on the true max‑overlap objective."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    step = step0
    for _ in range(max_iter):
        old_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
        grad = max_shift_gradient(h)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        new_corr = np.max(np.correlate(h_new, 1.0 - h_new, mode="full"))
        if new_corr < old_corr:
            h = h_new
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.5
        if step < 1e-8:
            break
    return h


def random_feasible_vector(n: int, rng: np.random.Generator):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    # start with uniform numbers, then project
    v = rng.random(n)
    target = n / 2.0
    v = project_feasible(v, target)
    return v
