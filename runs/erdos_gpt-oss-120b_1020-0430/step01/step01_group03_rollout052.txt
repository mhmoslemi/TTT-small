def hill_climb_binary(h_start, rng, max_iters=2000):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        # propose swap
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if h[i]==h[j]:
            continue
        # swap
        new_h = h.copy()
        new_h[i], new_h[j] = h[j], h[i]
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        else:
            # maybe accept some worsening? ignore for now.
            pass
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the soft‑max surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None   # sum of h must equal this (set before each SLSQP)
CURRENT_BETA = None     # smoothing parameter for soft‑max (set before each SLSQP)

# ----------------------------------------------------------------------
# Constraint and objective for SLSQP
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i h_i (1‑h_{i+k}) using log‑sum‑exp."""
    # full correlation of h with 1‑h
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# Evaluation utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """
    Generate a random feasible vector (0≤h_i≤1, Σh_i = n/2).
    Starts from the constant 0.5 vector and redistributes mass randomly.
    """
    h = np.full(n, 0.5, dtype=float)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

    """Binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    rng.shuffle(h)
    return h

# Small perturbations used by hill climbing
# ----------------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """
    Simple deterministic hill‑climb (accept only strict improvements).
    The step size shrinks periodically.
    """
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    shrink_interval = max(1, max_iters // 8)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % shrink_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

# SLSQP driver with the smooth max surrogate

def slsqp_optimize(h0, beta, max_iter=300):
    """Run a single SLSQP optimisation with the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the integral exactly (numerical drift may appear)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
