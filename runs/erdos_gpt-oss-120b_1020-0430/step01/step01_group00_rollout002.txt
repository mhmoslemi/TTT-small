import time
import numpy as np
from scipy import optimize

# -------------------- Helper functions --------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """
    Clip to [0,1] and enforce sum(h)=n/2 by water‑filling.
    """
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction to guarantee exactness
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """
    Compute C₅ for a discrete step function h on [0,2].
    """
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _solve_slsqp(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """
    Local optimisation via SLSQP on the formulation:
        minimise t  subject to  t >= correlation(h,1-h)
        and 0<=h<=1, sum(h)=n/2.
    Returns (h_opt, c5_val).
    """
    n = initial_h.size
    # ensure feasibility before start
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    constraints = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opt = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opt,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """
    Linear interpolation from a coarse grid to a finer one.
    """
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _de_optimize(n: int, time_left, seed: int) -> tuple:
    """
    Differential Evolution on the feasible set, followed by a polishing SLSQP.
    Returns (h_best, c5_best).
    """
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    # DE parameters – modest to stay within the time budget
    popsize = 7
    max_iter = 30

    # objective projects to feasibility before computing C5
    def obj(x):
        h = _project_to_feasible(x, n)
        return _c5_from_h(h)

    # run DE (may be skipped if there is insufficient time)
    if time_left() < 2.0:
        # fallback: use a random feasible start
        x0 = np.random.rand(n)
        x0 = _project_to_feasible(x0, n)
        return x0, _c5_from_h(x0)

    result = optimize.differential_evolution(
        obj,
        bounds,
        strategy="best1bin",
        maxiter=max_iter,
        popsize=popsize,
        polish=False,
        seed=seed,
        updating="deferred",
        disp=False,
        atol=0,
        tol=0,
    )
    h_de = _project_to_feasible(result.x, n)
    # Polish with SLSQP
    h_opt, c5_opt = _solve_slsqp(h_de, max_iter=200)
    return h_opt, c5_opt


# -------------------- Main entry point --------------------

    Returns (h_values, c5_bound, n_points).

    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Grab the optional parent construction
