import time, numpy as np

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _smooth_max_grad(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = corr - m
    # prevent overflow
    exp_shift = np.exp(alpha * shifted)
    sum_exp = exp_shift.sum()
    probs = exp_shift / sum_exp
    grad = np.zeros_like(h)
    # loop over shifts
    # shift index s = idx - (n-1)
    idx_offset = n - 1
    for idx, w in enumerate(probs):
        if w == 0.0:
            continue
        s = idx - idx_offset
        if s >= 0:
            grad[: n - s] += -w * h[s:]
            grad[s:] += -w * h[: n - s]
        else:
            s_abs = -s
            grad[s_abs:] += -w * h[: n - s_abs]
            grad[: n - s_abs] += -w * h[s_abs:]
    return grad, corr.max()

def _adam_optimize(h0, target, dx, max_iter, alpha_start, alpha_end, rng):
    h = h0.copy()
    n = h.shape[0]
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)
    for t in range(1, max_iter + 1):
        # schedule smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        grad, _ = _smooth_max_grad(h, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.5 / (np.sqrt(v_hat) + eps)   # learning rate scaled per coordinate
        h = h - step * m_hat
        # project back to feasible set
        h = _project_onto_capped_simplex(h, target)
        if t % 200 == 0:
            val = _compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    # final evaluation
    val = _compute_c5(h, dx)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    return best_h, best_val

def _local_refine(h, dx, rng, max_trials=2000):
    n = len(h)
    target = np.sum(h)
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)
    for _ in range(max_trials):
        i, j = rng.choice(n, size=2, replace=False)
        # possible delta limited by bounds
        max_inc = 1.0 - best_h[i]
        max_dec = best_h[j]  # decrease j
        delta = rng.rand() * min(max_inc, max_dec) * 0.5  # small step

        h_new[i] += delta
        h_new[j] -= delta
        val = _compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            best_h = h_new
    return best_h, best_val

import time
import numpy as np

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _smooth_max_grad(h: np.ndarray, alpha: float):
    """gradient of smoothed max_{shift} corr(h)"""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = corr - m
    # stable softmax
    exp_shift = np.exp(alpha * shifted)
    probs = exp_shift / exp_shift.sum()
    grad = np.zeros_like(h)
    off = n - 1
    for idx, w in enumerate(probs):
        if w == 0.0:
            continue
        s = idx - off
        if s >= 0:
            grad[: n - s] += -w * h[s:]
            grad[s:] += -w * h[: n - s]
        else:
            s = -s
            grad[s:] += -w * h[: n - s]
            grad[: n - s] += -w * h[s:]
    return grad, corr.max()


def _adam_optimize(h0: np.ndarray, target: float, dx: float,
                   max_iter: int, alpha_start: float, alpha_end: float,
                   rng: np.random.RandomState):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)

    for t in range(1, max_iter + 1):
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        grad, _ = _smooth_max_grad(h, alpha)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.5 / (np.sqrt(v_hat) + eps)          # 0.5 is learning‑rate factor
        h = h - step * m_hat

        h = _project_onto_capped_simplex(h, target)

        if t % 200 == 0:
            val = _compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()

    val = _compute_c5(h, dx)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    return best_h, best_val


def _local_refine(h: np.ndarray, dx: float,
                  rng: np.random.RandomState, max_trials: int = 500):
    n = len(h)
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)
    for _ in range(max_trials):
        i, j = rng.choice(n, size=2, replace=False)
        inc = min(1.0 - best_h[i], best_h[j])
        if inc <= 0:
            continue
        delta = rng.rand() * inc * 0.5
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val = _compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            best_h = h_new
    return best_h, best_val


def _upscale(vec: np.ndarray, n: int) -> np.ndarray:
    if len(vec) == n:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, vec)
