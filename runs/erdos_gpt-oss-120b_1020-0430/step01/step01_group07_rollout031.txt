def _c5_metric(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _initial_vector(n):
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        # upscale to n
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)
    # enforce sum = n/2
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    delta = target - h.sum()
    if abs(delta) > 1e-12:
        for idx in np.random.permutation(n):
            if 0 <= h[idx] + delta <= 1:
                h[idx] += delta
                break
    return h

def _anneal(start_h, max_seconds):
    # same as described
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start_time = time.time()
    # schedule parameters
    T0 = 0.2   # initial temperature
    T_end = 1e-6
    max_step = 0.5

    while time.time() - start_time < max_seconds:
        elapsed = time.time() - start_time
        progress = elapsed / max_seconds
        # exponential decay
        T = T0 * (T_end / T0) ** progress

        # choose move type
        if random.random() < 0.7:
            # gradient-based move
            # compute worst shift
            # compute correlation for mismatches
            corr = np.correlate(h, 1 - h, mode='full')
            idx_max = int(np.argmax(corr))
            k = idx_max - (n - 1)   # shift index
            # compute gradient for this shift
            grad = np.zeros(n, dtype=np.float64)
            # for i where i+k in range
            if k >= 0:
                grad[:n - k] += (1 - h[k:])   # (1 - h_{i+k})
                grad[k:] -= h[:n - k]        # ( - h_{i - k})? Wait derive correctly

                # For i >= k: i-k >=0 => subtract h_{i - k}

            else:
                # k negative
                kp = -k
                # For i+ k >=0 same
                grad[kp:] += (1 - h[:n - kp])  # i from kp..n-1 => i+k = i - kp (>=0)

                best_c = new_c
                best_h = h.copy()

            h[i] = hi
            h[j] = hj

    return best_h, best_c

import time
import math
import random
import numpy as np

# -------------------------------------------------------------
# core metric
def _c5_metric(h):
    """C5 value for a sampled step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


# -------------------------------------------------------------
# initial construction (interpolates if a parent is present)
def _initial_vector(n):
    """Return a feasible vector of length n (sum = n/2, 0≤h≤1)."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


# -------------------------------------------------------------
# gradient for a given shift (the shift that currently gives max overlap)
def _grad_for_shift(h, k):
    """Gradient of the current worst C5 term with respect to h."""
    n = h.size
    grad = np.zeros(n, dtype=np.float64)
    if k >= 0:
        idx = np.arange(0, n - k)
        grad[idx] += (1.0 - h[idx + k])
        grad[idx + k] -= h[idx]
    else:
        kp = -k
        idx = np.arange(kp, n)
        grad[idx] += (1.0 - h[idx - kp])
        grad[idx - kp] -= h[idx]
    return grad


# -------------------------------------------------------------
def _anneal(start_h, max_seconds):
    """Simulated‑annealing on the continuous box with sum constraint."""
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0

    # initial correlation, max value and the shift that attains it
    cur_corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = cur_corr.max()
    cur_c = cur_max * dx
    cur_k = int(np.argmax(cur_corr)) - (n - 1)

    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    T0 = 0.12                # start temperature
    T_end = 1e-6
    max_step = 0.45          # max absolute delta per move
    prob_grad = 0.55         # fraction of gradient‑guided moves

    while time.time() - start < max_seconds:
        # temperature (exponential decay)
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        T = T0 * (T_end / T0) ** prog

        # ----- choose a pair -------------------------------------------------
        if random.random() < prob_grad:
            # use gradient of the worst shift
            grad = _grad_for_shift(h, cur_k)
            pos = np.where(grad > 1e-9)[0]
            neg = np.where(grad < -1e-9)[0]
            if pos.size and neg.size:
                i = int(np.random.choice(pos))
                j = int(np.random.choice(neg))
            else:
                i, j = random.sample(range(n), 2)
        else:
            i, j = random.sample(range(n), 2)

        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)          # feasible interval for delta
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # restrict to a modest step most of the time
        cap_low = max(low, -max_step)
        cap_high = min(high, max_step)
        if cap_low < cap_high:
            delta = random.uniform(cap_low, cap_high)
        else:
            delta = random.uniform(low, high)

        # ----- apply swap ----------------------------------------------------
        h[i] = hi + delta
        h[j] = hj - delta

        # ----- evaluate -------------------------------------------------------
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = new_corr.max()
        new_c = new_max * dx
        new_k = int(np.argmax(new_corr)) - (n - 1)

        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / (T if T > 0 else 1e-12)):
            # accept
            cur_corr = new_corr
            cur_max = new_max
            cur_c = new_c
            cur_k = new_k
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # reject – restore
            h[i] = hi
            h[j] = hj

    return best_h, best_c


# -------------------------------------------------------------
def _refine(h, max_seconds):
    """Very small deterministic hill‑climbing passes."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    start = time.time()
    step = 0.07

    while time.time() - start < max_seconds:
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * step
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_c = _c5_metric(cur_h)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = hi
            cur_h[j] = hj

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C₅ bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # --- choose a reasonable resolution ---------------------------------------
    n = 500                     # even length, gives finer control than 100
    if n % 2:
        n += 1

    # --- initial feasible vector (interpolated parent if any) -----------------
    h0 = _initial_vector(n)

    # --- primary simulated annealing (most of the budget) --------------------
    primary_time = total_budget * 0.78
    best_h, best_c = _anneal(h0, primary_time)

    # --- remaining time: a few random restarts --------------------------------
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    while remaining > 6.0:                     # need a few seconds per run
        # binary start with exactly n/2 ones
        target = n // 2
        cand = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        cand[perm[:target]] = 1.0
        slice_time = min(remaining * 0.32, 28.0)
        cand_h, cand_c = _anneal(cand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        elapsed = time.time() - start_all
        remaining = total_budget - elapsed

    # --- final deterministic refinement ---------------------------------------
    refine_time = max(0.0, total_budget - (time.time() - start_all))
    best_h, best_c = _refine(best_h, refine_time)

    target_sum = n / 2.0

        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    try:
