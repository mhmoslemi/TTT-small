def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, cvxpy as cp, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # Determine n
    n = 70
    target_sum = n // 2 # integer

    # Try MILP
    best_h = None
    best_c5 = float('inf')

    try:
        # Build pairs
        pair_i = []
        pair_j = []
        shift_idx = [[] for _ in range(n)]  # shift_idx[s] list of pair indices
        for i in range(n):
            for j in range(i+1, n):
                shift = j - i
                pair_i.append(i)
                pair_j.append(j)
                shift_idx[shift].append(len(pair_i)-1)
        num_pairs = len(pair_i)
        # Variables
        v = cp.Variable(n, boolean=True)
        w = cp.Variable(num_pairs, boolean=True)
        M = cp.Variable()
        constraints = []
        # sum
        constraints.append(cp.sum(v) == target_sum)
        # symmetry breaking
        constraints.append(v[0] == 1)
        # w constraints
        for k in range(num_pairs):
            i = pair_i[k]
            j = pair_j[k]
            constraints.append(w[k] <= v[i])
            constraints.append(w[k] <= v[j])
            constraints.append(w[k] >= v[i] + v[j] - 1)
        # shift constraints
        for s in range(1, n):
            idxs = shift_idx[s]
            # product sum
            prod_sum = cp.sum(w[idxs]) if idxs else 0
            pre = cp.sum(v[:n-s])
            suffix = cp.sum(v[s:])
            constraints.append(pre - prod_sum <= M)
            constraints.append(suffix - prod_sum <= M)
        constraints.append(M >= 0)
        prob = cp.Problem(cp.Minimize(M), constraints)
        # Solve with CBC if available else ECOS_BB
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            prob.solve(solver=cp.ECOS_BB, warm_start=True, verbose=False)
        # Check status
        if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            v_val = np.rint(v.value).astype(float)  # round
            # Ensure exactly target sum (maybe rounding error)
            # If not, adjust: simple projection
            cur_sum = v_val.sum()
            if cur_sum != target_sum:
                # fix by toggling some bits

        v = cp.Variable(n, boolean=True)           # h_i ∈ {0,1}

        cons.append(cp.sum(v) == target_sum)
