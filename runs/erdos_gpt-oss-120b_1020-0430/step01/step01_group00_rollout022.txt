import time
import math
import numpy as np
from scipy import optimize as opt

# ---------- helper utilities ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given (possibly continuous) h."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- continuous refinement (SLSQP) ----------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """SLSQP: minimise maximal overlap."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = opt.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                       constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- smooth‑max optimisation (L‑BFGS‑B) ----------
def _smooth_optimize(start_h: np.ndarray, max_iter: int = 250,
                     alpha: float = 8.0, penalty: float = 500.0) -> tuple:
    n = start_h.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mx = np.max(corr)
        shifted = corr - mx
        exps = np.exp(alpha * shifted)
        smooth = mx + np.log(np.sum(exps)) / alpha
        pen = penalty * (np.sum(x) - target) ** 2
        return smooth + pen

    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(obj, start_h, method="L-BFGS-B", bounds=bounds,
                       options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    h = _project_to_feasible(res.x, n)
    return h, _c5_from_h(h)


# ---------- binary hill‑climbing (simulated annealing) ----------
def _binary_sa(n: int, iters: int, rng: np.random.Generator,
               target_raw: float) -> tuple:
    # start from random balanced binary vector
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1

    cur_score = np.max(np.correlate(h, 1 - h, mode="full"))
    best_h, best_score = h.copy(), cur_score
    # initial temperature based on distance to target
    T = max(1.0, cur_score - target_raw)
    for _ in range(iters):
        if T < 1e-12:
            break
        # pick a 1 and a 0
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = np.max(np.correlate(h, 1 - h, mode="full"))
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_raw:
                    break
        else:
            h[i] = 1
            h[j] = 0
        T *= 0.9995
    return best_h, best_score
