        ones_idx = np.where(cur == 1)[0]

                best = cur.copy()

def continuous_hill_climb(h, rng, max_iters=2000, step=0.2):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()

def binary_anneal_time(h, rng, max_iters, init_temp, time_limit):
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_val = compute_max_corr(best_h)
    cur_h = h.copy()
    cur_val = best_val
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t = init_temp * (1 - it / max_iters) + 1e-6
        # pick indices
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = compute_max_corr(cur_h)
        delta = new_val - cur_val
        if delta <= 0 or rng.random() < np.exp(-delta / t):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_val

    n = h_start.size

    cur_val = compute_max_corr(cur)
    best = cur.copy()
    best_val = cur_val
    for it in range(max_iters):
        if time.time() > time_limit:

        # temperature linear decay

        cur[j] = 1.0

                best = cur.copy()
        else:

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_max_corr(h):
    """Maximum overlap count (before scaling by dx)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr)


def binary_random(n, rng):
    """Random 0/1 vector with exactly n/2 ones."""
    half = n // 2

    h[idx] = 1.0
    return h

    """Turn any feasible h into a binary vector with the same total mass."""
    half = n // 2
    idx = np.argpartition(h, -half)[-half:]

    return bh

    """Simulated‑annealing on binary vectors, stops when time_limit (abs) is hit."""
    n = h_start.size
    cur = h_start.copy()
    cur_val = compute_max_corr(cur)
    best = cur.copy()
    best_val = cur_val
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        # linear temperature decay

                trial = slsqp_optimize(best_h, beta, rng, max_iter=300)
                c_trial = compute_c5(trial)
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = trial.copy()
            except Exception:
                pass
            if time.time() - start > budget_s * 0.97:
                break

    # ------------------------------------------------------------------
    # Final feasibility clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
