import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

# ----------------------------------------------------------------------
# Random feasible vector (sum = target, bounds [0,1])
# ----------------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

# ----------------------------------------------------------------------
# Fast full correlation using FFT (equivalent to np.correlate(h,1-h,'full'))
# ----------------------------------------------------------------------
def _full_corr_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    size = 2 * n - 1
    # correlation a⋆b = conv(a_rev, b)
    ha = np.fft.rfft(h[::-1], n=size)
    hb = np.fft.rfft(1.0 - h, n=size)
    conv = np.fft.irfft(ha * hb, n=size)
    return conv

# ----------------------------------------------------------------------
# C5 value for a given step vector
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = _full_corr_fft(h)
    return float(corr.max() * dx)

# ----------------------------------------------------------------------
# Gradient of a single shift (the shift that attains the max)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

# ----------------------------------------------------------------------
# Smooth approximation of max (used by the optional SLSQP refinement)
# ----------------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 600) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = _full_corr_fft(v)
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
