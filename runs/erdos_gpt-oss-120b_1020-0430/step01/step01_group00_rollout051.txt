def refine_h(h, time_left_func, max_iters=10000, step0=0.5):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    it = 0
    while it < max_iters and time_left_func() > 0.5:
        # compute c5 and kmax
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        c5 = corr[max_idx] * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # get shift
        k = max_idx - (n - 1)
        # gradient
        # shift plus/minus
        if k >= 0:
            shift_plus = np.empty_like(h)
            shift_plus[:n - k] = h[k:]
            shift_plus[n - k:] = 0.0
            shift_minus = np.empty_like(h)
            shift_minus[:k] = 0.0
            shift_minus[k:] = h[:n - k]
        else:
            kk = -k
            shift_plus = np.empty_like(h)
            shift_plus[:kk] = 0.0
            shift_plus[kk:] = h[:n - kk]
            shift_minus = np.empty_like(h)
            shift_minus[:n - kk] = h[kk:]
            shift_minus[n - kk:] = 0.0
        g = (1.0 - shift_plus) - shift_minus
        # identify pos and neg
        pos_idx = np.where(g > 1e-12)[0]
        neg_idx = np.where(g < -1e-12)[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            # fallback random transfer
            i = np.random.randint(n)
            j = np.random.randint(n)
        else:
            i = np.random.choice(pos_idx)
            j = np.random.choice(neg_idx)
        # compute max delta respecting bounds
        max_delta = min(h[i], 1.0 - h[j])
        # step size schedule: step = step0 * (1 - it / max_iters)
        delta = min(step0 * (1 - it / max_iters), max_delta)
        if delta <= 1e-12:
            # try random small delta
            delta = 1e-4 * max_delta
        # perform transfer
        h[i] -= delta
        h[j] += delta
        it += 1
    # after loop compute final c5
    final_c5 = _c5_from_h(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# -------------------- helper utilities --------------------

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # simple proportional correction loop
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not free.any():
            break
        delta = diff / free.sum()
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 = (2/n) * max_k correlation of h with (1-h)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _gradient_transfer(h: np.ndarray, k: int, step: float) -> np.ndarray:
    """One projected gradient‑like transfer for the shift k that currently dominates."""
    n = h.shape[0]
    # compute shifted arrays with zero padding
    if k >= 0:
        shift_plus = np.empty_like(h)
        shift_plus[: n - k] = h[k:]
        shift_plus[n - k :] = 0.0
        shift_minus = np.empty_like(h)
        shift_minus[:k] = 0.0
        shift_minus[k:] = h[: n - k]
    else:
        kk = -k
        shift_plus = np.empty_like(h)
        shift_plus[:kk] = 0.0
        shift_plus[kk:] = h[: n - kk]
        shift_minus = np.empty_like(h)
        shift_minus[: n - kk] = h[kk:]
        shift_minus[n - kk :] = 0.0
    g = (1.0 - shift_plus) - shift_minus

    pos = np.where(g > 1e-12)[0]
    neg = np.where(g < -1e-12)[0]

    if pos.size == 0 or neg.size == 0:
        # fallback: random pair
        i = np.random.randint(n)
        j = np.random.randint(n)
    else:
        i = np.random.choice(pos)
        j = np.random.choice(neg)

    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return h
    delta = min(step, max_delta)
    h[i] -= delta
    h[j] += delta
    return h


def _refine_h(
    h0: np.ndarray,
    max_iters: int,
    time_left,
    target: float = 0.3808,
) -> tuple:
    """Projected gradient‑like local search."""
    h = h0.copy()
    n = h.size
    best_h, best_c5 = h.copy(), _c5_from_h(h)

    step0 = 0.4  # initial transfer size
    for it in range(max_iters):
        if time_left() < 0.5:
            break

        # correlation and dominant shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        c5 = corr[idx] * (2.0 / n)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
            if best_c5 < target:
                break

        k = idx - (n - 1)
        step = step0 * (1.0 - it / max_iters)
        h = _gradient_transfer(h, k, step)

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            sigma = 0.02
            h = np.clip(h + sigma * np.random.randn(n), 0.0, 1.0)
            h = _project_to_feasible(h, n)

    final_c5 = _c5_from_h(h)
    if final_c5 < best_c5:
        best_c5, best_h = final_c5, h.copy()
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1 that minimises
    C5 = max_k ∫ h(x) (1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # try to obtain a parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
        base_n = init_h.size
    except NameError:
        init_h = None
        base_n = 128

    # candidate grid sizes – grow modestly, never exceed 500 points
    n_candidates = [base_n]
    for mul in (1.5, 2, 2.5, 3):
        n = int(round(base_n * mul))
        if n <= 500:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    target_bound = 0.3808
    best_h, best_c5, best_n = None, float("inf"), None

    for n in n_candidates:
        if time_left() < 5.0:
            break

        # build starting vector for this resolution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = np.full(n, 0.5)

        base_h = _project_to_feasible(base_h, n)

        # number of local‑search restarts (adapt to remaining time)
        restarts = 4
        max_iters = 4000

        for r in range(restarts):
            if time_left() < 2.0:
                break
            if r == 0:
                start_h = base_h.copy()
            else:
                # noisy perturbation of the base
                sigma = 0.15
                start_h = base_h + sigma * np.random.randn(n)
                start_h = np.clip(start_h, 0.0, 1.0)
                start_h = _project_to_feasible(start_h, n)

            h_opt, c5_val = _refine_h(start_h, max_iters, time_left, target=target_bound)

            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_val, n
                if best_c5 < target_bound:
                    break

        # optional short polish with SLSQP (few iterations) if time permits
        if best_h is not None and time_left() > 4.0:
            try:
                # reuse the previous _solve_h logic (lightweight version)
                n_opt = best_h.size
                # initial t is the current max correlation
                corr0 = np.correlate(best_h, 1.0 - best_h, mode="full")
                t0 = np.max(corr0)
                x0 = np.concatenate([best_h, [t0]])

                bounds = [(0.0, 1.0)] * n_opt + [(0.0, None)]

                def ineq(x):
                    h = x[:n_opt]
                    t = x[-1]
                    return t - np.correlate(h, 1.0 - h, mode="full")

                def eq(x):
                    return np.sum(x[:n_opt]) - (n_opt / 2.0)

                cons = [{"type": "ineq", "fun": ineq},
                        {"type": "eq", "fun": eq}]

                opt_res = optimize.minimize(
                    lambda x: x[-1],
                    x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 80, "ftol": 1e-9, "disp": False},
                )
                if opt_res.success:
                    h_polished = _project_to_feasible(opt_res.x[:n_opt], n_opt)
                    c5_polished = _c5_from_h(h_polished)
                    if c5_polished < best_c5:
                        best_h, best_c5, best_n = h_polished, c5_polished, n_opt
            except Exception:
                pass

        if best_c5 < target_bound:
            break

    # final safety: if nothing found, fall back to uniform 0.5
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    # final projection to guarantee feasibility before returning
    best_h = _project_to_feasible(np.array(best_h, dtype=float), best_n)

    return best_h.tolist(), float(best_c5), int(best_n)
