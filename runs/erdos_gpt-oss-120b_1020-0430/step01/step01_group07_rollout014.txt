def c5_metric(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def gradient_descent_opt(h, target, time_budget):
    n = h.size
    dx = 2.0 / n
    start = time.time()
    best_h = h.copy()
    best_c = c5_metric(h)
    # hyperparameters
    # schedule for beta and eta
    while time.time() - start < time_budget:
        # compute correlation
        cross = np.correlate(h, h, mode='full')
        # overlap for each lag
        overlap = target - cross
        # compute c5 current
        cur_c = overlap.max() * dx
        if cur_c < best_c - 1e-9:
            best_c = cur_c
            best_h = h.copy()
        # softmax weights
        # schedule beta: start low, increase
        elapsed = time.time() - start
        beta = 5.0 + 20.0 * (elapsed / time_budget)  # from 5 to 25
        # compute stable softmax
        w_raw = np.exp(beta * overlap - np.max(beta * overlap))
        w = w_raw / w_raw.sum()
        # compute M_i = sum_k w_k * (h_{i+k} + h_{i-k})
        M = np.zeros_like(h)
        n = h.size
        offset = n - 1
        # iterate over all lags
        # using slicing
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            L = idx - offset
            if L >= 0:
                # i from 0 to n-L-1: i+L within range
                M[:n-L] += wk * h[L:]
                M[L:]   += wk * h[:n-L]
            else:
                L2 = -L
                M[:n-L2] += wk * h[L2:]
                M[L2:]   += wk * h[:n-L2]
        # gradient of smooth max (objective) w.r.t h
        grad = 1.0 - M
        # learning rate schedule
        eta = 0.2 * (1 - elapsed / time_budget) + 0.02
        h_new = h - eta * grad
        h = project_bounds_sum(h_new, target)
    return best_h, best_c

def binary_hill_climb(h, target, time_budget):
    n = h.size
    # convert to binary version:
    # select top target entries
    k = int(round(target))
    idx_sorted = np.argpartition(h, -k)[-k:]   # top k indices
    h_bin = np.zeros_like(h)
    h_bin[idx_sorted] = 1.0
    # if target not integer (rare), assign fractional to next index
    # but we assume integer
    best_h = h_bin.copy()
    best_c = c5_metric(best_h)
    start = time.time()
    # precompute indices of ones and zeros
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]
    while time.time() - start < time_budget:
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        best_h[i] = 0.0
        best_h[j] = 1.0
        cur_c = c5_metric(best_h)
        if cur_c + 1e-12 < best_c:
            best_c = cur_c
            # update ones/zeros arrays
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
        else:
            # revert
            best_h[i] = 1.0
            best_h[j] = 0.0
        # optionally random restart if no improvement for many tries
    return best_h, best_c

import time, random, math
import numpy as np

def _c5_metric(h):
    """Exact C5 evaluation for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(v, target, max_iter=10):
    """Clamp to [0,1] and enforce sum(v)=target (Euclidean projection)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    eps = 1e-12
    if abs(cur - target) <= eps:
        return v
    for _ in range(max_iter):
        diff = target - cur
        free = np.where((v > eps) & (v < 1.0 - eps))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        np.clip(v, 0.0, 1.0, out=v)
        cur = v.sum()
        if abs(cur - target) <= eps:
            break
    if abs(v.sum() - target) > eps:
        diff = target - v.sum()
        cand = np.where((v + diff >= 0.0) & (v + diff <= 1.0))[0]
        if cand.size:
            v[np.random.choice(cand)] += diff
    return v

def _gradient_descent(h, target, budget):
    """Soft‑max gradient descent on the smooth max‑overlap objective."""
    n = h.size
    dx = 2.0 / n
    start = time.time()
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    offset = n - 1

    while time.time() - start < budget:
        # correlation of h with itself
        cross = np.correlate(h, h, mode="full")
        overlap = target - cross               # unscaled overlap for each lag
        cur_c = overlap.max() * dx
        if cur_c < best_c - 1e-12:
            best_c, best_h = cur_c, h.copy()

        # soft‑max over the lags (β grows linearly)
        elapsed = time.time() - start
        beta = 5.0 + 20.0 * (elapsed / budget)       # 5 → 25
        # stable soft‑max
        w_raw = np.exp(beta * overlap - np.max(beta * overlap))
        w = w_raw / w_raw.sum()

        # compute M_i = Σ_k w_k (h_{i+k}+h_{i‑k})
        M = np.zeros_like(h)
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            L = idx - offset
            if L >= 0:
                # i+L term
                M[: n - L] += wk * h[L:]
                # i‑L term
                M[L:] += wk * h[: n - L]
            else:
                L2 = -L
                M[: n - L2] += wk * h[L2:]
                M[L2:] += wk * h[: n - L2]

        # gradient of the smooth max (see analysis)
        grad = 1.0 - M

        # learning‑rate schedule
        eta = 0.2 * (1.0 - elapsed / budget) + 0.02
        h_new = h - eta * grad
        h = _project_bounds_sum(h_new, target)

    return best_h, best_c

def _binary_hill_climb(h, target, budget):
    """Greedy hill‑climbing on a binary vector (exactly target ones)."""
    n = h.size
    k = int(round(target))
    # turn the current real‑valued vector into a binary one
    idx = np.argpartition(h, -k)[-k:]
    h_bin = np.zeros_like(h)
    h_bin[idx] = 1.0

    best_h = h_bin.copy()
    best_c = _c5_metric(best_h)
    start = time.time()
    while time.time() - start < budget:
        # pick a random 1 and a random 0
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        best_h[i] = 0.0
        best_h[j] = 1.0
        cur_c = _c5_metric(best_h)
        if cur_c + 1e-12 < best_c:
            best_c = cur_c
        else:
            # revert the swap
            best_h[i] = 1.0
            best_h[j] = 0.0
    return best_h, best_c
