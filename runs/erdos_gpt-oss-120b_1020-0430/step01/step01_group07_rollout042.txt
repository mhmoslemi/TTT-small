def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math
    random.seed(seed)
    np.random.seed(seed)
    # Set n (samples)
    n = 400  # maybe adjust depending on budget
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size

        x = np.linspace(0, 2, n, endpoint=False)

    else:
        h = np.full(n, 0.5, dtype=np.float64)
    # enforce sum to n/2

    np.clip(h, 0.0, 1.0, out=h)

                h[idx] += residual
                break
    best_h = h.copy()
    best_c = _c5_metric(best_h)

    start = time.time()

    T0 = 0.1

    max_step = 0.2

    it = 0
    while time.time() - start < total:

        T = T0 * (T_end / T0) ** progress
        # pick indices i,j
        i, j = random.sample(range(n), 2)

        high = min(1.0 - hi, hj)

            high_cap = min(high, max_step)

                best_h = cur_h.copy()

    for _ in range(100):
        # shift v by lambda
        x = np.clip(v - lo, 0.0, 1.0)
        if x.sum() < target:
            lo -= 1.0
        else:
            break
    for _ in range(100):
        x = np.clip(v - hi, 0.0, 1.0)
        if x.sum() > target:
            hi += 1.0
        else:
            break
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return x

    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # term (1 - h[i+k]) for i = 0..n-1-k
        g[:n - k] += 1.0 - h[k:]
        # term -h[i - k] for i = k..n-1
        g[k:] -= h[:n - k]
