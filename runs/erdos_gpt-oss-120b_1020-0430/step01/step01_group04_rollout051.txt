import time

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log-sum-exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(conv)
    e = np.exp(alpha * (conv - maxc))
    return float((maxc + np.log(e.sum() / len(conv))) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape (min(x,2-x)) sampled on n points and normalised."""
    dx = 2.0 / n
    x = np.linspace(0.0, 2.0 - dx, n) + dx / 2.0
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    np.clip(tri, 0.0, 1.0, out=tri)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        np.clip(tri, 0.0, 1.0, out=tri)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1] and exact sum)."""
    a = rng.random(n)
    target = n / 2.0
    a *= target / a.sum()
    np.clip(a, 0.0, 1.0, out=a)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        np.clip(a, 0.0, 1.0, out=a)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a longer grid, then re‑normalise."""
    old_n = arr.shape[0]
    old_dx = 2.0 / old_n
    new_dx = 2.0 / new_n
    old_x = np.linspace(0.0, 2.0 - old_dx, old_n) + old_dx / 2.0
    new_x = np.linspace(0.0, 2.0 - new_dx, new_n) + new_dx / 2.0
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up *= target / up.sum()
    np.clip(up, 0.0, 1.0, out=up)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        np.clip(up, 0.0, 1.0, out=up)
    return up


def project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum(x)=target} using a simple bisection."""
    lo = -np.max(v) - 1.0     # guarantees all entries become 1 after shift
    hi = np.max(v) + 1.0      # guarantees all entries become 0 after shift
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def gradient_from_weights(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute ∇_h  (log‑sum‑exp) using the softmax weights w."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)      # base 1 from each f_k
    # w length = 2n-1, indexed by shift k = idx-(n-1)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # term1: subtract wk * h_{j+k}
            if k < n:
                g[: n - k] -= wk * h[k:]
                # term2: subtract wk * h_{j-k}
                g[k:] -= wk * h[: n - k]
        else:
            kk = -k
            # term1: subtract wk * h_{j-kk}
            g[kk:] -= wk * h[: n - kk]
            # term2: subtract wk * h_{j+kk}
            g[: n - kk] -= wk * h[kk:]
    return g


def gradient_descent_opt(start_h: np.ndarray, target: float,
                         deadline: float, rng: np.random.Generator):
    """Simple projected gradient descent on the smooth max objective."""
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    max_iters = 600
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = np.correlate(h, 1.0 - h, mode="full")
        # linearly increase smoothing parameter
        alpha = 2.0 + 78.0 * it / max_iters   # ends near 80
        # softmax weights (stable)
        maxc = np.max(conv)
        e = np.exp(alpha * (conv - maxc))
        w = e / e.sum()
        # gradient
        g = gradient_from_weights(h, w)
        # step size schedule (decay)
        lr = 0.3 * (1.0 - it / max_iters) + 0.02
        v = h - lr * g
        h = project_to_sum_and_bounds(v, target)

        if it % 30 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if best_c5 < 0.3808:
                # good enough – stop early
                break
            if no_improve >= 150:
                break
    return best_h, best_c5


def final_refine(h: np.ndarray, n: int, deadline: float):
    """One quick SLSQP refinement using the sharp softmax."""
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: softmax_c5(x, alpha=80.0),
                       h, method="SLSQP", bounds=bounds,
                       constraints=cons,
                       options={"maxiter": 500, "ftol": 1e-12, "disp": False})
        if res.success and time.time() < deadline:
            h2 = np.clip(res.x, 0.0, 1.0)
            # enforce exact sum (tiny correction)
            diff = target - h2.sum()
            if abs(diff) > 1e-12:
                h2 += diff / n
                np.clip(h2, 0.0, 1.0, out=h2)
            return h2
    except Exception:
        pass
    return h

# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95   # safety margin

    # candidate discretisations
    candidates = [80, 100, 120, 150, 180, 200, 240]

    # parent construction, if supplied
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidates:
            candidates.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in candidates:
        if time.time() > deadline:
            break
        target = n / 2.0

        # build a small collection of diverse starts
        starts = []

        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        starts.append(generate_triangular(n))

        for _ in range(4):
            starts.append(random_start(n, rng))

        tri = generate_triangular(n)
        for _ in range(3):
            pert = tri + rng.uniform(-0.15, 0.15, size=n)
            np.clip(pert, 0.0, 1.0, out=pert)
            diff = target - pert.sum()
            if abs(diff) > 1e-12:
                pert += diff / n
                np.clip(pert, 0.0, 1.0, out=pert)
            starts.append(pert)

        # optimise each start
        for s in starts:
            if time.time() > deadline:
                break
            h_opt, c5_opt = gradient_descent_opt(s, target, deadline, rng)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_opt, n

    # final SLSQP polish
    if best_h is not None and time.time() < deadline:
        refined = final_refine(best_h, best_n, deadline)
        refined_c5 = compute_c5(refined)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined, refined_c5

    # safety fallback
    if best_h is None:
        n = 80
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final projection to guarantee constraints
    target_sum = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = project_to_sum_and_bounds(best_h, target_sum)

    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), best_n
