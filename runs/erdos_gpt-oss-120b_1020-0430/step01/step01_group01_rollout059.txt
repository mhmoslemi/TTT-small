def optimize_h(h0, target_sum, dx, max_iter=2000, alpha=30.0, step0=1.0, time_limit=..., start_time=...):
    h = h0.copy()
    step = step0
    best_h = h.copy()
    best_val = compute_c5(h, dx)

    for it in range(max_iter):
        # compute corr and smooth_max
        corr_raw = np.convolve(h, 1 - h, mode='full')
        corr = corr_raw * dx
        max_corr = corr.max()
        shifted = corr - max_corr
        exp_alpha = np.exp(alpha * shifted)
        sum_exp = exp_alpha.sum()
        smooth_val = max_corr + np.log(sum_exp) / alpha

        # compute weights
        weights = exp_alpha / sum_exp  # shape (2n-1,)

        # compute gradient
        L = 2 * len(h) - 1
        # conv_a
        conv_a = np.convolve(weights, 1 - h, mode='full')
        conv_b_rev = np.convolve(weights[::-1], h, mode='full')
        grad = (conv_a[:len(h)] - conv_b_rev[L-1:L-1+len(h)]) * dx

        # project step: line search
        # propose new h
        new_h = h - step * grad
        # project onto feasible set
        new_h = project_onto_capped_simplex(new_h, target_sum)
        # evaluate smooth objective
        corr_raw_new = np.convolve(new_h, 1 - new_h, mode='full')
        corr_new = corr_raw_new * dx
        max_corr_new = corr_new.max()
        shifted_new = corr_new - max_corr_new
        exp_alpha_new = np.exp(alpha * shifted_new)
        smooth_val_new = max_corr_new + np.log(exp_alpha_new.sum()) / alpha

        if smooth_val_new <= smooth_val:
            h = new_h
            # maybe increase step slightly
            step *= 1.02
        else:
            step *= 0.5
        # occasionally evaluate actual C5 and track best
        if it % 10 == 0:
            val = compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        # time check break

    return best_h, best_val

def _smooth_gradient(h, dx, target_sum, alpha):
    # compute corr and weights
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_corr = corr.max()
    shifted = corr - max_corr
    exp_ = np.exp(alpha * shifted)  # values <=1
    sum_exp = exp_.sum()
    weights = exp_ / sum_exp

    # term1: w reversed convolve (1-h)
    a = 1 - h
    conv1 = np.convolve(weights[::-1], a, mode='full')
    # term2: w convolve h
    conv2 = np.convolve(weights, h, mode='full')
    n = len(h)
    term1 = conv1[2*n - 2 : 3*n - 2]
    term2 = conv2[n - 1 : 2*n - 1]
    grad = (term1 - term2) * dx
    smooth_val = max_corr + np.log(sum_exp) / alpha
    return grad, smooth_val, corr

import time
import numpy as np

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto {0 <= x <= 1, sum(x)=target} using bisection on the dual."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Generate a random point and project onto feasibility set."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Exact objective: max_k ∫ h(x)(1−h(x+k)) dx."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _smooth_gradient(h: np.ndarray, dx: float, alpha: float):
    """
    Gradient of the smooth‑max approximation of the objective.
    Returns (grad, smooth_val, corr_full) where corr_full is the full
    correlation array (already multiplied by dx).
    """
    n = h.shape[0]
    # full correlation (as in the objective)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = corr.max()
    shifted = corr - max_corr                      # ≤ 0
    exp_shift = np.exp(alpha * shifted)           # stable because shifted ≤ 0
    sum_exp = exp_shift.sum()
    weights = exp_shift / sum_exp                  # shape (2n‑1,)

    # term1 = Σ_s w_s (1‑h[j+s])
    a = 1.0 - h
    conv1 = np.convolve(weights[::-1], a, mode='full')
    # term2 = Σ_s w_s h[j‑s]
    conv2 = np.convolve(weights, h, mode='full')

    # slices extracting length‑n vectors
    # conv1 starts at index 2n‑2 (i.e. 2*n-2) for j=0
    term1 = conv1[2 * n - 2 : 3 * n - 2]
    # conv2 starts at index n‑1 for j=0
    term2 = conv2[n - 1 : 2 * n - 1]

    grad = (term1 - term2) * dx
    smooth_val = max_corr + np.log(sum_exp) / alpha
    return grad, smooth_val, corr


def _smooth_gd_optimize(h0: np.ndarray, target: float, dx: float,
                        alpha: float = 30.0, max_iter: int = 1500,
                        step0: float = 0.8,
                        start_time: float = None, budget_s: float = None):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    best_h = h.copy()
    best_val = _compute_c5(h, dx)

    for it in range(max_iter):
        # time guard
        if budget_s is not None and start_time is not None:
            if time.time() - start_time > budget_s * 0.92:
                break

        grad, smooth_val, _ = _smooth_gradient(h, dx, alpha)
        # trial step
        trial = h - step * grad
        trial = _project_onto_capped_simplex(trial, target)
        _, smooth_val_new, _ = _smooth_gradient(trial, dx, alpha)

        if smooth_val_new <= smooth_val:
            h = trial
            step = min(step * 1.02, 1.0)
        else:
            step *= 0.5
            if step < 1e-6:
                step = 1e-6

        if it % 10 == 0:
            cur = _compute_c5(h, dx)
            if cur < best_val:
                best_val = cur
                best_h = h.copy()

    return best_h, best_val


def _subgrad_optimize(h0: np.ndarray, target: float, dx: float,
                      max_iter: int = 500, step0: float = 0.4,
                      decay: float = 0.995):
    """Projected sub‑gradient descent using the actual max‑correlation."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    best_h = h.copy()
    best_val = _compute_c5(h, dx)

    for it in range(max_iter):
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr_raw))
        shift = idx - (n - 1)

        g = np.empty_like(h)
        if shift >= 0:
            s = shift
            g[: n - s] = 1.0 - h[s:]
            g[n - s:] = 1.0
            g[s:] += -h[: n - s]
        else:
            s = -shift
            g[s:] = 1.0 - h[: n - s]
            g[: s] = 1.0
            g[: n - s] += -h[s:]

        g *= dx
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay

        if it % 5 == 0:
            cur = _compute_c5(h, dx)
            if cur < best_val:
                best_val = cur
                best_h = h.copy()

    return best_h, best_val
