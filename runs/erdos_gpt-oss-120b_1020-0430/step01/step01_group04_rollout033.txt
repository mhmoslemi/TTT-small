def project_onto_capped_simplex(v, target):
    # v is any array
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi)/2
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2, 0.0, 1.0)
    # adjust final sum slight error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust small amount proportionally among non-bound entries
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: distribute equally
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

    # Use binary search over lam.

    # lam_high = 1.0 - np.min(v) + 1.0 (ensuring v+lam >=1 -> all one)

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    return project_onto_capped_simplex(tri, target)

    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0

def softmax_c5_val(h: np.ndarray, alpha: float) -> float:
    """Smooth approximation of max overlap."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(conv)
    w = np.exp(alpha * (conv - max_c))
    return float(max_c + np.log(w.sum()) / alpha)


def softmax_c5_grad(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the smooth surrogate."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(conv)
    w_raw = np.exp(alpha * (conv - max_c))
    p = w_raw / w_raw.sum()          # normalized weights
    grad = np.zeros_like(h)

    # contributions for each lag s = -(n-1)...(n-1)
    for s in range(-(n - 1), n):
        w_s = p[s + n - 1]
        if s >= 0:
            if n - s > 0:
                grad[: n - s] += w_s * (1.0 - h[s:])
                grad[s:] += w_s * (-h[: n - s])
        else:
            if n + s > 0:
                grad[-s:] += w_s * (1.0 - h[: n + s])
                grad[: n + s] += w_s * (-h[-s:])
    return grad

    h = start.copy()

# Simple local‑search refinement
# ------------------------------------------------------------
def local_refine(h: np.ndarray, target: float,
                 time_limit: float, rng: np.random.Generator):
    """Hill‑climbing moves that keep the sum constraint."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    while time.time() < time_limit - 0.1:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small amount δ from i to j
        max_minus = best_h[i]               # cannot go below 0
        max_plus = 1.0 - best_h[j]          # cannot exceed 1
        delta = rng.uniform(-0.2, 0.2)
        delta = np.clip(delta, -max_minus, max_plus)
        if abs(delta) < 1e-5:
            continue
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection only to clean tiny rounding errors
        cand = project_onto_capped_simplex(cand, target)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5 - 1e-7:
            best_h, best_c5 = cand, cand_c5
    return best_h, best_c5

# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95   # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (even numbers give integer target)
    candidate_ns = [150, 180, 200]

    # parent construction, if supplied by the harness
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    # -----------------------------------------------------------------
    # Loop over resolutions
    # -----------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > time_limit:
            break
        target = n / 2.0

        # ---- build a diverse set of starts ---------------------------------
        starts = []

        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        starts.append(generate_triangular(n))

        for _ in range(4):
            starts.append(random_feasible(n, rng))

        # ---- optimise each start --------------------------------------------
        for s in starts:
            if time.time() > time_limit:
                break
            s = project_onto_capped_simplex(s, target)
            h_opt, c5_opt = optimise_h(s, n, target, time_limit, rng)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_opt, n

    # -----------------------------------------------------------------
    # Final local refinement on the current best solution (if time permits)
    # -----------------------------------------------------------------
    if best_h is not None and time.time() < time_limit - 1.0:
        refined_h, refined_c5 = local_refine(best_h, best_n / 2.0,
                                            time_limit, rng)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # -----------------------------------------------------------------
    # Fallback – simple binary block if nothing succeeded
    # -----------------------------------------------------------------
    if best_h is None:
        n = 120
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_h = project_onto_capped_simplex(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final sanity check (tiny numerical drift)
    best_h = project_onto_capped_simplex(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
