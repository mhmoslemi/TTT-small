def perturb(h, rng, max_delta=0.1):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # max allowed delta: limited by bounds
    delta_max = min(max_delta, 1 - h[i], h[j])
    delta_min = -min(max_delta, h[i], 1 - h[j])
    # Choose delta uniformly between delta_min and delta_max
    delta = rng.uniform(delta_min, delta_max)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def compute_c5_fft(h):
    n = len(h)
    # Use real FFT size next_pow2 maybe
    # Compute full convolution length 2n-1: we can pad to 2n.
    m = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)  # linear convolutionlength m, not circular
    # conv length m, we need first 2n-1 entries.
    conv = conv[: 2*n - 1]
    # max
    dx = 2.0 / n
    return float(conv.max() * dx)

def project_to_box_and_sum(v, target):
    # find lambda such that sum(np.clip(v + lam, 0, 1)) = target
    # using binary search
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo+hi)/2, 0.0, 1.0)
    # adjust small rounding errors
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # Distribute diff among non-bound entries
        free = (w > 0) & (w < 1)
        cnt = free.sum()
        if cnt > 0:
            w[free] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w

def upsample(arr, new_n):
    old_n = len(arr)
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)

def compute_c5(h):

    G = np.fft.rfft(1 - h, n=m)

    conv = conv[:2*n-1]
    return float(conv.max() * dx)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 value using full linear correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def project_to_box_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while forcing sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 0.0) & (w < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            w[mask] += diff / cnt
        else:
            w += diff / w.size
        w = np.clip(w, 0.0, 1.0)
    return w


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from size len(arr) to new_n, then project."""
    old_n = arr.shape[0]
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    interp = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    return project_to_box_and_sum(interp, target)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular function h(x)=min(x,2-x) sampled on n points, normalized."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    # final exact correction
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_binary_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary feasible vector with exactly n/2 ones."""
    idx = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[idx[: n // 2]] = 1.0
    return h


def perturb(h: np.ndarray, rng: np.random.Generator, max_delta: float) -> np.ndarray:
    """Swap a small amount of mass between two random coordinates."""
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    # feasible delta respecting bounds
    delta_up = min(max_delta, 1.0 - h[i], h[j])
    delta_down = -min(max_delta, h[i], 1.0 - h[j])
    if delta_up < delta_down:  # no feasible move (should not happen)
        return h.copy()
    delta = rng.uniform(delta_down, delta_up)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h


def local_search(start_h: np.ndarray, deadline: float, rng: np.random.Generator):
    """Simulated‑annealing style hill climbing keeping sum and box constraints."""
    n = len(start_h)
    target = n / 2.0
    best_h = start_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    init_temp = 0.005
    delta0 = 0.2
    delta_min = 0.001
    max_iters = 2_000_000

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t = it / max_iters
        temp = init_temp * (1.0 - t) ** 3
        max_delta = delta0 * (1.0 - t) + delta_min

        new_h = perturb(cur_h, rng, max_delta)
        new_val = compute_c5(new_h)

        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        else:
            if temp > 0.0:
                prob = math.exp(-(new_val - cur_val) / temp)
                if rng.random() < prob:
                    cur_h, cur_val = new_h, new_val

        # occasional random reset from best to diversify
        if it % 10000 == 0 and rng.random() < 0.01:
            cur_h, cur_val = best_h.copy(), best_val

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    # Target resolution – 200 points gives a good trade‑off
    n = 200
    target_sum = n / 2.0

    # Build a pool of initial candidates
    starts = []

    # 1. Parent (if supplied)
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n != n:
            starts.append(upsample(parent_h, n))
        else:
            starts.append(parent_h.copy())
    except NameError:
        parent_h = None

    # 2. Triangular shape
    starts.append(generate_triangular(n))

    # 3. Random binary patterns
    for _ in range(3):
        starts.append(random_binary_start(n, rng))

    # 4. Random continuous feasible vectors
    for _ in range(2):
        v = rng.random(n)
        v = v * (target_sum / v.sum())
        v = project_to_box_and_sum(v, target_sum)
        starts.append(v)

    best_h = None
    best_val = np.inf
    best_n = n

    # Allocate time roughly equally among start points
    remaining_starts = len(starts)
    for s in starts:
        if time.time() > time_limit:
            break
        remaining_time = time_limit - time.time()
        # reserve a tiny margin for the final adjustment
        per_deadline = time.time() + remaining_time * 0.9 / remaining_starts
        h_opt, val_opt = local_search(s, per_deadline, rng)
        if h_opt is not None and val_opt < best_val:
            best_h, best_val = h_opt, val_opt
            best_n = n
        remaining_starts -= 1

    # Fallback in the unlikely event that nothing succeeded
    if best_h is None:
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_val = compute_c5(best_h)

    # Final projection onto the feasible set (eliminate tiny drifts)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 0.0) & (best_h < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            best_h[mask] += diff / cnt
        else:
            best_h += diff / best_h.size
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
