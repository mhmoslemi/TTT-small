    # water-filling
    lower = -np.max(v) - 1.0
    upper = np.max(v) + 1.0
    for _ in range(max_iter):

        s = proj.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lower = lam
        else:
            upper = lam
    return proj

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0/old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
    up = np.interp(new_x, old_x, arr)
    # renormalize sum
    target = new_n / 2.0
    up = up * (target / up.sum())
    # clip and repair
    up = np.clip(up, 0.0, 1.0)
    # adjust sum exactly using projection.
    up = project_onto_capped_simplex(up, target)
    return up

def subgradient_descent(h0, target, deadline, max_iter=5000, rng=None):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # correlation & max
        conv = np.correlate(h, 1.0 - h, mode='full')
        max_idx = conv.argmax()
        max_val = conv[max_idx]
        cur_c5 = max_val * dx
        shift = max_idx - (n - 1)
        # gradient of max conv
        g = np.ones(n)
        if shift >= 0:
            s = shift
            if s < n:
                g[s:] -= h[:n - s]
                g[:n - s] -= h[s:]
        else:
            s = -shift
            if s < n:
                g[:n - s] -= h[s:]
                g[s:] -= h[:n - s]

        step0 = 0.5
        step = step0 * (1.0 - it / max_iter) + 0.01
        new_h = h - step * g
        # projection
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_capped_simplex(new_h, target)
        new_c5 = compute_c5(new_h)

import time, math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max overlap."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def project_onto_capped_simplex(v: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto {x | sum(x)=target, 0<=x<=1}.
    Solved by a simple bisection on the Lagrange multiplier.
    """
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= 1e-12:
            return proj
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) on n points, exact integral =1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    return project_onto_capped_simplex(tri, target)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(arr) to new_n points, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    return project_onto_capped_simplex(up, target)


# ------------------------------------------------------------
# Optimization primitives
# ------------------------------------------------------------
def subgradient_descent(h0: np.ndarray, target: float, deadline: float,
                        max_iter: int = 8000, rng: np.random.Generator = None):
    """Simple sub‑gradient descent on the max‑overlap objective."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        cur_c5 = max_val * dx
        shift = max_idx - (n - 1)

        # sub‑gradient of the max‑correlation term
        g = np.ones(n)
        if shift >= 0:
            s = shift
            if s < n:
                g[s:] -= h[:n - s]
                g[:n - s] -= h[s:]
        else:
            s = -shift
            if s < n:
                g[:n - s] -= h[s:]
                g[s:] -= h[:n - s]

        # decreasing step size
        step0 = 0.8
        step = step0 * (1.0 - it / max_iter) + 0.02

        new_h = h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_capped_simplex(new_h, target)

        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = new_h.copy()

        # accept only improvements (simple descent)
        if new_c5 <= cur_c5:
            h = new_h

        # occasional random perturbation of the current best
        if (it + 1) % 2000 == 0 and rng is not None:
            pert = best_h + rng.normal(scale=0.04, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_onto_capped_simplex(pert, target)
            h = pert

    return best_h, best_c5


def refine_slsqp(h_start: np.ndarray, deadline: float):
    """One round of SLSQP using a smooth max‑approximation."""
    n = h_start.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        return softmax_c5(x, alpha=80.0)

    # allocate a small portion of remaining time
    max_iter = 300
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = minimize(obj, h_start, method="SLSQP", bounds=bounds,
                       constraints=cons, options=opts)
    except Exception:
        return None, None

    if not res.success:
        return None, None

    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_onto_capped_simplex(h_opt, target)
    return h_opt, compute_c5(h_opt)
