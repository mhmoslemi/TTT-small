               best_h = h_new.copy()

           T *= 0.95
           scale = max(scale*0.9, 1e-4)

import time, math
import numpy as np
from scipy.optimize import minimize

# globals used by callbacks (avoid closures)
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h0 with SLSQP on a soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible vector with 0≤h_i≤1 and Σh=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_anneal(h0, rng, start_time, budget_s, max_iters=15000):
    """Metropolis‑style annealing preserving sum and bounds."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    T = 0.12
    scale = 0.4
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.94:
            break
        cand = perturb_h(h, rng, scale=scale)
        cand_c = compute_c5(cand)
        if cand_c < cur_c or rng.random() < math.exp(-(cand_c - cur_c) / max(T, 1e-12)):
            h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        if (it + 1) % 500 == 0:
            T *= 0.88
            scale = max(scale * 0.85, 5e-5)
    return best_h, best_c


def parabola_h(n, rng):
    """Parabolic bump h(x)=max(0,1-((x-1)/a)^2) with a=0.75, then mass‑corrected."""
    a = 0.75
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - ((xs - 1.0) / a) ** 2)
    # repair sum (target = n/2)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # small uniform tweak then redistribution
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def upsample(h):
    """Duplicate each entry – preserves total mass."""
    return np.repeat(h, 2)

            continue

    # ---------- global search: annealing + short SLSQP ----------
    while time.time() - start_time < budget_s * 0.97:
        # start either from the current best (small perturb) or fresh random
        if rng.random() < 0.35:
            h_start = best_h.copy()
            for _ in range(5):
                h_start = perturb_h(h_start, rng, scale=0.12)
        else:
            h_start = random_feasible_vector(n, rng)

        h_anneal, c_anneal = simulated_anneal(h_start, rng, start_time, budget_s)

        # cheap SLSQP polishing
        for beta in betas[:2]:
            try:
                h_anneal = slsqp_optimize(h_anneal, beta=beta, max_iter=180)
            except Exception:
                pass
        c_anneal = compute_c5(h_anneal)

        if c_anneal < best_c:
            best_h, best_c = h_anneal, c_anneal
            if best_c < 0.38085:  # target already reached
                break

    # ---------- optional up‑sampling ----------
    if time.time() - start_time < budget_s * 0.99 and n * 2 <= 360:
        h_up = upsample(best_h)
        n_up = h_up.size
        # brief hill‑climb on finer grid
        h_up, c_up = hill_climb(h_up, rng, max_iters=2000, start_step=0.3)
        for beta in betas:
            if time.time() - start_time > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta=beta, max_iter=250)
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ---------- final feasibility cleanup ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
