import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1 - h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _adjust_sum(x, target, tol=1e-12, max_iter=1000):
    """
    Project x onto the intersection of the hyper‑plane sum(x)=target
    and the box 0≤x≤1.
    Simple iterative redistribution that respects bounds.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < tol:
        return x
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        if diff > 0:
            mask = x < 1.0 - tol
        else:
            mask = x > tol
        if not np.any(mask):
            break
        cand = np.where(mask)[0]
        per = diff / len(cand)
        for i in cand:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < tol:
                break
    # one final clip to guarantee bounds
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h, c5, deadline, rng):
    """Random pairwise mass‑exchange hill climbing (only accepting improvements)."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    """Hybrid simulated‑annealing / gradient search for C₅."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible step vector
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_vec = np.asarray(init_vec, dtype=np.float64)
        m0 = init_vec.size
        n = m0 * 2                     # double resolution for more freedom
        # linear interpolation onto the finer grid
        x0 = np.linspace(0.0, 2.0, m0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, init_vec)
    else:
        n = 90
        h = rng.random(n)

    target = n / 2.0                     # because ∫h = 1  →  sum h = n/2
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (simulated annealing + gradient steps)
    # ------------------------------------------------------------------
    deadline = start + budget_s - 0.5   # leave tiny margin for final cleanup

    # annealing schedule
    T0 = 8e-3
    Tf = 5e-7
    # gradient step schedule
    alpha0 = 0.18
    alpha_f = 0.006

    while time.time() < deadline:
        # fraction of elapsed time (0 → 1)
        t_frac = (time.time() - start) / (deadline - start)
        T = T0 * (Tf / T0) ** t_frac
        alpha = alpha0 * (alpha_f / alpha0) ** t_frac

        # decide move type
        if rng.random() < 0.75:                      # random pairwise exchange
            i, j = rng.choice(n, size=2, replace=False)
            if rng.random() < 0.5:
                src, dst = i, j
            else:
                src, dst = j, i
            max_move = min(cur_h[src], 1.0 - cur_h[dst])
            if max_move <= 0.0:
                continue
            # move size scaled a bit by temperature (larger early)
            scale = min(1.0, T / T0 * 5.0 + 0.2)
            delta = rng.random() * max_move * scale
            new_h = cur_h.copy()
            new_h[src] -= delta
            new_h[dst] += delta
        else:                                         # gradient step on worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            k = int(np.argmax(corr))
            r = k - (n - 1)                           # integer shift (can be negative)

            # compute sub‑gradient of the active shift
            g = np.ones_like(cur_h)
            if r >= 0:
                # i+r exists for i < n-r
                g[:n - r] -= cur_h[r:]
                # i-r exists for i >= r
                g[r:] -= cur_h[:n - r]
            else:
                r_abs = -r
                g[r_abs:] -= cur_h[:n - r_abs]
                g[:n - r_abs] -= cur_h[r_abs:]

            new_h = _adjust_sum(new_h, target)

        delta_c5 = new_c5 - cur_c5

            if new_c5 < best_c5 - 1e-12:
                best_h, best_c5 = new_h.copy(), new_c5

    # 3. Final pure hill‑climb to polish the best candidate
    # ------------------------------------------------------------------
    best_h, best_c5 = _hill_climb(best_h, best_c5, deadline, rng)

    # ------------------------------------------------------------------
    # 4. Return feasible result
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
