import time, random, math
import numpy as np
from scipy.optimize import minimize

# ----- global for equality constraint ---------------------------------
TARGET_SUM_GLOBAL = None


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


# ----- projection onto [0,1]^n ∩ {sum = target} -----------------------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Bisection on Lagrange multiplier to enforce box and sum constraints."""
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


# ----- basic shapes ----------------------------------------------------
def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)  # peak at centre, zero at edges
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangular(n: int) -> np.ndarray:
    """Block of height 1 on an interval of length 1 (half the domain)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    return h


def upsample_initial(old_h: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing vector to length n."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return project_onto_bounds_and_sum(new_h, n / 2.0)


# ----- objective utilities ---------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretised."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth surrogate for the max‑overlap objective."""
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return smooth_max(conv, alpha) * dx

    """Search for a step function h on [0,2] with a small Erdős C₅ value."""
    global TARGET_SUM_GLOBAL
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    n = 250                     # number of discretisation points (even)
    dx = 2.0 / n
    target_sum = n / 2.0
