def project_onto_bounds_and_sum(y, target):

    # So there's unique λ.

    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

        # subtract contributions

        if s >= 0:

            # h[i-s] is not present; for i>=s, use h[i-s] which is h[0:n-s]

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # --------------------------------------------------------------
    # Helpers
    # --------------------------------------------------------------
    def evaluate_raw(v):
        return float(np.correlate(v, 1 - v, mode="full").max())

    def project_sum(v, target):
        """Clip to [0,1] and adjust a single interior coordinate to meet the sum."""
        v = np.clip(v, 0.0, 1.0)
        diff = v.sum() - target
        if abs(diff) < 1e-12:
            return v
        # indices not at a bound
        interior = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
        if interior.size == 0:          # fall back – adjust any feasible entry
            for i in range(len(v)):
                if diff > 0 and v[i] > 0:
                    d = min(v[i], diff)
                    v[i] -= d
                    diff -= d
                elif diff < 0 and v[i] < 1:
                    d = min(1 - v[i], -diff)
                    v[i] += d
                    diff += d
                if abs(diff) < 1e-12:
                    break
            return v
        # adjust the first interior point; it stays within bounds because |diff|≤1
        i = interior[0]
        v[i] -= diff
        v[i] = min(max(v[i], 0.0), 1.0)
        # tiny residual correction if needed
        diff = v.sum() - target
        if abs(diff) > 1e-12:
            for i in interior:
                adj = -diff
                if adj > 0:
                    adj = min(1 - v[i], adj)
                else:
                    adj = max(-v[i], adj)
                v[i] += adj
                diff += adj
                if abs(diff) < 1e-12:
                    break
        return v

    # --------------------------------------------------------------
    # Initial vector (respect the length of the provided seed)
    # --------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    except NameError:
        n = 70
        h = np.random.rand(n)

    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, target_sum)

    best_h = h.copy()
    best_raw = evaluate_raw(best_h)

    # --------------------------------------------------------------
    # Parameters for the hybrid search
    # --------------------------------------------------------------
    T = max(1e-4, best_raw * 0.02)          # temperature for Metropolis steps
    decay_T = 0.995
    step = 0.25                              # gradient step magnitude
    decay_step = 0.998
    it = 0
    no_improve = 0

    cur_raw = best_raw
    while time.time() - start < budget_s * 0.96:
        # ----- correlation & worst shift -----
        corr_full = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr_full))
        s = idx_max - (n - 1)                # shift attaining the current max

        # ----- gradient of the worst shift (to decrease it) -----
        g = np.ones(n)
        if s >= 0:
            g[: n - s] -= h[s:]
            g[s:]     -= h[: n - s]
        else:
            d = -s
            g[d:]     -= h[: n - d]
            g[: n - d] -= h[d:]

        # Make the update preserve the sum (zero‑mean direction)
        g_adj = g - np.mean(g)
        h_candidate = h - step * g_adj
        h_candidate = project_sum(h_candidate, target_sum)
        cand_raw = evaluate_raw(h_candidate)

        if cand_raw < cur_raw or random.random() < math.exp(-(cand_raw - cur_raw) / max(T, 1e-12)):
            h = h_candidate
            cur_raw = cand_raw
            if cand_raw < best_raw:
                best_raw = cand_raw
                best_h = h.copy()
                no_improve = 0
        else:
            no_improve += 1

        # ----- occasional random two‑coordinate move (preserves sum) -----
        if random.random() < 0.35:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            max_up   = min(1.0 - hi, hj)    # increase i, decrease j
            max_down = min(hi, 1.0 - hj)    # decrease i, increase j
            if max_up + max_down > 0:
                delta = (random.random() * (max_up + max_down)) - max_down
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                # (sum unchanged, bounds already respected)
                new_raw = evaluate_raw(nh)
                if new_raw < cur_raw or random.random() < math.exp(-(new_raw - cur_raw) / max(T, 1e-12)):
                    h = nh
                    cur_raw = new_raw
                    if new_raw < best_raw:
                        best_raw = new_raw
                        best_h = h.copy()
                        no_improve = 0

        # ----- targeted move on the worst shift -----
        if it % 70 == 0:
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:

                        no_improve = 0

        it += 1
        if it % 200 == 0:
            T *= decay_T
            step *= decay_step
            if step < 5e-6:

        # ----- restart if stagnating -----
        if no_improve > 8000:
            h = best_h + np.random.normal(scale=0.07, size=n)
            h = project_sum(h, target_sum)
            cur_raw = evaluate_raw(h)
            no_improve = 0

    # Final small correction (numerical drift)
    # --------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-8:
        for _ in range(100):
            k = random.randrange(n)
            if err > 0 and best_h[k] >= err:
                best_h[k] -= err
                break
            if err < 0 and best_h[k] <= 1 + err:
                best_h[k] -= err   # err negative => addition
                break

    final_c5 = best_raw * (2.0 / n)
    return (best_h.tolist(), final_c5, n)
