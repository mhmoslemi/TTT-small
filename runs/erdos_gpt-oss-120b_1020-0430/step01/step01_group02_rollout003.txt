import time
import numpy as np
from math import exp

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """Project x onto 0≤x≤1 with sum(x)=target (water‑filling)."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / idx.size
        # distribute
        for i in idx:
            if diff > 0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h, cur_c5, deadline, rng):
    """Simple pairwise mass‑exchange hill climbing."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # direction randomly chosen
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
    return h, cur_c5

def _simulated_annealing(h, cur_c5, target_sum, deadline, rng):
    """Linear‑schedule simulated annealing with mass‑swap proposals."""
    n = h.size
    best_h, best_c5 = h.copy(), cur_c5
    T0 = 5e-3                    # initial temperature
    while time.time() < deadline:
        # temperature falls linearly to zero
        fraction = (time.time() - (deadline - (deadline - start_time))) / (deadline - start_time)
        T = max(T0 * (1.0 - fraction), 0.0)

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # move mass from i to j (or reverse) respecting bounds
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)

        # accept if better or with Metropolis probability
        if new_c5 < cur_c5 or rng.random() < exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            h, cur_c5 = h_new, new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = h.copy(), cur_c5
        # occasionally re‑project to eliminate drift
        if rng.random() < 0.001:
            h = _adjust_sum(h, target_sum)
            cur_c5 = _evaluate_c5(h)
    return best_h, best_c5
