def compute_c5(h): 
    n = len(h)

    fft_len = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, fft_len)

def project_to_feasible(h):
    n = h.size
    target = n/2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo+hi)*0.5, 0.0, 1.0)
    # adjust residual
    residual = target - p.sum()
    if abs(residual) > 1e-12:
        # adjust some free variable
        # Find index not at bounds
        free = np.where((p > 1e-12) & (p < 1 - 1e-12))[0]
        if free.size > 0:
            idx = free[0]
            p[idx] += residual
            p[idx] = min(max(p[idx], 0.0), 1.0)
        else:
            # just adjust first element limited by bounds
            idx = 0
            p[idx] = min(max(p[idx] + residual, 0.0), 1.0)
    return p

        h = x[:n]
        t = x[-1]
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return t - corr_vals

import time
import math
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Projection onto the feasible set: 0≤h≤1 and sum(h)=n/2
def _project_feasible(h: np.ndarray) -> np.ndarray:
    n = h.size
    target = n / 2.0
    # binary search for lambda s.t. sum(clip(h-lambda,0,1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction (rare)
    residual = target - p.sum()
    if abs(residual) > 1e-12:
        free = np.where((p > 1e-12) & (p < 1 - 1e-12))[0]
        if free.size:
            idx = free[0]
            p[idx] = min(max(p[idx] + residual, 0.0), 1.0)
        else:
            p[0] = min(max(p[0] + residual, 0.0), 1.0)
    return p


# ------------------------------------------------------------
# C5 value (scaled max correlation)
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    # FFT‑based full correlation of h with (1‑h)
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, size)
    R = np.fft.rfft((1.0 - h)[::-1], size)
    conv = np.fft.irfft(H * R, size)[: 2 * n - 1]
    max_corr = conv.max()
    return max_corr * (2.0 / n)


# ------------------------------------------------------------
# Linear up‑sampling of a coarse step function
def _upscale(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


# ------------------------------------------------------------
# Local refinement with SLSQP (same formulation as the reference)
def _slsqp_refine(start_h: np.ndarray, max_iter: int = 200) -> tuple:
    n = start_h.size
    corr0 = np.correlate(start_h, 1.0 - start_h, mode="full")
    t0 = np.max(corr0)

    x0 = np.concatenate([start_h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        vals = np.correlate(h, 1.0 - h, mode="full")
        return t - vals

    def _eq(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(lambda x: x[-1], x0, method="SLSQP",
                            bounds=bounds, constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else start_h
    h_opt = _project_feasible(h_opt)
    return h_opt, _c5_from_h(h_opt)

                if cur_c5 < local_best_c5:
                    local_best, local_best_c5 = cur.copy(), cur_c5
                    if cur_c5 < best_c5:
                        best_h, best_c5, best_n = cur.copy(), cur_c5, n
                        if best_c5 <= target:
                            return best_h.tolist(), float(best_c5), int(best_n)

                sigma = max(sigma * sigma_decay, 0.001)
                temp *= temp_decay
                if sigma < 0.005:
                    sigma = 0.1  # occasional restart of step size

        # ------------------------------------------------------------
        # One SLSQP refinement on the best local candidate
        if time_left() > 5.0:
            refined_h, refined_c5 = _slsqp_refine(local_best, max_iter=250)
            if refined_c5 < best_c5:
                best_h, best_c5, best_n = refined_h, refined_c5, n
                if best_c5 <= target:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # --------------------------------------------------------------------
    # Fallback (should never be needed)
    if best_h is None:
        best_h = parent.copy()
        best_c5 = _c5_from_h(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c5), int(best_n)
