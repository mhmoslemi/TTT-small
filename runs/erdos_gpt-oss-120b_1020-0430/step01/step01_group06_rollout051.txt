import numpy as np

def _apply_delta(h: np.ndarray, C: np.ndarray, offset: int, idx: int, delta: float) -> None:
    """Modify h[idx] by delta (keeping it in [0,1]) and update the full correlation array C."""
    if abs(delta) < 1e-15:
        return
    n = h.shape[0]
    old = float(h[idx])
    new = old + delta
    # contributions from all other positions
    q = np.arange(n)
    mask = q != idx
    q_other = q[mask]

    # i = idx, j = q_other  -> shift = q_other - idx
    C[q_other - idx + offset] += delta * (1.0 - h[q_other])

    # i = q_other, j = idx  -> shift = idx - q_other
    C[idx - q_other + offset] += -h[q_other] * delta

    # self term (shift 0)
    C[offset] += new * (1.0 - new) - old * (1.0 - old)

    h[idx] = new


def _project_sum_to_target(v: np.ndarray, target: float) -> np.ndarray:

    v = np.clip(v, 0.0, 1.0)
    cur = float(v.sum())
    # simple scaling if possible
    if cur > 0 and abs(cur - target) > 1e-12:
        scale = target / cur
        v *= scale
        v = np.clip(v, 0.0, 1.0)
        cur = float(v.sum())

        if diff > 0:
            cand = np.where(v > 1e-12)[0]

            best_h = h_opt.copy()
            best_n = n

    # ----------------------------------------------------------------------
    # First, if an initial construction is provided, use it (n=70)
    # ----------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float).copy()
        n0 = init.shape[0]
        init = _project_sum_to_target(init, n0 / 2.0)
        # give it a modest amount of time
        remaining = budget_s - (time.time() - start)
        if remaining > 8.0:
            try_one(n0, init, min(30.0, 0.6 * remaining))
    except NameError:
        pass

    # ----------------------------------------------------------------------
    # Main loop: repeatedly launch independent searches with n=200
    # ----------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        remaining = budget_s - (time.time() - start)
        if remaining < 6.0:
            break
        n = 200
        # random binary start with exactly n/2 ones
        h0 = np.zeros(n)
        idx = np.arange(n)
        np.random.shuffle(idx)
        h0[idx[: n // 2]] = 1.0
        alloc = min(30.0, 0.7 * remaining)   # 30 s or a share of the left‑over budget
        try_one(n, h0, alloc)
        if best_score < 0.3808:           # goal already reached
            break

    # ----------------------------------------------------------------------
    # Final clean‑up: ensure feasibility and compute the final bound
    # ----------------------------------------------------------------------
    if best_h is None:
        # fallback – a simple alternating pattern (never optimal but ensures a return)
        n = 70
        best_h = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)], dtype=float)
        best_n = n
        best_score = (np.correlate(best_h, 1 - best_h, mode='full').max()) * (2.0 / n)

    best_h = np.clip(best_h, 0.0, 1.0)
    # tiny correction for the sum (should already be exact)
    target = best_n / 2.0
    err = float(best_h.sum() - target)
    if abs(err) > 1e-9:
        for _ in range(10):
            k = random.randrange(best_n)
            if err > 0 and best_h[k] >= err:
                best_h[k] -= err
                break
            if err < 0 and best_h[k] <= 1.0 + err:
                best_h[k] -= err   # err negative ==> addition
                break

    final_c5 = (np.correlate(best_h, 1 - best_h, mode='full').max()) * (2.0 / best_n)
    return (best_h.tolist(), float(final_c5), best_n)
