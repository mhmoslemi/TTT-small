import time, numpy as np

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _full_corr_fft(h, L):
    # correlation of h with (1-h) using FFT, length 2*len(h)-1
    n = h.shape[0]
    h_f = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    rev_f = np.fft.rfft(rev, L)
    conv = np.fft.irfft(h_f * rev_f, L)[:2 * n - 1]
    return conv

def _adam_optimize(h0, target, steps, lr):

    beta2 = 0.999

    L = int(2 ** np.ceil(np.log2(2 * n - 1)))

        corr = _full_corr_fft(h, L)
        max_val = corr.max()
        # active set (all shifts achieving max up to tiny tolerance)

        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h -= step
        # projection
        h = _project_onto_capped_simplex(h, target)
    # final value
    final_corr = _full_corr_fft(h, L).max() * (2.0 / n)
    return h, final_corr

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    """Project onto {0≤x≤1, sum x = target} via bisection on the Lagrange multiplier."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng, n, target):
    """Random point in the capped simplex."""
    v = rng.rand(n)

def _gradient_for_shift(h, shift):
    """Sub‑gradient of correlation for a given shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0

    else:

        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g
