def compute_c5_fft(h, N=None, fft_one=None):
    n = h.shape[0]
    if N is None:
        N = 1 << ((2*n - 1).bit_length())
    if fft_one is None:
        fft_one = np.fft.fft(np.concatenate([np.ones(n), np.zeros(N-n)]))
    fft_h = np.fft.fft(h, N)
    fft_c = fft_one - fft_h
    conv = np.fft.ifft(fft_h * fft_c).real[:2*n-1]
    return conv.max() * (2.0 / n)
```

def upsample(arr, new_n):
    old_n = arr.shape[0]
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    return np.interp(x_new, x_old, arr)

def anneal_binary(h, time_limit):
    n = h.shape[0]
    dx = 2.0 / n
    N = 1 << ((2*n - 1).bit_length())
    fft_one = np.fft.fft(np.concatenate([np.ones(n), np.zeros(N-n)]))
    cur_h = h.copy()
    # compute current cost
    cur_c5 = compute_c5_fft(cur_h, N, fft_one)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # maintain ones and zeros list
    ones = list(np.flatnonzero(cur_h > 0.5))  # should be exact ones
    zeros = [i for i in range(n) if cur_h[i] < 0.5]

    rng = np.random.default_rng()
    start = time.time()
    total_allowed = time_limit - start
    # if total_allowed <= 0, return
    T0 = 0.02
    T_end = 1e-4

    it = 0
    while True:
        now = time.time()
        if now >= time_limit:
            break
        elapsed = now - start
        progress = elapsed / total_allowed
        T = max(T_end, T0 * (1 - progress))
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        # compute new cost
        new_c5 = compute_c5_fft(cur_h, N, fft_one)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            # update lists
            # i moves from ones->zeros, j from zeros->ones
            ones.remove(i)

def slsqp_optimize(start, n, time_limit):
    target = n/2
    bounds = [(0.,1.)]*n
    cons = {'type':'eq','fun': lambda x: x.sum() - target}
    # function for softmax
    def obj_alpha(alpha):
        def f(x):
            return softmax_c5(x, alpha=alpha)
        return f
    # stage schedule
    alphas = [5., 30., 100.]
    maxiters = [300, 300, 300]
    x = start
    for alpha, maxit in zip(alphas, maxiters):
        if time.time() > time_limit:
            break
        res = minimize(obj_alpha(alpha), x, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxit, 'ftol':1e-9, 'disp':False})
        if not res.success:
            # continue anyway
            x = res.x
        else:

# Core evaluation utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)

# Construction helpers
# ------------------------------------------------------------
def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation + renormalisation to exact integral."""
    old_n = arr.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(x_new, x_old, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up = up + diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

    """Triangular shape h(x)=min(x,2‑x), normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n

def slsqp_optimize(start: np.ndarray, n: int, deadline: float) -> np.ndarray:
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    x = start.copy()
    # progressive smoothing
    schedule = [(5.0, 400), (30.0, 400), (100.0, 400)]
    for alpha, maxit in schedule:
        if time.time() > deadline:
            break
        res = minimize(
            lambda v: softmax_c5(v, alpha=alpha),
            x,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxit, "ftol": 1e-9, "disp": False},
        )
        x = res.x if hasattr(res, "x") else x
    # final projection onto the feasible set
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = x + diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

            # binary from a continuous starter (top‑half threshold)
            src = parent_up if parent_up is not None else random_continuous_start(n, rng)
            idx = np.argpartition(-src, int(target))[: int(target)]
            start = np.zeros(n, dtype=float)
            start[idx] = 1.0

        # enforce exact integral
        diff = target - start.sum()
        if abs(diff) > 1e-12:
            start = start + diff / n
            start = np.clip(start, 0.0, 1.0)

        # local continuous optimisation
        h_opt = slsqp_optimize(start, n, time_limit)
        c5 = compute_c5(h_opt)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h = c5, h_opt.copy()

        attempts += 1
        if best_c5 < 0.3808:  # good enough – stop early
            break

    # final hill‑climbing tweak (if time permits)
    if best_h is not None and time.time() < time_limit:
        best_h, best_c5 = local_improve(best_h, time_limit)

    # fallback in the unlikely event no feasible point was found
    if best_h is None:
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_c5 = compute_c5(best_h)

    # final sanity projection
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
