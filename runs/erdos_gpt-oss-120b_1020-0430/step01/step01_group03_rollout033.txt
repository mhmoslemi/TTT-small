def grad_softmax(h, beta):
    corr = np.correlate(h, 1-h, mode='full')
    # shape 2n-1, index s = n-1 + shift
    # compute weights
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / np.sum(exp_bc)  # shape 2n-1
    
    # gradient init
    n = h.size
    grad = np.zeros(n)
    # For each shift
    for shift in range(-(n-1), n):
        weight = w[shift + n - 1]
        if shift >= 0:
            # i from 0 to n-shift-1 inclusive (n-shift length)
            i_end = n - shift
            # contribution to grad[i] (i from 0..i_end-1)
            grad[:i_end] += weight * (1 - h[shift:])   # shape i_end
            # contribution to grad[i+shift] (j = i+shift)
            grad[shift:] -= weight * h[:i_end]
        else: # shift negative
            k = -shift
            # i from 0 to n - k -1 (since i and i - k in range)
            i_end = n - k
            # Here expression: corr_{-k} = sum_i h_i (1 - h_{i - k}), where i - k >=0; i.e., i >= k.
            # Better treat symmetrical: corr_{-k} = sum_i h_{i} (1 - h_{i - k}) for i >= k.
            # Equivalent to shift positive k but swapping arguments.
            # We can compute contributions similarly by swapping indices:
            # For i from k to n-1 (i is index of first factor h_i)
            # grad[i] from first term: (1 - h[i - k])
            # grad[i - k] from second term: -h[i]
            # Implementation: i_slice = slice(k, n), j_slice = slice(0, n - k)
            grad[k:] += weight * (1 - h[:n - k])
            grad[:n - k] -= weight * h[k:]
    return grad

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_sum_feasible(v):
    """Project onto {0≤x_i≤1, Σx_i = len(v)/2} by simple redistribution."""
    n = v.size
    target = n / 2.0
    diff = target - v.sum()
    # tiny adjustment on a single coordinate when possible
    if abs(diff) > 1e-12:
        for i in range(n):
            new = v[i] + diff
            if 0.0 <= new <= 1.0:
                v[i] = new
                diff = target - v.sum()
                break
    # if still not exact (should be extremely rare) use a small random redistribution
    if abs(diff) > 1e-12:
        rng = np.random.default_rng(0)
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - v[i])
            else:
                add = max(diff, -v[i])
            v[i] += add
            diff = target - v.sum()
    return v

def softmax_gradient(h, beta):
    """Gradient of (1/beta)·log ∑ exp(beta·corr) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / np.sum(exp_bc)                     # soft‑max weights
    grad = np.zeros_like(h)
    for shift in range(-(n - 1), n):
        wk = w[shift + n - 1]
        if wk == 0.0:
            continue
        if shift >= 0:
            end = n - shift
            grad[:end] += wk * (1 - h[shift:])
            grad[shift:] -= wk * h[:end]
        else:
            k = -shift
            end = n - k
            grad[k:] += wk * (1 - h[:end])
            grad[:end] -= wk * h[k:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """
    Build a simple monotone step function (linear ramp) and polish it.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # choose resolution
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 150                       # moderate size, well within limits

    # linear ramp: h_i = i/(n-1)  (sum equals n/2 exactly)
    h = np.arange(n, dtype=float) / (n - 1)
    h = project_sum_feasible(h)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # brief soft‑max gradient descent to smooth any tiny irregularities
    # (mostly a safety net; the linear shape is already excellent)
    beta = 120.0
    step = 0.12
    max_iters = 300
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.95:
            break
        grad = softmax_gradient(best_h, beta)
        # move opposite to gradient (we minimise the soft‑max of the corr)
        cand = best_h - step * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_sum_feasible(cand)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c
        step *= 0.995                     # slowly shrink step size

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_feasible(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
