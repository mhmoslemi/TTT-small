    # compute gradient:
    mid = N-1

    conv = np.correlate(h, 1 - h, mode='full')
    # stable exponent
    a_conv = alpha * conv
    a_max = np.max(a_conv)
    exp_term = np.exp(a_conv - a_max)
    w = exp_term / exp_term.sum()
    smooth = np.dot(w, conv)
    # gradient
    mid = N - 1
    grad = np.zeros(N)
    # contribution for t=0
    w0 = w[mid]
    grad += w0 * (1 - 2 * h)
    # contributions for t>=1
    for t in range(1, N):
        wp = w[mid + t]
        wn = w[mid - t]
        # positive shift
        # indices i[:N-t] and i[t:]
        # (1 - h[i+t]) for i in [0,N-t-1]
        grad[:N-t] += wp * (1 - h[t:])
        # -h[i-t] for i in [t,N-1]
        grad[t:] += wp * (-h[:N-t])
        # negative shift
        # (1 - h[i - t]) for i in [t,N-1]
        grad[t:] += wn * (1 - h[:N-t])
        # -h[i + t] for i in [0,N-t-1]
        grad[:N-t] += wn * (-h[t:])
    return smooth, grad

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Core evaluation
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


# ------------------------------------------------------------
# Simple constructions
# ------------------------------------------------------------
def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0 / n)
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + (1.0 / old_n)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + (1.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def random_binary_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 0/1 vector with exactly n/2 ones (n must be even)."""
    a = np.zeros(n)
    a[: n // 2] = 1.0
    rng.shuffle(a)
    return a


# ------------------------------------------------------------
# Smooth max and its gradient
# ------------------------------------------------------------
def softmax_c5_and_grad(h: np.ndarray, alpha: float):
    """
    Smooth approximation of max overlap and its gradient.
    Returns (value, grad) where value = log‑sum‑exp(alpha*conv)/alpha.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")          # length 2n‑1
    a_conv = alpha * conv
    # stable log‑sum‑exp
    a_max = np.max(a_conv)
    exp_shift = np.exp(a_conv - a_max)
    w = exp_shift / exp_shift.sum()                     # softmax weights
    smooth = float(np.dot(w, conv))

    # gradient of smooth = sum_j w_j * d conv_j / d h
    mid = n - 1
    grad = np.zeros(n)

    # shift 0
    w0 = w[mid]
    grad += w0 * (1.0 - 2.0 * h)

    # shifts |s| >= 1
    for t in range(1, n):
        wp = w[mid + t]     # shift +t
        wn = w[mid - t]     # shift -t

        # positive shift (+t)
        grad[: n - t] += wp * (1.0 - h[t:])      # (1 - h[i+t])
        grad[t:]      += wp * (-h[: n - t])     # -h[i-t]

        # negative shift (‑t)
        grad[t:]      += wn * (1.0 - h[: n - t])   # (1 - h[i‑t])
        grad[: n - t] += wn * (-h[t:])            # -h[i+t]

    return smooth, grad


# ------------------------------------------------------------
# Projection onto [0,1] box with exact sum
# ------------------------------------------------------------
def project_to_box_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of x onto the intersection of the box [0,1]^n
    and the affine hyperplane sum = target.
    """
    lo, hi = -2.0, 2.0   # generous bounds for binary search on Lagrange multiplier
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


# ------------------------------------------------------------
# Optimisation of one start
# ------------------------------------------------------------
def optimise_start(start: np.ndarray,
                   alphas,
                   penalty,
                   target,
                   maxiter_stage):
    """
    Multi‑stage L‑BFGS‑B optimisation using the smooth max.
    Returns the final feasible vector.
    """
    h = start.copy()
    n = h.shape[0]
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        # objective with penalty for the sum constraint
        def fun(x):
            smooth, grad = softmax_c5_and_grad(x, alpha)
            diff = x.sum() - target
            val = smooth + penalty * diff * diff
            grad = grad + (2.0 * penalty * diff) * np.ones_like(x)
            return val, grad

        res = minimize(fun,
                       h,
                       method="L-BFGS-B",
                       jac=True,
                       bounds=bounds,
                       options={"maxiter": maxiter_stage, "ftol": 1e-12})
        # keep whatever we got; the optimizer may stop early
        h = np.clip(res.x, 0.0, 1.0)

    # final projection onto the exact feasible set
    h = project_to_box_and_sum(h, target)
    return h


# ------------------------------------------------------------
# Simple local improvement (swap mass between two points)
# ------------------------------------------------------------
def local_improvement(h: np.ndarray,
                      target: float,
                      time_limit: float):
    """Hill‑climbing that preserves the sum by moving mass between two coordinates."""
    n = h.shape[0]
    best = h.copy()
    best_c5 = compute_c5(best)
    rng = np.random.default_rng()
    while time.time() < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # maximum transferable amount without violating bounds
        max_move = min(best[i], 1.0 - best[j])
        if max_move < 1e-12:
            continue
        delta = rng.random() * max_move
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        c5 = compute_c5(cand)
        if c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = c5
    return best, best_c5


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.96  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (all even for binary starts)
    candidate_ns = [80, 100, 120, 150, 180, 200]

    # include parent size if provided
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    # main search loop
    for n in candidate_ns:
        if time.time() > time_limit:
            break

        target = n / 2.0
        starts = []

        # 1. parent (upsampled / downsampled)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # 2. triangular baseline
        starts.append(generate_triangular(n))

        # 3. random binary patterns
        if n % 2 == 0:
            for _ in range(3):
                starts.append(random_binary_feasible(n, rng))

        # 4. few random continuous starts
        for _ in range(2):
            a = rng.random(n)
            a = a * (target / a.sum())
            a = np.clip(a, 0.0, 1.0)
            diff = target - a.sum()
            if abs(diff) > 1e-12:
                a += diff / n
                a = np.clip(a, 0.0, 1.0)
            starts.append(a)

        # optimisation of each start
        for s in starts:
            if time.time() > time_limit:
                break
            h_opt = optimise_start(s,
                                   alphas=[5.0, 15.0, 30.0, 60.0, 100.0],
                                   penalty=1e3,
                                   target=target,
                                   maxiter_stage=150)
            c5_val = compute_c5(h_opt)
            if c5_val + 1e-12 < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                best_n = n

    # fallback if nothing succeeded
    if best_h is None:
        n = 80
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_n = n
        best_c5 = compute_c5(best_h)

    # final local refinement with remaining time
    remaining = time_limit - time.time()
    if remaining > 5.0:
        best_h, best_c5 = local_improvement(best_h,
                                            target=best_n / 2.0,
                                            time_limit=time.time() + remaining)

    # final projection to guarantee feasibility
    best_h = project_to_box_and_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
