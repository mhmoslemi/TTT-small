def _pairwise_hill_climb(h, cur_score, deadline, rng):
    n = len(h)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # move random amount from i to j
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 0:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_score = _evaluate_c5(h_new)
        if new_score + 1e-12 < cur_score:
            h, cur_score = h_new, new_score
    return h, cur_score

import time
import numpy as np

def _evaluate_c5(h):
    """Integral approximation of max overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x, target, eps=1e-12):
    """Project onto box [0,1] while fixing the sum to target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < eps:
        return x
    # water‑filling style redistribution
    while abs(diff) > eps:
        free = (x > eps) & (x < 1.0 - eps)
        nz = np.count_nonzero(free)
        if nz == 0:
            # cannot move any further; break
            break
        per = diff / nz
        x[free] += per
        # re‑clip any overflow/underflow
        np.clip(x, 0.0, 1.0, out=x)
        diff = target - x.sum()
    return x


def _triangle_h(n):
    """Triangular tent: h(x)=1-|x-1| on [0,2]."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    target = n / 2.0
    return _adjust_sum(h, target)


def _upsample(h_small, n):
    """Linear interpolation from a smaller grid to length n."""
    n0 = h_small.shape[0]
    # source grid (midpoint samples)
    xs = (np.arange(n0) + 0.5) * (2.0 / n0)
    xt = (np.arange(n) + 0.5) * (2.0 / n)
    h_big = np.interp(xt, xs, h_small, left=h_small[0], right=h_small[-1])
    target = n / 2.0
    return _adjust_sum(h_big, target)


def _directed_hill_climb(h, cur_score, deadline, rng):
    """Greedy moves guided by the gradient of the worst shift."""
    n = h.shape[0]
    while time.time() < deadline:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        idx = np.argmax(corr)                # single worst shift
        k = idx - (n - 1)                    # shift (may be negative)

        # gradient of that shift
        g = np.zeros_like(h)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            m = -k
            g[m:] += 1.0 - h[: n - m]
            g[: n - m] -= h[m:]

        # choose source (positive gradient) and destination (negative)
        src_cand = np.where(g > 1e-12)[0]
        dst_cand = np.where(g < -1e-12)[0]
        if src_cand.size == 0 or dst_cand.size == 0:
            break

        src = src_cand[np.argmax(g[src_cand])]
        dst = dst_cand[np.argmin(g[dst_cand])]

        # move as much as possible, but keep some randomness
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-14:
            break
        delta = max_move * rng.uniform(0.3, 1.0)

        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_score = _evaluate_c5(h_new)

        if new_score + 1e-12 < cur_score:
            h, cur_score = h_new, new_score
            # continue improving the same worst shift
        else:
            # if this move does not help, stop the directed phase
            break
    return h, cur_score


def _hill_climb(h, cur_score, deadline, rng):
    """Simple random pairwise mass exchange (fallback refinement)."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-14:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_score = _evaluate_c5(h_new)
        if new_score + 1e-12 < cur_score:
            h, cur_score = h_new, new_score
    return h, cur_score


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine resolution and initial candidates
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        base = np.asarray(init_vec, dtype=np.float64)
        n0 = base.size
        if n0 < 80:
            n = n0 * 2               # finer grid
        else:
            n = n0
        # upsample the given vector to the working resolution
        h_base = _upsample(base, n)
    else:
        n = 90
        h_base = rng.random(n)
        h_base = _adjust_sum(np.clip(h_base, 0.0, 1.0), n / 2.0)

    # ensure feasibility
    target = n / 2.0
    h_base = _adjust_sum(np.clip(h_base, 0.0, 1.0), target)

    # additional seeds: a triangular shape and a random binary pattern
    h_tri = _triangle_h(n)
    h_rand = rng.random(n)
    h_rand = _adjust_sum(np.clip(h_rand, 0.0, 1.0), target)

    candidates = [h_base, h_tri, h_rand]
    best_h = None
    best_score = float("inf")
    for cand in candidates:
        sc = _evaluate_c5(cand)
        if sc < best_score:
            best_h, best_score = cand.copy(), sc

    # ------------------------------------------------------------------
    # 2. Main stochastic optimisation loop
    # ------------------------------------------------------------------
    deadline = start + budget_s - 0.5
    outer_iter = 0
    while time.time() < deadline:
        # pick a starting point: perturb the current best or fresh random
        if rng.random() < 0.6:
            h_curr = best_h.copy()
            h_curr += rng.normal(scale=0.04, size=n)
            h_curr = np.clip(h_curr, 0.0, 1.0)
            h_curr = _adjust_sum(h_curr, target)
        else:
            h_curr = rng.random(n)
            h_curr = _adjust_sum(np.clip(h_curr, 0.0, 1.0), target)

        # directed hill climbing on the worst shift
        cur_score = _evaluate_c5(h_curr)
        h_curr, cur_score = _directed_hill_climb(h_curr, cur_score, deadline, rng)

        # occasional random hill climb for fine tweaking
        if rng.random() < 0.3:
            h_curr, cur_score = _hill_climb(h_curr, cur_score, deadline, rng)

        if cur_score + 1e-12 < best_score:
            best_h, best_score = h_curr.copy(), cur_score
            # early exit if target reached
            if best_score < 0.3808:
                break
        outer_iter += 1

    # ------------------------------------------------------------------
    # 3. Final refinement with a pure random hill climb
    # ------------------------------------------------------------------
    final_deadline = start + budget_s - 0.1
    best_h, best_score = _hill_climb(best_h, best_score, final_deadline, rng)

    # ------------------------------------------------------------------
    # 4. Return result
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_score = float(_evaluate_c5(best_h))
    return best_h, best_score, n
