def _smooth_optimize(h, max_iter=200, alpha_start=1.0, alpha_end=40.0, lr_start=1.0, lr_end=0.01):
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    for it in range(max_iter):
        alpha = alpha_start + (alpha_end - alpha_start) * it / (max_iter-1)
        lr = lr_start * (lr_end / lr_start) ** (it / (max_iter-1))
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute softmax weights
        # compute logsumexp
        max_corr = np.max(alpha * corr)  # for stability
        exp_vals = np.exp(alpha * corr - max_corr)
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp  # has shape (2n-1)
        # gradient
        grad = np.zeros(n)
        # iterate over shifts and accumulate contributions using slices
        # s from -(n-1) ... (n-1)
        for idx_s, s in enumerate(range(-n+1, n)):
            ws = w[idx_s]
            if ws == 0:
                continue
            # compute i_start, i_end
            i_start = max(0, -s)
            i_end = min(n-1, n-1-s)   # inclusive
            if i_start > i_end:
                continue
            # 1 term
            grad[i_start:i_end+1] += ws
            # -h[i+s] term: subtract ws * h[i+s] for each i
            # For slice i range, compute slice for h[i+s]
            if s >= 0:
                grad[i_start:i_end+1] -= ws * h[i_start+s:i_end+1+s]
                # -h[i] term at index i+s
                grad[i_start+s:i_end+1+s] -= ws * h[i_start:i_end+1]
            else:
                s_abs = -s
                # For s negative, i range is [s_abs, n-1], i_start = s_abs
                # Subtract h[i - s_abs] (i-s) factor from i range
                grad[i_start:i_end+1] -= ws * h[i_start - s_abs:i_end+1 - s_abs]
                # second term: subtract h[i] from index i - s_abs
                grad[i_start - s_abs:i_end+1 - s_abs] -= ws * h[i_start:i_end+1]
        # scale by dx
        dx = 2.0 / n
        grad *= dx
        # step
        h_new = h - lr * grad
        # project to feasible
        h_new = _project_to_feasible(h_new, n)

            best_h = h_new.copy()

        h = h_new
        # early stop if improvement small:
        # optional break.
    return best_h, best_c5

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ----- feasibility projection -------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ----- evaluation ------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ----- up‑sampling -----------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----- SLSQP refinement ------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x, n=n):
        return x[-1] - np.correlate(x[:n], 1.0 - x[:n], mode="full")

    def eq(x, n=n):
        return np.sum(x[:n]) - (n / 2.0)

    constraints = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = opt.minimize(lambda x: x[-1], x0, method="SLSQP",
                           bounds=bounds, constraints=constraints, options=opts)
    except Exception:
        res = None

    h_opt = res.x[:n] if res is not None and res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ----- smooth (log‑sum‑exp) optimisation ------------------------------------
def _smooth_optimize(h0: np.ndarray,
                     max_iter: int = 200,
                     alpha_start: float = 5.0,
                     alpha_end: float = 80.0,
                     lr_start: float = 0.5,
                     lr_end: float = 0.001) -> tuple:
    n = h0.size
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    for it in range(max_iter):
        # continuation parameters
        alpha = alpha_start + (alpha_end - alpha_start) * it / (max_iter - 1)
        lr = lr_start * (lr_end / lr_start) ** (it / (max_iter - 1))

        corr = np.correlate(h, 1.0 - h, mode="full")
        # soft‑max weights
        w = np.exp(alpha * corr - logsumexp(alpha * corr))

        grad = np.zeros(n)
        # accumulate weighted gradients for each shift
        for idx_s, s in enumerate(range(-n + 1, n)):
            ws = w[idx_s]
            if ws == 0.0:
                continue
            i_start = max(0, -s)
            i_end = min(n - 1, n - 1 - s)          # inclusive
            if i_start > i_end:
                continue
            # +1 term
            grad[i_start:i_end + 1] += ws
            if s >= 0:
                grad[i_start:i_end + 1] -= ws * h[i_start + s:i_end + 1 + s]
                grad[i_start + s:i_end + 1 + s] -= ws * h[i_start:i_end + 1]
            else:
                s_abs = -s
                grad[i_start:i_end + 1] -= ws * h[i_start - s_abs:i_end + 1 - s_abs]
                grad[i_start - s_abs:i_end + 1 - s_abs] -= ws * h[i_start:i_end + 1]

        # gradient step and projection
        h_new = h - lr * grad
        h_new = _project_to_feasible(h_new, n)

        c5_val = _c5_from_h(h_new)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


# ----- simple local refinement ------------------------------------------------
def _local_refine(h: np.ndarray,
                 max_iter: int = 2000,
                 delta_max: float = 0.05,
                 time_limit: callable = None) -> tuple:
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    for _ in range(max_iter):
        if time_limit is not None and time_limit() <= 0:
            break
        # pick donor and receiver
        donors = np.where(best_h > 1e-12)[0]
        receivers = np.where(best_h < 1 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = np.random.choice(donors)
        j = np.random.choice(receivers)
        max_delta = min(delta_max, best_h[i], 1 - best_h[j])
        if max_delta <= 0:

        delta = max_delta * np.random.rand()
        candidate = best_h.copy()
        candidate[i] -= delta
        candidate[j] += delta
        c5_new = _c5_from_h(candidate)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = candidate
    return best_h, best_c5

                lr_start=0.5,
                lr_end=0.001,
            )

            # SLSQP refinement
            h_opt, c5_opt = _solve_h(h_opt, max_iter=300)

            # local greedy refinement
            if time_left() > 2.0:
                h_opt, c5_opt = _local_refine(
                    h_opt,
                    max_iter=2000,
                    delta_max=0.03,
                    time_limit=time_left,
                )

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                if best_c5 <= target:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # fallback if nothing better was found
    if best_h is None:
        if init_coarse is not None:
            n = init_coarse.size
            best_h = _project_to_feasible(init_coarse.copy(), n)
            best_c5 = _c5_from_h(best_h)
            best_n = n
        else:
            n = 128
            best_h = np.full(n, 0.5)
            best_c5 = _c5_from_h(best_h)
            best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
