def grad_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # compute softmax weights
    # using stable exponentials:
    maxc = np.max(corr)
    exps = np.exp(alpha * (corr - maxc))
    w = exps / exps.sum()
    # compute gradient:
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, 1 - h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    base = n - 1
    termA = conv1[base:base + n]
    termB = conv2[base:base + n]
    grad = termA - termB
    return grad

def _adam_optimize(h0, target, alphas, steps_per_alpha, lr0=0.1):

    beta2 = 0.999
    eps = 1e-8
    t = 0
    for alpha, iters in zip(alphas, steps_per_alpha):
        for _ in range(iters):
            t += 1
            grad = _grad_smooth_max(h, alpha)  # not multiplied by dx? We'll include after.

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Objective evaluation
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Gradient of a single shift (used for final refinement)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Projected gradient descent on the true max‑correlation objective
# ----------------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 500, step0: float = 0.5,
                 decay: float = 0.995) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# Smoothed max gradient (log‑sum‑exp) using convolution for speed
# ----------------------------------------------------------------------
def _grad_smooth_max(h: np.ndarray, alpha: float) -> np.ndarray:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    # stable softmax weights
    exp_shift = np.exp(alpha * (corr - maxc))
    w = exp_shift / exp_shift.sum()

    # gradient = termA - termB where
    # termA = sum_s w_s * (1 - h_{j+s})
    # termB = sum_s w_s * h_{j-s}
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, 1.0 - h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    base = n - 1
    termA = conv1[base : base + n]
    termB = conv2[base : base + n]
    return termA - termB


# ----------------------------------------------------------------------
# Adam optimizer on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(h0: np.ndarray, target: float,
                   alphas, steps_per_alpha,
                   lr0: float = 0.05) -> np.ndarray:
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    dx = 2.0 / h.shape[0]

    for alpha, iters in zip(alphas, steps_per_alpha):
        for _ in range(iters):
            t += 1
            grad = _grad_smooth_max(h, alpha) * dx
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr0 / (np.sqrt(v_hat) + eps)
            h -= step * m_hat
            h = _project_onto_capped_simplex(h, target)
    return h

    # --------------------------------------------------------------
