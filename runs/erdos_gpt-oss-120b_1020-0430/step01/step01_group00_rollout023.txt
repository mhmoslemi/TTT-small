import time, numpy as np
from scipy import optimize

def _project_to_feasible(h, n, tol=1e-9):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    # adjust sum by water-filling
    # compute diff
    diff = h.sum() - target
    if abs(diff) <= tol:
        return h
    # We'll adjust uniformly among free variables.
    # Use iterative approach limited iterations.
    for _ in range(500):
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to decrease sum: reduce from those >0
            idx = np.where(h > 0)[0]
        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] -= delta
        # clip after change
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
    # final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1.0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0.0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

            continue
        # choose i with largest grad among inc indices
        i = inc_idxs[np.argmax(grad[inc_idxs])]
        # choose j with smallest grad among dec indices
        j = dec_idxs[np.argmin(grad[dec_idxs])]
        if i == j:
            # fallback random
            i, j = np.random.choice(n, 2, replace=False)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            # random jitter
            h = _project_to_feasible(h + 0.01*np.random.randn(n), n)
            continue
        # step size: start bigger, decrease over time
        elapsed = time.time() - start
        frac = elapsed / time_budget
        # step_scale from 0.5 to 0.05 linearly
        step_scale = 0.5 * (1 - frac) + 0.05 * frac
        delta = max_delta * step_scale
        # update
        h[i] += delta
        h[j] -= delta
        # evaluate
        new_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
        # if new max correlation increased, revert and try smaller step
        if new_corr > max_corr:
            # revert
            h[i] -= delta
            h[j] += delta
            # try smaller step
            delta = max_delta * 0.05
            if delta > 1e-12:
                h[i] += delta
                h[j] -= delta
                new_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if new_corr > max_corr:
                    # revert again
                    h[i] -= delta
                    h[j] += delta
                    # random jitter
                    h = _project_to_feasible(h + 0.01*np.random.randn(n), n)
        cur_c5 = new_corr * (2.0 / n)
        if cur_c5 < best_c5 - 1e-9:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0

        if no_improve >= 5000:
            # jitter
            h = _project_to_feasible(best_h + 0.02*np.random.randn(n), n)
            no_improve = 0
        it += 1
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# Global for SLSQP constraints
_CURRENT_N = None


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final fixing
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1−h(x+k)) dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _objective_slsqp(x: np.ndarray) -> float:
    return x[-1]


def _ineq_constraint_slsqp(x: np.ndarray) -> np.ndarray:
    n = _CURRENT_N
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr


def _eq_constraint_slsqp(x: np.ndarray) -> float:
    n = _CURRENT_N
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 200) -> tuple:
    """One shot of SLSQP starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    global _CURRENT_N
    _CURRENT_N = n
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint_slsqp},
        {"type": "eq", "fun": _eq_constraint_slsqp},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective_slsqp, x0, method="SLSQP",
                            bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_value(h_opt)


def _local_opt(initial_h: np.ndarray, time_budget: float) -> tuple:
    """Gradient‑guided local search with occasional random jitter."""
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_value(h)
    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        k = int(corr.argmax() - (n - 1))
        # gradient of the inner product for this shift
        grad = np.zeros(n)
        if k >= 0:
            if k < n:
                grad[: n - k] += h[k:]
                grad[k:] += h[: n - k]
        else:
            kk = -k
            if kk < n:
                grad[kk:] += h[: n - kk]
                grad[: n - kk] += h[kk:]
        # candidate indices
        inc = np.where(h < 1.0 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            # jitter if stuck
            h = _project_to_feasible(h + 0.01 * np.random.randn(n), n)
            continue
        i = inc[np.argmax(grad[inc])]
        j = dec[np.argmin(grad[dec])]
        if i == j:
            # fallback
            i, j = np.random.choice(n, 2, replace=False)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            # jitter
            h = _project_to_feasible(h + 0.01 * np.random.randn(n), n)
            continue
        # step size schedule
        elapsed = time.time() - start
        frac = elapsed / time_budget
        step_scale = 0.5 * (1 - frac) + 0.05 * frac
        delta = max_delta * step_scale
        # apply move
        h[i] += delta
        h[j] -= delta
        new_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
        if new_corr > max_corr + 1e-12:
            # revert and try a tiny step
            h[i] -= delta
            h[j] += delta
            delta = max_delta * 0.05
            if delta > 1e-12:
                h[i] += delta
                h[j] -= delta
                new_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if new_corr > max_corr + 1e-12:
                    h[i] -= delta
                    h[j] += delta
                    # jitter
                    h = _project_to_feasible(h + 0.01 * np.random.randn(n), n)
                    continue
        cur_c5 = new_corr * (2.0 / n)
        if cur_c5 < best_c5 - 1e-9:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 5000:
            # escape plateau
            h = _project_to_feasible(best_h + 0.02 * np.random.randn(n), n)
            no_improve = 0
    return best_h, best_c5
