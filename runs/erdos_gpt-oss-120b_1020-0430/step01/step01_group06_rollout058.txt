def project_onto_bounds_and_sum(y, target):
    # y is unconstrained after gradient step.
    # Find lambda s.t. sum_i clip(y_i + lambda, 0, 1) = target.
    low = -1e6
    high = 1e6
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = lam
        else:
            low = lam
    return np.clip(y + (low + high) / 2.0, 0.0, 1.0)

import numpy as np
import random
import time
import math

def project_onto_bounds_and_sum(y, target):
    # projection onto [0,1] with sum constraint
    low = -np.max(y) - 1.0
    high = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = lam
        else:
            low = lam
    return np.clip(y + (low + high) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    return np.correlate(h, 1 - h, mode='full').max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # problem size
    n = 256  # number of points, < 1000
    dx = 2.0 / n
    target_sum = n / 2.0

    # initialise h
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        m = init_arr.shape[0]
        if m != n:
            x_old = np.linspace(0.0, 2.0, m, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, init_arr)
        else:
            h = init_arr.copy()
    else:
        h = np.random.rand(n)

    h = np.clip(h, 0.0, 1.0)
    h = project_onto_bounds_and_sum(h, target_sum)

    # pre‑compute index pairs for each shift
    shift_idx = []
    for s in range(-(n - 1), n):
        if s >= 0:
            i = np.arange(0, n - s, dtype=int)
            j = i + s
        else:
            s_abs = -s
            i = np.arange(s_abs, n, dtype=int)
            j = i - s_abs
        shift_idx.append((i, j))

    cur_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # annealing / gradient parameters
    T = max(1e-6, cur_c5 * 0.05)
    alpha = 1.0
    lr = 0.1
    it = 0

    while time.time() - start < budget_s * 0.95:
        # correlation and current c5
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c5 = corr.max() * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            # optional early exit if already strong
            if best_c5 <= 0.3808:
                break

        # soft‑max weights approximating the max
        max_corr = corr.max()
        w = np.exp(alpha * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad = np.zeros_like(h)
        for wk, (i, j) in zip(w, shift_idx):
            if wk == 0.0:
                continue
            grad[i] += wk * (1.0 - h[j])
            grad[j] -= wk * h[i]

        # gradient step with projection
        h_new = h - lr * grad
        h_new = project_onto_bounds_and_sum(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)

        # Metropolis acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            h = h_new
            cur_c5 = new_c5

        # occasional targeted move on the worst shift
        if random.random() < 0.1:
            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)
            if s >= 0:
                i = np.arange(0, n - s, dtype=int)
                j = i + s
            else:
                s_abs = -s
                i = np.arange(s_abs, n, dtype=int)
                j = i - s_abs
            contrib = h[i] * (1.0 - h[j])
            k = int(np.argmax(contrib))
            i_best, j_best = i[k], j[k]
            eps = min(h[i_best], 1.0 - h[j_best]) * 0.5
            if eps > 1e-12:
                h2 = h.copy()
                h2[i_best] -= eps
                h2[j_best] += eps
                new_c5_tm = compute_c5(h2, dx)
                if new_c5_tm < cur_c5 or random.random() < math.exp(-(new_c5_tm - cur_c5) / max(T, 1e-12)):
                    h = h2
                    cur_c5 = new_c5_tm
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()

        # update schedules
        it += 1
        if it % 200 == 0:
            T = max(1e-9, T * 0.995)
            lr *= 0.995
            alpha = min(alpha * 1.005, 200.0)

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), float(final_c5), n)
