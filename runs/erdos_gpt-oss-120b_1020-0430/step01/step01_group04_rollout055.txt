def local_search(h, max_iter, time_limit):
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    # schedule: T = initial_temp
    init_temp = 0.001  # adjust
    for it in range(max_iter):
        if time.time() > time_limit: break
        # adapt temperature linearly decreasing
        T = init_temp * (1 - it / max_iter)
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # max transfer from i to j
        max_transfer = min(cur_h[i], 1 - cur_h[j])
        if max_transfer == 0: continue
        # propose delta random up to max_transfer * 0.5 (or *1)
        delta = rng.uniform(0, max_transfer) * 0.5
        # we also consider possibly moving from j to i (reverse) by randomly picking direction
        # Not needed as we choose direction based on which is donor.
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # compute new C5
        new_c5 = compute_c5(new_h)
        # accept if better or with prob
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):

            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
    return best_h, best_c5

def hill_climb(initial_h, time_end):
    cur_h, cur_c5 = initial_h, compute_c5(initial_h)
    best_h, best_c5 = cur_h.copy(), cur_c5
    n = len(cur_h)
    iter_no_improve = 0
    step_counter = 0
    T0 = 0.005
    while time.time() < time_end:
        step_counter += 1
        T = T0 * (0.9995 ** step_counter)   # exponential cooling
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # define donor as i (or j) randomly direction
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(cur_h[donor], 1 - cur_h[receiver])
        if max_transfer <= 1e-12:
            continue
        # randomly choose fraction
        delta = rng.uniform(0, max_transfer) * 0.5   # up to half capacity
        # optionally allow larger delta early
        delta *= 1.0 + (rng.random() - 0.5) * 0.1
        new_h = cur_h.copy()
        new_h[donor] -= delta
        new_h[receiver] += delta
        # compute c5
        new_c5 = compute_c5(new_h)
        # acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5 - 1e-9:
                best_h, best_c5 = cur_h.copy(), cur_c5
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        else:
            iter_no_improve += 1

        # maybe restart if stuck
        if iter_no_improve > 10000:
            # restart from best or random
            cur_h = best_h.copy()
            cur_c5 = best_c5
            iter_no_improve = 0
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core evaluation
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


# ------------------------------------------------------------
# Construction helpers
# ------------------------------------------------------------
def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular hat, already normalized (integral 1)."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)               # peak 1, zero at ends
    # exact integral is already 1, but enforce the sum constraint precisely
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri[0] += diff
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, then re‑normalise."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up[0] += diff
        up = np.clip(up, 0.0, 1.0)
    return up


def random_binary_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exact half‑ones binary feasible vector (requires even n)."""
    if n % 2 != 0:
        raise ValueError("random_binary_start requires even n")
    arr = np.zeros(n, dtype=float)
    arr[: n // 2] = 1.0
    rng.shuffle(arr)
    return arr


def perturb_start(base: np.ndarray, rng: np.random.Generator, scale: float = 0.1) -> np.ndarray:
    """Add Gaussian noise, clip, and renormalise to the integral."""
    n = base.shape[0]
    target = n / 2.0
    pert = base + rng.normal(scale=scale, size=n)
    np.clip(pert, 0.0, 1.0, out=pert)
    pert = pert * (target / pert.sum())
    diff = target - pert.sum()
    if abs(diff) > 1e-12:
        pert[0] += diff
        np.clip(pert, 0.0, 1.0, out=pert)
    return pert


# ------------------------------------------------------------
# Local stochastic optimisation (simulated annealing)
# ------------------------------------------------------------
def hill_climb(start_h: np.ndarray, deadline: float, rng: np.random.Generator):
    """Simple SA that preserves the integral exactly via transfers."""
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5

    # parameters for the annealing schedule
    init_T = 0.004
    it = 0
    stagnant = 0

    while time.time() < deadline:
        it += 1
        # geometric cooling
        T = init_T * (0.9995 ** it)

        # propose a mass transfer between two distinct sites
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # random direction of transfer
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(h[donor], 1.0 - h[receiver])
        if max_transfer <= 0.0:
            continue
        # usually move a modest fraction of what is possible
        delta = rng.random() * max_transfer * 0.5
        if delta <= 0.0:
            continue

        new_h = h.copy()
        new_h[donor] -= delta
        new_h[receiver] += delta

        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            h, cur_c5 = new_h, new_c5
            stagnant = 0
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = h.copy(), cur_c5
        else:
            stagnant += 1

        # occasional restart from the best known point
        if stagnant > 20000:
            h = best_h.copy()
            cur_c5 = best_c5
            stagnant = 0

    return best_h, best_c5


def refine_with_slsqp(h: np.ndarray, deadline: float):
    """Two‑stage SLSQP polish using the smooth max objective."""
    n = h.shape[0]
    target = n / 2.0

    def eq_con(x):
        return np.sum(x) - target

    cons = {"type": "eq", "fun": eq_con}
    bounds = [(0.0, 1.0)] * n

    # stage 1 – smoother objective
    res1 = minimize(
        lambda x: softmax_c5(x, alpha=5.0),
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-9, "disp": False},
    )
    if not res1.success or time.time() > deadline:
        return h

    # stage 2 – sharper approximation
    res2 = minimize(
        lambda x: softmax_c5(x, alpha=30.0),
        res1.x,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-9, "disp": False},
    )
    if not res2.success:
        return h

    opt = np.clip(res2.x, 0.0, 1.0)
    diff = target - opt.sum()
    if abs(diff) > 1e-12:
        opt[0] += diff
        np.clip(opt, 0.0, 1.0, out=opt)
    return opt

# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    # -----------------------------------------------------------------
    # Determine the base dimension; aim for a finer grid (200 points)
    # -----------------------------------------------------------------
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        parent_n = parent.shape[0]
    except NameError:
        parent = None
        parent_n = None

    final_n = 200                       # even, well within the “< 1k” guideline
    target_sum = final_n / 2.0

    # -----------------------------------------------------------------
    # Gather diverse starting points at the final resolution
    # -----------------------------------------------------------------
    starts = []

    if parent is not None:
        if parent_n == final_n:
            starts.append(parent.copy())
        else:
            starts.append(upsample(parent, final_n))

    starts.append(generate_triangular(final_n))
    # a few binary random starts (exact half‑ones)
    for _ in range(3):
        starts.append(random_binary_start(final_n, rng))

    # perturbed copies of the best deterministic start
    for _ in range(2):
        starts.append(perturb_start(starts[0], rng, scale=0.08))

    best_h = None
    best_c5 = np.inf

    # -----------------------------------------------------------------
    # Run a hill‑climbing SA from each start, respecting the global deadline
    # -----------------------------------------------------------------
    for s in starts:
        if time.time() > time_limit:
            break
        h_opt, c5_opt = hill_climb(s, time_limit, rng)
        if c5_opt < best_c5 - 1e-12:
            best_h, best_c5 = h_opt, c5_opt

    # -----------------------------------------------------------------
    # Final polishing with a short SLSQP run if time permits
    # -----------------------------------------------------------------
    if best_h is not None and time.time() < time_limit - 5.0:
        best_h = refine_with_slsqp(best_h, time_limit)

    # -----------------------------------------------------------------
    # Enforce constraints exactly (tiny numeric drift possible)
    # -----------------------------------------------------------------
    if best_h is None:
        # fallback – simple symmetric binary block
        best_h = np.concatenate([np.ones(final_n // 2), np.zeros(final_n - final_n // 2)])
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        best_h[0] += diff
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, final_n
