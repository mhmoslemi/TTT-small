import time, math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-9,
                                 max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Evaluation of the C5 objective (max correlation)
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Correct gradient of a single correlation term (shift may be negative)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # first part: +1 - h[i+s] for i < n-s
        g[: n - s] += 1.0 - h[s:]
        # second part: -h[i-s] for i >= s
        g[s:] += -h[: n - s]
    else:
        s = -shift
        # +1 for indices >= s
        g[s:] += 1.0
        # -h[i-s] for i >= s
        g[s:] += -h[: n - s]
        # -h[i+s] for i < n-s
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the (non‑smooth) max‑correlation
# ----------------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 2000,
                 step0: float = 0.5, decay: float = 0.999) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) utilities for optional SLSQP refinement
# ----------------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float, alpha: float = 30.0,
                    max_iter: int = 800) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)

# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # ----- discretisation size -------------------------------------------------
    try:

    target = n / 2.0          # ∫ h = 1 on interval of length 2  → average 0.5

    # ----- helper to upscale a low‑resolution vector ---------------------------

        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ----- initial candidate ----------------------------------------------------
    if base_h is not None:
        best_h = _upscale(base_h)
        best_h = _project_onto_capped_simplex(best_h, target)
        best_val = _compute_c5(best_h, dx)
    else:

        best_val = float("inf")

    # ----- optimisation loop ----------------------------------------------------
    restart = 0
    while time.time() - start < budget_s * 0.92:

        if val < best_val:

            best_h = h_opt.copy()

        restart += 1

    # ----- final simulated‑annealing polishing ---------------------------------
    if best_h is not None and time.time() - start < budget_s * 0.96:
        cur_h = best_h.copy()
        cur_val = best_val
        sigma = 0.05
        T0 = 0.015
        steps = 8000
        for i in range(steps):
            if time.time() - start > budget_s * 0.98:

            cand = cur_h + rng.normal(scale=sigma, size=n)

            frac = i / steps
            T = T0 * (1.0 - frac)

                if cur_val < best_val:

            sigma *= 0.9998   # slowly cool the proposal size

    if best_h is None:

    best_h = _project_onto_capped_simplex(best_h, target)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
