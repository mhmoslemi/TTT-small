def anneal_binary(n, rng, time_limit):
    half = n//2
    # start from random binary
    h = np.zeros(n)
    h[rng.choice(n, half, replace=False)] = 1
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    # parameters
    T0 = 0.01 # or maybe 0.05
    # total moves we aim for: maybe 500k
    moves = 0
    while time.time() - start < time_limit:
        # compute cooling ratio
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac)  # linear cooling
        # propose swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # perform swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            # probability exp(-delta/T)
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        moves += 1
    return best_h, best_c
```

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 used by the evaluator."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def anneal_binary(start_h, rng, time_budget):
    """
    Simulated‑annealing style local search for a binary (0/1) step function.
    Returns the best vector seen and its C5 value.
    """
    h = start_h.copy()
    n = h.size
    target = n // 2
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c

    start = time.time()
    # temperature parameters (tuned experimentally)
    T0 = 0.02
    Tmin = 0.0005

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # linear cooling
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + Tmin

        # propose a swap of a 1 and a 0
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue

        # perform swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    return best_h, best_c

            idx = np.flatnonzero(h0)
            drop = cur - target
            if drop:
                rm = rng.choice(idx, size=drop, replace=False)
                h0[rm] = 0.0
        elif cur < target:

                h0[add_idx] = 1.0

    else:
