def gradient_for_shift(h, k):
    n = len(h)
    # use np.zeros for shift contributions.
    # compute term1: (1 - h shifted left by -k)
    shifted_forward = np.zeros(n)  # zeros assume out-of-range
    if -k >= 0:
        # shift forward (h[i+k] exists for i from 0..n-k-1)
        # we want array a[i] = h[i+k]
        shifted_forward[:n-k] = h[k:]  # if k >=0, shift forward; careful with sign.
    else:
        # k negative: shift forward means i+k is smaller index
        # we want h[i+k] for i from -k..n-1
        shifted_forward[-k:] = h[:n+k]  # because k negative => n+k < n
    # compute term2: h shifted by +k (i.e., h[i-k])
    shifted_backward = np.zeros(n)
    if k >= 0:
        # i-k >=0 for i >= k => assign
        shifted_backward[k:] = h[:n-k]
    else:
        # k negative => i-k = i+|k|
        shifted_backward[:n+k] = h[-k:]
    # gradient = (1 - shifted_forward) - shifted_backward
    grad = (1.0 - shifted_forward) - shifted_backward
    return grad

def project_onto_constraints(h, target):
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    diff = target - h.sum()
    # if diff is zero within tolerance, done
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # find indices that can accept adjustments in direction of diff
        if diff > 0:
            capable = np.where(h < 1.0)[0]
            if capable.size == 0:
                break
            delta = diff / capable.size
            # limit delta to not exceed capacity
            inc = np.minimum(delta, 1.0 - h[capable])
            h[capable] += inc
            diff = target - h.sum()
        else:
            capable = np.where(h > 0.0)[0]
            if capable.size == 0:
                break
            delta = diff / capable.size  # negative
            dec = np.maximum(delta, -h[capable])  # ensure not go below 0
            h[capable] += dec
            diff = target - h.sum()
    # final clip just in case
    h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h0, rng, max_iters=5000, eta0=0.5):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        # compute correlation vector
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)  # returns index in 0..2n-2
        # compute shift k
        k = idx_max - (n - 1)
        grad = gradient_for_shift(h, k)
        step = eta0 / np.sqrt(it + 1)
        h_new = h + step * (-grad)  # move opposite to gradient to reduce the term
        # project
        h_new = project_onto_constraints(h_new, target)
        c = compute_c5(h_new)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def simulated_annealing(h0, rng, max_iters, init_T, final_T):
    n = len(h0)
    target = n/2
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    # precompute step scale? We'll use random epsilon from uniform [-scale,scale] limited by bounds.
    for it in range(max_iters):
        # temperature schedule
        T = init_T * ((final_T/init_T) ** (it / max_iters))
        # propose move: select i, j distinct, compute max epsilon allowed.
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # max epsilon = min(h[i], 1-h[j])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # sample epsilon within [-max_eps, max_eps] scaled by factor maybe typical magnitude ~ max_eps * 0.5?
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.5
        # apply move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # compute C5

                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # simple proportional adjustment
        mask = (diff > 0) * (h < 1.0) + (diff < 0) * (h > 0.0)  # boolean
        # compute per-element capacity
        if diff > 0:
            capacities = 1.0 - h
        else:
            capacities = -h
        total_cap = capacities[mask].sum()
        if total_cap > 0:
            # allocate proportionally
            add = np.zeros_like(h)
            add[mask] = capacities[mask] * diff / total_cap
            h += add
        # after adjustment maybe still diff small, can do loop fallback
        # final clip
        h = np.clip(h, 0.0, 1.0)
        # final difference adjust using random redistribution loop as earlier
        diff2 = target - h.sum()
        # loop fallback
        for _ in range(1000):
            if abs(diff2) < 1e-12:
                break
            i = rng.integers(0, len(h))
            if diff2 > 0:
                inc = min(diff2, 1.0 - h[i])
                h[i] += inc
            else:
                dec = min(-diff2, h[i])
                h[i] -= dec
            diff2 = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Create a random feasible vector with values in [0,1] and sum = n/2."""
    # start from a binary vector (exact half ones)
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    # add a few small fractional transfers to break symmetry
    for _ in range(2 * n):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.1
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb: keep changes only if they improve the bound."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        # shrink step periodically
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h0, rng, max_iters=40000, init_T=0.1, final_T=1e-6):
    """Simple SA that moves mass between two cells, accepting uphill moves with prob."""
    n = h0.size
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    for it in range(max_iters):
        T = init_T * ((final_T / init_T) ** (it / max_iters))
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        accept = False
        if c_new < c_curr:
            accept = True
        else:
            # Metropolis criterion
            if T > 0 and rng.random() < np.exp(-(c_new - c_curr) / T):
                accept = True
        if accept:
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c


# ----- SLSQP with smooth max (soft‑max) -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (tiny drift)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_for_shift(h, k):
    """Gradient of the worst‐shift term w.r.t. h."""
    n = h.size
    shifted_forward = np.zeros(n)
    if k >= 0:
        shifted_forward[: n - k] = h[k:]
    else:
        shifted_forward[-k :] = h[: n + k]
    shifted_backward = np.zeros(n)
    if k >= 0:
        shifted_backward[k:] = h[: n - k]
    else:
        shifted_backward[: n + k] = h[-k:]
    return (1.0 - shifted_forward) - shifted_backward


def project_feasible(h, target, rng):
    """Clip to [0,1] and repair the sum exactly."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
        else:
            dec = min(-diff, h[i])
            h[i] -= dec
        diff = target - h.sum()
    return h


def subgradient_descent(h0, rng, max_iters=3000, eta0=0.5):
    """Improve a vector by moving opposite to the gradient of the worst shift."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        grad = gradient_for_shift(h, k)
        step = eta0 / np.sqrt(it + 1)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target, rng)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
        h = h_new
    return best_h, best_c

    return best_h.tolist(), float(final_c), int(n_final)
