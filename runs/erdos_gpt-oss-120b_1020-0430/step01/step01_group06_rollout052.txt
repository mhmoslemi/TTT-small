def project_onto_capped_simplex(v, target):
    lo = -np.max(v)  # ensure v+lo <=0 => all clipped to 0 -> sum=0
    hi = 1 - np.min(v)  # ensure v+hi >=1 => all clipped to 1 -> sum=n
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

    T = cur_c5 * 0.02 + 1e-6
    T_decay = 0.99996
    while time.time() - start < budget_s * 0.95:
        # --------------------------------------------------------------
        # Identify the worst shift (maximising the objective)
        # --------------------------------------------------------------
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(corr_full.argmax())
        shift = idx_max - (n - 1)          # actual lag

        # --------------------------------------------------------------
        # Sub‑gradient of the active term L_shift
        # --------------------------------------------------------------
        g = np.ones_like(h)
        if shift >= 0:
            k = shift
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        else:
            k = -shift
            g[k:] -= h[: n - k]
            g[: n - k] -= h[k:]

        # --------------------------------------------------------------
        # Gradient step + projection
        # --------------------------------------------------------------
        step = 0.22 / math.sqrt(it + 1.0)   # diminishing step size
        h_candidate = h - step * g
        # project back onto feasible set
        h_candidate = proj_sum_to_target(h_candidate, target_sum)

        new_c5 = c5_val(h_candidate)

        # --------------------------------------------------------------
        # Acceptance rule (mostly greedy, occasional uphill)
        # --------------------------------------------------------------
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            if T > 1e-12 and random.random() < math.exp(-(new_c5 - cur_c5) / T):
                accept = True

        if accept:
            h = h_candidate
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

        # --------------------------------------------------------------
        # Periodic random two‑coordinate exchange (helps escape plateaus)
        # --------------------------------------------------------------
        if it % 800 == 0:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            max_up = min(1.0 - hi, hj)   # increase i, decrease j
            max_down = min(hi, 1.0 - hj) # decrease i, increase j
            if max_up + max_down > 0:
                delta = (random.random() * (max_up + max_down)) - max_down
                if abs(delta) > 1e-14:
                    nh = h.copy()
                    nh[i] = hi + delta
                    nh[j] = hj - delta
                    new_c5_sw = c5_val(nh)
                    if new_c5_sw < cur_c5 or (T > 1e-12 and random.random() <
                                             math.exp(-(new_c5_sw - cur_c5) / T)):
                        h = nh
                        cur_c5 = new_c5_sw
                        if new_c5_sw < best_c5:
                            best_c5 = new_c5_sw
                            best_h = h.copy()

        # --------------------------------------------------------------
        # Targeted tiny transfer along the worst shift (refinement)
        # --------------------------------------------------------------
        if it % 1200 == 0:
            # recompute worst shift after possible moves
            corr_full = np.correlate(h, 1.0 - h, mode="full")
            idx_max = int(corr_full.argmax())
            shift = idx_max - (n - 1)
            if shift >= 0:
                i_range = np.arange(n - shift)
                j_range = i_range + shift
            else:
                k = -shift
                i_range = np.arange(k, n)
                j_range = i_range - k
            contrib = h[i_range] * (1.0 - h[j_range])
            if contrib.size:
                worst_idx = int(np.argmax(contrib))
                i_best = i_range[worst_idx]
                j_best = j_range[worst_idx]
                eps_max = min(h[i_best], 1.0 - h[j_best])
                if eps_max > 1e-12:
                    eps = eps_max * 0.5
                    nh = h.copy()
                    nh[i_best] -= eps
                    nh[j_best] += eps
                    new_c5_t = c5_val(nh)
                    if new_c5_t < cur_c5:
                        h = nh
                        cur_c5 = new_c5_t
                        if new_c5_t < best_c5:
                            best_c5 = new_c5_t
                            best_h = h.copy()

        # --------------------------------------------------------------
        # Cooling & iteration counter
        # --------------------------------------------------------------
        T *= T_decay
        it += 1

    # ------------------------------------------------------------------
    # Final safety projection (tiny drift correction)
    # ------------------------------------------------------------------
    best_h = proj_sum_to_target(best_h, target_sum)
    final_c5 = c5_val(best_h)

    return (best_h.tolist(), final_c5, n)
