def _solve_h(initial_h, max_iter=250):
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # inequality: t - corr >= 0 for each lag
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode='full')
    # equality sum constraint
    def eq_fun(x):
        return np.sum(x[:n]) - (n / 2.0)

    constraints = [{'type': 'ineq', 'fun': ineq_fun},
                   {'type': 'eq', 'fun': eq_fun}]

                            bounds=bounds,
                            constraints=constraints,
                            options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5

import time
import numpy as np
from scipy import optimize


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Simple iterative scaling
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = h > 0
        else:
            free = h < 1
        if not np.any(free):
            break
        delta = diff / free.sum()
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        for i in range(n):
            if residual > 0 and h[i] < 1:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible h (∫h = 1)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """Local optimisation of C5 using SLSQP."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_f, x_c, coarse_h)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1
    that minimises  C5 = max_k ∫ h(x)(1-h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Load optional parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target = 0.3808

    # ------------------------------------------------------------------
    # Choose grid sizes
    n_set = set()
    if init_h is not None:
        n_set.add(init_h.size)
        base_n = init_h.size
    else:
        base_n = 64

    for m in (1, 2, 3, 4):
        n = base_n * m
        if n <= 300:
            n_set.add(int(n))
    for n in (80, 96, 108, 120, 128, 144, 160, 176, 192, 200, 224, 240, 256, 280, 300):
        if n <= 300:
            n_set.add(n)

    n_candidates = sorted(n_set)

    # ------------------------------------------------------------------
    # Scan simple symmetric three‑level patterns (edges = 1, middle = p)
    for n in n_candidates:
        if time_left() < 2.0:
            break
        max_e = n // 4
        for e in range(max_e + 1):
            denom = n - 2 * e
            if denom == 0:
                continue
            p = (n / 2.0 - 2 * e) / denom
            if not (0.0 <= p <= 1.0):
                continue
            h = np.full(n, p, dtype=float)
            if e:
                h[:e] = 1.0
                h[-e:] = 1.0
            c5 = _c5_from_h(h)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n
                if best_c5 < target:
                    break
        if best_c5 < target:
            break

    # ------------------------------------------------------------------
    # Include the provided parent construction if it improves the bound
    if init_h is not None:
        c5_par = _c5_from_h(init_h)
        if c5_par < best_c5:
            best_h, best_c5, best_n = init_h.copy(), c5_par, init_h.size

    # ------------------------------------------------------------------
    # Local refinement with SLSQP and stochastic restarts
    if best_h is not None:
        # First local optimisation
        h_opt, c5_opt = _solve_h(best_h, max_iter=400)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

        # Random restarts
        sigma = 0.2
        for _ in range(8):
            if time_left() < 2.0:
                break
            perturb = sigma * np.random.randn(best_n)
            start_h = best_h + perturb
            start_h = _project_to_feasible(start_h, best_n)
            h_opt, c5_opt = _solve_h(start_h, max_iter=300)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
            sigma *= 0.85

        # Simple hill‑climbing moves
        steps = 500
        for _ in range(steps):
            if time_left() < 1.0:
                break
            i, j = np.random.randint(0, best_n, size=2)
            if i == j:
                continue
            delta = (np.random.rand() - 0.5) * 0.05
            hi, hj = best_h[i], best_h[j]
            new_hi, new_hj = hi - delta, hj + delta
            if not (0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0):
                continue
            best_h[i], best_h[j] = new_hi, new_hj
            c5_new = _c5_from_h(best_h)
            if c5_new < best_c5:
                best_c5 = c5_new
            else:
                best_h[i], best_h[j] = hi, hj  # revert

    # ------------------------------------------------------------------
    # Fallback to a uniform construction if nothing succeeded
    if best_h is None:
        best_n = 64
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(best_c5), int(best_n)
