def binary_hill_climb(h_bin, rng, max_iters=5000):
    best = h_bin.copy()
    best_max = compute_c5(best)

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def project_box_simplex(v, target, iters=50):
    """
    Projection of v onto the intersection of the box [0,1]^n
    and the hyperplane sum(x)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = v.min() - 1.0   # low enough so that clipping gives all 1's
    hi = v.max()         # high enough so that clipping gives all 0's
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    v = rng.random(n)
    return project_box_simplex(v, n / 2.0)


def thue_morse(n):
    """Balanced Thue‑Morse sequence (0/1) of length n (good for powers of two)."""
    # property: for n a power of two the sequence contains exactly n/2 ones.
    return np.array([i.bit_count() & 1 for i in range(n)], dtype=float)

def grad_descent_time(h0, rng, max_seconds, n):
    """
    Gradient descent on the smooth log‑sum‑exp surrogate.
    Returns the best vector found within max_seconds seconds.
    """
    h = h0.copy()
    target = n / 2.0
    best = h.copy()
    best_c = compute_c5(h)

    beta = 5.0          # start β for the soft‑max
    lr = 0.5            # initial step size
    start = time.time()
    offset = n - 1
    while time.time() - start < max_seconds:
        # correlation and soft‑max weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        bc = beta * corr
        m = bc.max()
        exp_bc = np.exp(bc - m)
        w = exp_bc / exp_bc.sum()   # weights sum to 1

        # gradient of the surrogate: 1 - Σ_s w_s (h_{i+s}+h_{i‑s})
        grad = np.ones(n)
        for s in range(-(n - 1), n):
            ws = w[s + offset]
            if ws == 0.0:
                continue
            # term h_{i+s}
            if s >= 0:
                grad[: n - s] -= ws * h[s:]
            else:
                grad[-s :]    -= ws * h[: n + s]
            # term h_{i‑s}

                grad[: n - s2] -= ws * h[s2:]

        c = compute_c5(h)

            best = h.copy()

        # schedule
        beta = min(beta * 1.02, 500.0)
        lr = max(lr * 0.999, 1e-5)

    return best, best_c

        elif up.size < n:
            up = np.concatenate([up, np.zeros(n - up.size)])
        init_vectors.append(project_box_simplex(up, n / 2.0))

    best_h = None
    best_c = float("inf")

    # -----------------------------------------------------------------
    # Run gradient‑descent refinements from each starter
    # -----------------------------------------------------------------
    for h0 in init_vectors:
        if time.time() - start_time > budget_s * 0.80:
            break
        # allocate a slice of the remaining budget for this run
        remaining = budget_s - (time.time() - start_time)
        alloc = min(8.0, remaining * 0.6)   # at most 8 s or 60 % of what's left
        h_opt, c_opt = grad_descent_time(h0, rng, alloc, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # -----------------------------------------------------------------
    # Optional binary hill‑climb on a rounded version
    # -----------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        h_bin = binarize(best_h)
        h_bin_opt, c_bin = binary_hill_climb(h_bin, rng, max_iters=15000)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin_opt.astype(float)

    # -----------------------------------------------------------------
    # Final feasibility cleanup
    # -----------------------------------------------------------------
    if best_h is None:
        # fallback – a random feasible vector
        best_h = random_feasible(n, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_simplex(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
