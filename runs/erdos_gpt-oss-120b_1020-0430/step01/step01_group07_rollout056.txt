        progress = elapsed / time_limit
        T = T0 * (T_end / T0) ** progress

        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)

        if max_step > 0:
            low_cap = max(low, -max_step)
            high_cap = min(high, max_step)
            if low_cap < high_cap:
                delta = random.uniform(low_cap, high_cap)
            else:
                delta = random.uniform(low, high)

            delta = random.uniform(low, high)

        h[j] = hj - delta

        else:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        # distribute to a random admissible entry
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h

def _anneal(start_h, max_seconds):
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c
    start = time.time()
    T0 = 0.08
    T_end = 1e-6
    max_step = 0.25
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        # choose indices: bias towards extremes
        # simple random pair
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)   # amount that can be moved from i to j (negative delta)
        high = min(1.0 - hi, hj)   # amount that can be moved from j to i (positive delta)
        if low >= high:
            continue
        # cap delta magnitude
        if max_step > 0:
            low_cap = max(low, -max_step)
            high_cap = min(high, max_step)
            if low_cap < high_cap:
                delta = random.uniform(low_cap, high_cap)
            else:
                delta = random.uniform(low, high)
        else:
            delta = random.uniform(low, high)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c

def _greedy_refine(start_h, time_budget):
    h = start_h.copy()
    n = h.shape[0]
    best_c = _c5_metric(h)
    start = time.time()
    step = 0.2
    no_improve = 0
    while time.time() - start < time_budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx] * (2.0 / n)
        # shift relative to center
        shift = max_idx - (n - 1)
        # compute weight w for each index as (1 - h[i+shift]) when overlapping
        if shift >= 0:
            w = np.zeros(n)
            w[:n-shift] = 1.0 - h[shift:]
        else:
            s = -shift
            w = np.zeros(n)
            w[s:] = 1.0 - h[:n-s]
        contrib = h * w
        # choose i (decrease) and j (increase)
        pos_i = np.where(h > 1e-12)[0]
        pos_j = np.where(h < 1.0 - 1e-12)[0]
        if pos_i.size == 0 or pos_j.size == 0:
            break
        i = pos_i[np.argmax(contrib[pos_i])]
        j = pos_j[np.argmin(contrib[pos_j])]
        if i == j:
            # random fallback
            i = np.random.choice(pos_i)
            j = np.random.choice(pos_j)
        delta = min(step, h[i], 1.0 - h[j])
        if delta <= 0:
            step *= 0.5
            continue
        h[i] -= delta
        h[j] += delta
        new_c = _c5_metric(h)
        if new_c + 1e-9 < best_c:
            best_c = new_c
            no_improve = 0
        else:
            # revert
            h[i] += delta
            h[j] -= delta
            no_improve += 1
        if no_improve >= 300:
            step *= 0.7
            no_improve = 0
            if step < 1e-5:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    total_budget = float(budget_s)
    start_all = time.time()
    # choose number of points
    n = 400
    # initialize vector
    h0 = _make_initial(n)
    # Phase 1 - primary annealing
    primary_time = total_budget * 0.70

    # Phase 1 - random restarts
    remaining = total_budget - (time.time() - start_all)
    while remaining > 12.0:

        ones = int(target)
        frac = target - ones
        h_rand = np.zeros(n, dtype=np.float64)
        idx = np.random.permutation(n)
        h_rand[idx[:ones]] = 1.0
        if frac > 1e-12:
            h_rand[idx[ones]] = frac
        slice_time = min(remaining * 0.25, 30.0)
        cand_h, cand_c = _anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        remaining = total_budget - (time.time() - start_all)
    # Phase 2 - greedy refinement
    remaining = total_budget - (time.time() - start_all)
    if remaining > 5.0:
        best_h, best_c = _greedy_refine(best_h, remaining)
    # final enforcement of exact sum
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # simple correction by adding diff to a random free entry
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= best_h[idx] + diff <= hi:
                best_h[idx] += diff
                break
    # compute final bound using helper if present
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

def _c5_metric(h):
    """C5 value for the sampled step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _make_initial(n):

    if "initial_h_values" in globals():
