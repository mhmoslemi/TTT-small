def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ---------- initialise h ----------
    try:
        # use the provided construction, upscale for more flexibility
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = h0.shape[0]
        n = n0 * 2                       # double the resolution
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x, x0, h0)         # linear interpolation
    except NameError:
        n = 140
        h = np.linspace(0, 1, n)         # simple linear profile

    target = n / 2.0                     # required integral = 1 ⇒ sum = n/2

    # ---------- projection onto the feasible set ----------
    def project(v):
        lo, hi = -2.0, 2.0                # enough to drive all entries to 0/1
        for _ in range(50):
            lam = (lo + hi) * 0.5
            p = np.clip(v - lam, 0.0, 1.0)
            s = p.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    h = project(h)

    # ---------- objective ----------
    def c5_of(v):
        corr = np.correlate(v, 1 - v, mode="full")
        return float(corr.max() * (2.0 / len(v)))

    best_h = h.copy()
    best_c5 = c5_of(best_h)

    # ---------- optimisation parameters ----------
    tau = 1.0
    tau_decay = 0.9995
    step = 0.5
    step_decay = 0.9995
    it = 0
    no_improve = 0

    # ---------- main loop ----------
    while True:
        if time.time() - start > budget_s * 0.95:
            break

        # correlation (max‑overlap values)
        I = np.correlate(h, 1 - h, mode="full")
        Imax = I.max()
        # smooth‑max weights
        expv = np.exp((I - Imax) / max(tau, 1e-12))
        p = expv / expv.sum()

        # gradient of the smooth‑max (minimise)
        grad = np.zeros_like(h)
        idx = 0
        for s in range(-(n - 1), n):
            w = p[idx]
            if s == 0:
                grad += w * (1 - 2 * h)
            elif s > 0:
                L = n - s
                grad[0:L]       += w * (1 - h[s:n])
                grad[s:n]       -= w * h[0:L]
            else:
                sp = -s
                L = n - sp
                grad[sp:n]      += w * (1 - h[0:L])
                grad[0:L]       -= w * h[sp:n]
            idx += 1

        # gradient step and projection
        h_new = project(h - step * grad)
        new_c5 = c5_of(h_new)

        # acceptance (always keep best, otherwise SA‑like uphill allowance)
        if new_c5 < best_c5 or random.random() < math.exp(-(new_c5 - best_c5) / max(1e-12, tau)):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
        else:
            no_improve += 1

        # targeted local move every 200 iterations
        if it % 200 == 0:
            Ifull = np.correlate(h, 1 - h, mode="full")
            imax = int(np.argmax(Ifull))
            s = imax - (n - 1)
            if s >= 0:
                i_idx = np.arange(n - s)
                j_idx = i_idx + s
            else:
                sp = -s
                i_idx = np.arange(sp, n)
                j_idx = i_idx - sp
            contrib = h[i_idx] * (1 - h[j_idx])
            k = int(np.argmax(contrib))
            i_best, j_best = i_idx[k], j_idx[k]
            eps = min(h[i_best], 1 - h[j_best]) * 0.5
            if eps > 1e-12:
                h2 = h.copy()
                h2[i_best] -= eps
                h2[j_best] += eps
                h2 = project(h2)
                c5_t = c5_of(h2)
                if c5_t < best_c5:
                    h = h2
                    best_c5 = c5_t
                    best_h = h.copy()
                    no_improve = 0

        # occasional random perturbation if stuck
        if no_improve > 3000:
            h_noisy = h + (np.random.rand(n) - 0.5) * 0.05
            h = project(np.clip(h_noisy, 0.0, 1.0))
            no_improve = 0

        # schedule decay
        tau = max(tau * tau_decay, 1e-6)
        step = max(step * step_decay, 1e-5)
        it += 1

    # final tiny correction of the total mass
    err = best_h.sum() - target
    if abs(err) > 1e-9:
        for _ in range(10):
            k = random.randrange(n)
            delta = -err
            if 0 <= best_h[k] + delta <= 1:
                best_h[k] += delta
                break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
