def project_onto_bounds_sum(y, target, eps=1e-9, max_iter=100):
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_max_shift(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    g = np.ones_like(h)  # start 1
    if shift > 0:
        # j+shift inside for j=0..n-shift-1 -> subtract h[j+shift]; j-shift inside for j=shift..n-1 -> subtract h[j-shift]
        g[:n-shift] -= h[shift:]      # subtract h_{j+shift}
        g[shift:]   -= h[:n-shift]    # subtract h_{j-shift}
    elif shift < 0:
        s = -shift
        g[s:]     -= h[:n-s]          # subtract h_{j+shift}=h_{j - s}
        g[:n-s]   -= h[s:]           # subtract h_{j-shift}=h_{j + s}
    else: # shift == 0
        g -= 2.0 * h  # since 1 - h_{j} - h_{j} = 1 - 2h_j
    return g, idx, shift, np.max(corr)
```

def project_onto_bounds_sum(y, target, max_iter=100):
    x = y.copy()
    # Clip to bounds initially
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # indices that are not at bounds in direction of diff
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        # distribute equally
        delta = diff / cnt
        # add delta but stay within bounds
        add = np.where(mask, np.clip(delta, -np.inf, np.inf), 0.0)
        # adjust values with clipping
        x[mask] += delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    # If still not correct, fallback to binary search
    if abs(diff) > 1e-8:
        # binary search low/high as before
        lo = np.min(x) - 1.0
        hi = np.max(x) + 1.0
        for _ in range(100):

            if abs(s - target) < 1e-12:

            if s > target:
                lo = mid
            else:
                hi = mid
        x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

# ----- helpers ---------------------------------------------------------------

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def max_shift_and_gradient(h):
    """
    Return gradient of the active shift (the one attaining the maximum),
    together with the corresponding correlation value and the shift amount.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    max_corr = corr[idx]

    g = np.ones_like(h)

    if shift > 0:
        # 1 - h_{j+shift} - h_{j-shift}
        g[: n - shift] -= h[shift:]
        g[shift:] -= h[: n - shift]
    elif shift < 0:
        k = -shift
        g[k:] -= h[: n - k]          # h_{j-shift}
        g[: n - k] -= h[k:]          # h_{j+shift}
    else:  # shift == 0
        g -= 2.0 * h

    return g, max_corr, shift


def project_onto_bounds_sum(y, target, eps=1e-12, max_iter=60):
    """
    Euclidean projection of y onto the set {x | 0<=x_i<=1, sum x = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def upsample_vector(v, n_target):
    """Linear interpolation of a vector v to length n_target."""
    n0 = v.size
    if n_target == n0:
        return v.copy()
    x0 = np.linspace(0.0, 1.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 1.0, n_target, endpoint=False)
    v_up = np.interp(x1, x0, v)
    return v_up


def subgradient_descent(
    h0, target, max_iters, step0, decay, rng, time_limit
):
    """
    Simple projected sub‑gradient descent on the max‑overlap objective.
    Returns the best point seen inside the allotted time.
    """
    h = h0.copy()
    n = h.size
    step = step0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        g, max_corr, _ = max_shift_and_gradient(h)
        c = max_corr * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

        # gradient step + projection
        y = h - step * g
        h = project_onto_bounds_sum(y, target)

        step *= decay

        # occasional tiny random jitter to escape flat regions
        if (it + 1) % 300 == 0:
            jitter = rng.uniform(-0.005, 0.005, size=n)
            h = project_onto_bounds_sum(h + jitter, target)

    # final evaluation
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c


def hill_climb_refine(h0, target, rng, max_iters, start_step=0.3):
    """
    Simple deterministic hill‑climb that moves a small ε from one coordinate
    to another (preserving the sum). Accepts only improving moves.
    """
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size

    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # slowly shrink the step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c
