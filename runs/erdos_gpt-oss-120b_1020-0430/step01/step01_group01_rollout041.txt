import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _smooth_max_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _smooth_max_jac(v, alpha):
    n = v.size
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    exp_terms = np.exp(alpha * (corr - m))
    w = exp_terms / np.sum(exp_terms)
    g = np.zeros_like(v)
    # weighted sum of shift gradients
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        g += wi * _gradient_for_shift(v, shift)
    return g

def _equality_constraint(v, target):
    return np.sum(v) - target

    rng = np.random.RandomState(seed)
    start = time.time()
    # Determine grid size
    try:
        base_arr = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]

        base_arr = None
        base_n = 0
    # Target resolution (increase modestly)
    n = max(250, base_n)
    target_sum = n / 2.0
    dx = 2.0 / n

    # Prepare initial candidate
    if base_arr is not None:
        # Interpolate to new size
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, base_arr)
        h0 = _project_onto_capped_simplex(h0, target_sum)
        best_h = h0.copy()
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = _random_feasible(rng, n, target_sum)
        best_val = _compute_c5(best_h, dx)

    # Bounds for SLSQP
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _equality_constraint, 'args': (target_sum,)},)

    # Stage 1: continuation SLSQP with increasing smoothing
    alphas = [12.0, 30.0, 60.0]

                           bounds=bounds,

            val = _compute_c5(h_opt, dx)

import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    """Project v onto {0 ≤ x ≤ 1, sum(x)=target} by bisection on a dual variable."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng, n, target):
    """Generate a random vector and project it onto the feasible simplex."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h, dx):
    """Return max_x ∫ h·(1−h(·+x))."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _gradient_for_shift(h, shift):
    """Derivative of a single correlation entry w.r.t. h."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _smooth_max_obj(v, alpha):
    """Smooth approximation of max via log‑sum‑exp."""
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def _smooth_max_jac(v, alpha):
    """Gradient of the smooth max."""
    n = v.size
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    exp_terms = np.exp(alpha * (corr - m))
    w = exp_terms / np.sum(exp_terms)          # softmax weights, sum to 1
    g = np.zeros_like(v)
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        g += wi * _gradient_for_shift(v, shift)
    return g


def _equality_constraint(v, target):
    """Linear equality ∑v = target."""
    return np.sum(v) - target
