def project_to_box_and_sum(h, target):
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Determine indices that can adjust in direction of diff
    if diff > 0:
        # need to increase some entries that are less than 1
        free = np.where(h < 1 - 1e-12)[0]
        # allocate diff proportionally? we can loop.
        while diff > 1e-12 and free.size > 0:
            # distribute equally
            inc = diff / free.size
            # ensure not exceed 1
            # For each idx, inc_i = min(inc, 1 - h[idx])
            # handle caps.
            # Use vectorized with np.minimum
            caps = 1 - h[free]
            added = np.minimum(inc, caps)
            h[free] += added
            diff = target - h.sum()
            # update free where h reaches capacity
            free = free[h[free] < 1 - 1e-12]
        # after loop there may be rounding diff small, ignore
    else:
        # diff < 0: need to decrease some entries >0
        free = np.where(h > 1e-12)[0]
        while diff < -1e-12 and free.size > 0:
            dec = -diff / free.size
            caps = h[free]  # max we can subtract
            sub = np.minimum(dec, caps)
            h[free] -= sub
            diff = target - h.sum()
            free = free[h[free] > 1e-12]
    return h

    h = h0.copy()
    best_h = h.copy()
    best_val = compute_C5(h)  # max correlation with 1-h
    n = len(h)
    for t in range(1, max_iters+1):
        # compute correlation and find max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        # if improvement:
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # shift k:
        k = max_idx - (n - 1)
        # compute gradient:
        g = np.zeros_like(h)
        idx = np.arange(n)
        shift1 = idx + k
        mask1 = (shift1 >= 0) & (shift1 < n)
        g[idx[mask1]] += 1 - h[shift1[mask1]]
        shift2 = idx - k
        mask2 = (shift2 >= 0) & (shift2 < n)
        g[idx[mask2]] -= h[shift2[mask2]]
        # subgradient descent: step size
        step = step0 / (1 + t * 1e-3)   # or step0 * (1 - t / max_iters)
        # update
        h = h - step * g
        # project
        h = project_feasible(h, target_sum)
    return best_h, best_val

def subgradient_opt(h0, target, max_iters, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c = evaluate_c5(h)
    n = len(h)
    for it in range(1, max_iters+1):
        # compute corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        cur_c = corr[idx_max] * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # compute lag
        k = idx_max - (n - 1)
        # gradient
        g = np.zeros_like(h)
        idx = np.arange(n)
        shift1 = idx + k
        mask1 = (shift1 >= 0) & (shift1 < n)
        g[idx[mask1]] += 1.0 - h[shift1[mask1]]
        shift2 = idx - k
        mask2 = (shift2 >= 0) & (shift2 < n)
        g[idx[mask2]] -= h[shift2[mask2]]
        # step size schedule (decreasing)
        # we can use step = 0.5 / math.sqrt(it) maybe.
        step = 0.25 / math.sqrt(it)
        # update
        h = h - step * g
        # project to feasible set
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # water-filling
            if diff > 0:
                free = np.where(h < 1.0 - 1e-12)[0]
                # if nothing free, break
                if free.size == 0:
                    pass
                else:
                    inc = diff / free.size
                    # compute capacity left
                    cap = 1.0 - h[free]
                    add = np.minimum(inc, cap)
                    h[free] += add
                    # recompute diff may still positive due to cap limit, iterate
                    # We'll loop
                    while True:
                        diff = target - h.sum()
                        if abs(diff) < 1e-12:
                            break
                        free = np.where(h < 1.0 - 1e-12)[0]
                        if free.size == 0:
                            break
                        inc = diff / free.size
                        cap = 1.0 - h[free]
                        add = np.minimum(inc, cap)
                        h[free] += add
            else:
                # diff <0
                free = np.where(h > 1e-12)[0]
                if free.size > 0:
                    dec = -diff / free.size
                    cap = h[free]
                    sub = np.minimum(dec, cap)
                    h[free] -= sub
                    while True:
                        diff = target - h.sum()
                        if abs(diff) < 1e-12:
                            break

                        if free.size == 0:
                            break
                        dec = -diff / free.size
                        cap = h[free]
                        sub = np.minimum(dec, cap)
                        h[free] -= sub
        # stop early if time exceeded
        if time.time() - start_time > time_limit:
            break
    return best_h, best_c

def local_refine(h, target, steps=5000):
    h = h.copy()
    n = len(h)
    best = h.copy()
    best_val = evaluate_c5(h)
    for _ in range(steps):
        i, j = random.sample(range(n), 2)
        # we consider transferring small delta
        # define delta range limited
        low = max(-h[i], h[j] - 1.0)
        high = min(1.0 - h[i], h[j])
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.1
        h[i] += delta
        h[j] -= delta
        # ensure box
        # compute val
        val = evaluate_c5(h)
        if val < best_val:
            best_val = val
            best = h.copy()
        else:
            # rollback
            h[i] -= delta
            h[j] += delta
    return best, best_val

def _eval_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _project_feasible(h, target):
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # water-filling
    if diff > 0:
        # increase
        free = np.where(h < 1.0)[0]
        while diff > 1e-12 and free.size:
            inc = diff / free.size
            caps = 1.0 - h[free]
            add = np.minimum(inc, caps)
            h[free] += add
            diff = target - h.sum()
            free = free[h[free] < 1.0 - 1e-12]
    else:
        # decrease
        free = np.where(h > 0.0)[0]
        while diff < -1e-12 and free.size:
            dec = -diff / free.size
            caps = h[free]
            sub = np.minimum(dec, caps)
            h[free] -= sub
            diff = target - h.sum()
            free = free[h[free] > 1e-12]

import time, math, random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """C_5 = max_k ∫ h(x)(1‑h(x+k)) dx  (discretised)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_feasible(h: np.ndarray, target: float) -> None:
    """Clip to [0,1] and enforce Σh = target exactly (in‑place)."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return

    if diff > 0.0:                     # need to increase some entries
        free = np.where(h < 1.0)[0]
        while diff > 1e-12 and free.size:
            inc = diff / free.size
            cap = 1.0 - h[free]
            add = np.minimum(inc, cap)
            h[free] += add
            diff = target - h.sum()
            free = free[h[free] < 1.0 - 1e-12]
    else:                               # need to decrease some entries
        free = np.where(h > 0.0)[0]
        while diff < -1e-12 and free.size:
            dec = -diff / free.size
            cap = h[free]
            sub = np.minimum(dec, cap)
            h[free] -= sub
            diff = target - h.sum()
            free = free[h[free] > 1e-12]


def _initial_vector(n: int) -> np.ndarray:
    """Resize the supplied parent construction (if any) to n points."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        n0 = base.size

        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, base)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    if h.sum() != 0.0:
        h *= target / h.sum()
    _project_feasible(h, target)
    return h

    return h


def _subgradient_opt(
    h0: np.ndarray, target: float, max_iters: int, max_time: float
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on   max_k ∫ h(1‑h_k) .
    Stops after max_iters or when max_time seconds have elapsed.
    Returns the best vector seen and its C5 value.
    """
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_value(h)
    idx = np.arange(n)

    for it in range(1, max_iters + 1):
        if time.time() - start > max_time:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(corr))
        cur_c = corr[k_idx] * (2.0 / n)
