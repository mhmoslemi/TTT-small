import time, random, numpy as np
from scipy.optimize import differential_evolution

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # problem size
    n = 250                               # total samples
    dx = 2.0 / n
    target_sum = n / 2.0                  # integral constraint

    # ----- helpers -------------------------------------------------
    def compute_c5(h: np.ndarray) -> float:
        # linear correlation of h with (1-h)
        corr = np.convolve(h, (1.0 - h)[::-1])
        return np.max(corr) * dx

    # ----- block representation ------------------------------------
    block_size = 5
    m = n // block_size                    # number of blocks (50)
    target_sum_blocks = target_sum / block_size  # =25

    def project_blocks(x):
        return project_onto_bounds_and_sum(np.clip(x, 0.0, 1.0), target_sum_blocks)

    def blocks_to_h(blocks):
        blocks = project_blocks(blocks)
        return np.repeat(blocks, block_size)

    def eval_blocks(blocks):
        h = blocks_to_h(blocks)
        return compute_c5(h)

    # ----- optional initial population -------------------------------
    init_pop = None
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        if init_h.size != n:
            x_old = np.linspace(0.0, 2.0, init_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init_h = np.interp(x_new, x_old, init_h)
        init_h = project_onto_bounds_and_sum(init_h, target_sum)
        blocks0 = init_h.reshape(m, block_size).mean(axis=1)
        popsize = 15
        init_pop = np.random.rand(popsize, m)
        init_pop[0] = blocks0
        for i in range(popsize):
            init_pop[i] = project_blocks(init_pop[i])

    # ----- differential evolution ----------------------------------
    # time for DE (reserve a few seconds for refinement)
    time_left = budget_s - (time.time() - start) - 3.0
    max_iter = 2000
    if time_left < 20.0:
        max_iter = max(5, int(time_left / 0.02))
    result = differential_evolution(
        func=eval_blocks,
        bounds=[(0.0, 1.0)] * m,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=15,
        init=init_pop,
        seed=seed,
        polish=True,
        disp=False,
        updating='deferred',
        workers=1,
    )
    best_h = blocks_to_h(result.x)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    # ----- directed local refinement --------------------------------
    while time.time() - start < budget_s - 0.5:
        # correlation and worst shift
        corr = np.convolve(best_h, (1.0 - best_h)[::-1])
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * dx
        if cur_val < best_val:
            best_val = cur_val
        k = max_idx - (n - 1)    # shift corresponding to max overlap

        # gradient for this shift: g_i = 1 - h[i+k] - h[i-k]
        if k >= 0:
            h_plus = np.concatenate([best_h[k:], np.zeros(k, dtype=best_h.dtype)])
            h_minus = np.concatenate([np.zeros(k, dtype=best_h.dtype), best_h[:n - k]])
        else:
            kk = -k
            h_plus = np.concatenate([np.zeros(kk, dtype=best_h.dtype), best_h[:n - kk]])
            h_minus = np.concatenate([best_h[kk:], np.zeros(kk, dtype=best_h.dtype)])
        g = 1.0 - h_plus - h_minus

        pos = np.where((g > 1e-12) & (best_h > 1e-12))[0]
        neg = np.where((g < -1e-12) & (best_h < 1 - 1e-12))[0]
        moved = False
        if pos.size and neg.size:
            i_pos = pos[np.argmax(g[pos])]
            i_neg = neg[np.argmin(g[neg])]
            delta_max = min(best_h[i_pos], 1.0 - best_h[i_neg])
            if delta_max > 0:
                delta = delta_max * np.random.rand()
                cand = best_h.copy()
                cand[i_pos] -= delta
                cand[i_neg] += delta
                cand = np.clip(cand, 0.0, 1.0)
                cand = project_onto_bounds_and_sum(cand, target_sum)
                cval = compute_c5(cand)
                if cval < best_val:
                    best_val, best_h = cval, cand
                moved = True
        if not moved:
            # random pair move fallback
            i, j = np.random.randint(0, n, 2)
            if i != j:
                hi, hj = best_h[i], best_h[j]
                delta_max = min(1.0 - hi, hj)
                delta_min = -min(hi, 1.0 - hj)
                if delta_max > delta_min:
                    delta = np.random.uniform(delta_min, delta_max)
                    cand = best_h.copy()
                    cand[i] = hi + delta
                    cand[j] = hj - delta
                    cand = np.clip(cand, 0.0, 1.0)
                    cand = project_onto_bounds_and_sum(cand, target_sum)
                    cval = compute_c5(cand)
                    if cval < best_val:
                        best_val, best_h = cval, cand
        if best_val <= 0.3808:
            break

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
