### Projection onto Box + Sum constraints (Algorithm bisection)

Goal: given x, find y satisfying:

Proof: As lambda increases, each clip(x_i + lambda, 0,1) monotonic nondecreasing; sum is monotonic increasing, from 0 at lambda << -max(x) to n at lambda >> 1 - min(x). So there is a unique lambda solving sum = target. We can find it.

Implementation:

def project_to_feasible(x, target):
   lo = -np.max(x) - 1.0  # ensures x_i + lo <= -1 => clip 0
   hi = 1.0 - np.min(x) + 1.0  # ensures x_i + hi >= 2 => clip 1
   for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
   y = np.clip(x + lo, 0.0, 1.0)   # final
   # adjust for numeric residual
   residual = target - y.sum()
   if abs(residual) > 1e-9:
       # distribute resid among non-bound variables: those not at 0 or 1.
       free = (y > 1e-12) & (y < 1 - 1e-12)
       if free.any():
           # add residual equally
           y[free] += residual / free.sum()
           # clip again
           y = np.clip(y, 0.0, 1.0)
   return y

def subgradient_step(h, target, step_size):
    # compute correlation vector
    corr = np.correlate(h, 1 - h, mode='full')
    k_idx = np.argmax(corr)
    # shift index offset: in full correlation, lag = idx - (n-1); shift = lag
    lag = k_idx - (len(h) - 1)
    # compute gradient:
    # g[i] = (1 - h[i+lag]) - h[i-lag] where indices valid (0<=i+lag<n, 0<=i-lag <n)
    g = np.zeros_like(h)
    n = len(h)
    # For each i where both terms defined:
    # compute forward term
    if lag >= 0:
        # i from 0 to n-lag-1: forward term exists for i: 1 - h[i+lag]
        g[:n-lag] += (1 - h[lag:])  # that's 1 - h[i+lag] for i in [0, n-lag-1]
        # backward term: -h[i-lag] for i from lag to n-1
        g[lag:] -= h[:n-lag]  # -h[i-lag] for i >= lag
    else:
        # lag negative
        l = -lag
        # forward term: 1 - h[i+lag] defined for i >= l (since i+lag >=0)
        g[l:] += (1 - h[:n-l])
        # backward term: -h[i-lag] defined for i < n-l (since i-lag < n)
        g[:n-l] -= h[l:]
    # now g is gradient of sum_i h_i(1 - h_{i+lag}) (full convolution). For boundaries, rely on zeros for indices outside.
    # Perform step.
    h_new = h - step_size * g
    h_new = project_feasible(h_new, target)
    return h_new

def optimize_h(h_start, n, time_left_func, rng):
    target = n/2
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    iter = 0
    # subgradient phase: fixed number iterations limited by time
    # We'll run e.g., 2000 iterations or until time reduces.
    sub_iters = 3000
    while iter < sub_iters and time_left_func() > 0.1:
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        lag = k_idx - (n - 1)
        # compute gradient:
        g = np.zeros_like(h)
        if lag >= 0:
            g[:n - lag] += (1 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            l = -lag
            g[l:] += (1 - h[:n - l])
            g[:n - l] -= h[l:]
        # step size schedule
        step = 0.2 / (1 + iter * 0.001)
        h_new = h - step * g
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new
        # accept if better; else sometimes accept
        if c5_new <= compute_c5(h) or rng.random() < 0.01:
            h = h_new
        iter += 1

    # annealing phase
    # start from best_h
    h = best_h.copy()
    cur_c5 = best_c5
    T = 0.002
    sigma = 0.1
    decay_T = 0.9995
    decay_sigma = 0.9997
    while time_left_func() > 0.05:
        perturb = rng.normal(scale=sigma, size=n)
        cand = h + perturb
        cand = project_feasible(cand, target)
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = h.copy(), cur_c5
        T = max(T * decay_T, 1e-9)
        sigma = max(sigma * decay_sigma, 1e-5)

            h = best_h + sigma * rng.normal(size=n)
            h = project_feasible(h, target)
            cur_c5 = compute_c5(h)

    return best_h, best_c5

def project_feasible(x, target):
    # Use lambda (shift) for projection onto box + hyperplane.
    lo = -np.max(x) - 5.0  # ensure below feasible
    hi = 1.0 - np.min(x) + 5.0
    # Use binary search for 50 iterations.
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    # final y from lo (or hi)
    y = np.clip(x + lo, 0.0, 1.0)
    # due to numeric, adjust residual maybe.
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        # If residual >0, distribute to free variables
        free = (y > 1e-12) & (y < 1 - 1e-12)
        if free.any():
            y[free] += residual / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretized step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_feasible(x: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0 ≤ y ≤ 1, sum(y)=target} by a water‑filling (lambda‑shift) method."""
    # binary search for the shift lambda
    lo = -np.max(x) - 5.0
    hi = 1.0 - np.min(x) + 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + lo, 0.0, 1.0)
    # tiny residual correction (numerical)
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += residual / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def thue_morse_bits(n: int) -> np.ndarray:
    """Balanced 0/1 Thue‑Morse sequence (n must be a power of two)."""
    seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)
    return seq

# Core optimisation routine (subgradient + simulated annealing)
# ------------------------------------------------------------
def optimise_h(initial_h: np.ndarray, n: int, time_left, rng: np.random.Generator):
    """
    Refine a feasible h using a few hundred subgradient steps followed by a
    simulated‑annealing phase.  Returns the best h and its C5 value found.
    """
    target = n / 2.0
    h = project_feasible(initial_h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # ---- subgradient phase (focus on current worst shift) ----
    max_sub_iters = 3000
    it = 0
    while it < max_sub_iters and time_left() > 0.1:
        # correlation and index of maximal overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(corr))
        lag = k_idx - (n - 1)               # shift corresponding to the max
        # subgradient of the active term
        g = np.zeros_like(h)
        if lag >= 0:
            g[: n - lag] += (1.0 - h[lag:])   # (1 - h[i+lag])
            g[lag:]     -= h[: n - lag]       # -h[i-lag]
        else:
            l = -lag
            g[l:]       += (1.0 - h[: n - l]) # (1 - h[i+lag])  (lag negative)
            g[: n - l]  -= h[l:]              # -h[i-lag]
        # step size that slowly decays
        step = 0.25 / (1.0 + it * 0.001)
        cand = h - step * g
        cand = project_feasible(cand, target)
        cand_c5 = compute_c5(cand)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand.copy(), cand_c5

        # simple acceptance rule (mostly downhill, occasional uphill)
        if cand_c5 <= compute_c5(h) or rng.random() < 0.01:
            h = cand

        it += 1

    # ---- simulated‑annealing phase ----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    T = 0.004
    sigma = 0.15
    decay_T = 0.9995
    decay_sigma = 0.9997

    while time_left() > 0.02:
        perturb = rng.normal(scale=sigma, size=n)
        cand = cur_h + perturb
        cand = project_feasible(cand, target)
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
        # cooling schedules
        T = max(T * decay_T, 1e-9)
        sigma = max(sigma * decay_sigma, 1e-5)
        # occasional jump back to the best point (small noise)
        if rng.random() < 0.001:
            cur_h = best_h + sigma * rng.normal(size=n)
            cur_h = project_feasible(cur_h, target)
            cur_c5 = compute_c5(cur_h)

    return best_h, best_c5

    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial construction supplied by the harness (if any)

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target_bound = 0.3808

    for n in n_candidates:

        # build a collection of starting points for this size
        starts = []

                starts.append(init_global.copy())
            else:
                starts.append(upscale_h(init_global, n))

        # 2. Thue‑Morse pattern for powers of two
        if (n & (n - 1)) == 0:                # power of two test
            starts.append(thue_morse_bits(n))

        # 3. a random balanced binary vector
        rand_bin = np.zeros(n, dtype=float)
        rand_bin[: n // 2] = 1.0
        rng.shuffle(rand_bin)
        starts.append(rand_bin)

        # 4. a smooth random start (continuous values)
        rand_cont = rng.random(n)
        starts.append(project_feasible(rand_cont, n / 2.0))

        # evaluate each start and improve it
        for h0 in starts:
            if time_left() < 2.0:
                break
            h0 = project_feasible(h0, n / 2.0)
            start_c5 = compute_c5(h0)
            if start_c5 < best_c5:
                best_h, best_c5, best_n = h0.copy(), start_c5, n
                if best_c5 < target_bound:
                    return best_h.tolist(), float(best_c5), int(best_n)

            # allocate a slice of remaining time for the optimisation of this start
            slice_budget = min(200.0, time_left() * 0.5)
            slice_start = time.time()
            slice_time_left = lambda: slice_budget - (time.time() - slice_start)

            h_opt, c5_opt = optimise_h(h0, n, slice_time_left, rng)

            if c5_opt < best_c5:

                if best_c5 < target_bound:

        best_c5 = compute_c5(best_h)
        best_n = n
