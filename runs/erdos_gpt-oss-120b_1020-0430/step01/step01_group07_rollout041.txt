def _grad_ascent(start_h, time_budget):
    h = start_h.copy()
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    # schedule:
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # compute autocorr and min
        corr = np.correlate(h, h, mode='full')
        min_val = np.min(corr)
        # compute C5: max diff = 1 - dx * min_val
        cur_c = 1 - (2.0/n) * min_val
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # gradient:
        # compute gradient for shifts close to min
        # tolerance:
        tol = 1e-9 * max(1.0, abs(min_val))
        # mask indices
        mask = np.abs(corr - min_val) <= tol
        idxs = np.where(mask)[0]
        shifts = idxs - (n-1)
        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                i_end = n - s
                hi = h[:i_end]
                hj = h[s:]
                grad[:i_end] += hj
                grad[s:] += hi
            else:
                s_abs = -s
                # i_start = s_abs
                hi = h[s_abs:]
                hj = h[:n-s_abs]
                grad[s_abs:] += hj
                grad[:n-s_abs] += hi
        # compute step size:
        max_abs = np.max(np.abs(grad))
        if max_abs == 0:
            # random perturb to escape
            delta = 0.01 * (np.random.rand(n) - 0.5)
            h += delta
        else:
            # step magnitude schedule: start larger, diminish
            progress = (time.time() - start) / time_budget
            # step decays linearly from initial 0.5 to 0.001
            step_max = 0.5
            step_min = 0.001
            step = step_max * (step_min/step_max) ** progress
            scale = step / max_abs
            h = h + grad * scale
        # project onto constraints:
        np.clip(h, 0.0, 1.0, out=h)
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # try to adjust a random eligible element
            # if diff >0 add to entry <1, else subtract from entry >0
            idxs = np.where((diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0))[0]
            if len(idxs) > 0:
                # distribute diff equally across several entries if diff large
                # We'll just adjust first eligible
                idx = random.choice(idxs)
                # ensure feasible
                delta = diff
                # Cap to bounds
                if delta > 0:
                    delta = min(delta, 1.0 - h[idx])
                else:
                    delta = max(delta, -h[idx])
                h[idx] += delta
            else:
                # if none eligible (rare), just scale
                h *= target_sum / cur_sum
        # maybe occasionally add random noise to escape stagnation
        it += 1
        if it % 500 == 0:
            # random jitter
            jitter = (np.random.rand(n) - 0.5) * 0.01
            h += jitter
            np.clip(h, 0.0, 1.0, out=h)
            # reproject sum again
            cur_sum = h.sum()
            diff = target_sum - cur_sum
            if abs(diff) > 1e-12:
                # adjust by diff in a random eligible entry
                idxs = np.where((diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0))[0]
                if len(idxs)>0:
                    idx = random.choice(idxs)
                    if diff>0:
                        delta = min(diff, 1.0 - h[idx])
                    else:
                        delta = max(diff, -h[idx])
                    h[idx] += delta
    return best_h, best_c

def _repair(h, target):
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    deficit = target - cur
    if abs(deficit) > 1e-12:
        # try to distribute to a random eligible entry
        for idx in np.random.permutation(len(h)):
            lo = 0.0
            hi = 1.0
            if lo <= h[idx] + deficit <= hi:
                h[idx] += deficit
                break
        else:
            # fallback: scale
            h *= target / cur
    return h

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# basic evaluation, same as the reference implementation
def _c5_metric(h):
    """C5 value for a step function sampled by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


# ----------------------------------------------------------------------
# repair utilities – keep h in [0,1] and enforce the exact sum constraint
def _repair_sum_and_bounds(h, target):
    """Clip to [0,1] and tweak a single entry to hit the target sum."""
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    if abs(diff) <= 1e-12:
        return h
    # try a single entry that can absorb the entire deficit
    for idx in np.random.permutation(h.size):
        lo, hi = 0.0, 1.0
        if lo <= h[idx] + diff <= hi:
            h[idx] += diff
            return h
    # fallback: distribute uniformly over all non‑bound entries
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    avail = np.where(mask)[0]
    if avail.size:
        inc = diff / avail.size
        # respect bounds when distributing
        for i in avail:
            if diff > 0:
                inc_i = min(inc, 1.0 - h[i])
            else:
                inc_i = max(inc, -h[i])
            h[i] += inc_i
            diff -= inc_i
    # final tiny fix (should be negligible)
    if abs(diff) > 1e-12:
        h += diff / h.size
    np.clip(h, 0.0, 1.0, out=h)
    return h

# gradient ascent directed at the *minimum* autocorrelation
def _grad_ascent(start_h, max_seconds):
    """Projected gradient ascent on the min‑autocorrelation objective."""
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    start = time.time()
    # step schedule – exponential decay
    step_start = 0.5
    step_end = 0.001
    while time.time() - start < max_seconds:
        # current autocorrelation (no dx scaling needed for the direction)
        corr = np.correlate(h, h, mode="full")
        min_val = corr.min()
        # current C5 (for bookkeeping)
        cur_c = 1.0 - (2.0 / n) * min_val
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # identify all shifts attaining the minimum (within a tiny tolerance)
        tol = 1e-9 * max(1.0, abs(min_val))
        mask = np.abs(corr - min_val) <= tol
        shifts = np.where(mask)[0] - (n - 1)

        # compute sub‑gradient for those worst shifts
        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                i_end = n - s
                hi = h[:i_end]
                hj = h[s:]
                grad[:i_end] += hj
                grad[s:] += hi
            else:
                a = -s
                hi = h[a:]
                hj = h[:n - a]
                grad[a:] += hj
                grad[:n - a] += hi

        max_abs = np.max(np.abs(grad))
        if max_abs == 0.0:
            # stagnation – inject a tiny random bump
            h += (np.random.rand(n) - 0.5) * 0.001
        else:
            # linear/exponential step decay in log‑scale
            progress = (time.time() - start) / max_seconds
            step = step_start * (step_end / step_start) ** progress
            scale = step / max_abs
            h += grad * scale

        # projection onto the feasible set
        _repair_sum_and_bounds(h, target)

        # occasional jitter to escape plateaus
        if random.random() < 0.001:
            jitter = (np.random.rand(n) - 0.5) * 0.01
            h += jitter
            _repair_sum_and_bounds(h, target)

    return best_h, best_c

# a tiny hill‑climbing post‑processor (simple pairwise swaps)
def _hill_climb(h, max_iters=5000):
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    for _ in range(max_iters):
        i, j = random.sample(range(n), 2)
        hi, hj = best_h[i], best_h[j]

        # feasible delta range for the (i,j) pair
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.05
        best_h[i] = hi + delta
        best_h[j] = hj - delta
        new_c = _c5_metric(best_h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
        else:
            best_h[i] = hi
            best_h[j] = hj
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C5 bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # primary start – interpolated parent or a uniform baseline
    h0 = _initial_vector()
    n = h0.size
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 1) long gradient ascent from the deterministic start
    primary_time = total_budget * 0.75
    best_h, best_c = _grad_ascent(h0, primary_time)

    # ------------------------------------------------------------------
    # 2) a few random‑restart gradient climbs (remaining time)
    remaining = total_budget - (time.time() - start_all)
    while remaining > 6.0:                     # guarantee a few seconds per run
        # choose a random feasible start (binary or continuous)
        if random.random() < 0.5:
            # binary start – exactly n/2 ones (plus a fractional entry if needed)
            ones = int(target_sum)
            frac = target_sum - ones
            h_start = np.zeros(n, dtype=np.float64)
            perm = np.random.permutation(n)
            h_start[perm[:ones]] = 1.0
            if frac > 0:
                h_start[perm[ones]] = frac
        else:
            # continuous random start
            h_start = np.random.rand(n)
            h_start *= target_sum / h_start.sum()
            _repair_sum_and_bounds(h_start, target_sum)

        slice_time = min(remaining * 0.3, 30.0)
        cand_h, cand_c = _grad_ascent(h_start, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        remaining = total_budget - (time.time() - start_all)

    # ------------------------------------------------------------------
    # 3) final deterministic hill‑climbing refinement
    best_h, best_c = _hill_climb(best_h, max_iters=3000)

    # final safeguard – enforce the exact sum (tiny rounding errors)
    cur = best_h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # compute the final bound (use the provided helper if it exists)
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
