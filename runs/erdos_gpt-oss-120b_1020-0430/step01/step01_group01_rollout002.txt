import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} via bisection on the dual variable."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the max‑overlap objective for a given integer shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_momentum_optimize(
    h0: np.ndarray,
    target: float,
    step0: float = 0.5,
    decay: float = 0.996,
    momentum: float = 0.9,
    max_iter: int = 1200,
) -> np.ndarray:
    """Projected gradient descent with simple momentum (sub‑gradient of max‑overlap)."""
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    n = h.shape[0]
    for t in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        v = momentum * v - step * g
        h = h + v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def _smooth_max(x: np.ndarray, alpha: float) -> float:
    """Log‑sum‑exp smooth max."""
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_continuation_optimize(
    h0: np.ndarray,
    target: float,
    alphas: list,
    max_iter: int = 500,
) -> np.ndarray:
    """Successive SLSQP refinements with increasing smoothing parameter."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)

    h = h0.copy()
    for alpha in alphas:
        def obj(v):
            corr = np.correlate(v, 1.0 - v, mode='full')
            return _smooth_max(corr, alpha)

        res = minimize(
            obj,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False},
        )
        h = np.clip(res.x, 0.0, 1.0)
        h = _project_onto_capped_simplex(h, target)
    return h

        cand = _project_onto_capped_simplex(cand, target)
        val = _compute_c5(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val


def _upscale(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger number of points."""
    if len(vec) == new_n:
        return vec.copy()
    old_x = np.linspace(0.0, 2.0, len(vec), endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, vec)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function (as a vector of samples) improving the Erdős C5 bound.
    The return format is (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Determine discretisation size. 400 points gives a good trade‑off.
    # --------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    n = max(400, len(base_h) if base_h is not None else 0)
    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise best solution
    if base_h is not None:
        h_best = _upscale(base_h, n)
        h_best = _project_onto_capped_simplex(h_best, target_sum)
        best_val = _compute_c5(h_best, dx)
    else:
        h_best = None
        best_val = float('inf')

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    restart = 0
    while time.time() - start < budget_s * 0.90:
        # ---- generate a starting point ---------------------------------
        if h_best is not None and rng.rand() < 0.25:
            h0 = h_best + rng.normal(scale=0.025, size=n)
        else:
            h0 = _random_feasible(rng, n, target_sum)

        # ---- gradient descent with momentum ----------------------------
        h_gd = _gd_momentum_optimize(
            h0,
            target_sum,
            step0=0.6,
            decay=0.997,
            momentum=0.85,
            max_iter=800,
        )

        # ---- continuation SLSQP refinement ----------------------------
        alphas = [5, 12, 25, 45, 80]
        h_opt = _slsqp_continuation_optimize(h_gd, target_sum, alphas, max_iter=400)

        # ---- local random perturbation (cheap hill climb) ------------
        if time.time() - start < budget_s * 0.80:
            h_opt, _ = _local_search(
                h_opt,
                target_sum,
                dx,
                iters=120,
                rng=rng,
                step=0.035,
            )

        # ---- evaluate -------------------------------------------------
        val = _compute_c5(h_opt, dx)

        if val < best_val:
            best_val = val
            h_best = h_opt.copy()

        restart += 1

    # final safety projection
    if h_best is None:
        h_best = np.full(n, 0.5)
        h_best = _project_onto_capped_simplex(h_best, target_sum)
        best_val = _compute_c5(h_best, dx)
    else:
        h_best = _project_onto_capped_simplex(h_best, target_sum)
        best_val = _compute_c5(h_best, dx)

    return (h_best, float(best_val), n)
