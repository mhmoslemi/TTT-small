   best_h = h.copy()
   best_c = compute_c5(h)
   step = init_step
   for it in range(max_iters):
        # compute correlation full
        corr_full = np.correlate(h, 1 - h, mode='full')
        # find shift index for max (ignore edges? include)
        k_max = np.argmax(corr_full)
        # shift offset actual index: offset = k_max - (n-1)
        offset = k_max - (h.size - 1)
        # compute contributions p_i
        # For each i, idx = i + offset
        n = h.size
        contrib = np.empty(n)

        contrib[mask] = h[mask] * (1 - h[idxs[mask]])

        # pick i_max among those with h>0
        i_max = np.argmax(contrib)  # max contribution
        # compute (1 - h_{j+offset}) for each candidate j
        # we want small (1 - h_{j+offset}) i.e., large h_{j+offset}

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    # Determine n (target 180)

            n = n0

            # sort descending (largest close to 0.5)

            flip = zero_indices[order[:diff]]
            h[flip] = 1.0

            one_indices = np.where(h == 1)[0]
            one_fracs = frac[one_indices]
            order = np.argsort(one_fracs)  # smallest fractions (closest to 0)
            flip = one_indices[order[: -diff]]
            h[flip] = 0.0

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

    """Equality constraint Σ x_i = CURRENT_TARGET."""

def softmax_obj(x: np.ndarray) -> float:
    """Smooth surrogate for max_k Σ_i x_i(1‑x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")

    m = np.max(bc)

    """Optimise a step function h on [0,2] to minimise the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    target_n = 180                      # high‑resolution grid (≤200)
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        if target_n % n0 == 0:
            h = np.repeat(parent, target_n // n0)
        elif n0 * 2 <= target_n:
            h = np.repeat(parent, 2)
            if h.size < target_n:
                extra = random_binary_pattern(target_n - h.size, rng)
                h = np.concatenate([h, extra])
        else:
            h = parent.copy()
    else:
        h = random_binary_pattern(target_n, rng)

    h = enforce_feasibility(h, rng)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1 – intensive local improvement (targeted + random)
    # ------------------------------------------------------------------
    time_limit_loc = budget_s * 0.55
    step = 0.4
    while time.time() - start_time < time_limit_loc:
        # a short burst of focussed improvements
        h_opt, c_opt = improve_h(best_h, rng, max_iters=3000, init_step=step)
        if c_opt < best_c - 1e-12:
            best_h, best_c = h_opt, c_opt
        # slowly shrink the step for the next burst
        step = max(step * 0.85, 1e-4)

    # ------------------------------------------------------------------
    # Phase 2 – SLSQP refinements with an expanding β ladder
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.85:
        for beta in [5.0, 20.0, 80.0, 200.0, 500.0]:
            if time.time() - start_time > budget_s * 0.90:
                break
            try:
                cand = slsqp_optimize(best_h, beta=beta, max_iter=250)
                cand = enforce_feasibility(cand, rng)
                cand_c = compute_c5(cand)
                if cand_c < best_c - 1e-12:
                    best_h, best_c = cand, cand_c
            except Exception:
                continue

    # ------------------------------------------------------------------
    # Phase 3 – random‑restart exploration
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.98:
        h_start = random_binary_pattern(best_h.size, rng)
        h_tmp, c_tmp = improve_h(h_start, rng, max_iters=1500, init_step=0.5)
        # a couple of quick SLSQP passes
        for beta in [5.0, 20.0]:
            if time.time() - start_time > budget_s * 0.99:
                break
            try:
                h_tmp = slsqp_optimize(h_tmp, beta=beta, max_iter=150)
                h_tmp = enforce_feasibility(h_tmp, rng)
            except Exception:
                pass
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c - 1e-12:
            best_h, best_c = h_tmp, c_tmp

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = enforce_feasibility(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
