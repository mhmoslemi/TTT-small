def solve_lp(n=70):
    import cvxpy as cp
    u = cp.Variable(n)
    t = cp.Variable()
    # w matrix length n x n
    # Create w variable size n*n as a cp.Variable((n,n))
    w = cp.Variable((n,n))
    constraints = [u >= 0, u <= 1, cp.sum(u) == n / 2]
    constraints += [w >= 0]
    # w <= u_i (vertical), w <= u_j (horizontal)
    # using broadcasting; but cvxpy may not support broadcasting directly; need constructing constraints per entry.
    for i in range(n):
        constraints.append(w[i,:] <= u[i])   # row i, all columns <= u_i
        constraints.append(w[:,i] <= u[i])   # column i, all rows <= u_i

    # shift constraints
    for s in range(-(n-1), n):
        i_start = max(0, -s)
        i_end = min(n, n - s)
        indices = list(range(i_start, i_end))
        sum_w_shift = cp.sum(cp.hstack([w[i, i+s] for i in indices]))
        constraints.append(t >= (n/2) - sum_w_shift)

    # objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver='ECOS', verbose=False) # maybe use ECOS; it's a conic, not MILP. Should be able to handle linear constraints.

    # After solving, retrieve u.value.
    h = np.array(u.value).flatten()
    t_corr = t.value
    c5 = t_corr * (2.0 / n)
    return h, c5

   start = time.time()

       init_h = np.array(initial_h_values, dtype=float)

   target_sum = n/2.0
   # Try MILP
   try:
       # Define variables
       u = cp.Variable(n, boolean=True)
       t = cp.Variable()
       constraints = [cp.sum(u) == target_sum]

       ws = []  # list of w_s variables
       for s in range(1, n):
           length = n - s
           w_s = cp.Variable(length, boolean=True)
           ws.append(w_s)

           # vectorized constraints: w_s <= u[0:length]; w_s <= u[s:]

           constraints.append(t >= target_sum - cp.sum(w_s))

       constraints.append(t >= 0)

    random.seed(seed)

    start = time.time()

        n = 70

    try:
        u = cp.Variable(n, boolean=True)
        t = cp.Variable()
        constraints = [cp.sum(u) == target_sum]

        # shift variables
        for s in range(1, n):

            constraints.append(w_s <= u[:length])
            constraints.append(w_s <= u[s:])
            constraints.append(w_s >= u[:length] + u[s:] - 1)
            # correlation bound for shift s
            constraints.append(t >= target_sum - cp.sum(w_s))
        constraints.append(t >= 0)
