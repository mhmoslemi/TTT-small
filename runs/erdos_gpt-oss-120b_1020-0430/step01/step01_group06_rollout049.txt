def improve_local(h, max_iters, cur_c5=None):
    if cur_c5 is None:
        cur_c5 = c5_of(h)
    best_c5 = cur_c5
    best_h = h.copy()
    n = len(h)
    for it in range(max_iters):
        # compute correlation and worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (n - 1)
        # compute mismatch vector
        if s >= 0:
            i_ind = np.arange(n - s)
            p_ind = i_ind + s
        else:
            i_ind = np.arange(-s, n)
            p_ind = i_ind + s
        mism = h[i_ind] * (1 - h[p_ind])
        k = int(np.argmax(mism))
        i_best = i_ind[k]
        p_best = p_ind[k]
        eps_max = min(h[i_best], 1.0 - h[p_best])
        if eps_max <= 1e-12:
            # random small perturb
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            max_up = min(1 - hi, hj)
            max_down = min(hi, 1 - hj)
            if max_up + max_down == 0:
                continue
            delta = (random.random() * (max_up + max_down)) - max_down
            nh = h.copy()
            nh[i] = hi + delta
            nh[j] = hj - delta
        else:
            eps = eps_max * random.random()
            if eps < 1e-12:
                eps = eps_max * 0.5
            nh = h.copy()
            nh[i_best] -= eps
            nh[p_best] += eps
        new_c5 = c5_of(nh)
        if new_c5 < cur_c5:
            h = nh
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math
    from scipy.optimize import minimize

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    time_limit = budget_s * 0.95

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def c5_of(vec):
        """scaled max correlation"""
        n = len(vec)
        max_corr = np.max(np.correlate(vec, 1 - vec, mode="full"))
        return float(max_corr * (2.0 / n))

    def shift_corr(h, s):
        """correlation for a single shift (unnormalised)"""
        n = h.shape[0]
        if s >= 0:
            return np.dot(h[: n - s], 1 - h[s:])
        else:
            k = -s
            return np.dot(h[k:], 1 - h[: n - k])

    def project_to_sum(vec, target):
        """clip to [0,1] and adjust to meet the exact sum (simple iterative)"""
        np.clip(vec, 0.0, 1.0, out=vec)
        diff = vec.sum() - target
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:                     # need to decrease total
                cand = np.where(vec > 1e-12)[0]
                if cand.size == 0:
                    break
                i = random.choice(cand.tolist())
                delta = min(vec[i], diff)
                vec[i] -= delta
                diff -= delta
            else:                             # need to increase total
                cand = np.where(vec < 1 - 1e-12)[0]
                if cand.size == 0:
                    break
                i = random.choice(cand.tolist())
                delta = min(1.0 - vec[i], -diff)
                vec[i] += delta
                diff += delta
        np.clip(vec, 0.0, 1.0, out=vec)
        return vec

    # ----------------------------------------------------------------------
    # Core local improvement (greedy on worst shift)
    # ----------------------------------------------------------------------
    def greedy_refine(h, max_iters, cur_c5=None):
        n = len(h)
        if cur_c5 is None:
            cur_c5 = c5_of(h)
        best_h = h.copy()
        best_val = cur_c5
        for _ in range(max_iters):
            corr = np.correlate(h, 1 - h, mode="full")
            idx = int(np.argmax(corr))
            s = idx - (n - 1)

            if s >= 0:
                i_ind = np.arange(n - s)
                p_ind = i_ind + s
            else:
                i_ind = np.arange(-s, n)
                p_ind = i_ind + s

            mism = h[i_ind] * (1 - h[p_ind])
            k = int(np.argmax(mism))
            i_best = i_ind[k]
            p_best = p_ind[k]

            eps_max = min(h[i_best], 1.0 - h[p_best])
            if eps_max <= 1e-12:
                # fall back: random two‑index exchange
                i, j = random.sample(range(n), 2)
                hi, hj = h[i], h[j]
                max_up = min(1.0 - hi, hj)
                max_down = min(hi, 1.0 - hj)
                if max_up + max_down == 0:
                    continue
                delta = (random.random() * (max_up + max_down)) - max_down
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
            else:
                # move as much as possible (full eps_max)
                nh = h.copy()
                nh[i_best] -= eps_max
                nh[p_best] += eps_max

            new_val = c5_of(nh)
            if new_val < cur_c5 - 1e-12:
                h = nh
                cur_c5 = new_val
                if cur_c5 < best_val:
                    best_val = cur_c5
                    best_h = h.copy()
        return best_h, best_val

    # ----------------------------------------------------------------------
    # Main optimisation loop (random restarts + SLSQP + greedy)
    # ----------------------------------------------------------------------
    # Use the size of the supplied initial vector if present, otherwise 70
    n = 70
    if 'initial_h_values' in globals():
        try:
            n = len(initial_h_values)
        except Exception:
            pass
    target_sum = n / 2.0
    best_h = None
    best_c5 = float('inf')
    restart = 0

    while time.time() - start < time_limit:
        # ---- initial point -------------------------------------------------
        if restart == 0 and 'initial_h_values' in globals():
            h0 = np.array(initial_h_values, dtype=float).copy()
            if len(h0) != n:
                # fallback to random start if size mismatch
                h0 = np.random.rand(n)
        else:
            h0 = np.random.rand(n)

        h0 = project_to_sum(h0, target_sum)

        # ---- start formulation for SLSQP ----------------------------------
        max_corr0 = np.max(np.correlate(h0, 1 - h0, mode="full"))
        x0 = np.concatenate([h0, [max_corr0]])          # last entry is t = max_corr

        # bounds for h_i and t
        bounds = [(0.0, 1.0)] * n + [(0.0, target_sum)]

        # equality: sum h_i = target_sum
        cons = [{'type': 'eq',
                 'fun': lambda xv, t=target_sum: float(xv[:n].sum() - t)}]

        # inequalities: shift_corr <= t  for every shift
        for s in range(-(n - 1), n):
            cons.append({'type': 'ineq',
                         'fun': lambda xv, s=s: float(shift_corr(xv[:n], s) - xv[-1])})

        # ---- run SLSQP ----------------------------------------------------
        try:
            res = minimize(fun=lambda xv: xv[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 400,
                                    'ftol': 1e-9,
                                    'disp': False})
        except Exception:
            # if SciPy fails for any reason, skip this restart
            restart += 1
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_sum(h_opt, target_sum)

        # ---- greedy refinement ---------------------------------------------
        h_ref, c5_ref = greedy_refine(h_opt, max_iters=3000)

        # ---- update global best --------------------------------------------
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

        # early exit if we already beat the target
        if best_c5 <= 0.3808:
            break

        restart += 1

    # ----------------------------------------------------------------------
    # Final safety projection & evaluation
    # ----------------------------------------------------------------------
    if best_h is None:
        # fallback deterministic construction (half ones)
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_to_sum(best_h, target_sum)

    best_h = project_to_sum(best_h, target_sum)
    final_c5 = c5_of(best_h)

    return (best_h.tolist(), final_c5, n)
