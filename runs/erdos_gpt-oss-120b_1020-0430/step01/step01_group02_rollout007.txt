import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution among free variables
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _hill_climb(h: np.ndarray, current_c5: float, deadline: float, rng: np.random.Generator):
    """Simple pairwise mass‑exchange hill climbing."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < current_c5:
            h, current_c5 = h_new, new_c5
    return h, current_c5

    """

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")

    # ------------------------------------------------------------------
    end_time = start + budget_s - 15.0  # keep ~15 s for final polishing
    temp_start = max(0.01, best_c5 * 0.5)  # initial temperature
    iteration = 0

    while time.time() < end_time:
        # occasional large random restart
        if rng.random() < 0.002:
            cand = rng.random(n)
            cand = cand * (target / cand.sum())
            cand = _adjust_sum(cand, target)
        else:
            # propose a small mass transfer between two coordinates
            cand = None
            for _ in range(8):
                i, j = rng.choice(n, size=2, replace=False)
                if rng.random() < 0.5:
                    src, dst = i, j
                else:
                    src, dst = j, i
                max_move = min(best_h[src], 1.0 - best_h[dst])
                if max_move > 1e-12:
                    delta = rng.random() * max_move
                    cand = best_h.copy()
                    cand[src] -= delta
                    cand[dst] += delta
                    break
            if cand is None:
                continue

        cand_c5 = _evaluate_c5(cand)
        diff = cand_c5 - best_c5

        if diff < -1e-12:
            best_h, best_c5 = cand, cand_c5
        else:
            # linear cooling
            elapsed = time.time() - start
            frac = elapsed / (budget_s - 15.0)
            temp = max(1e-8, temp_start * (1.0 - frac))
            if rng.random() < np.exp(-diff / temp):
                best_h, best_c5 = cand, cand_c5

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Refine with a constrained local optimiser (SLSQP)
    # ------------------------------------------------------------------
    def obj(x):
        return _evaluate_c5(x)

    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n

    res = minimize(
        obj,
        best_h,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons],
        options={"maxiter": 800, "ftol": 1e-12, "disp": False},
    )
    refined = np.clip(res.x, 0.0, 1.0)
    refined = _adjust_sum(refined, target)
    refined_c5 = _evaluate_c5(refined)

    # ------------------------------------------------------------------
    # 4. Final short hill‑climb if time permits
    # ------------------------------------------------------------------
    final_deadline = start + budget_s - 0.5
    refined, refined_c5 = _hill_climb(refined, refined_c5, final_deadline, rng)

    # ------------------------------------------------------------------
    # 5. Return best found solution
    # ------------------------------------------------------------------
    final_h = _adjust_sum(np.clip(refined, 0.0, 1.0), target)
    final_c5 = float(_evaluate_c5(final_h))
    return final_h, final_c5, n
