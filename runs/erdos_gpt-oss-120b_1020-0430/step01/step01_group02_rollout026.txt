```
def _adjust_sum(x, target):
    x = np.clip(x, 0, 1)
    diff = target - x.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        candidates = np.where(mask)[0]
        per = diff / len(candidates)
        for i in candidates:
            if diff > 0:
                cap = 1 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0, 1)
```

```
def _gradient_move(h, cur_c5, target, rng):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n-1)
    # compute gradient
    grad = np.ones(n)
    if shift >= 0:
        if shift < n:
            grad[0:n-shift] -= h[shift:]
            grad[shift:] -= h[0:n-shift]
    else:
        s = -shift
        if s < n:
            grad[s:] -= h[0:n-s]
            grad[0:n-s] -= h[s:]
    # candidates
    src_candidates = np.where((grad > 1e-8) & (h > 1e-8))[0]
    dst_candidates = np.where((grad < -1e-8) & (h < 1-1e-8))[0]
    if src_candidates.size ==0 or dst_candidates.size==0:
        return h, cur_c5, False
    # choose src/dst
    # could select weighted randomly with prob proportional to |grad|
    src = src_candidates[rng.integers(src_candidates.size)]
    dst = dst_candidates[rng.integers(dst_candidates.size)]
    max_move = min(h[src], 1 - h[dst])
    if max_move < 1e-12:
        return h, cur_c5, False
    # try few deltas
    best_h = None
    best_val = cur_c5
    for factor in (0.3, 0.6, 0.9):
        delta = max_move * factor * rng.random()
        if delta <= 0: continue
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        val = _evaluate_c5(h_new)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new
    if best_h is not None:
        return best_h, best_val, True
    else:
        return h, cur_c5, False
```

import time
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto {0≤x≤1, sum(x)=target} by simple water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _gradient_move(h: np.ndarray, cur_val: float, target: float, rng: np.random.Generator):
    """
    One deterministic move guided by the gradient of the currently maximal shift.
    Returns (new_h, new_val, improved_flag).
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    s_idx = int(np.argmax(corr))
    shift = s_idx - (n - 1)

    grad = np.ones(n)
    if shift >= 0:
        if shift < n:
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] -= h[: n - s]
            grad[: n - s] -= h[s:]

    src_cand = np.where((grad > 1e-9) & (h > 1e-9))[0]
    dst_cand = np.where((grad < -1e-9) & (h < 1.0 - 1e-9))[0]
    if src_cand.size == 0 or dst_cand.size == 0:
        return h, cur_val, False

    src = src_cand[rng.integers(src_cand.size)]
    dst = dst_cand[rng.integers(dst_cand.size)]
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-12:
        return h, cur_val, False

    best_h = None
    best_val = cur_val
    for factor in (0.3, 0.6, 0.9):
        delta = max_move * factor * rng.random()
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        val = _evaluate_c5(h_new)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new
    if best_h is not None:
        return best_h, best_val, True
    return h, cur_val, False


def _optimize_n(
    n: int,
    init_vec: np.ndarray | None,
    rng: np.random.Generator,
    time_limit: float,
):
    """Simulated‑annealing + gradient descent for a given dimension."""
    target = n / 2.0
    if init_vec is not None and init_vec.size == n:
        h = _adjust_sum(np.clip(init_vec.astype(np.float64), 0, 1), target)
    else:
        h = rng.random(n)
        h = _adjust_sum(np.clip(h, 0, 1), target)

    best_h = h.copy()
    best_val = _evaluate_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val

    T = 0.02
    decay = 0.9997
    no_improve = 0
    start = time.time()
    deadline = start + time_limit - 0.5

    while time.time() < deadline:
        # gradient guided move
        new_h, new_val, improved = _gradient_move(cur_h, cur_val, target, rng)
        if improved:

            no_improve = 0
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
            T = max(1e-12, T * decay)
            continue

        # random mass exchange
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 1e-12:

        h_new = cur_h.copy()

        h_new[j] += delta

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):

    if time_left > 10.0:
        n1 = 81
        if init_vec is not None and n1 == best_n:
            n1 = 101
        h1, v1 = _optimize_n(n1, None, rng, time_left)
        if v1 < best_val:
            best_h, best_val, best_n = h1, v1, n1

    # Optional final local SLSQP polishing if time permits
    time_spent = time.time() - start_all
    time_left = budget_s - time_spent
    if time_left > 2.0:
        try:
            from scipy.optimize import minimize

            n = best_n
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n
            eq = {"type": "eq", "fun": lambda x: x.sum() - target}

            def obj(x):
                corr = np.correlate(x, 1.0 - x, mode="full")
                return float(np.max(corr) * (2.0 / n))

            x0 = best_h + rng.normal(scale=0.01, size=n)
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = _adjust_sum(x0, target)

            res = minimize(
                obj,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[eq],
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = _adjust_sum(np.clip(res.x, 0.0, 1.0), target)
                v_opt = _evaluate_c5(h_opt)
                if v_opt < best_val:
                    best_h, best_val = h_opt, v_opt
        except Exception:
            pass

    # final safety projection
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    best_val = float(_evaluate_c5(best_h))
    return best_h, best_val, best_n
