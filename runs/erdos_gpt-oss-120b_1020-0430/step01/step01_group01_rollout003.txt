import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Uniform random point in the hypercube, then project onto the simplex."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# --------------------------------------------------------------
# Objective: max correlation of h with (1-h) shifted
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# --------------------------------------------------------------
# Subgradient for the shift that attains the current max
# --------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 1500, step0: float = 0.8,
                 decay: float = 0.997) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑correlation objective."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h on [0,2] that (hopefully) improves the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    budget = float(budget_s)

    # ----------------------------------------------------------
    # Choose discretisation size (prefer more points for flexibility)
    # ----------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    min_n = 400
    if base_h is not None:
        n = max(min_n, len(base_h))
    else:
        n = min_n
    # keep n modest (<1000) as recommended
    n = min(n, 800)

    target_sum = n / 2.0
    dx = 2.0 / n

    # ----------------------------------------------------------
    # Helper to upscale a parent vector to the chosen resolution
    # ----------------------------------------------------------
    def _upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ----------------------------------------------------------
    # Initialise best candidate (parent if available)
    # ----------------------------------------------------------
    if base_h is not None:
        h_start = _upscale(base_h)
        h_start = _project_onto_capped_simplex(h_start, target_sum)
    else:
        h_start = _random_feasible(rng, n, target_sum)

    best_h = h_start.copy()
    best_val = _compute_c5(best_h, dx)

    # ----------------------------------------------------------
    # Phase 1: multiple projected sub‑gradient descents
    # ----------------------------------------------------------
    # Use remaining time wisely: stop when ~70% of budget spent
    phase1_deadline = start + 0.65 * budget
    gd_iters = 1800
    while time.time() < phase1_deadline:
        # random start, occasionally perturb current best
        if rng.rand() < 0.2 and best_h is not None:
            h0 = best_h + rng.normal(scale=0.02, size=n)
        else:
            h0 = _random_feasible(rng, n, target_sum)
        h0 = _project_onto_capped_simplex(h0, target_sum)

        h_opt = _gd_optimize(h0, target_sum,
                             max_iter=gd_iters,
                             step0=0.9,
                             decay=0.996)
        h_opt = _project_onto_capped_simplex(h_opt, target_sum)
        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()

    # ----------------------------------------------------------
    # Phase 2: Simulated annealing style refinement
    # ----------------------------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val
    SA_deadline = start + 0.97 * budget
    # Parameters for the annealing schedule
    init_step = 0.35
    final_step = 0.01
    init_temp = 1.0e-3
    final_temp = 1.0e-6

    while time.time() < SA_deadline:
        elapsed = time.time() - start
        frac = elapsed / (budget * 0.97)  # progress fraction within SA
        step = init_step * (1 - frac) + final_step * frac
        temp = init_temp * (1 - frac) + final_temp * frac

        cand = cur_h + step * rng.normal(size=n)
        cand = _project_onto_capped_simplex(cand, target_sum)
        cand_val = _compute_c5(cand, dx)

        if cand_val < cur_val or rng.rand() < np.exp(-(cand_val - cur_val) / max(temp, 1e-12)):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_h = cand_val, cand.copy()

        # occasional fresh perturbation around the best solution
        if int(elapsed * 100) % 20000 == 0:
            cur_h = best_h + rng.normal(scale=0.05, size=n)
            cur_h = _project_onto_capped_simplex(cur_h, target_sum)
            cur_val = _compute_c5(cur_h, dx)

    # ----------------------------------------------------------
    # Final projection & evaluation
    # ----------------------------------------------------------
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
