def anneal_optimize(h0, rng, max_iter, start_temp, end_temp):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    c = best_c

            c = new_c
            if c < best_c:

                best_h = h.copy()

def greedy_balance(h, rng, max_iters=2000, step=0.01):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        # compute autocorrelation for all k:
        # We can compute correlation of h with itself (full) -> conv.
        # C_corr = np.correlate(h, h, mode='full')
        C_full = np.correlate(h, h, mode='full')
        # This gives C_k for k from -(n-1) to (n-1)
        # We need only non-negative shifts? Since objective uses all shifts.
        # Let's consider all shifts.
        # min C_k:
        minc = np.min(C_full)
        # index with min:
        idx = np.argmin(C_full)
        # Convert idx to shift:
        shift = idx - (n - 1)  # shift value negative allowed
        # Determine direction: we want to increase C_k for this shift.
        # For each i, if we increase h[i] by eps, C_k changes by:
        # delta = eps * (h[i+shift] + h[i-shift]) (periodic)
        # For decreasing h[j] by eps, delta = -eps * (h[j+shift] + h[j-shift]).
        # So net change in C_k due to transfer epsilon from j to i is:
        # eps*(h[i+shift] + h[i-shift] - h[j+shift] - h[j-shift]).
        # We want positive change.
        # Choose i with max sum, j with min sum.
        idxs = np.arange(n)
        plus = h[(idxs + shift) % n] + h[(idxs - shift) % n]
        i = np.argmax(plus)
        j = np.argmin(plus)
        if i == j:
            break
        max_eps = min(h[j], 1.0 - h[i])  # can transfer from j to i
        if max_eps <= 0:
            break
        eps = min(step, max_eps)
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-9:
            h = h_new
            best_c = c_new
            best_h = h_new.copy()
        # else maybe accept with some prob? optional.
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly (project small residual)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def random_feasible_vector(n, rng):
    # start at 0.5 each
    h = np.full(n, 0.5)
    # mix by random transfers
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        if abs(eps) > max_eps:
            eps = np.sign(eps) * max_eps
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # clamp eps
    if eps > max_eps:
        eps = max_eps
    elif eps < -max_eps:
        eps = -max_eps
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # shrink step periodically
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def anneal_optimize(h0, rng, max_iter=2000, start_temp=0.02, end_temp=1e-6):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    c = best_c
    n = h.size
    for it in range(max_iter):
        # linear temperature schedule
        T = start_temp * (1.0 - it / max_iter) + end_temp * (it / max_iter)
        # scale decreasing from 0.4 to 0.01
        scale = 0.4 * (1.0 - it / max_iter) + 0.01 * (it / max_iter)
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def greedy_balance(h0, rng, max_iters=5000, step=0.01):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        # autocorrelation of h with itself
        corr = np.correlate(h, h, mode="full")
        min_idx = np.argmin(corr)
        shift = min_idx - (n - 1)  # signed shift
        # compute contribution for each position
        idxs = np.arange(n)
        contrib = h[(idxs + shift) % n] + h[(idxs - shift) % n]
        i = np.argmax(contrib)
        j = np.argmin(contrib)
        if i == j:
            break
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            break
        eps = min(step, max_eps)
        h[i] += eps
        h[j] -= eps
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

    # Determine discretisation size
    if "initial_h_values" in globals():

        if abs(diff) > 1e-12:

                if diff > 0:

                else:
                    add = max(diff, -h0[i])
                h0[i] += add
                diff = target - h0.sum()
    else:
        n = 180
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # deterministic candidate patterns
    x = (np.arange(n) + 0.5) * (2.0 / n)
    patterns = [
        np.full(n, 0.5),
        0.5 + 0.5 * np.cos(np.pi * x),               # half-wave cosine (range 0..1)
        np.arange(n) / n + 0.5 / n,                  # ramp up (exact sum)
        np.arange(n)[::-1] / n + 0.5 / n,            # ramp down
        np.clip(1 - np.abs(x - 1.0), 0.0, 1.0),       # symmetric triangle

            pat = np.clip(pat, 0.0, 1.0)
            diff = target - pat.sum()

import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP callbacks
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    """Run a single SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # tiny post‑processing to enforce the linear sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5(h):
    """Exact discrete C5 for a step function (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # shrink step size periodically
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def anneal_optimize(h0, rng, max_iter=2000, start_temp=0.02, end_temp=1e-6):
    """Simulated‑annealing style optimizer."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    for it in range(max_iter):
        T = start_temp * (1.0 - it / max_iter) + end_temp * (it / max_iter)
        scale = 0.4 * (1.0 - it / max_iter) + 0.01 * (it / max_iter)
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c


def greedy_balance(h0, rng, max_iters=5000, step=0.01):
    """Move mass from positions that hurt the worst autocorrelation shift to those that help."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        # autocorrelation of h with itself
        corr = np.correlate(h, h, mode="full")
        min_idx = np.argmin(corr)
        shift = min_idx - (n - 1)          # signed shift giving the smallest C_k
        idxs = np.arange(n)
        contrib = h[(idxs + shift) % n] + h[(idxs - shift) % n]
        i = np.argmax(contrib)            # best place to receive mass
        j = np.argmin(contrib)            # worst place to donate mass
        if i == j:
            break
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            break
        eps = min(step, max_eps)
        h[i] += eps
        h[j] -= eps
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise discretisation (up‑sample parent if present)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        up = 2
        h0 = np.repeat(base, up)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h0[i])
                else:
                    add = max(diff, -h0[i])
                h0[i] += add
                diff = target - h0.sum()
    else:
        n = 180
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Deterministic analytic candidates
    # ------------------------------------------------------------------
    x = (np.arange(n) + 0.5) * (2.0 / n)
    patterns = [
        np.full(n, 0.5),                                 # flat
        0.5 + 0.5 * np.cos(np.pi * x),                  # half‑wave cosine
        np.arange(n) / n + 0.5 / n,                     # ramp up (exact sum)
        np.arange(n)[::-1] / n + 0.5 / n,               # ramp down
        np.clip(1 - np.abs(x - 1.0), 0.0, 1.0),          # symmetric triangle
    ]
    for pat in patterns:
        pat = np.clip(pat, 0.0, 1.0)
        target = n / 2.0
        diff = target - pat.sum()
        if abs(diff) > 1e-12:
            pat = pat * (target / pat.sum())
            pat = np.clip(pat, 0.0, 1.0)
            diff = target - pat.sum()
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - pat[i])
                else:
                    add = max(diff, -pat[i])
                pat[i] += add
                diff = target - pat.sum()
        c = compute_c5(pat)
        if c < best_c:
            best_h, best_c = pat.copy(), c

    # ------------------------------------------------------------------
    # Main stochastic search (annealing + short hill‑climb)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        if rng.random() < 0.2:
            h_start = perturb_h(best_h, rng, scale=0.1)
        else:
            h_start = random_feasible_vector(n, rng)

        # annealing
        max_iter = 2500
        if time.time() - start_time > budget_s * 0.80:
            max_iter = max(200, int(max_iter * 0.2))
        h_a, c_a = anneal_optimize(h_start, rng, max_iter=max_iter,
                                   start_temp=0.02, end_temp=1e-6)

        # a brief deterministic refinement
        h_h, c_h = hill_climb(h_a, rng, max_iters=600, start_step=0.2)
        if c_h < best_c:
            best_h, best_c = h_h, c_h

    # ------------------------------------------------------------------
    # Greedy balancing refinement
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.90:
        h_g, c_g = greedy_balance(best_h, rng, max_iters=5000, step=0.01)
        if c_g < best_c:
            best_h, best_c = h_g, c_g

    # ------------------------------------------------------------------
    # More aggressive deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.96:
        h_h2, c_h2 = hill_climb(best_h, rng, max_iters=2000, start_step=0.15)
        if c_h2 < best_c:
            best_h, best_c = h_h2, c_h2

    # ------------------------------------------------------------------
    # Final soft‑max SLSQP refinement with increasing β
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.97:
        for beta in (20, 80, 320):
            if time.time() - start_time > budget_s * 0.97:
                break
            try:
                h_s = slsqp_optimize(best_h, beta=beta, max_iter=500)
                c_s = compute_c5(h_s)
                if c_s < best_c:
                    best_h, best_c = h_s, c_s
            except Exception:
                continue

    # ------------------------------------------------------------------
    # Final exact feasibility correction
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
