def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution similar to earlier
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        candidates = np.where(mask)[0]
        per = diff / len(candidates)
        for i in candidates:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple redistribution loop
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _worst_shift_info(h: np.ndarray):
    """Return (k, max_corr, contribution_vector) for the worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    k = idx - (n - 1)                # signed shift
    contrib = np.zeros(n, dtype=np.float64)
    if k >= 0:
        i = np.arange(n - k)
        contrib[i] = h[i] * (1.0 - h[i + k])
    else:
        i = np.arange(-k, n)
        contrib[i] = h[i] * (1.0 - h[i + k])
    return k, max_val, contrib


def _propose_random_move(h: np.ndarray, rng: np.random.Generator):
    src = np.where(h > 1e-12)[0]
    dst = np.where(h < 1.0 - 1e-12)[0]
    if src.size == 0 or dst.size == 0:
        return None
    i = src[rng.integers(src.size)]
    j = dst[rng.integers(dst.size)]
    max_move = min(h[i], 1.0 - h[j])
    if max_move <= 1e-12:
        return None
    delta = rng.random() * max_move
    return i, j, delta


def _propose_move(h: np.ndarray, rng: np.random.Generator, directed_prob: float = 0.3):
    """Hybrid move: with prob directed, otherwise random."""
    if rng.random() < directed_prob:
        k, _, contrib = _worst_shift_info(h)
        src_candidates = np.where(contrib > 1e-12)[0]
        if src_candidates.size == 0:
            return _propose_random_move(h, rng)
        i = src_candidates[rng.integers(src_candidates.size)]
        n = h.shape[0]
        if k >= 0:
            valid = np.arange(n - k)
        else:
            valid = np.arange(-k, n)
        mask = h[valid] < 1.0 - 1e-12
        if not np.any(mask):
            return _propose_random_move(h, rng)
        cand = valid[mask]
        shifted_i = h[i + k]
        shifted_vals = h[cand + k]
        better = shifted_vals > shifted_i + 1e-12
        if np.any(better):
            cand = cand[better]
        j = cand[rng.integers(cand.size)]
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 1e-12:
            return _propose_random_move(h, rng)
        # take the full possible amount for a directed step
        delta = max_move
        return i, j, delta
    else:
        return _propose_random_move(h, rng)


def _initialize_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an initial feasible vector of length n."""
    init = globals().get("initial_h_values")
    if init is not None:
        init_arr = np.asarray(init, dtype=np.float64)
        if init_arr.size == n:
            h0 = init_arr.copy()
        else:
            x_old = np.linspace(0.0, 2.0, init_arr.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init_arr)
    else:
        # random binary with exactly n/2 ones
        h0 = np.zeros(n, dtype=np.float64)
        ones = int(n // 2)
        idx = rng.permutation(n)[:ones]
        h0[idx] = 1.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _adjust_sum(h0, n / 2.0)
    return h0


def _refine_slsqp(h: np.ndarray, target: float, time_left: float) -> np.ndarray:
    """One short SLSQP refinement (if time permits)."""
    if time_left < 0.1:
        return h
    n = h.shape[0]
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr) * dx)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h
    if not res.success:
        return h
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = _adjust_sum(h_opt, target)
    return h_opt


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # -------------------- parameters --------------------
    n = 120                     # number of discretisation points
    target = n / 2.0
    directed_prob = 0.35        # chance of a directed move
    T0 = 0.02                   # initial temperature for simulated annealing

    # -------------------- initial solution --------------------
    h = _initialize_h(n, rng)
    cur_c5 = _evaluate_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # -------------------- simulated annealing --------------------
    while time.time() - start < budget_s - 1.0:
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = max(0.0, T0 * (1.0 - frac))

        mv = _propose_move(h, rng, directed_prob)
        if mv is None:
            continue
        i, j, delta = mv
        # apply move
        h[i] -= delta
        h[j] += delta
        new_c5 = _evaluate_c5(h)
        delta_c = new_c5 - cur_c5
        if delta_c < 0 or (T > 0 and rng.random() < np.exp(-delta_c / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta

    # -------------------- short SLSQP refinement --------------------
    time_left = start + budget_s - time.time()
    best_h = _refine_slsqp(best_h, target, time_left)

    # -------------------- greedy hill climbing --------------------
    deadline = start + budget_s - 0.2
    while time.time() < deadline:
        mv = _propose_random_move(best_h, rng)
        if mv is None:
            continue
        i, j, delta = mv
        best_h[i] -= delta
        best_h[j] += delta
        new_c5 = _evaluate_c5(best_h)
        if new_c5 + 1e-12 < best_c5:
            best_c5 = new_c5
        else:
            best_h[i] += delta
            best_h[j] -= delta

    # -------------------- final feasibility fix --------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum(best_h, target)
    best_c5 = float(_evaluate_c5(best_h))

    return best_h, best_c5, n
