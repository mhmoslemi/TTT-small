   import cvxpy as cp, time, numpy as np
   x = cp.Variable(n, boolean=True)
   constraints = [cp.sum(x) == n//2]  # integer division only works because n even.
   # For each L create y var
   for L in range(1, n):
       m = n - L
       y = cp.Variable(m, boolean=True)
       constraints.append(y <= x[:m])
       constraints.append(y <= x[L:])
       constraints.append(y >= x[:m] + x[L:] - 1)

    random.seed(seed)

    start = time.time()
    time_limit = budget_s * 0.95   # keep safe margin

    # determine n
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    except NameError:
        n = 70
        init = None

    target_sum = n/2.0

    # helper: project onto [0,1] and sum constraint
    def project(v):
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = cur - target_sum
        # adjust iteratively
        # decreasing
        while abs(diff) > 1e-12:
            if diff > 0:
                idx = np.where(v > 1e-12)[0]
                if idx.size == 0:
                    break
                i = random.choice(idx.tolist())
                delta = min(v[i], diff)
                v[i] -= delta
                diff -= delta
            else:
                need = -diff
                idx = np.where(v < 1 - 1e-12)[0]
                if idx.size == 0:
                    break
                i = random.choice(idx.tolist())
                delta = min(1.0 - v[i], need)
                v[i] += delta
                diff += delta
        return v

    # compute c5
    def c5_of(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return float(corr.max() * (2.0 / n))

    best_h = None
    best_c5 = float('inf')

    # number of restarts determined by remaining time
    while time.time() - start < time_limit:
        # Initialize h
        if best_h is None:
            if init is not None:
                h = init.copy()
            else:
                # random binary vector with exactly target_sum ones
                h = np.zeros(n)
                idx = np.arange(n)
                np.random.shuffle(idx)
                h[idx[:int(target_sum)]] = 1.0
        else:
            # perturb best_h slightly
            h = best_h + 0.02 * np.random.randn(n)
            h = project(h)

        # gradient descent parameters
        lr = 0.5
        lr_decay = 0.9995
        alpha = 2.0
        alpha_inc = 1.005
        max_local_iters = 3000

        for it_local in range(max_local_iters):
            # time check
            if time.time() - start >= time_limit:
                break
            # compute correlation vector
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = corr.max()
            # compute softmax weights
            # stability shift
            exp_vals = np.exp(alpha * (corr - max_corr))
            w = exp_vals / np.sum(exp_vals)

            # gradient
            g = np.zeros_like(h)

                w_neg = w[n - 1 - L]

def _c5_of(h):
    """Return C5 = max correlation * (2 / n) for a vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def _project_sum_and_bounds(v, target):
    """Clip v to [0,1] and adjust its sum to exactly target."""
    v = np.clip(v, 0.0, 1.0)
    diff = v.sum() - target
    # simple water‑filling without solving a linear program
    while abs(diff) > 1e-12:
        if diff > 0:                     # need to decrease total mass
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
            i = random.choice(idx.tolist())
            delta = min(v[i], diff)
            v[i] -= delta
            diff -= delta
        else:                            # need to increase total mass
            need = -diff
            idx = np.where(v < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = random.choice(idx.tolist())
            delta = min(1.0 - v[i], need)
            v[i] += delta
            diff += delta
    return v

def _gradient(h, alpha):
    """
    Gradient of the smooth approximation
        f(h) = (1/alpha) * log sum_k exp(alpha * C_k(h))
    where C_k are the correlation values for all integer shifts.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    # softmax weights, numerically stable
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)

    g = np.zeros_like(h)
    # handle positive and negative shifts together
    for L in range(1, n):
        w_pos = w[n - 1 + L]   # weight for shift +L
        w_neg = w[n - 1 - L]   # weight for shift -L
        m = n - L
        if w_pos != 0.0:
            g[:m] += w_pos * (1.0 - h[L:])   # ∂/∂h_i for i in [0,m)

            g[:m]  -= w_neg * h[L:]          # ∂/∂h_{i} for i in [0,m)
    return g
