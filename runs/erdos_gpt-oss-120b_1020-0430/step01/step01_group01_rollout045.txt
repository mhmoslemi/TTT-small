import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto {0 <= x <= 1, sum(x) = target} using bisection on the dual."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of max‑correlation w.r.t. h for a given shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation to max via log‑sum‑exp."""
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    step = step0
    m = np.zeros_like(h)
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        m = beta * m + (1.0 - beta) * g
        h = _project_onto_capped_simplex(h - step * m, target)
        step *= decay
    return h


def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 80.0,
                    max_iter: int = 1200) -> np.ndarray:
    """SLSQP on a smooth max approximation to fine‑tune the solution."""
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({"type": "eq", "fun": lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function on [0,2] that improves the Erdős C5 bound."""
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ----------------------------------------------------------------
    # Determine discretisation size (prefer at least 400 points)
    # ----------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    base_n = len(base_h) if base_h is not None else 0
    n = max(400, base_n, 200)          # target number of points
    target_sum = n / 2.0                # ∑ h = 1  →  sum = n/2 because dx = 2/n
    dx = 2.0 / n

    # ----------------------------------------------------------------
    # Helper to upscale a lower‑resolution vector if needed
    # ----------------------------------------------------------------
    def upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ----------------------------------------------------------------
    # Initial candidate
    # ----------------------------------------------------------------
    if base_h is not None:
        h0 = upscale(base_h)
        h0 = _project_onto_capped_simplex(h0, target_sum)
    else:
        h0 = _project_onto_capped_simplex(rng.rand(n), target_sum)

    best_h = h0.copy()
    best_val = _compute_c5(best_h, dx)

    # ----------------------------------------------------------------
    # Main optimisation loop (random restarts + PGD + occasional SLSQP)
    # ----------------------------------------------------------------
    restart = 0
    while time.time() - start_time < budget_s * 0.92:
        # ----- generate a starting vector ---------------------------------
        if rng.rand() < 0.3:
            # sinusoidal pattern for diversity
            phi = rng.rand() * 2.0 * np.pi
            amp = 0.3
            start_vec = 0.5 + amp * np.sin(2.0 * np.pi * np.arange(n) / n + phi)
            start_h = _project_onto_capped_simplex(start_vec, target_sum)
        elif rng.rand() < 0.3 and best_h is not None:
            # perturb current best
            start_h = best_h + rng.normal(scale=0.05, size=n)
            start_h = _project_onto_capped_simplex(start_h, target_sum)
        else:
            # fresh random feasible point
            start_h = _project_onto_capped_simplex(rng.rand(n), target_sum)

        # ----- projected gradient descent ---------------------------------
        h_opt = _pgd_optimize(start_h, target_sum,
                               max_iter=2000,
                               step0=0.3,
                               decay=0.998,
                               beta=0.9)

        # ----- occasional SLSQP refinement --------------------------------
        if restart % 4 == 0:
            h_opt = _slsqp_optimize(h_opt, target_sum,
                                    alpha=80.0,
                                    max_iter=1200)

        # ----- evaluate ----------------------------------------------------
        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            # early stop if we already beat the target
            if best_val <= 0.3808:
                break

        restart += 1

    # ----------------------------------------------------------------
    # Final safety projection and evaluation
    # ----------------------------------------------------------------
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
