import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points, then normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1] and sum = n/2)."""
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(arr) to new_n, then re‑normalise."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def max_corr_and_grad(h: np.ndarray):
    """Return (max correlation value, gradient of that value, shift) for h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    d = idx - (n - 1)          # integer shift corresponding to max
    term = np.zeros_like(h)
    if d >= 0:
        term[d:] += h[:n - d]      # h_{j-d}
        term[:n - d] += h[d:]      # h_{j+d}
    else:
        d2 = -d
        term[:n - d2] += h[d2:]    # h_{j+d}
        term[d2:] += h[:n - d2]    # h_{j-d}
    grad = 1.0 - term
    return max_val, grad, d


def propose_move(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, step: float):
    """Propose a feasible move that respects the sum constraint."""
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    n = h.shape[0]
    if pos.size and neg.size:
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            return None
        delta = rng.uniform(0.0, min(max_delta, step))
        if delta <= 0:
            return None
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
    # fallback: random feasible exchange
    i, j = rng.integers(0, n, size=2)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None
    delta = rng.uniform(0.0, max_delta * 0.5)
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def improve_h_sa(h_start: np.ndarray, time_deadline: float, rng: np.random.Generator):
    """Simple simulated‑annealing based improvement."""
    h = h_start.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    step = 0.08               # max mass moved in one proposal
    T = 0.012                 # initial temperature
    cooling = 0.995
    step_decay = 0.999
    iter_cnt = 0

    while time.time() < time_deadline:
        max_val, grad, _ = max_corr_and_grad(h)
        # propose
        h_new = propose_move(h, grad, rng, step)
        if h_new is None:
            continue
        new_c5 = compute_c5(h_new)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5
        iter_cnt += 1
        if iter_cnt % 200 == 0:
            T *= cooling
            step = max(step * step_decay, 1e-5)
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    # Determine a working resolution (larger gives more flexibility)
    base_n = 200
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    # Pick the resolution: use the larger of base_n and parent_n (if known)
    target_n = max(base_n, parent_n if parent_n is not None else 0)

    # Build a pool of diverse starting points
    starts = []
    if parent_h is not None:
        if parent_n == target_n:
            starts.append(parent_h.copy())
        else:
            starts.append(upsample(parent_h, target_n))
    starts.append(generate_triangular(target_n))
    for _ in range(3):
        starts.append(random_start(target_n, rng))

    best_h = None
    best_c5 = np.inf
    best_n = target_n

    # Allocate time slices to each start (≈ equal share)
    slice_time = max(2.0, (time_limit - time.time()) / max(len(starts), 1))

    for s in starts:
        remaining = time_limit - time.time()
        if remaining < 2.0:
            break
        run_deadline = time.time() + min(slice_time, remaining)
        h_opt, c5_opt = improve_h_sa(s, run_deadline, rng)
        if c5_opt < best_c5 - 1e-9:
            best_h = h_opt
            best_c5 = c5_opt

    # If nothing succeeded (unlikely), fall back to a simple binary block
    if best_h is None:
        n = target_n
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final sanity check / projection onto the feasible hyperplane
    target_sum = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h + (target_sum - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
