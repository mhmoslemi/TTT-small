def greedy_step(h, k, step_limit=None):
   n = len(h)
   v = np.zeros(n)
   # compute v efficiently using shifted slices:
   if k >= 0:
       v[:n-k] += h[k:]   # h[i + k] for i up to n-k-1
       v[k:] += h[:n-k]   # h[i - k] for i = k..n-1
   else: # k negative
       kk = -k
       v[kk:] += h[:n-kk]

def smooth_gd(h, target_sum, dx, time_limit, init_alpha=5, final_alpha=20, step0=0.01):

def project_box_sum(v, target):
    # water-filling algorithm
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0, 1)

import time
import random
import math
import numpy as np

def _c5_metric(h):
    """Maximum overlap integral C5 for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_box_sum(v, target):
    """Euclidean projection of v onto [0,1]^n with sum(v)=target."""
    lo = np.min(v) - 1.0       # enough low so that all values are clipped to 1
    hi = np.max(v)             # enough high so that all values are clipped to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _random_binary_h(n, target):
    """Binary (or one fractional) start vector with given sum."""
    ones = int(math.floor(target))
    frac = target - ones
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.permutation(n)
    if ones:
        h[idx[:ones]] = 1.0
    if frac > 1e-12:
        h[idx[ones]] = frac
    return h

def _greedy_search(h, target_sum, time_end):
    """Greedy mass‑transport search aimed at the worst lag."""
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    max_move = 0.05
    steps_without_improve = 0
    restart_limit = 5000

    while time.time() < time_end:
        # correlation of h with itself
        corr = np.correlate(h, h, mode="full")
        O = 1.0 - corr * dx                # overlap values for every shift
        maxO = O.max()

        # keep global best
        if maxO < best_c - 1e-12:
            best_c = maxO
            best_h = h.copy()
            steps_without_improve = 0
            if best_c <= 0.3808:          # early success
                break
        else:
            steps_without_improve += 1

        # pick one of the worst lags
        worst = np.where(O >= maxO - 1e-12)[0]
        k_idx = np.random.choice(worst)
        k = k_idx - (n - 1)                # actual shift amount

        # neighbour‑sum for each index (h[i+k] + h[i-k])
        v = np.zeros(n)
        if k >= 0:
            v[:n - k] += h[k:]
            v[k:] += h[:n - k]
        else:
            kk = -k
            v[kk:] += h[:n - kk]
            v[:n - kk] += h[kk:]

        donors = np.where(h > 1e-12)[0]
        receivers = np.where(h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break

        donor = donors[np.argmin(v[donors])]
        receiver = receivers[np.argmax(v[receivers])]
        if donor == receiver:
            continue

        delta = min(h[donor], 1.0 - h[receiver], max_move)
        if delta <= 1e-12:
            # tiny random tweak if stuck
            i, j = np.random.choice(n, 2, replace=False)
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo < hi:
                d = random.uniform(lo, hi) * 0.2
                h[i] += d
                h[j] -= d
            continue

        h[donor] -= delta
        h[receiver] += delta

        # occasional correction of accumulated rounding error
        if steps_without_improve % 1000 == 0:
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                for idx in np.random.permutation(n):
                    if 0.0 <= h[idx] + diff <= 1.0:
                        h[idx] += diff
                        break

        # restart if stuck too long
        if steps_without_improve >= restart_limit:
            h = _random_binary_h(n, target_sum)
            steps_without_improve = 0

    return best_h, best_c

def _local_refine(h, target_sum, time_end):
    """Tiny hill‑climbing moves that keep the sum fixed."""
    n = h.shape[0]
    best_c = _c5_metric(h)
    while time.time() < time_end:
        i, j = np.random.choice(n, 2, replace=False)
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue
        delta = random.uniform(lo, hi) * 0.02
        h[i] += delta
        h[j] -= delta
        c = _c5_metric(h)
        if c < best_c:
            best_c = c
        else:
            h[i] -= delta
            h[j] += delta
    return h, best_c
