import time

# -------------------------------------------------
# Helper utilities

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    # simple iterative correction – fast for the tiny residuals we get
    for _ in range(10):
        diff = h.sum() - target
        if abs(diff) <= tol:
            break
        # decide where we can move the excess/deficit
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix (should be almost never needed)
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-14:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-14:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

    cur_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # temperature schedule (linear to zero)
    T0 = 1e-2
    start_time = time.time()

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        # temperature decreases linearly
        T = T0 * (1.0 - it / max_iters)

        lower = max(-(1 - hi), -hj)          # minimal allowed δ

        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
