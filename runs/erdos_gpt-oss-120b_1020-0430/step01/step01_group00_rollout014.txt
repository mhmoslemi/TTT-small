def compute_corr_and_shift(h):
    n = h.shape[0]
    nfft = 2 * n
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(1.0 - h, nfft)
    corr_circ = np.fft.ifft(H * np.conj(G)).real
    # half-length for negative lags
    neg = corr_circ[n:2*n-1][::-1]
    pos = corr_circ[:n]
    corr = np.concatenate([neg, pos])  # shape 2n-1
    max_idx = np.argmax(corr)
    shift = max_idx - (n-1)
    max_val = corr[max_idx]
    return max_val, shift, corr

def hill_climb(h0, max_iters, time_budget, rnd_state, target=None):
    h = h0.copy()

    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    step_factor = 0.5
    iteration = 0
    start = time.time()
    while iteration < max_iters and (time.time() - start) < time_budget:
        # compute shift and gradient
        max_val, shift, corr = compute_corr_and_shift(h)
        cur_val = max_val * (2.0 / n)  # could also use max_val * 2/n

        #gradient
        g = gradient_for_shift(h, shift)

        donors = np.where((g > 1e-12) & (h > 1e-12))[0]
        receivers = np.where((g < -1e-12) & (h < 1 - 1e-12))[0]
        if donors.size == 0 or receivers.size == 0:
            # fallback random move
            i = rnd_state.choice(n, p=(h/np.sum(h)))  # pick from positive mass? Not needed.
            # Instead random indices
            i = rnd_state.choice(np.where(h > 1e-12)[0])
            j = rnd_state.choice(np.where(h < 1 - 1e-12)[0])
        else:
            i = rnd_state.choice(donors)
            j = rnd_state.choice(receivers)

        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            # can't move; continue
            iteration += 1
            continue
        epsilon = max_move * step_factor * rnd_state.rand()
        if epsilon <= 0:
            epsilon = max_move * step_factor
        # apply move
        h[i] -= epsilon
        h[j] += epsilon

        # Evaluate new C5
        new_max_val, _, _ = compute_corr_and_shift(h)
        new_val = new_max_val * (2.0 / n)

        if new_val < cur_val - 1e-12:  # improvement
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            # optionally increase step factor slightly
            step_factor = min(step_factor * 1.02, 0.9)
        else:
            # revert move
            h[i] += epsilon
            h[j] -= epsilon
            # optionally reduce step factor
            step_factor = max(step_factor * 0.99, 0.01)

        iteration += 1

        # optional early exit if target met
        if target is not None and best_val <= target:
            break

def _corr_and_shift(h):
    n = h.shape[0]
    nfft = 2 * n
    H = np.fft.fft(h, n=nfft)
    G = np.fft.fft(1.0 - h, n=nfft)
    corr_circ = np.fft.ifft(H * np.conj(G)).real
    # Build linear correlation values for lag range -(n-1) .. (n-1)
    # negative lags part:
    neg = corr_circ[n:2*n-1][::-1]  # shape n-1
    pos = corr_circ[:n]  # shape n
    corr = np.concatenate([neg, pos])

import time
import numpy as np

# ---------- feasibility utilities ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 via a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 using numpy.correlate (matches the evaluator)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- correlation helpers ----------
def _corr_and_shift(h: np.ndarray):
    """
    Return (max_corr, shift, full_corr_array) where max_corr is the
    (unscaled) maximum of the linear correlation h * (1-h) shifted,
    and shift is the integer lag (negative … positive) attaining it.
    """
    n = h.shape[0]
    nfft = 2 * n
    H = np.fft.fft(h, n=nfft)
    G = np.fft.fft(1.0 - h, n=nfft)
    circ = np.fft.ifft(H * np.conj(G)).real
    # Build linear correlation: lags -(n-1)…(n-1)
    neg = circ[n:2 * n - 1][::-1]      # length n-1, negative lags
    pos = circ[:n]                     # length n, non‑negative lags
    lin = np.concatenate([neg, pos])
    idx = np.argmax(lin)
    max_corr = lin[idx]
    shift = idx - (n - 1)              # 0 corresponds to lag 0
    return max_corr, shift, lin


def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Subgradient of the correlation at the given shift."""
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        if shift < n:
            g[: n - shift] += 1.0 - h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[: n - s]
            g[: n - s] -= h[s:]
    return g


# ---------- local search ----------
def _hill_climb(initial_h: np.ndarray, max_iters: int, budget_s: float,
                rng: np.random.RandomState, target: float = None):
    """
    Simple projected sub‑gradient descent.
    Returns (best_h, best_c5).
    """
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(h)
    cur_val = best_val
    step_factor = 0.5
    start = time.time()

    for it in range(max_iters):
        if (time.time() - start) > budget_s:
            break
        max_corr, shift, _ = _corr_and_shift(h)
        cur_val = max_corr * (2.0 / n)

        # gradient for the active shift
        g = _gradient_shift(h, shift)

        donors = np.where((g > 1e-12) & (h > 1e-12))[0]
        receivers = np.where((g < -1e-12) & (h < 1 - 1e-12))[0]

        if donors.size == 0 or receivers.size == 0:
            # fall back to a random admissible move
            pos_i = np.where(h > 1e-12)[0]
            pos_j = np.where(h < 1 - 1e-12)[0]
            if pos_i.size == 0 or pos_j.size == 0:
                break
            i = rng.choice(pos_i)
            j = rng.choice(pos_j)
        else:
            i = rng.choice(donors)
            j = rng.choice(receivers)

        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        eps = max_move * step_factor * rng.random()
        if eps <= 0:
            eps = max_move * step_factor

        # apply move
        h[i] -= eps
        h[j] += eps

        # evaluate
        new_corr, _, _ = _corr_and_shift(h)
        new_val = new_corr * (2.0 / n)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if target is not None and best_val <= target:
                    break
            step_factor = min(step_factor * 1.02, 0.9)
        else:
            # revert
            h[i] += eps
            h[j] -= eps
            step_factor = max(step_factor * 0.99, 0.01)

        # re‑project sum occasionally to avoid drift
        if it % 1000 == 0:
            diff = h.sum() - n / 2.0
            if abs(diff) > 1e-10:
                # spread the correction over a random feasible coordinate
                if diff > 0:
                    cand = np.where(h > 0)[0]
                else:
                    cand = np.where(h < 1)[0]
                if cand.size:
                    k = rng.choice(cand)
                    h[k] -= diff
                    h = np.clip(h, 0.0, 1.0)

    return best_h, best_val


# ---------- main driver ----------

    """
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # obtain optional parent construction
    try:
        parent_h = np.array(initial_h_values, dtype=float)

    best_h = None

    best_n = None
    target_c5 = 0.3808

    # candidate grid sizes (even, ≤ 300)
    sizes = []
    if parent_h is not None:
        sizes.append(parent_h.size)
    # multiples of the parent size
    if parent_h is not None:
        for mul in (2, 3):
            n = parent_h.size * mul
            if n <= 300:
                sizes.append(int(n))
    # some handcrafted even sizes
    for n in (160, 200, 256, 300):
        if n <= 300:
            sizes.append(n)
    sizes = sorted(set(sizes))

    # ----- optimisation over each size -----
    for n in sizes:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        remaining = budget_s - elapsed
        # allocate a modest slice to this size
        slice_budget = min(250.0, remaining * 0.4)

        # prepare start vectors
        starts = []
        if parent_h is not None:
            if parent_h.size != n:
                starts.append(_upscale_h(parent_h, n))
            else:
                starts.append(parent_h.copy())

        # random feasible start
        rng0 = np.random.RandomState(seed + n * 7)
        rand_vec = rng0.rand(n)
        rand_vec = _project_to_feasible(rand_vec, n)
        starts.append(rand_vec)

        # a slightly smoothed random start
        smooth = np.convolve(rand_vec, np.ones(3) / 3.0, mode="same")
        smooth = _project_to_feasible(smooth, n)
        starts.append(smooth)

        # run a short hill‑climb from each start
        per_start_budget = slice_budget / len(starts)
        for idx, h0 in enumerate(starts):
            rng = np.random.RandomState(seed + n * 31 + idx)
            h_opt, c5_opt = _hill_climb(
                h0,
                max_iters=10_000_000,
                budget_s=per_start_budget,
                rng=rng,
                target=target_c5,

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                if best_c5 <= target_c5:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # fallback – uniform half‑ones
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
