def project_to_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # if s > target, need to subtract from some interior entries
    # if s < target, need to add
    # Use water-filling: while abs(s - target) > tol:
    for _ in range(1000):
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        # Determine indices where we can adjust
        if diff > 0:  # need to reduce sum
            adjustable = np.where(h > 0)[0]
        else:
            adjustable = np.where(h < 1)[0]
        if adjustable.size == 0:
            break
        delta = diff / adjustable.size
        h[adjustable] -= delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    # final small correction: adjust one element precisely
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # find index not at bound
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
import scipy.optimize as opt

# ----- Helper functions ------------------------------------------------------


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # Tiny final correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Scaled max correlation for step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _max_corr_and_shift(h: np.ndarray):
    """Return (scaled max correlation, integer shift) for h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_corr = corr[idx] * (2.0 / n)
    shift = idx - (n - 1)
    return max_corr, shift


def _grad_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of C(s) w.r.t. h (unscaled)."""
    n = h.size
    g = np.zeros_like(h)
    for i in range(n):
        j = i + s
        if 0 <= j < n:
            g[i] += 1.0 - h[j]
        j2 = i - s
        if 0 <= j2 < n:
            g[i] -= h[j2]
    return g


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _gradient_descent(
    h0: np.ndarray, max_iter: int, time_left, target_bound: float = 0.3808
):
    """Sub‑gradient descent on max‑correlation objective."""
    n = h0.size
    dx = 2.0 / n
    h = _project_to_feasible(h0, n)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    step_base = 0.5 * dx
    step = step_base
    stagnation = 0

    for it in range(max_iter):
        if time_left() < 0.02:
            break

        cur_c, s = _max_corr_and_shift(h)

        if cur_c < best_c - 1e-12:
            best_c, best_h = cur_c, h.copy()
            stagnation = 0
        else:
            stagnation += 1

        # gradient of the active shift (unscaled)
        grad = _grad_for_shift(h, s)

        improved = False
        for _ in range(5):
            h_new = _project_to_feasible(h - step * grad, n)
            # enforce symmetry (helps the problem)
            h_new = (h_new + h_new[::-1]) * 0.5
            h_new = _project_to_feasible(h_new, n)

            new_c = _c5_from_h(h_new)
            if new_c < cur_c - 1e-12:
                h = h_new
                step = min(step * 1.1, step_base * 8)
                improved = True
                break
            else:
                step *= 0.5

        if not improved and stagnation > 300:
            break

        if best_c < target_bound:
            break

    return best_h, best_c


def _slsqp_refine(h: np.ndarray):
    """One SLSQP refinement starting from h."""
    n = h.size
    h0 = _project_to_feasible(h, n)
    corr = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr)

    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        hh = x[:n]
        tt = x[-1]
        return tt - np.correlate(hh, 1.0 - hh, mode="full")

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": ineq},
        {"type": "eq", "fun": eq},
    ]

    res = opt.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 500, "ftol": 1e-9, "disp": False},
    )
    if res.success:
        h_opt = _project_to_feasible(res.x[:n], n)
    else:
        h_opt = h0
    return h_opt, _c5_from_h(h_opt)
