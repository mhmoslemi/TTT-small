import random, time, math
import numpy as np

def _c5_of(v):
    """C5 value for vector v (dx = 2/len(v))."""
    return np.correlate(v, 1.0 - v, mode="full").max() * (2.0 / len(v))

def _project_to_sum(v, target):
    """Clip to [0,1] and adjust to exact sum = target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    err = v.sum() - target
    if abs(err) < 1e-12:
        return v
    # iterative water‑filling on free variables
    while True:
        free = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
        if free.size == 0:
            # everything at a bound – distribute the tiny remainder uniformly
            delta = err / len(v)
            v += delta
            v = np.clip(v, 0.0, 1.0)
            err = v.sum() - target
            if abs(err) < 1e-12:
                break
            continue
        delta = err / free.size
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
        err = v.sum() - target
        if abs(err) < 1e-12:
            break
    return v

def _random_binary(n):
    """Half‑ones binary vector of length n."""
    h = np.zeros(n, dtype=float)
    idx = np.arange(n)
    np.random.shuffle(idx)
    h[idx[: n // 2]] = 1.0
    return h

def _targeted_move(h, shift):
    """Move mass from the source of the worst product to its partner."""
    n = len(h)
    if shift == 0:
        return None
    if shift > 0:
        prod = h[: n - shift] * (1.0 - h[shift:])
        k = int(np.argmax(prod))
        src, dst = k, k + shift
    else:
        d = -shift
        prod = h[d:] * (1.0 - h[: n - d])
        k = int(np.argmax(prod))
        src, dst = k + d, k
    eps_max = min(h[src], 1.0 - h[dst])
    if eps_max <= 1e-12:
        return None
    eps = eps_max * 0.5
    nh = h.copy()
    nh[src] -= eps
    nh[dst] += eps
    return nh

def _gradient_move(h, shift):
    """Use gradient of the worst shift to transfer mass."""
    n = len(h)
    grad = np.zeros(n, dtype=float)
    if shift >= 0:
        for i in range(n - shift):
            grad[i] += 1.0 - h[i + shift]
            grad[i + shift] -= h[i]
    else:
        d = -shift
        for i in range(n - d):
            grad[i + d] += 1.0 - h[i]
            grad[i] -= h[i + d]
    inc = np.where(grad < -1e-12)[0]
    dec = np.where(grad > 1e-12)[0]
    if inc.size == 0 or dec.size == 0:
        return None
    i_inc = int(np.random.choice(inc))
    i_dec = int(np.random.choice(dec))
    eps_max = min(1.0 - h[i_inc], h[i_dec])
    if eps_max <= 1e-12:
        return None
    eps = eps_max * 0.5
    nh = h.copy()
    nh[i_inc] += eps
    nh[i_dec] -= eps
    return nh

def _swap_move(h):
    """Random two‑coordinate exchange preserving the sum."""
    n = len(h)
    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]
    max_up = min(1.0 - hi, hj)      # increase i, decrease j
    max_down = min(hi, 1.0 - hj)    # decrease i, increase j
    total = max_up + max_down
    if total <= 0.0:
        return None
    delta = (random.random() * total) - max_down
    if abs(delta) < 1e-14:
        return None
    nh = h.copy()
    nh[i] = hi + delta
    nh[j] = hj - delta
    return nh

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # Initialise problem size and starting vector

    return (best_h.tolist(), float(final_c5), n)
