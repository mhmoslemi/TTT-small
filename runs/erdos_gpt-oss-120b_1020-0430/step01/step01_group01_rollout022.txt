def grad_weighted(h, w):
    n = len(h)
    g = np.zeros_like(h)
    for s in range(-n+1, n):
        ws = w[s + n - 1]
        if ws == 0:
            continue
        if s >= 0:
            # term1: B_{i+s}
            # B = 1 - h
            # For i in [0, n-s-1], add ws * (1 - h[i+s])
            g[:n-s] += ws * (1 - h[s:])
            # term2: -h_{i-s} for i >= s
            g[s:] += ws * (-h[:n-s])
        else:
            # s negative => s = -t
            t = -s
            # term1: B_{i-s} = B_{i+t}
            g[t:] += ws * (1 - h[:n-t])
            # term2: -h_{i+s} = -h_{i - t}
            g[:n-t] += ws * (-h[t:])
    return g
```

def weighted_gradient(h, alpha):
    # compute corr (raw sum no dx)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # compute softmax weights
    maxc = corr.max()
    exps = np.exp(alpha * (corr - maxc))
    w = exps / exps.sum()

def _weighted_gradient(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()

def _gd_optimize(h0, target, max_iter=500, step0=0.2, decay=0.995):
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h
```

import time

# ----------------------------------------------------------------------
# Projection onto {0 <= x <= 1, sum(x) = target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]

    else:

    return g

def _weighted_gradient(h: np.ndarray, alpha: float):
    """Return (grad, corr) where grad is ∇_h (soft‑max of correlations)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    exp_shifted = np.exp(alpha * (corr - maxc))
    w = exp_shifted / exp_shifted.sum()    # soft‑max weights, sum to 1

    B = 1.0 - h
    w_rev = w[::-1]

    term1 = np.convolve(w_rev, B, mode='full')
    term2 = np.convolve(w, h, mode='full')
    off = n - 1
    grad = term1[off:off + n] - term2[off:off + n]
    return grad, corr
