def _slsqp_optimize(h0, target, n, dx, nfft, alpha=50.0, max_iter=800):
    # objective returns smooth max
    def obj(v):
        corr = _full_corr_fft(v, 1 - v, n, nfft) * dx
        max_val = np.max(corr)
        shifted = corr - max_val
        exp_a = np.exp(alpha * shifted)
        sum_exp = np.sum(exp_a)
        return max_val + np.log(sum_exp) / alpha
    def grad(v):
        corr = _full_corr_fft(v, 1 - v, n, nfft) * dx
        max_val = np.max(corr)
        shifted = corr - max_val
        exp_a = np.exp(alpha * shifted)
        sum_exp = np.exp(alpha * shifted).sum()
        w = exp_a / sum_exp
        # compute gradient similar to _smooth_max_and_grad but with w already.
        g = np.zeros_like(v)
        for idx, weight in enumerate(w):
            shift = idx - (n - 1)
            if shift >= 0:
                s = shift
                g[:n-s] += weight * (1 - v[s:])
                g[n-s:] += weight * 1.0
                g[s:] += weight * (-v[:n-s])
            else:
                s = -shift
                g[s:] += weight * (1 - v[:n-s])
                g[:s] += weight * 1.0
                g[:n-s] += weight * (-v[s:])
        return g
    cons = ({'type':'eq', 'fun': lambda v: np.sum(v) - target,
             'jac': lambda v: np.ones_like(v)},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # final projection onto sum constraint (since rounding may disturb)
    return _project_onto_capped_simplex(x, target)

import numpy as np
from scipy.optimize import minimize

# ---------- projection onto the capped simplex ----------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} using a bisection on the dual."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ---------- gradient for a given shift ----------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the max‑correlation term for a specific shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ---------- FFT based full correlation ----------
def _full_corr_fft(a: np.ndarray, b: np.ndarray, n: int, nfft: int) -> np.ndarray:
    """Linear full correlation a * b (like np.correlate(a,b,'full'))."""
    fa = np.fft.rfft(a, nfft)
    fb_rev = np.fft.rfft(b[::-1], nfft)
    conv = np.fft.irfft(fa * fb_rev, nfft)
    return conv[: 2 * n - 1]


# ---------- C5 evaluation ----------
def _compute_c5(h: np.ndarray, dx: float, n: int, nfft: int) -> float:
    corr = _full_corr_fft(h, 1.0 - h, n, nfft) * dx
    return float(corr.max())


# ---------- projected gradient descent on the true max ----------
def _gd_optimize(h0: np.ndarray, target: float, n: int, dx: float, nfft: int,
                 max_iter: int = 1200, step0: float = 0.5, decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    for _ in range(max_iter):
        corr = _full_corr_fft(h, 1.0 - h, n, nfft) * dx
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ---------- smooth max and its gradient ----------
def _smooth_max_and_grad(v: np.ndarray, alpha: float, dx: float,
                         n: int, nfft: int):
    """Return smooth max of correlations and its gradient (exact)."""
    corr = _full_corr_fft(v, 1.0 - v, n, nfft) * dx
    m = np.max(corr)
    shifted = corr - m
    exp_a = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_a)
    smooth = m + math.log(sum_exp) / alpha
    w = exp_a / sum_exp

    grad = np.zeros_like(v)
    for idx, weight in enumerate(w):
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            grad[: n - s] += weight * (1.0 - v[s:])
            grad[n - s :] += weight * 1.0
            grad[s:] += weight * (-v[: n - s])
        else:
            s = -shift
            grad[s:] += weight * (1.0 - v[: n - s])
            grad[: s] += weight * 1.0
            grad[: n - s] += weight * (-v[s:])
    return smooth, grad


# ---------- smooth gradient descent ----------
def _smooth_gd_optimize(h0: np.ndarray, target: float, n: int, dx: float,
                        nfft: int, alpha: float = 80.0,
                        max_iter: int = 200, step0: float = 0.2,
                        decay: float = 0.99) -> np.ndarray:
    h = h0.copy()
    step = step0
    for _ in range(max_iter):
        _, grad = _smooth_max_and_grad(h, alpha, dx, n, nfft)
        h = h - step * grad
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ---------- SLSQP refinement on smooth max ----------
def _slsqp_optimize(h0: np.ndarray, target: float, n: int, dx: float,
                    nfft: int, alpha: float = 80.0, max_iter: int = 600):
    def obj(v):
        corr = _full_corr_fft(v, 1.0 - v, n, nfft) * dx
        m = np.max(corr)
        shifted = corr - m
        exp_a = np.exp(alpha * shifted)
        return m + math.log(np.sum(exp_a)) / alpha

    def jac(v):
        corr = _full_corr_fft(v, 1.0 - v, n, nfft) * dx
        m = np.max(corr)
        shifted = corr - m
        exp_a = np.exp(alpha * shifted)
        w = exp_a / np.sum(exp_a)
        grad = np.zeros_like(v)
        for idx, weight in enumerate(w):
            shift = idx - (n - 1)
            if shift >= 0:
                s = shift
                grad[: n - s] += weight * (1.0 - v[s:])
                grad[n - s :] += weight * 1.0
                grad[s:] += weight * (-v[: n - s])
            else:
                s = -shift
                grad[s:] += weight * (1.0 - v[: n - s])
                grad[: s] += weight * 1.0
                grad[: n - s] += weight * (-v[s:])
        return grad

    cons = ({
        'type': 'eq',
        'fun': lambda v: np.sum(v) - target,
        'jac': lambda v: np.ones_like(v)
    },)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-11, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h on [0,2] that (hopefully) improves the Erdos C5 bound.
    Output: (h_values, c5_bound, n_points)
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    # ----- base vector (if provided) -----
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    n0 = len(base_h) if base_h is not None else 0
    # choose a richer discretisation; at least 400 points
    n = max(400, n0)
    target = n / 2.0
    dx = 2.0 / n
    nfft = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ----- helper to interpolate base solution onto the new grid -----
    def _upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ----- initialise best candidate -----
    if base_h is not None:
        best_h = _upscale(base_h)
        best_h = _project_onto_capped_simplex(best_h, target)
        best_val = _compute_c5(best_h, dx, n, nfft)
    else:
        best_h = None
        best_val = float('inf')

    restart = 0
    while time.time() - start < budget_s * 0.90:
        # ------- generate a starting point -------
        if best_h is not None and rng.rand() < 0.15:
            h0 = best_h + rng.normal(scale=0.015, size=n)
        else:
            v = rng.rand(n)
            h0 = _project_onto_capped_simplex(v, target)

        # ------- deterministic GD phase -------
        h_opt = _gd_optimize(h0, target, n, dx, nfft,
                             max_iter=1200, step0=0.55, decay=0.997)

        # ------- occasional smooth‑GD refinement -------
        if restart % 4 == 0:
            h_opt = _smooth_gd_optimize(h_opt, target, n, dx, nfft,
                                        alpha=80.0, max_iter=200,
                                        step0=0.22, decay=0.99)

        # ------- occasional SLSQP fine‑tune -------
        if restart % 5 == 0:
            h_opt = _slsqp_optimize(h_opt, target, n, dx, nfft,
                                    alpha=80.0, max_iter=600)

        # ------- evaluate -------
        val = _compute_c5(h_opt, dx, n, nfft)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        restart += 1

    # ----- final safety projection -----
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target)
        best_val = _compute_c5(best_h, dx, n, nfft)
    else:
        best_h = _project_onto_capped_simplex(best_h, target)
        best_val = _compute_c5(best_h, dx, n, nfft)

    return (best_h, float(best_val), n)
