import time, random, math
import numpy as np
from scipy.optimize import minimize

TARGET_SUM_GLOBAL = None  # used by sum_eq_constraint


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n intersect sum(x)=target via Lagrange bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretized C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Log‑sum‑exp smooth maximum."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth objective for SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular bump of height 1 centred at 1."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Indicator of interval [0.5,1.5] (width 1)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = ((pts >= 0.5) & (pts <= 1.5)).astype(float)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def random_feasible(n: int) -> np.ndarray:
    """Uniform random point projected to feasibility."""
    x = np.random.rand(n)
    return project_onto_bounds_and_sum(x, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function yielding a low C5 bound."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # problem size – a modest increase from 250 gives more flexibility
    n = 300                      # must be even for half‑mass constraint
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # Construct several diverse seeds
    seed_vectors = []

    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))
    seed_vectors.append(make_rectangle(n))

    # ------------------------------------------------------------------
    best_h = None
    best_val = np.inf
    restart = 0
    alphas = [20.0, 100.0, 500.0]          # continuation of smoothing

    # ------------------------------------------------------------------
    # Main global search – SLSQP from many starts
    while time.time() - start < budget_s * 0.7:
        if restart < len(seed_vectors):
            h0 = seed_vectors[restart]
        else:

        restart += 1

        for alpha in alphas:

                    h0,
