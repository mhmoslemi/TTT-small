def upsample_h(h, n_new):
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new = np.interp(x_new, x_old, h)
    # enforce sum scaling
    target = n_new/2.0
    cur = new.sum()
    if cur != 0:
        new *= target / cur
    np.clip(new, 0.0, 1.0, out=new)
    deficit = target - new.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n_new):
            if 0.0 <= new[idx] + deficit <= 1.0:
                new[idx] += deficit
                break
    return new

def _c5_metric(h: np.ndarray) -> float:
    """Integral upper bound C5 for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _make_initial_h(n: int) -> np.ndarray:
    """Create a feasible start vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)

    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h


def _upsample_h(h: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of h onto a finer grid of size n_new."""
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)

    target = n_new / 2.0
    cur = h_new.sum()
    if cur != 0.0:
        h_new *= target / cur
    np.clip(h_new, 0.0, 1.0, out=h_new)

    deficit = target - h_new.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n_new):
            if 0.0 <= h_new[idx] + deficit <= 1.0:
                h_new[idx] += deficit
                break
    return h_new


def _anneal(start_h: np.ndarray, max_seconds: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing optimisation of C5."""
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n

    # initial state
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_c = np.max(corr) * dx
    best_h = h.copy()
    best_c = cur_c

    # SA schedule parameters
    T0 = 0.06              # initial temperature
    T_end = 1e-6
    base_step = 0.2        # maximal move size at high temperature

    start = time.time()
    while True:
        now = time.time()
        if now - start >= max_seconds:
            break

        progress = (now - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        max_step = base_step * (T / T0 + 0.1)

        # identify the worst shift and its contribution profile
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)          # integer shift in samples

        cont = np.zeros(n, dtype=np.float64)
        if shift >= 0:
            if shift < n:
                cont[: n - shift] = h[: n - shift] * (1.0 - h[shift:])
        else:
            s = -shift

                cont[s:] = h[s:] * (1.0 - h[: n - s])

        # decide move type

                else:

        hi, hj = h[i], h[j]

        # apply the proposed move

            h[i] = hi
            h[j] = hj
