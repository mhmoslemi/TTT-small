def upsample(arr, new_n):
    old_n = len(arr)
    old_x = np.linspace(0, 2, old_n, endpoint=False)+ 1.0/old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False)+1.0/new_n
    up = np.interp(new_x, old_x, arr)
    # adjust sum to exact target
    target = new_n/2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

def slsqp_refine(h0, n, time_deadline):
    if time.time() > time_deadline - 1: return None, None
    target_sum = n / 2.0
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n
    # use high alpha
    res = minimize(lambda h: softmax_c5(h, alpha=60.0),
                   h0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
    if not res.success:
        return None, None
    h_opt = np.clip(res.x, 0.0, 1.0)
    c5 = compute_c5(h_opt)
    return h_opt, c5

def generate_triangular(n):
    x = np.linspace(0,2,n,endpoint=False)+1.0/n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def perturb_vector(v, eps, rng, target):
    new = v + rng.uniform(-eps, eps, size=v.shape)
    new = np.clip(new, 0.0, 1.0)
    diff = target - new.sum()
    if abs(diff) > 1e-12:
        new += diff / len(v)
        new = np.clip(new, 0.0, 1.0)
    return new

def local_search(start_h, time_deadline, rng):
    n = len(start_h)
    target = n/2.0
    best_h = start_h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step = 0.2
    iter = 0
    no_improve = 0
    while time.time() < time_deadline:
        iter += 1
        # maybe adapt step
        if iter % 50000 == 0:
            step *= 0.5
            step = max(step, 1e-5)
        # propose move
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        hi = cur_h[i]; hj = cur_h[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            continue
        delta = (rng.random() - 0.5) * 2 * step
        dt = np.clip(delta, lower, upper)
        if abs(dt) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] = hi + dt
        new_h[j] = hj - dt
        new_c5 = compute_c5(new_h)
        if new_c5 + 1e-12 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        else:
            no_improve += 1

        # Random restart if stuck
        if no_improve >= 200000:
            # perturb best
            eps = step * 2
            cur_h = perturb_vector(best_h, eps, rng, target)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0
    return best_h, best_c5

    time_limit = start_time + budget_s * 0.95

    n = 200

            start_h = upsample(parent_h, n)
        else:

        parent_h = None
        start_h = generate_triangular(n)

    # allocate time for local search and refinement
    # Ensure at least 10 seconds for refine
    refine_time = 20.0
    local_search_deadline = time_limit - refine_time
    if local_search_deadline < time.time():
        local_search_deadline = time.time() + 0.1

    # Run local search
    best_h, best_c5 = local_search(start_h, local_search_deadline, rng)

    # Run refinement if time left
    remaining = time_limit - time.time()
    if remaining > 5.0:
        refined_h, refined_c5 = slsqp_refine(best_h, n, time_limit)
        if refined_h is not None and refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # Ensure sum constraint exact
    target = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def softmax_c5(h: np.ndarray, alpha: float = 60.0) -> float:
    """Smooth approximation of max_k overlap for SLSQP refinement."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def perturb_vector(v: np.ndarray, eps: float, rng: np.random.Generator, target: float) -> np.ndarray:
    """Add small uniform noise and re‑project to the feasible affine subspace."""
    w = v + rng.uniform(-eps, eps, size=v.shape)
    w = np.clip(w, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        w += diff / w.size
        w = np.clip(w, 0.0, 1.0)
    return w

# Local stochastic search (annealed hill‑climb)
# ------------------------------------------------------------
def local_search(start_h: np.ndarray, deadline: float, rng: np.random.Generator):
    n = start_h.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    total_time = deadline - start_time
    directed_prob = 0.3          # probability of a gradient‑guided move
    max_step = 0.2               # initial transfer amount
    min_step = 1e-5

    iter_no_improve = 0
    while time.time() < deadline:
        now = time.time()
        remaining = deadline - now
        if remaining <= 0:
            break

        # temperature for simulated annealing (linear schedule)
        T = max(1e-6, remaining / total_time) * 0.01

        # step size shrinks with remaining time
        step = max(min_step, max_step * remaining / total_time)

        # current correlation – needed for directed moves
        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(conv.argmax())
        shift = max_idx - (n - 1)          # integer shift attaining current max

        # decide move type
        if rng.random() < directed_prob:
            # compute derivative d_i = 1 - (h_{i+shift}+h_{i-shift})
            d = np.ones(n, dtype=float)
            if shift >= 0:
                if shift < n:
                    d[:n - shift] -= cur_h[shift:]
                    d[shift:]   -= cur_h[:n - shift]
            else:
                s = -shift
                if s < n:
                    d[s:]      -= cur_h[:n - s]
                    d[:n - s]  -= cur_h[s:]

            pos = np.where((d > 1e-12) & (cur_h > 1e-12))[0]
            neg = np.where((d < -1e-12) & (cur_h < 1.0 - 1e-12))[0]
            if pos.size == 0 or neg.size == 0:
                # fall back to a random transfer
                i, j = rng.integers(0, n, size=2, endpoint=False)
                while i == j:
                    j = rng.integers(0, n, endpoint=False)
                max_dec = cur_h[i]
                max_inc = 1.0 - cur_h[j]
                delta = min(step, max_dec, max_inc)
                if delta <= 0:
                    continue
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_dec = cur_h[i]
                max_inc = 1.0 - cur_h[j]
                delta = min(step, max_dec, max_inc)
                if delta <= 0:
                    continue
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
        else:
            # pure random transfer between two distinct indices
            i, j = rng.integers(0, n, size=2, endpoint=False)
            while i == j:
                j = rng.integers(0, n, endpoint=False)
            max_dec = cur_h[i]
            max_inc = 1.0 - cur_h[j]
            delta = min(step, max_dec, max_inc)
            if delta <= 0:
                continue
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        # record global best
        if cur_c5 < best_c5:
            best_h = cur_h.copy()
            best_c5 = cur_c5
            iter_no_improve = 0

        # occasional rejuvenation if stuck
        if iter_no_improve >= 200_000:
            eps = step * 2.0
            cur_h = perturb_vector(best_h, eps, rng, target)
            cur_c5 = compute_c5(cur_h)
            iter_no_improve = 0

    return best_h, best_c5


# ------------------------------------------------------------
# Final SLSQP refinement (smooth max)
# ------------------------------------------------------------
def slsqp_refine(h0: np.ndarray, n: int, deadline: float):
    if time.time() > deadline - 5.0:
        return None, None
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    opts = {"maxiter": 400, "ftol": 1e-12, "disp": False}
    try:
        res = minimize(
            lambda h: softmax_c5(h, alpha=80.0),
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options=opts,
        )
    except Exception:
        return None, None
    if not res.success:
        return None, None
    h_opt = np.clip(res.x, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    # --------------------------------------------------------
    # Choose a working resolution (even number helps feasibility)
    # --------------------------------------------------------
    n = 200
    target_sum = n / 2.0

    # Build an initial feasible vector
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        if parent_h.shape[0] != n:
            start_h = upsample(parent_h, n)
        else:
            start_h = parent_h.copy()
    except NameError:
        parent_h = None
        start_h = generate_triangular(n)

    # --------------------------------------------------------
    # Stochastic annealed hill‑climb
    # --------------------------------------------------------
    hill_deadline = time_limit - 20.0  # reserve a slice for final refinement
    if hill_deadline < time.time() + 1.0:
        hill_deadline = time.time() + 1.0
    best_h, best_c5 = local_search(start_h, hill_deadline, rng)

    # --------------------------------------------------------
    # Optional refinement via SLSQP (smooth approximation)
    # --------------------------------------------------------
    if time.time() < time_limit - 5.0:
        refined_h, refined_c5 = slsqp_refine(best_h, n, time_limit)
        if refined_h is not None and refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # --------------------------------------------------------
    # Final feasibility clean‑up
    # --------------------------------------------------------
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h + (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
