def project_feasible(h, target):
    # target = n/2
    # clip
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = total - target
    # To correct, we can add/subtract diff uniformly among entries not at bounds.
    # While diff != 0 and there are adjustable entries:
    while abs(diff) > 1e-12:
        # Find indices where we can adjust (if diff>0 need to reduce values >0)
        if diff > 0:
            adjustable = np.where(h > 0)[0]
            if adjustable.size == 0: break
            # reduction per element
            delta = diff / adjustable.size
            # can't go below zero
            reduction = np.minimum(h[adjustable], delta)
            h[adjustable] -= reduction
            diff = h.sum() - target
        else:
            adjustable = np.where(h < 1)[0]
            if adjustable.size == 0: break
            delta = -diff / adjustable.size
            increase = np.minimum(1 - h[adjustable], delta)
            h[adjustable] += increase
            diff = h.sum() - target
        # break if no adjustment
        if np.allclose(diff, 0, atol=1e-12):
            break
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def gradient_of_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        k = lag
        # term1
        g[:n-k] += 1 - h[k:]
        # term2
        g[k:] -= h[:n-k]
    else:
        k = -lag
        g[k:] += 1 - h[:n-k]
        g[:n-k] -= h[k:]
    return g

def descent(h, max_iters, time_check, target_bound):
    n = len(h)
    target = n/2.0
    best = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    # step schedule:
    alpha = 0.5  # may tune
    for it in range(max_iters):
        if time_check() < 0.5:
            break
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = corr.max()
        lag = int(np.argmax(corr) - (n-1))
        # If max_corr < best_c5? Actually we are evaluating same.
        c5 = max_corr * (2.0 / n)
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best = h.copy()
            if best_c5 < target_bound:
                break

        # compute gradient of current worst lag
        g = gradient_of_lag(h, lag)
        # try step with line search
        # Choose list of alphas: decreasing
        for factor in (0.5, 0.25, 0.125, 0.0625):
            step = alpha * factor

                h = h_new
                break
        else:
            # if no improvement, reduce alpha
            alpha *= 0.5
            if alpha < 1e-6:
                break
    return best, best_c5

def random_local_search(h, max_steps, time_left):
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for s in range(max_steps):
        if time_left() < 0.5:
            break
        # pick two indices different
        i, j = np.random.randint(0, n, size=2)
        if i==j:
            continue
        # Determine possible transfer amount
        # want to move delta from i to j (i reduces, j increases)
        # cap delta to ensure bounds
        max_decrease = best_h[i]  # cannot go below 0
        max_increase = 1 - best_h[j] # cannot exceed 1
        max_delta = min(max_decrease, max_increase)
        if max_delta <= 0:
            continue
        delta = max_delta * (np.random.rand() * 0.5 + 0.5)  # random fraction 0.5-1
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # sum remains same, bounds remain.
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5

def project_feasible(h, target):

    total = h.sum()
    diff = target - total
    # tolerance
    tol = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h < 1)[0]
            if idx.size == 0:
                break
            # amount each gets
            inc = diff / idx.size
            # limit inc to not exceed 1
            inc_limited = np.minimum(inc, 1 - h[idx])
            # Update
            h[idx] += inc_limited
            # diff becomes target - sum after adding limited inc (some may be capped)
            total = h.sum()
            diff = target - total
        else:
            idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            dec = -diff / idx.size
            dec_limited = np.minimum(dec, h[idx])
            h[idx] -= dec_limited
            total = h.sum()
            diff = target - total
    # final clip
    h = np.clip(h,0.0,1.0)
    return h

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target - total
    eps = 1e-12
    # iterative water-filling
    for _ in range(1000):
        if abs(diff) <= eps:
            break
        if diff > 0:
            # Need to increase
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            # Increase but not exceed 1
            inc_possible = np.minimum(inc, 1.0 - h[idx])
            # Some entries may not take full inc, re-evaluate diff.
            h[idx] += inc_possible
            diff = target - h.sum()
        else:
            # Need to decrease
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            dec = -diff / idx.size
            dec_possible = np.minimum(dec, h[idx])
            h[idx] -= dec_possible
            diff = target - h.sum()
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

def softmin_ascent(h, max_iters, target_bound, time_left):
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 1.0
    for it in range(max_iters):
        if time_left() < 0.5:
            break
        # compute autocorr
        autoc = np.correlate(h, h, mode='full')
        # mask zero lag
        center = n-1
        # compute weights for all non-zero lags
        excl = np.arange(2*n-1) != center
        # to avoid overflow large exponent may produce underflow, but fine.
        # Compute softmin weights
        # Use stability: shift autoc values by min
        # But exponential negative large still fine.
        w_arr = np.zeros_like(autoc)
        masked_vals = autoc[excl]
        # we want softmin: weight_i = exp(-beta * val_i) / sum_exp
        # For numerical stability, subtract min:
        m = masked_vals.min()
        exp_vals = np.exp(-beta * (masked_vals - m))
        sum_exp = exp_vals.sum()
        if sum_exp == 0:
            # all weight zero? then break.
            break
        w_arr[excl] = exp_vals / sum_exp
        # compute gradient
        grad = np.zeros(n)
        # Loop over k from 1..n-1
        for k in range(1, n):
            wk = w_arr[center + k] + w_arr[center - k]
            if wk == 0:
                continue
            # forward shift
            grad[:n-k] += wk * h[k:]
            # backward shift
            grad[k:] += wk * h[:n-k]
        # set step size
        # step scaling maybe 0.1 / (beta+1)
        step = 0.5 / (beta + 1.0)
        # update
        h_new = h + step * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
            if best_c5 < target_bound:
                break
        # accept update if improves or sometimes
        if c5_new <= compute_c5(h) or np.random.rand() < 0.1:
            h = h_new
        # increase beta slowly
        beta = min(beta * 1.02, 200.0)  # limit sigma
    return best_h, best_c5

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Feasibility projector: clip to [0,1] and enforce sum(h)=target (n/2)
def _project_feasible(h: np.ndarray, target: float) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    eps = 1e-12
    for _ in range(1000):
        if abs(diff) <= eps:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            inc_possible = np.minimum(inc, 1.0 - h[idx])
            h[idx] += inc_possible
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            dec = -diff / idx.size
            dec_possible = np.minimum(dec, h[idx])
            h[idx] -= dec_possible
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


# ----------------------------------------------------------------------
# C5 evaluation for a discrete step function
def _c5(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


# ----------------------------------------------------------------------
# Gradient of the worst‑lag correlation (used in a simple projected GD)
def _grad_worst_lag(h: np.ndarray, lag: int) -> np.ndarray:
    n = h.size
    g = np.zeros(n)
    if lag >= 0:
        k = lag
        g[: n - k] += 1.0 - h[k:]
        g[k:] -= h[: n - k]
    else:
        k = -lag
        g[k:] += 1.0 - h[: n - k]
        g[: n - k] -= h[k:]
    return g


def _gd_worst(h0: np.ndarray, max_iters: int, target: float,
              time_left, target_bound) -> tuple:
    h = h0.copy()
    n = h.size
    best_h, best_c5 = h.copy(), _c5(h)
    step = 0.6
    for _ in range(max_iters):
        if time_left() < 0.5:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        lag = int(np.argmax(corr) - (n - 1))
        cur_c5 = corr.max() * (2.0 / n)
        if cur_c5 < best_c5 - 1e-12:
            best_c5, best_h = cur_c5, h.copy()
            if best_c5 < target_bound:
                break
        g = _grad_worst_lag(h, lag)
        improved = False
        for factor in (0.5, 0.25, 0.125, 0.0625):
            nh = _project_feasible(h - step * factor * g, target)
            new_c5 = _c5(nh)
            if new_c5 <= cur_c5 + 1e-12:
                h, improved = nh, True
                break
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
    return best_h, best_c5


# ----------------------------------------------------------------------
# Soft‑min ascent (maximises the minimal auto‑correlation)
def _softmin_ascent(h0: np.ndarray, max_iters: int, target: float,
                    time_left, target_bound) -> tuple:
    h = h0.copy()
    n = h.size
    best_h, best_c5 = h.copy(), _c5(h)
    beta = 0.5
    for it in range(max_iters):
        if time_left() < 0.5:
            break
        autoc = np.correlate(h, h, mode="full")
        center = n - 1
        mask = np.arange(2 * n - 1) != center
        vals = autoc[mask]
        # stability shift
        m = vals.min()
        exp_vals = np.exp(-beta * (vals - m))
        sum_exp = exp_vals.sum()
        if sum_exp == 0:
            break
        w = np.zeros_like(autoc)
        w[mask] = exp_vals / sum_exp

        grad = np.zeros(n)
        for k in range(1, n):
            wk = w[center + k] + w[center - k]
            if wk == 0.0:
                continue
            grad[: n - k] += wk * h[k:]
            grad[k:] += wk * h[: n - k]

        step = 0.5 / (beta + 1.0)
        nh = _project_feasible(h + step * grad, target)
        cur_c5 = _c5(h)
        new_c5 = _c5(nh)

        if new_c5 < best_c5:
            best_c5, best_h = new_c5, nh.copy()
            if best_c5 < target_bound:
                break

        # accept if not worsening (or with small probability)
        if new_c5 <= cur_c5 or np.random.rand() < 0.08:
            h = nh
        beta = min(beta * 1.07, 5.0)   # keep beta moderate to avoid underflow
    return best_h, best_c5


# ----------------------------------------------------------------------
# Simple random local improvement (swap a small mass)
def _random_local_search(h0: np.ndarray, max_steps: int,
                         time_left) -> tuple:
    h = h0.copy()
    n = h.size
    best_h, best_c5 = h.copy(), _c5(h)
    for _ in range(max_steps):
        if time_left() < 0.5:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer from i to j
        delta = min(h[i], 1.0 - h[j])
        if delta <= 0:
            continue
        delta *= 0.5 + 0.5 * np.random.rand()   # random fraction
        nh = h.copy()
        nh[i] -= delta
        nh[j] += delta
        new_c5 = _c5(nh)
        if new_c5 < best_c5:
            best_c5, best_h = new_c5, nh.copy()
        if new_c5 < _c5(h):
            h = nh
    return best_h, best_c5


# ----------------------------------------------------------------------
# Upscale / interpolate a coarse step function
def _upscale(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse.size
    if n_fine == n_coarse:
        return coarse.copy()
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse)


# ----------------------------------------------------------------------
# Analytic initial shapes (triangular, cosine)
def _triangular(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    return _project_feasible(h, n / 2.0)


def _cosine(n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = (1.0 + np.cos(np.pi * x)) / 2.0
    return _project_feasible(h, n / 2.0)

    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------

    base_n = init_h.size if init_h is not None else 64

        # Build a base function for this resolution
        if init_h is not None:

        else:

        base = _project_feasible(base, n / 2.0)

        # also keep a couple of analytic seeds
        seeds = [base, _triangular(n), _cosine(n)]

        # ----------------------------------------------------------------
        # Main optimisation attempts for this grid size
        for seed_h in seeds:
            if time_left() < 6.0:

            # random perturbations + soft‑min ascent

                if time_left() < 5.0:

                if restart > 0:

                    h0 = _project_feasible(seed_h + noise, n / 2.0)

                # soft‑min ascent (global search)
                h_opt, c5_opt = _softmin_ascent(

                    max_iters=6000,

            if best_c5 < target_bound:

        if best_c5 < target_bound:
            break

    # ----------------------------------------------------------------------
    # Final polishing with a small SLSQP (if time permits)
    if best_h is not None and time_left() > 30.0:
        n = best_h.size
        def obj(x):
            return x[-1]

        def eqc(x):
            return np.sum(x[:n]) - n / 2.0

        def ineqc(x):
            t = x[-1]
            h = x[:n]
            corr = np.correlate(h, 1.0 - h, mode="full")
            return t - corr

        x0 = np.concatenate([best_h, [np.max(np.correlate(best_h, 1 - best_h, mode="full"))]])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        cons = [{"type": "eq", "fun": eqc}, {"type": "ineq", "fun": ineqc}]

    # Fallback (unlikely)
    if best_h is None:
        n = 64
        best_h = np.full(n, 0.5)
        best_c5 = _c5(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
