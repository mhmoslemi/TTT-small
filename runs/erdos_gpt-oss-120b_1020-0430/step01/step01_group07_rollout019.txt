def full_corr(h):
    n = len(h)
    m = 2*n-1
    N = 1 << (m-1).bit_length()
    fft_h = np.fft.rfft(np.pad(h, (0,N-n), 'constant'), n=N)

def compute_c5(h):
    n = h.size
    dx = 2.0/n
    # compute correlation h * (1 - h) quickly:
    # correlate h with (1-h) via FFT as conv of h with rev(1-h)?
    # or compute via correlation of h and ones then subtract self-correlation.
    # Corr_h_1mh = Corr_h_ones - Corr_h_h
    # Corr_h_ones is sum of overlapping h values, i.e., cross-correlation of h with all-ones array.
    # But we can compute Corr_h_h via FFT, and Corr_h_ones via prefix sums maybe easier.
    # Since Corr_h_ones for shift s can be computed quickly: It's sum_{i} h[i] where i and i+s in range.
    # That's similar to convolution of h with ones reversed, which can be computed with FFT as well.
    # Let's simply compute Corr_h_1mh = np.correlate(h, 1-h, mode='full') using FFT with subtraction.

    # correlate h with (1-h) = correlate h with ones - correlate h with h
    # correlate h with ones: length n array of ones.
    n_full = 2*n - 1
    N = 1 << (n_full - 1).bit_length()
    # pad h and ones
    padded_h = np.pad(h, (0,N-n), 'constant')
    fft_h = np.fft.rfft(padded_h)
    # ones array:
    ones = np.ones_like(h)
    padded_ones = np.pad(ones, (0,N-n), 'constant')
    fft_ones = np.fft.rfft(padded_ones)
    corr_h_ones = np.fft.irfft(fft_h * np.conj(fft_ones), n=N)[:n_full].real
    corr_h_h = np.fft.irfft(fft_h * np.conj(fft_h), n=N)[:n_full].real
    corr = corr_h_ones - corr_h_h
    c5 = np.max(corr) * dx
    return c5

import time, random, math
import numpy as np

# -------------------------------------------------------------
# Helper: compute C5 using FFT (fast autocorrelation)
# -------------------------------------------------------------
def _c5_metric(h, fft_len, target_sum, dx):
    """Return C5 for vector h (values in [0,1]) using FFT."""
    # autocorrelation of h with itself
    H = np.fft.rfft(np.pad(h, (0, fft_len - h.size), 'constant'))
    corr = np.fft.irfft(H * np.conj(H), n=fft_len)[: 2 * h.size - 1].real
    min_corr = corr.min()
    return (target_sum - min_corr) * dx


# -------------------------------------------------------------
# Build an initial feasible vector, up‑sample if a parent exists
# -------------------------------------------------------------
def _initial_vector(n):
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        # linear interpolation onto the new grid
        x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, base)
    else:
        # simple start: half the interval filled with ones, rest zeros
        h = np.zeros(n, dtype=np.float64)
        half = n // 2
        h[:half] = 1.0

    # enforce the integral constraint exactly (sum = n/2)
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur                      # global scaling keeps ratios
    np.clip(h, 0.0, 1.0, out=h)

    # tiny correction to hit the target sum precisely
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # find any entry that can absorb the deficit safely
        ok = np.where((h + diff >= 0) & (h + diff <= 1))[0]
        if ok.size:
            h[ok[0]] += diff
    return h

# Simulated annealing on the pair‑move neighbourhood
# -------------------------------------------------------------
def _anneal(start_h, time_budget, fft_len):
    n = start_h.size
    dx = 2.0 / n
    target = n / 2.0

    h = start_h.copy()
    cur_c = _c5_metric(h, fft_len, target, dx)
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    # temperature schedule
    T0 = 0.12
    T_end = 1e-6
    max_step = 0.18

    while time.time() - start < time_budget:
        # linear progress → exponential temperature decay
        prog = (time.time() - start) / time_budget
        T = T0 * (T_end / T0) ** prog

        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        # feasible delta for (i receives, j loses)
        lo = max(-hi, hj - 1.0)
        hi_cap = min(1.0 - hi, hj)
        if lo >= hi_cap:
            continue

        # usually stay inside a modest step, occasionally allow larger jumps
        lo2 = max(lo, -max_step)
        hi2 = min(hi_cap, max_step)
        if lo2 < hi2:
            delta = random.uniform(lo2, hi2)
        else:
            delta = random.uniform(lo, hi_cap)

        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _c5_metric(h, fft_len, target, dx)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

        # occasional tiny renormalisation to keep the sum exact
        if random.random() < 0.001:
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                cand = np.where((h + diff >= 0) & (h + diff <= 1))[0]
                if cand.size:
                    h[cand[0]] += diff
    return best_h, best_c

# Greedy refinement that directly attacks the worst lag
# -------------------------------------------------------------
def _greedy_refine(start_h, time_budget, fft_len):
    n = start_h.size
    dx = 2.0 / n
    target = n / 2.0

    h = start_h.copy()
    best_c = _c5_metric(h, fft_len, target, dx)

    start = time.time()
    while time.time() - start < time_budget:
        # autocorrelation (full)
        H = np.fft.rfft(np.pad(h, (0, fft_len - n), 'constant'))
        corr = np.fft.irfft(H * np.conj(H), n=fft_len)[: 2 * n - 1].real

        min_corr = corr.min()
        lag_idx = corr.argmin()
        lag = lag_idx - (n - 1)          # signed shift where overlap is minimal

        # weight = contribution of a site to the overlap at this lag
        w = np.zeros(n)
        if lag > 0:
            w[: n - lag] += h[lag:]
            w[lag:] += h[: n - lag]
        elif lag < 0:
            s = -lag
            w[: n - s] += h[s:]
            w[s:] += h[: n - s]
        else:  # lag == 0
            w = 2.0 * h

        inc_mask = h < 1.0 - 1e-12
        dec_mask = h > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            break

        inc_cands = np.where(inc_mask)[0]
        dec_cands = np.where(dec_mask)[0]

        inc = inc_cands[np.argmax(w[inc_cands])]
        dec = dec_cands[np.argmin(w[dec_cands])]

        benefit = w[inc] - w[dec]
        if benefit <= 1e-9:                        # no productive pair found
            # fallback: tiny random reshuffle
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_cap = min(1.0 - hi, hj)
            if lo >= hi_cap:
                continue
            delta = random.uniform(lo, hi_cap) * 0.05
            h[i] = hi + delta
            h[j] = hj - delta
            cur_c = _c5_metric(h, fft_len, target, dx)
            if cur_c + 1e-12 < best_c:
                best_c = cur_c
            else:
                h[i] = hi
                h[j] = hj
            continue

        max_delta = min(0.2, 1.0 - h[inc], h[dec])
        delta = min(max_delta, benefit * 0.35)   # step proportional to expected gain

        h[dec] -= delta

        cur_c = _c5_metric(h, fft_len, target, dx)
        if cur_c + 1e-12 < best_c:
            best_c = cur_c
        else:
            # revert if no improvement (rare)
            h[inc] -= delta
            h[dec] += delta
    return h, best_c
