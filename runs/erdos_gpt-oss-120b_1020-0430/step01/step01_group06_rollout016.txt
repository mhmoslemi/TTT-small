import numpy as np, cvxpy as cp, time

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    random = np.random.RandomState(seed)
    n = 70
    # Check if initial_h_values exists
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    except NameError:
        init = None
    # set target sum
    target = n // 2
    # Build MILP model
    h = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    w = cp.Variable(integer=True)
    constraints = [cp.sum(h) == target]
    # constraints linking Y with h
    for i in range(n):
        for j in range(n):
            constraints.append(Y[i, j] <= h[i])
            constraints.append(Y[i, j] <= h[j])
            constraints.append(Y[i, j] >= h[i] + h[j] - 1)
    # overlap constraints for each shift
    for s in range(-n+1, n):
        if s >= 0:
            idx = range(n - s)
            sum_expr = cp.sum(cp.hstack([Y[i, i+s] for i in idx]))
        else:
            s_abs = -s
            idx = range(s_abs, n)
            sum_expr = cp.sum(cp.hstack([Y[i, i - s_abs] for i in idx]))
        constraints.append(sum_expr >= w)
    objective = cp.Maximize(w)
    prob = cp.Problem(objective, constraints)
    # Solve with CBC or GLPK_MI
    try:
        prob.solve(solver=cp.CBC, timeLimit=budget_s-5)
    except cp.error.SolverError:
        prob.solve(solver=cp.GLPK_MI, timeLimit=budget_s-5)
    # Get solution h
    h_vals = np.round(h.value).astype(float)  # ensure 0/1
    # ensure exact sum due to rounding issues
    cur_sum = h_vals.sum()
    diff = cur_sum - target
    if abs(diff) > 1e-6:
        # adjust by flipping some bits to meet sum exactly (should not happen)
        # Simple fix: sort indices by fractional part closeness
        fracs = h.value - np.floor(h.value)
        if diff > 0:
            # need to turn some 1's to 0
            idx = np.argsort(fracs)  # smallest fractions first (i.e., near 0)
            for i in idx:
                if h_vals[i] == 1 and diff > 0:
                    h_vals[i] = 0
                    diff -= 1
                if diff <= 0:
                    break
        else:
            idx = np.argsort(-fracs)  # largest fractions first (near 1)
            for i in idx:
                if h_vals[i] == 0 and diff < 0:
                    h_vals[i] = 1
                    diff += 1
                if diff >= 0:
                    break
    # compute final C5 bound
    def c5_of(v):
        corr = np.correlate(v, 1 - v, mode="full")
        return float(corr.max() * (2.0 / len(v)))
    final_c5 = c5_of(h_vals)
    return (h_vals.tolist(), final_c5, n)

import math
import random
import time
import numpy as np


def _c5_of(v: np.ndarray) -> float:
    """Maximum overlap integral for a given step function v."""
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(corr.max() * (2.0 / v.size))


def _project_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the hyperplane sum(v)=target while staying in [0,1]."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) <= 1e-12:
        return v
    if cur > 0:
        v *= target / cur
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    # simple iterative correction
    while abs(cur - target) > 1e-9:
        diff = cur - target
        if diff > 0:
            cand = np.where(v > 1e-12)[0]
            i = random.choice(cand.tolist())
            delta = min(v[i], diff)
            v[i] -= delta
        else:
            cand = np.where(v < 1.0 - 1e-12)[0]
            i = random.choice(cand.tolist())
            delta = min(1.0 - v[i], -diff)
            v[i] += delta
        cur = v.sum()
    return v


def _targeted_move(h: np.ndarray, s_idx: int) -> tuple:
    """Perform one targeted transfer on the worst shift."""
    n = h.size
    s = s_idx - (n - 1)          # actual shift
    if s >= 0:
        i_inds = np.arange(n - s)
        p_inds = i_inds + s
    else:
        s_abs = -s
        i_inds = np.arange(s_abs, n)
        p_inds = i_inds + s      # negative shift
    contrib = h[i_inds] * (1.0 - h[p_inds])
    k = int(np.argmax(contrib))
    i, p = i_inds[k], p_inds[k]
    eps_max = min(h[i], 1.0 - h[p])
    if eps_max <= 1e-12:
        return None, None, None
    # try a large step, but keep it a bit conservative
    eps = eps_max * 0.75
    nh = h.copy()
    nh[i] -= eps
    nh[p] += eps
    return nh, i, p


def _local_improve(h: np.ndarray, budget: float) -> np.ndarray:
    """Greedy improvement using full transfers on the worst shift."""
    start = time.time()
    n = h.size
    while time.time() - start < budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        s_idx = int(np.argmax(corr))
        s = s_idx - (n - 1)
        if s >= 0:
            i_inds = np.arange(n - s)
            p_inds = i_inds + s
        else:
            s_abs = -s
            i_inds = np.arange(s_abs, n)
            p_inds = i_inds + s
        contrib = h[i_inds] * (1.0 - h[p_inds])
        k = int(np.argmax(contrib))
        i, p = i_inds[k], p_inds[k]
        eps_max = min(h[i], 1.0 - h[p])
        if eps_max <= 1e-12:
            break
        # try full transfer; accept only if it improves
        nh = h.copy()
        nh[i] -= eps_max
        nh[p] += eps_max
        if _c5_of(nh) < _c5_of(h) - 1e-12:
            h = nh
            continue
        # try half transfer
        eps = eps_max * 0.5
        nh[i] -= eps
        nh[p] += eps
        if _c5_of(nh) < _c5_of(h) - 1e-12:
            h = nh
            continue
        break
    return h

        h = np.random.rand(n)

    target = n / 2.0
    h = _project_sum(h, target)

    cur_c5 = _c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # Simulated annealing with three move types
    # ----------------------------------------------------------------------
    T = max(0.01, cur_c5 * 0.05)      # initial temperature
    decay = 0.9997
    it = 0

    while time.time() - start < budget_s * 0.95:
        move_type = random.random()
        if move_type < 0.4:
            # -------- random two‑coordinate transfer (mass conservation) -------
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            max_up = min(1.0 - hi, hj)               # increase i, decrease j
            max_down = min(hi, 1.0 - hj)             # decrease i, increase j
            if max_up + max_down == 0:
                continue
            delta = (random.random() * (max_up + max_down)) - max_down
            nh = h.copy()
            nh[i] = hi + delta
            nh[j] = hj - delta
        elif move_type < 0.8:
            # --------------------- targeted transfer -------------------------
            corr = np.correlate(h, 1.0 - h, mode="full")
            s_idx = int(np.argmax(corr))
            nh, i, p = _targeted_move(h, s_idx)
            if nh is None:
                continue
        else:
            # ----------------------- tiny perturbation -----------------------
            i = random.randrange(n)
            # pick a second index different from i
            j = random.randrange(n - 1)
            if j >= i:
                j += 1
            hi, hj = h[i], h[j]
            max_up = min(1.0 - hi, hj)
            max_down = min(hi, 1.0 - hj)
            if max_up + max_down == 0:
                continue
            delta = (random.random() * (max_up + max_down)) - max_down
            nh = h.copy()
            nh[i] = hi + delta
            nh[j] = hj - delta

        new_c5 = _c5_of(nh)

        # Acceptance rule
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

        it += 1
        if it % 500 == 0:
            T = max(1e-6, T * decay)

        # occasional full restart if we haven't improved for a long stretch
        if it % 20000 == 0 and best_c5 < cur_c5 * 0.99:
            # keep current best, but re‑seed a fresh candidate
            h = np.random.rand(n)
            h = _project_sum(h, target)
            cur_c5 = _c5_of(h)

    # ----------------------------------------------------------------------
    # Final deterministic fine‑tuning (short budget)
    # ----------------------------------------------------------------------
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        best_h = _local_improve(best_h, budget=5.0)

    # ----------------------------------------------------------------------
    # Small numerical clean‑up of the sum constraint
    # ----------------------------------------------------------------------
    err = best_h.sum() - target
    if abs(err) > 1e-8:
        # distribute the error over a few coordinates
        for _ in range(100):
            if abs(err) < 1e-12:
                break
            i = random.randrange(n)
            if err > 0 and best_h[i] > 0:
                delta = min(err, best_h[i])
                best_h[i] -= delta
                err -= delta
            elif err < 0 and best_h[i] < 1:
                delta = min(-err, 1.0 - best_h[i])
                best_h[i] += delta
                err += delta

    final_c5 = _c5_of(best_h)

    return (best_h.tolist(), final_c5, n)
