import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    # correlation of h with 1-h, full mode
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct any drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_binary_feasible(n, rng):
    h = np.zeros(n)
    h[: n // 2] = 1.0
    rng.shuffle(h)
    return h

def neighbor_transfer(h, rng, max_eps=0.5):
    """Transfer a random amount from i to j, preserving sum."""
    n = h.size
    # pick source i that is >0 and dest j that is <1
    # try a few times to find suitable pair
    for _ in range(10):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] <= 1e-12 or h[j] >= 1 - 1e-12:
            continue
        limit = min(h[i], 1.0 - h[j])
        if limit <= 0:
            continue
        eps = rng.random() * limit * max_eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h
    # fallback: simple swap for binary vectors
    new_h = h.copy()
    return new_h

def simulated_annealing(h0, rng, max_iter, T0=0.02, Tf=0.0005):
    h_curr = h0.copy()
    c_curr = compute_c5(h_curr)
    best_h = h_curr.copy()
    best_c = c_curr
    # exponential cooling schedule
    for it in range(max_iter):
        # temperature
        T = T0 * (Tf / T0) ** (it / max_iter)
        h_new = neighbor_transfer(h_curr, rng, max_eps=0.9)
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-16)):
            h_curr = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

# ----------------------------------------------------------------------
# Global helpers for the SLSQP surrogate (soft‑max of the overlaps)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for a step vector h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # correct any tiny drift in the sum
    diff = CURRENT_TARGET - x.sum()

        h_pat = np.clip(h_pat, 0.0, 1.0)
        h_pat = adjust_sum(h_pat, base_n / 2.0, rng)
        c_pat = compute_c5(h_pat)
        if c_pat < best_c:
            best_h, best_c = h_pat.copy(), c_pat
        # short annealing refinement
        h_ref, c_ref = simulated_annealing(h_pat, rng, max_iter=8000,
                                           T0=0.025, Tf=0.0005)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Main stochastic search loop (SA + occasional SLSQP polishing)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.85:
        # pick a starting vector: either a perturbed best or a fresh binary sample
        if best_h is not None and rng.random() < 0.30:
            start_h = neighbor_transfer(best_h, rng, max_eps=0.9)
        else:
            start_h = random_binary_feasible(base_n, rng)

        # allocate iterations proportionally to remaining time
        elapsed = time.time() - start
        remaining = budget_s * 0.85 - elapsed
        max_iter = int(min(25000, max(6000, remaining * 1800)))

        h_sa, c_sa = simulated_annealing(start_h, rng, max_iter,
                                         T0=0.03, Tf=0.0005)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

        # quick SLSQP polishing (soft‑max surrogate)
        if time.time() - start < budget_s * 0.90:
            try:
                h_ref = slsqp_optimize(best_h, beta=15.0, max_iter=200)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref, c_ref
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Optional up‑sampling to 200 points if time permits
    # ------------------------------------------------------------------
    if time.time() - start < budget_s * 0.97 and base_n * 2 <= 200:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        remaining = budget_s - (time.time() - start)
        max_iter_up = int(min(12000, max(4000, remaining * 1500)))
        h_up, c_up = simulated_annealing(h_up, rng, max_iter_up,
                                         T0=0.02, Tf=0.0005)
        for beta in (5.0, 20.0, 80.0):
            if time.time() - start > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta=beta, max_iter=200)
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, base_n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final polishing passes with increasing beta values
    # ------------------------------------------------------------------
    for beta in (5.0, 20.0, 80.0):
        if time.time() - start > budget_s:
            break
        try:
            best_h = slsqp_optimize(best_h, beta=beta, max_iter=200)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Ensure exact feasibility before reporting
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, base_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(base_n)
