def improve_by_gradient(h, target_sum, step=0.05):
    s = argmin_autocorr_shift(h)
    n = len(h)
    g = np.zeros(n)
    if s >= 0:

import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _adjust_sum(x, target):
    """Project x onto the hyper‑plane sum(x)=target while respecting 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # water‑filling style redistribution
    while abs(diff) > 1e-12:
        if diff > 0:      # need to add mass
            mask = x < 1.0 - 1e-12
        else:             # need to remove mass
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _hill_climb(h, c5, deadline, rng):
    """Simple pairwise mass‑exchange hill climbing."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. problem size and initial construction
    # --------------------------------------------------------------
    n = 180                     # a moderate resolution
    target = n / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(new_x, old_x, h0)
    else:
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = 0.5 + 0.5 * np.cos(np.pi * x)   # smooth start within [0,1]

    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
    cur_f = np.max(np.correlate(h, 1.0 - h, mode="full"))   # unscaled max
    best_h, best_f = h.copy(), cur_f

    # --------------------------------------------------------------
    # 2. Smoothed‑max gradient descent
    # --------------------------------------------------------------
    beta = 1.0

    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
