def softmax_obj_and_grad(h, beta):
    n = h.size
    total = h.sum()
    # autocorr: use np.correlate; shape 2n-1
    auto = np.correlate(h, h, mode='full')
    obj = total - auto  # shape 2n-1
    # stable log-sum-exp
    scaled = beta * obj
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    f = (max_scaled + np.log(sum_exp)) / beta
    w = exp_scaled / sum_exp
    # convolution
    conv = np.convolve(w, h, mode='full')
    conv_center = conv[n-1 : n-1 + n]
    g = np.ones_like(h) - 2 * conv_center
    return f, g

import time, numpy as np
from scipy.optimize import minimize

# --- Global variables for SLSQP callbacks ---
CURRENT_TARGET = None
CURRENT_BETA = None

# -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_sum_box(x, target):
    """Projection of x onto {0≤y≤1, sum(y)=target} via bisection on lambda."""
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def softmax_obj_and_grad(h, beta):
    """Smooth max objective and its gradient."""
    n = h.size
    s = h.sum()
    auto = np.correlate(h, h, mode="full")      # sum_i h_i h_{i+k}
    obj = s - auto                              # vector of size 2n-1
    scaled = beta * obj
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    f = (max_scaled + np.log(sum_exp)) / beta
    w = exp_scaled / sum_exp                     # weights, length 2n-1, symmetric
    conv = np.convolve(w, h, mode="full")
    conv_center = conv[n - 1 : n - 1 + n]        # length n
    g = np.ones_like(h) - 2.0 * conv_center
    return f, g


def smooth_anneal_optimize(h0, beta_seq, steps_per_beta, base_step, time_limit):
    """Gradient descent on the smoothed objective with projection."""
    h = np.array(h0, copy=True)
    n = h.size
    target = n / 2.0
    start = time.time()
    for beta in beta_seq:
        step = base_step
        for _ in range(steps_per_beta):
            if time.time() - start > time_limit:
                return h
            f, g = softmax_obj_and_grad(h, beta)
            h_new = project_to_sum_box(h - step * g, target)
            f_new, _ = softmax_obj_and_grad(h_new, beta)
            if f_new < f:
                h = h_new
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.5
                if step < 1e-9:
                    break
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector then project to feasibility."""
    x = rng.random(n)
    return project_to_sum_box(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a small amount ε from one coordinate to another (keeps sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb using the exact C5 objective."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max used by SLSQP (uses globals CURRENT_TARGET, CURRENT_BETA)."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * (CURRENT_TARGET - corr)
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of the smoothed objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact projection onto feasible set
    target = n / 2.0
    x = project_to_sum_box(x, target)
    return x


def upsample_linear(h):
    """Double resolution by linear interpolation."""
    n = h.size
    h_up = np.empty(n * 2, dtype=float)
    h_up[0::2] = h
    # averages between neighbours for odd positions (except last)
    h_up[1:-1:2] = 0.5 * (h[:-1] + h[1:])
    h_up[-1] = h[-1]  # repeat the last value
    return h_up


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_sum_box(h0, n / 2.0)
    else:
        n = 90
        xs = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        h0 = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
        h0 = project_to_sum_box(h0, n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- phase 1: smooth gradient descent ----
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 20.0, 80.0]
    steps_per_beta = 200
    best_h = smooth_anneal_optimize(
        best_h,
        beta_schedule,
        steps_per_beta,
        base_step=0.1,
        time_limit=start + budget_s * 0.70,
    )
    c_now = compute_c5(best_h)
    if c_now < best_c:
        best_c = c_now

    # ---- phase 2: SLSQP refinements ----
    for beta in [5.0, 20.0, 80.0, 200.0]:
        if time.time() - start > budget_s * 0.80:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=500)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            pass

    # ---- phase 3: random restarts with hill‑climb ----
    while time.time() - start < budget_s * 0.90:
        h_rand = random_feasible_vector(n, rng)
        h_rand = smooth_anneal_optimize(
            h_rand,
            beta_seq=[0.5, 2.0, 5.0],
            steps_per_beta=80,
            base_step=0.2,
            time_limit=start + budget_s * 0.90,
        )
        h_rand, c_rand = hill_climb(h_rand, rng, max_iters=800, start_step=0.3)
        if c_rand < best_c:
            best_h, best_c = h_rand, c_rand

    # ---- optional up‑sampling ----
    if time.time() - start < budget_s * 0.95 and n * 2 <= 200:
        h_up = upsample_linear(best_h)
        n_up = h_up.size
        target_up = n_up / 2.0
        h_up = project_to_sum_box(h_up, target_up)

        # fine‑tune at higher resolution
        h_up = smooth_anneal_optimize(
            h_up,
            beta_seq=[0.5, 2.0, 5.0, 20.0, 80.0],
            steps_per_beta=150,
            base_step=0.08,
            time_limit=start + budget_s * 0.98,
        )
        for beta in [5.0, 20.0, 80.0]:
            if time.time() - start > budget_s:
                break
            try:
                h_up = slsqp_optimize(h_up, beta=beta, max_iter=350)
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_box(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
