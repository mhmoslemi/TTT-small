import time, numpy as np

# ---------- helper utilities ----------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while keeping sum(x) = target.
    Uses a bisection on the Lagrange multiplier."""
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised objective."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def upsample(vec, n):
    """Linear interpolation of a vector defined on [0,2)."""
    old_n = len(vec)
    if old_n == n:
        return np.array(vec, dtype=float)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, vec)


def two_level_shape(n, w):
    """Central plateau of height 1 with width w (≤1),
    outer region of constant height a = (1-w)/(2-w)."""
    if w <= 0.0:
        w = 0.0
    if w >= 1.0:
        w = 1.0
    a = (1.0 - w) / (2.0 - w) if w < 2.0 else 0.0
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.full(n, a, dtype=float)
    mask = np.abs(x - 1.0) <= w / 2.0
    h[mask] = 1.0
    # projection fixes the tiny discretisation error in the integral
    target = n / 2.0
    return project_onto_bounds_and_sum(h, target)

            cand = project_onto_bounds_and_sum(cand, target_sum)
            cand_val = compute_c5(cand)

            # accept if better, otherwise occasionally accept worse (SA)
            accept = cand_val < cur_val
            if not accept:
                # simple temperature schedule (decreases linearly)
                elapsed = time.time() - start
                T = max(0.001, (budget_s - elapsed) / budget_s)  # from ~1 down to ~0
                if np.random.rand() < np.exp(-(cand_val - cur_val) / T):
                    accept = True
            if accept:
                h = cand
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best_h = cand.copy()
                if best_val <= 0.3808:
                    break

        # early exit if we already beat the target
        if best_val <= 0.3808:
            break

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
