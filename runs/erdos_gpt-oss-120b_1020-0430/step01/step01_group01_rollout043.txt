def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=100):
    lo = -np.max(v) - 1
    hi = np.max(v) + 1
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _gradient_for_shift(h, shift):

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

        else:

        h_opt = _gd_optimize(
            h0,
            target,
            max_iter=max_iter,
            step0=0.8,
            decay=0.996,
            rng=rng,
        )
        if rng.rand() < 0.15:
            h_opt = _slsqp_optimize(h_opt, target, alpha=35.0, max_iter=500)

        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt

        # early exit if we already beat the target bound
        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed) and final polishing.
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target)

    best_h = _project_onto_capped_simplex(best_h, target)
    best_val = _compute_c5(best_h, dx)

    if time.time() - start < budget_s * 0.98:
        final_h = _slsqp_optimize(best_h, target, alpha=45.0, max_iter=800)
        final_val = _compute_c5(final_h, dx)
        if final_val < best_val:
            best_h = final_h
            best_val = final_val

    return (best_h, float(best_val), n)
