def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize
    from scipy.special import logsumexp

    start_time = time.time()
    time_deadline = start_time + budget_s * 0.95

    rng = np.random.default_rng(seed)

    # compute target sum factor for each n later

    def compute_c5(arr):
        n=arr.shape[0]
        dx=2.0/n
        conv=np.convolve(arr, (1-arr)[::-1], mode='full')
        return np.max(conv)*dx

    # softmax objective
    def softmax_c5(arr, alpha=30.0):
        conv=np.convolve(arr, (1-arr)[::-1], mode='full')
        return logsumexp(alpha * conv)/alpha

    # rounding to binary
    def round_to_binary(arr, target_sum):
        # arr shape (n,)
        idx = np.argpartition(-arr, target_sum-1)[:target_sum]
        bin_vec = np.zeros_like(arr)
        bin_vec[idx]=1.0
        return bin_vec

    # upsample
    def upsample(arr, new_n):
        old_n = len(arr)
        old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0/old_n
        new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
        up = np.interp(new_x, old_x, arr)
        # renormalize to sum target
        target = new_n/2.0
        up = up * (target / up.sum())
        # clip
        up = np.clip(up, 0.0, 1.0)
        diff = target - up.sum()
        if abs(diff) > 1e-12:
            up += diff/new_n
            up = np.clip(up, 0.0, 1.0)
        return up

    # recall parent
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    # candidate n's
    candidate_ns = [80, 100, 120, 140, 160, 180, 200]

    if parent_h is not None:
        parent_n = len(parent_h)
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Function for binary annealing
    def anneal_binary(start_bin, n, time_deadline):
        h = start_bin.astype(float)  # 0/1 floats
        cur_c5 = compute_c5(h)
        best_local_h = h.copy()
        best_local_c5 = cur_c5
        # number of iterations: choose adaptively: maybe 40000 * n/100?
        # Let's compute max_iters = int(1e5)
        max_iters = 200_000  # many
        # temperature schedule
        T0 = 0.02
        T_end = 1e-5
        for t in range(max_iters):
            if time.time() > time_deadline:
                break
            # propose swap
            i = rng.integers(n)
            j = rng.integers(n)
            if h[i] == h[j]:
                continue
            # swap
            h[i], h[j] = h[j], h[i]  # swap values
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            # compute temperature
            T = T0 * (1 - t / max_iters) + T_end * (t / max_iters)
            if delta <= 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if new_c5 < best_local_c5:
                    best_local_c5 = new_c5
                    best_local_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
        return best_local_h, best_local_c5

    # Loop over n
    for n in candidate_ns:
        if time.time() > time_deadline:
            break
        target_sum = n // 2
        # Generate starting points

    # final recompute
    final_c5 = compute_c5(best_h) if best_h is not None else np.inf

    # Return as required
    return best_h, final_c5, best_n

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def compute_c5(arr: np.ndarray) -> float:
    """Exact C5 for a discrete step function (full linear convolution)."""
    n = arr.shape[0]
    dx = 2.0 / n
    conv = np.convolve(arr, (1.0 - arr)[::-1], mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(arr: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max overlap via log‑sum‑exp."""
    conv = np.convolve(arr, (1.0 - arr)[::-1], mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a finer grid, then renormalise to exact integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up = up + diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def round_to_binary(arr: np.ndarray, target_sum: int) -> np.ndarray:
    """Project a real vector onto the set of binary vectors with a given sum."""
    idx = np.argpartition(-arr, target_sum - 1)[:target_sum]
    out = np.zeros_like(arr)
    out[idx] = 1.0
    return out


# ----------------------------------------------------------------------
# Simulated annealing on binary vectors
# ----------------------------------------------------------------------
def anneal_binary(start_bin: np.ndarray, n: int, deadline: float) -> tuple:
    """
    Simple simulated‑annealing on a binary (0/1) vector.
    Returns the best binary vector seen and its C5 value.
    """
    h = start_bin.astype(float)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    max_iters = 20000                     # enough for a quick local search
    T0, T_end = 0.02, 1e-6                # temperature schedule

    for t in range(max_iters):
        if time.time() > deadline:
            break
        # propose swapping a 1 and a 0
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]          # effect the swap
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        T = T0 * (1.0 - t / max_iters) + T_end * (t / max_iters)
        if delta <= 0.0 or np.random.rand() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_deadline = start_time + budget_s * 0.95   # safety margin

    # Try to retrieve a parent solution supplied by the harness
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    # Candidate grid sizes (all even, so the integral can be satisfied exactly)
    candidate_ns = [80, 100, 120, 140, 160, 180, 200]
    if parent_h is not None:
        pn = parent_h.shape[0]
        if pn not in candidate_ns:
            candidate_ns.insert(0, pn)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Exploration stage – many short annealing runs on binary vectors
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > time_deadline:
            break
        target_sum = n // 2

        # Build a small set of diverse binary starts for this resolution
        starts = []

        if parent_h is not None:
            # Upsample/downsample parent, then binarise
            if parent_h.shape[0] != n:
                up = upsample(parent_h, n)
            else:
                up = parent_h.copy()
            starts.append(round_to_binary(up, target_sum))

        # A few random binary vectors
        for _ in range(4):
            perm = rng.permutation(n)
            bin_vec = np.zeros(n)
            bin_vec[perm[:target_sum]] = 1.0
            starts.append(bin_vec)

        # Simple deterministic pattern (fill even positions first)
        pat = np.zeros(n)
        pat[::2][:target_sum] = 1.0
        if pat.sum() < target_sum:
            need = target_sum - int(pat.sum())
            pat[1::2][:need] = 1.0
        starts.append(pat)

        # Run annealing from each start
        for s in starts:
            if time.time() > time_deadline:
                break
            h_opt, c5_opt = anneal_binary(s, n, time_deadline)
            if c5_opt < best_c5 - 1e-9:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = n

    # ------------------------------------------------------------------
    # Continuous refinement – SLSQP with increasingly sharp softmax
    # ------------------------------------------------------------------
    if best_h is not None and time.time() < time_deadline:
        n = best_n
        target_sum = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}

        # Stagewise increase of the softmax sharpness
        for alpha in (5.0, 30.0, 100.0, 300.0):
            if time.time() > time_deadline:
                break
            res = minimize(
                lambda x: softmax_c5(x, alpha=alpha),
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-10, "disp": False},
            )
            if res.success:
                best_h = np.clip(res.x, 0.0, 1.0)

        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Fallback – a simple binary block if everything else failed
    # ------------------------------------------------------------------
    if best_h is None:
        n = 100
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final safety: enforce exact integral
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
