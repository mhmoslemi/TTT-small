def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-12) -> np.ndarray:

    h = np.clip(h, 0.0, 1.0)

    # water‑filling style correction
    for _ in range(500):

        diff = target - s
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]

            free = np.where(h > 0.0)[0]

            break
        delta = diff / free.size

    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible h (∫h =1)."""
    n = h.size
    dx = 2.0 / n
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * dx


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - correlation >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x: np.ndarray) -> float:
    """sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x: np.ndarray) -> float:
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 1200) -> tuple:
    """Run a single SLSQP optimisation starting from `initial_h`."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# Main optimisation loop
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h=1 that minimises
          C5 = max_k ∫ h(x)(1-h(x+k))dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # obtain the optional parent construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = np.full(64, 0.5)          # fallback uniform
    n0 = init_h.size

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target = 0.3808

    # ------------------------------------------------------------------
    # candidate grid sizes – start coarse and progressively refine
    # ------------------------------------------------------------------
    n_candidates = [n0]
    # multiples of the base size (up to ~400 points)
    for mul in (2, 3, 4):
        n = int(round(n0 * mul))
        if n % 2 == 0 and n <= 400:
            n_candidates.append(n)
    # some hand‑picked even sizes
    for n in (200, 256, 300, 352, 384):
        if n % 2 == 0 and n <= 400:
            n_candidates.append(n)

    n_candidates = sorted(set(n_candidates))

    # ------------------------------------------------------------------
    # main search over the grid sizes
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time_left() < 8.0:
            break

        # ----- build a starting function for this resolution ----------
        if init_h.size != n:
            base_h = _upscale_h(init_h, n)
        else:
            base_h = init_h.copy()
        base_h = _project_to_feasible(base_h, n)

        # ----- first deterministic optimisation (no noise) ------------
        h_opt, c5_val = _solve_h(base_h, max_iter=1800)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target:
                return best_h.tolist(), float(best_c5), int(best_n)

        # ----- stochastic restarts ------------------------------------
        attempts = 12
        sigma = 0.22
        for _ in range(attempts):
            if time_left() < 6.0:
                break
            perturb = sigma * np.random.randn(n)
            start_h = base_h + perturb
            start_h = _project_to_feasible(start_h, n)
            h_opt, c5_val = _solve_h(start_h, max_iter=1500)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 < target:
                    return best_h.tolist(), float(best_c5), int(best_n)
            sigma *= 0.85  # gradually focus the search

        # ----- local refinement of the current best on this grid -----
        if best_h is not None and best_n == n:
            h_opt, c5_val = _solve_h(best_h, max_iter=2500)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                if best_c5 < target:
                    return best_h.tolist(), float(best_c5), int(best_n)

    # ------------------------------------------------------------------
    # fallback – if no improvement was found (unlikely)
    # ------------------------------------------------------------------
    if best_h is None:
        n = n0
        best_h = _project_to_feasible(init_h, n)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
