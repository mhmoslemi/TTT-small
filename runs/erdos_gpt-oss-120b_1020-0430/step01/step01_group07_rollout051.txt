def compute_c5(h):
    n = h.shape[0]
    # pad to length 2n-1 for linear convolution
    pad_len = 2*n - 1
    H = np.fft.rfft(np.concatenate([h, np.zeros(n-1)]))
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(n-1)]))
    corr = np.fft.irfft(H * G, n=pad_len)
    return corr.max() * (2.0 / n)

def worst_lag_indices(h):
    n = len(h)
    # compute correlation S_c = np.correlate(h, 1-h, mode='full')
    S = np.correlate(h, 1.0 - h, mode='full')
    idx_max = S.argmax()
    lag = idx_max - (n - 1)
    # compute product terms for this lag
    if lag >= 0:
        i_vals = np.arange(0, n - lag)
        j_vals = i_vals + lag
    else:
        lag_abs = -lag
        i_vals = np.arange(lag_abs, n)
        j_vals = i_vals - lag_abs
    # compute contributions = h[i] * (1 - h[j])
    contributions = h[i_vals] * (1.0 - h[j_vals])
    max_idx = contributions.argmax()
    i = i_vals[max_idx]
    j = j_vals[max_idx]
    return i, j, lag

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def initial_vector(n):
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        # interpolate to n points
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)  # uniform with sum = n/2?
    # enforce sum
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    # adjust small sum difference
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # find some index that can accommodate
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
    return h

def anneal(h_start, time_budget, rng):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_c = best_c

    start = time.time()
    # temperature schedule
    T0 = 0.05
    T_end = 1e-5
    max_step = 0.3

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        progress = elapsed / time_budget
        T = T0 * (T_end / T0) ** progress  # exponential decay
        # choose move type: random pair (80%) or worst-lag guided (20%)
        if rng.random() < 0.2:
            # worst-lag guided move:
            # compute correlation of h and 1-h to find max shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx = corr.argmax()
            lag = idx - (n - 1)
            if lag == 0:
                # fallback random
                i, j = rng.choice(n, size=2, replace=False)
            else:
                if lag > 0:
                    i_candidates = np.arange(0, n - lag)
                else:
                    i_candidates = np.arange(-lag, n)
                # compute contributions
                j_candidates = i_candidates + lag
                # contributions = h[i] * (1 - h[j])
                contrib = h[i_candidates] * (1.0 - h[j_candidates])
                # pick index where contribution is largest
                max_idx = np.argmax(contrib)
                i = int(i_candidates[max_idx])
                j = int(j_candidates[max_idx])
        else:
            # random pair
            i, j = rng.choice(n, size=2, replace=False)

        hi, hj = h[i], h[j]
        # feasible delta range such that 0 <= hi+Δ <= 1 and 0 <= hj-Δ <= 1
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # limit step size
        step_cap = max_step
        low_cap = max(low, -step_cap)
        high_cap = min(high, step_cap)
        if low_cap < high_cap:
            delta = rng.uniform(low_cap, high_cap)
        else:
            delta = rng.uniform(low, high)

        # apply move
        h[i] = hi + delta
        h[j] = hj - delta

        new_c = compute_c5(h)
        dC = new_c - cur_c
        if dC < 0 or rng.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_c

def enforce_sum(h):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()

import time
import math
import random
import numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _initial_vector(n: int) -> np.ndarray:
    """Build a feasible start of length n, using the supplied parent if present."""
    if "initial_h_values" in globals():
        src = np.asarray(initial_h_values, dtype=np.float64)
        m = src.size
        xs = np.linspace(0.0, 2.0, m, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xt, xs, src)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
    return h


def _resize_vector(h: np.ndarray, n_new: int) -> np.ndarray:
    """Resize vector h to length n_new by linear interpolation and re‑normalise."""
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    cur = h2.sum()
    if cur != 0.0:
        h2 *= target / cur
    np.clip(h2, 0.0, 1.0, out=h2)
    diff = target - h2.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n_new):
            if 0.0 <= h2[idx] + diff <= 1.0:
                h2[idx] += diff
                break
    return h2


def _anneal(start_h: np.ndarray, time_budget: float, rng: np.random.Generator) -> tuple:
    """Simulated‑annealing improvement."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0, T_end = 0.5, 1e-5         # linear schedule
    max_step = 0.3

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        prog = elapsed / time_budget
        T = T0 * (1 - prog) + T_end * prog

        # guided move with probability 0.2
        if rng.random() < 0.2:
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = corr.argmax()
            lag = idx - (n - 1)
            if lag == 0:
                i, j = rng.choice(n, size=2, replace=False)
            else:
                if lag > 0:
                    i_cand = np.arange(0, n - lag)
                else:
                    i_cand = np.arange(-lag, n)
                j_cand = i_cand + lag
                prod = h[i_cand] * (1.0 - h[j_cand])
                maxpos = prod.argmax()
                i = int(i_cand[maxpos])
                j = int(j_cand[maxpos])
        else:
            i, j = rng.choice(n, size=2, replace=False)

        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # restrict step size
        lo_cap = max(low, -max_step)
        hi_cap = min(high, max_step)
        if lo_cap < hi_cap:
            delta = rng.uniform(lo_cap, hi_cap)
        else:
            delta = rng.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or rng.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    return best_h, best_c


def _deterministic_refine(h: np.ndarray, time_budget: float, rng: np.random.Generator) -> tuple:
    """Hill‑climbing focused on the worst lag."""
    n = h.size
    best_c = _c5_metric(h)
    start = time.time()

    while time.time() - start < time_budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = corr.argmax()
        lag = idx - (n - 1)
        if lag == 0:
            break
        if lag > 0:
            i_cand = np.arange(0, n - lag)
        else:
            i_cand = np.arange(-lag, n)
        j_cand = i_cand + lag
        contrib = h[i_cand] * (1.0 - h[j_cand])
        maxpos = contrib.argmax()
        i = int(i_cand[maxpos])
        j = int(j_cand[maxpos])

        hi, hj = h[i], h[j]
        delta_max = min(hi, 1.0 - hj)
        if delta_max <= 0:
            break
        # try a moderate step
        delta = delta_max * 0.5
        h[i] = hi - delta
        h[j] = hj + delta
        new_c = _c5_metric(h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
            continue
        # revert and try smaller step
        h[i] = hi
        h[j] = hj
        delta = delta_max * 0.1
        if delta < 1e-8:
            break
        h[i] = hi - delta
        h[j] = hj + delta
        new_c = _c5_metric(h)
        if new_c + 1e-12 < best_c:
            best_c = new_c
            continue
        h[i] = hi
        h[j] = hj
        break
    return h, best_c


def _repair_sum(h: np.ndarray) -> None:
    """Enforce exact integral constraint (sum = n/2)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in random.sample(range(n), n):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + diff <= hi:
                h[idx] += diff
                break
    np.clip(h, 0.0, 1.0, out=h)
