def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)

    n = 200
    half = n // 2

    # initialize binary h
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=np.int8)
    h[perm[:half]] = 1

    # pad
    pad_len = 3*n - 2
    pad = np.zeros(pad_len, dtype=np.int8)
    pad[n-1:n-1+n] = h

    # compute correlation using np.correlate, but we can also compute via convolution fast
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    # Since h binary, we can compute with int:
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    best_corr = cur_max
    best_h = h.copy()
    best_max = cur_max

    # Precompute L array
    L = np.arange(-(n-1), n, dtype=int)  # shape (2n-1,)

    # Set temperature schedule
    max_temp = 1.0
    min_temp = 1e-4

    # Main loop
    while True:
        now = time.time()
        if now - start > budget_s - 0.5:
            break
        # gradually compute temperature based on elapsed fraction
        frac = (now - start) / budget_s
        T = max_temp * (1 - frac) + min_temp * frac

        # select random indices
        # choose i_one from where h==1, i_zero where h==0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # compute delta using pad
        i1 = n-1 + i_one
        i0 = n-1 + i_zero

        # compute values using fancy indexing
        v1 = pad[i1 + L]
        v2 = pad[i0 + L]
        v3 = pad[i1 - L]
        v4 = pad[i0 - L]
        delta = (v1 - v2) + (v3 - v4)
        new_corr = corr + delta
        new_max = new_corr.max()

        # acceptance
        delta_max = new_max - cur_max
        accept = False
        if delta_max <= 0:
            accept = True
        else:
            # acceptance prob
            if T > 0:
                accept = rng.random() < np.exp(-delta_max / T)
        if accept:
            # commit swap
            h[i_one] = 0
            h[i_zero] = 1
            # update pad

def greedy_reduce(h, pad, corr, target, L, n, rng):
    cur_max = corr.max()
    while cur_max > target:
        idx = np.argmax(corr)
        lag = idx - (n-1)
        # compute source indices (1->0 for this lag)
        # source condition: h[i]==1 and i+lag in bounds and h[i+lag]==0
        if lag >=0:
            src_mask = (h[:n-lag]==1) & (h[lag:]==0)
            src_indices = np.where(src_mask)[0]
            # dest condition: h[j]==0 and j+lag in bounds and h[j+lag]==1
            dst_mask = (h[:n-lag]==0) & (h[lag:]==1)
            dst_indices = np.where(dst_mask)[0]
        else:
            # lag negative: treat similarly with shift left
            l = -lag
            src_mask = (h[l:]==1) & (h[:n-l]==0)
            src_indices = np.where(src_mask)[0] + l  # relative to original index?
            # Actually when lag negative: check i+lag = i - l. So i>=l needed.
            src_mask = (h[l:]==1) & (h[:n-l]==0)
            src_indices = np.where(src_mask)[0] + l
            # dest for swapping 0->1: h[j]==0 and h[j+lag]==1 => j>=l
            dst_mask = (h[l:]==0) & (h[:n-l]==1)
            dst_indices = np.where(dst_mask)[0] + l
        if src_indices.size == 0 or dst_indices.size == 0:
            # cannot improve this lag; break
            break
        # Choose random src and dst
        i_one = rng.choice(src_indices)
        i_zero = rng.choice(dst_indices)
        # compute delta as before and apply
        i1 = n-1 + i_one
        i0 = n-1 + i_zero
        v1 = pad[i1 + L]
        v2 = pad[i0 + L]
        v3 = pad[i1 - L]
        v4 = pad[i0 - L]
        delta = (v1 - v2) + (v3 - v4)
        corr = corr + delta
        cur_max = corr.max()
        # update h and pad
        h[i_one] = 0
        h[i_zero] = 1
        pad[i1] = 0
        pad[i0] = 1
        # maybe break if no improvement (cur_max not reduced)
        # continue loop

def reduce_max_lag(h, pad, corr, n, L, target, rng):
    cur_max = corr.max()
    while cur_max > target:
        idx = np.argmax(corr)
        lag = idx - (n-1)
        # compute source indices (ones that see zero at lag)
        if lag >= 0:
            src_mask = (h[:n-lag] == 1) & (h[lag:] == 0)
            srcs = np.where(src_mask)[0]
            dst_mask = (h[:n-lag] == 0) & (h[lag:] == 1)
            dsts = np.where(dst_mask)[0]
        else:
            l = -lag
            src_mask = (h[l:] == 1) & (h[:n-l] == 0)
            srcs = np.where(src_mask)[0] + l
            dst_mask = (h[l:] == 0) & (h[:n-l] == 1)
            dsts = np.where(dst_mask)[0] + l
        if srcs.size == 0 or dsts.size == 0:
            break  # cannot improve this lag
        # choose random pair
        i_one = rng.choice(srcs)
        i_zero = rng.choice(dsts)
        # compute delta and apply similar to earlier
        i1 = n-1 + i_one
        i0 = n-1 + i_zero
        d = i_zero - i_one
        base = (pad[i1 + L] + pad[i1 - L]) - (pad[i0 + L] + pad[i0 - L])

            idx_neg = -d + (n-1)

            delta[idx_neg] += 1

        # update h and pad
        h[i_one] = 0
        h[i_zero] = 1
        pad[i1] = 0

import time
import numpy as np

def _evaluate_c5_vec(h):
    """C5 for a binary/float step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_binary(n, rng):
    """Create a random binary vector with exactly n/2 ones."""
    half = n // 2
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=np.int8)
    h[perm[:half]] = 1
    return h

def _prepare_state(h):
    """Return padded array and full correlation for a binary h."""
    n = h.size
    pad = np.zeros(3 * n - 2, dtype=np.int8)
    pad[n - 1 : n - 1 + n] = h  # central region holds h
    corr = np.correlate(h, 1 - h, mode="full").astype(np.int16)
    return pad, corr

def _delta_for_swap(pad, i_one, i_zero, L, n):
    """Vectorised delta of the correlation after swapping positions i_one (1) and i_zero (0)."""
    i1 = n - 1 + i_one
    i0 = n - 1 + i_zero
    # base contribution without the cross‑term
    v1 = pad[i1 + L]
    v2 = pad[i0 + L]
    v3 = pad[i1 - L]
    v4 = pad[i0 - L]
    delta = (v1 + v3 - v2 - v4).astype(np.int16)   # shape (2n‑1,)
    # centre (lag 0) must stay zero
    delta[n - 1] = 0
    # cross‑term (distance d = i_zero‑i_one)
    d = i_zero - i_one
    if d != 0:
        idx1 = d + n - 1
        idx2 = -d + n - 1
        delta[idx1] += 1
        delta[idx2] += 1
    return delta, i1, i0

def _swap_state(h, pad, corr, i_one, i_zero, delta, i1, i0):
    """Apply the swap to h, pad and the correlation."""
    h[i_one] = 0
    h[i_zero] = 1
    pad[i1] = 0
    pad[i0] = 1
    corr += delta
    return h, pad, corr

def _greedy_improve(h, pad, corr, L, n, target_raw, rng):
    """Deterministic hill‑climbing: try to lower the current maximum until it reaches target_raw."""
    cur_max = int(corr.max())
    while cur_max > target_raw:
        idx = int(corr.argmax())
        lag = idx - (n - 1)            # signed lag
        # indices that contribute to this lag (1 → 0)
        if lag >= 0:
            src_mask = (h[: n - lag] == 1) & (h[lag:] == 0)
            dst_mask = (h[: n - lag] == 0) & (h[lag:] == 1)
            src = np.flatnonzero(src_mask)
            dst = np.flatnonzero(dst_mask)
        else:
            l = -lag
            src_mask = (h[l:] == 1) & (h[: n - l] == 0)
            dst_mask = (h[l:] == 0) & (h[: n - l] == 1)
            src = np.flatnonzero(src_mask) + l
            dst = np.flatnonzero(dst_mask) + l
        if src.size == 0 or dst.size == 0:
            break   # cannot improve this lag any more
        # try a few random candidates before giving up
        improved = False
        for _ in range(20):
            i_one = rng.choice(src)
            i_zero = rng.choice(dst)
            delta, i1, i0 = _delta_for_swap(pad, i_one, i_zero, L, n)
            new_max = int((corr + delta).max())
            if new_max <= cur_max:
                h, pad, corr = _swap_state(h, pad, corr, i_one, i_zero, delta, i1, i0)
                cur_max = new_max
                improved = True
                break
        if not improved:
            break
    return h, pad, corr, cur_max

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 200                     # even, moderate resolution
    half = n // 2
    L = np.arange(-(n - 1), n, dtype=np.int32)   # all possible lags
    target_raw = int(np.floor(0.3808 * n / 2))   # raw count bound to beat

    best_h = None
    best_max = n * 2               # large sentinel

    # ------------------------------------------------------------------
    # Build a list of starting vectors (random + structured)
    # ------------------------------------------------------------------
    start_vectors = []

    # a few random starts
    for _ in range(4):
        start_vectors.append(_initial_binary(n, rng))

    # Thue‑Morse sequence (balanced for powers of two)
    if (n & (n - 1)) == 0:
        tm = np.fromfunction(lambda i: (np.unpackbits(i.astype(np.uint32).view(np.uint8))[::-1].sum() % 2), (n,), dtype=int).astype(np.int8)
        # ensure exactly half ones – Thue‑Morse already does for powers of two
        start_vectors.append(tm)

    # use parent construction if supplied
    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        m = parent.size
        # up‑sample to length n by linear interpolation, then binarise
        x_parent = np.linspace(0.0, 2.0, m, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_target, x_parent, parent)
        binary = (interp > 0.5).astype(np.int8)
        # adjust the count to exactly n/2
        cur = int(binary.sum())
        if cur < half:
            zero_idx = np.where(binary == 0)[0]
            add = rng.choice(zero_idx, size=half - cur, replace=False)
            binary[add] = 1
        elif cur > half:
            one_idx = np.where(binary == 1)[0]
            rem = rng.choice(one_idx, size=cur - half, replace=False)
            binary[rem] = 0
        start_vectors.append(binary)

    # ------------------------------------------------------------------
    # Main optimisation loop (simulated annealing + greedy clean‑up)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 0.8:
        # select a start vector (cycle through the list)
        h0 = start_vectors[rng.integers(len(start_vectors))].copy()
        pad, corr = _prepare_state(h0)
        cur_max = int(corr.max())

        # simulated annealing
        # temperature falls exponentially from 2.0 to 1e‑4 over the remaining budget
        while True:
            now = time.time()
            if now - start > budget_s - 0.8:
                break
            elapsed = now - start
            frac = elapsed / budget_s
            T = 2.0 * (1e-4 / 2.0) ** frac   # exponential decay
            # pick a random 1 and a random 0
            i_one = rng.integers(n)
            while h0[i_one] != 1:
                i_one = rng.integers(n)
            i_zero = rng.integers(n)
            while h0[i_zero] != 0:
                i_zero = rng.integers(n)
            # propose swap
            delta, i1, i0 = _delta_for_swap(pad, i_one, i_zero, L, n)
            new_max = int((corr + delta).max())
            delta_max = new_max - cur_max
            accept = False
            if delta_max <= 0:
                accept = True
            else:
                if T > 0.0:
                    if rng.random() < np.exp(-delta_max / T):
                        accept = True
            if accept:
                h0, pad, corr = _swap_state(h0, pad, corr, i_one, i_zero, delta, i1, i0)
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h0.copy()
                if best_max <= target_raw:
                    # optional early exit from SA if we already reached the bound
                    break
        # after SA, try a deterministic greedy descent from the current state
        if time.time() - start < budget_s - 0.5:
            h0, pad, corr, cur_max = _greedy_improve(h0, pad, corr, L, n, target_raw, rng)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h0.copy()
        # stop early if we have met the target
        if best_max <= target_raw:
            break

    # ------------------------------------------------------------------
    # Finalise result
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback – simple balanced block (unlikely to happen)
        best_h = np.zeros(n, dtype=np.int8)
        best_h[:half] = 1
        best_max = int(np.correlate(best_h, 1 - best_h, mode="full").max())
    c5_bound = best_max * (2.0 / n)

    return best_h.astype(np.float64), float(c5_bound), n
