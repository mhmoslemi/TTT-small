def propose_swap(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # max amount we can shift from i to j: limited by h[i] (can't go below 0) and (1-h[j]) (can't exceed 1)
    max_delta = min(h[i], 1 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = rng.uniform(0, max_delta)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
```

def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    max_c = np.max(corr)
    exp_term = np.exp(alpha * (corr - max_c))
    w = exp_term / np.sum(exp_term)
    grad = np.zeros_like(h)
    # loops
    for idx, weight in enumerate(w):
        if weight == 0: continue
        s = idx - (n - 1)
        if s >= 0:
            # for i from 0 to n-s-1: contribution (1-h[i+s])
            grad[:n-s] += weight * (1.0 - h[s:])
            # for i from s to n-1: contribution -h[i-s]
            grad[s:] += weight * (-h[:n-s])
        else:
            s_abs = -s
            grad[s_abs:] += weight * (1.0 - h[:n-s_abs])
            grad[:s_abs] += weight * (-h[s_abs:])
    return grad
```

def subgradient_descent_max(h0, target_sum, max_iters=500, step0=0.5, decay=0.995):
    h = h0.copy()
    step = step0
    n = len(h)
    for _ in range(max_iters):
        corr = compute_corr(h)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # compute gradient for shift
        g = np.zeros_like(h)
        s = shift
        if s >= 0:
            g[:n-s] = 1.0 - h[s:]
            g[s:] += -h[:n-s]
        else:
            s_abs = -s
            g[s_abs:] = 1.0 - h[:n-s_abs]
            g[:s_abs] += -h[s_abs:]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_capped_simplex(h, target_sum)
        step *= decay
    return h
```

def upscale_vec(vec, new_n):
    old_n = len(vec)
    if old_n == new_n:

def project_onto_capped_simplex(v, target, tol=1e-9, max_iter=100):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

import time
import numpy as np

# -------------------------------------------------------
# Projection onto the capped simplex {0 ≤ x ≤ 1, sum(x)=target}
# -------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# -------------------------------------------------------
# Random feasible vector (sum = target, 0‑1 bounds)
# -------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# -------------------------------------------------------
# Correlation used in the objective (zero‑padded second array)
# -------------------------------------------------------
def _corr(h: np.ndarray) -> np.ndarray:
    return np.correlate(h, 1.0 - h, mode='full')


def _c5_value(h: np.ndarray, dx: float) -> float:
    return float(_corr(h).max() * dx)


# -------------------------------------------------------
# Linear interpolation to change resolution
# -------------------------------------------------------
def _upscale(vec: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(vec)
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, vec)


# -------------------------------------------------------
# Gradient of the smooth‑max of the correlation
# -------------------------------------------------------
def _smooth_max_grad(h: np.ndarray, alpha: float) -> np.ndarray:
    """
    Returns ∇_h  (1/α) log Σ_s exp(α * C_s)   where C = corr(h,1‑h).
    Uses two convolutions (C‑level) for efficiency.
    """
    n = len(h)
    corr = _corr(h)                     # length 2n‑1
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    w_sum = w.sum()
    w /= w_sum                           # softmax weights, sum to 1
    # reverse for the term (1‑h) contribution
    w_rev = w[::-1]

    # convolution of w_rev with (1‑h) and of w with h
    conv1 = np.convolve(w_rev, 1.0 - h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    start = n - 1                         # index where i aligns
    grad = conv1[start:start + n] - conv2[start:start + n]
    return grad


# -------------------------------------------------------
# Gradient of a single shift (sub‑gradient for the true max)
# -------------------------------------------------------
def _grad_shift(s: int, h: np.ndarray) -> np.ndarray:
    n = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[: n - s] = 1.0 - h[s:]
        g[s:] += -h[: n - s]
    else:
        t = -s
        g[t:] = 1.0 - h[: n - t]
        g[: n - t] += -h[t:]
    return g


# -------------------------------------------------------
# Projected gradient descent on the smooth‑max objective
# -------------------------------------------------------
def _smooth_descent(h0: np.ndarray, target: float,
                    alphas, iters_per_alpha,
                    step0: float = 0.5, decay: float = 0.998) -> np.ndarray:
    h = h0.copy()
    step = step0
    for alpha, it in zip(alphas, iters_per_alpha):
        for _ in range(it):
            g = _smooth_max_grad(h, alpha)
            h = h - step * g
            h = np.clip(h, 0.0, 1.0)
            h = _project_onto_capped_simplex(h, target)
            step *= decay
    return h


# -------------------------------------------------------
# Sub‑gradient descent on the true max‑correlation
# -------------------------------------------------------
def _max_descent(h0: np.ndarray, target: float,
                 max_iters: int = 300,
                 step0: float = 0.4, decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = len(h)
    for _ in range(max_iters):
        corr = _corr(h)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _grad_shift(shift, h)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# -------------------------------------------------------
# Simple local search: move a small amount of mass between two points
# -------------------------------------------------------
def _swap_move(h: np.ndarray, rng: np.random.RandomState, scale: float = 0.02) -> np.ndarray:
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = rng.uniform(0.0, max_delta * scale)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
