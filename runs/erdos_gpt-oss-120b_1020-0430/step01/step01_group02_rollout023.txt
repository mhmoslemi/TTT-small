    best_c5 = _evaluate_c5(best_h)

    # -----------------------------------------------------------------
    # 2. Guided stochastic search (gradient of the worst shift)
    # -----------------------------------------------------------------
    no_improve = 0
    m_worst = 3                     # how many worst shifts to combine

    while True:
        now = time.time()
        if now >= start + budget_s - 30.0:   # keep 30 s for refinements
            break

        # current overlaps
        corr = np.correlate(h, 1.0 - h, mode="full")
        # indices of the worst (largest) overlaps
        worst_idx = np.argsort(corr)[-m_worst:]

        # accumulated gradient of the product term for the selected shifts
        grad = np.zeros_like(h)
        for idx in worst_idx:
            s = idx - (n - 1)               # signed shift
            if s >= 0:
                grad[: n - s] += h[s:]
                grad[s:] += h[: n - s]
            else:
                t = -s
                grad[t:] += h[: n - t]
                grad[: n - t] += h[t:]

        # choose source (decrease) and destination (increase) indices
        dst_cand = np.where((grad == grad.max()) & (h < 1.0 - 1e-12))[0]
        src_cand = np.where((grad == grad.min()) & (h > 1e-12))[0]

        if dst_cand.size == 0 or src_cand.size == 0:
            i, j = rng.choice(n, size=2, replace=False)
            src, dst = (i, j) if h[i] > h[j] else (j, i)
        else:
            dst = rng.choice(dst_cand)
            src = rng.choice(src_cand)
            if src == dst:
                src = rng.choice(np.where(h > 1e-12)[0])

        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            i, j = rng.choice(n, size=2, replace=False)
            src, dst = (i, j) if h[i] > h[j] else (j, i)
            max_move = min(h[src], 1.0 - h[dst])
            if max_move <= 1e-12:
                continue

        delta = max_move * rng.uniform(0.2, 0.9)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)

        # acceptance
        if new_c5 + 1e-12 < best_c5:
            best_h, best_c5 = h_new.copy(), new_c5
            h = h_new
            no_improve = 0
        else:
            elapsed = now - start
            prog = elapsed / (budget_s - 30.0)
            T = best_c5 * (1.0 - prog) + 1e-9
            if rng.random() < np.exp(-(new_c5 - best_c5) / T):
                h = h_new
                no_improve = 0
            else:
                no_improve += 1

        # occasional restart if stuck
        if no_improve > 3000:
            h = rng.random(n)
            h = _adjust_sum(h, target)
            no_improve = 0

    # -----------------------------------------------------------------
    # 3. Local refinement with SLSQP (few short runs)
    # -----------------------------------------------------------------
    slsq_deadline = start + budget_s - 25.0
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    it = 0
    while time.time() < slsq_deadline:
        if it % 5 == 0:
            x0 = best_h + rng.normal(scale=0.05, size=n)
        else:
            x0 = best_h + rng.normal(scale=0.02, size=n)
        x0 = np.clip(x0, 0.0, 1.0)
        x0 = _adjust_sum(x0, target)

        res = minimize(
            obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        cand = np.clip(res.x, 0.0, 1.0) if res.success else x0
        cand = _adjust_sum(cand, target)
        cand_c5 = _evaluate_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand.copy(), cand_c5
        it += 1

    # -----------------------------------------------------------------
    # 4. Final fine‑grained hill climbing
    # -----------------------------------------------------------------
    final_deadline = start + budget_s - 0.2
    h = best_h.copy()
    while time.time() < final_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = max_move * rng.uniform(0.05, 0.5)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < best_c5:
            best_h, best_c5 = h_new.copy(), new_c5
            h = h_new

    # -----------------------------------------------------------------
    # 5. Return feasible solution
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum(best_h, target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
