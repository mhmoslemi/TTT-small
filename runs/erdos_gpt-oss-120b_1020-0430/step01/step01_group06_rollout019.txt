def neighbor_sum(v, shift):
    n = v.shape[0]
    d = np.zeros(n)
    if shift >= 0:
        d[:n-shift] += v[shift:]
        d[shift:] += v[:n-shift]

        d[s:] += v[:n-s]
        d[:n-s] += v[s:]
    return d
