import cvxpy as cp
import numpy as np

def solve_lp(n):
    target = n/2  # float
    h = cp.Variable(n)
    W = cp.Variable((n, n))
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == target,
                   W >= 0,
                   W <= cp.reshape(h, (n,1)),
                   W <= cp.reshape(h, (1,n)),
                   W >= cp.reshape(h, (n,1)) + cp.reshape(h, (1,n)) - 1,
                   t >= 0]
    # shift constraints
    for s in range(-n+1, n):
        # mask for shift s
        mask = np.eye(n, k=s)  # shape (n, n), dtype float
        if mask.sum() == 0:
            continue
        # sum of entries of mask * W
        constraints.append(cp.sum(cp.multiply(mask, W)) >= target - t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
    return h.value, t.value

def _adjust_sum(x, target, eps=1e-12):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # Tolerance loop
    while abs(diff) > eps:
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        indices = np.where(mask)[0]
        per = diff / len(indices)
        for i in indices:

                x[i] += add
                diff -= add
            else:

                x[i] -= sub
                diff += sub

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _adjust_sum(x: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """Project x onto {0≤x≤1, sum(x)=target} by simple redistribution."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < eps:
        return x
    # iterative redistribution
    while abs(diff) > eps:
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < eps:
                break
    return np.clip(x, 0.0, 1.0)


def _hill_climb(
    h: np.ndarray, cur_c5: float, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """Simple pair‑wise mass exchange hill climbing (accepts only improvements)."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
    return h, cur_c5


def _solve_lp(n: int, timeout: float = 300.0):
    """Linear‑programming relaxation of the min‑max formulation."""
    target = n / 2.0
    h = cp.Variable(n)
    W = cp.Variable((n, n))
    t = cp.Variable()
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target,
        W >= 0,
        W <= cp.reshape(h, (n, 1)),
        W <= cp.reshape(h, (1, n)),
        W >= cp.reshape(h, (n, 1)) + cp.reshape(h, (1, n)) - 1,
        t >= 0,
    ]
    # shift constraints: for each integer offset s, the overlapping sum of W
    for s in range(-n + 1, n):
        mask = np.eye(n, k=s)  # 1 on the s‑th diagonal, 0 elsewhere
        if mask.sum() == 0:
            continue
        constraints.append(cp.sum(cp.multiply(mask, W)) >= target - t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    except Exception:
        return None, None
    if h.value is None:
        return None, None
    return np.asarray(h.value).flatten(), float(t.value)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Produce a step function h on [0,2] giving an improved upper bound
    for the Erdős minimum‑overlap constant C₅.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose resolution
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        n = len(init_vec)          # respect the provided size if any
    else:
        n = 80                     # a modest even size giving extra flexibility
    target = n / 2.0

    # ------------------------------------------------------------------
    # 2. Obtain a good starting vector via the LP relaxation
    # ------------------------------------------------------------------
    h_lp, t_val = _solve_lp(n)
    if h_lp is None or np.any(np.isnan(h_lp)):
        # fallback: random feasible point
        h0 = rng.random(n)
        h0 = _adjust_sum(h0 * (target / h0.sum()), target)
    else:
        h0 = _adjust_sum(np.clip(h_lp, 0.0, 1.0), target)

    # ------------------------------------------------------------------
    # 3. Also consider a binary rounding of the LP solution (if any)
    # ------------------------------------------------------------------
    num_ones = int(round(target))
    h_bin = np.zeros(n)
    idx = np.argsort(h0)[-num_ones:]

    c_lp = _evaluate_c5(h0)
    c_bin = _evaluate_c5(h_bin)

    if c_bin < c_lp:

    # ------------------------------------------------------------------
    # 5. Final feasibility enforcement and return
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
