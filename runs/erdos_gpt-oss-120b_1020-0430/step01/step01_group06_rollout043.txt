def solve_milp(n):
    h = cp.Variable(n, boolean=True)
    lam = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2]  # n half maybe integer; for n even.
    # lam bounds
    constraints += [lam >= 0, lam <= n//2]
    for s in range(1, n):  # shift >0; ignore s=0
        length = n - s
        # product variables
        y = cp.Variable(length, boolean=True)
        # add product constraints
        constraints += [y <= h[:length],
                        y <= h[s:],
                        y >= h[:length] + h[s:] - 1]
        # sum overlapping at shift s
        constraints += [cp.sum(y) >= lam]
    # optional: for s=0, lam <= sum(h) i.e., lam <= n/2; already.
    prob = cp.Problem(cp.Maximize(lam), constraints)
    prob.solve(solver=cp.CBC, warm_start=True, verbose=False, max_iters=10000)
    return h, lam, prob

def solve_h(n):
    h = cp.Variable(n, boolean=True)
    lam = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n/2,
                   lam >= 0,
                   lam <= n/2]

        # product variables
        y = cp.Variable(length, boolean=True)
        constraints += [y <= h[:length],
                        y <= h[s:],
                        y >= h[:length] + h[s:] - 1,
                        cp.sum(y) >= lam]
    prob = cp.Problem(cp.Maximize(lam), constraints)
    prob.solve(solver=cp.CBC, verbose=False)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math, cvxpy as cp

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # Determine n candidates - try larger first for potentially better bound.
    n_candidates = [100, 84, 70]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    dx_list = {}
    target_sum_cache = {}
    # compute for each n candidate.
    for n in n_candidates:
        # Check time.
        if time.time() - start > budget_s * 0.9:
            break
        # adjust n to be even.
        if n % 2 != 0:
            n += 1
        target_sum = n // 2
        dx = 2.0 / n

        # Build LP.
        # define variables
        h = cp.Variable(n)
        lam = cp.Variable()
        constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum, lam >= 0, lam <= target_sum]
        # For each shift s
        for s in range(1, n // 2 + 1):
            length = n - s
            y = cp.Variable(length)
            # constraints
            constraints += [y >= 0,
                            y <= h[:length],
                            y <= h[s:],
                            y >= h[:length] + h[s:] - 1,
                            cp.sum(y) >= lam]
        # Objective
        prob = cp.Problem(cp.Maximize(lam), constraints)
        # Solve with ECOS or GLPK
        solver_used = None
        for solver in ('ECOS', 'GLPK', 'CBC'):
            try:
                prob.solve(solver=solver, verbose=False, max_iters=1000000)
                if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                    solver_used = solver
                    break
            except Exception:
                continue
        if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            continue

        h_val = np.array(h.value).astype(float)
        # Clip to [0,1], adjust sum
        h_val = np.clip(h_val, 0.0, 1.0)
        cur_sum = h_val.sum()
        # Adjust to exact target_sum using small correction.
        diff = target_sum - cur_sum
        if abs(diff) > 1e-8:
            # distribute diff uniformly across entries that are not at bounds.
            # Find indices where 0 < h_val < 1
            free = np.where((h_val > 1e-8) & (h_val < 1 - 1e-8))[0]
            if free.size > 0:
                adjust = diff / free.size
                h_val[free] += adjust
                h_val = np.clip(h_val, 0.0, 1.0)
            else:
                # fallback: adjust first element
                h_val[0] = min(max(h_val[0] + diff, 0.0), 1.0)
        # reclip after adjustment
        h_val = np.clip(h_val, 0.0, 1.0)
        # Compute C5
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        c5_val = float(corr.max() * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_val.copy()
            best_n = n
    # If still not found a solution (unlikely), fallback to simple random.
    if best_h is None:
        n = 70
        target_sum = n // 2
        h = np.zeros(n)
        h[:target_sum] = 1.0
        np.random.shuffle(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = float(corr.max() * dx)
        best_h = h
        best_n = n

    return (best_h.tolist(), float(best_c5), best_n)

def solve_lp(n):
    h = cp.Variable(n)
    lam = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2, lam >= 0, lam <= n/2]
    for s in range(1, n//2 + 1):
        length = n - s
        y = cp.Variable(length)
        constraints += [y >= 0,
                        y <= h[:length],
                        y <= h[s:],
                        y >= h[:length] + h[s:] - 1,
                        cp.sum(y) >= lam]
    prob = cp.Problem(cp.Maximize(lam), constraints)
    prob.solve(solver='ECOS', verbose=False)
    return h.value, lam.value
```

            diff = cur - target_sum
            if diff > 0:
                cand = np.where(v > 1e-12)[0]
                i = random.choice(cand.tolist())
                delta = min(v[i], diff)
                v[i] -= delta
            else:
                cand = np.where(v < 1 - 1e-12)[0]
                i = random.choice(cand.tolist())
                delta = min(1 - v[i], -diff)
                v[i] += delta
            cur = v.sum()
        return v

    # ------------------------------------------------------------------
    # Initial vector (random but feasible)
    # ------------------------------------------------------------------
    h = np.random.rand(n)
    h = project(h)

    # ------------------------------------------------------------------
    # Objective evaluation
    # ------------------------------------------------------------------
    def c5_of(v):
        corr = np.correlate(v, 1 - v, mode="full")
        return float(corr.max() * dx)

    cur_c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Simulated annealing / targeted moves
    # ------------------------------------------------------------------
    T = max(1e-5, cur_c5 * 0.02)       # initial temperature
    decay = 0.99997
    it = 0
    last_improve = 0
    max_it_without_improve = 30000   # restart if stagnates

    while time.time() - start < budget_s * 0.94:
        # --------------------------------------------------------------
        # Find worst shift (maximal complement)
        # --------------------------------------------------------------
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr))
        s = idx_max - (n - 1)          # signed shift
        s_abs = abs(s)

        # --------------------------------------------------------------
        # Targeted move: try to reduce the dominating contribution
        # --------------------------------------------------------------
        if s_abs <= n // 2:
            # compute contribution of each index for this shift
            if s >= 0:
                overlap_len = n - s
                contrib = np.empty(n, dtype=float)
                contrib[:overlap_len] = h[:overlap_len] * (1.0 - h[s:])
                contrib[overlap_len:] = h[overlap_len:]  # partner out of range → (1-0)=1
                i_best = int(np.argmax(contrib))
                if i_best < overlap_len:
                    p_best = i_best + s
                    eps_max = min(h[i_best], 1.0 - h[p_best])
                    if eps_max > 1e-12:
                        eps = eps_max * 0.5          # move half of the feasible amount
                        nh = h.copy()
                        nh[i_best] -= eps
                        nh[p_best] += eps
                        new_c5 = c5_of(nh)
                        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                            h = nh
                            cur_c5 = new_c5
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = h.copy()
                                last_improve = it
            else:
                # negative shift: symmetric handling
                s = -s
                overlap_len = n - s
                contrib = np.empty(n, dtype=float)
                contrib[s:] = h[s:] * (1.0 - h[:overlap_len])
                contrib[:s] = h[:s]                 # partner out of range
                i_best = int(np.argmax(contrib))
                if i_best >= s:
                    p_best = i_best - s
                    eps_max = min(h[i_best], 1.0 - h[p_best])
                    if eps_max > 1e-12:
                        eps = eps_max * 0.5
                        nh = h.copy()
                        nh[i_best] -= eps
                        nh[p_best] += eps
                        new_c5 = c5_of(nh)
                        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                            h = nh
                            cur_c5 = new_c5
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = h.copy()
                                last_improve = it

        # --------------------------------------------------------------
        # Random two‑coordinate exchange (preserve total mass)
        # --------------------------------------------------------------
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)        # increase i, decrease j
        max_down = min(hi, 1.0 - hj)      # decrease i, increase j
        if max_up + max_down > 1e-12:
            delta = (random.random() * (max_up + max_down)) - max_down
            if abs(delta) > 1e-14:
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_c5 = c5_of(nh)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        last_improve = it

        # --------------------------------------------------------------
        # Temperature decay
        # --------------------------------------------------------------
        T *= decay
        it += 1

        # --------------------------------------------------------------
        # Restart if stagnated too long
        # --------------------------------------------------------------
        if it - last_improve > max_it_without_improve:
            # re‑initialise a fresh feasible vector
            h = np.random.rand(n)
            h = project(h)
            cur_c5 = c5_of(h)
            last_improve = it

    # ------------------------------------------------------------------
    # Final safety: enforce exact sum (tiny numerical drift)
    # ------------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        # simple correction on the first element that can absorb the error
        for _ in range(10):
            k = random.randrange(n)
            if err > 0 and best_h[k] >= err:
                best_h[k] -= err
                break
            if err < 0 and best_h[k] <= 1 + err:
                best_h[k] -= err   # err negative → addition
                break

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), float(final_c5), n)
