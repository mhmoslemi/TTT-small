def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    # get initial_h if defined
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    target = 0.3808

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Candidate grid sizes
    n_candidates = [128, 256, 300]  # could adjust.

    # Use provided init size if matches.
    if init_h is not None and init_h.size not in n_candidates:
        n_candidates.insert(0, init_h.size)

    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        # check time left
        if time.time() - start_time > budget_s - 5:
            break

        # Build base solution
        if init_h is not None and init_h.size == n:
            base_h = init_h.copy()
        elif init_h is not None:
            base_h = _upscale_h(init_h, n)
        else:
            # start with uniform 0.5
            base_h = np.full(n, 0.5)

        # Project just in case
        base_h = _project_to_feasible(base_h, n)

        # Run SLSQP for refinement
        h_opt, c5_val = _solve_h(base_h, max_iter=400)

        # Keep best
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
            if best_c5 < target:
                break

        # Hill climbing further improve
        # allocate time portion: remaining time per candidate?
        time_remaining = budget_s - (time.time() - start_time)
        # allocate maybe half of remaining time for hill climbing on this n
        if time_remaining > 10:
            hc_time = min(0.4 * time_remaining, time_remaining - 2)
            h_hc, c5_hc = _hill_climb_h(h_opt, hc_time, time.time())
            if c5_hc < best_c5:
                best_h, best_c5, best_n = h_hc, c5_hc, n
                if best_c5 < target:
                    break

    if best_h is None:
        # fallback uniform
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_feasible(v: np.ndarray, n: int, tol: float = 1e-12, max_iter: int = 200) -> np.ndarray:
    """
    Clip v to [0,1] and enforce sum(v) == n/2 via water‑filling.
    """
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # decide which entries can move in the needed direction
        if diff > 0:
            free = np.where(v > 0)[0]
        else:
            free = np.where(v < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction (should be negligible)
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and v[i] < 1.0 - 1e-12:
                add = min(residual, 1.0 - v[i])
                v[i] += add
                residual -= add
            elif residual < 0 and v[i] > 1e-12:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(v, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Compute the C5 value for a given step function h (already feasible)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ----------------------------------------------------------------------
# SLSQP optimisation (same formulation as the reference code)
# ----------------------------------------------------------------------
def _ineq_constraint(x):
    """t - correlation >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    """Enforce integral (sum) constraint."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    """Minimise auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    """Run a single SLSQP optimisation starting from initial_h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective, x0, method="SLSQP", bounds=bounds, constraints=constraints, options=opts
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

def _hill_climb_h(start_h: np.ndarray, time_budget: float, start_time: float) -> tuple:
    """
    Greedy pairwise mass‑transfer hill climbing.
    Returns best_h, best_val found within the allocated time budget.
    """
    n = start_h.size
    best_h = start_h.copy()
    best_val = _c5_from_h(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    step_scale = 0.2  # typical magnitude of a move (relative to max possible delta)
    it = 0

    while time.time() - start_time < time_budget:
        it += 1

        # Every few thousand iterations reduce step size a bit
        if it % 4000 == 0:
            step_scale = max(step_scale * 0.85, 1e-4)

        # Random pair of indices
        i, j = np.random.choice(n, 2, replace=False)

        # Random direction of mass transfer
        if np.random.rand() < 0.5:
            # move from j to i
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                continue
            delta = step_scale * np.random.rand() * max_delta
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # move from i to j
            max_delta = min(1.0 - cur_h[j], cur_h[i])
            if max_delta <= 0:
                continue
            delta = step_scale * np.random.rand() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_val = _c5_from_h(new_h)

        # Greedy acceptance for the current state
        if new_val + 1e-14 < cur_val:
            cur_h = new_h
            cur_val = new_val

        # Record global improvement
        if new_val + 1e-14 < best_val:
            best_h = new_h
            best_val = new_val

        # Occasional random restart to avoid stagnation
        if it % 8000 == 0:
            rand = np.random.rand(n)
            rand = _project_to_feasible(rand, n)
            cur_h = rand
            cur_val = _c5_from_h(cur_h)

    return best_h, best_val
