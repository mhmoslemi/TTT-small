def corr_full_fft(h):
    n = len(h)
    N = 1 << (2*n-1).bit_length() # next power of two >= 2n
    # Actually to compute linear correlation: use FFT of length N where N >= 2n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1-h, n=N)
    conv = np.fft.irfft(H * G, n=N)
    # conv[i] = sum_{j} h[j] * (1-h[i-j]) with zero-padding (circular with length N), but linear valid for i from 0 to 2n-2
    # The result for full correlation consistent: conv[0:2*n-1]
    return conv[:2*n-1]

def gradient_of_corr_at_shift(h, k):
    n = len(h)
    g = np.zeros_like(h)
    # term 1: for i where i + k < n: g[i] += 1 - h[i+k]
    if k >= 0:
        g[:n - k] += 1 - h[k:]
        # term 2: for i where i - k >= 0: g[i] -= h[i - k]
        g[k:] -= h[:n - k]
    else:
        # negative shift
        kk = -k
        g[kk:] += 1 - h[:n - kk]
        g[:n - kk] -= h[kk:]
    return g

def project_to_sum_bounds(v, target):
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    # We'll use iterative loop.
    while abs(diff) > 1e-10:
        # identify free indices (0<x<1)
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if not np.any(free):
            # cannot adjust further; distribute to nearest bound?
            # fallback: distribute to all not at bound, but may saturate.
            # Here just break
            break
        per = diff / np.sum(free)
        # update free indices
        x[free] += per
        # clip to bounds
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return x

import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(v, target):
    """
    Project v onto the hyper‑plane sum(v)=target while respecting 0≤v≤1.
    Simple water‑filling style iteration.
    """
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # redistribute the residual uniformly among free variables
    for _ in range(1000):
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if not np.any(free):
            break
        per = diff / np.sum(free)
        x[free] += per
        # enforce bounds again
        np.clip(x, 0.0, 1.0, out=x)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    return np.clip(x, 0.0, 1.0)


def _gradient_and_corr(h):
    """
    Return gradient of the current maximal correlation term and the value of that term.
    The maximal term corresponds to some shift k (positive or negative).
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)          # shift (can be negative)
    g = np.zeros_like(h)

    if k >= 0:
        # part from h[i] * (1 - h[i+k])
        g[: n - k] += 1.0 - h[k:]
        # part from -h[i-k] * h[i]  (when i >= k)
        g[k:] -= h[: n - k]
    else:
        kk = -k
        g[kk:] += 1.0 - h[: n - kk]
        g[: n - kk] -= h[kk:]

    return g, corr[idx]


def _hill_climb(h, c5, deadline, rng):
    """Simple random pairwise mass‑exchange hill climbing."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose resolution and initialise
    # --------------------------------------------------------------
    # we use a finer grid than the supplied 45 points
    n = 180                     # even, gives target sum = n/2 = 90
    target = n / 2.0

    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_vec = np.asarray(init_vec, dtype=np.float64)
        m = init_vec.size
        # linear interpolation onto the finer grid
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init_vec)
        h0 = _adjust_sum(h0, target)
    else:
        # random start if no parent solution
        h0 = rng.random(n)
        h0 = _adjust_sum(h0, target)

    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    # --------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the maximal correlation term
    # --------------------------------------------------------------
    # allocate the bulk of the time budget
    grad_deadline = start + budget_s * 0.80
    step0 = 0.6                 # initial step size
    decay = 0.9995
    iteration = 0
    last_imp = 0
    restart_gap = 2000           # iterations without improvement before a reset

    cur_h = best_h.copy()
    cur_c5 = best_c5

    while time.time() < grad_deadline:
        # gradient of the currently active shift
        g, cur_corr = _gradient_and_corr(cur_h)

        # step size schedule
        step = step0 * (decay ** iteration)

        # trial move
        h_trial = cur_h - step * g
        h_trial = _adjust_sum(h_trial, target)

        trial_c5 = _evaluate_c5(h_trial)

        # accept if not worse (simple descent)
        if trial_c5 <= cur_c5 + 1e-14:
            cur_h, cur_c5 = h_trial, trial_c5

        # record global best
        if cur_c5 < best_c5 - 1e-12:
            best_h, best_c5 = cur_h.copy(), cur_c5
            last_imp = iteration

        # restart logic
        if iteration - last_imp > restart_gap:
            # perturb the best solution or start anew
            if rng.random() < 0.6:
                noise = rng.normal(scale=0.07, size=n)
                cur_h = _adjust_sum(best_h + noise, target)
            else:
                cur_h = _adjust_sum(rng.random(n), target)
            cur_c5 = _evaluate_c5(cur_h)
            last_imp = iteration

        iteration += 1

    # --------------------------------------------------------------
    # 3. Fine‑grained random hill climbing (uses remaining budget)
    # --------------------------------------------------------------
    hill_deadline = start + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, hill_deadline, rng)

    # final sanity projection
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))

    # --------------------------------------------------------------
    # 4. Return result
    # --------------------------------------------------------------
    return best_h, best_c5, n
