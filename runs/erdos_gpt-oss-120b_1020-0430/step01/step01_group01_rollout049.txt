def _gradient_shift(h, shift):
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[:n - s] = 1.0 - h[s:]   # deriv w.r.t h_i from first term (i from 0 to n-s-1)
        g[n - s:] = 1.0
        g[s:] += -h[:n - s]       # derivative from second term (i from s to n-1)
    else:
        s = -shift
        g[s:] = 1.0 - h[:n - s]
        g[:s] = 1.0
        g[:n - s] += -h[s:]
    return g
```

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# -------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    return _project_onto_capped_simplex(rng.rand(n), target)


# -------------------------------------------------
# Objective utilities
# -------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation for the given shift (the one attaining the max)."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# -------------------------------------------------
# Projected (sub)gradient descent on the max‑correlation objective
# -------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 1500, step0: float = 0.7,
                 decay: float = 0.997) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# -------------------------------------------------
# Smooth max for SLSQP
# -------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0, max_iter: int = 800) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# -------------------------------------------------
# Interpolation helper
# -------------------------------------------------
def _interpolate_to(v: np.ndarray, n_new: int) -> np.ndarray:
    n_old = len(v)
    if n_new == n_old:
        return v.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, v)

    # ----- final safety projection & return ----------
    h_best = _project_onto_capped_simplex(h_best, final_target)
    final_val = _compute_c5(h_best, dx_final)
    return (h_best, float(final_val), final_n)
