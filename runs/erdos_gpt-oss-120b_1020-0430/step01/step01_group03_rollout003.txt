def improve_h(h, rng, time_budget, start_time):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    T0 = 0.01   # maybe set based on initial c?
    T = T0
    it = 0
    while time.time() - start_time < time_budget:
        # decide move type: 70% random, 30% targeted
        if rng.random() < 0.3:
            # targeted
            # compute shift with max correlation
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            idx = np.argmax(corr)
            s = idx - (cur_h.size - 1)
            # compute w
            w = np.zeros_like(cur_h)
            n = cur_h.size
            if s >= 0:
                w[:n-s] += cur_h[s:]
                w[s:] += cur_h[:n-s]
            else:
                t = -s
                w[t:] += cur_h[:n-t]
                w[:n-t] += cur_h[t:]
            # choose indices
            # low weight among indices where cur_h > 1e-9
            low_candidates = np.where(cur_h > 1e-9)[0]
            high_candidates = np.where(cur_h < 1 - 1e-9)[0]
            if low_candidates.size == 0 or high_candidates.size == 0:
                # fallback to random
                new_h = perturb_pair(cur_h, rng)
                new_c = compute_c5(new_h)
            else:
                i_low = low_candidates[np.argmin(w[low_candidates])]
                i_high = high_candidates[np.argmax(w[high_candidates])]
                max_delta = min(cur_h[i_low], 1 - cur_h[i_high])
                if max_delta <= 0:
                    # fallback random
                    new_h = perturb_pair(cur_h, rng)
                    new_c = compute_c5(new_h)
                else:
                    # delta scaled
                    delta = max_delta * (0.5 + 0.5 * rng.random())  # up to max
                    new_h = cur_h.copy()
                    new_h[i_low] -= delta
                    new_h[i_high] += delta
                    new_c = compute_c5(new_h)
        else:
            # random perturb (pair transfer)
            i, j = rng.integers(0, cur_h.size, size=2)
            if i == j:
                continue
            max_eps = min(cur_h[i], 1 - cur_h[j])
            if max_eps <= 0:
                # try opposite direction maybe swap i/j
                # but we can skip
                continue
            eps = max_eps * (rng.random() * 0.5)  # random small
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
        # acceptance:
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T,1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        # update temperature every iteration
        it += 1
        T = T0 * (0.99 ** it)  # decays
    return best_h, best_c

def upsample(h, factor):
    n = h.size
    new_n = n * factor
    dx = 2.0 / n
    x_old = np.arange(n) * dx + dx/2.0
    x_new = np.arange(new_n) * (2.0 / new_n) + (2.0 / new_n)/2.0
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    # enforce sum: need to scale because linear interpolation changes integral.
    # current sum of h_new * dx_new = sum(h_new) * (2.0 / new_n) . We need it to be 1.
    # So we need to adjust: target sum should be new_n/2.
    target = new_n / 2.0
    diff = target - h_new.sum()
    # adjust by adding diff equally to all points within bounds? Might overshoot.
    # simplest: add diff / new_n to each (hard) then clip to [0,1] and then adjust more.
    # But we can just rescale linearly: h_new = h_new * (target / h_new.sum()). This preserves shape: if scaling factor >1 may cause some >1; we can clip after scaling and then re-balance with a correction loop like before.
    scale = target / h_new.sum()
    h_new = h_new * scale
    # clip then repair
    h_new = np.clip(h_new, 0.0, 1.0)
    # correct sum to target
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        # distribute diff randomly
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, new_n)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
    return h_new

def random_feasible_vector(n, rng):
    target = n/2.0
    h = np.zeros(n)
    remaining = target
    for i in range(n-1):
        max_val = min(1.0, remaining)
        val = rng.random() * max_val
        h[i] = val
        remaining -= val
    h[-1] = remaining  # should be <=1
    # slight random shuffling to break order
    rng.shuffle(h)
    return h

def random_pair_move(h, rng, max_step):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return None, None
    eps = max_eps * rng.random() * max_step
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h, eps

    delta = max_delta * max_step * rng.random()
    new_h = h.copy()
    new_h[i_low] -= delta

import time, math, numpy as np
from scipy.optimize import minimize

# globals for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def enforce_feasibility(h, target, rng):
    """Adjust h so that sum(h)=target while keeping 0≤h_i≤1."""
    diff = target - h.sum()
    n = h.size
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    target = n / 2.0
    h = np.zeros(n)
    remaining = target
    for i in range(n - 1):
        max_val = min(1.0, remaining)
        h[i] = rng.random() * max_val
        remaining -= h[i]
    h[-1] = remaining
    rng.shuffle(h)
    return h

def slsqp_optimize(h0, beta, max_iter=250):
    """Run a single SLSQP optimisation using a soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_feasibility(x, CURRENT_TARGET, np.random.default_rng())
    return x

def _weight_for_shift(h, shift):
    n = h.size
    w = np.zeros(n)
    if shift >= 0:
        w[:n - shift] += h[shift:]
        w[shift:] += h[:n - shift]
    else:
        t = -shift
        w[t:] += h[:n - t]
        w[:n - t] += h[t:]
    return w

def targeted_move(h, rng, max_step=0.2):
    """Move mass from a low‑weight index to a high‑weight one for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    w = _weight_for_shift(h, shift)
    low = np.where(h > 1e-12)[0]
    high = np.where(h < 1.0 - 1e-12)[0]
    if low.size == 0 or high.size == 0:
        return None
    i_low = low[np.argmin(w[low])]
    i_high = high[np.argmax(w[high])]
    max_delta = min(h[i_low], 1.0 - h[i_high])
    if max_delta <= 0:
        return None
    delta = max_delta * max_step * rng.random()
    new_h = h.copy()
    new_h[i_low] -= delta
    new_h[i_high] += delta
    return new_h

def anneal_opt(h0, rng, time_budget, start_time):
    """Simple simulated‑annealing optimiser."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    it = 0
    T0 = 0.02
    while time.time() - start_time < time_budget:
        if rng.random() < 0.3:
            new_h = targeted_move(cur_h, rng, max_step=0.2)
            if new_h is None:
                continue
        else:
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.2
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            accept = True
        else:
            T = T0 * (0.995 ** it)
            accept = rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12))
        if accept:
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        it += 1
    return best_h, best_c
