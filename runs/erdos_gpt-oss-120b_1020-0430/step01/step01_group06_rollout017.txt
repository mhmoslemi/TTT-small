                err += inc
    return h


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    except NameError:
        n = 70
        init = None
    target_sum = n // 2

    best_h = None
    best_c5 = float("inf")

    # ------------------------------------------------------------------
    # If an initial construction is supplied, normalise and improve it
    # ------------------------------------------------------------------
    if init is not None:
        v = np.clip(init, 0.0, 1.0)
        cur = v.sum()
        if abs(cur - target_sum) > 1e-12:
            if cur > 0:
                v *= target_sum / cur
            v = np.clip(v, 0.0, 1.0)
            cur = v.sum()
            while abs(cur - target_sum) > 1e-9:
                diff = cur - target_sum
                if diff > 0:
                    cand = np.where(v > 1e-12)[0]
                    i = random.choice(cand.tolist())
                    delta = min(v[i], diff)
                    v[i] -= delta
                else:
                    cand = np.where(v < 1 - 1e-12)[0]
                    i = random.choice(cand.tolist())
                    delta = min(1 - v[i], -diff)
                    v[i] += delta
                cur = v.sum()
        v = greedy_improve(v, max_iters=2000)
        c = c5_of(v)
        best_h = v.copy()
        best_c5 = c

    # ------------------------------------------------------------------
    # Build initial population
    # ------------------------------------------------------------------
    pop_size = 140
    population = []
    scores = []

    # deterministic seeds
    tm = thue_morse_seq(n)
    tm = repair_binary(tm, target_sum)
    population.append(tm.copy())
    scores.append(c5_of(tm))
    if scores[-1] < best_c5:
        best_c5 = scores[-1]
        best_h = tm.copy()

    phi = (1 + math.sqrt(5)) / 2.0
    st = sturmian_seq(phi, n)
    st = repair_binary(st, target_sum)
    population.append(st.copy())
    scores.append(c5_of(st))
    if scores[-1] < best_c5:
        best_c5 = scores[-1]
        best_h = st.copy()

    if init is not None:
        population.append(best_h.copy())
        scores.append(best_c5)

    while len(population) < pop_size:
        cand = random_balanced(n)
        population.append(cand)
        sc = c5_of(cand)
        scores.append(sc)
        if sc < best_c5:
            best_c5 = sc
            best_h = cand.copy()

    # ------------------------------------------------------------------
    # Evolutionary loop
    # ------------------------------------------------------------------
    generation = 0
    while time.time() - start < budget_s * 0.95:
        # tournament selection (size 5)
        def tournament():
            idxs = random.sample(range(pop_size), 5)
            best_i = idxs[0]
            best_score = scores[best_i]
            for i in idxs[1:]:
                if scores[i] < best_score:
                    best_i = i
                    best_score = scores[i]
            return population[best_i]

        p1 = tournament()
        p2 = tournament()
        child = crossover_uniform(p1, p2)
        child = mutate_flip(child, prob=0.02)
        child = np.rint(child)               # ensure binary
        child = np.clip(child, 0, 1)
        if int(child.sum()) != target_sum:
            child = repair_binary(child, target_sum)

        child = greedy_improve(child, max_iters=30)
        child_score = c5_of(child)

        # replace the worst individual if the child is better
        worst_i = 0
        worst_val = scores[0]
        for i in range(1, pop_size):
            if scores[i] > worst_val:
                worst_i = i
                worst_val = scores[i]
        if child_score < worst_val:
            population[worst_i] = child
            scores[worst_i] = child_score
            if child_score < best_c5:
                best_c5 = child_score
                best_h = child.copy()

        # occasional extra polishing of the global best
        if generation % 250 == 0 and generation > 0:
            best_h = greedy_improve(best_h, max_iters=100)
            best_c5 = c5_of(best_h)

        generation += 1

    # ------------------------------------------------------------------
    # Final intensive refinement
    # ------------------------------------------------------------------
    best_h = greedy_improve(best_h, max_iters=5000)
    final_c5 = c5_of(best_h)

    # final sum correction (tiny rounding errors)
    err = best_h.sum() - target_sum
    if abs(err) > 1e-8:
        if err > 0:
            while err > 1e-12:
                i = random.choice(np.where(best_h > 1e-12)[0])
                dec = min(best_h[i], err)
                best_h[i] -= dec
                err -= dec
        else:
            while err < -1e-12:
                i = random.choice(np.where(best_h < 1 - 1e-12)[0])
                inc = min(1.0 - best_h[i], -err)
                best_h[i] += inc
                err += inc
        final_c5 = c5_of(best_h)

    return (best_h.tolist(), final_c5, n)
