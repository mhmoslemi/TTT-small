def project_to_constraints(h):
    # ensure values in [0,1] and sum = target exactly
    target = n/2.0
    # clip
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    # distribute diff while respecting bounds
    if abs(diff) < 1e-12:
        return
    # we will add or subtract diff equally among random entries with capacity
    # find indices where we can adjust in the direction
    if diff > 0:
        # need to increase sum: find capacity to increase (1 - h[i])
        idx = np.random.permutation(n)
        for i in idx:
            cap = 1.0 - h[i]
            if cap <= 0: continue
            add = min(cap, diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # need to decrease sum: find capacity to decrease (h[i])
        idx = np.random.permutation(n)
        for i in idx:
            cap = h[i]  # amount we can subtract
            if cap <= 0: continue
            sub = min(cap, -diff)
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    # after loop diff should be close to zero

def anneal(start_h, time_limit):
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_metric = metric(h)
    cur_metric = best_metric
    start = time.time()
    # Param
    T0 = 0.06
    T_end = 1e-6
    max_step_initial = 0.5
    min_step = 1e-4
    # main loop
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        # temperature
        T = T0 * (T_end / T0) ** progress
        # step size decreasing linearly in log maybe
        max_step = max_step_initial * (0.01 ** progress)  # decays to 0.005? Wait
        # Actually we want max_step to go down from 0.5 to maybe 0.01 after full run:
        max_step = max_step_initial * (0.02 ** progress)  # at end ~0.5*0.02=0.01
        # Choose two indices
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        # compute allowable delta
        low = max(-hi, hj - 1.0)  # min delta so that hi+delta >=0 and hj-delta <=1
        high = min(1.0 - hi, hj) # max delta so hi+delta <=1 and hj-delta >=0
        if low >= high:
            continue
        # Use max_step to limit magnitude
        low_capped = max(low, -max_step)
        high_capped = min(high, max_step)
        if low_capped < high_capped:
            delta = random.uniform(low_capped, high_capped)
        else:
            delta = random.uniform(low, high)
        # apply move
        h[i] = hi + delta
        h[j] = hj - delta
        new_metric = metric(h)
        dC = new_metric - cur_metric
        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):

            if new_metric < best_metric:
                best_metric = new_metric
                best_h = h.copy()
        else:
            # reject: revert
            h[i] = hi
            h[j] = hj
    return best_h, best_metric

def refine(h, steps=5000, step_factor=0.02):
    n = h.shape[0]
    cur_metric = metric(h)
    for _ in range(steps):
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * step_factor
        if abs(delta) < 1e-12:
            continue
        h[i] = hi + delta
        h[j] = hj - delta
        new_metric = metric(h)
        if new_metric + 1e-12 < cur_metric:
            cur_metric = new_metric
        else:

            h[j] = hj
    return h, cur_metric
