def grad_shift(h, k):
    n = len(h)
    g = np.zeros_like(h)
    # term where p is first factor: p + k in range
    idx1 = np.arange(n)
    idx2 = idx1 + k
    mask = (idx2 >=0) & (idx2 < n)
    g[mask] = (1 - h[idx2[mask]])
    # term where p is second factor: p - k in range
    idx3 = np.arange(n)
    idx4 = idx3 - k
    mask2 = (idx4 >=0) & (idx4 < n)
    g[mask2] -= h[idx4[mask2]]
    return g
```

import time
import numpy as np
from scipy.optimize import minimize

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect {sum = target} using bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5: max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def grad_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation term at a given integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    idx = np.arange(n)

    # contribution where h[i] is the first factor
    idx2 = idx + shift
    mask = (0 <= idx2) & (idx2 < n)
    g[mask] = 1.0 - h[idx2[mask]]

    # contribution where h[i] appears as the second factor (1‑h)
    idx2 = idx - shift
    mask = (0 <= idx2) & (idx2 < n)
    g[mask] -= h[idx2[mask]]
    return g

def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

    np.random.seed(seed)
    start_time = time.time()

    n = 250                         # resolution
    target = n / 2.0                # integral constraint
    dx = 2.0 / n

    # ---- build initial pool -------------------------------------------------
    seeds = [make_triangle(n), make_sin2(n)]
    if "initial_h_values" in globals():
        try:

        except Exception:
            pass
    for _ in range(3):
        rnd = np.random.rand(n)
        seeds.append(project_onto_bounds_and_sum(rnd, target))

    # ---- pick the best seed -------------------------------------------------
    best_h = None
    best_val = np.inf
    for s in seeds:
        v = compute_c5(s)
        if v < best_val:
            best_val = v
            best_h = s.copy()
    h = best_h.copy()
    cur_val = best_val

    iter_cnt = 0
    # ---- main subgradient loop --------------------------------------------
    while time.time() - start_time < budget_s - 0.5:

        max_idx = int(np.argmax(corr))

        if cur_val < best_val:

            if best_val <= 0.3808:
