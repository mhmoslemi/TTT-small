import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation (same as np.correlate)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_binary_solution(n, target_int, fraction, rng):
    h = np.zeros(n, dtype=np.float64)
    perm = rng.permutation(n)
    h[perm[:target_int]] = 1.0
    if fraction > 0.0:
        h[perm[target_int]] = fraction
    return h

def _mutate(h, rng, swaps=1):
    """Move mass between two positions swaps times."""
    h_new = h.copy()
    n = h_new.shape[0]
    for _ in range(swaps):
        # positive entries
        pos = np.where(h_new > 1e-12)[0]
        # negative entries
        neg = np.where(h_new < 1.0 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        # ensure j != i
        possible = neg[neg != i]
        if possible.size == 0:
            continue
        j = rng.choice(possible)
        delta = min(h_new[i], 1.0 - h_new[j])
        if delta <= 1e-12:
            continue
        transfer = rng.random() * delta
        h_new[i] -= transfer
        h_new[j] += transfer
    return h_new

def _directed_move(h, shift, rng):
    """Transfer mass from a position with h>0 to its partner at given shift to reduce O(k)."""
    n = h.shape[0]
    if shift == 0:
        return h.copy()
    h_new = h.copy()
    if shift > 0:
        idx = np.where(
            (h_new > 1e-12) &
            (np.arange(n) + shift < n) &
            (h_new[np.arange(n) + shift] < 1.0 - 1e-12)
        )[0]
    else:
        idx = np.where(
            (h_new > 1e-12) &
            (np.arange(n) + shift >= 0) &
            (h_new[np.arange(n) + shift] < 1.0 - 1e-12)
        )[0]
    if idx.size == 0:
        return h_new
    i = rng.choice(idx)
    j = i + shift
    delta = min(h_new[i], 1.0 - h_new[j])
    if delta <= 1e-12:
        return h_new
    transfer = rng.random() * delta
    h_new[i] -= transfer
    h_new[j] += transfer
    return h_new

def _fix_sum(h, target):
    """Simple redistribution to enforce exact sum while respecting bounds."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    # If we need to add mass, prefer elements <1
    # If we need to remove, prefer elements >0
    if diff > 0:
        # add mass
        idx = np.where(h < 1.0 - 1e-12)[0]
        for i in idx:
            cap = 1.0 - h[i]
            add = min(diff, cap)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # remove mass
        idx = np.where(h > 1e-12)[0]
        for i in idx:
            cap = h[i]
            sub = min(-diff, cap)
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_vec = np.asarray(init_vec, dtype=np.float64)
        n = init_vec.size
    else:
        n = 100  # default resolution when no seed provided
        init_vec = None

    target = n / 2.0
    target_int = int(np.floor(target))
    fraction = target - target_int

    # Build initial population
    pop_size = 12
    population = []
    scores = []

    # include deterministic seeds
    if init_vec is not None:
        # produce a binary projection of the given vector
        order = np.argsort(-init_vec)
        h_proj = np.zeros(n, dtype=np.float64)
        h_proj[order[:target_int]] = 1.0
        if fraction > 0.0:
            h_proj[order[target_int]] = fraction
        population.append(h_proj)
        scores.append(_evaluate_c5(h_proj))

    # add a few structured patterns (alternating)
    base_pat = np.array([1,0,1,0,0], dtype=np.float64)  # density 0.6, will be trimmed
    rep = int(np.ceil(n / base_pat.size))
    pat = np.tile(base_pat, rep)[:n]
    # scale to required sum by random swapping of excess ones/zeros
    excess = pat.sum() - target
    if excess > 0:
        # turn some ones to zero
        ones_idx = np.where(pat == 1.0)[0]
        turn = rng.choice(ones_idx, size=int(excess), replace=False)
        pat[turn] = 0.0
    elif excess < 0:
        zeros_idx = np.where(pat == 0.0)[0]
        turn = rng.choice(zeros_idx, size=int(-excess), replace=False)
        pat[turn] = 1.0
    population.append(pat)
    scores.append(_evaluate_c5(pat))

    # fill the rest with random solutions
    while len(population) < pop_size:
        sol = _random_binary_solution(n, target_int, fraction, rng)
        population.append(sol)
        scores.append(_evaluate_c5(sol))

    # keep best answer
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # parameters for simulated annealing on the best candidate
    T = max(best_score * 0.2, 0.01)
    cooling = 0.9997

    # main optimisation loop
    while time.time() - start < budget_s - 0.5:
        # ---------- evolutionary step ----------
        # pick best and produce offspring
        parent_idx = int(np.argmin(scores))
        parent = population[parent_idx]
        offspring = _mutate(parent, rng, swaps=1)
        # occasionally perform larger mutation
        if rng.random() < 0.07:
            offspring = _mutate(offspring, rng, swaps=3)
        off_score = _evaluate_c5(offspring)

        # replace worst if improvement
        worst_idx = int(np.argmax(scores))
        if off_score < scores[worst_idx]:
            population[worst_idx] = offspring
            scores[worst_idx] = off_score
            # update global best
            if off_score < best_score:
                best_h = offspring.copy()
                best_score = off_score

        # ---------- directed improvement ----------
        # compute overlap vector for current best
        overlap_vec = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / n)
        k_max = int(np.argmax(overlap_vec) - (n - 1))
        # attempt a directed move targeting that shift
        candidate = _directed_move(best_h, k_max, rng)
        cand_score = _evaluate_c5(candidate)
        if cand_score < best_score:
            best_h = candidate

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Maximum overlap integral for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_binary_solution(n, target_int, fraction, rng):
    """Binary (or one fractional) feasible vector."""
    h = np.zeros(n, dtype=np.float64)
    perm = rng.permutation(n)
    h[perm[:target_int]] = 1.0
    if fraction > 0.0:
        h[perm[target_int]] = fraction
    return h

def _mutate(h, rng, swaps=1):
    """Move mass between two positions `swaps` times."""
    h_new = h.copy()
    n = h_new.shape[0]
    for _ in range(swaps):
        pos = np.where(h_new > 1e-12)[0]
        neg = np.where(h_new < 1.0 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        cand = neg[neg != i]
        if cand.size == 0:
            continue
        j = rng.choice(cand)
        delta = min(h_new[i], 1.0 - h_new[j])
        if delta <= 1e-12:
            continue
        transfer = rng.random() * delta
        h_new[i] -= transfer
        h_new[j] += transfer
    return h_new

def _directed_move(h, shift, rng):
    """Transfer mass from a positive entry to its partner at `shift`."""
    if shift == 0:
        return h.copy()
    n = h.shape[0]
    h_new = h.copy()
    if shift > 0:
        idx = np.where(
            (h_new > 1e-12)
            & (np.arange(n) + shift < n)
            & (h_new[np.arange(n) + shift] < 1.0 - 1e-12)
        )[0]
    else:
        idx = np.where(
            (h_new > 1e-12)
            & (np.arange(n) + shift >= 0)
            & (h_new[np.arange(n) + shift] < 1.0 - 1e-12)
        )[0]
    if idx.size == 0:
        return h_new
    i = rng.choice(idx)
    j = i + shift
    delta = min(h_new[i], 1.0 - h_new[j])
    if delta <= 1e-12:
        return h_new
    transfer = rng.random() * delta
    h_new[i] -= transfer
    h_new[j] += transfer
    return h_new

def _fix_sum(h, target):
    """Simple redistribution to hit the exact sum while respecting bounds."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        for i in idx:
            cap = 1.0 - h[i]
            add = min(diff, cap)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        idx = np.where(h > 1e-12)[0]
        for i in idx:
            cap = h[i]
            sub = min(-diff, cap)
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h
