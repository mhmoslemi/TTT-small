    T_end = 1e-6
    max_step = 0.2
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress

        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c = corr[max_idx] * (2.0 / n)  # Actually c5_metric computes max*dx, we can just compute directly.
        # but we need to compute after each move? We'll compute C each iteration to find worst lag.
        # compute L_star
        L_star = max_idx - (n - 1)
        # compute weight w
        w = np.zeros(n)
        if L_star >= 0:
            L = L_star
            if L < n:
                w[:n-L] += h[L:]
                w[L:] += h[:n-L]
        else:
            L = -L_star
            if L < n:
                w[L:] += h[:n-L]
                w[:n-L] += h[L:]

        # select i and j
        weights_i = w + 1e-12
        prob_i = weights_i / weights_i.sum()
        i = np.random.choice(n, p=prob_i)

        max_w = w.max()
        # for low weights
        low_weights = (max_w - w) + 1e-12
        prob_j = low_weights / low_weights.sum()
        j = np.random.choice(n, p=prob_j)
        if i == j:
            continue

        hi = h[i]; hj = h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # limit step
        low_cap = max(low, -max_step)
        high_cap = min(high, max_step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta

        # compute new C
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c = new_corr.max() * (2.0 / n)

import time

def _c5_metric(h):
    """Max overlap C5 for sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_vector(n):
    """Create a feasible starting vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h

def _biased_anneal(start_h, max_seconds):
    """Simulated‑annealing that prefers moves improving the worst lag."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0 = 0.05
    T_end = 1e-6
    max_step = 0.25
    # number of worst lags to bias against
    k_worst = 4

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress

        # correlation and worst lags
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_c = np.max(corr) * (2.0 / n)
        # indices of the k largest overlaps (worst lags)
        if k_worst < corr.size:
            worst_idx = np.argpartition(corr, -k_worst)[-k_worst:]
        else:
            worst_idx = np.arange(corr.size)
        lags = worst_idx - (n - 1)

        # compute weight for each position: sum of neighbours for the considered lags
        w = np.zeros(n)
        for L in lags:
            if L >= 0:
                Lp = L
                if Lp < n:
                    w[:n - Lp] += h[Lp:]          # forward neighbour
                    w[Lp:] += h[:n - Lp]          # backward neighbour
            else:
                Lp = -L
                if Lp < n:
                    w[Lp:] += h[:n - Lp]          # forward (i+L)
                    w[:n - Lp] += h[Lp:]          # backward (i-L)

        # pick i (receiver) where weight is large, j (donor) where weight is small
        wi = w + 1e-12
        prob_i = wi / wi.sum()
        i = np.random.choice(n, p=prob_i)

        max_w = w.max()
        wj = (max_w - w) + 1e-12
        prob_j = wj / wj.sum()
        j = np.random.choice(n, p=prob_j)

        if i == j:

        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)

            h[i] = hi

        # binary (or near‑binary) random start respecting the sum constraint
        target = n / 2.0
        ones = int(target)
        frac = target - ones
        h_rand = np.zeros(n, dtype=np.float64)
        idx = np.random.permutation(n)
        h_rand[idx[:ones]] = 1.0
        if frac > 0:
            h_rand[idx[ones]] = frac

        # allocate a slice of the remaining budget
        remaining = total_budget - (time.time() - start_all)
        slice_time = min(remaining * 0.3, 25.0)
        cand_h, cand_c = _biased_anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h

    # final deterministic hill‑climbing with tiny moves
    for _ in range(4):
        improved = False
        for _ in range(2500):
            i, j = random.sample(range(n), 2)
            hi, hj = best_h[i], best_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = random.uniform(low, high) * 0.02
            best_h[i] = hi + delta
            best_h[j] = hj - delta
            new_c = _c5_metric(best_h)
            if new_c + 1e-11 < best_c:
                best_c = new_c
                improved = True
            else:
                best_h[i] = hi
                best_h[j] = hj
        if not improved:
            break

    # enforce exact sum (tiny rounding errors)
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # final evaluation
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
