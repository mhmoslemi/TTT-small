import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(v: np.ndarray, n: int, alpha: float, smooth: float) -> float:
    """Smooth approximation of max overlap, using the half‑vector v."""
    # build full h from its half‑vector
    m = n // 2
    h = np.empty(n, dtype=float)
    h[:m] = v
    h[m:] = 1.0 - v[::-1]

    conv = np.correlate(h, 1.0 - h, mode="full")
    smooth_pen = smooth * np.sum(np.diff(v) ** 2)
    return float(logsumexp(alpha * conv) / alpha + smooth_pen)


def build_h_from_v(v: np.ndarray, n: int) -> np.ndarray:
    """Construct full h from its first half v (complement symmetry)."""
    m = n // 2
    h = np.empty(n, dtype=float)
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape on n points, normalised to integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    # renormalise (integral must be 1)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

# Optimisation of a half‑vector
# ------------------------------------------------------------

def optimise_half(v0: np.ndarray, n: int, deadline: float,
                  smooth: float = 5e-4) -> tuple:
    """Two‑stage L‑BFGS‑B optimisation of the half‑vector."""
    m = n // 2
    bounds = [(0.0, 1.0)] * m

    # stage 1 – smooth landscape
    res1 = minimize(
        lambda v: softmax_c5(v, n, alpha=6.0, smooth=smooth),
        v0,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 180, "ftol": 1e-9},
    )
    if not res1.success or time.time() > deadline:
        return None, None

    # stage 2 – sharper approximation
    res2 = minimize(
        lambda v: softmax_c5(v, n, alpha=30.0, smooth=smooth),
        res1.x,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 180, "ftol": 1e-9},
    )
    if not res2.success or time.time() > deadline:
        return None, None

    v_opt = np.clip(res2.x, 0.0, 1.0)
    h_opt = build_h_from_v(v_opt, n)
    c5_val = compute_c5(h_opt)
    return v_opt, c5_val


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.94  # safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Candidate sample sizes (must be even for the symmetry trick)
    candidate_ns = [80, 100, 120, 150, 180, 200, 256, 300]

    # Attempt to incorporate a parent construction if supplied
    parent_h = None
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    for n in candidate_ns:
        if time.time() > deadline:
            break
        if n % 2 != 0:
            continue
        m = n // 2

        starts = []

        # 1. Linear monotone start (0 → 1)
        starts.append(np.linspace(0.0, 1.0, m))

        # 2. Triangular‑derived start (first half of a triangular shape)
        tri_h = generate_triangular(n)
        starts.append(tri_h[:m])

        # 3. Random feasible starts
        for _ in range(5):
            starts.append(rng.random(m))

        # 4. Perturbed linear starts
        base = np.linspace(0.0, 1.0, m)
        for _ in range(3):
            pert = base + rng.uniform(-0.18, 0.18, size=m)
            starts.append(np.clip(pert, 0.0, 1.0))

        # 5. Parent‑derived start if available
        if parent_h is not None:
            ph = parent_h
            if ph.size != n:
                ph = upsample(ph, n)
            starts.append(ph[:m])

        # Deduplicate very close starts (optional, cheap)
        uniq = []
        for s in starts:
            if not any(np.allclose(s, u, atol=1e-12) for u in uniq):
                uniq.append(s)
        starts = uniq

        # Optimise each start
        for v0 in starts:
            if time.time() > deadline:
                break
            v_opt, c5_val = optimise_half(v0, n, deadline)
            if v_opt is not None and c5_val < best_c5 - 1e-9:
                best_c5 = c5_val
                best_n = n
                best_h = build_h_from_v(v_opt, n)
                # early exit if we beat the target bound
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # Fallback in the unlikely event that nothing succeeded
    if best_h is None:
        n = 80
        m = n // 2
        v = np.linspace(0.0, 1.0, m)
        best_h = build_h_from_v(v, n)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final sanity check – ensure the integral is correct up to numeric tolerance
    target_sum = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # very slight projection onto the feasible hyperplane
        best_h = best_h + (target_sum - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
