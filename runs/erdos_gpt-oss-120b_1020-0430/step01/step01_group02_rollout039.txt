import time
import numpy as np
from scipy.optimize import minimize

def _c5_val(h):
    """Maximum overlap integral for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """Project x onto the simplex sum(x)=target with 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple redistribution loop
    while abs(diff) > 1e-12:
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0.0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Simulated‑annealing + hill‑climb optimisation for the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # initialise
    # --------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h = np.asarray(init_vec, dtype=np.float64).copy()
        n = h.size
    else:
        n = 45                     # default size
        h = rng.random(n)
        target = n / 2.0
        h = h * (target / h.sum())
    target = n / 2.0
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)

    # best known solution
    best_h = h.copy()
    best_score = _c5_val(best_h)

    # --------------------------------------------------------------
    # simulated annealing (majority of the budget)
    # --------------------------------------------------------------
    dx = 2.0 / n
    total_budget = budget_s
    end_anneal = start + total_budget * 0.80
    T0 = 0.05                     # initial temperature
    T_end = 1e-6
    log_factor = np.log(T_end / T0)

    cur_h = h.copy()
    # initial score and correlation
    corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_score = np.max(corr) * dx

    while time.time() < end_anneal:
        elapsed = time.time() - start
        T = T0 * np.exp(log_factor * elapsed / total_budget)

        # compute worst lag (the one attaining the max)
        max_idx = int(np.argmax(corr))
        L = max_idx - (n - 1)          # signed lag
        L_abs = abs(L)

        # ------------------------------------------------------------------
        # propose a move: 60% targeted, 40% random pair
        # ------------------------------------------------------------------
        if rng.random() < 0.6 and L_abs < n:
            # region that participates in the worst lag
            if L >= 0:
                i_start, i_end = 0, n - L_abs
                shift_start = L_abs
            else:
                i_start, i_end = L_abs, n
                shift_start = 0
            length = i_end - i_start
            if length > 0:
                seg_h = cur_h[i_start:i_end]
                seg_shift = cur_h[shift_start:shift_start + length]
                contrib = seg_h * (1.0 - seg_shift)

                src_candidates = np.where((seg_h > 1e-12) & (contrib > 0))[0]
                dst_candidates = np.where((seg_h < 1.0 - 1e-12) & (contrib < 1.0))[0]

                if src_candidates.size and dst_candidates.size:
                    # pick among a few extreme contributors to avoid deterministic loops
                    k_src = min(3, src_candidates.size)
                    k_dst = min(3, dst_candidates.size)

                    src_idx = src_candidates[np.argpartition(-contrib[src_candidates], k_src-1)[:k_src]]
                    src_off = rng.choice(src_idx)
                    src = i_start + src_off

                    dst_idx = dst_candidates[np.argpartition(contrib[dst_candidates], k_dst-1)[:k_dst]]
                    dst_off = rng.choice(dst_idx)
                    dst = i_start + dst_off
                else:
                    src, dst = rng.integers(0, n, size=2)
                    while src == dst:
                        dst = rng.integers(0, n)
            else:
                src, dst = rng.integers(0, n, size=2)
                while src == dst:
                    dst = rng.integers(0, n)
        else:
            src, dst = rng.integers(0, n, size=2)
            while src == dst:
                dst = rng.integers(0, n)

        # ------------------------------------------------------------------
        # amount of mass to transfer
        # ------------------------------------------------------------------
        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            # try a tiny random jitter instead of a full move
            cur_h += (rng.random(n) - 0.5) * 0.001
            cur_h = _adjust_sum(np.clip(cur_h, 0.0, 1.0), target)
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            cur_score = np.max(corr) * dx
            continue
        delta = rng.random() * max_move
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta

        # ------------------------------------------------------------------
        # evaluate
        # ------------------------------------------------------------------
        new_score = _c5_val(new_h)

        # accept?
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
            cur_h, cur_score = new_h, new_score
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score

    # --------------------------------------------------------------
    # greedy hill climbing (use the remainder of the budget)
    # --------------------------------------------------------------
    end_hill = start + total_budget * 0.95
    while time.time() < end_hill:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(best_h[src], 1.0 - best_h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        cand = best_h.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _c5_val(cand)
        if cand_score + 1e-12 < best_score:
            best_h, best_score = cand, cand_score

    # --------------------------------------------------------------
    # optional final SLSQP refinement (few short restarts)
    # --------------------------------------------------------------
    remaining = start + total_budget - time.time()
    if remaining > 4.0:
        def obj(x):
            return _c5_val(x)

        eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
        bounds = [(0.0, 1.0)] * n
        # a handful of jittered starts
        for _ in range(8):
            if time.time() > start + total_budget - 1.0:
                break
            x0 = best_h + rng.normal(scale=0.03, size=n)
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = _adjust_sum(x0, target)
            res = minimize(
                obj,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[eq_cons],
                options={"maxiter": 150, "ftol": 1e-9, "disp": False},
            )
            cand = np.clip(res.x, 0.0, 1.0)
            cand = _adjust_sum(cand, target)
            cand_score = _c5_val(cand)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score

    # final feasibility fix
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_score = float(_c5_val(best_h))
    return best_h, best_score, n
