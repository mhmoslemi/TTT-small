def hill_climb_binary(b, n, rng, time_limit):
    best = b.copy()
    best_val = compute_c5_sym(best)
    cur = b.copy()
    cur_val = best_val
    T0 = 0.1
    # schedule: T = T0 * (time_left / total_time)
    while time.time() < time_limit:
        # pick swap indices
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        cur[i], cur[j] = cur[j], cur[i]
        new_val = compute_c5_sym(cur)
        # Accept if lower, else revert with probability exp(-(new_val - cur_val)/T)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T0, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best = cur.copy()
                best_val = new_val
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
        # decay temperature based on elapsed ratio
        elapsed = time.time() - start_time
        T = T0 * (1 - elapsed / (time_limit - start_time)) # linear decay
    return best, best_val

def hill_climb_binary(b_init, n, rng, deadline, global_start):

    h = np.empty(n, dtype=float)

    cur_c5 = compute_c5(h)
    best_b = b.copy()
    best_c5 = cur_c5

    # We'll perform swaps until time out or no improvement after many tries
    # Keep a counter for non-improving attempts
    no_improve_counter = 0
    max_no_improve = 5000   # after many tries without improvement, exit
    while time.time() < deadline:
        # select i,j where b[i]!=b[j]

            continue
        # swap
        b[i], b[j] = b[j], b[i]
        # update full h at positions i, j and their mirrors
        i_m = n - 1 - i
        j_m = n - 1 - j
        h[i] = b[i]
        h[i_m] = b[i]
        h[j] = b[j]
        h[j_m] = b[j]

def improve_continuous(h0, n, rng, deadline):
    target_sum = n / 2.0
    cons = {"type":"eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0,1.0)]*n
    # first stage soft objective
    def obj_alpha(alpha):
        return lambda h: softmax_c5(h, alpha)
    # attempt stage 1
    try:
        res1 = minimize(obj_alpha(5.0), h0, method="SLSQP", bounds=bounds, constraints=cons,
                        options={"maxiter":500, "ftol":1e-9, "disp":False})
        if not res1.success:
            return h0, compute_c5(h0)
        # stage2
        res2 = minimize(obj_alpha(30.0), res1.x, method="SLSQP", bounds=bounds, constraints=cons,
                        options={"maxiter":500, "ftol":1e-9, "disp":False})
        if not res2.success:
            return res1.x, compute_c5(res1.x)
        h_opt = np.clip(res2.x, 0.0, 1.0)
        # enforce sum exactly
        cur_sum = h_opt.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            h_opt = h_opt + diff / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        return h_opt, compute_c5(h_opt)
    except Exception as e:
        return h0, compute_c5(h0)

def anneal_h(h_init, n, rng, deadline):
    h = h_init.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    max_step = 0.1  # initial step size
    temperature = 0.1
    # schedule parameters:
    start_time = time.time()
    while time.time() < deadline:
        # adapt temperature linearly decreasing with time
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        T = max(0.001, temperature * (1 - fraction))
        step = max_step * (1 - fraction) + 0.001
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(-step, step)
        # adjust if would violate bounds
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0:
            delta = -h[i]  # set hi=0
            hj = h[j] - delta
        elif hi > 1:
            delta = 1 - h[i]
            hj = h[j] - delta
        if hj < 0:
            delta = h[j]  # set hj=0 (i.e., subtract delta = h[j])
            hi = h[i] + delta

            delta = h[j] - 1  # ensure hj=1

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Basic evaluation utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation to max using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


# ------------------------------------------------------------
# Simple helpers
# ------------------------------------------------------------
def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    return np.interp(new_x, old_x, arr)


def random_binary(half: int, ones: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length `half` with exactly `ones` entries = 1."""
    b = np.zeros(half, dtype=int)
    idx = rng.choice(half, size=ones, replace=False)
    b[idx] = 1
    return b


def binary_start_from_initial(half: int, ones: int, rng: np.random.Generator):
    """Derive a binary half‑vector from the global `initial_h_values` if present."""
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except NameError:
        return None
    # upsample to the full length that will be used later
    n_full = half * 2
    parent_up = upsample(parent, n_full)
    # keep only the first half for symmetry
    vals = parent_up[:half]
    # take the largest `ones` entries
    idx = np.argpartition(vals, -ones)[-ones:]
    b = np.zeros(half, dtype=int)
    b[idx] = 1
    return b


# ------------------------------------------------------------
# Binary hill‑climbing
# ------------------------------------------------------------
def hill_climb_binary(b_start: np.ndarray, n: int,
                      rng: np.random.Generator, deadline: float) -> tuple:
    """Greedy 0/1 swap search on a symmetric function."""
    half = n // 2
    target_ones = half // 2
    b = b_start.copy()
    # build full symmetric h
    h = np.empty(n, dtype=float)
    h[:half] = b
    h[half:] = b[::-1]
    cur_c5 = compute_c5(h)
    best_b = b.copy()
    best_c5 = cur_c5
    # counters for stagnation
    no_improve = 0
    max_no_improve = 5000
    while time.time() < deadline:
        i = rng.integers(half)
        j = rng.integers(half - 1)
        while j == i:
            j = rng.integers(half)
        if b[i] == b[j]:
            continue
        # perform the swap
        b[i], b[j] = b[j], b[i]
        i_m = n - 1 - i
        j_m = n - 1 - j
        h[i] = b[i]
        h[i_m] = b[i]
        h[j] = b[j]
        h[j_m] = b[j]
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            # revert
            b[i], b[j] = b[j], b[i]
            h[i] = b[i]
            h[i_m] = b[i]
            h[j] = b[j]
            h[j_m] = b[j]
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return best_b, best_c5


# ------------------------------------------------------------
# Continuous simulated‑annealing refinement
# ------------------------------------------------------------
def anneal_continuous(h_start: np.ndarray, n: int,
                     rng: np.random.Generator, deadline: float) -> tuple:
    """Two‑point random walk on the simplex with a simple temperature schedule."""
    h = h_start.astype(float).copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # schedule parameters
    T0 = 0.02                # initial temperature
    step0 = 0.2              # initial max delta
    eps = 1e-12
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time + eps)
        T = max(eps, T0 * (1.0 - frac))
        step = max(eps, step0 * (1.0 - frac))

        i, j = rng.choice(n, size=2, replace=False)

        # feasible delta range that keeps both entries in [0,1]
        delta_min = max(-h[i], h[j] - 1.0)
        delta_max = min(1.0 - h[i], h[j])
        if delta_min > delta_max:
            continue

        # propose a delta (biased towards zero by shrinking step)
        delta = rng.uniform(-step, step)
        delta = np.clip(delta, delta_min, delta_max)

        if abs(delta) < 1e-15:
            continue

        # apply the move
        h[i] += delta
        h[j] -= delta

        new_c5 = compute_c5(h)
        dE = new_c5 - cur_c5

        if dE < -1e-12 or rng.random() < np.exp(-dE / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        # early exit if we already beat the target
        if best_c5 <= 0.3808:
            break

    return best_h, best_c5


# ------------------------------------------------------------
# Optional final local refinement (SLSQP on smooth surrogate)
# ------------------------------------------------------------
def refine_with_slsqp(h0: np.ndarray, n: int, rng: np.random.Generator,
                     deadline: float) -> tuple:
    """Two‑stage SLSQP on the soft‑max surrogate, starting from `h0`."""
    target_sum = n / 2.0
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n

    def run_stage(alpha, start):
        if time.time() >= deadline:
            return None
        res = minimize(lambda h: softmax_c5(h, alpha),
                       start, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": 400, "ftol": 1e-9, "disp": False})
        return res

    res1 = run_stage(5.0, h0)
    if res1 is None or not res1.success:
        return h0, compute_c5(h0)

    res2 = run_stage(30.0, res1.x)
    if res2 is None or not res2.success:
        return res1.x, compute_c5(res1.x)

    h_opt = np.clip(res2.x, 0.0, 1.0)
    # enforce exact sum (tiny correction)
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-12:
        h_opt += diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # keep a safety margin

    # --------------------------------------------------------
    # Choose a resolution (larger = finer, still cheap enough)
    # --------------------------------------------------------
    n = 200                           # must be even
    half = n // 2
    target_ones = half // 2          # exactly n/4 ones in the half‑vector

    # --------------------------------------------------------
    # Build a pool of starting binary vectors
    # --------------------------------------------------------
    starts = []

    b0 = binary_start_from_initial(half, target_ones, rng)
    if b0 is not None:
        starts.append(b0)

    # a few handcrafted seeds (e.g. a block of ones in the centre)
    centre = np.zeros(half, dtype=int)
    centre[half // 4: 3 * half // 4] = 1
    centre[:target_ones] = 1  # ensure exactly the right number
    starts.append(centre.copy())

    # random binaries
    for _ in range(12):
        starts.append(random_binary(half, target_ones, rng))

    best_h = None
    best_c5 = np.inf
    best_n = n

    # --------------------------------------------------------
    # Binary hill‑climbing on each start
    # --------------------------------------------------------
    for b_start in starts:
        if time.time() >= time_limit:
            break
        b_opt, c5_opt = hill_climb_binary(b_start, n, rng, time_limit)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = np.concatenate([b_opt, b_opt[::-1]]).astype(float)

    # --------------------------------------------------------
    # Continuous simulated‑annealing refinement
    # --------------------------------------------------------
    if best_h is not None and time.time() < time_limit:
        h_ref, c5_ref = anneal_continuous(best_h, n, rng, time_limit)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # --------------------------------------------------------
    # Final SLSQP polish (optional, runs only if time permits)
    # --------------------------------------------------------
    if best_h is not None and time.time() < time_limit:
        h_polished, c5_polished = refine_with_slsqp(best_h, n, rng, time_limit)
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_h = h_polished

    # --------------------------------------------------------
    # Safety fallback (should never be needed)
    # --------------------------------------------------------
    if best_h is None:
        # simple half‑block construction
        half_block = np.concatenate([np.ones(half), np.zeros(half)])
        best_h = half_block.astype(float)
        best_c5 = compute_c5(best_h)

    # Final sanity checks
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), best_n
