def project(x, target):
   # clip then adjust sum using waterfilling algorithm.
   x = np.clip(x, 0.0, 1.0)
   # loop
   for _ in range(100):
       diff = target - x.sum()
       if abs(diff) < 1e-12:
               break
       # compute free indices (where we can move in direction)
       if diff > 0:
           free = np.where(x < 1)[0]
           # capacity of each is 1-x[i]; total capacity sum_cap
           cap = 1 - x[free]
           total_cap = cap.sum()
           if total_cap < diff - 1e-12:
               # cannot reach target, cap them all to 1
               x[free] = 1.0
               continue
           # allocate amount proportionally to capacities?
           # we can give per index: per = diff / len(free). Then adjust if exceed cap.
           per = diff / len(free)
           inc = np.minimum(per, cap)
           x[free] += inc
       else:
           free = np.where(x > 0)[0]
           cap = x[free]
           total_cap = cap.sum()
           if total_cap < -diff - 1e-12:
               x[free] = 0.0
               continue
           per = diff / len(free) # diff negative, per negative
           dec = np.minimum(-per, cap) # positive amount to subtract
           x[free] -= dec
   return x

def hill_climb(h, steps, rng, temp_schedule=None):
   current = h.copy()
   current_c5 = eval_c5(current)
   n = len(current)
   # for each step
   for step in range(steps):
       i, j = rng.integers(0, n, size=2)
       if i == j: continue
       # random direction:
       if rng.random() < 0.5:
           src, dst = i, j
       else:
           src, dst = j, i
       max_move = min(current[src], 1 - current[dst])
       if max_move <= 1e-12:
           continue
       # step size: maybe random uniform fraction of max_move
       delta = rng.random() * max_move
       # optionally scale with temperature
       if temp_schedule is not None:
           T = temp_schedule(step)
           # propose larger moves as T high
       # apply move
       new = current.copy()
       new[src] -= delta
       new[dst] += delta
       # evaluate
       new_c5 = eval_c5(new)
       if new_c5 < current_c5 - 1e-12:
           current, current_c5 = new, new_c5
       else:
           # maybe accept with probability
           if temp_schedule is not None:
               T = temp_schedule(step)
               # accept if exp(-(new_c5-current_c5)/T) > rand
               if rng.random() < np.exp(-(new_c5 - current_c5) / T):
                   current, current_c5 = new, new_c5
   return current, current_c5

def refine_with_slsqp(h, target, rng):

   def obj(x):
       # compute correlation and max
       corr = np.correlate(x, 1.0 - x, mode='full')
       return np.max(corr) * (2.0 / len(x))

   # constraints: equality sum = target; bounds [0,1]
   bounds = [(0.0, 1.0)] * len(h)
   eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
   # Use best as starting point
   x0 = h.copy()
   # optionally small jitter
   x0 = np.clip(x0 + rng.normal(scale=0.01, size=len(h)), 0.0, 1.0)
   x0 = project(x0, target)
   res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=[eq_cons],
                  options={'maxiter': 300, 'ftol':1e-10, 'disp': False})
   if res.success:
       new_h = project(res.x, target)
       new_c5 = _evaluate_c5(new_h)
       if new_c5 < _evaluate_c5(h):
           return new_h, new_c5
   return h, _evaluate_c5(h)

def project(x, target, eps=1e-12):
    # x is np.array float.
    # Clip:
    x = np.clip(x, 0.0, 1.0)
    # Use iterative method:
    # We'll try to adjust sum by moving mass among free indexes.
    # We'll loop up to maybe 2 * n times.
    for _ in range(2 * len(x)):
        diff = target - x.sum()
        if abs(diff) <= eps:
            break
        if diff > 0:
            # need to increase some entries
            free = np.where(x < 1 - eps)[0]
            if free.size == 0:
                break
            # capacity available each
            cap = 1 - x[free]
            total_cap = cap.sum()
            if total_cap <= diff:
                # fill all
                x[free] = 1.0
                continue
            # Distribute proportionally to capacity
            inc = diff * cap / total_cap
            x[free] += inc
        else:
            # need to decrease some entries
            free = np.where(x > eps)[0]
            if free.size == 0:
                break
            cap = x[free]  # amount we can subtract
            total_cap = cap.sum()
            if total_cap <= -diff:
                # zero out all
                x[free] = 0.0
                continue
            dec = -diff * cap / total_cap
            x[free] -= dec
    # final clip just to be safe
    return np.clip(x, 0.0, 1.0)

import time, math
import numpy as np
from scipy.optimize import minimize

def _eval_c5(h: np.ndarray) -> float:
    """max overlap integral for step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_to_sum(x: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and adjust sum to target via water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(2 * x.size):
        diff = target - x.sum()
        if abs(diff) <= eps:
            break
        if diff > 0:
            free = np.where(x < 1.0 - eps)[0]
            if free.size == 0:
                break
            cap = 1.0 - x[free]
            total = cap.sum()
            if total <= diff:
                x[free] = 1.0
                continue
            inc = diff * cap / total
            x[free] += inc
        else:
            free = np.where(x > eps)[0]
            if free.size == 0:
                break
            cap = x[free]
            total = cap.sum()
            if total <= -diff:
                x[free] = 0.0
                continue
            dec = -diff * cap / total
            x[free] -= dec
    return np.clip(x, 0.0, 1.0)

def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with prescribed sum."""
    x = rng.random(n)
    x = x * (target / x.sum())
    return _project_to_sum(x, target)

def _hill_climb(h: np.ndarray, moves: int, rng: np.random.Generator) -> tuple:
    """Greedy pairwise mass‑exchange hill climbing."""
    n = h.size
    best = h.copy()
    best_c5 = _eval_c5(best)
    for _ in range(moves):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(best[src], 1.0 - best[dst])
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        new = best.copy()
        new[src] -= delta
        new[dst] += delta
        new_c5 = _eval_c5(new)
        if new_c5 + 1e-12 < best_c5:
            best, best_c5 = new, new_c5
    return best, best_c5

def _slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator) -> tuple:
    n = h.size
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr) * dx)
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    x0 = h + rng.normal(scale=0.01, size=n)
    x0 = _project_to_sum(x0, target)
    res = minimize(obj, x0, method="SLSQP", bounds=bounds,
                   constraints=[cons], options={"maxiter": 300, "ftol": 1e-10, "disp": False})
    if res.success:
        x = _project_to_sum(res.x, target)
        return x, _eval_c5(x)
    return h, _eval_c5(h)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # resolution – larger gives finer shapes, 180 is a good compromise
    n = 180
    target = n / 2.0
    dx = 2.0 / n

    # ---- gather seed vectors -------------------------------------------------
    seeds = []

    init = globals().get("initial_h_values")
    if init is not None:
        init = np.asarray(init, dtype=np.float64)
        if init.size != n:
            old_n = init.size
            old_pos = (np.arange(old_n) + 0.5) * (2.0 / old_n)
            new_pos = (np.arange(n) + 0.5) * dx
            init = np.interp(new_pos, old_pos, init)
        seeds.append(init)

    # uniform 0.5 (already satisfies sum)
    seeds.append(np.full(n, 0.5))

    # random binary half ones
    bin_vec = np.zeros(n)
    bin_vec[rng.choice(n, size=n // 2, replace=False)] = 1.0
    seeds.append(bin_vec)

    # smooth sinusoidal start
    pos = (np.arange(n) + 0.5) * dx
    sin_vec = (np.sin(np.pi * pos / 2.0) + 1.0) / 2.0   # in [0,1]
    sin_vec = _project_to_sum(sin_vec, target)
    seeds.append(sin_vec)

    # evaluate seeds
    best_h = None
    best_c5 = np.inf
    for s in seeds:
        s = _project_to_sum(s, target)
        c5 = _eval_c5(s)
        if c5 < best_c5:
            best_h, best_c5 = s, c5

    # ---- main stochastic search ---------------------------------------------
    while time.time() - start < budget_s - 20.0:
        mode = rng.integers(0, 5)
        if mode == 0:
            # jittered copy of one of the seed vectors
            base = rng.choice(seeds)
            cand = base + rng.normal(scale=0.05, size=n)
        elif mode == 1:
            # fresh uniform random
            cand = _random_feasible(n, target, rng)
        elif mode == 2:
            # fresh random binary half
            cand = np.zeros(n)
            cand[rng.choice(n, size=n // 2, replace=False)] = 1.0
        elif mode == 3 and best_h is not None:
            # perturb current best
            cand = best_h + rng.normal(scale=0.1, size=n)
        else:
            # random convex combo of two seeds
            a, b = rng.choice(seeds, size=2, replace=False)
            alpha = rng.random()
            cand = alpha * a + (1 - alpha) * b

        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_to_sum(cand, target)

        # hill climb
        remaining = budget_s - (time.time() - start)
        moves = 1500 if remaining > 60 else 500 if remaining > 30 else 200
        cand, cand_c5 = _hill_climb(cand, moves, rng)

        # occasional SLSQP refinement
        if remaining > 40 and rng.random() < 0.12:
            cand, cand_c5 = _slsqp_refine(cand, target, rng)

        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_c5

    # ---- intensive final hill climbing ---------------------------------------
    deadline = start + budget_s - 2.0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(best_h[i], 1.0 - best_h[j])
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        new = best_h.copy()
        new[i] -= delta
        new[j] += delta
        new_c5 = _eval_c5(new)
        if new_c5 + 1e-12 < best_c5:
            best_h, best_c5 = new, new_c5

    # ---- final SLSQP polishing ---------------------------------------------
    for _ in range(2):
        best_h, best_c5 = _slsqp_refine(best_h, target, rng)

    best_h = _project_to_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_eval_c5(best_h))

    return best_h, best_c5, n
