def random_feasible_vector(n, rng):
    target=n/2
    # start with zeros
    h=np.zeros(n)
    remaining=target
    remaining_idxs = list(range(n))
    # draw random numbers sequentially: allocate random amount up to min(1, remaining)
    while remaining_idxs:
        i = remaining_idxs.pop()
        max_val = min(1.0, remaining)
        if len(remaining_idxs)==0:
            # last index gets all remaining
            val = remaining
        else:
            val = rng.random()*max_val
        h[i] = val
        remaining -= val
    # Now random distribution but order may be biased; we can shuffle.
    rng.shuffle(h)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function (dx = 2 / len(h))."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return float((m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA)


def adjust_sum(vec: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass so that Σ vec = target while keeping 0 ≤ vec ≤ 1."""
    diff = target - vec.sum()
    n = vec.size
    # simple random redistribution
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - vec[i])
        else:
            add = max(diff, -vec[i])
        vec[i] += add
        diff = target - vec.sum()
    # very small residual – spread uniformly
    if abs(diff) > 1e-12:
        vec += diff / n
    vec = np.clip(vec, 0.0, 1.0)
    # final quick fix
    diff = target - vec.sum()
    for _ in range(10000):
        if abs(diff) < 1e-12:

        i = rng.integers(0, n)

            add = min(diff, 1.0 - vec[i])
        else:
            add = max(diff, -vec[i])
        vec[i] += add
        diff = target - vec.sum()
    return vec


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random vector with entries in [0,1] and sum = n/2."""
    target = n / 2.0
    # Dirichlet‑like start
    a = rng.random(n) + 1e-6
    h = a / a.sum() * target
    h = np.clip(h, 0.0, 1.0)
    return adjust_sum(h, target, rng)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.3):
    """Deterministic hill‑climb (accept only strict improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # decay step size periodically
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = adjust_sum(x, target, rng)
    return x
