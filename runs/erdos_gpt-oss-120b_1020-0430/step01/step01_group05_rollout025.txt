def shape_from_L(L, n):
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx/2.0
    # Ensure L between 0 and 1
    H = 1.0 / (2.0 - L)
    h = np.empty_like(xs)
    left_ramp = xs <= L
    right_ramp = xs >= (2.0 - L)
    middle = (~left_ramp) & (~right_ramp)
    # left ramp: linear from 0 to H
    # hx = H * xs / L
    # handle L=0 edge (then no ramp, shape constant H=0.5)
    if L > 1e-12:
        h[left_ramp] = (xs[left_ramp] / L) * H
    else:
        h[left_ramp] = H  # L=0, but left_ramp empty anyway
    # middle region
    h[middle] = H
    # right ramp: linear down from H to 0: h = H * (2 - xs)/L
    if L > 1e-12:
        h[right_ramp] = ( (2.0 - xs[right_ramp]) / L ) * H
    else:
        h[right_ramp] = H
    return h

import time, random, math
import numpy as np
from scipy.optimize import minimize


def _project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while fixing sum(x)=target (bisection on λ)."""
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretised."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _smooth_max(arr: np.ndarray, alpha: float) -> float:
    """log‑sum‑exp smooth max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def shape_from_L(L: float, n: int) -> np.ndarray:
    """Symmetric ramp‑plateau shape defined by parameter L∈[0,1].
    L = length of each ramp; height H = 1/(2‑L)."""
    if L <= 0.0:
        # uniform 0.5 (area already 1)
        return np.full(n, 0.5)
    H = 1.0 / (2.0 - L)
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx / 2.0
    h = np.empty_like(xs)

    left = xs <= L
    right = xs >= (2.0 - L)
    middle = (~left) & (~right)

    # left ramp linear from 0 to H
    h[left] = (xs[left] / L) * H
    # plateau
    h[middle] = H
    # right ramp linear down to 0
    h[right] = ((2.0 - xs[right]) / L) * H
    return h


def _hill_climb(initial_h: np.ndarray, target_sum: float, end_time: float):
    """Simple pairwise mass moves driven by the current worst shift."""
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    n = best_h.shape[0]

    while time.time() < end_time:
        # decide move type
        if random.random() < 0.75:
            # move mass along the shift that currently gives the worst overlap
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_max = np.argmax(corr) - (n - 1)
            if k_max >= 0:
                i = np.random.randint(0, n - k_max)
                j = i + k_max
            else:
                i = np.random.randint(-k_max, n)
                j = i + k_max
        else:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue

        # try moving a random amount from i to j (or opposite)
        if random.random() < 0.5:
            delta_max = min(best_h[i], 1.0 - best_h[j])
            if delta_max <= 0:
                continue
            delta = random.random() * delta_max
            cand = best_h.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            delta_max = min(best_h[j], 1.0 - best_h[i])
            if delta_max <= 0:
                continue
            delta = random.random() * delta_max
            cand = best_h.copy()
            cand[i] += delta
            cand[j] -= delta

        # re‑enforce feasibility (sum and bounds)
        cand = _project_onto_bounds_and_sum(cand, target_sum)

        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand
            if best_val <= 0.3808:
                break

    return best_h, best_val
