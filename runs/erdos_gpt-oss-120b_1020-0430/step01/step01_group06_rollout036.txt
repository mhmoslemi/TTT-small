import numpy as np, random, time, math

def project_capped_simplex(v, target, ub=1.0, tol=1e-9, max_iter=100):
    low = -np.max(v) - ub - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        h = np.clip(v - lam, 0.0, ub)
        cur = h.sum()
        if abs(cur - target) <= tol:
            return h
        if cur > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) / 2.0, 0.0, ub)

def compute_bound(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max() * (2.0 / n))

def gradient_from_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        g[:n-shift] += h[shift:]
        g[shift:] += h[:n-shift]
    else:
        s = -shift
        g[s:] += h[:n-s]
        g[:n-s] += h[s:]
    return g

def fine_tune(h, best_bound, target_sum, n, start_time, budget_s):
    # local random swaps with SA for remaining time
    T = 0.01
    decay = 0.9995
    it = 0
    cur_h = h.copy()
    # compute current bound
    C = np.correlate(cur_h, cur_h, mode='full')
    cur_bound = (target_sum - C.min()) * (2.0 / n)
    best_h_local = cur_h.copy()
    best_bound_local = cur_bound

    while time.time() - start_time < budget_s * 0.95:
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        max_up = min(1.0 - hi, hj)
        max_down = min(hi, 1.0 - hj)
        if max_up + max_down == 0:
            continue
        delta = (random.random() * (max_up + max_down)) - max_down
        nh = cur_h.copy()
        nh[i] = hi + delta
        nh[j] = hj - delta
        Cn = np.correlate(nh, nh, mode='full')
        new_bound = (target_sum - Cn.min()) * (2.0 / n)
        if new_bound < cur_bound or random.random() < math.exp(-(new_bound - cur_bound) / max(T, 1e-12)):
            cur_h = nh
            cur_bound = new_bound
            if new_bound < best_bound_local:
                best_bound_local = new_bound
                best_h_local = nh.copy()
        it += 1
        if it % 1000 == 0:
            T *= decay
    return best_h_local, best_bound_local

import numpy as np, random, time, math

def _project_capped_simplex(v, target, ub=1.0, tol=1e-12, max_iter=80):
    """
    Euclidean projection of v onto the set {h: 0<=h_i<=ub, sum(h)=target}.
    Uses the fact that the solution has the form h_i = clip(v_i - lam, 0, ub)
    for some lambda. Finds lambda by binary search.
    """
    lo = -np.max(v) - ub - 1.0          # guarantees all entries become ub
    hi = np.max(v) + 1.0                # guarantees all entries become 0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, ub)
        s = h.sum()
        if abs(s - target) <= tol:
            return h
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, ub)


def _bound_from_h(h):
    """C5 bound = max correlation * dx."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _soft_weights(C, beta):
    """Soft‑min weights for the vector C (larger weight for smaller C)."""
    # w_i ∝ exp(-beta * C_i)
    ex = -beta * C
    ex -= np.max(ex)            # shift for numerical stability
    w = np.exp(ex)
    return w / w.sum()


def _gradient_from_weights(h, weights):
    """∇ (∑ w_s C_s) where C_s = Σ h_i h_{i+s}.  Returns the positive direction."""
    n = h.shape[0]
    g = np.zeros_like(h)
    off = n - 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - off
        if shift >= 0:
            # i from 0 .. n-1-shift  -> h_i contributes via h_{i+shift}
            g[: n - shift] += w * h[shift:]
            # i from shift .. n-1  -> h_i contributes via h_{i-shift}
            g[shift:] += w * h[: n - shift]
        else:
            s = -shift
            g[s:] += w * h[: n - s]
            g[: n - s] += w * h[s:]
    return g


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # initialise a feasible vector
    # ------------------------------------------------------------------
    try:

    h = np.clip(h, 0.0, 1.0)
    h = _project_capped_simplex(h, target)

    best_h = h.copy()
    best_bound = _bound_from_h(h)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    beta = 1.0                       # start soft‑min temperature
    beta_max = 200.0
    step = 0.5                       # raw gradient step size
    step_min = 1e-4
    T = 2e-2                        # simulated‑annealing temperature
    decay_T = 0.9994
    decay_step = 0.9995
    beta_inc = 1.02
    it = 0

    while time.time() - start < budget_s * 0.95:
        # current autocorrelation
        C = np.correlate(h, h, mode="full")
        cur_bound = (target - C.min()) * (2.0 / n)   # same as _bound_from_h
        if cur_bound < best_bound:
            best_bound = cur_bound
            best_h = h.copy()

        # soft‑min weighting – focus on the smallest autocorrelations
        w = _soft_weights(C, beta)

        # gradient of weighted sum of autocorrelations
        g = _gradient_from_weights(h, w)

        # propose a move
        h_new = h + step * g
        h_new = _project_capped_simplex(h_new, target)

        # evaluate
        C_new = np.correlate(h_new, h_new, mode="full")
        new_bound = (target - C_new.min()) * (2.0 / n)

        if nb < cur_bound or random.random() < math.exp(-(nb - cur_bound) / max(T2, 1e-12)):
            cur_h = nh
            cur_bound = nb
        if i % 500 == 0:
            T2 *= 0.99

    # ensure feasibility of the best vector
    if abs(best_h.sum() - target) > 1e-9:
        best_h = _project_capped_simplex(best_h, target)

    final_bound = _bound_from_h(best_h)
    return (best_h.tolist(), final_bound, n)
