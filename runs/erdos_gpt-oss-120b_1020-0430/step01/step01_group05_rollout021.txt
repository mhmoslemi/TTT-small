# Helper functions

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the box [0,1]^n intersected with the hyperplane sum(x)=target.
    Uses bisection on the Lagrange multiplier.
    """
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape: peak 1 at centre, linear to 0 at edges."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2] – already has integral 1."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_cosine(n: int) -> np.ndarray:
    """Half‑period cosine, zero at the ends, peak 1 in the centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 0.5 * (1.0 + np.cos(np.pi * (pts - 1.0)))
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

    return project_onto_bounds_and_sum(h, n / 2.0)
