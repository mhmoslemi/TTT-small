import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core evaluation helpers
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth upper bound of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    # stable log‑sum‑exp
    return float(logsumexp(alpha * conv) / alpha)


# ------------------------------------------------------------
# Simple constructions
# ------------------------------------------------------------
def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2‑x) sampled on n points and normalised."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    # enforce exact integral
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def generate_monotonic(n: int) -> np.ndarray:
    """Linear decreasing shape h(x)=1‑x/2 on [0,2]; already has integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = 1.0 - x / 2.0
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h = h * (target / cur)
        h = np.clip(h, 0.0, 1.0)
    return h


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector with entries in [0,1] and exact sum n/2."""
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary (0/1) vector with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n, dtype=float)
    idx = rng.permutation(n)[:half]
    arr[idx] = 1.0
    return arr


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new resolution, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


# ------------------------------------------------------------
# Optimisation utilities
# ------------------------------------------------------------
def optimise_h(start: np.ndarray, n: int, deadline: float) -> tuple:
    """Two‑stage L‑BFGS‑B optimisation using a penalised softmax objective."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = start.copy()
    lam = 1.0
    for alpha in (5.0, 30.0, 100.0, 300.0):
        if time.time() > deadline:
            break
        lam *= 10.0  # strengthen sum‑penalty

        def obj(x):
            return softmax_c5(x, alpha) + lam * (x.sum() - target) ** 2

        res = minimize(
            obj,
            h,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 500, "ftol": 1e-12, "disp": False},
        )
        if not res.success:
            break
        h = np.clip(res.x, 0.0, 1.0)

        # cheap projection onto the exact hyperplane
        cur = h.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)

    return h, compute_c5(h)


def hill_climb(h: np.ndarray, n: int, deadline: float,
               rng: np.random.Generator, max_iter: int = 20000,
               delta: float = 0.12) -> tuple:
    """Simple swap‑move hill climbing respecting the sum constraint."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    target = n / 2.0

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move mass from j to i (increase i, decrease j)
        max_move = min(delta, 1.0 - cur_h[i], cur_h[j])
        if max_move <= 0:
            continue
        d = rng.random() * max_move
        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d

        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 - 1e-12:
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cur_h.copy(), cur_c5
                # occasionally tighten delta when we improve
                delta = max(delta * 0.99, 0.02)
    return best_h, best_c5
