import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto {0≤x≤1, sum(x)=target}."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple water‑filling style fix
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        ids = np.where(mask)[0]
        per = diff / len(ids)
        for i in ids:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h, c5, deadline, rng):
    """Random pairwise mass exchange hill climbing."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5

def _lp_initialize(n, rng):
    """
    Linear‑programming relaxation that maximises the minimum
    sum_i min(h_i , h_{i+k})  (a convex surrogate for the true objective).
    Returns a feasible h satisfying the sum constraint.
    """
    target = n / 2.0
    num_y = n * (n - 1) // 2          # pairs (i,i+k), k≥1
    total_vars = n + num_y + 1        # last variable is t
    t_idx = total_vars - 1

    # equality: sum h_i = target
    A_eq = sp.csr_matrix(
        (np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
        shape=(1, total_vars)
    )
    b_eq = np.array([target], dtype=float)

    data = []
    rows = []
    cols = []
    b_ub = []

    row_cnt = 0
    shift_y = [[] for _ in range(n)]   # y indices belonging to each shift k
    y_idx = n

    for k in range(1, n):
        for i in range(n - k):
            # y - h_i ≤ 0
            data.extend([1.0, -1.0])
            rows.extend([row_cnt, row_cnt])
            cols.extend([y_idx, i])
            b_ub.append(0.0)
            row_cnt += 1

            # y - h_{i+k} ≤ 0
            data.extend([1.0, -1.0])
            rows.extend([row_cnt, row_cnt])
            cols.extend([y_idx, i + k])
            b_ub.append(0.0)
            row_cnt += 1

            # -y + h_i + h_{i+k} - 1 ≤ 0   (McCormick lower bound)
            data.extend([-1.0, 1.0, 1.0])
            rows.extend([row_cnt, row_cnt, row_cnt])
            cols.extend([y_idx, i, i + k])
            b_ub.append(1.0)
            row_cnt += 1

            shift_y[k].append(y_idx)
            y_idx += 1

    # constraints:  sum_i y_{i,k} ≥ t   →   -sum y + t ≤ 0
    for k in range(1, n):
        for y_var in shift_y[k]:
            data.append(-1.0)
            rows.append(row_cnt)
            cols.append(y_var)
        data.append(1.0)          # coefficient of t
        rows.append(row_cnt)
        cols.append(t_idx)
        b_ub.append(0.0)
        row_cnt += 1

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(row_cnt, total_vars))
    b_ub = np.array(b_ub, dtype=float)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * num_y + [(0.0, target)]

    c = np.zeros(total_vars, dtype=float)
    c[t_idx] = -1.0                     # maximise t

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP initialization failed")
    h = res.x[:n]
    return h

def _smooth_weights(h, beta):
    """Return max correlation, weights (soft‑max) and full correlation vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    maxc = np.max(corr)
    # stabilise exponentials
    exp_term = np.exp(beta * (corr - maxc))
    w = exp_term / exp_term.sum()
    return maxc, w, corr

def _gradient_from_weights(h, w):
    """Gradient of the smooth‑max objective (weights w over shifts)."""
    n = h.size
    grad = np.zeros(n, dtype=float)
    # every shift contributes +ws to all entries
    grad += w.sum()                     # =1, but keep for clarity
    offset = n - 1
    for idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        s = idx - offset                # shift value (negative … positive)
        if s >= 0:
            # subtract ws * h[i+s] for i = 0 … n-1-s
            grad[: n - s] -= ws * h[s:]
            # subtract ws * h[i-s] for i = s … n-1
            grad[s:] -= ws * h[: n - s]
        else:  # s < 0
            t = -s
            # subtract ws * h[i-s] = h[i+t] for i = 0 … n-1-t
            grad[: n - t] -= ws * h[t:]
            # subtract ws * h[i+s] = h[i-t] for i = t … n-1
            grad[t:] -= ws * h[: n - t]
    return grad

def _smooth_proj_descent(h0, target, deadline):
    """
    Projected gradient descent on a smooth approximation of max overlap.
    beta grows during the run to focus the approximation on the true maximum.
    """
    h = h0.copy()
    best_h, best_c5 = h.copy(), _evaluate_c5(h)
    beta = 1.0
    step = 0.4
    it = 0
    while time.time() < deadline:
        it += 1
        # gradually tighten the approximation
        if it % 200 == 0:
            beta = min(beta * 1.15, 30.0)
            step = max(step * 0.95, 0.02)

        _, w, _ = _smooth_weights(h, beta)
        grad = _gradient_from_weights(h, w)

        h_new = h - step * grad
        h_new = _adjust_sum(np.clip(h_new, 0.0, 1.0), target)
        new_c5 = _evaluate_c5(h_new)

        # simple acceptance rule
        if new_c5 < best_c5:
            best_h, best_c5 = h_new.copy(), new_c5
            h = h_new
        else:
            # occasional stochastic move to escape flat regions
            if np.random.rand() < 0.05:
                h = h_new
            else:
                # keep current point
                pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest resolution that balances quality and speed.
    n = 80                     # even → sum constraint is integral
    target = n / 2.0

    # Try to initialise from the LP relaxation; fall back to random if it fails.
    try:
        h0 = _lp_initialize(n, rng)
    except Exception:
        h0 = rng.random(n)
        h0 = h0 * (target / h0.sum())
        h0 = _adjust_sum(h0, target)

    h0 = _adjust_sum(np.clip(h0, 0.0, 1.0), target)
    best_h, best_c5 = h0, _evaluate_c5(h0)

    # --- Smooth projected gradient descent ---------------------------------
    deadline = start + budget_s - 2.0
    h_opt, c5_opt = _smooth_proj_descent(best_h, target, deadline)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    # --- Final fine‑grained hill climbing ----------------------------------
    deadline = start + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, deadline, rng)

    # Ensure feasibility one last time
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))

    return best_h, best_c5, n
