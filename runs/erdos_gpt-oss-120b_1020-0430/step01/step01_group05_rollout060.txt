    n = len(a)
    n2 = 1 << (2*n-1).bit_length()  # next power of 2.
    fa = np.fft.rfft(np.concatenate([a, np.zeros(n2-n)]))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    N = 1 << ((2*n-1).bit_length())
    a = np.fft.rfft(h, N)
    b = np.fft.rfft(1.0 - h, N)
    c = np.fft.irfft(a * np.conj(b), N)
    corr = c[:2*n-1]
    return np.max(corr) * dx

    cand = project_onto_bounds_and_sum(cand, target_sum)

```python
import time, random, math, numpy as np
from scipy.optimize import minimize

def project_onto_bounds_and_sum(x, target):
    # bisection on lambda
    low, high = -1.0, 1.0
    # ensure that low: sum(proj) > target; high: sum(proj) < target
    # We'll just loop a number of times.
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    return np.clip(x - lam, 0.0, 1.0)
```

Now correlation function:

```python
def correlation_h_1mh(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())
    a = np.fft.rfft(h, N)
    b = np.fft.rfft(1.0 - h, N)
    c = np.fft.irfft(a * np.conj(b), N)
    # linear correlation length 2n-1
    return c[:2*n-1]
```

```python
def compute_c5(h):
    n = h.shape[0]
```

```python
def smooth_max(arr, alpha):
    m = np.max(arr)
    # shift for numerical stability
    shifted = arr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
```

import time, random, math
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while enforcing sum(x)=target by bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def correlation_h_1mh(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (full length 2n-1)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    a = np.fft.rfft(h, N)
    b = np.fft.rfft(1.0 - h, N)
    c = np.fft.irfft(a * np.conj(b), N)
    return c[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k)) dx, discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = correlation_h_1mh(h)
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth objective used by SLSQP (smooth max of the correlation)."""
    corr = correlation_h_1mh(h)
    return smooth_max(corr, alpha) * dx


def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing vector to new length n."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, n / 2.0)


# ---------- deterministic seeds ----------
def make_triangle(n: int) -> np.ndarray:
    """Triangular hump, peak 1 at centre, linear down to 0 at edges."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2]."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * xs / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_cos_edge(n: int) -> np.ndarray:
    """High at the edges, low in the centre: h(x)=0.5+0.5·cos(πx)."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = 0.5 + 0.5 * np.cos(np.pi * xs)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_plateau(n: int) -> np.ndarray:
    """Flat block of height 1 on [0.5,1.5], zero elsewhere."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.where((xs >= 0.5) & (xs <= 1.5), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)
