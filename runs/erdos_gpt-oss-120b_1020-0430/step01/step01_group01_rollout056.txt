        if len(vec) == n:

        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    if base_h is not None:
        h0 = upsample(base_h)
        h0 = project_to_feasible(h0, target_sum)
        best_h = h0.copy()
        best_val = _compute_c5(best_h, dx)

    # Parameter schedule
    alphas = [5, 10, 20, 40, 80]
    iters_per_alpha = 300   # can adjust based on remaining time
    step0 = 0.8
    decay = 0.995

    # Main loop with restarts until time

        # start from best with perturbation or random
        if rng.rand() < 0.4:

        else:
            h = _random_feasible(rng, n, target_sum)
        # Project feasible
        h = project_to_feasible(h, target_sum)

        # Continuation gradient descent
        for alpha in alphas:
            step = step0
            for _ in range(iters_per_alpha):
                # compute gradient
                grad = _smooth_max_grad(h, dx, alpha)
                h = h - step * grad
                h = project_to_feasible(h, target_sum)

        # Evaluate exact C5

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# --------------------------------------------------------------
# Random feasible point (uniform then project)
# --------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# --------------------------------------------------------------
# Exact C5 evaluation (max correlation)
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# --------------------------------------------------------------
# Gradient of the smooth‑max approximation of the C5 objective
# --------------------------------------------------------------
def _smooth_max_grad(h: np.ndarray, dx: float, alpha: float) -> np.ndarray:
    n = h.shape[0]
    # correlation of h with itself
    corr_hh = np.correlate(h, h, mode="full")
    # f_i = (sum(h) - corr_hh_i) * dx  ; sum(h)=target_sum = n/2
    f = 1.0 - dx * corr_hh                     # because sum(h)*dx = 1
    # softmax weights
    max_f = np.max(f)
    exps = np.exp(alpha * (f - max_f))
    w = exps / exps.sum()
    # gradient = dx - dx * sum_i w_i * (h_{k+s}+h_{k-s})
    grad = np.full_like(h, dx, dtype=np.float64)
    # iterate over shifts
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        s = idx - (n - 1)          # shift corresponding to this entry
        if s >= 0:
            # term h_{k+s}
            if s < n:
                grad[: n - s] -= dx * wi * h[s:]
            # term h_{k-s}
            if s > 0:
                grad[s:] -= dx * wi * h[: n - s]
        else:
            t = -s
            # term h_{k-s} becomes h_{k+t}
            if t < n:
                grad[: n - t] -= dx * wi * h[t:]
            # term h_{k+s} becomes h_{k-t}
            if t > 0:
                grad[t:] -= dx * wi * h[: n - t]
    return grad


# --------------------------------------------------------------
# Gradient of a single worst shift (sub‑gradient of the exact max)
# --------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

# Main driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h on [0,2] that (hopefully) improves the Erdős C₅ bound."""
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ---- base solution ------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    n0 = len(base_h) if base_h is not None else 0
    n = max(400, n0)                     # a modest but expressive grid
    dx = 2.0 / n
    target_sum = n / 2.0                 # sum of h values (integral = 1)

    # up‑sample if needed
    def _upsample(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    if base_h is not None:
        h0 = _upsample(base_h)
        h0 = _project_onto_capped_simplex(h0, target_sum)
        best_h = h0.copy()
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = _random_feasible(rng, n, target_sum)
        best_val = _compute_c5(best_h, dx)

    # ---- optimisation hyper‑parameters --------------------------------
    alphas = [5, 10, 20, 40, 80]          # continuation schedule
    iters_per_alpha = 300                # iterations per alpha stage
    step0 = 0.6
    decay = 0.995

    # --------------------------------------------------------------
    # Main optimisation loop (random restarts)
    # --------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.88:
        # initialise a fresh candidate
        if rng.rand() < 0.35:
            cand = best_h + rng.normal(scale=0.015, size=n)
        else:
            cand = _random_feasible(rng, n, target_sum)

        cand = _project_onto_capped_simplex(cand, target_sum)

        # continuation gradient descent on smoothed max
        for alpha in alphas:
            step = step0
            for _ in range(iters_per_alpha):
                grad = _smooth_max_grad(cand, dx, alpha)
                cand -= step * grad
                cand = _project_onto_capped_simplex(cand, target_sum)
                step *= decay

        # short sub‑gradient polishing on the exact max
        for _ in range(120):

            shift = int(np.argmax(corr)) - (n - 1)
            g = _gradient_for_shift(cand, shift)
            cand -= 0.07 * g
            cand = _project_onto_capped_simplex(cand, target_sum)

        val = _compute_c5(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

        # early exit if we are already very good

            break

    # final safety projection & evaluation
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
