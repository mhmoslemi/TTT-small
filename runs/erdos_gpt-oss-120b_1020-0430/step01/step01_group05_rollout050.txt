    # h: np.ndarray length n
    L = np.correlate(h, 1-h, mode='full')  # length 2n-1
    L_max = np.max(L)

    sum_exp = np.sum(expL)

    # weights
    w = expL / sum_exp  # shape (2n-1,)
    # gradient: g = 1 - (forward+backward)

    backward = backward_full[n-1:n-1+len(h)]
    grad = (1.0 - (forward + backward)) * dx
    obj = smooth * dx
    return obj, grad

def smooth_obj_and_grad(h, dx, alpha):
    L = np.correlate(h, 1.0 - h, mode='full')
    L_max = L.max()
    # shift L for stable exp
    L_shift = L - L_max
    expL = np.exp(alpha * L_shift)
    sum_exp = expL.sum()
    smooth = L_max + np.log(sum_exp) / alpha
    # gradient weights
    w = expL / sum_exp
    # forward sum:
    forward_full = np.correlate(w, h, mode='full')
    start = len(w) // 2  # which is (2n-1)//2 = n-1
    forward = forward_full[start:start+len(h)]
    # backward sum:
    w_rev = w[::-1]
    backward_full = np.correlate(w_rev, h, mode='full')
    backward = backward_full[start:start+len(h)]
    grad = (1.0 - (forward + backward)) * dx
    obj = smooth * dx
    return obj, grad

def optimize_seed(seed_h, target_sum, dx, bounds, time_limit):
    cur = seed_h.copy()
    # phases
    alphas = [30.0, 200.0, 2000.0]
    max_iter_per_alpha = 150
    init_step = 0.5
    for alpha in alphas:

        cur = gradient_descent_phase(cur, target_sum, dx, alpha, max_iter_per_alpha, init_step, time_limit)
    # SLSQP refinement
    if time.time() < time_limit:
        def obj_func(x):
            return smooth_obj_and_grad(x, dx, 2000.0)[0]
        def jac_func(x):
            return smooth_obj_and_grad(x, dx, 2000.0)[1]
        try:
            res = minimize(obj_func, cur, jac=jac_func, method='SLSQP', bounds=bounds,
                           constraints={'type':'eq','fun':lambda x: np.sum(x)-target_sum},
                           options={'maxiter':200, 'ftol':1e-12, 'disp':False})
            cur = project_onto_bounds_and_sum(res.x, target_sum)
        except Exception:
            pass
    # Local random pairwise refinement
    # Use while loop with time check
    while time.time() < time_limit - 0.1:
        i, j = np.random.randint(0, len(cur), size=2)
        if i==j: continue
        hi, hj = cur[i], cur[j]
        # keep sum constant while staying in [0,1]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = np.clip(cand, 0.0, 1.0)

        cval = compute_c5(cand)
        best_cand_val = compute_c5(cur)
        if cval < best_cand_val:
            cur = cand
    return cur

import time, random, math
import numpy as np
from scipy.optimize import minimize

# global for the equality constraint
TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect {sum = target} using a bisection on λ."""
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_obj_and_grad(h: np.ndarray, dx: float, alpha: float):
    """Smooth max (log‑sum‑exp) of the overlap integral and its gradient."""
    # linear values L_k = ∫ h(x)(1‑h(x+k)) dx   (without the dx factor)
    L = np.correlate(h, 1.0 - h, mode="full")
    L_max = L.max()
    L_shift = L - L_max
    expL = np.exp(alpha * L_shift)
    sum_exp = expL.sum()
    smooth = L_max + np.log(sum_exp) / alpha

    # weights of the soft‑max
    w = expL / sum_exp                     # length = 2n-1
    n = h.shape[0]
    start = n - 1                          # index where shift = 0 appears
    # forward term Σ w_s h_{i+s}
    forward_full = np.correlate(w, h, mode="full")
    forward = forward_full[start:start + n]
    # backward term Σ w_s h_{i-s}
    w_rev = w[::-1]
    backward_full = np.correlate(w_rev, h, mode="full")
    backward = backward_full[start:start + n]

    grad = (1.0 - (forward + backward)) * dx
    obj = smooth * dx
    return obj, grad


def objective_smooth(h, dx, alpha):
    obj, _ = smooth_obj_and_grad(h, dx, alpha)
    return obj


def grad_smooth(h, dx, alpha):
    _, g = smooth_obj_and_grad(h, dx, alpha)
    return g


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σ h_i = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def upsample_initial(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an old vector to new_n points, then project."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, new_n / 2.0)


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.where(np.abs(pts - 1.0) <= 0.5, 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def gradient_descent_phase(h, target, dx, alpha, max_iter, init_step, deadline):
    cur = h.copy()
    step = init_step
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        obj, grad = smooth_obj_and_grad(cur, dx, alpha)
        # simple backtracking line search
        for _ in range(5):
            cand = project_onto_bounds_and_sum(cur - step * grad, target)
            cand_obj, _ = smooth_obj_and_grad(cand, dx, alpha)
            if cand_obj <= obj:
                cur = cand
                break
            step *= 0.5
        else:
            step *= 0.5
        step = min(step * 1.02, 3.0)   # avoid blowing up
    return cur


def optimize_from(seed_h, target, dx, bounds, deadline):
    cur = seed_h.copy()
    # continuation on smoothing parameter
    for alpha, iters, step in [(30.0, 150, 1.0),
                               (200.0, 150, 1.0),
                               (2000.0, 100, 1.0)]:
        if time.time() >= deadline:
            break
        cur = gradient_descent_phase(cur, target, dx, alpha, iters, step, deadline)

    # final SLSQP refinement on the hardest smooth objective
    if time.time() < deadline:
        try:
            res = minimize(objective_smooth, cur,
                           args=(dx, 2000.0),
                           jac=grad_smooth,
                           method="SLSQP",
                           bounds=bounds,
                           constraints={"type": "eq", "fun": sum_eq_constraint},
                           options={"maxiter": 300, "ftol": 1e-12, "disp": False})
            cur = project_onto_bounds_and_sum(res.x, target)
        except Exception:
            pass

    # cheap random pairwise tweaks
    n = cur.shape[0]
    while time.time() < deadline - 0.1:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = cur.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target)
        if compute_c5(cand) < compute_c5(cur):
            cur = cand
    return cur


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h with a low Erdős C₅ constant."""
    global TARGET_SUM_GLOBAL
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # problem size – a power of two makes FFT‑friendly, but we stay with O(n²) ops
    n = 512
    dx = 2.0 / n
    target_sum = n / 2.0
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ---- seeds ---------------------------------------------------------
    seed_vectors = []

    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))
    seed_vectors.append(make_rectangle(n))

    best_h = None
    best_val = np.inf

    # ---- main optimisation loop ----------------------------------------
    idx = 0
    while time.time() < deadline:
        if idx < len(seed_vectors):
            h0 = seed_vectors[idx]
        else:
            # create a random feasible vector
            rnd = np.random.rand(n)
            h0 = project_onto_bounds_and_sum(rnd, target_sum)
        idx += 1

        h_opt = optimize_from(h0, target_sum, dx, bounds, deadline)
        val = compute_c5(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            # early exit if record is beaten
            if best_val <= 0.3808:
                break

    # fallback if nothing succeeded
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
