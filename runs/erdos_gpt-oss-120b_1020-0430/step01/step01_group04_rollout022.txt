def hill_climb(initial_h, time_deadline):
    h = initial_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    # parameters
    base_step = 0.2
    temp0 = 0.01
    start_time = time.time()
    while time.time() < time_deadline:
        # annealing schedule based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / (time_deadline - start_time)
        step = base_step * (1 - frac) + 1e-5
        temp = temp0 * (1 - frac) + 1e-6
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        hi = h[i]; hj = h[j]
        lower = max(-hi, hj - 1, -step)
        upper = min(1 - hi, hj, step)
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        # apply
        h[i] = hi + delta
        h[j] = hj - delta
        new_val = compute_c5(h)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / temp):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_val

def hill_climb(start_h, deadline, rng):
    h = start_h.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_val = cur
    n = len(h)
    base_step = 0.3 # dynamic
    temp0 = 0.01
    start = time.time()
    while time.time() < deadline:
        # compute schedule
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total>0 else 1.0
        step = base_step * (1 - frac) + 1e-6
        temp = temp0 * (1 - frac) + 1e-9
        # choose indices
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        hi = h[i]; hj = h[j]
        lower = max(-hi, hj - 1, -step)
        upper = min(1 - hi, hj, step)
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        h[i] = hi + delta
        h[j] = hj - delta
        new = compute_c5(h)
        if new < cur or rng.random() < np.exp(-(new - cur) / temp):
            cur = new
            if new < best_val:
                best_val = new
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_val

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ------------------------------------------------------------
# Basic evaluation
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 using FFT‑based correlation (fast for many calls)."""
    n = h.shape[0]
    L = 2 * n - 1
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(H * R, n=L)
    return float(corr.max() * (2.0 / n))


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max via log‑sum‑exp (used by SLSQP)."""
    n = h.shape[0]
    L = 2 * n - 1
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)
    return float(logsumexp(alpha * conv) / alpha * (2.0 / n))

def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangle reaching height 1 (area exactly 1)."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, keeping the integral exact."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random feasible vector (sum = n/2, values in [0,1])."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def generate_piecewise_linear(n: int, a: float) -> np.ndarray:
    """
    Symmetric piecewise‑linear shape:
        - linear rise from 0 to height t over [0, a]
        - flat plateau of height t on [a, 2‑a]
        - symmetric descent.
    Parameter a ∈ [0,1]; the height t = 1/(2‑a) guarantees integral 1.
    """
    t = 1.0 / (2.0 - a)          # plateau height
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = np.empty_like(x)
    if a <= 0.0:
        h.fill(t)
    else:
        left = x <= a
        right = x >= (2.0 - a)
        middle = ~left & ~right
        h[left] = (t / a) * x[left]
        h[middle] = t
        h[right] = (t / a) * (2.0 - x[right])
    h = np.clip(h, 0.0, 1.0)

    # enforce exact integral (tiny correction)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb(start_h: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Simple simulated‑annealing style hill‑climber.
    Modifies two entries at a time while preserving the total sum.
    """
    h = start_h.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_val = cur
    n = h.shape[0]

    base_step = 0.25
    temp0 = 0.012
    start = time.time()
    while time.time() < deadline:
        # schedule
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        step = base_step * (1.0 - frac) + 1e-6
        temp = temp0 * (1.0 - frac) + 1e-9

        # pick two different indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        hi, hj = h[i], h[j]
        # feasible delta range respecting bounds and step size
        low = max(-hi, hj - 1.0, -step)
        high = min(1.0 - hi, hj, step)
        if low >= high:
            continue
        delta = rng.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta
        new = compute_c5(h)

        if new < cur or rng.random() < np.exp(-(new - cur) / temp):
            cur = new
            if new < best_val:
                best_val = new
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

    return best_h, best_val


def refine_with_slsqp(start_h: np.ndarray, deadline: float):
    """Two‑stage SLSQP using the smooth max approximation."""
    n = start_h.shape[0]
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

    opts = {"maxiter": 400, "ftol": 1e-12, "disp": False}

    res1 = opt.minimize(lambda x: softmax_c5(x, alpha=10.0), start_h,
                        method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    if not res1.success or time.time() >= deadline:
        h_opt = np.clip(res1.x, 0.0, 1.0)
        return h_opt, compute_c5(h_opt)

    res2 = opt.minimize(lambda x: softmax_c5(x, alpha=80.0), res1.x,
                        method="SLSQP", bounds=bounds, constraints=cons, options=opts)
    h_opt = np.clip(res2.x if res2.success else res1.x, 0.0, 1.0)

    # final exact sum correction (tiny)
    diff = target - h_opt.sum()
    if abs(diff) > 1e-12:
        h_opt += diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)

    return h_opt, compute_c5(h_opt)
