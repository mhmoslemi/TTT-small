def improve_h(h, time_end, target_val=0.3808):
    best_h = h.copy()
    best_val = compute_c5_fft(best_h)
    n = h.shape[0]
    while time.time() < time_end:
        # compute correlation and max shift
        corr = compute_corr_fft(best_h) # get linear correlation
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * (2.0/n)
        if max_val < best_val:
            best_val = max_val
            # best_h remains same
        if best_val <= target_val:
            break
        # shift k calculation
        k = max_idx - (n - 1)  # shift can be negative
        # compute product contributions p_i
        # valid i range: i from max(0,-k) to min(n-1, n-1 - k)
        i_start = max(0, -k)
        i_end = min(n, n - k)  # exclusive
        i_range = np.arange(i_start, i_end)
        # compute contributions
        p = best_h[i_range] * (1 - best_h[i_range + k])
        # if sum(p)==0 skip
        total_p = p.sum()
        if total_p == 0:
            # random pair
            i = random.randint(0, n-1)
            j = random.randint(0, n-1)
            # ensure i != j and can transfer
        else:
            # sample i weighted by p
            # Use random choice: compute cumulative distribution
            r = random.random() * total_p
            cum = np.cumsum(p)
            idx = np.searchsorted(cum, r)
            i = i_range[idx]
        # target j = i + k (if within bounds else choose random)
        j = i + k
        if j < 0 or j >= n:
            # fallback random j
            j = random.randint(0, n-1)
        if i == j:
            continue
        # compute max transferable delta
        delta_max = min(best_h[i], 1.0 - best_h[j])
        # also limit step size to some epsilon (like 0.02)
        delta_max = min(delta_max, 0.02) # choices
        if delta_max <= 1e-9:
            continue
        delta = random.random() * delta_max
        # apply move
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute c5
        cand_val = compute_c5_fft(cand)
        # accept if improvement (or maybe accept with some probability)
        if cand_val < best_val - 1e-9:  # improvement
            best_h = cand
            best_val = cand_val
        # else maybe accept occasionally? but skip.
    return best_h, best_val

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())  # next power of 2
    # compute FFT
    H = np.fft.rfft(h, n=L)
    rev = (1 - h)[::-1]
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[:2*n-1]  # real part
    return conv

import time, random, math

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Projects x onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1−h) using FFT (same as np.correlate with mode='full')."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, n=L)
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[: 2 * n - 1]
    return conv


def compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 for a discretised h (max_k ∫ h(x)(1‑h(x+k))dx)."""
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)


def make_triangle(n: int) -> np.ndarray:
    """Triangular bump with peak 1 at x=1; integral equals 1."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2]; integral equals 1."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)
