def binary_from_initial():
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        # Interpolate to n
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        # uniform 0.5 start -> random binary
        h = np.full(n, 0.5)
    
    # Now convert to binary: set to 1 if h > 0.5 else 0
    binary = (h > 0.5).astype(np.float64)
    # adjust to have exactly n/2 ones:
    ones = int(binary.sum())
    target = n // 2
    diff = ones - target
    if diff > 0:
        # too many ones: set diff of ones to 0 (choose random)
        idx = np.where(binary == 1)[0]
        rng.shuffle(idx)
        binary[idx[:diff]] = 0
    elif diff < 0:
        # need extra ones
        idx = np.where(binary == 0)[0]
        rng.shuffle(idx)
        binary[idx[: -diff]] = 1
    return binary
```

def anneal_binary(start_h, time_limit, rng, T0=0.04, T_end=1e-6):

def anneal_continuous(start_h, time_limit, rng, T0=0.04, T_end=1e-6, max_step=0.2):
    h = start_h.copy()
    cur_c = _c5_metric(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (T_end / T0) ** progress
        # select two distinct indices
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # clamp to max_step
        low_cap = max(low, -max_step)
        high_cap = min(high, max_step)
        if low_cap < high_cap:
            delta = rng.uniform(low_cap, high_cap)
        else:
            delta = rng.uniform(low, high)
        # apply
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        d = new_c - cur_c
        if d < 0 or rng.random() < math.exp(-d / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_c

import time, math
import numpy as np

def _c5_metric(h):
    """Discrete C5 metric (max overlap) for a step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_vector(n, rng):
    """Create a feasible start on a grid of size n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in rng.permutation(n):
            if 0.0 <= h[idx] + deficit <= 1.0:
                h[idx] += deficit
                break
    return h

def _random_binary(n, rng):
    """Random binary vector with exactly n/2 ones (fractional if n odd)."""
    h = np.zeros(n, dtype=np.float64)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    if n % 2:
        # one fractional entry to achieve exact sum
        extra = rng.choice(np.where(h == 0)[0])
        h[extra] = 0.5
    return h

def _anneal_binary(start_h, time_limit, rng, T0=0.04, T_end=1e-6):
    """Simulated‑annealing with swaps between 0/1 entries."""
    h = start_h.copy()
    cur_c = _c5_metric(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = elapsed / time_limit
        T = T0 * (T_end / T0) ** prog

        i, j = rng.integers(n, size=2)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = _c5_metric(h)
        d = new_c - cur_c
        if d < 0 or rng.random() < math.exp(-d / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def _anneal_continuous(start_h, time_limit, rng,
                      T0=0.04, T_end=1e-6, max_step=0.2):
    """Simulated‑annealing with fractional mass transfers."""
    h = start_h.copy()
    cur_c = _c5_metric(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = elapsed / time_limit
        T = T0 * (T_end / T0) ** prog

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        low_cap = max(low, -max_step)
        high_cap = min(high, max_step)
        if low_cap < high_cap:
            delta = rng.uniform(low_cap, high_cap)
        else:
            delta = rng.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric(h)
        d = new_c - cur_c
        if d < 0 or rng.random() < math.exp(-d / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    n = 300

    best_c = float('inf')

    time_per_start = binary_budget / len(starts)

        elapsed = time.time() - start_all
        if elapsed >= total_budget - 5:

        cand_h, cand_c = _anneal_binary(vec, slice_time, rng)
        if cand_c < best_c:

                                            max_step=0.15)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()

    # final tiny‑step polishing with a very low temperature
    remaining = total_budget - (time.time() - start_all)
    if remaining > 2 and best_h is not None:
        cand_h, cand_c = _anneal_continuous(best_h, remaining, rng,
                                            T0=1e-6, T_end=1e-9, max_step=0.05)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()

    # enforce the exact integral constraint
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in rng.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # final evaluation (the helper may be more accurate)
    try:
        final_c = evaluate_erdos_solution(best_h)  # provided by the harness
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
