def improve_by_worst_shift(h, current_c5, rng):
    n = len(h)
    s, max_corr, corr = worst_shift_info(h)
    # get overlapping indices
    if s >= 0:
        src_range = np.arange(0, n - s)
        dst_range = src_range + s
    else:
        s_abs = -s
        src_range = np.arange(s_abs, n)
        dst_range = src_range - s_abs
    # identify eligible pairs
    eps = 1e-9
    src_vals = h[src_range]
    dst_vals = h[dst_range]
    eligible = np.where((src_vals > eps) & (dst_vals < 1 - eps))[0]
    if eligible.size == 0:
        return h, current_c5, False
    # try random eligible pair
    idx = rng.choice(eligible)
    src = src_range[idx]
    dst = dst_range[idx]
    max_move = min(h[src], 1 - h[dst])
    # try moving fraction
    delta = max_move * rng.random()
    if delta < 1e-12:
        return h, current_c5, False
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    new_c5 = evaluate_c5(h_new)
    if new_c5 + 1e-12 < current_c5:
        return h_new, new_c5, True
    else:
        # try half delta maybe?
        delta2 = delta * 0.5
        h_new = h.copy()
        h_new[src] -= delta2
        h_new[dst] += delta2
        new_c5 = evaluate_c5(h_new)
        if new_c5 + 1e-12 < current_c5:
            return h_new, new_c5, True
        else:
            return h, current_c5, False

import time
import numpy as np
from scipy.optimize import minimize

_eq_target = None  # will be set inside run


def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyperplane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple redistribution
    while abs(diff) > 1e-12:
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / idx.size
        for i in idx:
            if diff > 0.0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _eq_sum_constraint(x: np.ndarray) -> float:
    return x.sum() - _eq_target  # uses global _eq_target set in run()


def _worst_shift_info(h: np.ndarray):
    """Return (shift, max_corr, full_corr_array)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)  # negative for left shift, positive for right shift
    return shift, corr[idx], corr


def _guided_move(h: np.ndarray, cur_c5: float, rng: np.random.Generator):
    """Try to reduce the current worst shift by moving mass from a source to a destination."""
    n = h.size
    shift, _, _ = _worst_shift_info(h)
    if shift >= 0:
        src = np.arange(0, n - shift)
        dst = src + shift
    else:
        s = -shift
        src = np.arange(s, n)
        dst = src - s
    src_vals = h[src]
    dst_vals = h[dst]
    eps = 1e-12
    eligible = np.where((src_vals > eps) & (dst_vals < 1.0 - eps))[0]
    if eligible.size == 0:
        return h, cur_c5, False
    i = rng.choice(eligible)
    i_src = src[i]
    i_dst = dst[i]
    max_move = min(h[i_src], 1.0 - h[i_dst])
    if max_move < eps:
        return h, cur_c5, False
    delta = max_move * rng.random()
    h_new = h.copy()
    h_new[i_src] -= delta
    h_new[i_dst] += delta
    new_c5 = _evaluate_c5(h_new)
    if new_c5 < cur_c5 - 1e-12:
        return h_new, new_c5, True
    return h, cur_c5, False


def _random_swap_move(h: np.ndarray, cur_c5: float, rng: np.random.Generator):
    """Random exchange of a small amount of mass between a 1‑ish and a 0‑ish entry."""
    n = h.size
    # pick two distinct indices
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    if h[i] == h[j]:
        return h, cur_c5, False
    # define direction: move from larger to smaller
    if h[i] > h[j]:
        src, dst = i, j
    else:
        src, dst = j, i
    max_move = min(h[src], 1.0 - h[dst])
    if max_move < 1e-12:
        return h, cur_c5, False
    delta = max_move * rng.random()
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    new_c5 = _evaluate_c5(h_new)
    if new_c5 < cur_c5 - 1e-12:
        return h_new, new_c5, True
    return h, cur_c5, False


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise

    init_vec = globals().get("initial_h_values")

        n = 45

    target = n / 2.0
    global _eq_target
    _eq_target = target

    deadline_greedy = start + budget_s * 0.55

        n_candidate = cur_h.size
        i = rng.integers(n_candidate)

        if max_move < 1e-12:

        new_c5 = _evaluate_c5(h_new)

        if new_c5 < cur_c5 - 1e-12:

    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
