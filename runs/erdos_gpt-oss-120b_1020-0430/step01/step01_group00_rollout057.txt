def max_overlap_and_gradient(h):
    n = len(h)
    O = np.correlate(h, 1.0 - h, mode='full')
    # O shape 2n-1
    idx = np.argmax(O)
    shift = idx - (n - 1)  # shift s such that O[i] = sum_{j} h_j (1 - h_{j+s}); note s may be negative.
    # compute gradient
    k = abs(shift)
    # Overlap region length L = n - k
    # Determine direction.
    if shift >= 0:
        # positive shift (right)
        mask = np.arange(n) <= n - 1 - k
        h_plus = np.concatenate([h[k:], np.zeros(k)])
        h_minus = np.concatenate([np.zeros(k), h[:n - k]])
        g = mask.astype(float) - h_plus - h_minus
    else:
        # negative shift (left)
        mask = np.arange(n) >= k
        # For negative shift, defining overlay region: i from k..n-1
        # h_minus: h_{i - k} term: which becomes h[i - k]; for i >= k, that's h[:n - k] shifted.
        # map to array of size n: at index i, want h[i - k] if i >= k else 0.
        h_minus = np.concatenate([np.zeros(k), h[:n - k]])  # same as before
        # h_plus term: -h_{i + k} for i where i + k < n: i <= n - 1 - k. We need to align accordingly.
        # So create array where at index i, value is h_{i + k} if i + k < n else 0.
        h_plus = np.concatenate([h[k:], np.zeros(k)])
        g = mask.astype(float) - h_plus - h_minus
    # The gradient sign: Do we need negative gradient to descent? 
    # Let's test: O_s = sum_i h_i (1 - h_{i+s}) (for overlapping region). Increasing h_i would increase O if (1 - h_{i+s}) positive.
    # The derivative dO/dh_i = 1 - h_{i+s} (positive if h_{i+s}<1) - maybe also -h_{i-s}. So gradient g is derivative.
    # To minimize O, we need to move opposite direction of gradient: new_h = h - lr * g.
    return O[idx], shift, g

import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 using a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a given discretised step function."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ------------------------------------------------------------
# Gradient‑based minimisation of the max overlap
# ------------------------------------------------------------
def _max_overlap_and_gradient(h: np.ndarray):
    """Return max overlap value, the shift attaining it, and the sub‑gradient."""
    n = h.size
    # full correlation gives O_s for all integer shifts
    O = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(O)
    shift = idx - (n - 1)          # shift ∈ [-(n-1), n-1]

    k = abs(shift)
    # masks for the linear part of the derivative
    if shift >= 0:
        mask = np.arange(n) <= n - 1 - k
    else:
        mask = np.arange(n) >= k

    # shifted copies (zero‑padded) for the product terms
    h_plus = np.concatenate([h[k:], np.zeros(k)])   # h_{i+k} (or h_{i-k} for negative)
    h_minus = np.concatenate([np.zeros(k), h[:n - k]])  # h_{i-k} (or h_{i+k} for negative)

    # derivative of O_shift w.r.t. h
    grad = mask.astype(float) - h_plus - h_minus
    return O[idx], shift, grad


def _gradient_descent(start_h: np.ndarray,
                      max_iter: int = 300,
                      lr0: float = 0.7,
                      lr_decay: float = 0.99):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = start_h.size
    h = _project_to_feasible(start_h, n)
    best_h = h.copy()
    best_val = _c5_from_h(h)
    lr = lr0

    for _ in range(max_iter):
        _, _, grad = _max_overlap_and_gradient(h)
        h = h - lr * grad
        h = _project_to_feasible(h, n)
        val = _c5_from_h(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        lr *= lr_decay
    return best_h, best_val


# ------------------------------------------------------------
# Optional quick polishing with SLSQP (same as previous solution)
# ------------------------------------------------------------
def _solve_h(initial_h: np.ndarray, max_iter: int = 180) -> tuple:
    """One SLSQP refinement starting from a feasible h."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    def _obj(x):
        return x[-1]

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    res = optimize.minimize(_obj, x0, method="SLSQP", bounds=bounds,
                            constraints=cons,
                            options={"maxiter": max_iter, "ftol": 1e-9,
                                     "disp": False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)
