def smooth_obj_and_grad(h, beta, target, mu):
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    # stable log-sum-exp
    max_corr = np.max(corr)
    # compute w
    exp_bc = np.exp(beta * (corr - max_corr))
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp
    # smooth max value
    smooth = (max_corr + np.log(sum_exp)) / beta

    # gradient of smooth max
    g = np.zeros_like(h)
    # w length = 2n -1
    offset = n - 1
    for idx, weight in enumerate(w):
        d = idx - offset
        if d >= 0:
            # left contribution: h_j part left factor
            if n - d > 0:
                g[:n-d] += weight * b[d:]
                g[d:] -= weight * h[:n-d]
        else:
            dp = -d
            if n - dp > 0:
                # left contribution for negative shift
                g[dp:] += weight * b[:n-dp]   # n-dp = n + d
                g[:n-dp] -= weight * h[dp:]
    # penalty gradient
    diff = h.sum() - target
    grad = g + 2.0 * mu * diff * np.ones_like(h)
    # objective with penalty
    obj = smooth + mu * diff * diff
    return obj, grad
```

def hill_climb(h_start, rng, max_iters=500, init_step=0.2):
    h = h_start.copy()

        j = rng.integers(0, h.size)

        eps = (rng.random() - 0.5) * 2.0 * max_eps * step

        new_h[i] -= eps
        new_h[j] += eps

        if it % (max_iters // 5 + 1) == 0:

import time
import numpy as np
from scipy.optimize import minimize

# ----- global placeholders for the smooth objective (avoid lambdas) -----
GLOBAL_BETA = None
GLOBAL_TARGET = None
GLOBAL_MU = None

def smooth_obj_and_grad_wrapper(x):
    """Wrapper used by scipy.optimize – reads globals for parameters."""
    return smooth_obj_and_grad(x, GLOBAL_BETA, GLOBAL_TARGET, GLOBAL_MU)

def compute_c5(h):
    """Exact discrete C₅ for a step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def fix_sum(h, target, rng):
    """Redistribute a tiny amount to enforce Σh = target (exactly)."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = fix_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=1500, init_step=0.3):
    """Simple hill‑climb (accept only strictly improving moves)."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            h, best_c = new_h, c
        if (it + 1) % (max_iters // 6 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def smooth_obj_and_grad(h, beta, target, mu):
    """Smooth approximation of max_{k} Σ h_i (1‑h_{i+k}) + quadratic sum penalty."""
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")          # length 2n‑1
    max_corr = np.max(corr)
    # stable log‑sum‑exp
    exp_bc = np.exp(beta * (corr - max_corr))
    sum_exp = exp_bc.sum()
    smooth = max_corr + np.log(sum_exp) / beta
    w = exp_bc / sum_exp                            # weights for the soft‑max
    # gradient of the smooth max
    g = np.zeros_like(h)
    offset = n - 1
    for idx, weight in enumerate(w):
        d = idx - offset
        if d >= 0:
            if n - d > 0:
                g[:n - d] += weight * b[d:]
                g[d:]     -= weight * h[:n - d]
        else:
            dp = -d
            if n - dp > 0:
                g[dp:]    += weight * b[:n - dp]
                g[:n - dp] -= weight * h[dp:]
    # quadratic penalty for the sum constraint
    diff = h.sum() - target
    grad = g + 2.0 * mu * diff * np.ones_like(h)
    obj = smooth + mu * diff * diff
    return obj, grad

def optimize_with_beta(h_start, beta, target, mu, max_iter, rng):
    """Run a single L‑BFGS‑B optimisation on the smooth surrogate."""
    global GLOBAL_BETA, GLOBAL_TARGET, GLOBAL_MU
    GLOBAL_BETA = beta
    GLOBAL_TARGET = target
    GLOBAL_MU = mu
    n = h_start.size
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_obj_and_grad_wrapper,
        h_start,

    """Linearly interpolate a parent solution to a different resolution."""
    n_parent = parent_h.size
    dx_parent = 2.0 / n_parent
    xs_parent = (np.arange(n_parent) + 0.5) * dx_parent
    dx = 2.0 / n_target
    xs_target = (np.arange(n_target) + 0.5) * dx
    return np.interp(xs_target, xs_parent, parent_h, left=0.0, right=0.0)

def triangular_initial(n):
    """Triangular shape: h(x)=max(0,1‑|x‑1|)."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    return h

def cosine_initial(n):
    """Cosine bump: h(x) = (1+cos(π x))/2."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = 0.5 * (1.0 + np.cos(np.pi * xs))
    return h
