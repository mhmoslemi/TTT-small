def hill_climb_binary(n, seed, max_iters, time_limit):
    np.random.seed(seed)
    # Start with random binary vector having exactly n//2 ones.
    h = np.zeros(n, dtype=np.int8)
    ones_idx = np.random.choice(n, n//2, replace=False)
    h[ones_idx] = 1

    max_corr = np.max(np.correlate(h, 1 - h, mode='full'))
    best_corr = max_corr
    best_h = h.copy()

    start_time = time.time()
    it = 0
    while it < max_iters and time.time() - start_time < time_limit:
        # compute correlation array and find worst shift index, and shift value
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        max_idx = corr.argmax()
        s = max_idx - (n - 1)  # shift
        
        # find contributions indices
        if s >= 0:
            h_i = h[:n-s]
            h_i_s = h[s:]
            contrib_i = np.where((h_i == 1) & (h_i_s == 0))[0]
        else:
            s_abs = -s
            h_i = h[s_abs:]
            h_i_s = h[:n-s_abs]
            contrib_i = np.where((h_i == 1) & (h_i_s == 0))[0] + s_abs
        
        if contrib_i.size == 0:
            # no contributions? possibly zero correlation, break
            break
        
        i = np.random.choice(contrib_i)
        # eligible zeros to swap
        if s >= 0:
            # compute mask for zeros where either out-of-range or partner is 1
            positions = np.arange(n)
            partner_in_range = positions + s < n
            partner_is_one = np.zeros(n, dtype=bool)
            partner_is_one[partner_in_range] = (h[positions[partner_in_range] + s] == 1)
            eligible_j = np.where((h == 0) & ((~partner_in_range) | partner_is_one))[0]
        else:
            s_abs = -s
            positions = np.arange(n)
            partner_in_range = positions - s_abs >= 0
            partner_is_one = np.zeros(n, dtype=bool)
            partner_is_one[partner_in_range] = (h[positions[partner_in_range] - s_abs] == 1)
            eligible_j = np.where((h == 0) & ((~partner_in_range) | partner_is_one))[0]
        if eligible_j.size == 0:
            # fallback: any zero
            eligible_j = np.where(h == 0)[0]
        j = np.random.choice(eligible_j)
        # perform swap
        h[i] = 0
        h[j] = 1
        
        new_max_corr = np.max(np.correlate(h, 1 - h, mode='full'))
        if new_max_corr <= max_val:
            # accept
            max_corr = new_max_corr
            if max_corr < best_corr:
                best_corr = max_corr
                best_h = h.copy()
        else:
            # reject: swap back
            h[i] = 1
            h[j] = 0
        it += 1

import time
import numpy as np
from scipy import optimize

# ---------- feasibility helpers ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 via water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def _ineq_constraint(x):
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 300) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        _objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = _project_to_feasible(res.x[:n] if res.success else h0, n)
    return h_opt, _c5_from_h(h_opt)

def _binary_from_continuous(arr: np.ndarray) -> np.ndarray:
    """Turn a real vector into a binary one with exactly n/2 ones."""
    n = arr.size
    half = n // 2
    idx = np.argsort(-arr)[:half]
    h = np.zeros(n, dtype=np.int8)
    h[idx] = 1
    return h
