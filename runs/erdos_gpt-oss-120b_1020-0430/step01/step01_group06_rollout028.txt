    # ------------------------------------------------------------------
    # Basic utilities
    # ------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)

        init = None

    # --------------------------------------------------------------
    # Evaluation of C5
    # --------------------------------------------------------------
    def eval_c5(v):
        """max correlation * dx"""
        return float(np.correlate(v, 1.0 - v, mode="full").max() * dx)

    # --------------------------------------------------------------
    # Projection onto [0,1] and sum = target_sum
    # --------------------------------------------------------------
    def project(v):
        """Euclidean projection onto the box [0,1]∩{sum = target_sum}."""
        x = np.clip(v, 0.0, 1.0)
        diff = x.sum() - target_sum

        for _ in range(5 * n):
            if abs(diff) < 1e-12:
                break

            nfree = int(free.sum())
            if nfree == 0:
                break
            delta = diff / nfree
            x[free] -= delta
            x = np.clip(x, 0.0, 1.0)
            diff = x.sum() - target_sum

        err = x.sum() - target_sum

            # B = h[i-s] (0 if out of range)
            B = np.empty(n)
            B[:s] = 0.0
            B[s:] = v[: n - s]
        else:
            t = -s
            A = np.empty(n)
            A[t:] = 1.0 - v[: n - t]
            A[:t] = 1.0
            B = np.empty(n)
            B[: n - t] = v[t:]
            B[n - t:] = 0.0
        grad = A - B
        return grad, s, corr[idx_max] * dx

    # --------------------------------------------------------------
    # Initial best candidate
    # --------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")
    if init is not None:
        h0 = project(init)
        c0 = eval_c5(h0)
        best_h, best_c5 = h0, c0

    # --------------------------------------------------------------
    # Main multi‑restart projected‑gradient phase
    # --------------------------------------------------------------
    RESTARTS = 30
    ITER_PER = 2000

    for _ in range(RESTARTS):
        if time.time() - start > budget_s * 0.92:
            break
        # random feasible start
        cand = project(np.random.rand(n))
        cur = eval_c5(cand)
        eta = 0.5
        for it in range(ITER_PER):
            if time.time() - start > budget_s * 0.92:
                break
            g, shift, wval = worst_gradient(cand)
            # trial move
            trial = project(cand - eta * g)
            new = eval_c5(trial)
            if new < cur - 1e-12:
                cand, cur = trial, new
                # keep step size modestly large after success
                eta = min(1.0, eta * 1.02)
            else:
                eta *= 0.7
                if eta < 1e-5:
                    eta = 0.5
            # occasional small random jitter to escape flat plateaus
            if it % 200 == 0:
                pert = (np.random.rand(n) - 0.5) * 0.02
                trial = project(cand + pert)
                new = eval_c5(trial)
                if new < cur:
                    cand, cur = trial, new
        if cur < best_c5:
            best_h, best_c5 = cand, cur

    # --------------------------------------------------------------
    # Follow‑up simulated‑annealing refinement (targeted moves omitted,
    # simple two‑coordinate swaps are sufficient)
    # --------------------------------------------------------------
    if best_h is None:
        # fallback – should never happen
        best_h = project(np.random.rand(n))
        best_c5 = eval_c5(best_h)

    h = best_h.copy()
    cur = best_c5
    T = max(1e-4, cur * 0.02)
    decay = 0.9996
    it = 0

    while time.time() - start < budget_s * 0.98:
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)         # raise i, lower j
        max_down = min(hi, 1.0 - hj)       # lower i, raise j
        if max_up + max_down == 0:
            it += 1
            continue
        delta = (random.random() * (max_up + max_down)) - max_down
        if abs(delta) < 1e-14:
            it += 1
            continue
        nh = h.copy()
        nh[i] = hi + delta
        nh[j] = hj - delta
        new = eval_c5(nh)
        if new < cur or random.random() < math.exp(-(new - cur) / T):
            h, cur = nh, new
            if new < best_c5:
                best_h, best_c5 = nh.copy(), new
        it += 1
        if it % 1000 == 0:
            T *= decay

    # --------------------------------------------------------------
    # Final safety projection and return
    # --------------------------------------------------------------
    best_h = project(best_h)
    final_c5 = eval_c5(best_h)

    return (best_h.tolist(), final_c5, n)
