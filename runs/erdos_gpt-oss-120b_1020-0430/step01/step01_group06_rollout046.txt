    import numpy as np, random, time, math
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # Determine n
    try:
        init = np.array(initial_h_values, dtype=float).copy()
        n = init.shape[0]
    except NameError:
        n = 70
        init = None
    target_ones = n // 2

    # Helper: max correlation count
    def max_corr(arr):
        # arr is binary 0/1 integer array
        c = np.correlate(arr, 1 - arr, mode='full')
        return int(c.max())

    # Helper: full correlation and max shift
    def corr_and_max(arr):
        c = np.correlate(arr, 1 - arr, mode='full')
        max_val = int(c.max())
        idx = int(np.argmax(c))
        shift = idx - (n - 1)
        return max_val, shift, c

    # Random binary generator with correct sum
    def random_binary():
        h = np.zeros(n, dtype=np.int8)
        ones = random.sample(range(n), target_ones)
        h[ones] = 1
        return h

    # Initialise h
    if init is not None:
        # Convert to binary via threshold
        h = (init > 0.5).astype(np.int8)
        cur_sum = int(h.sum())
        if cur_sum != target_ones:
            diff = cur_sum - target_ones
            if diff > 0:
                # turn diff ones to zeros
                idx_one = np.where(h == 1)[0]
                to_zero = random.sample(list(idx_one), diff)
                h[to_zero] = 0
            else:
                # turn -diff zeros to ones
                idx_zero = np.where(h == 0)[0]
                to_one = random.sample(list(idx_zero), -diff)
                h[to_one] = 1
    else:
        h = random_binary()

    best_h = h.copy()
    best_max = max_corr(best_h)

    # Simulated annealing parameters
    T = 4.0
    decay = 0.9997
    it = 0
    since_improve = 0
    # Main loop
    while time.time() - start < budget_s * 0.95:
        cur_max, worst_shift, _ = corr_and_max(h)

        if random.random() < 0.6:

            s = worst_shift
            if s >= 0:

                contrib_idxs = i_vals[cond]
                if contrib_idxs.size == 0:
                    i = random.choice(np.where(h == 1)[0])
                else:
                    i = int(random.choice(contrib_idxs.tolist()))
                zero_idxs = np.where(h == 0)[0]
                j = int(random.choice(zero_idxs.tolist()))

            j = random.choice(np.where(h == 0)[0].tolist())

        h_new[i] = 0

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float).copy()
        n = init.shape[0]
    except NameError:
        n = 70
        init = None

    target_ones = n // 2                     # because ∫h = 1  →  sum = n/2
    # --------------------------------------------------------------
    # Helper functions
    # --------------------------------------------------------------
    def max_corr(arr):
        """maximum value of correlate(arr, 1-arr) (integer count)."""
        return int(np.correlate(arr, 1 - arr, mode="full").max())

    def corr_and_max(arr):
        """return max count, shift achieving it, and the full correlation."""
        c = np.correlate(arr, 1 - arr, mode="full")
        max_val = int(c.max())
        idx = int(np.argmax(c))
        shift = idx - (len(arr) - 1)
        return max_val, shift, c

    def random_binary():
        """balanced 0/1 vector."""
        h = np.zeros(n, dtype=np.int8)
        ones = random.sample(range(n), target_ones)
        h[ones] = 1
        return h

    # --------------------------------------------------------------
    # Initialise h (binary) – try to reuse initial_h_values if present
    # --------------------------------------------------------------
    if init is not None:
        h = (init > 0.5).astype(np.int8)
        cur = int(h.sum())
        if cur != target_ones:
            diff = cur - target_ones
            if diff > 0:                     # too many ones → turn some off
                idx = np.where(h == 1)[0]
                to_zero = random.sample(list(idx), diff)
                h[to_zero] = 0
            else:                             # too few ones → turn some on
                idx = np.where(h == 0)[0]
                to_one = random.sample(list(idx), -diff)
                h[to_one] = 1
    else:
        h = random_binary()

    best_h = h.copy()
    best_max = max_corr(best_h)

    # --------------------------------------------------------------
    # Simulated annealing (targeted swaps + random swaps)
    # --------------------------------------------------------------
    T = 4.0                     # initial temperature
    decay = 0.9997
    it = 0

    while time.time() - start < budget_s * 0.95:
        cur_max, worst_shift, _ = corr_and_max(h)

        # ----- choose move type -------------------------------------------------
        if random.random() < 0.6:                           # targeted
            s = worst_shift
            if s >= 0:
                i_vals = np.arange(n - s)
                cond = (h[i_vals] == 1) & (h[i_vals + s] == 0)
                contrib = i_vals[cond]
                if contrib.size == 0:
                    i = int(random.choice(np.where(h == 1)[0].tolist()))
                else:
                    i = int(random.choice(contrib.tolist()))
                j = int(random.choice(np.where(h == 0)[0].tolist()))
            else:
                s_abs = -s

                cond = (h[i_vals] == 1) & (h[i_vals - s_abs] == 0)

                    i = int(random.choice(np.where(h == 1)[0].tolist()))
                else:
                    i = int(random.choice(contrib.tolist()))
