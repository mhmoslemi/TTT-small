import time, math, numpy as np

from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max overlap (log‑sum‑exp trick)."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular seed, scaled to satisfy the integral constraint."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector (values in [0,1] and ∑h = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, preserving the integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

# Optimization primitives
# ------------------------------------------------------------
def slsqp_optimize(start: np.ndarray, deadline: float) -> np.ndarray:
    """Two‑stage SLSQP from a feasible start; respects the time deadline."""
    n = start.shape[0]
    target = n / 2.0

    def eq_sum(h):
        return np.sum(h) - target

    cons = {"type": "eq", "fun": eq_sum}
    bounds = [(0.0, 1.0)] * n

    # Stage 1 – smooth objective, modest α
    res1 = minimize(
        lambda h: softmax_c5(h, alpha=5.0),
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res1.success or time.time() > deadline:
        return None

    # Stage 2 – sharper objective, larger α
    res2 = minimize(
        lambda h: softmax_c5(h, alpha=30.0),
        res1.x,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res2.success or time.time() > deadline:
        return None
    return np.clip(res2.x, 0.0, 1.0)


def simulated_annealing(h0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple:
    """Continuous SA that preserves the integral exactly (pairwise adjustments)."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5

    # temperature schedule
    temp = 0.1
    final_temp = 1e-5
    cool = 0.9995
    max_step = 0.2  # maximal absolute change per move

    # pre‑compute index arrays for speed
    idx = np.arange(n)

    while time.time() < deadline:
        # pick distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # admissible interval for delta (add to i, subtract from j)
        lo = max(-h[i], h[j] - 1.0, -max_step)
        hi = min(1.0 - h[i], h[j], max_step)
        if lo >= hi:
            continue
        delta = rng.uniform(lo, hi)

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # numeric safety
        if h_new[i] < -1e-12 or h_new[i] > 1.0 + 1e-12 or h_new[j] < -1e-12 or h_new[j] > 1.0 + 1e-12:
            continue

        new_c5 = compute_c5(h_new)

        improve = new_c5 < cur_c5
        if improve or rng.random() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h, best_c5 = h_new.copy(), new_c5

        # cool down
        temp = max(final_temp, temp * cool)

    # final tiny projection onto the feasible hyperplane (sum may drift by 1e-12)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic search for a step function with a low C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.96  # cushion

    # decide on a working resolution
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        n_parent = parent_h.shape[0]
        # ensure even length (needed for exact half‑mass)
        if n_parent % 2 == 1:
            n_parent += 1
            parent_h = upsample(parent_h, n_parent)
    except NameError:
        parent_h = None
        n_parent = None

    # candidate sizes – keep them modest for speed
    candidate_ns = [80, 100, 120, 140, 160]
    if n_parent and n_parent not in candidate_ns:
        candidate_ns.insert(0, n_parent)

    best_h, best_c5, best_n = None, np.inf, None

    for n in candidate_ns:
        if time.time() > time_limit:
            break

        target_sum = n / 2.0

        # build a short list of diverse starts
        starts = []

        # parent (upsampled / downsampled)
        if parent_h is not None:
            if n == n_parent:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # classic triangular seed
        starts.append(generate_triangular(n))

        # a few random feasible vectors
        for _ in range(3):
            starts.append(random_start(n, rng))

        # try each start
        for s in starts:
            if time.time() > time_limit:
                break

            # ---- SLSQP refinement -------------------------------------------------
            slsqp_h = slsqp_optimize(s, time_limit)
            if slsqp_h is None:
                continue

            # ---- Simulated annealing -----------------------------------------------
            sa_h, sa_c5 = simulated_annealing(slsqp_h, time_limit, rng)

            if sa_c5 < best_c5 - 1e-9:
                best_h, best_c5, best_n = sa_h, sa_c5, n

        # if we obtained a promising bound quickly, try a second SA run from the best so far
        if best_h is not None and time.time() < time_limit:
            extra_h, extra_c5 = simulated_annealing(best_h, time_limit, rng)
            if extra_c5 < best_c5 - 1e-9:
                best_h, best_c5 = extra_h, extra_c5
                best_n = n

    # -------------------------------------------------------------------------
    # Fallback: a simple binary block (guaranteed feasible) if nothing succeeded
    # -------------------------------------------------------------------------
    if best_h is None:
        n = 100
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final sanity – enforce the integral constraint exactly
    target = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-12:
        best_h = best_h + (target - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
