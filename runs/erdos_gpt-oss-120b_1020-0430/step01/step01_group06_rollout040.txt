def solve_mip(n):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(1, integer=True)
    constraints = [cp.sum(h) == n//2]  # n is even
    # For each shift
    for s in range(-(n-1), n):
        if s == 0:
            # O(0) = sum_i h_i * (1 - h_i) = 0 automatically because product term = h_i
            # Actually O(0) = sum h_i - sum h_i*h_i = sum h_i - sum h_i = 0
            continue
        # overlapping pairs
        if s > 0:
            i_range = range(0, n - s)
        else:
            i_range = range(-s, n)
        i_idx = list(i_range)
        j_idx = [i + s for i in i_idx] # j in [0, n-1]
        m = len(i_idx)
        # create w binary vector of length m
        w = cp.Variable(m, boolean=True)
        # w <= h[i], w <= h[j], w >= h[i] + h[j] - 1
        constraints += [w <= h[i_idx], w <= h[j_idx]]
        constraints += [w >= h[i_idx] + h[j_idx] - 1]
        # overlap = sum_i h_i - sum w
        overlap = cp.sum(h[i_idx]) - cp.sum(w)
        constraints += [overlap <= t]
    # t upper bound maybe n, lower bound 0
    constraints += [t >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.GLPK_MI, verbose=False)
    return h.value, t.value

def solve_mip(n, target_sum, start, budget):

    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == target_sum]
    for s in range(1, n):
        # positive shift s
        i = np.arange(0, n - s)
        j = i + s
        m = i.shape[0]
        if m > 0:
            w = cp.Variable(m, boolean=True)
            # constraints
            constraints.append(w <= h[i])
            constraints.append(w <= h[j])
            constraints.append(w >= h[i] + h[j] - 1)
            overlap = cp.sum(h[i]) - cp.sum(w)
            constraints.append(overlap <= t)
        # negative shift -s
        i2 = np.arange(s, n)
        j2 = i2 - s
        m2 = i2.shape[0]

            constraints.append(w2 <= h[i2])
            constraints.append(w2 <= h[j2])
            constraints.append(w2 >= h[i2] + h[j2] - 1)
            overlap2 = cp.sum(h[i2]) - cp.sum(w2)
            constraints.append(overlap2 <= t)
    constraints.append(t >= 0)
    constraints.append(t <= n)  # optional bound

    prob = cp.Problem(cp.Minimize(t), constraints)
    # compute remaining time

    else:

                one_idx = np.where(hv > 0.5)[0]
                if one_idx.size:
                    hv[one_idx[: -diff]] = 0.0
        return hv.astype(float)

    # --------------------------------------------------------------
    # Try binary bound 13 (gives C5 = 13*2/n = 0.3714 for n=70)
    # --------------------------------------------------------------
    best_h = None
    best_c5 = None
    for bound in (13, 14):
        hv = feasible_binary(bound)
        if hv is not None:
            cur_c5 = c5_of(hv)
            if best_c5 is None or cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = hv
            # stop trying larger bounds if we already succeeded with the smallest
            break

    # --------------------------------------------------------------
    # If MILP failed or did not meet target, fall back to SA
    # --------------------------------------------------------------
    if best_c5 is not None and best_c5 <= 0.3808:
        return (best_h.tolist(), best_c5, n)

    # ----------- helper to create a random feasible vector ------------
    def random_feasible():
        v = np.random.rand(n)
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target_sum - cur
        # simple projection onto the hyperplane with box constraints
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                cand = np.where(v < 1 - 1e-12)[0]
                if not cand.size:
                    break
                i = random.choice(cand.tolist())
                delta = min(1 - v[i], diff)
                v[i] += delta
                diff -= delta
            else:
                cand = np.where(v > 1e-12)[0]
                if not cand.size:
                    break
                i = random.choice(cand.tolist())
                delta = min(v[i], -diff)
                v[i] -= delta
                diff += delta
        return v

    # -------------------- simulated annealing -----------------------
    # start from the best binary solution if we have one,
    # otherwise from a fresh random vector.
    if best_h is not None:
        h = best_h.copy()
    else:
        h = random_feasible()
    cur_c5 = c5_of(h)
    best_local_h = h.copy()
    best_local_c5 = cur_c5

    T = max(1e-4, cur_c5 * 0.02)
    decay = 0.9997
    it = 0
    last_improve = it

    while time.time() - start < budget_s * 0.95:
        # ---- targeted improvement on the worst shift ----------------
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = int(np.argmax(corr))
        s = idx_max - (n - 1)                # shift that attains the max
        if s >= 0:
            i_inds = np.arange(0, n - s, dtype=int)
            p_inds = i_inds + s
        else:
            s_abs = -s
            i_inds = np.arange(s_abs, n, dtype=int)
            p_inds = i_inds - s_abs
        contrib = h[i_inds] * (1.0 - h[p_inds])
        k = int(np.argmax(contrib))
        i_best, p_best = i_inds[k], p_inds[k]
        eps_max = min(h[i_best], 1.0 - h[p_best])
        if eps_max > 1e-12:
            eps = eps_max * 0.5
            nh = h.copy()
            nh[i_best] -= eps
            nh[p_best] += eps
            new_c5 = c5_of(nh)
            if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                h = nh
                cur_c5 = new_c5
                if new_c5 < best_local_c5:
                    best_local_c5 = new_c5
                    best_local_h = h.copy()
                    last_improve = it

        # ---- random two‑coordinate exchange preserving the sum -----
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)      # increase i, decrease j
        max_down = min(hi, 1.0 - hj)    # decrease i, increase j
        if max_up + max_down > 0:
            delta = (random.random() * (max_up + max_down)) - max_down
            if abs(delta) > 1e-14:
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_c5 = c5_of(nh)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_local_c5:
                        best_local_c5 = new_c5
                        best_local_h = h.copy()
                        last_improve = it

        it += 1
        if it % 1000 == 0:
            T *= decay
        # restart if no progress for a while
        if it - last_improve > 20000:
            h = random_feasible()
            cur_c5 = c5_of(h)
            last_improve = it

    # final correction of possible numerical drift
    err = best_local_h.sum() - target_sum
    if abs(err) > 1e-8:
        for _ in range(100):
            k = random.randrange(n)
            if err > 0 and best_local_h[k] >= err:
                best_local_h[k] -= err
                break
            if err < 0 and best_local_h[k] <= 1 + err:
                best_local_h[k] -= err
                break

    final_c5 = c5_of(best_local_h)
    return (best_local_h.tolist(), final_c5, n)
