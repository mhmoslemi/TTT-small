def improve_with_slsqp(h, dx, target_sum, alpha_start=5, alpha_end=500, steps=5):
    h_cur = h.copy()
    alphas = np.geomspace(alpha_start, alpha_end, num=steps)
    for alpha in alphas:
        res = minimize(objective_smooth, h_cur, args=(dx, alpha), method='SLSQP',
                       bounds=bounds, constraints={'type':'eq', 'fun':sum_eq_constraint},
                       options={'maxiter':300, 'ftol':1e-9})
        h_cur = project_onto_bounds_and_sum(res.x, target_sum)
    return h_cur
```

def local_refine(h, target_sum, n_iter, dx):
    best = h.copy()
    best_val = compute_c5(best)
    n = len(h)
    for _ in range(n_iter):
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        hi, hj = best[i], best[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        # choose delta randomly within range, maybe scaled
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        # ensure within bounds and sum (should be)
        # but due to floating errors may need projection
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best = cand
            # early exit if below target 0.3808
            if best_val <= 0.3808:
                break
    return best, best_val
```

def smooth_max(arr, alpha):
    m = np.max(arr)
    return m + np.log(np.sum(np.exp(alpha * (arr - m)))) / alpha

def local_refine(h, target_sum, n_iter_big, n_iter_small, dx):
    # big moves
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(n_iter_big):
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        hi, hj = best[i], best[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12: continue
        cand = best.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        # projection step may not be needed as sum unchanged ignoring rounding
        # but keep to be safe
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best = cand
    # small moves
    for _ in range(n_iter_small):
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        hi, hj = best[i], best[j]
        # restrict delta to small magnitude
        max_delta = min(0.02, 1.0 - hi, hj)

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:

    low, high = -1.0, 1.0

            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(arr)
    return m + np.log(np.sum(np.exp(alpha * (arr - m)))) / alpha


def compute_c5(h: np.ndarray) -> float:
    """Exact objective used for final evaluation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth surrogate for SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.where((pts >= 0.5) & (pts <= 1.5), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def improve_with_slsqp(h0, dx, target_sum, bounds, alphas):
    """Continuation SLSQP: start with low α, increase to high α."""
    h = h0.copy()
    for alpha in alphas:
        try:
            res = minimize(
                objective_smooth,
                h,
                args=(dx, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": 400, "ftol": 1e-9, "disp": False},
            )
            h = project_onto_bounds_and_sum(res.x, target_sum)
        except Exception:
            continue
    return h


def local_refine(h, target_sum, big_iters, small_iters, dx):
    """Two‑phase pairwise mass transfer refinement."""
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)

    # large moves
    for _ in range(big_iters):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best[i], best[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val, best = val, cand

    # fine moves
    for _ in range(small_iters):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best[i], best[j]
        max_delta = min(0.02, 1.0 - hi, hj)
        min_delta = -min(0.02, hi, 1.0 - hj)
        if max_delta < min_delta:
            continue
        delta = np.random.uniform(min_delta, max_delta)
        if abs(delta) < 1e-14:
            continue
        cand = best.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val, best = val, cand
    return best, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    n = 500                     # more points → finer shape
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # collect diverse seeds
    seed_vecs = []
    if "initial_h_values" in globals():
        try:
            seed_vecs.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seed_vecs.append(make_triangle(n))
    seed_vecs.append(make_sin2(n))
    seed_vecs.append(make_rectangle(n))

    best_h = None
    best_val = np.inf
    alphas = np.geomspace(5.0, 200.0, num=5)  # smoothing schedule

    # deterministic seeds
    for h0 in seed_vecs:
        if time.time() - start > budget_s - 5:
            break
        h_opt = improve_with_slsqp(h0, dx, target_sum, bounds, alphas)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()
        # allocate a modest amount of refinement time
        remaining = budget_s - (time.time() - start)
        big = int(min(5000, max(0, remaining * 1500)))
        small = int(min(20000, max(0, remaining * 6000)))
        best_h, best_val = local_refine(best_h, target_sum, big, small, dx)
        if best_val <= 0.3808:
            break

    # random restarts while time permits
    while time.time() - start < budget_s - 5:
        rnd = np.random.rand(n)
        h0 = project_onto_bounds_and_sum(rnd, target_sum)
        h_opt = improve_with_slsqp(h0, dx, target_sum, bounds, alphas)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()
        remaining = budget_s - (time.time() - start)
        big = int(min(3000, max(0, remaining * 1200)))
        small = int(min(15000, max(0, remaining * 4800)))
        best_h, best_val = local_refine(best_h, target_sum, big, small, dx)
        if best_val <= 0.3808:
            break

    # final safety projection and exact evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
