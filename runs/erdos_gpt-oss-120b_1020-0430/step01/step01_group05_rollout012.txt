def gradient_smooth_max(h, alpha=20):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w = w / w.sum()
    g = np.zeros_like(h)
    # shift k from -(n-1) to n-1
    for k in range(-(n-1), n):
        weight = w[k + n - 1]
        if weight == 0:
            continue
        if k >= 0:
            # overlapping i = 0..n-1-k
            i = slice(0, n - k)
            j = slice(k, n)
        else:
            # k negative, overlapping i = -k..n-1
            i = slice(-k, n)
            j = slice(0, n + k)  # Since k negative, n + k < n
        # g[i] += weight * (1 - h[j])
        g[i] += weight * (1 - h[j])
        # g[j] -= weight * h[i]
        g[j] -= weight * h[i]
    return g

Note: For k negative, j slice of length n + k (k negative) correct.

def gradient_smooth_max(h, alpha=20.0):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w = w / w.sum()
    g = np.zeros_like(h)
    # shift from -(n-1) to n-1
    for shift in range(-(n-1), n):
        weight = w[shift + n - 1]
        if weight == 0:
            continue
        if shift >= 0:
            i_slice = slice(0, n - shift)
            j_slice = slice(shift, n)
        else:
            i_slice = slice(-shift, n)
            j_slice = slice(0, n + shift)
        # update g
        g[i_slice] += weight * (1 - h[j_slice])
        g[j_slice] -= weight * h[i_slice]
    return g

def random_pairwise_move(h, target_sum):
   n = len(h)
   i, j = np.random.randint(0, n, 2)
   if i == j: return None
   hi, hj = h[i], h[j]
   delta_max = min(1.0 - hi, hj)
   delta_min = -min(hi, 1.0 - hj)
   if delta_max <= delta_min: return None
   delta = np.random.uniform(delta_min, delta_max)
   h_new = h.copy()
   h_new[i] = hi + delta
   h_new[j] = hj - delta
   # after adjusting, still within [0,1] because delta within bounds.

import time, random, numpy as np

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while forcing sum(x)=target (simple bisection)."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective (max overlap) for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def gradient_smooth_max(h: np.ndarray, alpha: float = 20.0) -> np.ndarray:
    """Gradient of smooth_max(corr,alpha) w.r.t. h (ignoring the outer dx factor)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()                     # softmax weights, sum to 1
    g = np.zeros_like(h)

    # shifts from -(n-1) to n-1
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        if k >= 0:
            i_slice = slice(0, n - k)
            j_slice = slice(k, n)
        else:
            i_slice = slice(-k, n)
            j_slice = slice(0, n + k)
        g[i_slice] += wk * (1.0 - h[j_slice])
        g[j_slice] -= wk * h[i_slice]
    return g


def symmetrize_and_project(h: np.ndarray, target: float) -> np.ndarray:
    """Force symmetry about the midpoint, then enforce constraints."""
    h = (h + h[::-1]) * 0.5
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, target)


def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new length."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, n / 2.0)
