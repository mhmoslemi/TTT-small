    # clip
    h = np.clip(h, 0.0, 1.0)
    # iterative.
    for _ in range(10):
        sum_h = h.sum()
        diff = target_sum - sum_h
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # final normalization due to rounding errors:
    # If still off, adjust greedily:
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        # reduce diff by adjusting interior points
        indices = np.where((h > 0) & (h < 1))[0]
        if len(indices) > 0:
            step = diff / len(indices)
            h[indices] += step
            h = np.clip(h, 0.0, 1.0)
        # if still diff, adjust extremes
        # ensure final sum via last resort:
        while abs(h.sum() - target_sum) > 1e-9:
            i = np.random.randint(len(h))
            if diff > 0 and h[i] < 1:
                delta = min(diff, 1 - h[i])
                h[i] += delta
                diff -= delta
            elif diff < 0 and h[i] > 0:
                delta = min(-diff, h[i])
                h[i] -= delta
                diff += delta
    return h
