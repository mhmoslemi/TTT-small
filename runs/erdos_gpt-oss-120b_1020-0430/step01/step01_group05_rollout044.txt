def gradient_smooth_max(weights, h):
    n = len(h)
    grad = np.full(n, np.sum(weights))  # should be 1

import time, random, math
import numpy as np
from scipy.optimize import minimize

# ---------- Global helpers (no closures) ----------
TARGET_SUM_GLOBAL = None   # set before optimization
CURRENT_DX = 1.0
CURRENT_ALPHA = 20.0

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n intersected with sum(x)=target using bisection."""
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised max overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    """Width 1 centred at 1, height 1."""
    dx = 2.0 / n
    width = 1.0
    cnt = int(round(width / dx))
    start = (n - cnt) // 2
    h = np.zeros(n)
    h[start:start + cnt] = 1.0
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    if old_n == n:
        return project_onto_bounds_and_sum(init.astype(float), n / 2.0)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - TARGET_SUM_GLOBAL


def sum_eq_jac(x: np.ndarray) -> np.ndarray:
    return np.ones_like(x)


def objective_and_grad(h: np.ndarray, dx: float, alpha: float):
    """Smooth max of the correlation, returning value and gradient."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # stable softmax
    exp_term = np.exp(alpha * (corr - m))
    sumexp = np.sum(exp_term)
    smooth_val = (m + np.log(sumexp) / alpha) * dx
    weights = exp_term / sumexp  # sum to 1

    n = h.shape[0]
    # gradient of smooth max (before dx)
    grad = np.full(n, 1.0)  # sum(weights) == 1
    for s_idx, w in enumerate(weights):
        s = s_idx - (n - 1)
        if s == 0:
            grad -= 2.0 * w * h
        elif s > 0:
            grad[:n - s] -= w * h[s:]
            grad[s:]      -= w * h[:n - s]
        else:  # s < 0
            t = -s
            grad[t:]      -= w * h[:n - t]
            grad[:n - t]  -= w * h[t:]
    grad *= dx
    return smooth_val, grad

    return objective_and_grad(x, CURRENT_DX, CURRENT_ALPHA)
