import time, numpy as np

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=60):
    lo, hi = -np.max(v) - 1.0, np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(_corr_full(h).max() * dx)

def _gradient_for_shift(h, shift):
    n = h.size
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g

def _subgradient(h):
    n = h.size
    corr = _corr_full(h)
    m = corr.max()
    mask = corr >= m - 1e-12
    shifts = np.where(mask)[0] - (n - 1)
    grad = np.zeros_like(h)
    for sh in shifts:
        grad += _gradient_for_shift(h, sh)
    if shifts.size:
        grad /= shifts.size
    return grad

def _gd_momentum_optimize(h0, target, max_iter=2500, init_step=0.8,
                          decay=0.9995, beta=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    step = init_step
    for _ in range(max_iter):
        g = _subgradient(h)
        v = beta * v + step * g
        h = h - v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

def _anneal(h_start, val_start, target, dx, rng, budget, start):

    T = max(val - 0.2, 0.1)

    # rough estimate of iterations we can afford
