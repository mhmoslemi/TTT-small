   m = np.zeros_like(h)
   v = np.zeros_like(h)
   beta1=0.9

      shift = idx - (len(h)-1)

      # adam updates
      m = beta1*m + (1-beta1)*g
      v = beta2*v + (1-beta2)*(g**2)

   return best_h, best_val

import time
import numpy as np
from math import ceil

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Correlation via FFT (linear convolution)
# ----------------------------------------------------------------------
def _corr_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr


def _compute_c5_val(h: np.ndarray, dx: float) -> float:
    corr = _corr_fft(h)
    return float(corr.max() * dx)


# ----------------------------------------------------------------------
# Gradient of the worst (maximal) shift term
# ----------------------------------------------------------------------
def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[:s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Adam‑style projected optimizer
# ----------------------------------------------------------------------
def _adam_optimize(h0: np.ndarray, target: float, dx: float,
                   max_iter: int, start_time: float, budget_s: float):
    h = h0.copy()
    n = h.shape[0]
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0.10
    best_h = h.copy()
    best_val = _compute_c5_val(h, dx)

    for t in range(1, max_iter + 1):
        # early stop check every 50 iterations
        if t % 50 == 0 and (time.time() - start_time) > budget_s * 0.95:
            break

        corr = _corr_fft(h)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_worst_shift(h, shift)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        h = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h = _project_onto_capped_simplex(h, target)

        # periodic evaluation
        if t % 10 == 0:
            val = _compute_c5_val(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()

    return best_h, best_val
