    # clip y to [0,1] first? Actually we need to find λ such that sum_i clip(y_i - λ, 0, 1) = target.

    for _ in range(max_iter):
        lam = (lo + hi) / 2.0

            lo = lam

import time
import numpy as np

def evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_sum_box(y: np.ndarray, target: float, tol: float = 1e-12,
                     max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of y onto the set
        {x | 0 <= x_i <= 1, sum x_i = target }.
    Solved by binary search on the Lagrange multiplier.
    """
    lo = np.min(y) - 1.0      # guarantee sum(lo) >= target
    hi = np.max(y)            # guarantee sum(hi) <= target
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Choose resolution and initialise
    # ----------------------------------------------------------------------
    n = 91                     # a modestly larger grid gives more flexibility
    target = n / 2.0           # ∑h = target  (since ∫h = 1 and dx = 2/n)

    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init = np.asarray(init_vec, dtype=float)
        n0 = init.size
        dx0 = 2.0 / n0
        # cell centres for the old and new discretisation
        old_x = np.arange(n0) * dx0 + dx0 / 2.0
        dx = 2.0 / n
        new_x = np.arange(n) * dx + dx / 2.0
        h = np.interp(new_x, old_x, init, left=0.0, right=0.0)
    else:
        h = rng.random(n)

    h = np.clip(h, 0.0, 1.0)
    h = _project_sum_box(h, target)

    cur_h = h.copy()
    cur_c5 = evaluate_c5(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # 2. Simulated‑annealing style random pairwise mass moves
    # ----------------------------------------------------------------------
    total_time = budget_s - 10.0          # reserve a small margin for final tidy‑up
    end_time = start + total_time
    T0 = 0.10                              # initial temperature
    it = 0

    while time.time() < end_time:
        # linear cooling
        elapsed = time.time() - start
        T = max(1e-6, T0 * (1.0 - elapsed / total_time))

        # propose a transfer of mass from i (source) to j (dest)
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move          # random fraction of the feasible amount

        cur_h[i] -= delta
        cur_h[j] += delta
        new_c5 = evaluate_c5(cur_h)

        diff = new_c5 - cur_c5
        if diff < -1e-12 or rng.random() < np.exp(-diff / T):
            # accept
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # reject – revert
            cur_h[i] += delta
            cur_h[j] -= delta

        # occasional large perturbation to shake the system
        if it % 20000 == 0 and it > 0:
            noise = rng.normal(scale=0.2, size=n)
            trial = cur_h + noise
            trial = np.clip(trial, 0.0, 1.0)
            trial = _project_sum_box(trial, target)
            trial_c5 = evaluate_c5(trial)

            if trial_c5 < best_c5:
                best_c5, best_h = trial_c5, trial.copy()
                cur_h, cur_c5 = trial.copy(), trial_c5
            else:
                if rng.random() < np.exp(-(trial_c5 - cur_c5) / T):
                    cur_h, cur_c5 = trial.copy(), trial_c5

        it += 1

    # ----------------------------------------------------------------------
    # 3. Deterministic gradient‑guided refinement (pairwise mass exchange)
    # ----------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5

    for _ in range(5000):
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)                     # shift that realises the maximum

        # gradient of the active term S_k with respect to each component
        g = np.zeros(n)
        if k >= 0:
            r1 = np.arange(0, n - k)
            g[r1] += 1.0 - cur_h[r1 + k]
            r2 = np.arange(k, n)
            g[r2] -= cur_h[r2 - k]
        else:
            kk = -k
            r1 = np.arange(kk, n)
            g[r1] += 1.0 - cur_h[r1 - kk]
            r2 = np.arange(0, n - kk)
            g[r2] -= cur_h[r2 + kk]

        # choose a source with large positive gradient and a sink with large negative
        src_cand = np.where(cur_h > 1e-12)[0]
        dst_cand = np.where(cur_h < 1.0 - 1e-12)[0]
        if src_cand.size == 0 or dst_cand.size == 0:
            break

        src = src_cand[np.argmax(g[src_cand])]
        dst = dst_cand[np.argmin(g[dst_cand])]

        if g[src] <= g[dst] + 1e-12:
            break   # no improving direction found

        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            break
        delta = 0.5 * max_move                     # step size – half of the feasible amount

        cur_h[src] -= delta
        cur_h[dst] += delta
        new_c5 = evaluate_c5(cur_h)

        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, cur_h.copy()
        else:
            # revert – no gain
            cur_h[src] += delta
            cur_h[dst] -= delta
            break

    # ----------------------------------------------------------------------
    # 4. Final projection and return
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_box(best_h, target)
    best_c5 = float(evaluate_c5(best_h))

    return best_h, best_c5, n
