def hill_climb_targeted(h_start, rng, max_iters=5000, start_step=0.5):

    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = len(h)

    for it in range(max_iters):
        # compute current correlation and max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        # compute contributions for max lag
        shift = max_idx - (n - 1)
        # produce list of candidate indices
        if shift >= 0:
            valid_i = np.arange(0, n - shift)
            # contributions
            contrib = h[valid_i] * (1 - h[valid_i + shift])
        else:
            valid_i = np.arange(-shift, n)
            contrib = h[valid_i] * (1 - h[valid_i + shift])
        # choose i with probability proportional to contribution or top set
        # if max(contrib) ==0, fall back random move.
        if np.max(contrib) > 0 and rng.random() < 0.9:
            # select among top 30% contributions
            thresh = np.percentile(contrib, 70)
            candidates = valid_i[contrib >= thresh]
            i = rng.choice(candidates)
        else:
            i = rng.integers(0, n)

        # choose j random anywhere (different)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # fallback to random move
            continue
        eps = rng.random() * max_eps * step
        # apply move
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_c = compute_c5(new_h)
        if new_c < best_c - 1e-12:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        else:
            # optional accept if equal?
            # keep old h
            pass

        # occasionally shrink step
        if (it+1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of maxₖ Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2 / n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def random_feasible(n, rng):
    """Generate a random feasible h (0 ≤ h_i ≤ 1, Σ h_i = n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 6):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def hill_climb_targeted(h0, rng, max_iters=8000, start_step=0.5):
    """Hill‑climb that always transfers mass from a high‑contributing index."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        # correlation & lag of current maximum
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        shift = max_idx - (n - 1)

        # indices contributing to this lag
        if shift >= 0:
            idx = np.arange(0, n - shift)
            contrib = h[idx] * (1.0 - h[idx + shift])
        else:
            idx = np.arange(-shift, n)
            contrib = h[idx] * (1.0 - h[idx + shift])

        # pick a source i among the top contributors (if any)
        if contrib.size and np.max(contrib) > 0 and rng.random() < 0.85:
            thresh = np.percentile(contrib, 70)
            cand = idx[contrib >= thresh]
            i = rng.choice(cand)
        else:
            i = rng.integers(0, n)

        # pick a destination j different from i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h

        # shrink step size slowly
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

def slsqp_refine(h_start, rng, n, betas, max_iter=400):
    """Run a short series of SLSQP passes with increasing beta."""
    global CURRENT_TARGET, CURRENT_BETA
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    for beta in betas:
        CURRENT_TARGET = n / 2.0
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        res = minimize(
            softmax_obj,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        # enforce exact sum (numerical drift)
        diff = CURRENT_TARGET - x.sum()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    TIME = lambda: time.time() - start_time

    # target resolution – larger gives more flexibility
    n = 180
    if n % 2:
        n += 1
    target_sum = n / 2.0

    # ---- build initial candidates ---------------------------------------
    candidates = []

    # 1. up‑sample the supplied parent if present
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        up = np.interp(xs, xs0, h0, left=0.0, right=0.0)
        up = np.clip(up, 0.0, 1.0)
        diff = target_sum - up.sum()
        for _ in range(3000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - up[i]) if diff > 0 else max(diff, -up[i])
            up[i] += add
            diff = target_sum - up.sum()
        candidates.append(up)

    # 2. a smooth “tent’’ shape (often a good starting point)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    tent = 1.0 - np.abs(xs - 1.0)
    tent = np.clip(tent, 0.0, 1.0)
    diff = target_sum - tent.sum()
    for _ in range(3000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - tent[i]) if diff > 0 else max(diff, -tent[i])
        tent[i] += add
        diff = target_sum - tent.sum()
    candidates.append(tent)

    # 3. a random feasible vector

    if best_h is not None and TIME() < budget_s * 0.9:

    while TIME() < budget_s * 0.98:

        h_tmp, c_tmp = hill_climb_targeted(h_start, rng, max_iters=3000, start_step=0.4)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ---- final clean‑up (exact feasibility) -----------------------------
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = target_sum - best_h.sum()
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()
        best_c = compute_c5(best_h)

    # ---- optional up‑sampling one more level if time permits ------------
    if best_h is not None and TIME() < budget_s * 0.99 and n * 2 <= 200:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        # sum is already correct after repetition
        h_up, c_up = hill_climb_targeted(h_up, rng, max_iters=4000, start_step=0.3)
        # a final light SLSQP polish
        betas = [200.0]
        h_up, c_up = slsqp_refine(h_up, rng, n_up, betas, max_iter=300)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    return best_h.tolist(), float(best_c), int(n)
