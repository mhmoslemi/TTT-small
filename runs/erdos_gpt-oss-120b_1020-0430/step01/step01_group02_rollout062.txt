import time, math, numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    n=len(h)
    dx=2.0/n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _adjust_sum(x, target):
    # similar to prior but maybe slower; use algorithm
    # Clip
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Simple redistribution:
    # if need to add mass: add equally to entries < 1
    # else subtract equally from entries >0
    # iterate few times
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
            if not np.any(mask):
                break
            per = diff / mask.sum()
            add = np.minimum(per, 1.0 - x[mask])
            x[mask] += add
        else:
            mask = x > 0.0
            if not np.any(mask):
                break
            per = diff / mask.sum()
            sub = np.maximum(per, -x[mask])
            x[mask] += sub
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h, c5, deadline, rng):
    n = len(h)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5

def _local_optimize_slsqp(h, target, rng):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    # objective: max correlation times dx
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * (2.0 / n)
    # start from h, perturb slightly
    x0 = h + rng.normal(scale=0.02, size=n)
    x0 = np.clip(x0, 0.0, 1.0)
    x0 = _adjust_sum(x0, target)
    res = minimize(obj, x0, method="SLSQP", bounds=bounds,
                   constraints=[eq_cons], options={"maxiter": 150, "ftol": 1e-9, "disp": False})
    if res.success:
        cand = _adjust_sum(np.clip(res.x, 0.0, 1.0), target)
        return cand, _evaluate_c5(cand)
    else:
        return h, _evaluate_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:

import time, math
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _adjust_sum(x, target, max_iter=10, eps=1e-12):
    """
    Project x onto the hyper‑plane sum(x)=target while respecting 0≤x≤1.
    Simple redistribution loop; sufficient for the modest dimensions used.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < eps:
        return x
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        per = diff / mask.sum()
        if diff > 0:
            add = np.minimum(per, 1.0 - x[mask])
            x[mask] += add
        else:
            sub = np.maximum(per, -x[mask])
            x[mask] += sub
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)


def _hill_climb(h, c5, deadline, rng):
    """Greedy pairwise mass‑exchange hill climbing until time expires."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5


def _local_slsqp(h, target, rng):
    """A short SLSQP refinement from a feasible point."""
    n = h.shape[0]
    bounds = [(0.0, 1.0)] * n
    eq = {"type": "eq", "fun": lambda x: x.sum() - target}
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    x0 = h + rng.normal(scale=0.02, size=n)
    x0 = np.clip(x0, 0.0, 1.0)
    x0 = _adjust_sum(x0, target)

    res = minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq],
        options={"maxiter": 150, "ftol": 1e-9, "disp": False},
    )
    if res.success:
        cand = _adjust_sum(np.clip(res.x, 0.0, 1.0), target)
        return cand, _evaluate_c5(cand)
    return h, _evaluate_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low Erdős C₅ bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initial vector (use the provided one if available)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.asarray(init, dtype=np.float64)
        n = h.size
    else:
        n = 45
        target = n / 2.0
        h = rng.random(n)
        h = h * (target / h.sum())
        h = _adjust_sum(h, target)

    target = n / 2.0
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing exploration (≈85 % of the time budget)
    # ------------------------------------------------------------------
    total = budget_s
    sa_deadline = start + total * 0.85
    cur_h, cur_c5 = best_h.copy(), best_c5

    while time.time() < sa_deadline:
        # linear schedules for noise size (σ) and temperature (T)
        elapsed = time.time() - start
        prog = elapsed / total
        sigma = max(0.05, 0.30 * (1.0 - prog))
        T = max(1e-4, 0.020 * (1.0 - prog))

        # propose a noisy neighbour
        cand = cur_h + rng.normal(scale=sigma, size=n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _adjust_sum(cand, target)

        cand_c5 = _evaluate_c5(cand)
        dE = cand_c5 - cur_c5
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
                # short SLSQP refinement

    final_deadline = start + total - 0.2

    # ------------------------------------------------------------------

    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
