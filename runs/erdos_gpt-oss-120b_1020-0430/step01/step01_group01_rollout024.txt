def _gd_optimize(h0, target, max_iter=1200, step0=0.5, decay=0.996):
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for t in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        # find subset of shifts close to max
        mask = corr >= max_val - 1e-12
        idxs = np.where(mask)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n - 1)
            grad += _gradient_for_shift(h, shift)
        grad /= len(idxs)
        h = h - step * grad
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Objective evaluation
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Gradient of the correlation for a *single* shift (the exact sub‑gradient)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Projected (sub)gradient descent using the *average* gradient over all
# shifts attaining the maximal correlation.
# ----------------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 800,
                 step0: float = 0.5,
                 decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        mask = corr >= max_val - 1e-12
        idxs = np.where(mask)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            shift = int(idx) - (n - 1)
            grad += _gradient_for_shift(h, shift)
        grad /= float(len(idxs))
        h = h - step * grad
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# Smooth max (used only inside the optional SLSQP refinement)
# ----------------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0,
                    max_iter: int = 1000) -> np.ndarray:
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# ----------------------------------------------------------------------
# Simple local pairwise exchange search (preserves the sum exactly)
# ----------------------------------------------------------------------
def _local_search(h: np.ndarray, target: float, dx: float,
                  max_steps: int = 200,
                  rng: np.random.RandomState = None) -> (np.ndarray, float):
    if rng is None:
        rng = np.random.RandomState()
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)
    n = h.shape[0]
    for _ in range(max_steps):
        i, j = rng.choice(n, size=2, replace=False)
        delta = (rng.rand() - 0.5) * 0.02  # +-0.01
        hi, hj = best_h[i], best_h[j]
        low = max(-hi, -(1.0 - hj))
        high = min(1.0 - hi, hj)
        delta = np.clip(delta, low, high)
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        val = _compute_c5(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val


# ----------------------------------------------------------------------
# Upscaling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upscale(vec: np.ndarray, new_n: int) -> np.ndarray:
    if len(vec) == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, vec)
