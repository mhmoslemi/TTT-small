def gradient_logsumexp(h, alpha):

    g = 1 - h
    conv = np.correlate(h, g, mode='full')

    scaled = alpha * conv

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def project_onto_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of x onto {0<=y<=1, sum(y)=target}."""
    # solve y_i = clip(x_i - lam,0,1) with sum(y)=target via bisection on lam
    lo, hi = -10.0, 10.0

    def s(lam):
        return np.clip(x - lam, 0.0, 1.0).sum()

    # expand bounds until they bracket the target
    while s(lo) < target:
        lo -= 10.0
    while s(hi) > target:
        hi += 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if s(mid) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, re‑normalised."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def generate_triangular(n: int) -> np.ndarray:

    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_feasible(n: int) -> np.ndarray:
    """Random vector in [0,1]^n with sum exactly n/2."""
    a = np.random.rand(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def gradient_logsumexp(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of (1/alpha) log‑sum‑exp(alpha*correlation) w.r.t. h."""
    n = h.shape[0]
    g = 1.0 - h
    conv = np.correlate(h, g, mode="full")
    scaled = alpha * conv
    mx = scaled.max()
    exp_scaled = np.exp(scaled - mx)
    w = exp_scaled / exp_scaled.sum()  # weights for each shift
    grad = np.zeros_like(h)

    # iterate over all possible shifts s = -(n-1)…(n-1)
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        if ws == 0.0:
            continue
        if s >= 0:
            # (1 - h[i+s]) term
            grad[: n - s] += ws * (1.0 - h[s:])
            # -h[i] term (affects index i+s)
            grad[s:] -= ws * h[: n - s]
        else:
            s_abs = -s
            grad[s_abs:] += ws * (1.0 - h[: n - s_abs])
            grad[: n - s_abs] -= ws * h[s_abs:]
    return grad


def optimize_h(start_h: np.ndarray, target: float, deadline: float,
               max_iters: int = 8000, alpha_max: float = 50.0):
    """Projected gradient descent on the smooth max‑overlap objective."""
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        p = it / max_iters
        alpha = 5.0 + (alpha_max - 5.0) * p
        lr = 0.5 * (1.0 - p) + 0.02 * p
        grad = gradient_logsumexp(h, alpha)
        h_new = project_onto_box_sum(h - lr * grad, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        h = h_new

        # occasional random jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.25:
            jitter = np.random.uniform(-0.05, 0.05, size=n)
            h = np.clip(best_h + jitter, 0.0, 1.0)
            h = project_onto_box_sum(h, target)

    return best_h, best_c5


def local_search(h: np.ndarray, target: float, deadline: float,
                 steps: int = 12000):
    """Simple mass‑transfer hill climbing."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while steps and time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            steps -= 1
            continue
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0.0:
            steps -= 1
            continue
        delta = np.random.uniform(0.0, max_transfer)
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        c5 = compute_c5(new_h)
        if c5 < best_c5:
            best_h, best_c5 = new_h, c5
        steps -= 1
    return best_h, best_c5


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    # Choose a moderate resolution; 200 points is a good trade‑off
    n = 200
    target_sum = n / 2.0

    # Build a diverse pool of starting vectors
    starts = []

    # 1. Parent construction, up‑sampled if needed
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.shape[0] != n:
            starts.append(upsample(parent, n))
        else:
            starts.append(parent.copy())
    except NameError:
        parent = None

    # 2. Classic triangular shape
    starts.append(generate_triangular(n))

    # 3. A few random feasible vectors
    for _ in range(4):
        starts.append(random_feasible(n))

    # 4. Perturbations of the triangular shape
    for _ in range(2):
        base = generate_triangular(n)
        pert = base + np.random.uniform(-0.12, 0.12, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = project_onto_box_sum(pert, target_sum)
        starts.append(pert)

    best_h = None
    best_c5 = np.inf

    # Optimise each start sequentially, respecting the time budget
    for s in starts:
        if time.time() >= deadline:
            break
        h_opt, c5_opt = optimize_h(s, target_sum, deadline)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final refinement with a simple hill‑climber if time permits
    if best_h is not None and time.time() < deadline:
        best_h, best_c5 = local_search(best_h, target_sum, deadline)

    # Final safety projection and C5 recomputation
    best_h = project_onto_box_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
