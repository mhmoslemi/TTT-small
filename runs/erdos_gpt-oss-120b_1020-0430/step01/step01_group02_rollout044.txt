def _project_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    # water-filling algorithm
    # We'll use loop
    diff = target - x.sum()
    # tolerance
    eps = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        # free indices where x not at bound in direction of diff
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            # cannot adjust
            break
        n_free = mask.sum()
        add = diff / n_free
        # apply with clipping
        before = x[mask].copy()
        if diff > 0:
            # add but not exceed 1
            inc = np.minimum(add, 1.0 - before)
        else:
            inc = np.maximum(add, -before)  # add negative
        x[mask] += inc
        diff = target - x.sum()
    # final clip
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """Project x onto [0,1]^n while fixing sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        # indices that can still move in the needed direction
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        free = np.where(mask)[0]
        add = diff / free.size
        # apply but respect bounds
        if diff > 0:
            inc = np.minimum(add, 1.0 - x[free])
        else:
            inc = np.maximum(add, -x[free])
        x[free] += inc
        diff = target - x.sum()
    # final safety clip
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h: np.ndarray, cur_score: float, deadline: float, rng: np.random.Generator) -> (np.ndarray, float):
    """Simple pairwise mass‑exchange hill climbing with optional uphill acceptance."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # direction: move from i to j or opposite
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        # small random step
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_score = _evaluate_c5(h_new)
        if new_score + 1e-12 < cur_score:
            h, cur_score = h_new, new_score
    return h, cur_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine dimension and initialise population
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        n = int(len(init_vec))
        h0 = np.asarray(init_vec, dtype=np.float64)
    else:
        n = 45
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = _adjust_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    # population size (moderate)
    pop_size = max(20, n // 2)
    pop = np.empty((pop_size, n), dtype=np.float64)
    scores = np.empty(pop_size, dtype=np.float64)

    # first individual = best found so far
    pop[0] = best_h
    scores[0] = best_c5

    # fill rest randomly
    for i in range(1, pop_size):
        x = rng.random(n)
        x = _adjust_sum(x, target_sum)
        pop[i] = x
        scores[i] = _evaluate_c5(x)

    # record current global best
    idx = int(np.argmin(scores))
    best_h = pop[idx].copy()
    best_c5 = scores[idx]

    # ------------------------------------------------------------------
    # 2. Differential Evolution loop
    # ------------------------------------------------------------------
    F = 0.8          # mutation factor
    CR = 0.7         # crossover probability
    # run until near budget end; keep a small margin for final hill climbing
    while time.time() - start < budget_s - 5.0:
        for i in range(pop_size):
            # mutation: select three distinct indices not equal to i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])

            # crossover
            jrand = rng.integers(n)
            cross_mask = rng.random(n) < CR
            cross_mask[jrand] = True
            trial = np.where(cross_mask, mutant, pop[i])

            # enforce bounds & sum constraint
            trial = _adjust_sum(trial, target_sum)

            trial_score = _evaluate_c5(trial)

            # selection
            if trial_score + 1e-12 < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_c5:
                    best_c5 = trial_score
                    best_h = trial.copy()

        # occasional diversification: replace worst with fresh random if stagnation
        if np.random.rand() < 0.02:
            worst = int(np.argmax(scores))
            x = rng.random(n)
            x = _adjust_sum(x, target_sum)
            pop[worst] = x
            scores[worst] = _evaluate_c5(x)

    # ------------------------------------------------------------------
    # 3. Fine‑grained hill climbing on the best individual
    # ------------------------------------------------------------------
    deadline = start + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, deadline, rng)

    # final safeguard
    best_h = _adjust_sum(best_h, target_sum)
    best_c5 = float(_evaluate_c5(best_h))

    return best_h, best_c5, n
