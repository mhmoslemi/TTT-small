import time
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target, max_iter=200):
    """
    Project vector x onto the hyper‑plane sum(x)=target while respecting 0≤x≤1.
    Uses a water‑filling style binary search for efficiency.
    """
    x = np.clip(x, 0.0, 1.0)
    lo = -np.max(x)          # makes all entries zero
    hi = 1.0 - np.min(x)     # makes all entries one
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if numerical error remains
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            # distribute proportionally to remaining capacity
            cap = np.where(diff > 0, 1.0 - y[mask], y[mask])
            per = diff / cap.sum()
            y[mask] += per * cap
            y = np.clip(y, 0.0, 1.0)
    return y

def _subgradient_step(h, target, eps, lr):
    """
    One projected sub‑gradient step for the smoothed max‑overlap objective.
    Returns the new vector and the raw (unscaled) maximal correlation.
    """
    n = len(h)
    # raw correlation values (not multiplied by dx)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()

    # soft‑max weights (numerically stable)
    shift = corr.max()
    w = np.exp((corr - shift) / eps)
    w /= w.sum()                     # sum to 1

    grad = np.ones(n)                # contribution from the linear sum term
    offset = n - 1                   # index of shift 0 in corr / w

    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - offset            # actual integer shift
        if s >= 0:
            if s < n:
                # subtract h[i+s] for i where i+s is inside the array
                grad[:n - s] -= weight * h[s:]
                # subtract h[i-s] for i where i-s is inside the array
                grad[s:]   -= weight * h[:n - s]
        else:
            t = -s
            # i+s = i-t inside for i >= t
            grad[t:]   -= weight * h[:n - t]
            # i-s = i+t inside for i <= n-1-t
            grad[:n - t] -= weight * h[t:]

    # gradient descent step
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _adjust_sum(h_new, target)
    return h_new, max_corr

def _hill_climb(h, c5, deadline, rng):
    """Simple random pairwise mass‑exchange hill‑climbing."""
    n = len(h)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h = np.asarray(init_vec, dtype=np.float64)
        n = h.size
    else:
        n = 80                         # a modest resolution when no parent is given
        h = np.full(n, 0.5)            # start from the uniform distribution

    target = n / 2.0
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
    best_h = h.copy()
    best_c5 = float(_evaluate_c5(best_h))

    # ------------------------------------------------------------------
    # 2. Smoothed sub‑gradient descent with continuation on epsilon
    # ------------------------------------------------------------------
    grad_deadline = start + budget_s * 0.85   # reserve ~15% for final hill‑climb
    eps_start, eps_end = 0.05, 0.004
    lr_start, lr_end = 0.5, 0.02

    it = 0
    while time.time() < grad_deadline:
        # linear progress fraction for schedules
        frac = (time.time() - start) / (grad_deadline - start)
        eps = eps_start * (eps_end / eps_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        h, cur_raw = _subgradient_step(h, target, eps, lr)
        cur_c5 = float(cur_raw * (2.0 / n))

        if cur_c5 + 1e-12 < best_c5:
            best_h, best_c5 = h.copy(), cur_c5

        # occasional random jitter to escape shallow plateaux
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=0.03, size=n)
            cand = h + jitter
            cand = np.clip(cand, 0.0, 1.0)
            cand = _adjust_sum(cand, target)
            cand_c5 = _evaluate_c5(cand)
            if cand_c5 < best_c5:
                h, best_h, best_c5 = cand, cand.copy(), cand_c5
        it += 1

    # ------------------------------------------------------------------
    # 3. Fine‑grained random hill climbing (as a final polish)
    # ------------------------------------------------------------------
    hill_deadline = start + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, hill_deadline, rng)

    # ------------------------------------------------------------------
    # 4. Return the final feasible solution
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
