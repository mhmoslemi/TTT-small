def compute_gradient(h, alpha):
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    # precompute exponentials with stabilization
    exp_alpha_conv = np.exp(alpha * conv - np.max(alpha * conv))
    w = exp_alpha_conv / exp_alpha_conv.sum()
    offset = n - 1
    # term1 (sum of w over valid ℓ)
    # compute using cumulative sum
    c = np.cumsum(np.concatenate([[0.0], w]))
    idx_start = offset - np.arange(n)  # shape n
    idx_end = offset + n - 1 - np.arange(n)
    term1 = c[idx_end + 1] - c[idx_start]
    # term2 and term3 via loop
    term2 = np.zeros(n, dtype=float)
    term3 = np.zeros(n, dtype=float)
    for ell in range(- (n-1), n):
        w_ell = w[ell + offset]
        if w_ell == 0.0:
            continue
        if ell >= 0:
            # term2: i from 0 to n-ell-1 includes h[i+ell]
            term2[:n-ell] += w_ell * h[ell:]
            # term3: i from ell to n-1 includes h[i-ell] (since i-ell >=0)
            term3[ell:] += w_ell * h[:n-ell]
        else:
            ell_neg = -ell
            term2[ell_neg:] += w_ell * h[:n-ell_neg]  # because i+ell = i - ell_neg
            term3[:n-ell_neg] += w_ell * h[ell_neg:] # i-ell = i+ell_neg
    grad = term1 - term2 - term3
    return conv, w, grad

def project_to_feasible(h):
    # Clip first
    h = np.clip(h, 0, 1)
    # Compute current sum
    s = h.sum()
    target = len(h)/2.
    # While sum not equal within tolerance:
    # Use iterative water-filling: if s < target, try to increase some entries that are below 1.
    # If s > target, decrease entries that are above 0.
    # Distribute diff uniformly across eligible entries; if any hitting bound, repeat.
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # eligible indices
    for _ in range(10):
        if diff > 0:
            # indices where h<1
            mask = h < 1 - 1e-12
        else:
            mask = h > 1e-12
        m = mask.sum()
        if m == 0:
            break
        incr = diff / m
        h[mask] += incr
        # clip again
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def softmax_obj_and_grad(h, alpha, target_sum, penalty):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    obj = (max_scaled + np.log(sum_exp)) / alpha
    diff_sum = h.sum() - target_sum
    obj += penalty * diff_sum * diff_sum

    # compute probabilities
    w = exp_scaled / sum_exp   # length 2n-1

    # compute gradient (softmax part)
    offset = n - 1
    # term1 using cumulative sum
    c = np.cumsum(np.concatenate([[0.0], w]))
    i = np.arange(n)
    idx_start = offset - i
    idx_end = offset + n - 1 - i
    term1 = c[idx_end + 1] - c[idx_start]

    term2 = np.zeros(n)
    term3 = np.zeros(n)
    # loop over ell
    # to improve speed, compute w values once
    for ell in range(- (n-1), n):
        w_ell = w[ell + offset]
        if w_ell == 0.0:
            continue
        if ell >= 0:
            term2[:n-ell] += w_ell * h[ell:]
            term3[ell:] += w_ell * h[:n-ell]
        else:
            ell_neg = -ell
            term2[ell_neg:] += w_ell * h[:n-ell_neg]
            term3[:n-ell_neg] += w_ell * h[ell_neg:]
    grad_soft = term1 - term2 - term3

    grad = grad_soft + 2.0 * penalty * diff_sum * np.ones_like(h)
    return obj, grad

import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * (2.0 / n))


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalized."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible start (values in [0,1] and sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, keeping the integral exact."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def project_to_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy the exact integral."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        inc = diff / cnt
        h[mask] += inc
        h = np.clip(h, 0.0, 1.0)
    return h


# ------------------------------------------------------------
# Smoothed objective with analytic gradient
# ------------------------------------------------------------
def softmax_obj_and_grad(h: np.ndarray, alpha: float, target_sum: float, penalty: float):
    """
    Returns (objective, gradient) of the log‑sum‑exp approximation of C5
    plus a quadratic penalty for the sum constraint.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")          # length 2n‑1
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)            # stabilised
    sum_exp = exp_scaled.sum()
    # soft‑max value = (max + log sum) / alpha
    obj = (max_scaled + np.log(sum_exp)) / alpha

    # quadratic penalty for the integral constraint
    diff_sum = h.sum() - target_sum
    obj += penalty * diff_sum * diff_sum

    # probabilities for the gradient (softmax weights)
    w = exp_scaled / sum_exp          # shape (2n‑1,)
    offset = n - 1

    # term1: sum of w over the shifts that keep i+ℓ inside the array
    c = np.cumsum(np.concatenate([[0.0], w]))
    i = np.arange(n)
    idx_start = offset - i
    idx_end = offset + n - 1 - i
    term1 = c[idx_end + 1] - c[idx_start]

    # term2 and term3: weighted sums involving h
    term2 = np.zeros(n, dtype=float)
    term3 = np.zeros(n, dtype=float)

    for ell in range(-(n - 1), n):
        w_ell = w[ell + offset]
        if w_ell == 0.0:
            continue
        if ell >= 0:
            term2[: n - ell] += w_ell * h[ell:]
            term3[ell:] += w_ell * h[: n - ell]
        else:
            ell_neg = -ell
            term2[ell_neg:] += w_ell * h[: n - ell_neg]
            term3[: n - ell_neg] += w_ell * h[ell_neg:]

    grad_soft = term1 - term2 - term3
    grad = grad_soft + 2.0 * penalty * diff_sum * np.ones_like(h)
    return obj, grad

# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    # parent solution, if provided by the harness
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    # candidate resolutions (include parent size if different)
    candidate_ns = [150, 180, 200, 240, 300]
    if parent_n and parent_n not in candidate_ns:
        candidate_ns.insert(0, parent_n)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # -----------------------------------------------------------------
    # Exploration over resolutions and starting points
    # -----------------------------------------------------------------
    for n in candidate_ns:
        if time.time() > time_limit:
            break
        target_sum = n / 2.0

        starts = []

        # upsampled parent (if any)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # classic triangular baseline
        starts.append(generate_triangular(n))

        # a few random feasible vectors
        for _ in range(3):
            starts.append(random_start(n, rng))

        # perturbed triangular shapes
        tri = generate_triangular(n)
        for _ in range(2):
            pert = tri + rng.uniform(-0.07, 0.07, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            diff = target_sum - pert.sum()
            pert += diff / n
            pert = np.clip(pert, 0.0, 1.0)
            starts.append(pert)

        # -----------------------------------------------------------------
        # Optimise each start via a multi‑stage smoothed‑max L‑BFGS schedule
        # -----------------------------------------------------------------
        for start in starts:
            if time.time() > time_limit:
                break
            h = project_to_feasible(start)
            bounds = [(0.0, 1.0)] * n

            for alpha in (5.0, 10.0, 20.0, 40.0, 80.0):
                if time.time() > time_limit:
                    break
                # penalty for sum‑constraint; kept modest
                penalty = 12.0

                res = optimize.minimize(
                    lambda x: softmax_obj_and_grad(x, alpha, target_sum, penalty),
                    h,
                    method="L-BFGS-B",
                    jac=True,
                    bounds=bounds,
                    options={"maxiter": 150, "ftol": 1e-9, "disp": False},

            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-9:
                best_h = h.copy()
                best_c5 = c5_val
                best_n = n

    # -----------------------------------------------------------------
    # Local hill‑climbing refinement on the best solution found
    # -----------------------------------------------------------------
    if best_h is not None and time.time() < time_limit:
        h = best_h.copy()
        current_c5 = best_c5
        n = h.shape[0]
        while time.time() < time_limit:
            i, j = rng.choice(n, size=2, replace=False)
            # decide direction of transfer
            if rng.random() < 0.5:
                # move from i to j
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta * 0.6
                new_i = h[i] - delta
                new_j = h[j] + delta
            else:
                # move from j to i
                max_delta = min(h[j], 1.0 - h[i])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta * 0.6
                new_i = h[i] + delta
                new_j = h[j] - delta
            h_new = h.copy()
            h_new[i] = new_i
            h_new[j] = new_j
            new_c5 = compute_c5(h_new)
            if new_c5 + 1e-12 < current_c5:
                h = h_new
                current_c5 = new_c5
        best_h = h
        best_c5 = current_c5
        best_n = n

    # -----------------------------------------------------------------
    # Final safety projection and exact C5 recomputation
    # -----------------------------------------------------------------
    if best_h is None:
        # fallback: simple binary block (always feasible)
        n = 80
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_h = block
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = project_to_feasible(best_h)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
