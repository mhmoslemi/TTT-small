def softmax_grad(x):
    n = x.size
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    p = exp_bc / sum_exp  # shape 2n-1

def slsqp_optimize_grad(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    cons = [{'type': 'eq', 'fun': sum_eq_constraint, 'jac': sum_eq_jac}]
    res = minimize(softmax_obj, h0, method='SLSQP', jac=softmax_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

        for _ in range(2000):

                break
            i = np.random.randint(0, n)

                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the optimizers (set before each call)
CURRENT_BETA = 1.0
CURRENT_TARGET = 1.0
RNG = None  # will be set in run()
# ----------------------------------------------------------------------


def compute_c5(h):
    """Exact C5 for a discrete step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def sum_eq_jac(x):
    """Jacobian of the equality constraint."""
    return np.ones_like(x)


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    global CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA


def softmax_grad(x):
    """Gradient of the soft‑max surrogate."""
    global CURRENT_BETA
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    p = exp_bc / np.sum(exp_bc)               # soft‑max weights, length 2n‑1
    grad = np.zeros_like(x)

    # iterate over all possible shifts s = -(n-1) … (n-1)
    for s in range(-(n - 1), n):
        ps = p[s + n - 1]
        if s >= 0:
            k = s
            # contribution from (1 - x_{i+k})
            grad[: n - k] += ps * (1.0 - x[k:])
            # contribution from -x_{i-k}
            grad[k:] += ps * (-x[: n - k])
        else:
            k = -s
            # i+s = i-k  (i ≥ k)
            grad[k:] += ps * (1.0 - x[: n - k])
            # i-s = i+k  (i ≤ n-1-k)
            grad[: n - k] += ps * (-x[k:])
    return grad


def adjust_sum(x, target):
    """Redistribute mass so that Σ x_i = target while staying in [0,1]."""
    diff = target - x.sum()
    n = x.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = RNG.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x


def random_feasible_vector(n):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    target = n / 2.0
    # start from the flat solution and perform random transfers
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = RNG.integers(0, n)
        j = RNG.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (RNG.random() - 0.5) * 2.0 * max_eps * 0.4
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return adjust_sum(h, target)


def perturb_h(h, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = RNG.integers(0, n)
    j = RNG.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (RNG.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h_start, max_iters=1000, start_step=0.3):
    """Simple deterministic hill‑climb using perturb_h."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # decay step size
        if (it + 1) % (max_iters // 6 + 1) == 0:
            step = max(step * 0.65, 1e-4)
    return best_h, best_c


def slsqp_optimize_grad(h0, beta, max_iter=300):
    """Run a single SLSQP optimisation using the soft‑max surrogate and its gradient."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_eq_constraint, "jac": sum_eq_jac}]

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        jac=softmax_grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = adjust_sum(x, CURRENT_TARGET)
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    global RNG
    RNG = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s

    # ------------------------------------------------------------------
    # Initialise from the supplied parent solution (if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            h0 = adjust_sum(h0, target)
    else:
        n = 90
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 0 – smooth‑max local refinement (ascending beta ladder)
    # ------------------------------------------------------------------
    beta_ladder = [0.5, 1.0, 3.0, 10.0, 30.0, 80.0]
    for beta in beta_ladder:
        if time.time() - start_time > time_limit * 0.70:
            break
        try:
            h_opt = slsqp_optimize_grad(best_h, beta=beta, max_iter=500)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
        except Exception:
            continue

    # ------------------------------------------------------------------
    # Phase 1 – random restarts with brief gradient polishing
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit * 0.85:
        h_start = random_feasible_vector(n)
        # short polishing with the first three betas
        for beta in beta_ladder[:3]:
            try:
                h_start = slsqp_optimize_grad(h_start, beta=beta, max_iter=200)
            except Exception:
                pass
        # a few hill‑climb steps to exploit the exact max‑overlap landscape
        h_start, c_start = hill_climb(h_start, max_iters=400, start_step=0.4)
        if c_start < best_c:
            best_h, best_c = h_start, c_start

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to 2× resolution (if time permits)
    # ------------------------------------------------------------------
    if n * 2 <= 300 and time.time() - start_time < time_limit * 0.95:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        target_up = n_up / 2.0
        diff = target_up - h_up.sum()
        if abs(diff) > 1e-12:
            h_up = adjust_sum(h_up, target_up)

        # run the beta ladder on the finer grid
        for beta in beta_ladder[2:]:
            if time.time() - start_time > time_limit:
                break
            try:
                h_up = slsqp_optimize_grad(h_up, beta=beta, max_iter=400)
            except Exception:
                pass

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ------------------------------------------------------------------
    # Final clean‑up (ensure exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = adjust_sum(best_h, target)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
