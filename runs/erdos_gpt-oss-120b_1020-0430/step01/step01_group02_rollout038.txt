def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 200  # maybe try 200 (even)
    # ensure n is even
    if n % 2 == 1:
        n += 1
    # initialize binary h random half 1s
    h = np.zeros(n, dtype=np.int8)

    h[ones_positions] = 1
    # compute initial
    current_max = evaluate_max_corr(h)
    best_h = h.copy()
    best_max = current_max
    # schedule parameters
    # Let's set temperature based on difference between current max and lower bound (maybe best max).
    # We'll use exponential cooling: T = T0 * alpha^{iteration}
    T0 = n  # start high enough e.g., n/2
    alpha = 0.9995   # cooling per iteration
    T = T0
    iteration = 0
    # maybe also implement restarts after some time
    # while time left > small
    while time.time() - start < budget_s - 1.0:
        # random swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # ensure values differ
        if h[i] == h[j]:
            continue
        # perform swap
        h[i], h[j] = h[j], h[i]
        new_max = evaluate_max_corr(h)
        delta = new_max - current_max
        accept = False
        if delta < 0:
            accept = True
        else:

            prob = np.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # cooling
        iteration += 1
        T = T0 * (alpha ** iteration)
        # occasional random restart
        if iteration % 10000 == 0:
            # if stuck, random shuffle a fraction
            # reinitialize random shuffle of a small portion maybe
            # but we keep best
            # optionally reinitialize entire solution if no improvement in some time.
            pass

    c5 = best_max * 2.0 / n

import cvxpy as cp

def solve_binary_milp(n, time_limit):
    # h binary variable shape n
    h = cp.Variable(n, boolean=True)
    # total ones
    constraints = [cp.sum(h) == n//2]  # ensures exactly half ones; n even
    max_overlap = cp.Variable(integer=True, nonneg=True)

    # for each shift s=1..n-1
    for s in range(1, n):
        length = n - s
        # create y variables for each i: y_i = h_i * h_{i+s}
        # shape length
        y = cp.Variable(length, boolean=True)
        # constraints linking y and h
        # y_i <= h_i
        constraints += [y <= h[:length]]
        constraints += [y <= h[s:]]
        constraints += [y >= h[:length] + h[s:] - 1]
        # compute c_s = sum_i h_i - sum_i y_i
        cs = cp.sum(h[:length]) - cp.sum(y)
        constraints += [cs <= max_overlap]
    # Objective: minimize max_overlap
    obj = cp.Minimize(max_overlap)
    prob = cp.Problem(obj, constraints)
    # Solve with time limit
    prob.solve(solver=cp.CBC, time_limit=time_limit)
    # if solution found
    if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
        h_solution = np.round(h.value).astype(np.int8)  # boolean rounding
        max_overlap_val = int(np.round(max_overlap.value))
        return h_solution, max_overlap_val, prob.status
    else:
        return None, None, prob.status

def solve_binary_via_mip(n, time_budget):
    # allocate time budget for MIP maybe 30 seconds
    # Build model.
    h = cp.Variable(n, boolean=True)

import time
import numpy as np

def _max_corr_and_shift(h):
    # h binary 0/1 as numpy array
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = int(corr.max())
    max_idx = int(np.argmax(corr))
    shift = max_idx - (len(h) - 1)
    return max_val, shift, corr

def _solve_mip_binary(n, time_limit):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    # binary var h
    h = cp.Variable(n, boolean=True)
    # max overlap variable
    t = cp.Variable(integer=True, nonneg=True)
    constraints = [cp.sum(h) == n // 2]
    for s in range(1, n):
        L = n - s
        # linking vars
        y = cp.Variable(L, boolean=True)
        constraints.append(y <= h[:L])
        constraints.append(y <= h[s:])
        constraints.append(y >= h[:L] + h[s:] - 1)
        cs = cp.sum(h[:L]) - cp.sum(y)
        constraints.append(cs <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        # some solvers use different kw name
        try:
            prob.solve(solver=cp.CBC, max_seconds=time_limit, warm_start=True, verbose=False)
        except Exception:
            return None, None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE, cp.FEASIBLE_INACCURATE):
        return None, None
    h_val = np.rint(h.value).astype(np.int8)
    # enforce exact half ones
    current = int(h_val.sum())
    target = n // 2
    if current != target:
        diff = target - current
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            if zeros.size >= diff:
                h_val[zeros[:diff]] = 1
        else:
            ones = np.where(h_val == 1)[0]
            if ones.size >= -diff:
                h_val[ones[: -diff]] = 0
    max_corr = int(np.max(np.correlate(h_val, 1 - h_val, mode='full')))
    return h_val, max_corr

def _hill_climb_binary(n, rng, time_limit, start_h=None):
    # simple random swap hill climbing, try to improve max_corr
    if start_h is None:
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1
    else:
        h = start_h.copy()

    best_corr = max_corr
    deadline = time.time() + time_limit

    while time.time() < deadline:
        # pick random differing indices
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if h[i] == h[j]:

        if cand_corr < max_corr:
            max_corr = cand_corr
            if cand_corr < best_corr:
                best_corr = cand_corr
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_corr

import time
import numpy as np

def _max_corr_and_shift(h):
    """Return max correlation value, the shift that attains it and the full vector."""
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = int(corr.max())
    max_idx = int(np.argmax(corr))
    shift = max_idx - (len(h) - 1)
    return max_val, shift, corr


def _solve_mip_binary(n, time_limit):
    """
    Solve the exact binary formulation with a MILP (CBC) if cvxpy is available.
    Returns a binary vector and its max‑overlap count, or (None, None) on failure.
    """
    try:
        import cvxpy as cp
    except Exception:                      # cvxpy not present
        return None, None

    # binary variables for the step function
    h = cp.Variable(n, boolean=True)
    # the maximal overlap (integer)
    t = cp.Variable(integer=True, nonneg=True)

    constraints = [cp.sum(h) == n // 2]   # exactly half ones

    # for each shift create auxiliary variables y_{i,s}=h_i*h_{i+s}
    for s in range(1, n):
        L = n - s
        y = cp.Variable(L, boolean=True)
        constraints += [y <= h[:L]]
        constraints += [y <= h[s:]]
        constraints += [y >= h[:L] + h[s:] - 1]
        # mismatch count for this shift
        cs = cp.sum(h[:L]) - cp.sum(y)
        constraints += [cs <= t]

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        try:                               # some installations use a different kw
            prob.solve(solver=cp.CBC, max_seconds=time_limit,
                       warm_start=True, verbose=False)
        except Exception:
            return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE,
                           cp.FEASIBLE, cp.FEASIBLE_INACCURATE):
        return None, None

    h_val = np.rint(h.value).astype(np.int8)

    # enforce exact half ones (tiny rounding issues may appear)
    target = n // 2
    cur = int(h_val.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            h_val[zeros[:diff]] = 1
        else:
            ones = np.where(h_val == 1)[0]
            h_val[ones[: -diff]] = 0

    max_corr = int(np.max(np.correlate(h_val, 1 - h_val, mode="full")))
    return h_val, max_corr


def _hill_climb_binary(n, rng, time_limit, start_h=None):
    """
    Simple random swap hill‑climbing for a binary vector.
    Tries to reduce the maximal overlap count.
    """
    if start_h is None:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, size=n // 2, replace=False)] = 1
    else:
        h = start_h.copy()

    best_h = h.copy()
    best_corr = int(np.max(np.correlate(h, 1 - h, mode="full")))
    deadline = time.time() + time_limit

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue

        h[i], h[j] = h[j], h[i]                     # perform swap
        cur_corr = int(np.max(np.correlate(h, 1 - h, mode="full")))

        if cur_corr < best_corr:
            best_corr = cur_corr
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]                 # revert

        # occasional small random perturbation to escape plateaus
        if rng.random() < 0.001:
            for _ in range(3):
                a = rng.integers(n)
                b = rng.integers(n - 1)
                if b >= a:
                    b += 1
                if h[a] != h[b]:
                    h[a], h[b] = h[b], h[a]

    return best_h, best_corr

    """
    Optimise a step function for the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)
    target_bound = 0.3808                     # we need to beat this

    n = 44                                    # even size, half ones ⇒ integer sum
    # ------------------------------------------------------------------
    # 1. Try an exact MILP solution (CBC) – usually finishes in seconds.
    # ------------------------------------------------------------------
    milp_time = min(300, max(0.0, budget_s - 200))
    h_bin, max_corr = _solve_mip_binary(n, milp_time)

        if c5 <= target_bound:               # success – optimal or good enough

        start_h = h_bin                       # use as warm start for local search
    else:
        start_h = None

    # ------------------------------------------------------------------
