import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper globals for the soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for the step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- feasibility utilities ----------
def make_feasible(h, rng):
    """Clip to [0,1] and enforce Σh = n/2 by random redistribution."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return make_feasible(h, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- SLSQP optimizer ----------
def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP run on the smooth surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny repair if sum drifted
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = np.array(x)
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- local deterministic improvement ----------
def improve_max_corr(h, step):
    """Push mass between paired indices of the shift that attains the current maximum."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # signed shift
    if shift >= 0:
        i_start, i_end = 0, n - shift
        for i in range(i_start, i_end):
            j = i + shift
            hi, hj = h[i], h[j]
            if hi > hj + 1e-12:
                eps = min(step, (hi - hj) / 2.0, hi, 1.0 - hj)
                if eps > 0:
                    h[i] -= eps
                    h[j] += eps
            elif hj > hi + 1e-12:
                eps = min(step, (hj - hi) / 2.0, hj, 1.0 - hi)
                if eps > 0:
                    h[j] -= eps
                    h[i] += eps
    else:
        shift = -shift
        i_start, i_end = shift, n
        for i in range(i_start, i_end):
            j = i - shift
            hi, hj = h[i], h[j]
            if hi > hj + 1e-12:
                eps = min(step, (hi - hj) / 2.0, hi, 1.0 - hj)
                if eps > 0:
                    h[i] -= eps
                    h[j] += eps
            elif hj > hi + 1e-12:
                eps = min(step, (hj - hi) / 2.0, hj, 1.0 - hi)
                if eps > 0:
                    h[j] -= eps
                    h[i] += eps
    return h


def local_optimize(h, max_iter=120, init_step=0.12, step_decay=0.85):
    """Iteratively reduce the max‑overlap by equalising the critical pair."""
    h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for _ in range(max_iter):
        h = improve_max_corr(h, step)
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
        else:
            step = max(step * step_decay, 1e-5)
            if step < 1e-6:
                break
    return h, best_c


def upsample_linear(h, factor):
    """Linear interpolation up‑sampling (preserves ordering)."""
    n = h.size
    n_up = n * factor
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xs_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
    return np.interp(xs_up, xs, h)
