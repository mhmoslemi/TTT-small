def subgradient_optimize(h, max_iter, step0, step_decay, time_func):
    n = len(h)
    best_h = h.copy()
    best_val = _c5_from_h(h)
    for it in range(1, max_iter+1):

            break

import time, numpy as np

def _project_to_feasible(h, n, tol=1e-9, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        free = mask.nonzero()[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final small adjustment
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        # distribute the residual greedily
        for i in range(n):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

    best_h = h.copy()

    for it in range(1, max_iter+1):

import time
import numpy as np
from scipy import optimize

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        for i in range(n):
            if resid > 0 and h[i] < 1 - 1e-12:
                add = min(resid, 1 - h[i])
                h[i] += add
                resid -= add
            elif resid < 0 and h[i] > 1e-12:
                sub = min(-resid, h[i])
                h[i] -= sub
                resid += sub
            if abs(resid) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """One SLSQP run (as in the reference solution)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = corr0.max()
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq(x):
        h = x[:n]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c

    def eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [{"type": "ineq", "fun": ineq}, {"type": "eq", "fun": eq}]
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from coarse to fine grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _subgradient_optimize(
    h: np.ndarray,
    max_iter: int,
    step0: float,
    step_decay: float,
    time_left,
) -> tuple:
    """Projected sub‑gradient descent on the max‑correlation objective."""
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(h)

    for it in range(1, max_iter + 1):
        if time_left() < 5.0:
            break
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr_full))
        k = max_idx - (n - 1)

        # gradient of the active correlation
        g = np.zeros_like(h)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            d = -k
            g[d:] += 1.0 - h[: n - d]

        h_new = h - step * g
        h_new = _project_to_feasible(h_new, n)

        val = _c5_from_h(h_new)
        if val < best_val:
            best_val, best_h = val, h_new.copy()
        h = h_new

    return best_h, best_val
