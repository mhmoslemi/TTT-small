def propose_move(h, rng, step_scale):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # direction: from i to j if rng.random() < 0.5
    if rng.random() < 0.5:
        donor, rec = i, j
    else:
        donor, rec = j, i
    max_eps = min(h[donor], 1.0 - h[rec])
    if max_eps <= 0:
        return None, None, None
    eps = rng.random() * max_eps * step_scale
    new_h = h.copy()
    new_h[donor] -= eps
    new_h[rec] += eps
    return new_h, donor, rec

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers for the soft‑max SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run a single SLSQP optimisation using the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- core problem utilities ----------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def random_feasible_vector(n, rng):
    """Generate a random feasible vector satisfying 0≤h≤1 and Σh=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two entries (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Greedy hill‑climb (accept only strictly improving moves)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def simulated_annealing(h0, rng, max_iters=4000, t0=0.02, tf=1e-5):
    """Simple geometric‑cooling annealer."""
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    if max_iters <= 1:
        factor = 1.0
    else:
        factor = (tf / t0) ** (1.0 / (max_iters - 1))
    T = t0
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        # random direction
        if rng.random() < 0.5:
            donor, rec = i, j
        else:
            donor, rec = j, i
        max_eps = min(h[donor], 1.0 - h[rec])
        if max_eps <= 0.0:
            T *= factor
            continue
        eps = rng.random() * max_eps * (T / t0)
        h_new = h.copy()
        h_new[donor] -= eps
        h_new[rec] += eps
        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
        T *= factor
    return best_h, best_c

        else:
            start_vec = random_feasible_vector(n, rng)
        trial_h, trial_c = simulated_annealing(start_vec, rng, max_iters=3000,
                                               t0=0.02, tf=1e-5)
        if trial_c < best_c:
            best_h, best_c = trial_h, trial_c
        # local refinement
        trial_h2, trial_c2 = hill_climb(best_h, rng, max_iters=600, start_step=0.2)
        if trial_c2 < best_c:
            best_h, best_c = trial_h2, trial_c2

    # ---- Phase 2 : up‑sample and repeat ----
    # repeat each element twice – preserves total mass exactly
    h_up = np.repeat(best_h, 2)
    n_up = h_up.size
    best_h, best_c = h_up, compute_c5(h_up)

    phase2_end = start_time + budget_s * 0.60
    while time.time() < phase2_end:
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n_up, rng)
        trial_h, trial_c = simulated_annealing(start_vec, rng,
                                               max_iters=4000,
                                               t0=0.015, tf=5e-6)
        if trial_c < best_c:
            best_h, best_c = trial_h, trial_c
        trial_h2, trial_c2 = hill_climb(best_h, rng, max_iters=800, start_step=0.15)
        if trial_c2 < best_c:
            best_h, best_c = trial_h2, trial_c2

    # ---- Optional third up‑sample if time permits ----
    if time.time() - start_time < budget_s * 0.85 and n_up * 2 <= 360:
        h_up2 = np.repeat(best_h, 2)
        n_up2 = h_up2.size
        best_h, best_c = h_up2, compute_c5(h_up2)
        phase3_end = start_time + budget_s * 0.95
        while time.time() < phase3_end:
            if rng.random() < 0.6:
                start_vec = perturb_h(best_h, rng, scale=0.2)
            else:
                start_vec = random_feasible_vector(n_up2, rng)
            trial_h, trial_c = simulated_annealing(start_vec, rng,
                                                   max_iters=3000,
                                                   t0=0.01, tf=2e-6)
            if trial_c < best_c:
                best_h, best_c = trial_h, trial_c
            trial_h2, trial_c2 = hill_climb(best_h, rng, max_iters=600, start_step=0.1)
            if trial_c2 < best_c:
                best_h, best_c = trial_h2, trial_c2
        n = n_up2
    else:
        n = n_up

    # ---- Phase 4 : final SLSQP refinement (soft‑max ladder) ----
    betas = [5.0, 20.0, 80.0, 200.0]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.99:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=500)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            continue

    # ---- final feasibility clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
