import numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_box_simplex(v, target):
    # use bisection with 60 iterations
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # softmax weights numerically stable
    z = beta * corr
    m = np.max(z)
    w = np.exp(z - m)
    w /= w.sum()
    # compute terms
    term1 = np.convolve(w[::-1], 1 - h, mode='full')
    term2 = np.convolve(w, h, mode='full')
    offset = n - 1
    grad = term1[offset:offset+n] - term2[offset:offset+n]
    return grad

def random_feasible_vector(n, rng):
    vec = rng.random(n)
    target = n / 2.0
    return project_box_simplex(vec, target)

    n = h.size

    step = init_step

    best_c = compute_c5(h)
    start = time.time()

            if c_val < best_c:

                best_h = h.copy()
            # decay step

import time
import numpy as np

def compute_c5(h):
    """Exact C₅ for discrete step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_simplex(v, target):
    """Project onto {0≤x_i≤1, Σx_i = target} via bisection on λ."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Draw a random vector in the feasible box-simplex."""
    v = rng.random(n)
    return project_box_simplex(v, n / 2.0)


def grad_and_max_corr(h, beta):
    """Gradient of smooth max‑overlap and the current max correlation."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")  # length 2n‑1
    z = beta * corr
    m = np.max(z)
    w = np.exp(z - m)
    w /= w.sum()
    # term1 = Σ w_s (1‑h_{j+s}), term2 = Σ w_s h_{j‑s}
    term1 = np.convolve(w[::-1], 1 - h, mode="full")
    term2 = np.convolve(w, h, mode="full")
    off = n - 1
    grad = term1[off : off + n] - term2[off : off + n]
    return grad, np.max(corr)


def gradient_descent(start_h, betas, step0, iters_per_beta, time_limit, rng):
    """Projected gradient descent on the smooth max‑overlap objective."""
    h = start_h.copy()

    step = step0
    best_h = h.copy()

        for _ in range(iters_per_beta):
            if time.time() - start > time_limit:
                return best_h, best_c
            grad, max_corr = grad_and_max_corr(h, beta)

            # occasional jitter to avoid plateaus
            if rng.random() < 0.005:
                h += rng.normal(scale=0.01, size=n)
                h = project_box_simplex(h, target)
            c_val = max_corr * (2.0 / n)
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()
            step *= 0.9996
    return best_h, best_c


def hill_climb(start_h, rng, max_iters=1200, scale=0.25):
    """Simple local search that swaps a small amount between two coordinates."""
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        h2 = h.copy()
        h2[i] -= eps
        h2[j] += eps
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2
        scale *= 0.9999
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initial feasible vector (use parent if provided)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_simplex(h0, n / 2.0)
    else:
        n = 90
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Core optimisation loop (many short restarts)
    # ------------------------------------------------------------------
    beta_schedule = [0.5, 2.0, 5.0, 15.0, 40.0, 120.0]
    iters_per_beta = 200
    step0 = 0.5

    while time.time() - start_time < budget_s * 0.94:
        # pick a start point: random or a perturbed best
        if rng.random() < 0.4:
            h_start = random_feasible_vector(n, rng)
        else:
            h_start = best_h.copy()
            h_start += rng.normal(scale=0.04, size=n)
            h_start = project_box_simplex(h_start, n / 2.0)

        remaining = budget_s - (time.time() - start_time)
        td = min(remaining * 0.12, 4.0)  # seconds for this trial
        h_opt, c_opt = gradient_descent(
            h_start, beta_schedule, step0, iters_per_beta, td, rng
        )
        # a short hill‑climb refinement
        h_opt2, c_opt2 = hill_climb(h_opt, rng, max_iters=800, scale=0.2)
        if c_opt2 < best_c:
            best_c, best_h = c_opt2, h_opt2.copy()
        if time.time() - start_time > budget_s * 0.90:
            break

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid (if time allows)
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.98 and n * 2 <= 200:
        h_up = np.repeat(best_h, 2)
        n_up = h_up.size
        h_up = project_box_simplex(h_up, n_up / 2.0)

        beta_up = [2.0, 10.0, 40.0, 120.0]
        iters_up = 250
        step_up = 0.35
        td = budget_s - (time.time() - start_time)
        h_up_opt, c_up_opt = gradient_descent(
            h_up, beta_up, step_up, iters_up, td, rng
        )
        h_up_opt, c_up_opt = hill_climb(h_up_opt, rng, max_iters=1000, scale=0.15)
        if c_up_opt < best_c:
            best_c, best_h, n = c_up_opt, h_up_opt.copy(), n_up

    # ------------------------------------------------------------------
    # Final projection to enforce exact feasibility
    # ------------------------------------------------------------------
    best_h = project_box_simplex(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
