    g = 1 - h
    corr = np.correlate(h, g, mode='full')
    max_corr = np.max(corr)
    # compute softmax weights
    exp_vals = np.exp(beta * (corr - max_corr))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros_like(h)
    # compute indicator part:
    # For each shift s from -(n-1) to n-1:
    for idx, weight in enumerate(w):

            continue

def _project_to_feasible(x, n, target=None, tol=1e-9, max_iter=1000):
    if target is None: target = n / 2.0
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # iterative water-filling
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to reduce sum
            idx = np.where(x > 0)[0]
        else:
            idx = np.where(x < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        x[idx] -= delta
        x = np.clip(x, 0.0, 1.0)
    # final correction
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # distribute residual greedily
        for i in range(n):
            if residual > 0 and x[i] < 1 - 1e-12:
                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 1e-12:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)
```

def _c5_from_h(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')

def _upscale_h(coarse_h, n_fine):
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

def _grad_softmax_max(h, beta):
    n = h.size
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    max_corr = corr.max()
    # to avoid overflow, shift
    exp_vals = np.exp(beta * (corr - max_corr))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros_like(h)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            # indices 0..n-s-1: (1 - h[i+s])
            grad[:n-s] += weight * (1 - h[s:])
            # indices s..n-1: -h[i-s]
            grad[s:] -= weight * h[:n-s]
        else:
            ss = -s
            # i >= ss: term (1 - h[i-ss])
            grad[ss:] += weight * (1 - h[:n-ss])
            # i <= n-ss-1: term -h[i+ss]
            grad[:n-ss] -= weight * h[ss:]
    return grad
```

import time
import numpy as np

def _project_to_feasible(x: np.ndarray, n: int, target: float = None,
                         tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(x)=target (default n/2) by water‑filling."""
    if target is None:
        target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(x > 0)[0]
        else:
            idx = np.where(x < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        x[idx] -= delta
        x = np.clip(x, 0.0, 1.0)
    # tiny correction if needed
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and x[i] < 1 - 1e-12:
                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
            elif residual < 0 and x[i] > 1e-12:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _grad_max_shift(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_{k} ∫h(x)(1‑h(x+k)) using the worst shift only."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)                # shift in samples, may be negative
    grad = np.zeros_like(h)
    if s >= 0:
        # (1 - h[i+s]) term for i = 0 .. n‑s‑1
        grad[: n - s] += 1.0 - h[s:]
        # -h[i‑s] term for i = s .. n‑1
        grad[s:] -= h[: n - s]
    else:
        ss = -s
        # (1 - h[i‑ss]) term for i = ss .. n‑1
        grad[ss:] += 1.0 - h[: n - ss]
        # -h[i+ss] term for i = 0 .. n‑ss‑1
        grad[: n - ss] -= h[ss:]
    return grad
