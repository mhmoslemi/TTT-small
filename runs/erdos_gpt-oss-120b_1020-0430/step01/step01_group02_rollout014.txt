def _evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation via FFT: corr[k] = sum_i h[i] * (1-h[i+k]), zero-padded.
    # Equivalent to cross-correlation: conv = ifft(fft(h) * conj(fft(g)))
    g = 1.0 - h
    m = 2 * n - 1
    H = np.fft.fft(h, m)
    G = np.fft.fft(g, m)
    # correlation = real(ifft(H * np.conj(G)))
    corr = np.fft.ifft(H * np.conj(G)).real
    # due to floating errors, ensure non-negative
    # Take max
    return np.max(corr) * dx

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    m = 2*n - 1
    H = np.fft.fft(h, m)
    G = np.fft.fft(g, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    # Ignore negligible negative due to rounding

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # water-filling style:
    # We'll loop until diff close to zero
    eps = 1e-12
    while abs(diff) > eps:
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        free_idx = np.where(mask)[0]
        # distribute equally
        per = diff / free_idx.size
        for i in free_idx:
            if diff > 0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) <= eps:
                break

def _simulated_annealing(h, target, budget, rng):
    n = len(h)
    cur_h = h.copy()
    cur_val = _evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # set temperatures
    init_temp = cur_val * 0.4 + 1e-6
    final_temp = cur_val * 0.001 + 1e-8
    while time.time() - start < budget:
        # progress fraction
        frac = (time.time() - start) / budget
        temp = init_temp * (1 - frac) + final_temp * frac
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            continue

        # use delta = rng.random() * max_move
        delta = rng.random() * max_move
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        new_val = _evaluate_c5(new_h)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (temp + 1e-16)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
    return best_h, best_val

def _slsqp_refine(h, target, deadline, rng):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type":"eq","fun": lambda x: x.sum() - target}
    best_h = h.copy()
    best_val = _evaluate_c5(best_h)
    while time.time() < deadline:
        # jitter start
        x0 = best_h + rng.normal(scale=0.02, size=n)
        x0 = np.clip(x0,0,1)
        x0 = _adjust_sum(x0, target)
        res = minimize(lambda x: _evaluate_c5(x), x0, method='SLSQP', bounds=bounds, constraints=[eq_cons], options={'maxiter':100,'ftol':1e-9,'disp':False})
        if not res.success:
            break
        cand = _adjust_sum(np.clip(res.x,0,1), target)
        cand_val = _evaluate_c5(cand)
        if cand_val < best_val:
            best_h, best_val = cand, cand_val
        # exit if time nearly done
        if time.time() > deadline:
            break
    return best_h, best_val

import time

from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    m = 2 * n - 1
    H = np.fft.fft(h, m)
    G = np.fft.fft(g, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    return np.max(corr) * dx

def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyper‑plane sum(x)=target while respecting 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    eps = 1e-12
    diff = target - x.sum()
    while abs(diff) > eps:
        if diff > 0:
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        if not np.any(mask):
            break
        free = np.where(mask)[0]
        per = diff / free.size
        for i in free:
            if diff > 0:
                add = min(per, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-per, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) <= eps:
                break
    return np.clip(x, 0.0, 1.0)
