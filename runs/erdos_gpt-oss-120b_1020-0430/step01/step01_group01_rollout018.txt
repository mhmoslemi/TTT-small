def _adam_optimize(h0, target, alpha=30.0, max_iters=3000, step0=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for t in range(1, max_iters+1):
        corr = np.correlate(h, 1 - h, mode='full')
        # compute softmax weights
        maxc = np.max(alpha*corr)
        exp = np.exp(alpha*corr - maxc)
        w = exp / np.sum(exp)
        # gradient is sum_s w_s * _gradient_for_shift
        grad = np.zeros_like(h)
        n = len(h)
        for shift, weight in enumerate(w):
            s = shift - (n-1)
            grad += weight * _gradient_for_shift(h, s)
        # directional gradient on smoothed max approximates gradient of smooth max.
        # Because smoothed objective: logsumexp( alpha * corr ) / alpha; gradient = sum w_s * grad_corr(s)
        # Indeed.
        # Update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        # project onto feasible set
        h = _project_onto_capped_simplex(h, target)
    return h

import time
import numpy as np

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} using bisection on the dual."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    """Uniform random point then projection."""
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the correlation for a given shift."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# --------------------------------------------------------------
# Optimisers
# --------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 1500,
                 step0: float = 0.8, decay: float = 0.998) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑correlation."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def _smooth_optimize(h0: np.ndarray, target: float,
                     max_iter: int = 500, step0: float = 0.1,
                     alpha: float = 30.0) -> np.ndarray:
    """Gradient descent on a smooth max (log‑sum‑exp) objective."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # soft‑max weights
        a_corr = alpha * corr
        m = np.max(a_corr)
        w = np.exp(a_corr - m)
        w /= w.sum()
        g = np.zeros_like(h)
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            shift = idx - (n - 1)
            g += weight * _gradient_for_shift(h, shift)
        h -= step * g
        h = _project_onto_capped_simplex(h, target)
        step *= 0.999
    return h


def _local_exchange_optimize(h: np.ndarray, target: float, dx: float,
                             max_iter: int, rng: np.random.RandomState) -> np.ndarray:
    """Hill‑climbing by moving a tiny amount of mass between two points."""
    best = h.copy()
    best_val = _compute_c5(best, dx)
    n = len(best)
    # indices that can give/take mass – recomputed lazily
    for _ in range(max_iter):
        # choose donor (h>0) and receiver (h<1)
        donors = np.where(best > 1e-8)[0]
        receivers = np.where(best < 1 - 1e-8)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        max_delta = min(best[i], 1.0 - best[j])
        if max_delta <= 0:
            continue
        delta = rng.rand() * max_delta
        new_h = best.copy()
        new_h[i] -= delta
        new_h[j] += delta
        val = _compute_c5(new_h, dx)
        if val + 1e-12 < best_val:
            best = new_h
            best_val = val
    return best

    """Return a step function h on [0,2] giving an improved C5 bound."""
    rng = np.random.RandomState(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96   # keep a safety margin

    # ------------------------------------------------------------------
    # Choose discretisation size
    # ------------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_h = None

    # Aim for a finer grid than the parent (>=400 points) but stay <1000.
    n = max(400, len(base_h) if base_h is not None else 0)
    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Helper to bring any vector to the required length
    # ------------------------------------------------------------------
    def _upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ------------------------------------------------------------------
    # Initial candidate
    # ------------------------------------------------------------------
    if base_h is not None:
        h0 = _upscale(base_h)
        h0 = _project_onto_capped_simplex(h0, target_sum)
    else:
        h0 = _random_feasible(rng, n, target_sum)

    best_h = h0.copy()
    best_val = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Main optimisation loop: many short restarts with GD + occasional
    # smooth refinement + local exchange.
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # ---- generate starting point ---------------------------------
        if best_h is not None and rng.rand() < 0.3:
            # slight Gaussian perturbation of the current best
            start = best_h + rng.normal(scale=0.02, size=n)
        else:
            start = _random_feasible(rng, n, target_sum)

        # ---- aggressive sub‑gradient descent -------------------------
        h_opt = _gd_optimize(start, target_sum,
                             max_iter=1500, step0=0.8, decay=0.998)

        # ---- occasional smooth refinement -----------------------------
        if restart % 4 == 0:
            h_opt = _smooth_optimize(h_opt, target_sum,
                                     max_iter=600, step0=0.07, alpha=35.0)

        # ---- local exchange hill‑climb --------------------------------
        remaining = deadline - time.time()
        # allocate a few seconds for the local search if we still have time
        if remaining > 4.0:
            iters = int(min(2000, remaining * 400))  # about 400 iters per second
            h_opt = _local_exchange_optimize(h_opt, target_sum, dx,
                                             max_iter=iters, rng=rng)

        # ---- evaluate -------------------------------------------------
        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        restart += 1
        # safety check – break if we are almost out of time
        if time.time() + 1.0 > deadline:
            break

    # ------------------------------------------------------------------
    # Final projection (just in case) and return
    # ------------------------------------------------------------------
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
