    for t in range(n_iter):
        p = np.correlate(h, h, mode='full')[n-1:]

def worst_shift_gradient(h):
    n = len(h)

    p = corr[n-1:]  # non-neg lags

import time
import numpy as np
from scipy.optimize import minimize

# ---------- feasibility projection ----------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (should be very small)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# ---------- C5 evaluation ----------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# ---------- SLSQP refinement (as a final polish) ----------
def _solve_h(initial_h: np.ndarray, max_iter: int = 500) -> tuple:
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    res = minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})

    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


# ---------- upscale (linear interpolation) ----------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# ---------- worst‑shift gradient ----------
def _worst_shift_gradient(h: np.ndarray):
    """Return index k with minimal self‑product and its gradient."""
    n = h.size
    corr = np.correlate(h, h, mode="full")
    prod = corr[n - 1:]               # non‑negative lags
    k = int(np.argmin(prod))          # worst shift
    g = np.zeros_like(h)
    if k < n:
        g[: n - k] += h[k:]           # contribution from forward partner
    if k > 0:
        g[k:] += h[: n - k]           # contribution from backward partner
    return k, g


# ---------- simple projected gradient ascent ----------
def _refine_h(h0: np.ndarray, n_iter: int = 1500, step0: float = 0.25, target_bound: float = 0.3808,
              time_left=lambda: 0):
    """Iteratively increase the minimum product p_k = h⋆h."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    # early exit if already below target
    if best_c5 <= target_bound:
        return best_h, best_c5
    for it in range(n_iter):
        if time_left() < 0.5:
            break
        k, g = _worst_shift_gradient(h)
        norm = np.linalg.norm(g)
        if norm == 0:
            break
        # linear step schedule
        step = step0 * (1.0 - it / n_iter)
        h = h + step * g / norm
        h = _project_to_feasible(h, n)
        cur = _c5_from_h(h)
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            if best_c5 <= target_bound:
                break
    return best_h, best_c5
