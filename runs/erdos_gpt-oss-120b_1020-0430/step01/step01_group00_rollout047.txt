import time, math
import numpy as np
from scipy import optimize

# ------------------- helper functions ------------------- #

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 = max correlation * 2/n (dx = 2/n)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _interpolate_h(src_h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of src_h (defined on [0,2]) to n_target points."""
    n_src = src_h.size
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, src_h)


def _binary_from_float(f: np.ndarray) -> np.ndarray:
    """Turn a float array into a binary 0/1 array with exactly half ones."""
    n = f.size
    idx = np.argsort(-f)[: n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def _beatty_binary(n: int) -> np.ndarray:
    """Sturmian (Beatty) binary sequence of density 1/2."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    frac = (np.arange(n) * phi) % 1.0
    ones = np.where(frac < 0.5)[0]
    # adjust to exact n/2
    if ones.size > n // 2:
        dist = np.abs(frac[ones] - 0.5)
        drop = ones[np.argsort(dist)[: ones.size - n // 2]]
        h = np.zeros(n)
        h[ones] = 1.0
        h[drop] = 0.0
    elif ones.size < n // 2:
        zeros = np.where(frac >= 0.5)[0]
        dist = np.abs(frac[zeros] - 0.5)
        add = zeros[np.argsort(dist)[: n // 2 - ones.size]]
        h = np.zeros(n)
        h[ones] = 1.0
        h[add] = 1.0
    else:
        h = np.zeros(n)
        h[ones] = 1.0
    return h


def _simulated_annealing_binary(start_h: np.ndarray, time_budget: float) -> tuple:
    """Simple SA on binary vectors (swap a 1 and a 0). Returns best h and its C5."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_c5 = _c5_from_h(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    # temperature schedule
    T = cur_c5 * 0.5 + 1e-6
    Tmin = 1e-6
    decay = 0.9997

    while time.time() - start < time_budget and T > Tmin:
        # pick random 1 and 0
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1.0 - cur_h)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0

        new_c5 = _c5_from_h(new_h)
        delta = new_c5 - cur_c5
        if delta <= 0 or np.random.rand() < math.exp(-delta / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
        T *= decay
    return best_h, best_c5


def _optimize_continuous(init_h: np.ndarray) -> tuple:
    """SLSQP refinement (same formulation as the reference solution)."""
    n = init_h.size
    h0 = _project_to_feasible(init_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def eq_fun(x):
        return np.sum(x[:n]) - (n / 2.0)

    def obj_fun(x):
        return x[-1]

    cons = [{"type": "ineq", "fun": ineq_fun},
            {"type": "eq",   "fun": eq_fun}]

    opts = {"maxiter": 400, "ftol": 1e-9, "disp": False}
    try:
        res = optimize.minimize(obj_fun, x0, method="SLSQP",
                                bounds=bounds, constraints=cons, options=opts)
    except Exception:
        res = None

    if res is not None and res.success:
        h_opt = res.x[:n]
    else:
        h_opt = h0
    h_opt = _project_to_feasible(h_opt, n)
    c5_val = _c5_from_h(h_opt)
    return h_opt, c5_val


# ------------------- main routine ------------------- #

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # retrieve optional parent construction
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    # target bound for early exit
    target_c5 = 0.3808

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate grid sizes (even, ≤300)
    candidates = set()
    if parent_h is not None:
        candidates.add(parent_h.size)
    for mul in (2, 3, 4):
        if parent_h is not None:
            candidates.add(parent_h.size * mul)
        else:
            candidates.add(128 * mul)
    for n in (160, 200, 256, 300):
        candidates.add(n)
    # keep only even numbers ≤300
    cand_list = sorted([n for n in candidates if n % 2 == 0 and n <= 300])

    # main search over grid sizes (large to small)
    for n in sorted(cand_list, reverse=True):
        if time_left() < 6.0:
            break

        # -------- initial binary pattern -------- #
        if parent_h is not None:
            if parent_h.size != n:
                base_float = _interpolate_h(parent_h, n)
            else:
                base_float = parent_h.copy()
            init_bin = _binary_from_float(base_float)
        else:
            init_bin = _beatty_binary(n)

        # -------- simulated annealing -------- #
        # allocate up to 30 % of remaining time (capped) for SA on this size
        sa_budget = min(time_left() * 0.30, 200.0)
        bin_h, bin_c5 = _simulated_annealing_binary(init_bin, sa_budget)

        if bin_c5 < best_c5:
            best_h, best_c5, best_n = bin_h.copy(), bin_c5, n
            if best_c5 < target_c5:
                break

        # -------- continuous refinement -------- #
        if time_left() > 5.0:
            cont_h, cont_c5 = _optimize_continuous(bin_h)
            if cont_c5 < best_c5:
                best_h, best_c5, best_n = cont_h.copy(), cont_c5, n
                if best_c5 < target_c5:
                    break

    # fallback: uniform half‑density step if nothing better found
    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    return best_h.tolist(), float(best_c5), int(best_n)
