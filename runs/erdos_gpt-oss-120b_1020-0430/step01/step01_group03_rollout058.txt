    cur_c = best_c

    # set schedule: total_iters approximate based on time remaining
    while time.time() - start < time_limit:
        # Temperature decays linearly with elapsed fraction
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        T = max(0.001, 1.0 - fraction) # start 1, end 0
        # step size scaling: maybe step = 0.5 * T

def normalize(x):
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = n / 2.0
    diff = target - x.sum()
    # use simple loop distributing diff
    if abs(diff) > 1e-12:
        # allow many iterations but diff small
        max_iters = 10000
        for _ in range(max_iters):
            if abs(diff) < 1e-12:
                break
            # choose indices with capacity
            if diff > 0:
                # need to add
                idx = np.where(x < 1.0)[0]
                if idx.size == 0:
                    break
                i = idx[np.random.randint(0, idx.size)]
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                idx = np.where(x > 0.0)[0]
                if idx.size == 0:
                    break
                i = idx[np.random.randint(0, idx.size)]
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub

def slsqp_optimize(h0, beta, max_iter=400):
    from scipy.optimize import minimize
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        # we use log-sum-exp
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': sum_eq}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})

    return x

    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1

def slsqp_optimize(h0, beta, max_iter=400):

                                              'ftol': 1e-12,
                                              'disp': False})

    # Adjust sum if needed

            add = min(diff, 1.0 - x[i])
            x[i] += add
        else:
            i = np.argmax(x > 0.0)
            sub = min(-diff, x[i])
            x[i] -= sub
    return x

import time

from scipy.optimize import minimize

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2 / n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def make_feasible(x, rng):
    """Project x onto the box [0,1] and the hyperplane Σx = n/2."""
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(10000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                idx = np.where(x < 1.0)[0]
                if idx.size == 0:
                    break
                i = rng.choice(idx)
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                idx = np.where(x > 0.0)[0]
                if idx.size == 0:
                    break
                i = rng.choice(idx)
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    x = rng.random(n)
    return make_feasible(x, rng)


def perturb_random(h, rng, scale):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

        c = compute_c5(new_h)

            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:

    return best_h, best_c


def gradient_worst_shift(h):
    """Gradient of the worst (maximal) overlap w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k_idx = np.argmax(corr)
    k_shift = k_idx - (n - 1)
    grad = np.zeros_like(h)
    if k_shift >= 0:
        i = np.arange(0, n - k_shift, dtype=int)
        j = i + k_shift
        grad[i] += (1 - h[j])
        grad[j] -= h[i]
    else:
        s = -k_shift
        i = np.arange(s, n, dtype=int)
        j = i - s
        grad[i] += (1 - h[j])
        grad[j] -= h[i]
    return grad, k_shift


def gradient_move(h, rng):
    """Single directed transfer based on the gradient of the worst shift."""
    grad, _ = gradient_worst_shift(h)
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h
    i = pos[np.argmax(grad[pos])]
    j = neg[np.argmin(grad[neg])]
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * 0.5
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
