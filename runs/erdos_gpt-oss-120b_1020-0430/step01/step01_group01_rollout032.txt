def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    # Determine n
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    # set n
    n = max(300, len(base) if base is not None else 0)
    target_sum = n / 2.0
    dx = 2.0 / n

    # upscale function
    def upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    if base is not None:
        h0_init = upscale(base)
        h0_init = _project_onto_capped_simplex(h0_init, target_sum)
        best_h = h0_init.copy()
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = None
        best_val = np.inf

    # Phase schedule for smooth descent
    phases = [
        (5.0, 0.9, 150, 50),
        (10.0, 0.7, 200, 50),
        (20.0, 0.5, 250, 30),
        (40.0, 0.3, 300, 20),
        (80.0, 0.2, 300, 15),
    ]

    restart = 0
    while time.time() - start_time < budget_s * 0.9:
        # generate start vector
        if best_h is not None and rng.rand() < 0.25:
            h_start = best_h + rng.normal(scale=0.02, size=n)
        else:
            h_start = _random_feasible(rng, n, target_sum)

        # smooth descent
        h_opt = h_start.copy()
        for alpha, step0, n_iter, K in phases:
            step = step0
            decay = 0.995
            for _ in range(n_iter):
                # compute correlation and weights
                corr = np.correlate(h_opt, 1.0 - h_opt, mode='full') * dx
                max_corr = corr.max()
                # weights
                w_raw = np.exp(alpha * (corr - max_corr))
                w = w_raw / w_raw.sum()
                # select top K shifts
                if K < len(w):
                    idxs = np.argpartition(w, -K)[-K:]
                else:
                    idxs = np.arange(len(w))
                g = np.zeros_like(h_opt)
                for idx in idxs:
                    shift = idx - (n - 1)
                    grad = _gradient_for_shift(h_opt, shift)
                    g += w[idx] * grad
                # update
                h_opt = h_opt - step * g
                # projection
                h_opt = _project_onto_capped_simplex(h_opt, target_sum)
                step *= decay
                # early exit if time low
                if time.time() - start_time > budget_s * 0.95:
                    break
            if time.time() - start_time > budget_s * 0.95:
                break

        # optional max-corr gradient descent fine tune
        # We'll run limited steps
        h_opt = _max_corr_descent(h_opt, target_sum, max_iter=200, step0=0.6, decay=0.997)

        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        restart += 1

    # final projection
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)
    else:
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)

def _max_corr_descent(h0, target, max_iter=400, step0=0.5, decay=0.996):
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

def _smooth_grad(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    w_raw = np.exp(alpha * (corr - max_corr))
    w = w_raw / w_raw.sum()
    w_arr = w   # shape (2n-1)
    w_rev = w_arr[::-1]
    f = 1.0 - h
    conv1 = np.convolve(w_arr, f, mode='full')
    conv2 = np.convolve(w_rev, h, mode='full')

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-9,
                                max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# Objective evaluation
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Gradient of a single correlation shift (subgradient of the max)
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _smooth_gradient(h: np.ndarray, alpha: float, target: float) -> np.ndarray:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    w_raw = np.exp(alpha * (corr - max_corr))
    w = w_raw / w_raw.sum()  # length 2n‑1, indexed by shift = idx-(n-1)

    # convolution implementation of the weighted sum of shift‑gradients
    f = 1.0 - h
    conv1 = np.convolve(w, f, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    start = n - 1
    grad = conv1[start:start + n] - conv2[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smoothed descent with a schedule of alphas and step sizes
# ----------------------------------------------------------------------
def _smooth_descent(h0: np.ndarray, target: float,
                    phases: list) -> np.ndarray:
    """phases = [(alpha, step0, n_iter), ...]"""
    h = h0.copy()
    n = h.shape[0]
    for alpha, step0, n_iter in phases:
        step = step0
        decay = 0.995
        for _ in range(n_iter):
            grad = _smooth_gradient(h, alpha, target)
            h = h - step * grad
            h = _project_onto_capped_simplex(h, target)
            step *= decay
    return h


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑correlation objective
# ----------------------------------------------------------------------
def _max_corr_descent(h0: np.ndarray, target: float,
                      max_iter: int = 300, step0: float = 0.6,
                      decay: float = 0.996) -> np.ndarray:
    h = h0.copy()
    n = h.shape[0]
    step = step0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):

        return np.interp(x_new, x_old, v)

    # --------------------------------------------------------------
    # Initial best candidate
    # --------------------------------------------------------------
    if base is not None:

    # --------------------------------------------------------------
