    best_h = cur_h.copy()

        cur_h[j] += delta
        new_c5 = _evaluate_c5(cur_h)

                best_h = cur_h.copy()

            cur_h[j] -= delta

def _adjust_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    current = x.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return x
    # if need to increase, add to entries below 1 equally
    # if need to decrease, subtract from entries above 0 equally
    # iterate a few times
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
            if not np.any(mask):
                break
            add = diff / np.count_nonzero(mask)
            # limit each addition to capacity
            capacity = 1.0 - x[mask]
            inc = np.minimum(add, capacity)
            x[mask] += inc
            diff = target - x.sum()
        else:
            mask = x > 0.0
            if not np.any(mask):
                break
            sub = (-diff) / np.count_nonzero(mask)
            capacity = x[mask]
            dec = np.minimum(sub, capacity)
            x[mask] -= dec
            diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

import time
import math
import numpy as np

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """Project x onto {0≤x_i≤1, sum x = target}."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple redistribution loop
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
            if not np.any(mask):
                break
            add = diff / np.count_nonzero(mask)
            cap = 1.0 - x[mask]
            inc = np.minimum(add, cap)
            x[mask] += inc
        else:
            mask = x > 0.0
            if not np.any(mask):
                break
            sub = (-diff) / np.count_nonzero(mask)
            cap = x[mask]
            dec = np.minimum(sub, cap)
            x[mask] -= dec
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

def _binary_sa_full(n, target, start_time, deadline, rng, target_c5=0.3808):
    """Simulated‑annealing on binary vectors (exactly target ones)."""
    best_h = None
    best_c5 = np.inf
    cur_h = None
    cur_c5 = None
    iters_since_best = 0
    total_iters = 0
    # pre‑compute when to give up
    while time.time() < deadline:
        # restart if stuck too long
        if cur_h is None or iters_since_best > 50000:
            cur_h = np.zeros(n, dtype=np.float64)
            cur_h[rng.choice(n, size=int(target), replace=False)] = 1.0
            cur_c5 = _evaluate_c5(cur_h)
            iters_since_best = 0

        # temperature decays very slowly – we only need a weak schedule
        T = max(1e-5, 0.05 * (0.9999 ** total_iters))

        # decide move type
        if rng.random() < 0.7:
            # targeted swap based on the currently worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            s_idx = np.argmax(corr)
            shift = s_idx - (n - 1)
            idx = np.arange(n)
            partner = idx + shift
            valid = (partner >= 0) & (partner < n)

            mism_one = idx[valid & (cur_h == 1) & (cur_h[partner] == 0)]
            mism_zero = idx[valid & (cur_h == 0) & (cur_h[partner] == 1)]

            if mism_one.size and mism_zero.size:
                i = rng.choice(mism_one)
                j = rng.choice(mism_zero)
            else:
                i, j = rng.integers(0, n, size=2, endpoint=False)
                if cur_h[i] == cur_h[j]:
                    total_iters += 1
                    continue
        else:
            i, j = rng.integers(0, n, size=2, endpoint=False)
            if cur_h[i] == cur_h[j]:
                total_iters += 1
                continue

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = _evaluate_c5(cur_h)
        dE = new_c5 - cur_c5

        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                iters_since_best = 0
                # stop early if we already beat the required bound
                if best_c5 <= target_c5:
                    return best_h, best_c5
            else:
                iters_since_best += 1
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            iters_since_best += 1

        total_iters += 1

    # fallback if nothing was found (should not happen)
    if best_h is None:
        best_h = np.zeros(n, dtype=np.float64)
        best_h[:int(target)] = 1.0
        best_c5 = _evaluate_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
