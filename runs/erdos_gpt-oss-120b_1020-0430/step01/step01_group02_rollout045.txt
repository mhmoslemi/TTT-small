def adjust_sum(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Compute diff = target - sum(v)
    diff = target - v.sum()
    # Loop up to some max iter.
    # Use mask of free variables: those not at bounds within tolerance.
    tolerance = 1e-12
    iteration = 0
    while abs(diff) > 1e-12 and iteration < 1000:
        # Determine indices that can be increased (if diff >0) or decreased (if diff<0)
        if diff > 0:
            mask = v < 1.0 - tolerance
        else:
            mask = v > tolerance
        count = np.count_nonzero(mask)
        if count == 0:
            break
        per = diff / count
        # Add per to masked elements
        # Clip changes
        # For each idx where mask: adjust
        # Use vectorized operation
        # delta = np.clip(per, lower, upper)
        # But per may be bigger than capacity. We'll compute capacity and adjust accordingly.
        # Simpler: adjustments = np.where(mask, np.clip(v + per, 0, 1) - v, 0)
        # However if capacity insufficient, after loop diff will be reduced less than full.
        adjustments = np.zeros_like(v)
        if diff > 0:
            # capacity to add
            cap = 1.0 - v
            add = np.minimum(per, cap)
            # If per > cap, we add only up to cap; diff reduces accordingly
            # Actually need to adjust per for each, but per is same for all free; could exceed cap for some
            # We'll add cap for those where per > cap, and add per for others.
            # For vectorized: add = np.where(per > cap, cap, per)
            add = np.where(per > cap, cap, per)
            adjustments[mask] = add[mask]
        else:
            # diff <0; need to subtract
            cap = v
            sub = np.where(-per > cap, cap, -per)  # per is negative
            adjustments[mask] = -sub[mask]  # subtract
        # Apply adjustments
        v += adjustments
        diff = target - v.sum()
        iteration += 1
    # Final clip
    return np.clip(v, 0.0, 1.0)

def random_move(h, rng, max_scale=0.5):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # Random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-12:
        return h.copy()
    # delta scale: upto max_scale fraction of max_move
    delta = rng.random() * max_move * max_scale
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

def targeted_move(h, rng):
    n = len(h)
    # compute correlation
    c = np.correlate(h, 1.0 - h, mode='full')
    # find index of max
    idx = np.argmax(c)
    shift = idx - (n - 1)  # shift value s
    # For positive shift s>0: terms with i in [0, n-1-s] contributions hi*(1 - h[i+s])
    # For negative shift s<0: contributions hi*(1 - h[i+s]) but i+s >=0. Could treat symmetrical.
    # Choose i that contributes most: hi*(1 - h[i+shift]).
    if shift >= 0:
        i_range = np.arange(n - shift)
        contrib = h[i_range] * (1.0 - h[i_range + shift])
    else:
        # shift negative, i_range = -shift .. n-1
        s = -shift
        i_range = np.arange(s, n)
        contrib = h[i_range] * (1.0 - h[i_range - s])
    # find maximum contribution and index i
    i_local = np.argmax(contrib)
    i = i_range[i_local]
    # Determine j = i+shift (if shift>=0) else i-shift
    if shift >= 0:
        j = i + shift
    else:
        j = i - shift
    # propose moving delta from i to j or vice versa to reduce product hi*(1 - hj)
    # Ideally reduce hi or increase hj.
    # Which direction reduces product more? If hi > (1 - hj), moving from i to j reduces hi and increases hj.
    # We'll decide to move from i (src) to j (dst) if h[i] > 0 and h[j] < 1.
    # But also consider reverse.
    # Choose direction that can move mass.
    if h[i] > 0 and h[j] < 1:
        src, dst = i, j
    elif h[j] > 0 and h[i] < 1:
        src, dst = j, i
    else:
        # can't move because both at bounds, return original
        return h.copy()
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-12:
        return h.copy()
    # delta fraction proportional to contribution maybe
    delta = min(max_move, 0.5 * max_move) * rng.random()
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

    # Project onto constraints sum=target, 0<=x<=1 via lambda method.
    # x is arbitrary real.
    # Use lambda search: find lambda such that sum(clamp(x+lambda,0,1)) = target

            lo = mid
        else:
            hi = mid

import time
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """
    Compute C_5 for the step vector h.
    """
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1 - h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the capped simplex:
        0 <= x_i <= 1,   sum_i x_i = target.
    The solution has the form x_i = clip(v_i + λ, 0, 1) for a suitable λ,
    which we find by binary search.
    """
    # Ensure a feasible interval for λ.
    lo = -np.max(v) - 1.0   # guarantees v+λ <= 0 -> all zeros
    hi = 1.0 - np.min(v) + 1.0   # guarantees v+λ >= 1 -> all ones
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    return np.clip(v + λ, 0.0, 1.0)


def _random_move(h: np.ndarray, rng: np.random.Generator, max_scale: float = 0.8) -> np.ndarray:
    """
    Transfer a random amount of mass from one index to another,
    preserving the total sum and the box constraints.
    """
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-14:
        return h.copy()
    delta = rng.random() * max_move * max_scale
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new


def _targeted_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the shift giving the maximal overlap and move mass from the
    index contributing most to that overlap towards its partner.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)        # integer shift s
    if shift == 0:
        # fall back to a random move if the worst shift is zero
        return _random_move(h, rng, max_scale=0.3)

    # Determine the range of i that participates in this shift
    if shift > 0:
        i_range = np.arange(n - shift)
        partner = i_range + shift
    else:
        s = -shift
        i_range = np.arange(s, n)
        partner = i_range - s

    # contribution hi * (1 - h_partner)
    contrib = h[i_range] * (1.0 - h[partner])
    i_local = int(np.argmax(contrib))
    i = i_range[i_local]
    j = partner[i_local]

    # Decide direction that can actually move mass
    if h[i] > 0.0 and h[j] < 1.0:
        src, dst = i, j
    elif h[j] > 0.0 and h[i] < 1.0:
        src, dst = j, i
    else:
        # both at bounds – give up
        return h.copy()

    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-14:
        return h.copy()
    # Move a random fraction, but not more than half of the allowed amount
    delta = rng.random() * max_move * 0.5
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Initialise vector h
    # --------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.asarray(init, dtype=np.float64).copy()
        n = h0.size
        h = h0
    else:
        n = 60                     # choose a modest resolution when no starter
        h = rng.random(n)

    target_sum = n / 2.0
    h = _project_to_sum_and_bounds(h, target_sum)

    # --------------------------------------------------------------
    # 2. Simulated annealing phase
    # --------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Temperature schedule parameters
    T_start = 0.012
    T_end = 1e-6
    total_time = budget_s - 2.0   # reserve time for final polishing

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = min(elapsed / total_time, 1.0)
        # Exponential cooling: T = T_start * (T_end/T_start)^{progress}
        T = T_start * ((T_end / T_start) ** progress)

        # Pick a move type
        if rng.random() < 0.78:
            h_new = _random_move(cur_h, rng, max_scale=0.85)
        else:
            h_new = _targeted_move(cur_h, rng)

        new_c5 = _evaluate_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c5 = h_new, new_c5
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5

        # Occasionally re‑seed a little bit around the current best
        if rng.random() < 0.001:
            jitter = best_h + rng.normal(scale=0.015, size=n)
            jitter = np.clip(jitter, 0.0, 1.0)
            jitter = _project_to_sum_and_bounds(jitter, target_sum)
            cur_h, cur_c5 = jitter, _evaluate_c5(jitter)

    # --------------------------------------------------------------
    # 3. Final deterministic hill‑climbing refinement
    # --------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # Random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-14:
            continue
        delta = rng.random() * max_move * 0.5
        h_new = cur_h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5

    # --------------------------------------------------------------
    # 4. Return the best feasible solution found
    # --------------------------------------------------------------
    best_h = _project_to_sum_and_bounds(best_h, target_sum)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
