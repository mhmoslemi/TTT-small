    decay = 0.9999
    it=0
    while time.time() - start < budget_s*0.95:
        # random pair of indices and perturb
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        # compute allowed delta, for exchange of epsilon from i to j.
        # we must keep each within [0,1] and sum constant.

        max_up = min(1.0 - hi, hj)  # moving from j to i (increase i)
        max_down = min(hi, 1.0 - hj) # moving from i to j (decrease i)
        if max_up + max_down == 0:
            continue
        # sample epsilon uniformly from [-max_down, max_up]
        delta = random.random() * (max_up + max_down) - max_down
        if abs(delta) < 1e-12: continue
        nh = cur_h.copy()
        nh[i] = hi + delta
        nh[j] = hj - delta
        new_c5 = get_c5(nh)
        # Acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / Ttemp):
            cur_h = nh
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = nh.copy()
        # periodic targeted move
        if it % 500 == 0:
            # Compute full correlation
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            idx_max = np.argmax(corr)
            s = idx_max - (n - 1)
            # find indices i with highest contribution at shift s: h[i]*(1 - h[i+s])
            if s >= 0:
                i_range = np.arange(n - s)
                p_range = i_range + s
            else:
                i_range = np.arange(-s, n)
                p_range = i_range + s
            contrib = cur_h[i_range] * (1 - cur_h[p_range])
            k = int(np.argmax(contrib))
            i_best = i_range[k]
            p_best = p_range[k]
            eps_max = min(cur_h[i_best], 1 - cur_h[p_best])
            if eps_max > 1e-12:
                eps = eps_max * 0.5
                nh2 = cur_h.copy()
                nh2[i_best] -= eps
                nh2[p_best] += eps
                new_c5b = get_c5(nh2)
                if new_c5b < cur_c5 or random.random() < math.exp(-(new_c5b - cur_c5) / Ttemp):
                    cur_h = nh2
                    cur_c5 = new_c5b
                    if new_c5b < best_c5:

def binary_milp_solution(n, target_sum, time_limit):
    import cvxpy as cp, numpy as np, time
    h = cp.Variable(n, boolean=True)
    T = cp.Variable()
    constraints = [cp.sum(h) == target_sum, T >= 0]
    for s in range(-n+1, n):
        if s == 0: continue
        start = max(0, -s)
        length = n - abs(s)
        i = h[start:start+length]
        j = h[start+s:start+s+length]
        w = cp.Variable(length, nonneg=True)
        constraints += [w <= i,
                        w <= 1 - j,
                        w >= i - j,
                        cp.sum(w) <= T]
    prob = cp.Problem(cp.Minimize(T), constraints)
    # time limit param (if solver supported)
    try:
        prob.solve(solver=cp.CBC, verbose=False, tee=False, # maybe time_limit?
            # there is an optional parameter "max_iters" or "timeLimit"
        )
    except Exception as e:
        return None
    if h.value is None:
        return None
    h_bin = np.rint(h.value).astype(int)
    # enforce sum
    if h_bin.sum() != target_sum:
        # adjust by flipping random bits
        diff = int(target_sum - h_bin.sum())
        # if diff>0: set some zero to 1, else set some 1 to 0
        zero_idx = np.where(h_bin == 0)[0]
        one_idx = np.where(h_bin == 1)[0]
        if diff > 0:
            choices = np.random.choice(zero_idx, diff, replace=False)
            h_bin[choices] = 1
        elif diff < 0:
            choices = np.random.choice(one_idx, -diff, replace=False)
            h_bin[choices] = 0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math, cvxpy as cp

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # get n
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    except NameError:
        n = 70
        init_h = None

    target_sum = n/2

    # Helper to compute C5
    def c5_of(v):
        return float(np.correlate(v, 1 - v, mode='full').max() * (2.0 / n))

    best_h = None
    best_c5 = math.inf

    # Try MILP binary solution if enough time
    time_left = budget_s - (time.time() - start)
    if time_left > 30: # allocate at least 30 sec for MILP
        # Build MILP
        h = cp.Variable(n, boolean=True)
        T = cp.Variable()
        constraints = [cp.sum(h) == target_sum, T >= 0]
        for s in range(-n+1, n):
            if s == 0:
                continue
            start_i = max(0, -s)
            length = n - abs(s)
            hi = h[start_i:start_i+length]
            hj = h[start_i+s:start_i+s+length]
            w = cp.Variable(length, nonneg=True)
            constraints.append(w <= hi)
            constraints.append(w <= 1 - hj)
            constraints.append(w >= hi - hj)
            constraints.append(cp.sum(w) <= T)
        prob = cp.Problem(cp.Minimize(T), constraints)
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            prob = None
        if prob is not None and h.value is not None:
            h_bin = np.rint(h.value).astype(float)
            # Ensure sum exactly target_sum (floating)
            diff = int(round(target_sum - h_bin.sum()))
            if diff != 0:
                # adjust
                zero_idx = np.where(h_bin < 0.5)[0]
                one_idx = np.where(h_bin >= 0.5)[0]
                if diff > 0 and zero_idx.size >= diff:
                    sel = np.random.choice(zero_idx, diff, replace=False)
                    h_bin[sel] = 1.0
                elif diff < 0 and one_idx.size >= -diff:
                    sel = np.random.choice(one_idx, -diff, replace=False)
                    h_bin[sel] = 0.0
            cur_c5 = c5_of(h_bin)
            best_h = h_bin.copy()
            best_c5 = cur_c5

    # If MILP failed, fall back to init or random
    if best_h is None:
        if init_h is not None:
            # ensure sum constraint
            cur = init_h.copy()
            # clip and adjust sum
            cur = np.clip(cur, 0.0, 1.0)
            cur_sum = cur.sum()
            if abs(cur_sum - target_sum) > 1e-9:
                # adjust by scaling then clipping
                cur *= target_sum / cur_sum
                cur = np.clip(cur, 0.0, 1.0)
                # final correction loop
                cur_sum = cur.sum()
                while abs(cur_sum - target_sum) > 1e-9:
                    diff = cur_sum - target_sum
                    if diff > 0:
                        cand = np.where(cur > 1e-12)[0]
                        i = random.choice(cand.tolist())
                        delta = min(cur[i], diff)
                        cur[i] -= delta
                    else:
                        cand = np.where(cur < 1 - 1e-12)[0]
                        i = random.choice(cand.tolist())
                        delta = min(1 - cur[i], -diff)
                        cur[i] += delta
                    cur_sum = cur.sum()
            best_h = cur.copy()
            best_c5 = c5_of(best_h)
        else:
            # random binary half ones
            cur = np.zeros(n)
            cur[:int(target_sum)] = 1.0
            np.random.shuffle(cur)
            best_h = cur.copy()
            best_c5 = c5_of(best_h)

    # Local refinement (simulated annealing)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Setup temperature
    Ttemp = max(1e-5, cur_c5 * 0.02)
    decay = 0.9997
    it = 0

    while time.time() - start < budget_s * 0.95:
        # Random exchange move
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        max_up = min(1.0 - hi, hj)   # increase i, decrease j
        max_down = min(hi, 1.0 - hj) # decrease i, increase j
        if max_up + max_down > 0:
            delta = random.random() * (max_up + max_down) - max_down
            if abs(delta) > 1e-12:
                nh = cur_h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_c5 = c5_of(nh)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / Ttemp):
                    cur_h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = nh.copy()
        # Targeted move occasionally
        if it % 400 == 0:
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)
            if s >= 0:
                i_range = np.arange(n - s)
                p_range = i_range + s
            else:
                i_range = np.arange(-s, n)
                p_range = i_range + s
            contrib = cur_h[i_range] * (1 - cur_h[p_range])
            k = int(np.argmax(contrib))
            i_best = i_range[k]
            p_best = p_range[k]
            eps_max = min(cur_h[i_best], 1 - cur_h[p_best])
            if eps_max > 1e-12:
                eps = eps_max * 0.5
                nh = cur_h.copy()
                nh[i_best] -= eps
                nh[p_best] += eps
                new_c5 = c5_of(nh)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / Ttemp):
                    cur_h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = nh.copy()
        it += 1
        if it % 1200 == 0:
            Ttemp *= decay
        # Early exit if we beat target bound
        if best_c5 <= 0.3808:
            break

    # Final adjustment to exact sum
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        # distribute error
        for _ in range(200):
            k = random.randrange(n)
            if err > 0 and best_h[k] > 0:
                delta = min(err, best_h[k])
                best_h[k] -= delta
                err -= delta
            elif err < 0 and best_h[k] < 1:
                delta = min(-err, 1 - best_h[k])
                best_h[k] += delta
                err += delta
            if abs(err) < 1e-12:
                break

    final_c5 = c5_of(best_h)

    return (best_h.tolist(), final_c5, n)

import numpy as np
import random
import math
import time
import cvxpy as cp

def _c5_of(v: np.ndarray, n: int) -> float:
    """Maximum overlap integral for a discretised step function."""
    return float(np.correlate(v, 1 - v, mode="full").max() * (2.0 / n))

def _max_shift_gradient(v: np.ndarray, n: int) -> np.ndarray:
    """Gradient of the worst‐shift overlap w.r.t. the vector v."""
    corr = np.correlate(v, 1 - v, mode="full")
    mval = corr.max()
    # shifts attaining the maximum (allow tiny numerical tolerance)
    idxs = np.where(np.abs(corr - mval) < 1e-12)[0]
    shifts = idxs - (n - 1)
    grad = np.zeros(n)
    for s in shifts:
        if s == 0:
            continue
        if s > 0:
            i = np.arange(n - s)
            j = i + s
        else:
            i = np.arange(-s, n)
            j = i + s          # j = i - |s|
        grad[i] += (1 - v[j])
        grad[j] -= v[i]
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine dimension and target sum
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = int(init_h.shape[0])
    except NameError:
        n = 70
        init_h = None

    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Helper to keep h within bounds and correct the sum exactly
    # ------------------------------------------------------------------
    def _repair(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        if abs(cur - target_sum) > 1e-12:
            if cur > 0:
                v *= target_sum / cur
            v = np.clip(v, 0.0, 1.0)
            cur = v.sum()
            while abs(cur - target_sum) > 1e-9:
                diff = cur - target_sum
                if diff > 0:
                    cand = np.where(v > 1e-12)[0]
                    i = random.choice(cand.tolist())
                    delta = min(v[i], diff)
                    v[i] -= delta
                else:
                    cand = np.where(v < 1 - 1e-12)[0]
                    i = random.choice(cand.tolist())
                    delta = min(1 - v[i], -diff)
                    v[i] += delta
                cur = v.sum()
        return v

    # ------------------------------------------------------------------
    # Initialise best solution (binary MILP if possible)
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = math.inf

    # ------------------------------------------------------------------
    # Try an exact binary MILP (only if n is even)
    # ------------------------------------------------------------------
    time_left = budget_s - (time.time() - start)
    if n % 2 == 0 and time_left > 30.0:
        # Build MILP
        h = cp.Variable(n, boolean=True)
        T = cp.Variable()
        cons = [cp.sum(h) == target_sum, T >= 0]

        for s in range(-n + 1, n):
            if s == 0:
                continue
            start_i = max(0, -s)
            length = n - abs(s)
            hi = h[start_i:start_i + length]
            hj = h[start_i + s:start_i + s + length]
            w = cp.Variable(length, nonneg=True)
            cons.extend([
                w <= hi,
                w <= 1 - hj,
                w >= hi - hj,
                cp.sum(w) <= T
            ])

        prob = cp.Problem(cp.Minimize(T), cons)
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=int(time_left))
        except Exception:
            prob = None

        if prob is not None and h.value is not None:
            hb = np.rint(h.value).astype(float)  # binary (0/1)
            # Fix any tiny deviation from the required number of ones
            diff = int(round(target_sum - hb.sum()))
            if diff != 0:
                zeros = np.where(hb < 0.5)[0]
                ones = np.where(hb > 0.5)[0]
                if diff > 0 and len(zeros) >= diff:
                    sel = np.random.choice(zeros, diff, replace=False)
                    hb[sel] = 1.0
                elif diff < 0 and len(ones) >= -diff:
                    sel = np.random.choice(ones, -diff, replace=False)
                    hb[sel] = 0.0
            cur_c5 = _c5_of(hb, n)
            best_h = hb.copy()
            best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Fallback initialisation (use provided init or a random half‑ones vector)
    # ------------------------------------------------------------------
    if best_h is None:
        if init_h is not None:
            cur = _repair(init_h.copy())
        else:
            cur = np.zeros(n)
            cur[:int(target_sum)] = 1.0
            np.random.shuffle(cur)
        best_h = cur.copy()
        best_c5 = _c5_of(best_h, n)

    # ------------------------------------------------------------------
    # Local optimisation – simulated annealing + gradient moves
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5

    Ttemp = max(1e-5, cur_c5 * 0.02)
    decay = 0.9997
    it = 0

    while time.time() - start < budget_s * 0.95:
        # ----------- random exchange move (preserves sum) ---------------
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        max_up = min(1.0 - hi, hj)          # increase i, decrease j
        max_down = min(hi, 1.0 - hj)        # decrease i, increase j
        if max_up + max_down > 0:
            delta = random.random() * (max_up + max_down) - max_down
            if abs(delta) > 1e-12:
                nh = cur_h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new_c5 = _c5_of(nh, n)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / Ttemp):
                    cur_h, cur_c5 = nh, new_c5
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, nh.copy()

        # ----------- targeted move on the worst shift -----------------
        if it % 400 == 0:
            corr = np.correlate(cur_h, 1 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            s = idx_max - (n - 1)
            if s >= 0:
                i_range = np.arange(n - s)
                p_range = i_range + s
            else:
                i_range = np.arange(-s, n)
                p_range = i_range + s
            contrib = cur_h[i_range] * (1.0 - cur_h[p_range])
            k = int(np.argmax(contrib))
            i_best, p_best = i_range[k], p_range[k]
            eps_max = min(cur_h[i_best], 1.0 - cur_h[p_best])
            if eps_max > 1e-12:
                eps = eps_max * 0.5
                nh = cur_h.copy()
                nh[i_best] -= eps
                nh[p_best] += eps
                new_c5 = _c5_of(nh, n)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / Ttemp):
                    cur_h, cur_c5 = nh, new_c5
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, nh.copy()

        # ----------- gradient‑driven move on the worst shift -----------
        if it % 300 == 0:
            grad = _max_shift_gradient(cur_h, n)
            pos = int(np.argmax(grad))
            neg = int(np.argmin(grad))
            if grad[pos] > 0 and grad[neg] < 0:
                eps_max = min(cur_h[pos], 1.0 - cur_h[neg])
                if eps_max > 1e-12:
                    eps = eps_max * 0.5
                    nh = cur_h.copy()
                    nh[pos] -= eps
                    nh[neg] += eps
                    new_c5 = _c5_of(nh, n)
                    if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / Ttemp):
                        cur_h, cur_c5 = nh, new_c5
                        if new_c5 < best_c5:
                            best_c5, best_h = new_c5, nh.copy()

        it += 1
        if it % 1200 == 0:
            Ttemp *= decay
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Final repair to honour the integral constraint exactly
    # ------------------------------------------------------------------
    err = best_h.sum() - target_sum
    if abs(err) > 1e-9:
        for _ in range(200):
            k = random.randrange(n)
            if err > 0 and best_h[k] > 0:
                delta = min(err, best_h[k])
                best_h[k] -= delta
                err -= delta
            elif err < 0 and best_h[k] < 1:
                delta = min(-err, 1 - best_h[k])
                best_h[k] += delta
                err += delta
            if abs(err) < 1e-12:
                break

    final_c5 = _c5_of(best_h, n)
    return (best_h.tolist(), final_c5, n)
