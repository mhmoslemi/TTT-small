import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the SLSQP helper (no closures)
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_{k} Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(vec, target, rng):
    """Clip to [0,1] and adjust to hit the exact sum target."""
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # simple random redistribution (few thousand steps is enough)
    n = vec.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - vec[i])
        else:
            add = max(diff, -vec[i])
        vec[i] += add
        diff = target - vec.sum()
    return vec


def slsqp_optimize(h0, beta, max_iter=200):
    """One SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny redistrib. to fix possible drift
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, target, rng)
    return x


def hill_climb(h_start, rng, max_iters=2000, step_scale=0.2):
    """Simple hill‑climb with pairwise mass transfers."""
    n = h_start.size
    target = n / 2.0
    h = enforce_sum(h_start.copy(), target, rng)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = step_scale
    for it in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a small ε from i to j
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_h = enforce_sum(new_h, target, rng)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
