def _project_sum(x, target, max_iter=50):
    lo = -2.0 # just safe
    hi = 2.0
    for _ in range(max_iter):
        mid = (lo+hi)/2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

import time

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the hyper‑plane sum(x)=target while respecting 0≤x≤1.
    Uses a simple bisection on the shift λ where y = clip(x+λ,0,1).
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)


def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a random vector in [0,1]^n with prescribed sum."""
    x = rng.random(n)
    return _project_sum(x, target)


def _pair_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Propose a mass‑exchange move between two coordinates (keeps the sum)."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 1e-14:
        return None
    delta = rng.random() * max_move
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new


def _hill_climb(h: np.ndarray, cur_score: float, deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Deterministic improvement by accepting only improving pair moves."""
    best_h, best_score = h.copy(), cur_score
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(best_h[src], 1.0 - best_h[dst])
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move
        new_h = best_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        new_score = _evaluate_c5(new_h)
        if new_score + 1e-12 < best_score:
            best_h, best_score = new_h, new_score
    return best_h, best_score


def _smooth_max_c5(h: np.ndarray, beta: float) -> float:
    """
    Smooth approximation of the max‑overlap functional.
    Uses soft‑max on the correlation values (no dx scaling).
    """
    n = h.size
    total = h.sum()
    conv = np.correlate(h, h, mode="full")          # h·h
    corr = total - conv                             # h·(1‑h) = total‑h·h
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    w = exp_vals / exp_vals.sum()
    smooth = np.dot(w, corr)                        # weighted average ≈ max
    return smooth * (2.0 / n)


def _gradient_step(h: np.ndarray, beta: float, eta: float,
                   target_sum: float) -> np.ndarray:
    """
    One gradient‑descent step on the smooth max‑overlap objective.
    The gradient is the weighted sum of shifted copies of h.
    """
    n = h.size
    total = h.sum()
    conv = np.correlate(h, h, mode="full")
    corr = total - conv
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    w = exp_vals / exp_vals.sum()                   # soft‑max weights

    grad = np.zeros_like(h)
    # accumulate -w * (h shifted left/right)
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)                       # shift = lag
        if shift >= 0:
            # contribution from h[i] * h[i+shift]
            grad[: n - shift] -= wi * h[shift:]
            grad[shift:]      -= wi * h[: n - shift]
        else:
            s = -shift
            grad[s:]         -= wi * h[: n - s]
            grad[: n - s]    -= wi * h[s:]

    # descent step (gradient of smooth max is negative of the above)
    h_new = h - eta * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project_sum(h_new, target_sum)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimise a step function for the Erdős C5 problem."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1. Initialise problem size and a feasible starting point
    # -----------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        n = 45
        target_sum = n / 2.0
        h0 = _random_feasible(n, target_sum, rng)

    target_sum = n / 2.0
    h0 = _project_sum(np.clip(h0, 0.0, 1.0), target_sum)
    best_h = h0.copy()
    best_score = _evaluate_c5(best_h)

    # -----------------------------------------------------------------
    # 2. Global stochastic search (simulated annealing)
    # -----------------------------------------------------------------
    cur_h = best_h.copy()
    cur_score = best_score
    T0 = 0.02                          # initial temperature
    no_improve = 0
    SA_deadline = start_time + budget_s * 0.55   # ~55 % of time for SA

        T = T0 * max(0.0, 1.0 - frac) + 1e-12

        cand = _pair_move(cur_h, rng)
        if cand is None:

        cand_score = _evaluate_c5(cand)
        delta = cand_score - cur_score

        if delta < -1e-14 or rng.random() < np.exp(-delta / T):
            cur_h, cur_score = cand, cand_score
            if cur_score + 1e-12 < best_score:
                best_h, best_score = cur_h.copy(), cur_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart when stuck
        if no_improve > 8000:
            cur_h = best_h + rng.normal(scale=0.07, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_sum(cur_h, target_sum)
            cur_score = _evaluate_c5(cur_h)
            no_improve = 0

    # -----------------------------------------------------------------
    # 3. Local refinement with smooth‑gradient steps
    # -----------------------------------------------------------------
    beta = 12.0            # softness of the max‑approximation
    eta = 0.04             # learning rate
    grad_deadline = start_time + budget_s * 0.85   # another 30 % of budget

    while time.time() < grad_deadline:
        new_h = _gradient_step(best_h, beta, eta, target_sum)
        new_score = _evaluate_c5(new_h)
        if new_score + 1e-12 < best_score:
            best_h, best_score = new_h, new_score
        else:
            # simple back‑off: shrink step size a little
            eta *= 0.99
            if eta < 1e-5:
                eta = 0.04
        # occasional random jitter to avoid stagnation
        if rng.random() < 0.003:
            jitter = rng.normal(scale=0.04, size=n)
            best_h = np.clip(best_h + jitter, 0.0, 1.0)
            best_h = _project_sum(best_h, target_sum)
            best_score = _evaluate_c5(best_h)

    # -----------------------------------------------------------------
    # 4. Final deterministic hill‑climbing
    # -----------------------------------------------------------------
    final_deadline = start_time + budget_s - 0.5
    best_h, best_score = _hill_climb(best_h, best_score, final_deadline, rng)

    # -----------------------------------------------------------------
    # 5. Return the best feasible solution found
    # -----------------------------------------------------------------
    best_h = _project_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_score = float(_evaluate_c5(best_h))
    return best_h, best_score, n
