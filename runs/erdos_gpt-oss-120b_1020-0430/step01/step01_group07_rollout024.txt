    n = h.shape[0]
    g = np.zeros_like(h)

        d = shift
        g[:n-d] += (1.0 - h[d:])
        g[d:] -= h[:n-d]
    else:
        d = -shift
        g[d:] += (1.0 - h[:n-d])

    return g

def propose_random_pair_move(h):
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_bound = min(1.0 - hi, hj)
    if lo >= hi_bound:
        return None, None, None
    # sample delta uniformly
    delta = random.uniform(lo, hi_bound)
    return i, j, delta

def propose_gradient_pair_move(h, grad):
    # indices where gradient positive and negative
    pos_idxs = np.where(grad > 1e-12)[0]
    neg_idxs = np.where(grad < -1e-12)[0]
    if pos_idxs.size == 0 or neg_idxs.size == 0:
        return None, None, None
    i = pos_idxs[np.argmax(grad[pos_idxs])]  # biggest positive
    j = neg_idxs[np.argmin(grad[neg_idxs])]  # most negative (most negative)
    hi, hj = h[i], h[j]

    n = h.shape[0]
    target = n / 2.0
    current = h.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return h

            break

    return h

import time, math, random
import numpy as np

# Utility functions

def _c5_metric(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _worst_shift_and_corr(h: np.ndarray):
    """Return (shift, correlation array) where the shift gives the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift ∈ [-(n-1), n-1]
    return shift, corr


def _gradient_worst(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst‑shift overlap w.r.t. h (assuming only that shift)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        d = shift
        g[: n - d] += (1.0 - h[d:])
        g[d:] -= h[: n - d]
    else:
        d = -shift
        g[d:] += (1.0 - h[: n - d])
        g[: n - d] -= h[d:]
    return g


def _propose_random_pair(h: np.ndarray):
    """Random feasible pair move (i receives, j loses)."""
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_bound = min(1.0 - hi, hj)
    if lo >= hi_bound:
        return None, None, None
    delta = random.uniform(lo, hi_bound)
    return i, j, delta


def _propose_gradient_pair(h: np.ndarray, g: np.ndarray, T: float, T0: float):
    """Move guided by the gradient of the worst shift."""
    pos = np.where(g > 1e-12)[0]
    neg = np.where(g < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return None, None, None
    i = pos[np.argmax(g[pos])]
    j = neg[np.argmin(g[neg])]
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_bound = min(1.0 - hi, hj)
    # step size that shrinks with temperature
    max_step = 0.5 * (T / T0) + 0.01
    lo = max(lo, -max_step)
    hi_bound = min(hi_bound, max_step)
    # we need a negative delta (i decreases, j increases)
    upper = min(hi_bound, -1e-12)
    lower = lo
    if lower >= upper:
        return None, None, None
    delta = random.uniform(lower, upper)
    return i, j, delta


def _adjust_sum(h: np.ndarray):
    """Enforce ∑h = n/2 exactly while staying in [0,1]."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for idx in np.random.permutation(n):
        lo = max(0.0, h[idx] + diff)
        hi = min(1.0, h[idx] + diff)
        if lo <= hi:
            h[idx] += diff
            break
    else:
        # fallback (unlikely)
        h *= target / h.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

    """Return (best_h, best_c) found from start_h within time_limit seconds."""
