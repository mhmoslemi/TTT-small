def refine_solution(h, max_iter, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    n = len(h)
    for _ in range(max_iter):
        # choose two distinct indices
        i, j = rng.integers(0,n, size=2)
        if i == j: continue
        # direction: move mass from i to j
        if best[i] <= 1e-12 or best[j] >= 1 - 1e-12: continue
        # potential delta range
        delta = min(best[i], 1 - best[j])
        if delta <= 0: continue
        # choose epsilon random fraction of delta, up to maybe 0.5*delta
        eps = rng.random() * delta * 0.5 + 1e-6
        new = best.copy()
        new[i] -= eps
        new[j] += eps
        # ensure bounds
        # evaluate
        c5 = compute_c5(new)
        if c5 < best_c5 - 1e-9:
            best = new
            best_c5 = c5
    return best, best_c5

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def normalize_h(vec: np.ndarray, n: int) -> np.ndarray:
    """Scale to exact integral, clip to [0,1], and correct any residual sum error."""
    target = n / 2.0
    if vec.sum() == 0:
        vec = np.full_like(vec, target / n)
    else:
        vec = vec * (target / vec.sum())
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) > 1e-12:
        vec += diff / n
        vec = np.clip(vec, 0.0, 1.0)
    return vec


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, then renormalise."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    return normalize_h(up, new_n)


def generate_random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1] with the required sum."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    # final uniform correction
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def generate_plateau_shape(n: int, a_frac: float) -> np.ndarray:
    """
    Piecewise linear shape:
      - plateau of height 1 on [0, a]
      - linear drop to 0 on [a, 2-a]
      - zero afterwards.
    a_frac ∈ [0,1] is the plateau length (in the continuous interval).
    """
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    a = a_frac
    b = 2.0 - a
    h = np.where(x <= a, 1.0,
                 np.where(x < b, (b - x) / (b - a), 0.0))
    return normalize_h(h, n)


# ------------------------------------------------------------
# Optimisation helpers
# ------------------------------------------------------------
def slsqp_optimize(start: np.ndarray, n: int, deadline: float) -> tuple:
    """
    Multi‑stage SLSQP: progressively increase the softmax α.
    Returns (h_opt, c5) or (None, None) if time runs out.
    """
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    alphas = [5.0, 15.0, 30.0, 80.0, 150.0]  # continuation schedule
    cur = start.copy()

    for alpha in alphas:
        if time.time() > deadline:
            return None, None
        res = minimize(
            lambda h: softmax_c5(h, alpha=alpha),
            cur,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            # if the optimizer fails, keep the current point and continue
            cur = np.clip(res.x, 0.0, 1.0)
        else:
            cur = np.clip(res.x, 0.0, 1.0)

    # final projection onto the feasible set
    cur = normalize_h(cur, n)
    c5 = compute_c5(cur)
    return cur, c5


def hill_climb_refine(h: np.ndarray, rng: np.random.Generator,
                      deadline: float, max_iter: int = 20000) -> tuple:
    """Simple pairwise mass‑transfer hill‑climb."""
    n = h.shape[0]
    best = h.copy()
    best_c5 = compute_c5(best)
    while max_iter > 0 and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random amount from i to j respecting bounds
        max_move = min(best[i], 1.0 - best[j])
        if max_move <= 0:
            continue
        eps = rng.random() * max_move
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        c5 = compute_c5(cand)
        if c5 < best_c5 - 1e-12:
            best = cand
            best_c5 = c5
        max_iter -= 1
    return best, best_c5
