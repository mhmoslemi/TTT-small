    total_time = budget_s

    # step schedule: start_step=0.5 maybe; decay factor
    init_step = 0.5

    iteration = 0

    while time.time() < deadline:

        shift, max_corr = _compute_worst_shift_and_corr(current_h)
        # if shift zero, we can break - but unlikely as max_corr may be zero only if perfect overlap.

            break

        # direction

def _slsqp_refine(h0, target, time_limit):
    n = len(h0)
    bounds = [(0.0,1.0)]*n
    eq_cons = {"type":"eq","fun":lambda x: x.sum() - target}
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * (2.0/n)
    # Use minimize with maxiter=200
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=[eq_cons], options={'maxiter':200, 'ftol':1e-9,'disp':False})
    if res.success:
        return np.clip(res.x,0,1)
    else:
        return h0

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """Project x onto the hyper‑plane sum(x)=target while staying in [0,1]."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _worst_shift(h):
    """Return (shift, max_correlation) for the worst overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr[idx]

def _hill_climb(h, cur_c5, deadline, rng):
    """Simple pairwise mass‑exchange hill climbing."""
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose number of points and initialise
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        # even length makes the integral exact with binary values
        n = 44
        target = n / 2.0
        h0 = np.zeros(n, dtype=np.float64)
        h0[:int(target)] = 1.0
        rng.shuffle(h0)
    target = n / 2.0
    best_h = _adjust_sum(np.clip(h0, 0.0, 1.0), target)
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Random exploration phase (quickly samples many candidates)
    # ------------------------------------------------------------------
    exploration_deadline = start + budget_s * 0.15
    while time.time() < exploration_deadline:
        cand = rng.random(n)
        cand = _adjust_sum(np.clip(cand, 0.0, 1.0), target)
        cand_c5 = _evaluate_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_c5

    # ------------------------------------------------------------------
    # 3. Gradient‑descent‑style refinement on the worst shift
    # ------------------------------------------------------------------
    current_h = best_h.copy()
    current_c5 = best_c5
    total_time = budget_s
    # leave a small margin for final hill‑climb
    grad_deadline = start + budget_s - 1.0
    init_step = 0.5
    no_improve = 0
    max_no_imp = 2000

    while time.time() < grad_deadline:
        # linear decay of step size
        remaining = grad_deadline - time.time()
        frac = remaining / total_time
        step = init_step * frac

        shift, _ = _worst_shift(current_h)
        if shift == 0:
            break  # cannot improve further

        l = abs(shift)
        d = np.zeros(n, dtype=np.float64)
        if l < n:
            d[:n - l] += current_h[l:]
            d[l:] += current_h[:n - l]

        d -= np.mean(d)                     # keep sum zero before projection
        new_h = current_h + step * d
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = _adjust_sum(new_h, target)
        new_c5 = _evaluate_c5(new_h)

        if new_c5 + 1e-12 < current_c5:
            current_h, current_c5 = new_h, new_c5
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = new_h, new_c5
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_imp:
            # restart near the best solution or completely random
            if rng.random() < 0.6:
                cand = best_h + rng.normal(scale=0.1, size=n)
            else:
                cand = rng.random(n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = _adjust_sum(cand, target)
            cand_c5 = _evaluate_c5(cand)
            current_h, current_c5 = cand, cand_c5
            no_improve = 0

    # ------------------------------------------------------------------
    # 4. A few SLSQP refinements (max‑overlap objective)
    # ------------------------------------------------------------------
    def obj(x):
        return _evaluate_c5(x)

    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n

    slsqp_deadline = start + budget_s - 0.5
    while time.time() < slsqp_deadline - 2.0:
        # jitter around the current best
        x0 = best_h + rng.normal(scale=0.04, size=n)
        x0 = np.clip(x0, 0.0, 1.0)
        x0 = _adjust_sum(x0, target)

        res = minimize(
            obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        cand = np.clip(res.x, 0.0, 1.0) if res.success else x0
        cand = _adjust_sum(cand, target)
        cand_c5 = _evaluate_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_c5
            current_h, current_c5 = cand, cand_c5

    # ------------------------------------------------------------------
    # 5. Final fine‑grained hill climbing
    # ------------------------------------------------------------------
    final_deadline = start + budget_s
    best_h, best_c5 = _hill_climb(best_h, best_c5, final_deadline, rng)

    # ------------------------------------------------------------------
    # 6. Return feasible solution
    # ------------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
