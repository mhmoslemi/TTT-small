def anneal(start_h, time_budget):
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = max_c5(h)

    cur_c = best_c
    start = time.time()
    T0 = 0.2
    T_end = 1e-5
    max_step = 0.2  # for delta moves
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        progress = elapsed / time_budget
        T = T0 * (T_end / T0) ** progress

        # decide move type
        if random.random() < 0.4:  # swap move
            i, j = random.sample(range(n), 2)
            # create new h by swapping
            hi, hj = h[i], h[j]
            # no need to check bounds because swapping values retains bounds.
            h[i], h[j] = hj, hi
            new_c = max_c5(h)
            accept = (new_c < cur_c) or (random.random() < math.exp(-(new_c - cur_c)/T))
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = hi, hj
        else:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)  # delta added to i, sub from j
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            # enforce step size
            low_c = max(low, -max_step)
            high_c = min(high, max_step)
            if low_c < high_c:
                delta = random.uniform(low_c, high_c)
            else:
                delta = random.uniform(low, high)
            # apply
            h[i] = hi + delta
            h[j] = hj - delta
            new_c = max_c5(h)
            accept = (new_c < cur_c) or (random.random() < math.exp(-(new_c - cur_c)/T))
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # revert
                h[i] = hi
                h[j] = hj
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    import time, random, math, numpy as np
    random.seed(seed)
    np.random.seed(seed)
    total_budget = float(budget_s)
    start_all = time.time()

    # Choose n points, maybe 400
    n = 400
    # Build initial vector from initial_h_values if exists else random
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x, x0, h0)
    else:
        h0 = np.full(n, 0.5, dtype=np.float64)
    # enforce sum exactly
    target = n / 2.0
    cur_sum = h0.sum()
    if cur_sum != 0:
        h0 = h0 * (target / cur_sum)
    np.clip(h0, 0.0, 1.0, out=h0)
    # Now adjust for any residual
    diff = target - h0.sum()
    if abs(diff) > 1e-12:
        # distribute to random index respecting bounds
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h0[idx] + diff <= hi:
                h0[idx] += diff
                break

    # allocate main annealing budget 0.85
    main_budget = total_budget * 0.85
    best_h, best_c = anneal(h0, main_budget)
    # rest of time: random restarts
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    # each restart with small budget maybe 0.05 of total remaining
    while remaining > 5.0:
        # generate random binary start
        start = np.zeros(n, dtype=np.float64)
        # choose exactly n/2 ones (some may be fractional if odd)
        target_ones = int(target)
        frac = target - target_ones
        indices = np.random.permutation(n)
        start[indices[:target_ones]] = 1.0
        if frac > 0:
            start[indices[target_ones]] = frac
        # run anneal on this start for limited time
        sub_budget = min(remaining * 0.3, 30.0)  # up to 30 seconds per restart
        cand_h, cand_c = anneal(start, sub_budget)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        elapsed = time.time() - start_all
        remaining = total_budget - elapsed

    # final polishing: small hill climbing passes
    for _ in range(5):
        improved = False
        for _ in range(4000):
            i, j = random.sample(range(n), 2)
            hi, hj = best_h[i], best_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:

            delta = random.uniform(low, high) * 0.02
            best_h[i] = hi + delta
            best_h[j] = hj - delta
            new_c = max_c5(best_h)
            if new_c + 1e-10 < best_c:
                best_c = new_c
                improved = True
            else:
                best_h[i] = hi
                best_h[j] = hj
        if not improved:
            break

    # enforce exact sum again
    diff = target - best_h.sum()
    if abs(diff) > 1e-9:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # final evaluation
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = max_c5(best_h)

    return best_h, float(final_c), int(n)

import time, random, math, numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """C5 value for a step function sampled by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_sum(h: np.ndarray, target: float) -> None:
    """Adjust h in‑place so that sum(h) == target while keeping 0≤h≤1."""
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in np.random.permutation(h.size):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + diff <= hi:
                h[idx] += diff
                break


def _initial_vector(n: int) -> np.ndarray:
    """Build a feasible start vector of length n."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        m = base.size
        x_src = np.linspace(0.0, 2.0, m, endpoint=False)
        x_dst = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_dst, x_src, base)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    _project_sum(h, n / 2.0)
    return h


def _anneal(start_h: np.ndarray, time_budget: float, bound_target: float = 0.38085) -> tuple:
    """Simulated‑annealing improvement."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0, T_end = 0.5, 1e-6          # temperature extremes
    max_step = 0.4                  # max absolute delta per move
    swap_prob = 0.3                 # probability to try a pure swap

    while time.time() - start < time_budget:
        prog = (time.time() - start) / time_budget
        T = T0 * (T_end / T0) ** prog

        # ---- move selection -------------------------------------------------
        if random.random() < swap_prob:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            # pure swap preserves sum and bounds automatically
            h[i], h[j] = hj, hi
            new_c = _c5_metric(h)
            dC = new_c - cur_c
            accept = dC < 0 or (T > 0 and random.random() < math.exp(-dC / T))
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h.copy()
                if best_c <= bound_target:
                    break
            else:
                h[i], h[j] = hi, hj
        else:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]

            low = max(-hi, hj - 1.0)          # keep both inside [0,1]
            high = min(1.0 - hi, hj)
            if low >= high:
                continue

            low_c = max(low, -max_step)
            high_c = min(high, max_step)
            if low_c < high_c:
                delta = random.uniform(low_c, high_c)
            else:
                delta = random.uniform(low, high)

            h[i] = hi + delta
            h[j] = hj - delta
            new_c = _c5_metric(h)
            dC = new_c - cur_c
            accept = dC < 0 or (T > 0 and random.random() < math.exp(-dC / T))
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h.copy()
                if best_c <= bound_target:
                    break
            else:
                h[i], h[j] = hi, hj

    return best_h, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C₅ bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # --- problem size -------------------------------------------------------
    n = 400                     # moderate resolution, still cheap to evaluate
    target_sum = n / 2.0

    # --- primary start -------------------------------------------------------
    h0 = _initial_vector(n)

    # --- main annealing (most of the time) ----------------------------------
    main_budget = total_budget * 0.85
    best_h, best_c = _anneal(h0, main_budget)

    # --- random binary restarts with shorter anneals ------------------------
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    while remaining > 6.0:                       # at least a few seconds per run
        # binary start with exactly n/2 ones (fractional tail if needed)
        start = np.zeros(n, dtype=np.float64)
        whole = int(target_sum)
        frac = target_sum - whole
        idx = np.random.permutation(n)
        start[idx[:whole]] = 1.0
        if frac > 0:
            start[idx[whole]] = frac

        sub_budget = min(remaining * 0.3, 30.0)
        cand_h, cand_c = _anneal(start, sub_budget)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        elapsed = time.time() - start_all
        remaining = total_budget - elapsed

    # --- final deterministic polishing ----------------------------------------
    for _ in range(4):
        improved = False
        for __ in range(3000):
            i, j = random.sample(range(n), 2)
            hi, hj = best_h[i], best_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = random.uniform(low, high) * 0.03
            best_h[i] = hi + delta
            best_h[j] = hj - delta
            new_c = _c5_metric(best_h)
            if new_c + 1e-10 < best_c:
                best_c = new_c
                improved = True
            else:
                best_h[i], best_h[j] = hi, hj
        if not improved:
            break

    # --- enforce exact sum one last time --------------------------------------
    _project_sum(best_h, target_sum)

    # --- final evaluation ------------------------------------------------------
    try:
        final_c = evaluate_erdos_solution(best_h)  # may be supplied by harness
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
