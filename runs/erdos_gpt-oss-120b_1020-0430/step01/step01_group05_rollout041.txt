   while time.time() < end_time:

       i,j = np.random.randint(0,n,2)

       if i == j: continue

       hi = h[i]; hj = h[j]

       # Determine allowed delta range as earlier
       delta_max = min(1.0 - hi, hj)
       delta_min = -min(hi, 1.0 - hj)

       if delta_max <= delta_min: continue

       # step size adapt based on some distribution; maybe use random normal with std 0.05 scaled to range.

       delta = np.random.uniform(delta_min, delta_max)

       # small magnitude cut:
       # Optionally incorporate probability for big moves.

       cand = h.copy()
       cand[i] = hi + delta
       cand[j] = hj - delta
       # No need to clamp as delta ensures within [0,1].
       # Compute C5

def generate_trapezoid(a,s,n):
    dx = 2/n
    # compute p = max(0, (1/a) - s) ; ensure non-negative
    p = max(0.0, (1.0/a) - s)
    # before shape, we need to position shape symmetrically at center.
    # total support length L = p + 2*s.
    L = p + 2*s
    # The shape extends from left = 1 - L/2 to right = 1 + L/2.

    x = np.arange(n)*dx + dx/2
    h = np.zeros(n)
    left = 1.0 - L/2
    right = 1.0 + L/2

    # compute for x between left and left+s: rising linear from 0 to a
    # for x between left+s and left+s+p: height a
    # for x between left+s+p to right: falling linear from a to 0

    # rising:
    mask_rise = (x >= left) & (x < left + s)
    if s>0:
        h[mask_rise] = a * (x[mask_rise] - left) / s
    # plateau:
    mask_plate = (x >= left + s) & (x <= left + s + p)
    h[mask_plate] = a
    # fall:
    mask_fall = (x > left + s + p) & (x <= right)
    if s>0:
        h[mask_fall] = a * (right - x[mask_fall]) / s

    # ensure bounds 0-1; should be.

    # Project onto sum = target (might already satisfy due to analytic area being exactly 1, but numeric errors). We'll call projection with target_sum.

    return project_onto_bounds_and_sum(h, target_sum)

import time, random, math
import numpy as np
from scipy.optimize import minimize

# Global for equality constraint
TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n ∩ {sum = target} by bisection on lambda."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Log‑sum‑exp smooth maximum."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth version of the max‑overlap objective."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Symmetric triangle of height 1 on [0,2]."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) (already has integral 1)."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to a new size."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h_new, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function on [0,2] with a low C₅ value."""
    global TARGET_SUM_GLOBAL
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # --------------------------------------------------------------
    # problem size – a moderate discretisation that keeps O(n²) cheap
    n = 300                     # even ⇒ target sum = n/2 integer
    dx = 2.0 / n
    target_sum = n / 2.0
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    best_h = None
    best_val = np.inf

    # --------------------------------------------------------------
    # seed vectors: parent (if any) + a few deterministic starts
    seed_vectors = []
    if "initial_h_values" in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))

    # --------------------------------------------------------------
    # ---------- 1. MULTIPLE SLSQP RESTARTS -----------------------
    restart = 0
    while time.time() - start < budget_s - 2.0:
        if restart < len(seed_vectors):
            h0 = seed_vectors[restart]
        else:
            # random feasible start
            rnd = np.random.rand(n)
            h0 = project_onto_bounds_and_sum(rnd, target_sum)

        restart += 1

        # homotopy on the smoothing parameter
        h_cur = h0
        for alpha in (5.0, 10.0, 20.0, 40.0):
            try:
                res = minimize(
                    objective_smooth,
                    h_cur,
                    args=(dx, alpha),
                    method="SLSQP",
                    bounds=bounds,
                    constraints={"type": "eq", "fun": sum_eq_constraint},
                    options={"maxiter": 400, "ftol": 1e-10, "disp": False},
                )
                h_cur = project_onto_bounds_and_sum(res.x, target_sum)
            except Exception:
                break
            if time.time() - start > budget_s - 2.0:
                break

        val = compute_c5(h_cur)
        if val < best_val:
            best_val = val
            best_h = h_cur.copy()
            if best_val <= 0.3808:
                break

    # --------------------------------------------------------------
    # ---------- 2. GREEDY PAIRWISE REFINEMENT --------------------
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    # greedy hill‑climb using pairwise mass transfers
    curr_h = best_h.copy()
    curr_val = best_val
    iters = 0
    step_limit = 0.02            # maximal absolute mass moved per step
    while time.time() - start < budget_s - 1.0:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = curr_h[i], curr_h[j]
        # feasible delta range preserving bounds
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        # restrict to a modest step size
        low = max(delta_min, -step_limit)
        high = min(delta_max, step_limit)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)
        cand = curr_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand_val = compute_c5(cand)
        # accept only if improvement
        if cand_val + 1e-12 < curr_val:
            curr_h = cand
            curr_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val <= 0.3808:
                    break
        # occasional re‑projection to correct tiny drift
        if iters % 5000 == 0:
            curr_h = project_onto_bounds_and_sum(curr_h, target_sum)
            curr_val = compute_c5(curr_h)
            if curr_val < best_val:
                best_val = curr_val
                best_h = curr_h.copy()
        iters += 1

    # final safety projection & exact evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
