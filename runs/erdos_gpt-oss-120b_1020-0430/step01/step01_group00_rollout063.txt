def hill_climb_binary(h, time_limit, target):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5

    # Precompute indices arrays:
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    # Ensure ones and zeros arrays updates after each swap (or we can maintain using set structures)
    # For simplicity recompute each iteration (cost O(n) negligible compared to correlation.

    start = time.time()
    T0 = 0.01
    while time.time() - start < time_limit:
        # random i from ones, j from zeros
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = _c5_from_h(cur_h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # cooling schedule
            elapsed = time.time() - start
            progress = elapsed / time_limit
            T = T0 * (1 - progress)  # linear cooling
            if T > 1e-9 and np.random.rand() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c5 = new_c5

            zeros = np.flatnonzero(cur_h == 0)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 < target:
                    break
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # maybe occasionally random restart? Not needed.

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # Get initial construction
    try:
        init_h_cont = np.array(initial_h_values, dtype=float)
        n0 = init_h_cont.size
    except NameError:
        init_h_cont = None
        n0 = None

    # Target bound
    target = 0.3808

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Helper compute C5
    def compute_c5(h):
        n = h.size
        dx = 2.0/n
        max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
        return max_corr * dx

    # Choose n candidates list: if init available, use its size, and 2x, 4x (<=300). Also include 256 maybe.
    n_candidates = []
    if n0:
        n_candidates.append(n0)
    # add multiples
    for mul in [2, 3]:
        if n0:
            n = n0 * mul
        else:
            n = 128 * mul
        if n <= 300:
            n_candidates.append(int(n))
    # Also maybe add 200,256 (if <=300)
    for n in (200,256):
        if n <= 300:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 5.0:
            break
        # Build starting binary h for this n
        if init_h_cont is not None and init_h_cont.size == n:
            # use continuous init, convert to binary
            init_binary = np.zeros(n, dtype=int)
            idx = np.argsort(-init_h_cont)[:n//2]

        elif init_h_cont is not None and init_h_cont.size != n:

        else:

        best_local_h = init_binary.copy()
        best_local_c5 = cur_c5

        # Record if better globally
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = init_binary.copy().astype(float), cur_c5, n
            if best_c5 < target:
                return best_h.tolist(), float(best_c5), int(best_n)

        # Parameters
        max_iter = int(5e6)  # not used

        time_for_n = min( (budget_s*0.7) / len(n_candidates), time_left()*0.8 )

        inner_start = time.time()

        T0 = 0.02

        while time.time() - inner_start < time_for_n and time_left() > 0.5:
            # Random swap
            # get indices of ones and zeros each time
            ones = np.flatnonzero(best_local_h == 1)
            zeros = np.flatnonzero(best_local_h == 0)
            if len(ones)==0 or len(zeros)==0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

            new_c5 = compute_c5(best_local_h)
            delta = new_c5 - best_local_c5

                if new_c5 < best_c5:
                    best_h = best_local_h.astype(float).copy()
                    best_c5 = new_c5
                    best_n = n

                progress = elapsed / time_for_n
                T = T0 * (1 - progress)
                if T > 1e-9 and np.random.rand() < np.exp(-delta / T):
                    # accept worse

                else:
                    # revert swap
                    best_local_h[i], best_local_h[j] = best_local_h[j], best_local_h[i]
            # occasionally random restart from best_h if stagnation
            # skip

def _project_to_feasible(h):
    # clip
    h = np.clip(h, 0.0, 1.0)
    target = h.size / 2.0
    # compute sum
    s = h.sum()
    if abs(s - target) <= 1e-12:
        return h
    # Use iterative water-filling:
    for _ in range(1000):
        diff = target - s
        # Determine indices that can be adjusted (not at bound in direction of diff)
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # update
        h[free] += delta
        # clip again
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    return h

import time

# Helper to enforce feasibility: 0 <= h <= 1 and sum(h) = n/2
def _project_to_feasible(h, max_iter=200):
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction (should be essentially zero)
    s = h.sum()
    if abs(s - target) > 1e-10:
        diff = target - s
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h

# C5 from a feasible vector (uses min autocorrelation)
def _c5_from_h(h):
    n = h.size
    # full autocorrelation of h with itself
    full = np.correlate(h, h, mode="full")
    # indices n-1 ... n-1+(n-1) correspond to non‑negative shifts 0..n-1
    corr_pos = full[n - 1 :]
    min_corr = np.min(corr_pos)
    # C5 = 1 - (2 * min_corr) / n   (since total integral of h is 1)
    return 1.0 - (2.0 * min_corr) / n


# ------------------------------------------------------------
# Gradient‑ascent on the *minimum* autocorrelation (i.e. minimise C5)
def _gradient_optimize(start_h, time_budget, target):
    n = start_h.size
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    start = time.time()
    step = 0.5  # initial step size

    while time.time() - start < time_budget:
        # autocorrelation
        full = np.correlate(h, h, mode="full")
        corr_pos = full[n - 1 :]
        min_corr = np.min(corr_pos)
        k_min = np.argmin(corr_pos)

        # gradient that increases the correlation for the worst shift
        g = np.zeros_like(h)
        if k_min > 0:
            g[: n - k_min] += h[k_min:]       # partner i+k
            g[k_min:] += h[: n - k_min]       # partner i‑k
        else:  # k_min == 0
            g = 2.0 * h

        # step and projection
        h_new = _project_to_feasible(h + step * g)
        c5_new = _c5_from_h(h_new)

        # keep the best ever seen
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if best_c5 < target:
                break

        # accept the move (gradient ascent) – even if slightly worse,
        # the projection keeps feasibility.
        h = h_new

        # slowly shrink the step
        step *= 0.9995

        # occasional tiny random kick to escape flat regions
        if np.random.rand() < 0.001:
            h = _project_to_feasible(h + 0.01 * np.random.randn(n))

    return best_h, best_c5
