def local_search(h, n, target, time_deadline):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    # initial step size fraction, maybe 0.1 relative to [0,1].
    step = 0.2
    # Temperature schedule: start at 0.01 maybe.
    # Since objective changes maybe ~0.1, we can set temperature start 0.001? We can ignore acceptance of worse for simplicity (only accept if better).
    # Use simple hill climbing with random increments.
    while time.time() < time_deadline:
        # choose i,j distinct
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # Determine max allowed delta for each to stay within [0,1]
        # delta positive means increase i decrease j.
        # compute lower/upper bound for delta:
        max_inc_i = 1.0 - cur_h[i]
        max_dec_i = -cur_h[i]  # negative
        max_inc_j = 1.0 - cur_h[j]
        max_dec_j = -cur_h[j]
        # permitted delta range: delta <= max_inc_i, delta >= max_dec_i, -delta <= max_inc_j, -delta >= max_dec_j
        # So delta in [max(max_dec_i, -max_inc_j), min(max_inc_i, -max_dec_j)]
        lower = max(-max_inc_j, max_dec_i)
        upper = min(max_inc_i, -max_dec_j)
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # ensure sum unchanged automatically.
        new_val = compute_c5(new_h)
        if new_val < cur_val - 1e-12:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
        # else maybe occasionally accept? We'll skip for simplicity.
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def project_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce sum == target by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h


def symmetrize(h: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry about the centre of the interval."""
    return (h + h[::-1]) * 0.5


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape (peak 1 at centre) normalised to integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    return project_feasible(tri, target)


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old grid to a finer one, with re‑normalisation."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    return project_feasible(up, new_n / 2.0)


def local_search(h: np.ndarray, n: int, target: float,
                 deadline: float, rng: np.random.Generator):
    """Simple hill‑climbing: swap mass between two points while keeping sum."""
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # feasible delta range so that both entries stay in [0,1]
        max_inc_i = 1.0 - cur_h[i]
        max_dec_i = -cur_h[i]
        max_inc_j = 1.0 - cur_h[j]
        max_dec_j = -cur_h[j]

        lower = max(-max_inc_j, max_dec_i)
        upper = min(max_inc_i, -max_dec_j)
        if lower >= upper:
            continue

        delta = rng.uniform(lower, upper)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        val = compute_c5(new_h)
        if val < cur_val - 1e-12:
            cur_h = new_h
            cur_val = val
            if val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = val

    # final symmetrisation & projection
    best_h = project_feasible(symmetrize(best_h), target)
    best_val = compute_c5(best_h)
    return best_h, best_val


def optimise_from_start(start: np.ndarray, n: int, target: float,
                        deadline: float, rng: np.random.Generator):
    """Multi‑stage SLSQP with increasing soft‑max sharpness, then local search."""
    h = project_feasible(start, target)
    h = symmetrize(h)

    best_h = h
    best_c5 = compute_c5(best_h)

    alphas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    cons = {"type": "eq", "fun": lambda x, t=target: np.sum(x) - t}
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() > deadline:
            break

        def obj(x):
            return softmax_c5(x, alpha)

        res = minimize(
            obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 600, "ftol": 1e-9, "disp": False},
        )
        h_candidate = res.x if res.success else h
        h_candidate = project_feasible(h_candidate, target)
        h_candidate = symmetrize(h_candidate)

        c5_val = compute_c5(h_candidate)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_candidate

        h = h_candidate  # warm‑start next stage

    # short hill‑climb, using at most half of the remaining budget (capped at 30 s)
    remaining = deadline - time.time()
    if remaining > 1.0:
        ls_deadline = time.time() + min(30.0, remaining * 0.5)
        h_ls, c5_ls = local_search(best_h, n, target, ls_deadline, rng)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls

    return best_h, best_c5

    """

    Returns (h_values, c5_bound, n_points).
    """

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s * 0.95

    # Try to use the parent construction if supplied

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in candidate_ns:
        if time.time() > deadline:
            break

        target_sum = n / 2.0
        starts = []

        # 1. Parent (or its up‑sample)
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # 2. Classic triangular shape
        starts.append(generate_triangular(n))

        # 3. Simple binary block (half ones, half zeros)
        blk = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        starts.append(project_feasible(blk, target_sum))

        # 4. A few random feasible vectors
        for _ in range(5):
            rand = rng.random(n)
            rand = rand * (target_sum / rand.sum())
            rand = project_feasible(rand, target_sum)
            starts.append(rand)

        # 5. Perturbed triangular shapes
        tri = generate_triangular(n)
        for _ in range(3):
            pert = tri + rng.uniform(-0.15, 0.15, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_feasible(pert, target_sum)
            starts.append(pert)

        # Optimize each start
        for s in starts:
            if time.time() > deadline:
                break
            h_opt, c5_opt = optimise_from_start(s, n, target_sum, deadline, rng)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt, c5_opt, n

    # If a decent solution was found, try a final refinement on a finer grid
    if best_h is not None and time.time() < deadline - 5.0:
        fine_n = best_n * 2
        if fine_n <= 500:  # keep the refinement modest
            fine_h = upsample(best_h, fine_n)
            fine_h, fine_c5 = optimise_from_start(
                fine_h, fine_n, fine_n / 2.0, deadline, rng
            )
            if fine_c5 < best_c5:

    # Final safety post‑processing
    if best_h is not None:
        best_h = project_feasible(best_h, best_n / 2.0)
        best_h = symmetrize(best_h)
        best_c5 = compute_c5(best_h)
    else:
        # Fallback: simple binary block (guaranteed feasible)
        best_n = 100
        best_h = np.concatenate([np.ones(best_n // 2), np.zeros(best_n - best_n // 2)]).astype(float)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
