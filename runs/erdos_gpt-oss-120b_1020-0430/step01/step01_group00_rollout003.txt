def gradient_descent(h0, n_iter=200, lr=0.5, lambda_start=5, lambda_mul=1.02):

import time
import numpy as np
from scipy import optimize

def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip h to [0,1] and enforce sum(h)=n/2 using water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny fix for any leftover error
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Maximum overlap integral for a given discretised h."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer one."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _weighted_corr_gradient(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of sum_k w_k * corr_k where corr_k = sum_i h_i (1-h_{i+k})."""
    n = h.size
    g = 1.0 - h
    grad = np.zeros_like(h)
    shift_center = n - 1
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        s = idx - shift_center          # shift value, could be negative
        if s >= 0:
            length = n - s
            # +wi * g[i+s] for i in [0, length)
            grad[:length] += wi * g[s:s + length]
            # -wi * h[i] added to position i+s
            grad[s:s + length] -= wi * h[:length]
        else:
            t = -s
            length = n - t
            # +wi * g[i] contributes to h[i+t]
            grad[t:] += wi * g[:length]
            # -wi * h[i+t] contributes to position i
            grad[:length] -= wi * h[t:t + length]
    return grad


def _gd_optimize(h0: np.ndarray, n: int, time_left, max_iter: int = 250) -> tuple:
    """Projected gradient descent on a smooth approximation of max‑correlation."""
    h = h0.copy()
    best_h, best_c5 = h.copy(), _c5_from_h(h)
    lr = 0.5            # initial step size
    lam = 1.0           # smoothing parameter
    for it in range(max_iter):
        if time_left() < 2.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = corr.max()
        # soft‑max weights (stable)
        w = np.exp(lam * (corr - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            break
        w /= w_sum
        grad = _weighted_corr_gradient(h, w)
        h -= lr * grad
        h = _project_to_feasible(h, n)
        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        # simple schedules
        lr *= 0.99
        lam *= 1.02
    return best_h, best_c5


def _ineq_constraint(x):
    """t - correlation >= 0 for all shifts."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full")


def _eq_constraint(x):
    """Sum of h must equal n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _objective(x):
    """Minimise the auxiliary variable t."""
    return x[-1]


def _solve_h(initial_h: np.ndarray, max_iter: int = 250) -> tuple:
    """Refine a feasible h using SLSQP (minimise the max correlation)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(_objective, x0, method="SLSQP", bounds=bounds,
                            constraints=cons, options=opts)
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _local_search(h: np.ndarray, n: int, time_left, max_iter: int = 500) -> tuple:
    """Simple pairwise transfer hill‑climb keeping the sum fixed."""
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    step = 0.05
    for _ in range(max_iter):
        if time_left() < 1.0:
            break
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        delta = min(step, best_h[i], 1.0 - best_h[j])
        if delta <= 0.0:
            continue
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        c5 = _c5_from_h(cand)
        if c5 < best_c5:
            best_h, best_c5 = cand, c5
        else:
            step *= 0.99
    return best_h, best_c5
