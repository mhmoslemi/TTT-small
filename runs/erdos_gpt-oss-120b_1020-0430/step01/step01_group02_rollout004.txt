    if res.success:

        x = _adjust_sum(x, target)  # ensure exactly sum
        return x
    else:
        return h

def _simulated_annealing(start_h, target, deadline, rng, start_time):
    n = len(start_h)

    cur_h = start_h.copy()
    cur_c5 = _evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    init_temp = 0.001
    final_temp = 1e-6
    iter_no_improve = 0
    restart_threshold = 50000  # maybe large

    iteration = 0
    while time.time() < deadline:
        # compute temperature
        temp = max(final_temp, init_temp * (0.9999 ** iteration))
        # choose random src,dst with valid transfer
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1  # ensure j!=i
        # direction random
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        # compute max_move
        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move < 1e-12:
            continue
        # choose delta
        if rng.random() < 0.25:
            delta = max_move
        else:
            delta = max_move * rng.random()
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        # evaluate
        new_c5 = _evaluate_c5(new_h)
        delta_score = new_c5 - cur_c5
        if delta_score < -1e-12 or rng.random() < math.exp(-delta_score / temp):
            cur_h = new_h
            cur_c5 = new_c5
            iter_no_improve = 0
        else:
            iter_no_improve += 1
        if cur_c5 < best_c5 - 1e-12:
            best_h = cur_h.copy()
            best_c5 = cur_c5
            iter_no_improve = 0
        # restart if stuck

            cur_h = _random_binary_initial(n, target, rng)
            cur_c5 = _evaluate_c5(cur_h)
            iter_no_improve = 0

def _random_binary_initial(n, target, rng):

    h = np.zeros(n, dtype=np.float64)

    return h

import time
import math
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        idx = np.where(mask)[0]
        per = diff / len(idx)
        for i in idx:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _random_initial_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (continuous values)."""
    v = rng.random(n)
    return _adjust_sum(v, target)

def _simulated_annealing(start_h: np.ndarray,
                         target: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple SA with pairwise mass transfers."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c5 = _evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    init_temp = 1e-3
    final_temp = 1e-6
    iter_no_improve = 0
    restart_after = 120_000  # iterations without improvement → restart
    it = 0

    while time.time() < deadline:
        # temperature schedule (exponential decay)
        temp = max(final_temp, init_temp * (0.9999 ** it))

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_move = min(cur_h[src], 1.0 - cur_h[dst])
        if max_move <= 1e-12:
            continue

        # sometimes take the full amount, otherwise a random fraction
        if rng.random() < 0.25:
            delta = max_move
        else:
            delta = max_move * rng.random()

        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta

        new_c5 = _evaluate_c5(new_h)
        dscore = new_c5 - cur_c5

        accept = False
        if dscore < -1e-12:
            accept = True
        elif temp > 1e-12 and rng.random() < math.exp(-dscore / temp):
            accept = True

        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        if cur_c5 < best_c5 - 1e-12:
            best_h = cur_h.copy()
            best_c5 = cur_c5
            iter_no_improve = 0
            # early exit if we already beat the target bound
            if best_c5 < 0.3808:
                break

        # occasional jitter to keep diversity
        if it % 15000 == 0 and rng.random() < 0.07:
            jitter = cur_h.copy()
            idxs = rng.choice(n, size=min(5, n), replace=False)
            jitter[idxs] += (rng.random(len(idxs)) - 0.5) * 0.2
            jitter = np.clip(jitter, 0.0, 1.0)
            jitter = _adjust_sum(jitter, target)
            cur_h = jitter
            cur_c5 = _evaluate_c5(cur_h)

        # restart if stuck
        if iter_no_improve >= restart_after:
            cur_h = _random_initial_vector(n, target, rng)
            cur_c5 = _evaluate_c5(cur_h)
            iter_no_improve = 0

        it += 1

    return best_h, best_c5

def _local_refine(h: np.ndarray, target: float) -> np.ndarray:
    """One short SLSQP refinement."""
    n = h.shape[0]
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    try:
        res = minimize(obj,
                       h,
                       method="SLSQP",
                       bounds=bounds,
                       constraints=[cons],
                       options={"maxiter": 300, "ftol": 1e-10, "disp": False})
    except Exception:
        return h
    if not res.success:
        return h
    x = np.clip(res.x, 0.0, 1.0)
    x = _adjust_sum(x, target)
    return x

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # 1. Initialise
    # -------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        n = 48                     # even size → integer target
        target = n / 2.0
        h0 = _random_initial_vector(n, target, rng)

    target = n / 2.0
    h0 = _adjust_sum(np.clip(h0, 0.0, 1.0), target)
    best_h, best_c5 = h0.copy(), _evaluate_c5(h0)

    # -------------------------------------------------------------
    # 2. Simulated annealing (global search)
    # -------------------------------------------------------------
    deadline = start_time + budget_s - 1.0   # keep a second for final steps
    h_sa, c5_sa = _simulated_annealing(best_h, target, deadline, rng)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    # -------------------------------------------------------------
    # 3. Local refinement via SLSQP
    # -------------------------------------------------------------
    refined = _local_refine(best_h, target)
    refined_c5 = _evaluate_c5(refined)
    if refined_c5 < best_c5:
        best_h, best_c5 = refined, refined_c5

    # -------------------------------------------------------------
    # 4. Final feasibility check and return
    # -------------------------------------------------------------
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
