def softmax_obj_and_grad(h, alpha):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_c = conv.max()
    shifted = conv - max_c
    exp_alpha = np.exp(alpha * shifted)
    sum_exp = exp_alpha.sum()
    obj = (np.log(sum_exp) / alpha) + max_c  # = log-sum-exp / alpha
    # compute weights
    w = exp_alpha / sum_exp
    # gradient
    grad = np.zeros_like(h)
    # Loop over all lags
    # Use for loop over l in range(2*n-1)
    for idx in range(2 * n - 1):
        wl = w[idx]
        if wl == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            i_slice = slice(0, n - s)
            j_slice = slice(s, n)
        else:
            t = -s
            i_slice = slice(t, n)
            j_slice = slice(0, n - t)
        # contributions
        grad[i_slice] += wl * (1.0 - h[j_slice])
        grad[j_slice] += wl * (-h[i_slice])
    return obj, grad

def project_box_sum(y, target, tol=1e-12, max_iter=60):
    # binary search for lambda to satisfy sum after clipping
    low = -np.max(y) - 1.0
    high = 1.0 - np.min(y) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = mid
        else:
            low = mid
    x = np.clip(y + (low + high) / 2.0, 0.0, 1.0)
    # Ensure sum within tolerance
    # If slight off due to clipping, adjust uniformly within allowed bounds
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff over free variables (not at bounds)
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback uniform adjust then clip
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_obj_and_grad(h: np.ndarray, alpha: float):
    """
    Smooth approximation of the max overlap and its gradient.
    Returns (obj, grad) where obj = (1/alpha) * logsumexp(alpha * conv).
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_c = conv.max()
    shifted = conv - max_c
    exp_a = np.exp(alpha * shifted)
    sum_exp = exp_a.sum()
    obj = (np.log(sum_exp) / alpha) + max_c  # = log‑sum‑exp / alpha

    # softmax weights
    w = exp_a / sum_exp

    grad = np.zeros_like(h)
    # loop over all lags
    for idx, wl in enumerate(w):
        if wl == 0.0:
            continue
        s = idx - (n - 1)   # shift amount (negative/positive)
        if s >= 0:
            i_slice = slice(0, n - s)
            j_slice = slice(s, n)
        else:
            t = -s
            i_slice = slice(t, n)
            j_slice = slice(0, n - t)

        # contributions to gradient
        grad[i_slice] += wl * (1.0 - h[j_slice])
        grad[j_slice] += wl * (-h[i_slice])
    return obj, grad


def obj_penalized(x: np.ndarray, alpha: float, target: float, lam: float):
    """Objective + quadratic penalty for the integral constraint."""
    obj, grad = softmax_obj_and_grad(x, alpha)
    diff = x.sum() - target
    obj += lam * diff * diff
    grad += 2.0 * lam * diff * np.ones_like(x)
    return obj, grad


def obj_penalized_with_grad(x, alpha, target, lam):
    """Wrapper returning (value, gradient) for scipy.optimize."""
    return obj_penalized(x, alpha, target, lam)


def project_box_sum(y: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 60):
    """
    Euclidean projection of y onto {x: sum x = target, 0 ≤ x ≤ 1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low = -np.max(y) - 1.0
    high = 1.0 - np.min(y) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = mid
        else:
            low = mid
    x = np.clip(y + (low + high) / 2.0, 0.0, 1.0)

    # tiny adjustment if still off (unlikely)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, then re‑project."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    up = project_box_sum(up, target)
    return up


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular profile, then re‑project to feasibility."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    tri = project_box_sum(tri, target)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector."""
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    a = project_box_sum(a, target)
    return a


# ------------------------------------------------------------
# Parametric shape (piecewise linear with flat top)
# ------------------------------------------------------------
def parametric_shape(n: int, t: float) -> np.ndarray:
    """
    Symmetric shape with slope length `t` (0 ≤ t ≤ 1) and flat top.
    t = 0 gives constant 0.5, t = 1 gives the pure triangle.
    """
    if t <= 0.0:
        h = np.full(n, 0.5)
    else:
        M = 1.0 / (2.0 - t)          # maximal height
        x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        h = np.empty_like(x)

        left = x < t
        right = x > (2.0 - t)
        middle = ~ (left | right)

        # left ramp
        h[left] = M * x[left] / t
        # flat part
        h[middle] = M
        # right ramp (mirror)
        h[right] = M * (2.0 - x[right]) / t

    target = n / 2.0
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h = h + (target - cur) / n
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, target)
    return h


def best_parametric(n: int, time_deadline: float):
    """
    Coarse‑to‑fine grid search for the optimal slope length `t`.
    Returns (h_best, c5_best, t_best).
    """
    best_c5 = np.inf
    best_h = None
    best_t = 0.0

    # coarse grid
    for t in np.linspace(0.0, 1.0, 51):
        if time.time() > time_deadline:
            break
        h = parametric_shape(n, t)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h, best_t = c5, h, t

    # refinement passes
    for _ in range(3):
        if time.time() > time_deadline:
            break
        step = 1.0 / (2 ** (5 + _))   # decreasing step size
        low = max(0.0, best_t - step)
        high = min(1.0, best_t + step)
        for t in np.linspace(low, high, 31):
            if time.time() > time_deadline:
                break
            h = parametric_shape(n, t)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h, best_t = c5, h, t

    return best_h, best_c5, best_t


# ------------------------------------------------------------
# Local optimisation with graduated smooth max
# ------------------------------------------------------------
def optimise_h(start: np.ndarray, n: int, target: float, deadline: float) -> np.ndarray:
    """
    Runs a few L‑BFGS‑B stages with increasing smoothness.
    Returns a feasible refined vector.
    """
    h = project_box_sum(start, target)
    bounds = [(0.0, 1.0)] * n
    alphas = [5.0, 15.0, 30.0, 80.0, 200.0]
    lam = 10.0

    for alpha in alphas:
        if time.time() > deadline:
            break
        res = minimize(obj_penalized_with_grad,
                       h,
                       args=(alpha, target, lam),
                       method="L-BFGS-B",
                       jac=True,
                       bounds=bounds,
                       options={"maxiter": 300, "ftol": 1e-12})
        h = np.clip(res.x, 0.0, 1.0)
        h = project_box_sum(h, target)
        lam *= 10.0
    return h


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    # -----------------------------------------------------------------
    # Base problem size – use parent size if available, otherwise 120.
    # -----------------------------------------------------------------
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        parent_n = None

    base_n = parent_n if parent_n is not None else 120
    if base_n % 2 != 0:
        base_n += 1

    target_base = base_n / 2.0

    # -----------------------------------------------------------------
    # Collect initial candidates at the base resolution.
    # -----------------------------------------------------------------
    candidates = []

    if parent_h is not None:
        if parent_n == base_n:
            candidates.append(parent_h.copy())
        else:
            candidates.append(upsample(parent_h, base_n))

    candidates.append(generate_triangular(base_n))

    for _ in range(5):
        candidates.append(random_start(base_n, rng))

    best_h = None
    best_c5 = np.inf
    best_n = base_n

    # -----------------------------------------------------------------
    # Optimise all candidates at the base resolution.
    # -----------------------------------------------------------------
    for start_vec in candidates:
        if time.time() > time_limit:
            break
        h_opt = optimise_h(start_vec, base_n, target_base, time_limit)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, base_n

    # -----------------------------------------------------------------
    # Parametric family search (piecewise linear with flat top).
    # -----------------------------------------------------------------
    param_n = 300 if best_n < 300 else best_n  # ensure at least moderate resolution
    if time.time() < time_limit:
        param_h, param_c5, _ = best_parametric(param_n, time_limit)
        if param_c5 < best_c5:
            best_h, best_c5, best_n = param_h, param_c5, param_n

    # -----------------------------------------------------------------
    # Refine the best found shape at higher resolutions if time permits.
    # -----------------------------------------------------------------
    refinement_levels = [200, 300, 400, 500]
    for n_target in refinement_levels:
        if time.time() > time_limit:
            break
        if n_target <= best_n:
            continue
        # upsample current best shape
        up_h = upsample(best_h, n_target)
        target_t = n_target / 2.0
        # a couple of starts (the upsampled version and a fresh random one)
        starts = [up_h, random_start(n_target, rng)]
        for s in starts:
            if time.time() > time_limit:
                break
            h_opt = optimise_h(s, n_target, target_t, time_limit)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n_target

    # -----------------------------------------------------------------
    # Final symmetrisation (helps marginally) and final feasibility fix.
    # -----------------------------------------------------------------
    if best_h is not None:
        sym = (best_h + best_h[::-1]) / 2.0
        sym = project_box_sum(sym, best_n / 2.0)
        c5_sym = compute_c5(sym)
        if c5_sym < best_c5:
            best_h, best_c5 = sym, c5_sym

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
