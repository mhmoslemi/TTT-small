def _gradient_smoothed_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    # shape 2n-1
    # compute softmax weights
    max_corr = corr.max()
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / exp_vals.sum()
    # compute weighted sum of gradients
    # For each shift index k (0..2n-2), shift = k-(n-1)
    # We'll accumulate gradient.
    grad = np.zeros_like(h)
    # Use loop over shifts. Could vectorize using convolution but loop is okay for n up to 400.
    for k, wk in enumerate(w):
        if wk == 0:
            continue
        shift = k - (n - 1)
        grad += wk * _gradient_for_shift(h, shift)
    return grad

def _adam_optimize(h0, target, max_iters=2000, step0=0.1, decay=0.9995, alpha_start=5, alpha_end=100, alpha_ramp_iters=1000, K=5, smoothed_interval=10):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    step = step0
    for t in range(1, max_iters+1):
        # compute alpha linearly ramping
        if t <= alpha_ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * (t-1) / alpha_ramp_iters
        else:
            alpha = alpha_end

        # compute gradient
        if t % smoothed_interval == 0:
            # smoothed gradient with top K shifts
            corr = np.correlate(h, 1-h, mode='full')
            # pick top K shifts

        # if C5 increased, reduce step
        # We'll compute C5 before and after but cost extra correlation; maybe cheap.
        # We'll check every 100 iterations.

        step *= decay

def _adam_optimize(h0, target, max_iters, step0, decay, alpha_start, alpha_end, alpha_ramp_iters, smoothed_interval, K, rng):

    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    step = step0
    n = len(h)

    for t in range(1, max_iters+1):
        # compute alpha
        if alpha_ramp_iters > 0:
            if t <= alpha_ramp_iters:
                alpha = alpha_start + (alpha_end - alpha_start) * (t-1) / alpha_ramp_iters
            else:
                alpha = alpha_end
        else:
            alpha = alpha_end

        # compute gradient
        if smoothed_interval > 0 and (t % smoothed_interval == 0):
            corr = np.correlate(h, 1-h, mode='full')
            K_use = min(K, corr.size)
            # select top K
            top_idx = np.argpartition(corr, -K_use)[-K_use:]
            top_vals = corr[top_idx]
            # weights using softmax with alpha
            max_top = top_vals.max()
            exp_vals = np.exp(alpha * (top_vals - max_top))
            w = exp_vals / exp_vals.sum()
            grad = np.zeros_like(h)
            for idx, wk in zip(top_idx, w):
                shift = idx - (n - 1)
                grad += wk * _gradient_for_shift(h, shift)
        else:
            # max shift only
            corr = np.correlate(h, 1-h, mode='full')
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            grad = _gradient_for_shift(h, shift)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - step * m_hat / (np.sqrt(v_hat) + epsilon)
        # project
        h = _project_onto_capped_simplex(h, target)
        # decay step
        step *= decay
    return h

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# --------------------------------------------------------------
# Correlation (C5) evaluation
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


# --------------------------------------------------------------
# Gradient of a single shift of the correlation
# --------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # first term: derivative of h_i * (1 - h_{i+s})
        g[: n - s] = 1.0 - h[s:]
        # second term: derivative via h_{i+s}
        g[s:] += -h[: n - s]
    else:
        s = -shift
        # first term (i >= s)
        g[s:] = 1.0 - h[: n - s]
        # second term (i <= n-s-1)
        g[: n - s] += -h[s:]
    return g


# --------------------------------------------------------------
# Adam optimizer with projection onto the feasible set
# --------------------------------------------------------------
def _adam_optimize(h0: np.ndarray,
                   target: float,
                   max_iters: int = 2000,
                   step0: float = 0.5,
                   decay: float = 0.9995,
                   alpha_start: float = 5.0,
                   alpha_end: float = 80.0,
                   alpha_ramp_iters: int = 800,
                   smoothed_interval: int = 10,
                   K: int = 5) -> np.ndarray:
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = step0
    n = len(h)

    for t in range(1, max_iters + 1):
        # schedule for the smoothness parameter
        if t <= alpha_ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / alpha_ramp_iters
        else:
            alpha = alpha_end

        # gradient computation
        if smoothed_interval > 0 and (t % smoothed_interval == 0):
            corr = np.correlate(h, 1.0 - h, mode='full')
            K_use = min(K, corr.size)
            # top‑K shifts
            top_idx = np.argpartition(corr, -K_use)[-K_use:]
            top_vals = corr[top_idx]
            # softmax weights (stable)
            max_top = top_vals.max()
            w_exp = np.exp(alpha * (top_vals - max_top))
            w = w_exp / w_exp.sum()
            grad = np.zeros_like(h)
            for idx, wk in zip(top_idx, w):
                shift = idx - (n - 1)
                grad += wk * _gradient_for_shift(h, shift)
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            grad = _gradient_for_shift(h, shift)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = h - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        h = _project_onto_capped_simplex(h, target)

        step *= decay

    return h


# --------------------------------------------------------------
# Simple up‑sampling (linear interpolation) utility
# --------------------------------------------------------------
def _upscale(vec: np.ndarray, n_target: int) -> np.ndarray:
    if len(vec) == n_target:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_new, x_old, vec)
