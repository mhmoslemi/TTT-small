def _c5_metric(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _make_feasible(h):
    # ensure sum = n/2
    # clip to [0,1], adjust sum by adding residual to random index
    target = len(h) / 2
    np.clip(h, 0, 1, out=h)
    cur = h.sum()
    diff = target - cur
    # if diff > 0, add to some index not at 1; else subtract from some >0
    if abs(diff) > 1e-12:
        # find indices that can accommodate diff
        # random shuffle indices
        idxs = np.random.permutation(len(h))
        for idx in idxs:
            lo, hi = 0.0, 1.0
            val = h[idx]
            if lo <= val + diff <= hi:
                h[idx] += diff
                break
    return h

def _anneal(start_h, time_limit):
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c
    start = time.time()
    T0 = 0.05
    T_end = 1e-5
    max_step = 0.3
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (T_end / T0) ** progress
        # choose move type
        if random.random() < 0.9:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            # bound delta:
            # Use temperature to limit step size: step_limit = max_step * (T/T0)  # if T decreases, step limit reduces
            step_limit = max_step * (T / T0)
            # but ensure step_limit positive >0
            low_cap = max(low, -step_limit)
            high_cap = min(high, step_limit)
            if low_cap < high_cap:
                delta = random.uniform(low_cap, high_cap)
            else:
                delta = random.uniform(low, high)
            # apply
            h[i] = hi + delta
            h[j] = hj - delta
        else:
            # swap
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            # No change in sum anyway
            delta = None

        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            if delta is not None:
                h[i] = hi
                h[j] = hj
            else:
                h[i], h[j] = hi, hj
    return best_h, best_c

import time
import math
import random
import numpy as np

def _c5_metric(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_feasible(h: np.ndarray) -> np.ndarray:
    """Clamp to [0,1] and fix the integral exactly (sum = n/2)."""
    n = h.shape[0]
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute the tiny remainder to a random admissible entry
    if diff > 0:
        candidates = np.where(h < 1.0)[0]
    else:
        candidates = np.where(h > 0.0)[0]
    if candidates.size:
        idx = np.random.choice(candidates)
        h[idx] += diff
        # final clamp just in case of tiny rounding errors
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _grid_search(n: int, max_time: float) -> tuple[np.ndarray, float]:
    """
    Very coarse 3‑step search.
    h = 1 on [0,a], = p on [a,a+b], = 0 elsewhere,
    with a + p*b = 1 and a in [0,1], p in [0,1].
    """
    target = n / 2.0
    dx = 2.0 / n
    best_c = float("inf")
    best_h = None

    a_vals = np.linspace(0.0, 1.0, 101)          # length of the 1‑segment
    p_vals = np.linspace(0.0, 1.0, 101)          # height of the middle segment

    start = time.time()
    for a in a_vals:
        for p in p_vals:
            # stop if time budget exhausted
            if time.time() - start > max_time:
                return best_h, best_c

            if p < 1e-12:
                if abs(a - 1.0) > 1e-9:
                    continue
                b = 0.0
            else:
                b = (1.0 - a) / p

            if b < -1e-12 or a + b > 2.0 + 1e-12:
                continue

            n1 = int(round(a / dx))
            n2 = int(round(b / dx))
            if n1 + n2 > n:
                n2 = n - n1

            h = np.zeros(n, dtype=np.float64)
            if n1 > 0:
                h[:n1] = 1.0
            if n2 > 0:
                h[n1:n1 + n2] = p
            _project_to_feasible(h)

            # test a few random rotations – the problem is not periodic
            for _ in range(3):
                shift = np.random.randint(0, n)
                h_shift = np.roll(h, shift)
                c = _c5_metric(h_shift)
                if c < best_c:
                    best_c = c
                    best_h = h_shift.copy()
    return best_h, best_c


def _anneal(start_h: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing with pairwise transfers (keeps sum exact)."""
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    T0 = 0.05
    T_end = 1e-6
    max_step_initial = 0.5

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (T_end / T0) ** progress          # exponential cooling
        max_step = max_step_initial * (1 - progress) + 1e-3

        # move type: transfer (90 %) or swap (10 %)
        if random.random() < 0.9:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            # restrict to modest steps most of the time
            low_cap = max(low, -max_step)
            high_cap = min(high, max_step)
            if low_cap < high_cap:
                delta = random.uniform(low_cap, high_cap)
            else:
                delta = random.uniform(low, high)
            h[i] = hi + delta
            h[j] = hj - delta
        else:
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            delta = None  # signals a pure swap

        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            if delta is not None:
                h[i] = hi
                h[j] = hj
            else:
                h[i], h[j] = hi, hj

    return best_h, best_c


def _hill_climb(h: np.ndarray, max_iter: int) -> tuple[np.ndarray, float]:
    """Zero‑temperature local search – keep only improving moves."""
    best = h.copy()
    n = best.shape[0]
    best_c = _c5_metric(best)

    for _ in range(max_iter):
        # find the lag giving the current maximum overlap
        corr = np.correlate(best, 1.0 - best, mode="full")
        lag_idx = int(np.argmax(corr))
        lag = lag_idx - (n - 1)

        i, j = random.sample(range(n), 2)
        hi, hj = best[i], best[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # direction that tends to reduce the maximal lag
        i_lag = i + lag
        j_lag = j + lag
        vi = best[i_lag] if 0 <= i_lag < n else 0.0
        vj = best[j_lag] if 0 <= j_lag < n else 0.0

        # delta should have sign opposite to (vi - vj)
        if vi < vj:
            low_dir, high_dir = max(low, 0.0), high
        elif vi > vj:
            low_dir, high_dir = low, min(high, 0.0)
        else:
            continue
        if low_dir >= high_dir:
            continue

        delta = random.uniform(low_dir, high_dir) * 0.05  # tiny tweak
        best[i] = hi + delta
        best[j] = hj - delta
        new_c = _c5_metric(best)
        if new_c + 1e-12 < best_c:
            best_c = new_c
        else:
            best[i] = hi
            best[j] = hj
    return best, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C₅ bound and the sample count."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # ------------------------------------------------------------
    # 1. Choose a decent resolution (even number, not too large)
    n = 600                     # 2 sec interval → dx≈0.00333
    if n % 2:
        n += 1

    # ------------------------------------------------------------
    # 2. Quick grid search for a good starting shape
    grid_time = total_budget * 0.07   # ~7 % of the budget
    best_h, best_c = _grid_search(n, grid_time)
    if best_h is None:                # fallback – uniform half density
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_h = _project_to_feasible(best_h)
        best_c = _c5_metric(best_h)

    # ------------------------------------------------------------
    # 3. Main simulated‑annealing phase
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    anneal_time = remaining * 0.80   # bulk of the remaining budget
    best_h, best_c = _anneal(best_h, anneal_time)

    # ------------------------------------------------------------
    # 4. Final deterministic hill‑climbing refinement
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 5.0:
        max_iter = int(remaining * 2000)   # a few thousand iterations per second
        best_h, best_c = _hill_climb(best_h, max_iter)

    # ------------------------------------------------------------
    # 5. Enforce exact integral (possible tiny drift)
    best_h = _project_to_feasible(best_h)

    # ------------------------------------------------------------
    # 6. Compute the final bound (use helper if present)
    try:
        final_c = evaluate_erdos_solution(best_h)  # pre‑imported by the harness
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
