def _project_sum(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # We need to adjust sum to target
    # Compute diff
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Use iterative approach: While diff not close, distribute diff equally across unfixed variables.
    # maintain mask of free variables not at bound (0 or 1)
    free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
    # if no free, then just adjust some bound elements? Should not happen.
    # Spread addition or subtraction across free variables uniformly, clamping.
    while abs(diff) > 1e-12 and free.size > 0:
        per = diff / free.size
        # Clip addition to stay within bounds
        add = np.clip(per, -x[free], 1 - x[free])
        x[free] += add
        diff = target - x.sum()
        # Update free mask
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
    # In rare case still diff (due to rounding), adjust by distributing to any variable within bound capacity.
    if abs(diff) > 1e-12:
        # assign to the first variable that can accommodate.
        for i in range(len(x)):
            cap_up = 1 - x[i]
            cap_down = x[i]
            if diff > 0 and cap_up > 1e-12:
                d = min(diff, cap_up)
                x[i] += d
                diff -= d
            elif diff < 0 and cap_down > 1e-12:
                d = min(-diff, cap_down)
                x[i] -= d
                diff += d
            if abs(diff) < 1e-12:
                break
    return x

def _project_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    n = x.size
    # Iteratively distribute diff among free variables (not at bounds)
    free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
    # If no free variables, just adjust one variable within bounds
    while abs(diff) > 1e-12 and free.size > 0:
        per = diff / free.size
        # compute allowed addition for each free variable
        cur = x[free]
        add = np.clip(per, -cur, 1.0 - cur)
        x[free] = cur + add
        diff = target - x.sum()
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
    if abs(diff) > 1e-12:
        # adjust any variable that can accommodate the remaining diff
        for i in range(n):
            if diff > 0:
                cap = 1.0 - x[i]
                if cap > 1e-12:
                    d = min(diff, cap)
                    x[i] += d
                    diff -= d
            else:
                cap = x[i]
                if cap > 1e-12:
                    d = min(-diff, cap)
                    x[i] -= d
                    diff += d
            if abs(diff) < 1e-12:
                break
    # final clip
    x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Return the maximal overlap integral for a step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to exactly target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x

    n = x.size
    # distribute the residual among the variables that are not at a bound
    while abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size == 0:
            break
        per = diff / free.size
        cur = x[free]
        add = np.clip(per, -cur, 1.0 - cur)
        x[free] = cur + add
        diff = target - x.sum()

    # if still a tiny residual remains, fix it on any feasible coordinate
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0:
                cap = 1.0 - x[i]
                if cap > 1e-12:
                    d = min(diff, cap)
                    x[i] += d
                    diff -= d
            else:
                cap = x[i]
                if cap > 1e-12:
                    d = min(-diff, cap)
                    x[i] -= d
                    diff += d
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)


def _objective(vec: np.ndarray) -> float:
    """Objective for local continuous optimisation (same as C₅)."""
    n = vec.shape[0]
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return np.max(corr) * (2.0 / n)


def _refine(h: np.ndarray, target: float) -> np.ndarray:
    """A few SLSQP steps starting from h."""
    n = h.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    res = minimize(
        _objective,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": 300, "ftol": 1e-10, "disp": False},
    )
    if not res.success:
        return h
    return _project_sum(np.clip(res.x, 0.0, 1.0), target)


def _hill_climb(h: np.ndarray, cur_c5: float, deadline: float, rng: np.random.Generator):
    """Simple mass‑exchange hill climbing until a deadline."""
    n = h.shape[0]
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 1e-12:
            continue
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h, cur_c5 = h_new, new_c5
    return h, cur_c5


def _sine_candidate(n: int, target: float) -> np.ndarray:
    """A smooth sinusoidal start (values stay inside [0,1])."""
    xs = np.sin(np.linspace(0.0, np.pi, n))
    xs = xs * (target / xs.sum())
    return _project_sum(xs, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function that gives a low C₅ bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise (use provided construction or fall back to a random one)
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.size
    else:
        n = 45
        h0 = rng.random(n)
        target = n / 2.0
        h0 = h0 * (target / h0.sum())
    target = n / 2.0
    best_h = _project_sum(np.clip(h0, 0.0, 1.0), target)
    best_c5 = _evaluate_c5(best_h)

    # Try a smooth sinusoid – often a good starting point.
    cand = _sine_candidate(n, target)
    cand_c5 = _evaluate_c5(cand)
    if cand_c5 + 1e-12 < best_c5:
        best_h, best_c5 = cand, cand_c5

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing style global search
    # ------------------------------------------------------------------
    deadline_sa = start_time + budget_s - 2.0  # reserve a little time for final polishing
    while time.time() < deadline_sa:
        remaining = deadline_sa - time.time()
        # allocate a modest chunk per restart (max 20 s, at least 4 s)
        run_len = min(20.0, max(4.0, remaining * 0.4))
        if run_len <= 0.0:
            break

        # start from the current elite with a modest jitter
        jitter = rng.normal(scale=0.15, size=n)
        cur_h = best_h + jitter
        cur_h = np.clip(cur_h, 0.0, 1.0)
        cur_h = _project_sum(cur_h, target)
        cur_c5 = _evaluate_c5(cur_h)

        # annealing schedule
        T_start, T_end = 0.018, 0.0004
        t0 = time.time()
        while time.time() - t0 < run_len:
            frac = (time.time() - t0) / run_len
            T = T_start * (1 - frac) + T_end * frac

            # choose a move type
            if rng.random() < 0.55:
                # mass exchange between two positions
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(cur_h[i], 1.0 - cur_h[j])
                if max_move <= 1e-12:
                    continue
                delta = rng.random() * max_move
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                # simple swap
                i, j = rng.choice(n, size=2, replace=False)
                if cur_h[i] == cur_h[j]:
                    continue
                new_h = cur_h.copy()
                new_h[i], new_h[j] = new_h[j], new_h[i]

            new_c5 = _evaluate_c5(new_h)
            diff = new_c5 - cur_c5
            if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
                cur_h, cur_c5 = new_h, new_c5
                if new_c5 + 1e-12 < best_c5:
                    best_h, best_c5 = new_h, new_c5

        # occasional local SLSQP refinement from the current best
        if rng.random() < 0.18:
            refined = _refine(best_h, target)
            refined_c5 = _evaluate_c5(refined)
            if refined_c5 + 1e-12 < best_c5:
                best_h, best_c5 = refined, refined_c5

    # ------------------------------------------------------------------
    # 3. Final polishing: short hill‑climb + a few SLSQP runs
    # ------------------------------------------------------------------
    final_deadline = start_time + budget_s - 0.5
    best_h, best_c5 = _hill_climb(best_h, best_c5, final_deadline, rng)

    for _ in range(3):
        if time.time() > start_time + budget_s - 0.3:
            break
        refined = _refine(best_h, target)
        refined_c5 = _evaluate_c5(refined)
        if refined_c5 + 1e-12 < best_c5:
            best_h, best_c5 = refined, refined_c5

    # ensure feasibility before returning
    best_h = _project_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = float(_evaluate_c5(best_h))
    return best_h, best_c5, n
