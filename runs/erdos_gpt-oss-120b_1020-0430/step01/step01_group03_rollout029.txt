def gradient_softmax(h, beta):
    n = len(h)
    r = np.correlate(h, 1 - h, mode='full')
    # Compute softmax weights p
    # Might cause overflow; use stable method.
    max_r = np.max(beta * r)
    exp_vals = np.exp(beta * r - max_r)
    p = exp_vals / exp_vals.sum()
    # compute gradient
    grad = np.zeros_like(h)
    # Use vectorized operations:
    # For each offset ℓ:
    # compute contributions:
    # w * (1 - h[j+ℓ]) adds to grad[j] for j where j+ℓ valid
    # subtract w * h[j-ℓ] for j where j-ℓ valid
    # We'll implement using slices.
    for offset in range(-n+1, n):
        w = p[offset + n - 1]
        if w == 0:
            continue
        if offset >= 0:
            # term (1 - h_{j+offset}) adds to grad[0:n-offset]
            grad[:n-offset] += w * (1 - h[offset:])
            # subtract h_{j-offset} from grad[offset:]
            grad[offset:] -= w * h[:n-offset]
        else:
            # offset negative
            off = -offset
            # term (1 - h_{j+offset}) where j+offset = j-off => valid for j>=off
            grad[off:] += w * (1 - h[:n-off])
            # subtract h_{j-offset} where j-offset = j+off => valid for j< n-off
            grad[:n-off] -= w * h[off:]
    return grad

def pgd_optimize(h0, beta, lr, steps, rng):
    h = h0.copy()
    n = len(h)
    target = n / 2
    for _ in range(steps):
        grad = gradient_softmax(h, beta)
        # we want to move opposite gradient since we minimize objective
        h_new = h - lr * grad
        # clip and adjust
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce sum exactly
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            # distribute diff greedily across points with margin
            # Use rng to pick indices to add/subtract
            for __ in range(200):
                if abs(diff) < 1e-12:
                    break
                idx = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[idx])
                else:
                    add = max(diff, -h_new[idx])
                h_new[idx] += add
                diff = target - h_new.sum()
        h = h_new
    return h

def gradient_softmax(h, beta):
    n = h.size
    r = np.correlate(h, 1 - h, mode='full')
    # stable softmax
    c = beta * r
    maxc = np.max(c)
    expc = np.exp(c - maxc)
    p = expc / np.sum(expc)

    grad = np.zeros_like(h)
    # for each offset
    # note: offset range: -n+1 ... n-1
    for offset in range(-n+1, n):
        w = p[offset + n - 1]
        if w == 0.0:
            continue
        if offset >= 0:
            # j from 0 to n-offset-1: term (1 - h[j+offset])
            grad[:n-offset] += w * (1 - h[offset:])
            # subtract h at indices for j >= offset: h[j-offset]
            grad[offset:] -= w * h[:n-offset]
        else:
            off = -offset
            # j from off to n-1: term (1 - h[j-offset]) = (1 - h[j-off])? Wait offset negative
            # offset = -off..
            # j+offset = j - off. valid when j >= off
            grad[off:] += w * (1 - h[:n-off])
            # subtract h_{j - offset} = h_{j + off}, valid when j < n - off
            grad[:n-off] -= w * h[off:]
    return grad

We need to verify correctness: For offset negative, we added w * (1 - h_{j+offset}) with j+offset >=0 => j >= -offset (=off). So j range [off, n-1] => grad[off:] plus term w * (1 - h[0:n-off])? Actually h_{j+offset} = h_{j - off}, index ranging 0..n-off-1 (since j+offset = j - off, j from off..n-1 → j-off from 0..n-off-1). That's implemented as (1 - h[:n-off]), broadcast across grad[off:].

Second subtraction: term - h_{j - offset} = -h_{j + off}, valid for j such that j+off < n => j < n-off. So j range [0, n-off-1] correspond to indices of grad[:n-off] subtracting h[off:] (since h_{j+off}). That's correct.

def adam_optimize(h_init, beta, steps, lr, rng):

    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    beta2 = 0.999

        grad = gradient_softmax(h, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        h = np.clip(h, 0.0, 1.0)

            c = compute_c5(h)
            if c < best_local_c:
                best_local_c = c

def perturb_h_move(h, rng, scale):

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2] (dx = 2/n)."""
    n = h.size

    return np.max(corr) * (2.0 / n)

def gradient_softmax(h, beta):
    """Gradient of the soft‑max surrogate of max_k Σ h_i(1‑h_{i+k})."""
    n = h.size
    r = np.correlate(h, 1 - h, mode="full")
    c = beta * r
    maxc = np.max(c)
    expc = np.exp(c - maxc)
    p = expc / expc.sum()                     # soft‑max weights, length 2n‑1
    grad = np.zeros_like(h)

    for offset in range(-n + 1, n):
        w = p[offset + n - 1]
        if w == 0.0:
            continue
        if offset >= 0:
            grad[: n - offset] += w * (1 - h[offset:])
            grad[offset:]       -= w * h[: n - offset]
        else:
            off = -offset
            grad[off:]          += w * (1 - h[: n - off])
            grad[: n - off]     -= w * h[off:]
    return grad

    for t in range(1, steps + 1):
        g = gradient_softmax(h, beta)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()

        if t % 10 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c


# ---------- main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # resolution (feel free to change if time permits)
    n = 360

    # ----- initialise from parent if available -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        factor = n // h0.size
        if factor * h0.size == n and factor > 1:
            h = np.repeat(h0, factor)
        else:
            xs_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n,    endpoint=False)
            h = np.interp(xs_new, xs_old, h0)
    else:
        h = random_feasible_vector(n, rng)

    # enforce exact feasibility
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---------- multi‑stage optimisation ----------
    betas = [1.0, 5.0, 20.0, 80.0, 200.0]
    steps_per_beta = 200
    lr = 0.1

    for beta in betas:
        if time.time() - start > budget_s * 0.80:
            break
        h_opt, c_opt = adam_optimize(best_h, beta, steps_per_beta, lr, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # final hill‑climb polishing
    if time.time() - start < budget_s * 0.95:
        h_pol, c_pol = hill_climb(best_h, rng, max_iters=3000, start_step=0.5)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol

    # ---------- random restarts (light) ----------
    restarts = 0
    while time.time() - start < budget_s * 0.99 and restarts < 20:
        restarts += 1
        h_start = random_feasible_vector(n, rng)
        # short Adam ladder
        for beta in betas[:3]:
            if time.time() - start > budget_s * 0.99:
                break
            h_start, _ = adam_optimize(h_start, beta, steps_per_beta // 2, lr, rng)
        # brief hill‑climb
        h_start, c_start = hill_climb(h_start, rng, max_iters=1200, start_step=0.3)
        if c_start < best_c:
            best_h, best_c = h_start, c_start

    # ----- final feasibility clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
