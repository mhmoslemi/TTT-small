def project_onto_box_simplex(v, s, lo=0., hi=1., tol=1e-12, max_iter=100):
    # Find lambda such that sum(clip(v - lambda, lo, hi)) = s.
    # The function is monotonic decreasing in lambda.
    # lower bound lambda_low = min(v) - hi
    # upper bound lambda_high = max(v) - lo
    lo_lambda = np.min(v) - hi
    hi_lambda = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lo_lambda + hi_lambda) / 2
        w = np.clip(v - lam, lo, hi)
        total = w.sum()
        if abs(total - s) < tol:
            return w
        if total > s:
            lo_lambda = lam
        else:
            hi_lambda = lam
    return np.clip(v - (lo_lambda+hi_lambda)/2, lo, hi)
```

def project_onto_box_simplex(v, s, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    # ensures sum w = s and lo <= w <= hi by solving w = clip(v - lambda, lo, hi)
    # binary search for lambda
    # left bound: lambda_low = np.min(v) - hi
    # right bound: lambda_high = np.max(v) - lo
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        w = np.clip(v - lam, lo, hi)
        total = w.sum()
        if abs(total - s) <= tol:
            return w
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    # final
    w = np.clip(v - (lam_low + lam_high) / 2.0, lo, hi)
    # adjust tiny residual if needed (could redistrib)
    return w
```

import time
import numpy as np

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def _evaluate_c5(h):
    """Maximum overlap integral for a step vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_onto_box_simplex(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=60):
    """
    Euclidean projection of v onto {x : lo <= x_i <= hi, sum x_i = target}.
    Solution has the form x = clip(v - λ, lo, hi) for a suitable λ.
    """
    # λ bounds: when λ is very negative, clip(v-λ)=hi; when λ very positive, =lo
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam, lo, hi)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    # final cut
    w = np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)
    # in rare case of tiny residual, adjust by simple redistribution
    diff = target - w.sum()
    if abs(diff) > tol:
        # distribute the residual uniformly on free variables
        free = (w > lo + 1e-12) & (w < hi - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, lo, hi)
    return w


def _gradient_at_lag(h, lag):
    """Gradient of the correlation at a given lag (worst shift)."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        # term from first factor: (1 - h[i+lag]) for i=0..n-1-lag
        g[: n - lag] = 1.0 - h[lag:]
        # term from second factor: -h[i-lag] for i=lag..n-1
        g[lag:] -= h[: n - lag]
    else:
        p = -lag
        # term from first factor for i=p..n-1: (1 - h[i-p])
        g[p:] = 1.0 - h[: n - p]
        # term from second factor for i=0..n-1-p: -h[i+p]
        g[: n - p] -= h[p:]
    return g


def _gradient_move(h, lag, target, step_scale):
    """Propose a gradient step for the worst lag."""
    g = _gradient_at_lag(h, lag)
    norm = np.linalg.norm(g)
    if norm == 0.0:
        return h, _evaluate_c5(h)
    step = step_scale / (norm + 1e-12)
    cand = h - step * g
    cand = _project_onto_box_simplex(cand, target)
    return cand, _evaluate_c5(cand)

    # step sizes (larger → earlier, smaller → fine‑tuning)

    # counter for stagnation
    no_improve = 0

    while time.time() < deadline:
