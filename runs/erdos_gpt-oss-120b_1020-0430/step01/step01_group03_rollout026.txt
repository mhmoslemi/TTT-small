def gradient_descent(h, max_iters, init_step, rng):

    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):

        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (n - 1)
        # compute direction
        if s >= 0:
            left = np.concatenate((np.zeros(s), h[:-s])) if s > 0 else h
            right = np.concatenate((h[s:], np.zeros(s))) if s > 0 else h
        else:
            ss = -s
            left = np.concatenate((h[ss:], np.zeros(ss)))
            right = np.concatenate((np.zeros(ss), h[:-ss]))
        direction = -1.0 + left + right
        # update
        h_new = h + step * direction
        # clip and sum fix
        h_new = np.clip(h_new, 0.0, 1.0)
        # fix sum
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[i])
                else:
                    add = max(diff, -h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()
        h = h_new
        # compute current C
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # decay step
        step *= 0.9995  # small decay
        if step < 1e-6:
            step = 1e-6
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the optimizer callbacks (avoid closures)
# ----------------------------------------------------------------------
CURRENT_TARGET = None   # set before each SLSQP run
CURRENT_BETA = None     # soft‑max temperature


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the integral exactly (numerical drift may appear)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    # start from a uniform vector and shuffle mass randomly
    h = np.full(n, 0.5)
    for _ in range(n * 4):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=4000, start_step=0.5):
    """Simple deterministic hill‑climb (accept improvements only)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # shrink step size slowly
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    # final feasibility correction
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(3000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()
    return best_h, best_c


def gradient_descent(h0, rng, max_iters, init_step, start_time, budget_s):
    """Projected gradient descent on the dominant shift."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.85:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        s = idx - (n - 1)          # shift producing the current maximum
        # build left and right shifted copies (zero outside)
        if s >= 0:
            if s:
                left = np.concatenate((np.zeros(s), h[:-s]))
                right = np.concatenate((h[s:], np.zeros(s)))
            else:
                left = h
                right = h
        else:
            ss = -s
            left = np.concatenate((h[ss:], np.zeros(ss)))
            right = np.concatenate((np.zeros(ss), h[:-ss]))
        direction = -1.0 + left + right   # -∇C(s)
        h_new = h + step * direction
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()
        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # exponential decay of step size
        step = max(step * 0.998, 1e-6)
    return best_h, best_c


def upsample_h(parent, factor):
    """Nearest‑neighbour up‑sampling (repeat each sample `factor` times)."""
    return np.repeat(parent, factor)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C₅ overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        # try to double the resolution if it stays ≤200 samples
        if n0 * 2 <= 200:
            h = upsample_h(parent, 2)
            n = h.size
        else:
            h = parent.copy()
            n = n0
    else:
        n = 150
        h = random_feasible_vector(n, rng)

    # enforce exact feasibility
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(3000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1 – gradient descent on the dominant shift
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.70:
        gd_h, gd_c = gradient_descent(best_h, rng, max_iters=6000,
                                      init_step=0.3,
                                      start_time=start_time,
                                      budget_s=budget_s)
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c

    # ------------------------------------------------------------------
    # Phase 2 – hill‑climb refinement
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.80:
        hc_h, hc_c = hill_climb(best_h, rng, max_iters=5000, start_step=0.6)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

    # ------------------------------------------------------------------
    # Phase 3 – SLSQP with increasing β
    # ------------------------------------------------------------------
    betas = [10.0, 40.0, 120.0, 400.0]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.90:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=400)
            c_trial = compute_c5(trial)
            if c_trial < best_c:
                best_h, best_c = trial, c_trial
        except Exception:
            continue

    # ------------------------------------------------------------------
    # Phase 4 – random restarts (short gradient + hill‑climb + SLSQP)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.98:
        # generate a fresh start
        h_start = random_feasible_vector(n, rng)
        # quick gradient descent
        gd_h, _ = gradient_descent(h_start, rng, max_iters=1200,
                                   init_step=0.25,
                                   start_time=start_time,
                                   budget_s=budget_s)
        # short hill‑climb
        hc_h, hc_c = hill_climb(gd_h, rng, max_iters=1500, start_step=0.5)
        # a couple of SLSQP passes with low β
        for beta in betas[:2]:
            try:
                hc_h = slsqp_optimize(hc_h, beta=beta, max_iter=200)
            except Exception:
                pass
        c_tmp = compute_c5(hc_h)
        if c_tmp < best_c:
            best_h, best_c = hc_h, c_tmp

    # ------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Return the best found solution
    # ------------------------------------------------------------------
    return best_h.tolist(), float(final_c), int(n)
