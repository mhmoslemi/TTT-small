import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the soft‑max surrogate used by SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = None  # will be set before each SLSQP call
CURRENT_BETA = None    # temperature for the log‑sum‑exp

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ h_i (1 - h_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def slsqp_optimize(h0, beta, max_iter=200):
    """One SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce exact sum (tiny drift may appear)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def random_feasible_vector(n, rng):
    """Return a random vector in [0,1]^n with sum = n/2."""
    x = rng.random(n)
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:

        i = rng.integers(0, n)

            add = min(diff, 1.0 - x[i])
            x[i] += add
            diff -= add
        else:
            sub = min(-diff, x[i])
            x[i] -= sub
            diff += sub
    return x

def gradient_worst_shift(h, shift):
    """Gradient of the worst correlation term with respect to h."""
    n = h.size
    g = np.zeros(n)
    if shift >= 0:
        k = shift
        idx = np.arange(0, n - k)
        g[idx] += 1.0 - h[idx + k]
        g[idx + k] -= h[idx]
    else:
        t = -shift
        idx = np.arange(0, n - t)
        g[idx] -= h[idx + t]
        g[idx + t] += 1.0 - h[idx]
    return g

def local_optimize(h0, time_limit, start_step=0.5):
    """
    Fast gradient‑driven hill‑climb that focuses on the shift attaining
    the maximal overlap.  Returns (h_opt, c5_opt).
    """
    h = h0.copy()
    n = h.size
    best_c = compute_c5(h)
    step = start_step
    start = time.time()

    while time.time() - start < time_limit:
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # gradient for that shift
        g = gradient_worst_shift(h, shift)

        # candidate indices (few extreme values)
        i_cand = np.argsort(-g)[:5]   # largest gradient
        j_cand = np.argsort(g)[:5]    # smallest gradient

        improved = False
        for i in i_cand:
            if h[i] <= 1e-14:
                continue
            for j in j_cand:
                if i == j or h[j] >= 1.0 - 1e-14:
                    continue
                if g[i] <= g[j] + 1e-12:
                    continue
                eps = min(step, h[i], 1.0 - h[j])
                if eps <= 1e-14:
                    continue
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                new_c = compute_c5(new_h)
                if new_c < best_c - 1e-12:
                    h = new_h
                    best_c = new_c
                    improved = True
                    break
            if improved:
                break

        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
    return h, best_c
