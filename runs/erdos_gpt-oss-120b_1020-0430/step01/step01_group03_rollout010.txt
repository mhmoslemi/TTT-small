def subgradient(h):
    n = h.size
    c = np.correlate(h, 1 - h, mode='full')
    max_c = c.max()

    idxs = np.where(c >= max_c - 1e-12)[0]

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
CURRENT_TARGET = None  # sum constraint for SLSQP
CURRENT_BETA = None    # temperature for soft‑max surrogate


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (possible tiny drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def shift_zero(arr, k):
    """Shift arr by k with zero padding (k>0 → shift right)."""
    n = arr.size
    if k > 0:
        out = np.empty_like(arr)
        out[:k] = 0.0
        out[k:] = arr[: n - k]
    elif k < 0:
        km = -k
        out = np.empty_like(arr)
        out[-km:] = 0.0
        out[:-km] = arr[km:]
    else:
        out = arr.copy()
    return out


def subgradient(h):
    """Sub‑gradient of the max‑overlap objective."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for idx in idxs:
        k = idx - (n - 1)
        grad += 1.0 - shift_zero(h, k) - shift_zero(h, -k)
    grad /= max(idxs.size, 1)  # avoid div‑by‑zero
    return grad, max_val * (2.0 / n)


def random_feasible_vector(n, rng):
    """Generate a random h ∈ [0,1]^n with Σ h = n/2."""
    # start from the centre then random mass transfers
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def gradient_descent(h0, rng, max_iters=2000, step0=0.5, step_decay=0.9995):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h0.size
    target = n / 2.0
    h = np.clip(h0, 0.0, 1.0)

    # enforce exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()

    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = step0

    for it in range(max_iters):
        grad, _ = subgradient(h)
        if np.allclose(grad, 0):
            # stagnation → random shake
            h = perturb_h(h, rng, scale=0.1)
            continue
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)

        # re‑balance the sum
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()

        h = h_new
        cnow = compute_c5(h)
        if cnow < best_c:
            best_c = cnow
            best_h = h.copy()
        step *= step_decay
    return best_h, best_c
