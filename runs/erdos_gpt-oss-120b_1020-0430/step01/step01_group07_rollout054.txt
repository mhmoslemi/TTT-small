def _optimize_h(h_start, time_budget, rng):
    h = h_start.copy()
    n = h.size
    dx = 2.0 / n
    # Track best
    best_h = h.copy()
    best_c = _c5_metric_fast(h)

    start = time.time()
    # define temperature schedule: maybe for random moves
    # We'll set T0 = 0.1, T_end = 1e-5.

    while time.time() - start < time_budget:
        # compute corr and max
        corr = _corr_fft(h)
        max_idx = np.argmax(corr)
        cur_c = corr[max_idx] * dx
        # update best
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # compute worst lag
        kstar = max_idx - (n - 1)
        # compute gradient
        g = np.zeros(n)
        if kstar >= 0:
            # term1
            g[:n - kstar] = 1.0 - h[kstar:]
            # term2
            g[kstar:] -= h[:n - kstar]
        else:
            k = -kstar
            g[k:] = 1.0 - h[:n - k]
            g[:n - k] -= h[k:]

        # find positive and negative gradient
        pos_idx = np.argmax(g)   # largest positive
        neg_idx = np.argmin(g)   # most negative
        # Try gradient move if beneficial
        if g[pos_idx] > 1e-12 and g[neg_idx] < -1e-12:
            # move mass from pos to neg
            delta_max = min(h[pos_idx], 1.0 - h[neg_idx])
            # step fraction maybe decreasing with time
            elapsed = time.time() - start
            progress = elapsed / time_budget
            # step fraction schedule: start high then low
            step_frac = 0.5 * (1 - progress) + 0.01  # between ~0.5 and 0.01
            step = min(delta_max * step_frac, delta_max)
            if step > 1e-12:
                h[pos_idx] -= step
                h[neg_idx] += step
                continue   # go next iteration after gradient move
        # else random move using simulated annealing
        # choose random pair
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # Temperature
        elapsed = time.time() - start
        progress = elapsed / time_budget
        T0, T_end = 0.1, 1e-5
        T = T0 * (T_end / T0) ** progress
        # propose delta uniform within [low, high]
        delta = rng.uniform(low, high)
        # modify
        h[i] = hi + delta
        h[j] = hj - delta
        # evaluate new C5
        new_corr = _corr_fft(h)
        new_c = new_corr.max() * dx
        dC = new_c - cur_c
        # accept if improvement or with probability
        if dC < 0 or rng.rand() < math.exp(-dC / max(T, 1e-12)):
            # accept, update cur_c (will be recomputed next iteration)
            pass
        else:
            # reject
            h[i] = hi
            h[j] = hj
        # loop continues

    return best_h, best_c

def _corr_fft(h):
    n = h.shape[0]
    N = 2 * n
    Fh = np.fft.fft(h, N)
    F1mh = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(Fh * np.conj(F1mh)).real[:2*n - 1]
    return corr

def _c5_metric_fast(h):
    n = h.shape[0]
    corr = _corr_fft(h)
    return corr.max() * (2.0 / n)

import time, math, random
import numpy as np

# ---------- core helpers -------------------------------------------------
def _corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (full mode)."""
    n = h.shape[0]
    N = 2 * n
    Fh = np.fft.fft(h, N)
    F1mh = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(Fh * np.conj(F1mh)).real[: 2 * n - 1]
    return corr


def _c5_metric_fast(h: np.ndarray) -> float:
    """C_5 value for the sampled step function."""
    n = h.shape[0]
    return _corr_fft(h).max() * (2.0 / n)


def _initial_vector_n(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Construct a feasible start vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        if n0 != n:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x, x0, h0)
        else:
            h = h0.copy()
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    cur = h.sum()
    if cur != 0.0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for idx in rng.permutation(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break
    return h


def _random_binary_vector(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Exact‑mass binary (or almost binary) start."""
    target = n / 2.0
    ones = int(np.floor(target))
    frac = target - ones
    h = np.zeros(n, dtype=np.float64)
    idx = rng.permutation(n)
    if ones:
        h[idx[:ones]] = 1.0
    if frac > 1e-12:
        h[idx[ones]] = frac
    return h


def _optimize_h(start_h: np.ndarray, budget_s: float,
                rng: np.random.RandomState) -> tuple[np.ndarray, float]:
    """Gradient‑guided simulated annealing."""
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5_metric_fast(h)

    start = time.time()
    stagnation = 0
    max_stagn = 5000               # restart trigger
    while time.time() - start < budget_s:

        corr = _corr_fft(h)
        cur_c = corr.max() * dx
        # update best
        if cur_c + 1e-12 < best_c:
            best_c = cur_c
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        # possible restart on long stagnation
        remaining = budget_s - (time.time() - start)
        if stagnation >= max_stagn and remaining > 30.0:
            h = _random_binary_vector(n, rng)
            stagnation = 0
            continue

        # worst lag (index of max correlation)
        max_idx = int(np.argmax(corr))
        kstar = max_idx - (n - 1)

        # gradient of the worst lag
        g = np.zeros(n, dtype=np.float64)
        if kstar >= 0:
            if n - kstar > 0:

                g[kstar:] -= h[: n - kstar]
        else:
            k = -kstar

        # temperature (for occasional uphill steps)
