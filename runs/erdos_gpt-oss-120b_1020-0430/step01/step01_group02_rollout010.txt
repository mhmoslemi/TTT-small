def _evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute correlation via FFT
    size = 2*n - 1
    # pad to next power of 2 for speed? Not necessary
    f = np.fft.rfft(h, size)
    g = np.fft.rfft(1 - h, size)
    corr = np.fft.irfft(f * g, n=size)
    # corr length size; convert to float, might have tiny imaginary errors
    max_corr = np.max(corr.real)
    return max_corr * dx

def _hill_climb(h, c5, deadline, rng, max_iter=20000):
    n = len(h)
    while time.time() < deadline and max_iter>0:
        i, j = rng.choice(n, size=2, replace=False)
        # Choose direction random
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 0:
            continue
        delta = rng.uniform(0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
        max_iter -= 1
    return h, c5

import time, numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    N = 2 * n - 1
    # FFT-based correlation
    fa = np.fft.rfft(h, n=N)
    fb = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(fa * fb, n=N)
    max_corr = np.max(corr.real)
    return max_corr * dx

def _adjust_sum(x, target, eps=1e-12):
    """Project onto [0,1] and sum=target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) <= eps:
        return x
    # iterative water-filling
    # keep track of which variables are not at a bound
    free = np.arange(x.size)
    while abs(diff) > eps and free.size > 0:
        # capacity to add or subtract per free variable
        if diff > 0:
            caps = 1.0 - x[free]
        else:
            caps = x[free]  # amount we can subtract
        total_cap = caps.sum()
        if total_cap <= 0:
            break
        # distribute diff proportionally to capacity
        # but limit to capacity per variable
        # use simple equal distribution
        per = diff / free.size
        # apply limited by capacity
        change = np.clip(per, -caps, caps)
        x[free] += change
        diff = target - x.sum()
        # update free set (variables that are not at bound)
        if diff > 0:
            free = free[x[free] < 1.0 - eps]
        else:
            free = free[x[free] > eps]
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h, c5, deadline, rng, max_steps=200000):
    n = h.shape[0]
    # simple hill climbing moving mass between two positions
    while time.time() < deadline and max_steps > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-14:
            max_steps -= 1
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
        max_steps -= 1
    return h, c5

def _resample_h(src, n_target):
    n_src = src.shape[0]
    if n_src == n_target:
        return src.copy()
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(xt, xs, src)

    start = time.time()
    rng = np.random.default_rng(seed)

    # discover if an initial construction is provided
    init_vec = globals().get("initial_h_values")
    # candidate resolutions (even numbers for exact half sum)
    n_candidates = [45, 60, 80, 100, 120]
    # optional: shuffle order to diversify search
    rng.shuffle(n_candidates)

    best_overall_h = None
    best_overall_c5 = float('inf')
    best_overall_n = None

    # allocate roughly half the budget for exploration, half for final polishing
    exploration_deadline = start + budget_s * 0.65

    for n in n_candidates:

        target = n / 2.0

                rnd_c5 = _evaluate_c5(rnd)
                if rnd_c5 < best_c5:
                    best_h, best_c5 = rnd, rnd_c5

        # after SLSQP, run a dedicated hill‑climb on this candidate
        hill_deadline = start + budget_s * 0.9
        best_h, best_c5 = _hill_climb(best_h, best_c5, hill_deadline, rng)

        # keep the global best
        if best_c5 < best_overall_c5:
            best_overall_c5 = best_c5
            best_overall_h = best_h.copy()
            best_overall_n = n

    # final polishing on the best found solution using a longer hill‑climb
    final_deadline = start + budget_s - 0.2
    if best_overall_h is not None:
        best_overall_h, best_overall_c5 = _hill_climb(
            best_overall_h,
            best_overall_c5,
            final_deadline,
            rng,
            max_steps=500000,
        )
        # final projection to guarantee feasibility
        best_overall_h = _adjust_sum(np.clip(best_overall_h, 0.0, 1.0), best_overall_n / 2.0)
        best_overall_c5 = float(_evaluate_c5(best_overall_h))

        n = 45
        target = n / 2.0
        best_overall_h = np.full(n, 0.5)
        best_overall_c5 = float(_evaluate_c5(best_overall_h))
