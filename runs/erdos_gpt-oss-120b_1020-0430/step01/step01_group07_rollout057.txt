    n = h.shape[0]
    size = 1 << ((2*n-1).bit_length())  # next power of two

def _c5_metric_fft(h, size):
    # h: 1D np.ndarray
    # compute correlation h with (1 - h) reversed
    fft_h = np.fft.rfft(h, size)
    fft_rev = np.fft.rfft((1 - h)[::-1], size)
    corr = np.fft.irfft(fft_h * fft_rev, size)[:2*h.shape[0] - 1]
    return np.max(corr) * (2.0 / h.shape[0])

import time
import math
import random
import numpy as np

# ---- Metric -------------------------------------------------------------
def _c5_metric_fft(h: np.ndarray, fft_size: int) -> float:
    """C_5 value using FFT based full correlation."""
    # correlation of h with (1‑h) (zero‑extended)
    fft_h = np.fft.rfft(h, fft_size)
    fft_rev = np.fft.rfft((1.0 - h)[::-1], fft_size)
    corr = np.fft.irfft(fft_h * fft_rev, fft_size)[: 2 * h.shape[0] - 1]
    return np.max(corr) * (2.0 / h.shape[0])


# ---- Worst lag ---------------------------------------------------------
def _worst_lag(h: np.ndarray, n: int, fft_size: int):
    """Return lag L (can be negative) that attains the maximal overlap."""
    fft_h = np.fft.rfft(h, fft_size)
    fft_rev = np.fft.rfft((1.0 - h)[::-1], fft_size)
    corr = np.fft.irfft(fft_h * fft_rev, fft_size)[: 2 * n - 1]
    k = int(np.argmax(corr))
    L = n - 1 - k          # shift sign convention matches the integral
    return L, float(corr[k])


# ---- Gradient -----------------------------------------------------------
def _gradient_wrt_h(h: np.ndarray, L: int) -> np.ndarray:
    """Gradient of f(L)=∫h(x)(1‑h(x+L))dx w.r.t. the discretised h."""
    n = h.shape[0]
    idx = np.arange(n)
    grad = np.zeros(n, dtype=np.float64)

    # contribution of the linear term (1‑h_{i+L})
    mask = (idx + L >= 0) & (idx + L < n)
    grad[mask] += 1.0 - h[idx[mask] + L]

    # contribution of the quadratic term (‑h_{i‑L})
    mask2 = (idx - L >= 0) & (idx - L < n)
    grad[mask2] -= h[idx[mask2] - L]
    return grad


# ---- Simulated annealing -------------------------------------------------
def _anneal(start_h: np.ndarray, time_limit: float, n: int, fft_size: int):
    """Pairwise mass‑transfer simulated annealing."""
    h = start_h.copy()
    cur_c = _c5_metric_fft(h, fft_size)
    best_h = h.copy()
    best_c = cur_c

    T0 = 0.08
    T_end = 1e-6
    max_step = 0.20
    start = time.time()

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        progress = elapsed / time_limit
        T = T0 * (T_end / T0) ** progress

        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)          # delta added to h[i]
        high = min(1.0 - hi, hj)          # delta added to h[i]
        if low >= high:
            continue

        # mostly modest steps, occasional full range
        if random.random() < 0.9:
            low_c = max(low, -max_step)
            high_c = min(high, max_step)
            if low_c < high_c:
                delta = random.uniform(low_c, high_c)
            else:
                delta = random.uniform(low, high)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _c5_metric_fft(h, fft_size)
        dC = new_c - cur_c

        if dC < 0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    return best_h, best_c


# ---- Gradient descent refinement -----------------------------------------
def _refine_gradient(h: np.ndarray, time_limit: float, n: int, fft_size: int):
    """Small deterministic moves guided by the gradient of the worst lag."""
    best_h = h.copy()
    best_c = _c5_metric_fft(best_h, fft_size)
    cur_c = best_c
    start = time.time()
    max_step = 0.05

    while time.time() - start < time_limit:
        L, _ = _worst_lag(best_h, n, fft_size)
        grad = _gradient_wrt_h(best_h, L)

        i_pos = int(np.argmax(grad))   # decrease this entry
        i_neg = int(np.argmin(grad))   # increase this entry

        if grad[i_pos] <= 1e-8 and -grad[i_neg] <= 1e-8:
            break

        hi, hj = best_h[i_pos], best_h[i_neg]
        delta = min(hi, 1.0 - hj)                # feasible amount
        if delta <= 0:
            # try the opposite orientation
            hi, hj = best_h[i_neg], best_h[i_pos]
            i_pos, i_neg = i_neg, i_pos
            delta = min(hi, 1.0 - hj)
            if delta <= 0:
                break

        delta = min(delta, max_step)

        # apply move: decrease i_pos, increase i_neg
        best_h[i_pos] = hi - delta
        best_h[i_neg] = hj + delta

        new_c = _c5_metric_fft(best_h, fft_size)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
            best_c = new_c
        else:
            # revert
            best_h[i_pos] = hi
            best_h[i_neg] = hj

    return best_h, best_c


# ---- Initial vectors -----------------------------------------------------
def _generate_initial_vectors(n: int):
    """Return a modest collection of feasible start vectors."""
    target = n / 2.0
    vectors = []

    # 1) interpolation of the supplied construction (if any)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        m0 = h0.size
        x0 = np.linspace(0.0, 2.0, m0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
        cur = h.sum()
        if cur != 0:
            h *= (target / cur)
        np.clip(h, 0.0, 1.0, out=h)
        deficit = target - h.sum()
        if abs(deficit) > 1e-12:
            for idx in np.random.permutation(n):
                lo, hi = -h[idx], 1.0 - h[idx]
                if lo <= deficit <= hi:
                    h[idx] += deficit
                    break
        vectors.append(h)

    # 2) several binary vectors with exactly n/2 ones
    ones = int(target)
    for _ in range(4):
        v = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        v[perm[:ones]] = 1.0
        vectors.append(v)

    # 3) a random continuous vector, scaled to the correct total mass
    v = np.random.rand(n)
    v = np.clip(v, 0.0, 1.0)
    v *= (target / v.sum())
    np.clip(v, 0.0, 1.0, out=v)
    vectors.append(v)

    return vectors


# ---- Main entry point ----------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and the number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.time()

    # problem size – 512 points give a good trade‑off between freedom and speed
    n = 512
    if n % 2:
        n += 1
    target_sum = n / 2.0
    fft_size = 1 << ((2 * n - 1).bit_length())  # power of two for FFT

    # ----- generate start vectors ---------------------------------------
    start_vectors = _generate_initial_vectors(n)

    # ----- allocate time for the annealing phase ------------------------
    anneal_budget = total_budget * 0.80
    best_h = None
    best_c = float('inf')

    for idx, vec in enumerate(start_vectors):
        elapsed = time.time() - start_all
        remaining = anneal_budget - elapsed
        if remaining <= 6.0:            # not enough time for another run
            break
        # equal remaining share for the rest of the starts
        slice_time = remaining / (len(start_vectors) - idx)
        cand_h, cand_c = _anneal(vec, slice_time, n, fft_size)
        if cand_c < best_c:
            best_c = cand_c
            best_h = cand_h.copy()

    if best_h is None:                 # fallback – should never happen
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_c = _c5_metric_fft(best_h, fft_size)

    # ----- refinement with gradient steps --------------------------------
    remaining = total_budget - (time.time() - start_all)
    if remaining > 2.0:
        refine_time = remaining * 0.90
        best_h, best_c = _refine_gradient(best_h, refine_time, n, fft_size)

    # ----- final exactness of the mass constraint -----------------------
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
            lo, hi = -best_h[idx], 1.0 - best_h[idx]
            if lo <= diff <= hi:
                best_h[idx] += diff
                break

    # ----- final evaluation (use the provided helper if available) -----
    try:
        final_c = evaluate_erdos_solution(best_h)   # noqa: F821
    except Exception:
        final_c = _c5_metric_fft(best_h, fft_size)

    return best_h, float(final_c), int(n)
