def compute_c5_fast(h):
    n = len(h)
    L = 1 << ((2*n - 2).bit_length()) (next power of two > 2n-1) maybe. Use L = 1 << ((2*n-1).bit_length()).

    # mixture: Want linear correlation: Output length 2n-1. Using rfft approach:

    ha = np.fft.rfft(h, L)
    hb = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(ha * hb, L)[:2*n-1]  # shape match

    max_val = np.max(conv)
    return max_val * (2.0 / n)

def compute_c5_fast(h):
    n = len(h)
    L = 1 << ((2*n - 1).bit_length())
    # ensure L >= 2n-1
    fft_h = np.fft.rfft(h, n=L)
    fft_r = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(fft_h * fft_r, n=L)[:2*n-1]
    return float(conv.max() * (2.0 / n))

def simulated_annealing_binary(n, initial_h, deadline, rng):
    # h is binary array (0/1) dtype float.
    cur_h = initial_h.copy()
    cur_c5 = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # Determine indices list
    # Actually we can maintain arrays of 1 and 0 indices.
    # At each swap we need update lists.

    ones = np.flatnonzero(cur_h > 0.5) # returns array of indices where h==1
    zeros = np.flatnonzero(cur_h < 0.5)

    total_time = deadline - time.time()
    # run while time left
    start = time.time()
    # compute initial T0
    T0 = cur_c5 * 0.5 if cur_c5 > 0 else 0.01

    while True:
        cur_time = time.time()
        if cur_time >= deadline:
            break
        elapsed = cur_time - start
        # fraction of allocated time:
        frac = elapsed / total_time if total_time > 0 else 1.0
        T = T0 * (1.0 - frac)  # linear cooling
        if T <= 1e-6:
            T = 1e-6

        # random swap
        # pick random index in ones
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        idx_one = ones[i]
        idx_zero = zeros[j]

        # perform swap
        cur_h[idx_one] = 0.0
        cur_h[idx_zero] = 1.0

        # compute new objective
        new_c5 = compute_c5_fast(cur_h)
        delta = new_c5 - cur_c5

        # accept?
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            # update ones/zeros arrays:
            # swap positions in arrays
            ones[i] = idx_zero
            zeros[j] = idx_one
            # No need to sort.
            # Track best
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[idx_one] = 1.0
            cur_h[idx_zero] = 0.0
            # nothing to update

    return best_h, best_c5

def simulated_annealing_binary(n, start_h, deadline, rng):
    cur_h = start_h.copy()
    cur_c5 = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # maintain indices arrays:
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    # total time for SA: remaining = deadline - now
    start = time.time()
    total_alloc = deadline - start

    # set initial temperature:
    T0 = max(0.01, cur_c5 * 0.5)  # at least 0.01
    # iteration count not predetermined; loop until time.

    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start
        frac = elapsed / total_alloc if total_alloc > 0 else 1.0
        T = T0 * (1.0 - frac)
        if T < 1e-6:
            T = 1e-6

        # choose random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5_fast(cur_h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            # no update to lists
    return best_h, best_c5

import time
import numpy as np
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core evaluation utilities
# ------------------------------------------------------------
def compute_c5_exact(h: np.ndarray) -> float:
    """Exact C5 using linear correlation (as used by the evaluator)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def compute_c5_fast(h: np.ndarray) -> float:
    """Fast C5 via FFT (same result, much quicker for many calls)."""
    n = h.shape[0]
    # next power of two ≥ 2*n‑1
    L = 1 << ((2 * n - 1).bit_length())
    ha = np.fft.rfft(h, n=L)
    hb = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(ha * hb, n=L)[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max(C5) used for SLSQP (not essential)."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def softmax_c5_a5(h: np.ndarray) -> float:
    return softmax_c5(h, alpha=5.0)


def softmax_c5_a30(h: np.ndarray) -> float:
    return softmax_c5(h, alpha=30.0)

# Constraint wrapper for scipy.optimize (avoids lambdas)
# ------------------------------------------------------------
class SumConstraint:
    def __init__(self, target):
        self.target = float(target)

    def __call__(self, h):
        return np.sum(h) - self.target


# ------------------------------------------------------------
# Random binary starter
# ------------------------------------------------------------
def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exact half‑filled binary vector."""
    half = n // 2
    vec = np.zeros(n, dtype=float)
    vec[rng.choice(n, size=half, replace=False)] = 1.0
    return vec


# ------------------------------------------------------------
# Initialisation using the parent construction if available
# ------------------------------------------------------------
def binary_start_from_parent(n: int, rng: np.random.Generator) -> np.ndarray:
    """Upsample the optional parent and binarise while keeping the exact sum."""
    try:
        parent = np.asarray(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        return random_binary(n, rng)

    old_n = parent.shape[0]
    if old_n != n:
        # linear interpolation on the interval [0,2)
        old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
        new_x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        up = np.interp(new_x, old_x, parent)
    else:
        up = parent.copy()

    bin_vec = (up > 0.5).astype(float)
    target = n // 2
    cur = int(bin_vec.sum())
    if cur > target:
        # too many ones – turn some off
        idx = np.flatnonzero(bin_vec == 1)
        off = rng.choice(idx, size=cur - target, replace=False)
        bin_vec[off] = 0.0
    elif cur < target:
        # too many zeros – turn some on
        idx = np.flatnonzero(bin_vec == 0)
        on = rng.choice(idx, size=target - cur, replace=False)
        bin_vec[on] = 1.0
    return bin_vec


# ------------------------------------------------------------
# Simulated annealing for binary vectors
# ------------------------------------------------------------
def simulated_annealing_binary(
    n: int,
    start_h: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Return best binary vector found before deadline."""
    cur_h = start_h.copy()
    cur_val = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # maintain explicit index lists for ones and zeros
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    start = time.time()
    total = max(deadline - start, 1e-12)
    T0 = max(0.01, cur_val * 0.5)  # initial temperature

    while True:
        now = time.time()
        if now >= deadline:
            break
        # linear cooling
        frac = (now - start) / total
        T = T0 * (1.0 - frac)
        if T < 1e-6:
            T = 1e-6

        # pick a random 1‑index and a random 0‑index
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = compute_c5_fast(cur_h)
        delta = new_val - cur_val

        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_val = new_val
            # update index lists in‑place
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = cur_h.copy()
        else:
            # reject – revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic optimisation for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin
    overall_deadline = start_time + budget_s * 0.94

    # --------------------------------------------------------
    # Choose a resolution – larger n gives more flexibility.
    # 400 points is a good trade‑off within the time budget.
    # --------------------------------------------------------
    n = 400
    target_sum = n // 2

    best_h = None
    best_c5 = np.inf
    best_n = None

    # -----------------------------------------------------------------
    # Repeated short annealing runs to avoid getting stuck in a local minima
    # -----------------------------------------------------------------
    min_cycle = 5.0  # seconds per SA run (at least)
    while time.time() + min_cycle < overall_deadline:
        # start from a fresh binary vector (sometimes from the parent)
        if rng.random() < 0.3:
            start_vec = binary_start_from_parent(n, rng)
        else:
            start_vec = random_binary(n, rng)

        # allocate a slice of the remaining budget for this SA run
        remaining = overall_deadline - time.time()
        # keep at least a few seconds for a final refinement pass
        slice_deadline = time.time() + min(remaining * 0.6, max(30.0, min_cycle * 2))
        if slice_deadline > overall_deadline:
            slice_deadline = overall_deadline

        h_candidate, c5_candidate = simulated_annealing_binary(
            n, start_vec, slice_deadline, rng
        )

        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            best_n = n

    # --------------------------------------------------------
    # If nothing succeeded (unlikely), fall back to a simple construction
    # --------------------------------------------------------
    if best_h is None:
        n = 200
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_c5 = compute_c5_exact(best_h)
        best_n = n

    # Final exact evaluation (to match the harness)
    final_c5 = compute_c5_exact(best_h)

    return best_h, final_c5, best_n
