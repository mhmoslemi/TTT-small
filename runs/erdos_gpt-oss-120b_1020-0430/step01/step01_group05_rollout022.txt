def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    n = 256
    dx = 2.0/n
    target_sum = n/2
    global TARGET_SUM_GLOBAL # not required

    # helper functions defined above

    # Build seeds list
    seeds = []
    # upsample parent if exists
    if 'initial_h_values' in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rect(n))
    # random seed
    seeds.append(project_onto_bounds_and_sum(np.random.rand(n), target_sum))

    # Evaluate all seeds
    best_h = None
    best_val = np.inf
    for h in seeds:
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    # Optional: Could also evaluate many random seeds as initial.

    # start hill climbing
    cur_h = best_h.copy()
    cur_val = best_val

    # iteration counters
    iters_without_improve = 0
    max_no_improve = 5000  # after which we restart randomly

    while time.time() - start < budget_s - 0.5:
        # pick i, j
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        hi, hj = cur_h[i], cur_h[j]
        eps_min = -min(hi, 1.0 - hj)   # decreasing i, increasing j (epsilon negative)
        eps_max = min(1.0 - hi, hj)    # increasing i, decreasing j (epsilon positive)
        if eps_max <= eps_min:
            continue
        # sample epsilon
        eps = np.random.uniform(eps_min, eps_max)
        new_h = cur_h.copy()
        new_h[i] += eps
        new_h[j] -= eps
        # evaluate
        new_val = compute_c5(new_h)

        # accept if improvement
        if new_val < cur_val or (np.random.rand() < np.exp(-(new_val - cur_val) / max(1e-9, (budget_s - (time.time() - start)) / budget_s * 0.005 + 1e-4))):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                iters_without_improve = 0
            else:
                iters_without_improve += 1
        else:
            iters_without_improve += 1

        # restart if stuck
        if iters_without_improve >= max_no_improve:
            # restart with random feasible vector
            cur_h = project_onto_bounds_and_sum(np.random.rand(n), target_sum)
            cur_val = compute_c5(cur_h)
            iters_without_improve = 0

    # final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    final_val = compute_c5(best_h)

    return (best_h.tolist(), float(final_val), int(n))

import time
import random
import numpy as np

# -----------------------------------------------------------
# Helper functions (top‑level, no closures)
# -----------------------------------------------------------

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while keeping sum(x)=target (bisection on lambda)."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 = max_k ∫ h(x)(1‑h(x+k)) dx, discretised."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_triangle(n: int) -> np.ndarray:
    """Triangular bump, peak 1 at centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(pi*x/2) on [0,2]."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rect(n: int) -> np.ndarray:
    """Block of ones of total mass 1 (contiguous interval)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a possibly different‑size parent."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, n / 2.0)


# -----------------------------------------------------------
# Main optimisation routine
# -----------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a list of
    values in [0,1] on a uniform grid of length n_points (default 256) that
    approximately minimise the Erdős constant C5.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # -------------------------------------------------------
    # Problem size – a power of two helps if we ever switch to FFT.
    # -------------------------------------------------------
    n = 256                     # even, sum = n/2
    target_sum = n / 2.0

    # -------------------------------------------------------
    # Build diverse initial seeds.
    # -------------------------------------------------------
    seeds = []

    if "initial_h_values" in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass

    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rect(n))
    # a purely random feasible vector
    rnd = np.random.rand(n)
    seeds.append(project_onto_bounds_and_sum(rnd, target_sum))

    # -------------------------------------------------------
    # Choose the best of the seeds (cheap evaluation).
    # -------------------------------------------------------
    best_h = None
    best_val = np.inf
    for h in seeds:
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # -------------------------------------------------------
    # Local stochastic optimisation (pairwise mass swaps).
    # -------------------------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val

    # parameters for the annealing schedule
    T_start = 1e-3
    T_end = 1e-5
    max_no_improve = 5000       # after this many iterations restart randomly
    no_improve_cnt = 0

    while time.time() - start < budget_s - 0.5:
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        hi, hj = cur_h[i], cur_h[j]

        # feasible epsilon range (mass moved from i to j)
        eps_min = -min(hi, 1.0 - hj)   # decrease i, increase j
        eps_max = min(1.0 - hi, hj)    # increase i, decrease j
        if eps_max <= eps_min:
            continue

        # mostly small moves, occasional large leaps
        if np.random.rand() < 0.05:
            eps = np.random.uniform(eps_min, eps_max)
        else:
            span = eps_max - eps_min
            step = min(0.12, span)      # restrict step size for fine tuning
            eps = np.random.uniform(-step, step)
            eps = max(eps_min, min(eps_max, eps))

        # propose new vector
        cand = cur_h.copy()
        cand[i] = hi + eps
        cand[j] = hj - eps

        # evaluate
        cand_val = compute_c5(cand)

        # simulated‑annealing acceptance
        if cand_val < cur_val:
            accept = True
        else:
            elapsed = time.time() - start
            frac = elapsed / budget_s
            T = (1.0 - frac) * T_start + frac * T_end
            prob = np.exp(-(cand_val - cur_val) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_h = cand
            cur_val = cand_val
            # update global best if improved
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                no_improve_cnt = 0
                # early exit if we already beat the target record
                if best_val <= 0.3808:
                    break
            else:
                no_improve_cnt += 1
        else:
            no_improve_cnt += 1

        # occasional full restart when stuck
        if no_improve_cnt >= max_no_improve:
            cur_h = project_onto_bounds_and_sum(np.random.rand(n), target_sum)
            cur_val = compute_c5(cur_h)
            no_improve_cnt = 0

    # -------------------------------------------------------
    # Final safety projection & evaluation
    # -------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    final_val = compute_c5(best_h)

    return (best_h.tolist(), float(final_val), int(n))
