def projected_gradient_descent(h0, target_sum, max_iter, step0, decay, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    for i in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        shift = np.argmax(corr) - (len(h)-1)
        g = gradient_for_shift(h, shift)
        # momentum
        v = momentum * v - step * g
        h = h + v
        # project onto box and sum constraint
        h = project_onto_capped_simplex(h, target_sum)
        step *= decay
    return h
```

def local_swap_refine(h, target, dx, rng, attempts=50):
    besth = h.copy()

        val = _compute_c5(newh, dx)

            besth = newh
            bestval = val

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ----------------------------------------------------------------------
# Random feasible vector (uniform then project)
# ----------------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# ----------------------------------------------------------------------
# C5 evaluation for a given discretisation
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ----------------------------------------------------------------------
# Sub‑gradient corresponding to the shift attaining the maximum correlation
# ----------------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with momentum (sub‑gradient of the max)
# ----------------------------------------------------------------------
def _gd_momentum_optimize(h0: np.ndarray, target: float,
                          max_iter: int = 1000, step0: float = 0.6,
                          decay: float = 0.998, beta: float = 0.9) -> np.ndarray:
    h = h0.copy()
    v = np.zeros_like(h)
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)
        g = _gradient_for_shift(h, shift)
        v = beta * v - step * g
        h = h + v
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# ----------------------------------------------------------------------
# Simple hill‑climbing on binary vectors (swap a 1 with a 0)
# ----------------------------------------------------------------------
def _binary_hill_climb(h_bin: np.ndarray, dx: float,
                       rng: np.random.RandomState,
                       time_limit: float, start: float) -> tuple:
    """Return (best_h, best_val) after exhaustive swaps until time runs out."""
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_val = _compute_c5(best_h, dx)
    cur_h = best_h.copy()
    cur_val = best_val

    # Pre‑compute positions of ones / zeros for quick sampling
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)

    while time.time() - start < time_limit:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = _compute_c5(new_h, dx)
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            # update index lists
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if new_val < best_val:
                best_h, best_val = new_h, new_val
        # occasional random restart from the current global best
        if rng.rand() < 0.005:
            cur_h = best_h.copy()
            cur_val = best_val
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
    return best_h, best_val
