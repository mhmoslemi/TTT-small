def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

import time, random, math
import numpy as np
from scipy.optimize import minimize

TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n ∩ {sum = target} by bisection on the Lagrange multiplier."""
    low, high = -1.0, 1.0  # guarantees sum at low > target, at high < target
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretized C5: max_k ∫ h(x)(1-h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Smooth approximation of max via log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth objective used by SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape: peak 1 at centre, linear to 0 at edges."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(πx/2) on [0,2] – already has integral 1."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

    """Return a step function h on [0,2] with a low C5 bound."""
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # problem size
    n = 250                       # keep even for exact half‑mass
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # Build initial seeds
    seeds = []
    if "initial_h_values" in globals():
        try:
            seeds.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Phase 1: SLSQP with increasing smoothing
    alphas = [5.0, 20.0, 60.0]          # start smooth, finish sharp
    for seed_vec in seeds:
        h0 = seed_vec.copy()
        for a in alphas:
            if time.time() - start > budget_s * 0.5:
                break
            try:
                res = minimize(
                    objective_smooth,
                    h0,
                    args=(dx, a),
                    method="SLSQP",
                    bounds=bounds,
                    constraints={"type": "eq", "fun": sum_eq_constraint},
                    options={"maxiter": 400, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                continue
            h_opt = project_onto_bounds_and_sum(res.x, target_sum)
            val = compute_c5(h_opt)
            if val < best_val:
                best_val = val
                best_h = h_opt.copy()
            h0 = h_opt  # continue from here
            if best_val <= 0.3808:
                break
        if best_val <= 0.3808 or time.time() - start > budget_s * 0.5:
            break

    # If nothing found yet, fall back to a triangle
    if best_h is None:
        best_h = make_triangle(n)
        best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 2: Stochastic hill climbing / simulated annealing
    end_time = start + budget_s - 0.3  # reserve a little margin
    cur = best_h.copy()
    cur_val = best_val
    best_local = best_h.copy()
    best_local_val = best_val

    # temperature schedule (linear)
    T_start = 0.02

        i, j = np.random.randint(0, n, size=2)
        if i == j:

        hi, hj = cur[i], cur[j]

        # feasible move: delta transferred from i to j (i loses, j gains)
        delta_max = min(hi, 1.0 - hj)
        delta_min = -min(hj, 1.0 - hi)
        if delta_max <= 0 and delta_min >= 0:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # bounds already respected by construction
        cand_val = compute_c5(cand)

        # accept if improvement or with Metropolis probability
        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / T):
            cur = cand
            cur_val = cand_val
            if cand_val < best_local_val:
                best_local = cand.copy()
                best_local_val = cand_val

        # occasional reset to best to avoid drift
        if (time.time() - start) % 30 < 0.02:  # roughly every 30 s
            cur = best_local.copy()
            cur_val = best_local_val

        if best_local_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Final projection & evaluation
    best_h = np.clip(best_local, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    final_val = compute_c5(best_h)

    return (best_h.tolist(), float(final_val), int(n))
