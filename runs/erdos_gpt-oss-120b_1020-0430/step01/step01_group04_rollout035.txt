Pseudo-code:

```
import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def softmax_c5(h, alpha):
    conv = np.correlate(h, 1.0 - h, mode='full')
    # stable log-sum-exp
    m = np.max(alpha * conv)
    # compute logsumexp

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of the max via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    # stable log‑sum‑exp
    m = np.max(alpha * conv)
    return float((m + np.log(np.sum(np.exp(alpha * conv - m)))) / alpha)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with exact sum n/2."""
    a = rng.random(n)
    target = n / 2.0
    a *= target / a.sum()
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of arr to length new_n, preserving integral."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up *= target / up.sum()
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def trapezoidal_shape(n: int, a: float) -> np.ndarray:
    """
    Symmetric trapezoidal (triangular limit) shape.
    a ∈ [0,1] is the half‑width of the sloping part.
    """
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    if a <= 0.0:
        h_max = 0.5
        h = np.full_like(x, h_max)
    else:
        h_max = 1.0 / (2.0 - a)          # ensures integral = 1
        left = x <= a
        right = x >= 2.0 - a
        middle = ~(left | right)
        h = np.empty_like(x)
        h[left] = h_max * (x[left] / a)
        h[right] = h_max * ((2.0 - x[right]) / a)
        h[middle] = h_max
    # renormalise (tiny discretisation error)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def sinusoidal_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random sinusoid with amplitude 0.5, shifted to satisfy the integral."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    freq = rng.uniform(0.5, 3.5)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


# ------------------------------------------------------------
# Local search (simulated annealing / hill climbing)
# ------------------------------------------------------------
def hill_climb(start: np.ndarray, n: int, deadline: float, rng: np.random.Generator):
    """Simple stochastic hill climbing with temperature decay."""
    h = start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    max_step = 0.2
    T = 0.015

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move mass from i to j
        max_delta = min(max_step, h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c = compute_c5(h_new)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            prob = np.exp(-(new_c - cur_c) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

        # gradual cooling / step reduction
        max_step = max(max_step * 0.9999, 0.001)
        T = max(T * 0.9995, 1e-6)

    return best_h, best_c


# ------------------------------------------------------------
# Refinement via SLSQP with a smooth max approximation
# ------------------------------------------------------------
def refine_slsqp(start: np.ndarray, n: int, deadline: float):
    target = n / 2.0

    def eq_constr(h):
        return np.sum(h) - target

    cons = {"type": "eq", "fun": eq_constr}
    bounds = [(0.0, 1.0)] * n

    h_cur = start.copy()
    for alpha in (5.0, 20.0, 80.0):
        if time.time() > deadline:
            break
        res = minimize(
            lambda h: softmax_c5(h, alpha),
            h_cur,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 800, "ftol": 1e-10, "disp": False},
        )
        if not res.success:
            # continue with the current point
            continue
        h_cur = np.clip(res.x, 0.0, 1.0)

    final_c = compute_c5(h_cur)
    return h_cur, final_c


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function to obtain an improved upper bound for C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96   # safety margin

    # possible discretisations (larger n gives finer resolution)
    candidate_ns = [80, 100, 120, 140]

    # use the parent construction if supplied
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        parent_n = parent.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent = None
        parent_n = None

    best_h = None
    best_c = np.inf
    best_n = None

    # -----------------------------------------------------------------
    # Main optimisation loop – iterate while time remains
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # pick a resolution (cycle through the list for fairness)
        n = candidate_ns[int(time.time() * 1e3) % len(candidate_ns)]

        # ----- build a set of diverse starting points for this n -----
        starts = []

        # parent up‑sample / down‑sample
        if parent is not None:
            starts.append(upsample(parent, n) if parent_n != n else parent.copy())

        # trapezoidal family (10 different slopes)
        for a in np.linspace(0.05, 0.95, 10):
            starts.append(trapezoidal_shape(n, a))

        # few sinusoidal random instances
        for _ in range(2):
            starts.append(sinusoidal_shape(n, rng))

        # a handful of pure random feasible vectors
        for _ in range(4):
            starts.append(random_feasible(n, rng))

        # -----------------------------------------------------------------
        # Optimise every start (hill‑climb → SLSQP) while we have time
        # -----------------------------------------------------------------
        for s in starts:
            if time.time() >= deadline:
                break
            # give each start a modest budget for hill climbing
            sub_deadline = time.time() + min(4.0, (deadline - time.time()) * 0.4)
            h_local, c_local = hill_climb(s, n, sub_deadline, rng)

            # refine with SLSQP if time permits
            if time.time() < deadline:
                h_ref, c_ref = refine_slsqp(h_local, n, deadline)
            else:
                h_ref, c_ref = h_local, c_local

            if c_ref < best_c - 1e-9:
                best_h, best_c, best_n = h_ref, c_ref, n
                # early exit if we already beat the target
                if best_c <= 0.3808:
                    break
        if best_c <= 0.3808:
            break

    # -----------------------------------------------------------------
    # Fallback solution (should never be reached)
    # -----------------------------------------------------------------
    if best_h is None:
        n = 100
        best_h = random_feasible(n, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final sanity‑check: enforce the integral exactly
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-10:
        best_h += (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        # a tiny residual fix
        cur_sum = best_h.sum()
        if abs(cur_sum - target) > 1e-9:
            idx = rng.integers(0, best_n)
            best_h[idx] += (target - cur_sum)
            best_h[idx] = np.clip(best_h[idx], 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
