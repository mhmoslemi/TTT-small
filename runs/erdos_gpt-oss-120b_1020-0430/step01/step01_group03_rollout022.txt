    if s >= 0:

    else:

    return g

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions -------------------------------------------------

def compute_c5(h):
    """Exact discrete C_5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(h, target, max_iter=50):
    """
    Project vector h onto the box [0,1] while fixing the sum to target.
    Finds λ such that x = clip(h + λ, 0, 1) satisfies sum(x)=target.
    """
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(h + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(h + (low + high) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random point in the feasible set {0≤h_i≤1, Σh_i=n/2}."""
    h = rng.random(n)
    return project_sum_box(h, n / 2.0)


# ---------- smooth optimisation (soft‑max) ----------------------------------

CURRENT_TARGET = None   # Σ h_i = CURRENT_TARGET
CURRENT_BETA = None     # softmax temperature inverse


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP refinement using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = project_sum_box(x, CURRENT_TARGET)
    return x


# ---------- sub‑gradient descent on the true max ---------------------------

def subgradient_max_shift(h, shift):
    """
    Sub‑gradient of the correlation term at the given integer shift.
    shift is the integer offset (negative allowed) corresponding to the
    position in np.correlate(..., mode='full').
    """
    n = h.size
    s = abs(shift)
    idx = np.arange(n)
    g = np.zeros_like(h)

    if shift >= 0:
        mask_ind = idx <= n - 1 - s
    else:
        mask_ind = idx >= s
    g[mask_ind] = 1.0

    forward = idx + s
    mask_fwd = forward < n
    g[mask_fwd] -= h[forward[mask_fwd]]

    backward = idx - s
    mask_bwd = backward >= 0
    g[mask_bwd] -= h[backward[mask_bwd]]

    return g


def subgradient_descent(h, time_limit, start_time):
    """
    Reduce the exact max‑overlap objective using sub‑gradient steps.
    Returns a (possibly) improved feasible vector.
    """
    n = h.size
    target = n / 2.0
    eta = 0.2                # initial step size
    decay = 0.9995           # geometric decay
    it = 0
    while time.time() - start_time < time_limit:
        # correlation and current max
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = corr.max()
        # pick a random shift attaining the maximum
        max_positions = np.flatnonzero(corr == max_val)
        shift = int(max_positions[np.random.randint(len(max_positions))] - (n - 1))
        # gradient step
        g = subgradient_max_shift(h, shift)
        h = h - eta * g
        # enforce bounds and sum
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, target)
        eta *= decay
        it += 1
        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = np.random.normal(scale=0.005, size=n)
            h = np.clip(h + jitter, 0.0, 1.0)
            h = project_sum_box(h, target)
    return h


# ---------- main optimisation driver ----------------------------------------

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    # ---- phase A : smooth SLSQP refinements --------------------------------

        if time.time() - start > 0.45 * budget_s:

        try:

            maybe_update(trial)
        except Exception:
            continue

    # ---- phase B : sub‑gradient descent on true max ------------------------
    time_for_sub = 0.30 * budget_s
    best_h = subgradient_descent(best_h, time_for_sub, start)
    maybe_update(best_h)

    # ---- phase C : progressive up‑sampling ---------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > 0.92 * budget_s:

        # double resolution

        h_up = project_sum_box(h_up, n_up / 2.0)

    final_c = compute_c5(best_h)
