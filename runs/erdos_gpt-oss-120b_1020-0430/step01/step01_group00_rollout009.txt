def compute_c5(h):
    n = len(h)
    # Next power of 2 >= 2*n-1
    m = 1 << ((2*n-1).bit_length())
    # Use real FFT
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)
    # conv length m, but linear convolution length = 2*n-1, located at first 2*n-1 elements.
    corr = conv[:2*n-1]
    max_corr = corr.max()
    dx = 2.0 / n
    return max_corr * dx

def simulated_annealing(initial_h, max_iters, budget_time_left):
    h = initial_h.copy()
    n = len(h)
    dx = 2.0/n
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    # schedule parameters
    start_time = time.time()
    for it in range(max_iters):
        if time.time() - start_time > budget_time_left:
            break
        # temperature schedule: linear decreasing
        progress = it / max_iters
        temperature = max(1e-6, (0.001 * (1 - progress) + 1e-7))
        # random pair
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # ensure i > j? not needed
        # Determine feasible delta
        if h[i] > 0 and h[j] < 1:
            max_delta = min(h[i], 1 - h[j])
            source = i
            target = j
        elif h[j] > 0 and h[i] < 1:
            max_delta = min(h[j], 1 - h[i])
            source = j
            target = i
        else:
            continue
        # Step magnitude: proportional to max_delta and schedule
        # Use random factor
        delta = max_delta * (0.1 + 0.9 * (1 - progress)) * np.random.rand()
        # Avoid too small delta that doesn't change val
        if delta < 1e-8:
            continue
        # propose
        h_new = h.copy()
        h_new[source] -= delta
        h_new[target] += delta
        # compute new val
        new_val = compute_c5(h_new)
        # acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / temperature):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
    return best_h, best_val

def compute_c5(h):
    n = len(h)
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)
    corr = conv[:2*n-1]
    max_corr = np.max(corr)
    dx = 2.0 / n
    return max_corr * dx

def slsqp_optimize(h0, max_iter=200):
    n = len(h0)
    # ensure feasible initial
    h0 = _project_to_feasible(h0, n)
    corr0 = np.correlate(h0, 1 - h0, mode='full')
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    def objective(x):
        return x[-1]

    def eq_constraint(x):
        return np.sum(x[:n]) - n/2

    def ineq_constraint(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full')
        return t - corr

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [{'type': 'eq', 'fun': eq_constraint},
                   {'type': 'ineq', 'fun': ineq_constraint}]

    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = optimize.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options=opts)
    h_opt = res.x[:n] if res.success else h0
    # Ensure feasibility final
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, compute_c5(h_opt)

    h = np.clip(h, 0.0, 1.0)

    total = h.sum()

    if abs(total - target) <= tol:

    #   else add to entries <1
    #   compute per-item delta = diff / count
    #   adjust with clipping.

        if abs(diff) <= tol:
            break
        if diff > 0:
            # Need to reduce

        else:
            idx = np.where(h < 1)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        # Apply
        h[idx] -= delta
        # Clip
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()

            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

    n = h.size
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)[: 2 * n - 1]
    return conv.max() * (2.0 / n)

# ---------- upscale (linear interpolation) ----------
def _upscale_h(coarse: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse.size
    x_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    fine = np.interp(x_f, x_c, coarse)
    return _project_to_feasible(fine, n_fine)

# ---------- simulated annealing ----------
def _simulated_annealing(h0: np.ndarray, max_iters: int,
                         budget_s: float, start_time: float) -> tuple:
    n = h0.size
    h = h0.copy()
    cur_val = _c5_from_h(h)
    best_h, best_val = h.copy(), cur_val

    # temperature schedule (linear)
    T_start, T_end = 2e-4, 1e-7

    for it in range(max_iters):
        if time.time() - start_time > budget_s - 0.2:
            break

        prog = it / max_iters
        T = T_start * (1 - prog) + T_end * prog

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # decide direction (source -> target)
        if h[i] > 0 and h[j] < 1:
            src, tgt = i, j
        elif h[j] > 0 and h[i] < 1:
            src, tgt = j, i
        else:
            continue

        max_delta = min(h[src], 1.0 - h[tgt])
        if max_delta <= 0:
            continue

        coeff = 0.2 * (1 - prog) + 0.01          # step size factor
        delta = max_delta * coeff * np.random.rand()
        if delta < 1e-13:
            continue

        h_new = h.copy()
        h_new[src] -= delta
        h_new[tgt] += delta

        new_val = _c5_from_h(h_new)

        if new_val < cur_val:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            h, cur_val = h_new, new_val
            if new_val < best_val:
                best_h, best_val = h_new.copy(), new_val

    return best_h, best_val

# ---------- SLSQP refinement ----------
def _refine_slsqp(h0: np.ndarray, max_iter: int = 300) -> tuple:
    n = h0.size
    h0 = _project_to_feasible(h0, n)

    t0 = np.correlate(h0, 1.0 - h0, mode="full").max()
    x0 = np.concatenate([h0, [t0]])

    def obj(x):
        return x[-1]

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "eq", "fun": eq},
            {"type": "ineq", "fun": ineq}]

    res = optimize.minimize(obj, x0, method="SLSQP",
                            bounds=bounds, constraints=cons,
                            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)

# ---------- main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    target = 0.3808

    # obtain possible parent construction
    try:
        init_h = np.array(initial_h_values, dtype=float)
        have_parent = True
    except NameError:
        init_h = None
        have_parent = False

    best_h, best_val, best_n = None, float("inf"), None

    # candidate grid sizes
    cand = []
    if have_parent:
        cand.append(init_h.size)
    for s in (256, 300, 384):
        cand.append(s)
    cand = sorted(set(cand))

    for n in cand:
        if time.time() - start > budget_s - 5:
            break

        # build base h for this size
        if have_parent:
            base = _upscale_h(init_h, n) if init_h.size != n else init_h.copy()
        else:
            base = np.full(n, 0.5)
        base = _project_to_feasible(base, n)

        # allocate iterations based on remaining time
        time_left = budget_s - (time.time() - start)
        # generous iteration count (up to ~200k for n≈256)
        max_iters = int(min(200_000, 5e5 * (256 / n) * (time_left / budget_s)))
        max_iters = max(30_000, max_iters)

        # several restarts with slight noise
        for restart in range(3):
            if time.time() - start > budget_s - 5:
                break
            if restart == 0:
                h0 = base.copy()
            else:
                noise = np.random.uniform(-0.08, 0.08, size=n)
                h0 = _project_to_feasible(base + noise, n)
            h_sa, val_sa = _simulated_annealing(h0, max_iters, budget_s, start)
            if val_sa < best_val:
                best_h, best_val, best_n = h_sa.copy(), val_sa, n
                if best_val < target:
                    break

        if best_h is not None and best_n == n and best_val < target:
            break

        # refine with SLSQP if time permits
        if best_h is not None and best_n == n and time.time() - start < budget_s - 5:
            h_ref, val_ref = _refine_slsqp(best_h, max_iter=300)
            if val_ref < best_val:
                best_h, best_val, best_n = h_ref.copy(), val_ref, n
                if best_val < target:
                    break

    # fallback
    if best_h is None:
        n = 64
        best_h = np.full(n, 0.5)
        best_h = _project_to_feasible(best_h, n)
        best_val = _c5_from_h(best_h)
        best_n = n

    # final sanity projection
    best_h = _project_to_feasible(np.array(best_h, dtype=float), best_n)
    best_val = _c5_from_h(best_h)

    return best_h.tolist(), float(best_val), int(best_n)
