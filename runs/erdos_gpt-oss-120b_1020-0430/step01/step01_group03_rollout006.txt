def softmax_obj_and_grad(x, beta):
    n = len(x)
    corr = np.correlate(x, 1-x, mode='full')
    z = beta * corr
    max_z = np.max(z)
    # stable exponentials
    exp_z = np.exp(z - max_z)
    sum_exp = np.sum(exp_z)
    w = exp_z / sum_exp  # shape 2n-1

    # objective
    obj = (max_z + np.log(sum_exp)) / beta

    # gradient
    grad = np.zeros_like(x)
    # shift range from -(n-1) to (n-1)
    for idx,s in enumerate(range(-(n-1), n)):
        weight = w[idx]
        # compute vector contribution for shift s
        # compute term "1 - x_{i+s} - x_{i-s}"
        # start with array of ones
        vec = np.ones(n)
        # i+s within bounds
        if s >= 0:
            # i from 0..n-1-s: i+s = i + s in [s .. n-1]
            vec[:n-s] -= x[s:]
            # i-s within bounds: i-s >=0 => i >= s
            vec[s:] -= x[:n-s]
        else:
            s2 = -s
            # s negative => s = -s2
            # i+s = i - s2, valid when i >= s2
            vec[s2:] -= x[:n-s2]   # x_{i+s} = x_{i-s2}
            # i-s = i + s2, valid when i <= n-1-s2 => i < n-s2
            vec[:n-s2] -= x[s2:]   # x_{i-s} = x_{i+s2}
        grad += weight * vec
    return obj, grad

def simulated_anneal(h0, rng, max_iters=2000, init_temp=0.01, decay=0.995):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    temp = init_temp
    for it in range(max_iters):
        # propose perturbation: transfer epsilon from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.2  # step scale
        # apply
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        delta = c - best_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h = new_h
            if c < best_c:
                best_c = c
                best_h = new_h.copy()
        temp *= decay
        if temp < 1e-8:
            temp = 1e-8
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# global placeholders for the current optimisation problem
CURRENT_BETA = None
CURRENT_N = None
CURRENT_TARGET = None


# ----------------------------------------------------------------------
#  Constraint & objective (soft‑max surrogate) with analytic gradient
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k})."""
    n = CURRENT_N
    beta = CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def softmax_grad(x):
    """Analytic gradient of the soft‑max objective."""
    n = CURRENT_N
    beta = CURRENT_BETA
    # correlation (full) and weights
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    ex = np.exp(bc - m)
    w = ex / np.sum(ex)          # shape 2n‑1, sum to 1
    grad = np.zeros_like(x)
    idxs = np.arange(n)
    shift0 = -(n - 1)
    for idx, weight in enumerate(w):
        shift = shift0 + idx
        vec = np.ones(n)
        ip = idxs + shift
        mask = (ip >= 0) & (ip < n)
        if mask.any():
            vec[mask] -= x[ip[mask]]
        im = idxs - shift
        mask2 = (im >= 0) & (im < n)
        if mask2.any():
            vec[mask2] -= x[im[mask2]]
        grad += weight * vec
    return grad

def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

    h = np.full(n, 0.5)

            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # slowly shrink step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def upsample_vector(h):
    """Linear up‑sampling by a factor 2 (preserves integral)."""
    n = h.size
    dx = 2.0 / n
    x_old = np.arange(n) * dx + dx / 2.0
    n2 = 2 * n
    dx2 = 2.0 / n2
    x_new = np.arange(n2) * dx2 + dx2 / 2.0
    return np.interp(x_new, x_old, h)


def slsqp_optimize_with_grad(h0, beta, max_iter=400):
    """Run SLSQP on the soft‑max surrogate with analytic gradient."""
    global CURRENT_N, CURRENT_BETA, CURRENT_TARGET
    n = h0.size
    CURRENT_N = n
    CURRENT_BETA = beta
    CURRENT_TARGET = n / 2.0

        method="SLSQP",
        jac=softmax_grad,
        bounds=bounds,
        constraints=cons,

            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
