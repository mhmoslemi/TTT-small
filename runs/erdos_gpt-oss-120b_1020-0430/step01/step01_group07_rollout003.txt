function optimize_h(h_init, total_time):
    start_time = time.time()
    best_h = h_init.copy(); best_c = compute_c5_and_shift(best_h)[0]
    h = h_init.copy()
    n = len(h)
    target = n/2.0
    # parameters
    init_alpha = 0.5
    min_alpha = 1e-4
    decay_rate = 0.999  # multiply each iteration
    alpha = init_alpha
    # iteration loop
    while time.time() - start_time < total_time:
        c, shift = compute_c5_and_shift(h)
        if c < best_c:
            best_c = c; best_h = h.copy()
        # compute gradient g
        # define shift offset
        s = shift - (n-1)  # shift s can be negative; integer
        # compute g
        g = np.ones_like(h)  # start with 1
        # subtract h_{i+s} where valid
        if s >= 0:
            g[:n-s] -= h[s:]  # h_{i+s}
            # For i-s part
            g[s:] -= h[:n-s]  # h_{i-s}
        else:
            s_abs = -s
            g[s_abs:] -= h[:-s_abs]  # h_{i+s}
            g[:n-s_abs] -= h[s_abs:]  # h_{i-s}
        # Now g = 1 - h_{i+s} - h_{i-s}, with missing terms zero automatically because we didn't subtract for out-of-bounds.
        # compute descent direction d = -g
        d = -g
        # project to zero-sum
        d -= d.mean()
        # choose step size adaptively
        # limit max delta change per component
        max_step = 0.2
        # compute step size limited by max_step and staying within bounds.
        # we can compute feasible α_max s.t. h + α d stays in [0,1]
        # For each i, compute α_i_max = (bound - h_i)/d_i for d_i>0 else (0 - h_i)/d_i for d_i<0. We want α so that all constraints satisfied.
        # α_max = min over positive d_i of (1 - h_i)/d_i and over negative d_i of (-h_i)/d_i (since d_i negative)
        # α_min = max over positive d_i of (0 - h_i)/d_i [but that's negative]; we just need max α that satisfies both.
        # Implementation:
        pos = d > 0
        neg = d < 0
        if np.any(pos):
            alpha_pos = np.min((1 - h[pos]) / d[pos])
        else:
            alpha_pos = np.inf
        if np.any(neg):
            alpha_neg = np.min((-h[neg]) / d[neg])
        else:
            alpha_neg = np.inf
        alpha_max = min(alpha_pos, alpha_neg)
        # also limit by max_step magnitude
        alpha_step = min(alpha, alpha_max, max_step)
        if alpha_step <= 0:
            # cannot move, break
            alpha = max(alpha * decay_rate, min_alpha)
            continue
        h = h + alpha_step * d
        # ensure numerical within bounds
        np.clip(h, 0.0, 1.0, out=h)
        # enforce sum constraint exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust one coordinate with slack
            # find index where adjustment won't violate bounds
            if diff > 0:
                idx_candidates = np.where(h < 1 - 1e-12)[0]
            else:
                idx_candidates = np.where(h > 1e-12)[0]
            if idx_candidates.size > 0:
                idx = idx_candidates[0]
                h[idx] += diff
                # clip again
                h[idx] = min(max(h[idx],0.0),1.0)
        # update alpha decay
        alpha = max(alpha * decay_rate, min_alpha)
    return best_h, best_c

def c5_and_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    cmax = corr[max_idx] * dx
    return cmax, max_idx

def subgradient_descent(h_init, time_limit):
    start = time.time()
    n = h_init.shape[0]
    target = n / 2.0
    best_h = h_init.copy()
    best_c, _ = c5_and_shift(best_h)
    h = h_init.copy()
    # parameters
    alpha = 0.5
    min_alpha = 1e-5
    decay = 0.9995
    max_step = 0.1   # limit step size
    while time.time() - start < time_limit:
        c, idx = c5_and_shift(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        shift = idx - (n - 1)  # integer shift
        # compute gradient
        g = np.ones_like(h)
        s = shift
        if s >= 0:
            # part for i+s
            g[:n-s] -= h[s:]
            # part for i-s
            g[s:] -= h[:n-s]
        else:
            s_abs = -s
            # i+s term: i-s_abs
            g[s_abs:] -= h[:-s_abs]
            # i-s term: i+s_abs
            g[:n-s_abs] -= h[s_abs:]

        d = -g
        # zero-sum projection
        d -= d.mean()
        # compute max feasible step size
        pos = d > 0
        neg = d < 0
        # compute upper bound for step
        if np.any(pos):
            step_pos = np.min((1.0 - h[pos]) / d[pos])
        else:
            step_pos = np.inf
        if np.any(neg):
            step_neg = np.min((-h[neg]) / d[neg])
        else:
            step_neg = np.inf
        step_max = min(step_pos, step_neg, max_step)
        step = min(alpha, step_max)
        if step <= 0:
            # reduce alpha and continue
            alpha *= decay
            if alpha < min_alpha:
                alpha = min_alpha
            continue
        h = h + step * d
        # clipping due to numeric
        np.clip(h, 0.0, 1.0, out=h)
        # enforce sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # find index with slack
            if diff > 0:
                idxs = np.where(h < 1 - 1e-12)[0]
            else:
                idxs = np.where(h > 1e-12)[0]
            if idxs.size > 0:
                idx_adj = idxs[0]
                h[idx_adj] += diff
                if h[idx_adj] < 0.0:
                    h[idx_adj] = 0.0
                elif h[idx_adj] > 1.0:
                    h[idx_adj] = 1.0
        # decay alpha
        alpha = max(alpha * decay, min_alpha)
    return best_h, best_c

# ----------------------------------------------------------------------
def _c5_metric(h: np.ndarray) -> float:
    """Return the C_5 value for a step function sampled by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _c5_metric_and_shift(h: np.ndarray):
    """Return (C_5, shift_index) where shift_index is the index in the
    full correlation array that attains the maximum."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * dx, idx


# ----------------------------------------------------------------------
def _subgradient_descent(start_h: np.ndarray, time_limit: float):
    """Projected sub‑gradient descent on the worst shift.
    Returns the best vector encountered and its C_5 value."""
    start = time.time()
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0

    best_h = h.copy()
    best_c, _ = _c5_metric_and_shift(best_h)

    # step‑size schedule
    alpha = 0.5
    min_alpha = 1e-5
    decay = 0.9995
    max_step = 0.12

    while time.time() - start < time_limit:
        # evaluate current worst shift
        cur_c, idx = _c5_metric_and_shift(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # shift corresponding to the maximal correlation
        shift = idx - (n - 1)          # integer shift (may be negative)

        # gradient of the active term: g_i = 1 - h_{i+shift} - h_{i-shift}
        g = np.ones_like(h)

        s = shift
        if s >= 0:
            # i+shift term
            g[: n - s] -= h[s:]
            # i-shift term
            g[s:] -= h[: n - s]
        else:
            s_abs = -s
            g[s_abs:] -= h[: n - s_abs]
            g[: n - s_abs] -= h[s_abs:]

        # descent direction (negative gradient) and zero‑sum projection
        d = -g
        d -= d.mean()

        # feasible step respecting box constraints
        pos = d > 0
        neg = d < 0
        if np.any(pos):
            step_pos = np.min((1.0 - h[pos]) / d[pos])
        else:
            step_pos = np.inf
        if np.any(neg):
            step_neg = np.min((-h[neg]) / d[neg])
        else:
            step_neg = np.inf
        step_max = min(step_pos, step_neg, max_step)

        step = min(alpha, step_max)
        if step <= 0:
            alpha = max(alpha * decay, min_alpha)
            continue

        h += step * d
        np.clip(h, 0.0, 1.0, out=h)

        # enforce exact integral (tiny correction)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                h[i] += diff
                h[i] = min(max(h[i], 0.0), 1.0)

        alpha = max(alpha * decay, min_alpha)

    return best_h, best_c
