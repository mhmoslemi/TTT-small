def compute_autocorr_fft(h):
    n = len(h)
    nfft = 1 << (2*n-1).bit_length()
    H = np.fft.rfft(h, nfft)

def shape_plateau(w, n):
    # generate h defined by plateau width w (0 <= w <= 2)
    # compute heights as h0 = 2/(w+2)
    # a = (2-w)/2
    a = (2 - w)/2
    h0 = 2.0 / (w + 2.0)
    # generate x coordinates midpoint of bins
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.zeros_like(xs)
    # left region: xs < a => linear from 0 to h0
    left = xs < a
    h[left] = h0 * (xs[left] / a)
    # plateau: a <= xs <= a + w
    plateau = (xs >= a) & (xs <= a + w)
    h[plateau] = h0
    # right region: xs > a + w
    right = xs > a + w
    h[right] = h0 * ((2 - xs[right]) / a)
    return h
```

import time, random, math
import numpy as np
from scipy.optimize import minimize

TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n with the equality sum(x)=target via bisection."""
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        p = np.clip(x - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact max overlap using FFT‑based autocorrelation."""
    n = h.shape[0]
    dx = 2.0 / n
    total = h.sum()
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    autocorr = np.fft.irfft(H * np.conj(H), n=nfft)[: 2 * n - 1]
    C = (total - autocorr) * dx
    return float(C.max())


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_plateau(w: float, n: int) -> np.ndarray:
    """Symmetric trapezoid: linear rise, flat top of width w, linear fall."""
    if w <= 0.0:
        return make_triangle(n)
    if w >= 2.0:
        return project_onto_bounds_and_sum(np.full(n, 0.5), n / 2.0)
    a = (2.0 - w) / 2.0
    h0 = 2.0 / (w + 2.0)            # ensures area = 1
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.empty_like(pts)
    left = pts < a
    h[left] = h0 * (pts[left] / a)
    plateau = (pts >= a) & (pts <= a + w)
    h[plateau] = h0
    right = pts > a + w
    h[right] = h0 * ((2.0 - pts[right]) / a)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_beta(p: float, n: int) -> np.ndarray:
    """Shape ~ x^p (2-x)^p, rescaled to unit area."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = (pts ** p) * ((2.0 - pts) ** p)
    target = n / 2.0
    total = h.sum()
    if total == 0:
        h = np.full(n, 0.5)
    else:
        h = h * (target / total)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, target)


def random_feasible(n: int) -> np.ndarray:
    rnd = np.random.rand(n)
    return project_onto_bounds_and_sum(rnd, n / 2.0)


def propose_move(h: np.ndarray, max_step: float = 0.05):
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    delta_low = max(-hi, hj - 1.0)
    delta_high = min(1.0 - hi, hj)
    if delta_high - delta_low < 1e-12:
        return None
    low = max(delta_low, -max_step)
    high = min(delta_high, max_step)
    if high <= low:
        return None
    delta = np.random.uniform(low, high)
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    return cand


def local_search(h: np.ndarray, cur_val: float, start: float, budget: float):
    """Simple SA / hill‑climb on the pairwise‑transfer manifold."""
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    target = n / 2.0
    T = 0.02
    it = 0
    while time.time() - start < budget - 0.5:
        cand = propose_move(best_h, max_step=0.07)
        if cand is None:
            continue
        cand = project_onto_bounds_and_sum(cand, target)
        val = compute_c5(cand)
        if val < best_val or np.random.rand() < math.exp(-(val - best_val) / max(T, 1e-12)):
            best_h = cand
            best_val = val
        T *= 0.99996
        it += 1
        if best_val <= 0.3808:
            break
        # occasional random restart to escape stagnation
        if it % 60000 == 0 and np.random.rand() < 0.02:
            cand = random_feasible(n)
            cand_val = compute_c5(cand)
            if cand_val < best_val:
                best_h = cand
                best_val = cand_val
    return best_h, best_val


def final_slsqp_refine(h: np.ndarray):
    """Brief SLSQP refinement using a smooth max approximation."""
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def sum_eq(x):
        return np.sum(x) - target

    def smooth_obj(x):
        # autocorrelation via FFT
        nfft = 1 << ((2 * n - 1).bit_length())
        X = np.fft.rfft(x, n=nfft)
        ac = np.fft.irfft(X * np.conj(X), n=nfft)[: 2 * n - 1]
        C = (np.sum(x) - ac) * dx
        alpha = 200.0
        m = np.max(C)
        # log‑sum‑exp for smooth max
        return m + np.log(np.exp(alpha * (C - m)).sum()) / alpha

    try:
        res = minimize(
            smooth_obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints={"type": "eq", "fun": sum_eq},
            options={"maxiter": 250, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h
    return project_onto_bounds_and_sum(res.x, target)
