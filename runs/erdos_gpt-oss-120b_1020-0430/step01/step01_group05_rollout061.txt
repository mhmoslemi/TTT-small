    n = h.shape[0]

    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1 - h, N)

def compute_c5_fft(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    # rfft returns real-valued FFT for real input

def make_rectangle(n, width=1.0):

def compute_corr_fft_full(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(A * np.conj(B), N)[:2*n - 1]
    return corr
```

import time
import numpy as np

# ---------- projection onto [0,1]^n with fixed sum ----------
def project_onto_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n intersected with sum(v)=target."""
    low, high = -1.0, 1.0  # guarantees sum at low > target, at high < target
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


# ---------- analytic seed constructions ----------
def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(x - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * x / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rectangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    left, right = 0.5, 1.5  # width 1 centred at 1
    h = np.where((x >= left) & (x <= right), 1.0, 0.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


# ---------- correlation utilities ----------
def correlation_full(h: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) correlation of h with 1‑h, length 2n‑1."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(A * np.conj(B), N)[: 2 * n - 1]
    return corr


def compute_c5_fft(h: np.ndarray) -> float:
    """C₅ evaluated via FFT (fast)."""
    n = h.shape[0]
    corr = correlation_full(h)
    return corr.max() * (2.0 / n)


# ---------- gradient for the worst shift ----------
def gradient_worst_shift(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of g(k) at the lag with maximal overlap."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            grad[: n - lag] += 1.0 - h[lag:]
            grad[lag:] -= h[: n - lag]
    else:
        l = -lag
        if l < n:
            grad[l:] += 1.0 - h[: n - l]
            grad[: n - l] -= h[l:]
    return grad


# ---------- local projected gradient descent ----------
def local_optimize(
    h0: np.ndarray,
    target: float,
    max_iter: int = 300,
    step: float = 0.15,
    decay: float = 0.95,
) -> tuple[np.ndarray, float]:
    """Iteratively reduce the worst overlap by projected gradient steps."""
    cur = h0.copy()
    cur_val = compute_c5_fft(cur)
    for _ in range(max_iter):
        corr = correlation_full(cur)
        idx = int(np.argmax(corr))
        lag = idx - (len(cur) - 1)
        grad = gradient_worst_shift(cur, lag)
        if np.all(np.abs(grad) < 1e-12):
            break
        cand = cur - step * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target)
        new_val = compute_c5_fft(cand)
        if new_val < cur_val:
            cur, cur_val = cand, new_val
        else:
            step *= decay
            if step < 1e-6:
                break
    return cur, cur_val


# ---------- random feasible start ----------
def random_feasible_vector(rng: np.random.Generator, n: int, target: float) -> np.ndarray:
    v = rng.random(n)
    return project_onto_bounds_and_sum(v, target)
