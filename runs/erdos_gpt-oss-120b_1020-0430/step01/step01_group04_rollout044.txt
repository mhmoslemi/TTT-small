import time, numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(alpha * conv) / alpha)

def generate_triangular(n):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri

def upsample(arr, new_n):
    old_n = arr.shape[0]
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

def anneal(h_start, rng, time_deadline):
    h = h_start.copy()
    n = h.shape[0]
    target = n / 2.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # Annealing parameters
    T0 = 0.1
    T_end = 1e-5
    max_iters = 200000  # large upper bound
    iter = 0
    while time.time() < time_deadline and iter < max_iters:
        iter += 1
        # pick two distinct indices
        i, j = rng.integers(n, size=2)
        while i == j:
            j = rng.integers(n)
        # compute feasible delta range
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi <= lo:
            continue
        # choose delta uniformly
        delta = rng.uniform(lo, hi)
        # construct new h
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # evaluate
        c5_new = compute_c5(h_new)
        # temperature schedule linearly decreasing based on iteration fraction
        frac = iter / max_iters
        T = T0 * (1 - frac) + T_end * frac
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T, 1e-12)):
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def refine_with_slsqp(h_init, n, rng, time_deadline):
    target_sum = n / 2.0
    def constraint_eq(h):
        return np.sum(h) - target_sum
    cons = {"type":"eq", "fun": constraint_eq}
    bounds = [(0.0, 1.0)] * n
    # two-stage smoothing
    # Stage 1: α = 10
    res1 = minimize(lambda h: softmax_c5(h, alpha=10.0), h_init,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter':500, 'ftol':1e-9, 'disp':False})
    if not res1.success or time.time() > time_deadline:
        return h_init
    # Stage 2: α = 50
    res2 = minimize(lambda h: softmax_c5(h, alpha=50.0), res1.x,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter':500, 'ftol':1e-9, 'disp':False})
    if not res2.success or time.time() > time_deadline:
        return res1.x
    return np.clip(res2.x, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95
    # Choose resolution
    n = 200
    # Build base start
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        parent_n = parent.shape[0]
        if parent_n != n:
            h0 = upsample(parent, n)
        else:
            h0 = parent.copy()
    except NameError:
        h0 = generate_triangular(n)
    # Ensure feasibility
    target = n / 2.0
    cur_sum = h0.sum()
    if abs(cur_sum - target) > 1e-12:
        h0 = h0 + (target - cur_sum) / n
        h0 = np.clip(h0, 0.0, 1.0)
    # Multiple annealing restarts
    best_h = None
    best_c5 = np.inf
    restarts = 5
    for r in range(restarts):
        if time.time() > time_limit:
            break
        # add random perturbation to starting point for diversity
        h_start = h0.copy()
        # small random shuffle: add small gaussian noise and renorm
        noise = rng.normal(scale=0.05, size=n)
        h_start = h_start + noise
        h_start = np.clip(h_start, 0.0, 1.0)
        # renormalize sum
        diff = target - h_start.sum()
        h_start = h_start + diff / n
        h_start = np.clip(h_start, 0.0, 1.0)
        # allocate part of remaining time
        remaining = time_limit - time.time()
        # allocate roughly equal time for each remaining restart
        alloc = remaining / max(1, restarts - r)
        deadline = time.time() + alloc * 0.9
        h_ann, c5_ann = anneal(h_start, rng, deadline)
        # refine
        if time.time() < time_limit:
            h_ref = refine_with_slsqp(h_ann, n, rng, time_limit)
            c5_ref = compute_c5(h_ref)
        else:
            h_ref = h_ann
            c5_ref = c5_ann
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
    # fallback
    if best_h is None:
        best_h = generate_triangular(n)
        best_c5 = compute_c5(best_h)
    # final check and repair sum
    final_sum = best_h.sum()
    if abs(final_sum - target) > 1e-8:
        best_h = best_h + (target - final_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap via log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, renormalised to satisfy ∑h = n/2."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, keeping the integral exact."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


# ------------------------------------------------------------
# Simulated annealing (keeps the sum constraint exactly)
# ------------------------------------------------------------
def anneal(h_start: np.ndarray, rng: np.random.Generator, deadline: float) -> tuple:
    h = h_start.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5

    T0, T_end = 0.12, 5e-6          # temperature schedule
    max_iters = 200_000
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = rng.integers(n, size=2)
        while i == j:
            j = rng.integers(n)

        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi <= lo:
            continue
        delta = rng.uniform(lo, hi)

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        c5_new = compute_c5(h_new)

        frac = it / max_iters
        T = T0 * (1.0 - frac) + T_end * frac
        accept = c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T, 1e-12))
        if accept:
            h, cur_c5 = h_new, c5_new
            if cur_c5 < best_c5:
                best_h, best_c5 = h.copy(), cur_c5
    return best_h, best_c5


# ------------------------------------------------------------
# Final refinement with SLSQP (smooth max approximation)
# ------------------------------------------------------------
def refine_slsqp(h0: np.ndarray, n: int, rng: np.random.Generator, deadline: float) -> np.ndarray:
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    bounds = [(0.0, 1.0)] * n

    # stage 1 – milder smoothing
    res1 = minimize(lambda h: softmax_c5(h, alpha=12.0),
                    h0, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 600, "ftol": 1e-9, "disp": False})
    if not res1.success or time.time() > deadline:
        return np.clip(h0, 0.0, 1.0)

    # stage 2 – sharper smoothing
    res2 = minimize(lambda h: softmax_c5(h, alpha=80.0),
                    res1.x, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 600, "ftol": 1e-9, "disp": False})
    if not res2.success or time.time() > deadline:
        return np.clip(res1.x, 0.0, 1.0)

    return np.clip(res2.x, 0.0, 1.0)
