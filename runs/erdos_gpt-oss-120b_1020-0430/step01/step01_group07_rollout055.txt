    dx = 2.0 / n

    T_end = 1e-9
    max_step_initial = 0.3
    eps = 1e-12
    while time.time() - start < time_limit:

        progress = elapsed / time_limit
        T = T0 * (T_end / T0) ** progress
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c = corr[max_idx] * dx
        shift = max_idx - (n - 1)
        if shift >= 0:
            i_start = 0
            i_end = n - shift
        else:
            i_start = -shift
            i_end = n
        # find source i
        partner = h[i_start + shift : i_end + shift]
        w = 1.0 - partner
        h_sub = h[i_start:i_end]
        mask_i = h_sub > eps
        if np.any(mask_i):
            w_i = w.copy()
            w_i[~mask_i] = -1.0
            i_local = int(np.argmax(w_i))
            i = i_start + i_local
            # choose j
            mask_outside = np.ones(n, dtype=bool)
            mask_outside[i_start:i_end] = False
            cand_out = np.where(mask_outside & (h < 1 - eps))[0]
            if cand_out.size > 0:
                j = int(np.random.choice(cand_out))
            else:
                # pick inside with minimal w where h<1
                inner_mask = ~mask_outside & (h < 1 - eps)
                if np.any(inner_mask):
                    # use w values for those indices within range
                    w_full = np.full(n, np.inf)
                    w_full[i_start:i_end] = w
                    w_full[~inner_mask] = np.inf  # invalid
                    j = int(np.argmin(w_full))
                else:
                    # fallback random
                    j = random.randrange(n)

            # fallback random pair
            i = random.randrange(n)
            j = random.randrange(n)

                j = random.randrange(n)

        low_cap = max(low, -max_step)
        high_cap = min(high, max_step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)
        # apply move

        h[j] -= delta
        # evaluate new C5

                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
    return best_h, best_c

import time, math, random, numpy as np

def _c5_metric(h):
    """C5 value for discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _solve_sa(start_h, time_limit):
    """Targeted simulated‑annealing on a fixed grid."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    # initial best
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c = np.max(corr) * dx
    best_h = h.copy()
    cur_c = best_c
    start = time.time()
    # schedule constants
    T0, Tend = 0.001, 1e-9
    max_step0 = 0.3
    eps = 1e-12
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = elapsed / time_limit
        T = T0 * (Tend / T0) ** prog
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_c = corr[idx] * dx
        shift = idx - (n - 1)
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n
        # source choice (i)
        partner = h[i_start + shift : i_end + shift]
        w = 1.0 - partner                     # larger -> better source
        h_sub = h[i_start:i_end]
        mask_src = h_sub > eps
        if np.any(mask_src):
            w_src = w.copy()
            w_src[~mask_src] = -1.0
            i_local = int(np.argmax(w_src))
            i = i_start + i_local
            # destination (j) – preferably outside the affected interval
            mask_out = np.ones(n, dtype=bool)
            mask_out[i_start:i_end] = False
            cand_out = np.where(mask_out & (h < 1.0 - eps))[0]
            if cand_out.size:
                j = int(np.random.choice(cand_out))
            else:
                inner_ok = (~mask_out) & (h < 1.0 - eps)
                if np.any(inner_ok):
                    w_full = np.full(n, np.inf)
                    w_full[i_start:i_end] = w
                    w_full[~inner_ok] = np.inf
                    j = int(np.argmin(w_full))
                else:
                    j = random.randrange(n)
                    while j == i:
                        j = random.randrange(n)
        else:
            # fallback random pair
            i, j = random.sample(range(n), 2)
        # feasible delta for move i += delta, j -= delta
        low = max(-h[i], h[j] - 1.0)
        high = min(1.0 - h[i], h[j])
        if low >= high:
            continue
        max_step = max_step0 * (1.0 - prog)
        lo = max(low, -max_step)
        hi = min(high, max_step)
        delta = random.uniform(lo, hi) if lo < hi else random.uniform(low, high)
        # apply
        h[i] += delta
        h[j] -= delta
        # evaluate
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        new_c = np.max(corr_new) * dx
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_c
