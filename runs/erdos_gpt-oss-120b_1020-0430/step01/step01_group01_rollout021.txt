def anneal_binary(h, target, dx, rng, time_limit):
    best_h = h.copy()
    best_val = compute_c5(best_h,dx)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    # schedule: T_i = T0 * (1 - i / max_iter)
    T0 = 0.02  # set based on typical C5 differences ~0.001
    # approximate max iterations: compute based on time budget.
    while time.time() - start < time_limit:
        # propose swap
        # choose i where cur_h[i]==1, j where cur_h[j]==0
        i = rng.choice(np.where(cur_h==1)[0])
        j = rng.choice(np.where(cur_h==0)[0])
        # swap
        cur_h[i], cur_h[j] = 0, 1
        new_val = compute_c5(cur_h,dx)
        if new_val < cur_val or rng.rand() < np.exp(-(new_val-cur_val)/T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1, 0
        # update temperature
        # maybe linear cooling: T = T0 * (1 - (time_elapsed)/time_limit)
        T = T0 * (1 - (time.time()-start)/time_limit)
        # ensure T > minimal
        if T < 1e-6: T = 1e-6
    return best_h, best_val

def _proj_capped_simplex(v, target, tol=1e-9, max_iter=100):
    lo = -np.max(v) - 1
    hi = np.max(v) + 1
    for _ in range(max_iter):
        lam = (lo+hi)/2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)
```

def _upscale(vec, n):
    x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Projection onto the capped simplex {0<=x<=1, sum(x)=target}
# --------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 tol: float = 1e-9,
                                 max_iter: int = 80) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# --------------------------------------------------------------
# Random feasible (continuous) vector
# --------------------------------------------------------------
def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


# --------------------------------------------------------------
# Correlation based C5 evaluation
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray, dx: float) -> float:
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


# --------------------------------------------------------------
# Gradient of the correlation at the shift attaining the maximum
# --------------------------------------------------------------
def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        g[s:] += -h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        g[: n - s] += -h[s:]
    return g


# --------------------------------------------------------------
# Projected sub‑gradient descent on the max‑correlation objective
# --------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, target: float,
                 max_iter: int = 1200,
                 step0: float = 1.0,
                 decay: float = 0.997) -> np.ndarray:
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


# --------------------------------------------------------------
# Smooth approximation of max (log‑sum‑exp)
# --------------------------------------------------------------
def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


# --------------------------------------------------------------
# SLSQP refinement of a smooth max objective
# --------------------------------------------------------------
def _slsqp_optimize(h0: np.ndarray, target: float,
                    alpha: float = 30.0,
                    max_iter: int = 1200) -> np.ndarray:
    n = h0.size

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # ----- determine discretisation size -----------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        n0 = len(base_h)
    except Exception:
        base_h = None
        n0 = 0

    n = max(400, n0)                # aim for a richer grid
    if n % 2:
        n += 1                       # make the target sum integer
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- simple linear up‑sampling ---------------------------
    def _upscale(vec):
        if len(vec) == n:
            return vec.copy()
        x_old = np.linspace(0.0, 2.0, len(vec), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        return np.interp(x_new, x_old, vec)

    # ----- initialise best candidate ---------------------------
    if base_h is not None:
        h_start = _upscale(base_h)
        h_start = _project_onto_capped_simplex(h_start, target_sum)
    else:
        h_start = _random_feasible(rng, n, target_sum)

    best_h = h_start.copy()
    best_val = _compute_c5(best_h, dx)

    # ------------------------------------------------------------
    # Phase 1 – many short GD + occasional SLSQP runs
    # ------------------------------------------------------------
    phase1_budget = budget_s * 0.42
    while time.time() - start < phase1_budget:
        # random start (sometimes a perturbed best)
        if rng.rand() < 0.18:
            h0 = best_h + rng.normal(scale=0.04, size=n)
            h0 = _project_onto_capped_simplex(h0, target_sum)
        else:
            h0 = _random_feasible(rng, n, target_sum)

        h_opt = _gd_optimize(h0, target_sum,
                             max_iter=1200,
                             step0=1.0,
                             decay=0.997)

        # occasional SLSQP polishing
        if rng.rand() < 0.33:
            h_opt = _slsqp_optimize(h_opt, target_sum,
                                    alpha=80.0,
                                    max_iter=1200)

        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

    # ------------------------------------------------------------
    # Phase 2 – tiny stochastic perturbations around the best
    # ------------------------------------------------------------
    phase2_budget = budget_s * 0.48
    while time.time() - start < phase1_budget + phase2_budget:
        cand = best_h + rng.normal(scale=0.02, size=n)
        cand = _project_onto_capped_simplex(cand, target_sum)
        val = _compute_c5(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    # ------------------------------------------------------------
    # Phase 3 – final smooth‑max refinement (if time permits)
    # ------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 6.0:
        # increase smoothing gradually
        for a in (30.0, 60.0, 120.0):
            best_h = _slsqp_optimize(best_h, target_sum,
                                     alpha=a,
                                     max_iter=1200)
            best_val = _compute_c5(best_h, dx)

    # final safe projection & evaluation
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
