def _adjust_sum(x, target):
    x = np.clip(x,0.0,1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    while True:
        if diff > 0:
            # need to add mass, find indices with x < 1
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not np.any(mask):
            break
        indices = np.where(mask)[0]
        per = diff / len(indices)
        # adjust
        add = np.clip(per, 0, 1 - x[indices]) if diff>0 else np.clip(per, -x[indices], 0)
        # Actually we need sign and cap
        if diff > 0:
            cap = 1.0 - x[indices]
            add = np.minimum(per, cap)
            x[indices] += add
            diff = target - x.sum()
        else:
            cap = x[indices]
            sub = np.minimum(-per, cap)
            x[indices] -= sub
            diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    return np.clip(x,0.0,1.0)

import time
import math
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h: np.ndarray) -> float:
    """Return the max overlap integral for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the hyper‑plane sum(x)=target while staying in [0,1].
    Simple iterative redistribution.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        per = diff / idx.size
        if diff > 0:
            cap = 1.0 - x[idx]
            add = np.minimum(per, cap)
            x[idx] += add
        else:
            cap = x[idx]
            sub = np.minimum(-per, cap)
            x[idx] -= sub
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

def _partner_array(h: np.ndarray, shift: int) -> np.ndarray:
    """Return array p where p[i] = h[i+shift] if inside bounds else 0."""
    n = h.size
    p = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            p[: n - shift] = h[shift:]
    else:
        s = -shift
        if s < n:
            p[s:] = h[: n - s]
    return p

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Choose resolution and initialise
    # ----------------------------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n0 = h0.size
        # if the provided resolution is modest, upscale to gain flexibility
        if n0 < 70:
            n = 81
            old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(new_x, old_x, h0)
        else:
            n = n0
    else:
        n = 81
        h0 = rng.random(n)
    target = n / 2.0
    h0 = _adjust_sum(h0, target)

    # ----------------------------------------------------------------------
    # 2. Keep the best solution seen so far
    # ----------------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ----------------------------------------------------------------------
    # 3. Main stochastic search (directed + random moves)
    # ----------------------------------------------------------------------
    iteration = 0
    # time left for the main loop (keep a small margin for final polishing)
    while time.time() - start < budget_s - 6.0:

        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(corr.argmax())

                    bounds=bounds,
                    constraints=[eq_cons],
                    options={"maxiter": 200, "ftol": 1e-9, "disp": False},
                )
                if res.success:
                    cand = _adjust_sum(np.clip(res.x, 0.0, 1.0), target)
                    cand_c5 = _evaluate_c5(cand)
                    if cand_c5 + 1e-12 < best_c5:
                        best_h, best_c5 = cand, cand_c5
                        cur_h, cur_c5 = cand, cand_c5
            except Exception:
                pass

        iteration += 1

    # ----------------------------------------------------------------------
    # 4. Final intensive local optimisation from the best found point
    # ----------------------------------------------------------------------
    def obj_final(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * (2.0 / n)

    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj_final,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 600, "ftol": 1e-12, "disp": False},
        )
        if res.success:
            best_h = _adjust_sum(np.clip(res.x, 0.0, 1.0), target)
    except Exception:
        pass

    best_c5 = float(_evaluate_c5(best_h))

    # ensure final feasibility (tiny numerical drift)
    best_h = _adjust_sum(np.clip(best_h, 0.0, 1.0), target)

    return best_h, best_c5, n
