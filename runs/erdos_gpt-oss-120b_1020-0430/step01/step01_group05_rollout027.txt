def objective_smooth_fft(h, dx, alpha):
    n = len(h)
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[:2*n-1]
    # smooth max
    m = np.max(corr)
    shifted = corr - m
    return (m + np.log(np.exp(alpha * shifted).sum()) / alpha) * dx

import time, random, math
import numpy as np
from scipy.optimize import minimize

# global target sum variable for constraints
TARGET_SUM_GLOBAL = None

def project_onto_bounds_and_sum(x, target):
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[:2 * n - 1].real
    return np.max(corr) * dx

def smooth_max(arr, alpha):
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def objective_smooth(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha) * dx

def sum_eq_constraint(x):
    return np.sum(x) - TARGET_SUM_GLOBAL

def make_triangle(n):
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)

def make_sin2(n):
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def upsample_initial(init, n):
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)

    global TARGET_SUM_GLOBAL
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    n = 500   # number of sample points (must be even)
    dx = 2.0 / n
    target_sum = n / 2.0
    TARGET_SUM_GLOBAL = target_sum

    # ------------------------------------------------------------------
    # Build initial seeds

    if 'initial_h_values' in globals():
        try:
            seed_vectors.append(upsample_initial(np.array(initial_h_values, dtype=float), n))
        except Exception:
            pass
    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))
    # a few random feasible vectors
    for _ in range(2):
        rnd = np.random.rand(n)
        seed_vectors.append(project_onto_bounds_and_sum(rnd, target_sum))

    # ------------------------------------------------------------------
    # Phase 1: SLSQP optimisation on each seed
    best_h = None
    best_val = np.inf
    bounds = [(0.0, 1.0)] * n
    alpha_schedule = [10.0, 30.0, 60.0]   # increasing smoothness

    # Use about half of the budget for this phase
    phase1_end = start_time + budget_s * 0.5

    while seed_vectors and time.time() < phase1_end:
        h0 = seed_vectors.pop(0)
        cur = h0.copy()
        for alpha in alpha_schedule:
            if time.time() >= phase1_end:
                break
            try:
                res = minimize(
                    objective_smooth,
                    cur,
                    args=(dx, alpha),
                    method='SLSQP',

                    constraints={'type': 'eq', 'fun': sum_eq_constraint},

                )
            except Exception:
                break
            cur = project_onto_bounds_and_sum(res.x, target_sum)
        val = compute_c5(cur)
        if val < best_val:
            best_val = val
            best_h = cur.copy()
        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # If nothing found, start from a random feasible vector
    if best_h is None:
        rnd = np.random.rand(n)
        best_h = project_onto_bounds_and_sum(rnd, target_sum)
        best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 2: Simulated annealing on continuous variables
    cur_h = best_h.copy()
    cur_val = best_val

    # allocate remaining time (except a small tail for final tweaks)
    phase2_end = start_time + budget_s * 0.9
    while time.time() < phase2_end:

        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        # feasible delta that keeps both variables inside [0,1]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        # linearly decreasing temperature
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        T = max(1e-6, 0.02 * (remaining / budget_s))

        delta = np.random.uniform(delta_min, delta_max)
        cand = cur_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cand_val = compute_c5(cand)

        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / T):
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Phase 3: Simple pairwise deterministic search
    phase3_end = start_time + budget_s - 0.5
    while time.time() < phase3_end:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cval = compute_c5(cand)
        if cval < best_val:
            best_val = cval
            best_h = cand.copy()
        if best_val <= 0.3808:
            break

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))

import time, random, math
import numpy as np
from scipy.optimize import minimize

# Global variable for the equality constraint
TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n with Σx = target by bisection on a Lagrange multiplier."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """
    Exact objective (linear cross‑correlation) using FFT.
    Returns max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid.
    """
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[: 2 * n - 1].real
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Smooth max via log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float) -> float:
    """Smooth surrogate of the max used by SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape (peak 1 at centre, linear to 0)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin²(πx/2) on [0,2] (already integrates to 1)."""
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an initial vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low Erdős‑overlap constant C₅."""
    global TARGET_SUM_GLOBAL
    np.random.seed(seed)
    random.seed(seed)

    start_total = time.time()

    # --------------------------------------------------------------
    # Discretisation
    n = 500                     # even, gives finer resolution than 250
    dx = 2.0 / n
    target_sum = n / 2.0
    TARGET_SUM_GLOBAL = target_sum

    # --------------------------------------------------------------
    # Seed vectors (deterministic + a few random ones)
    seed_vectors = []
    if "initial_h_values" in globals():
        try:
            seed_vectors.append(
                upsample_initial(np.array(initial_h_values, dtype=float), n)
            )
        except Exception:
            pass
    seed_vectors.append(make_triangle(n))
    seed_vectors.append(make_sin2(n))

    # --------------------------------------------------------------
    # Phase 1 – many SLSQP runs with increasing smoothness
    best_h = None
    best_val = np.inf
    bounds = [(0.0, 1.0)] * n
    alphas = [10.0, 30.0, 60.0]                     # smoother → tighter max
    phase1_deadline = start_total + budget_s * 0.5

    while time.time() < phase1_deadline and best_val > 0.3808:
        if seed_vectors:
            h0 = seed_vectors.pop(0)
        else:
            rnd = np.random.rand(n)
            h0 = project_onto_bounds_and_sum(rnd, target_sum)

        cur = h0
        for a in alphas:
            if time.time() >= phase1_deadline or best_val <= 0.3808:
                break
            try:
                res = minimize(
                    objective_smooth,
                    cur,
                    args=(dx, a),
                    method="SLSQP",
                    bounds=bounds,
                    constraints={"type": "eq", "fun": sum_eq_constraint},
                    options={"maxiter": 500, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                break
            cur = project_onto_bounds_and_sum(res.x, target_sum)

        val = compute_c5(cur)
        if val < best_val:
            best_val = val
            best_h = cur.copy()
        if best_val <= 0.3808:
            break

    # --------------------------------------------------------------
    # Fallback if no feasible start succeeded
    if best_h is None:
        rnd = np.random.rand(n)
        best_h = project_onto_bounds_and_sum(rnd, target_sum)
        best_val = compute_c5(best_h)

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing‑style continuous refinement
    cur_h = best_h.copy()
    cur_val = best_val
    phase2_deadline = start_total + budget_s * 0.9

    while time.time() < phase2_deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        # feasible delta (keeps both entries in [0,1])
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue

        # simple linear cooling schedule
        elapsed = time.time() - start_total
        remain = budget_s - elapsed
        T = max(1e-6, 0.02 * (remain / budget_s))

        delta = np.random.uniform(delta_min, delta_max)
        cand = cur_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cand_val = compute_c5(cand)

        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / T):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_h = cand_val, cand.copy()
        if best_val <= 0.3808:
            break

    # --------------------------------------------------------------
    # Phase 3 – deterministic pairwise fine‑tuning
    phase3_deadline = start_total + budget_s - 0.5
    while time.time() < phase3_deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target_sum)
        cval = compute_c5(cand)
        if cval < best_val:
            best_val, best_h = cval, cand.copy()
        if best_val <= 0.3808:
            break

    # --------------------------------------------------------------
    # Final safety projection & return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
