```
def simulated_annealing(initial_h, rng, time_deadline, max_step_init=0.3):
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # temperature schedule
    T0 = 0.01
    T = T0
    iteration = 0
    while time.time() < time_deadline:
        # adjust step size gradually
        max_step = max_step_init * (1 - (time.time() - start) / (time_deadline - start))
        h_new = neighbor_move(h, rng, max_step)
        c5_new = compute_c5(h_new)
        delta = c5_new - best_c5
        # If better than best, accept; also if better than current (c5_current). 
        # Use SA accept.
        # For simplicity: if c5_new < best_c5: accept and update best; also set h = h_new.
        # else accept with probability exp(-delta / T)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            h = h_new
        else:
            # accept with probability:
            if T > 1e-9 and rng.random() < np.exp(-delta / T):
                h = h_new
        # update temperature: linear cooling
        T = T0 * (1 - (time.time() - start) / (time_deadline - start))
        iteration += 1
    return best_h, best_c5
```

```
def neighbor_move(h, rng, max_step):
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # direction: sign of delta: we can decide to move from j to i or i to j.
    # compute max possible positive delta (move from j to i):
    max_pos = min(max_step, 1.0 - h[i], h[j])
    # compute max possible negative delta (move from i to j):
    max_neg = min(max_step, h[i], 1.0 - h[j])
    # if both zero, return original (no move)
    if max_pos <= 0 and max_neg <= 0:
        return h
    # sample delta:
    if max_pos > 0 and max_neg > 0:
        # choose direction random
        if rng.random() < 0.5:
            delta = rng.random() * max_pos
        else:
            delta = -rng.random() * max_neg
    elif max_pos > 0:
        delta = rng.random() * max_pos
    else:
        delta = -rng.random() * max_neg
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Ensure clipping within [0,1] due to rounding errors
    new_h[i] = min(1.0, max(0.0, new_h[i]))
    new_h[j] = min(1.0, max(0.0, new_h[j]))
    return new_h
```

```
def simulated_annealing(start_h, rng, deadline, init_step=0.3, init_temp=0.02):
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start_time = time.time()
    # linear schedule for temperature and step size decreasing over time
    while time.time() < deadline:
        remaining = deadline - time.time()
        # compute progress fraction
        progress = (time.time() - start_time) / (deadline - start_time)
        max_step = init_step * (1 - progress)
        temp = init_temp * (1 - progress)
        # propose neighbor
        h_new = neighbor_move(h, rng, max_step)
        new_c5 = compute_c5(h_new)
        # accept if better or with probability
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            # update best
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            # accept with probability exp(-(new - cur) / temp) if temp>0
            if temp > 0 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
                h = h_new
                cur_c5 = new_c5
        # end loop
    return best_h, best_c5
```

import time
import numpy as np

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    m = 2 * n - 1
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * G, m)
    return float(np.max(conv) * (2.0 / n))


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalized."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible vector with sum = n/2 and entries in [0,1]."""
    a = rng.random(n)
    target = n / 2.0
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, preserving the integral."""
    old_n = arr.shape[0]
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up

# Simple feasible neighbour move
# ------------------------------------------------------------
def neighbor_move(h: np.ndarray, rng: np.random.Generator, max_step: float) -> np.ndarray:
    """Transfer a small amount of mass between two distinct coordinates."""
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1  # guarantee i != j

    # Max possible transfer from j to i (positive delta)
    max_pos = min(max_step, 1.0 - h[i], h[j])
    # Max possible transfer from i to j (negative delta)
    max_neg = min(max_step, h[i], 1.0 - h[j])

    if max_pos <= 0 and max_neg <= 0:
        return h

    if max_pos > 0 and max_neg > 0:
        if rng.random() < 0.5:
            delta = rng.random() * max_pos
        else:
            delta = -rng.random() * max_neg
    elif max_pos > 0:
        delta = rng.random() * max_pos
    else:
        delta = -rng.random() * max_neg

    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Clip for numerical safety
    new_h[i] = min(1.0, max(0.0, new_h[i]))
    new_h[j] = min(1.0, max(0.0, new_h[j]))
    return new_h

# Simulated annealing on the exact C5 objective
# ------------------------------------------------------------
def simulated_annealing(start_h: np.ndarray, rng: np.random.Generator,
                        deadline: float, init_step: float = 0.30,
                        init_temp: float = 0.020) -> tuple:
    """
    Simple SA that respects the sum constraint via pairwise transfers.
    Returns (best_h, best_c5) discovered before ``deadline``.
    """
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    total_span = max(deadline - start_time, 1e-12)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_span
        max_step = init_step * (1.0 - progress)
        temp = init_temp * (1.0 - progress)

        h_new = neighbor_move(h, rng, max_step)
        new_c5 = compute_c5(h_new)

        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            if temp > 0 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
                h = h_new
                cur_c5 = new_c5

    return best_h, best_c5
