import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def gradient_for_lag(h, lag):
    n = h.size
    if lag >= 0:
        # term1
        term1 = np.empty(n)
        term1[:n-lag] = 1 - h[lag:]
        term1[n-lag:] = 1.0
        term2 = np.empty(n)
        term2[:lag] = 0.0
        term2[lag:] = h[:n-lag]
    else:
        lag = -lag
        term1 = np.empty(n)
        term1[lag:] = 1 - h[:n-lag]
        term1[:lag] = 1.0
        term2 = np.empty(n)
        term2[:n-lag] = h[lag:]
        term2[n-lag:] = 0.0
    return term1 - term2

def enforce_sum(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    # simple binary random then adjust
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    # allow fractional tweaks
    # random small transfers
    for _ in range(n*3):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n/2.0, rng)
    np.clip(h, 0.0, 1.0, out=h)
    return h

    i = rng.integers(0, n)

    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):

    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/n)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def gradient_for_lag(h, lag):
    """Gradient of the overlap sum for a fixed lag."""
    n = h.size
    if lag >= 0:
        term1 = np.empty(n)
        term1[: n - lag] = 1 - h[lag:]
        term1[n - lag :] = 1.0
        term2 = np.empty(n)
        term2[:lag] = 0.0
        term2[lag:] = h[: n - lag]
    else:
        lag = -lag
        term1 = np.empty(n)
        term1[lag:] = 1 - h[: n - lag]
        term1[:lag] = 1.0
        term2 = np.empty(n)
        term2[: n - lag] = h[lag:]
        term2[n - lag :] = 0.0
    return term1 - term2


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector (0≤h_i≤1, Σh_i=n/2)."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    for _ in range(n * 3):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    enforce_sum(h, n / 2.0, rng)
    np.clip(h, 0.0, 1.0, out=h)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """Refine h using a smooth surrogate of the max‑overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


def targeted_grad_descent(h, rng, max_iters, time_limit):
    """Move mass from indices that increase the worst lag to those that decrease it."""
    n = h.size
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_corr = corr[max_idx]
        cur_c = cur_corr * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        lag = max_idx - (n - 1)
        d = gradient_for_lag(h, lag)
        pos = np.where(d > 1e-8)[0]
        neg = np.where(d < -1e-8)[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_eps = min(h[i], 1 - h[j])
            if max_eps > 0:
                factor = 0.2 * (1 - it / max_iters) + 0.01
                eps = max_eps * factor
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                new_c = compute_c5(new_h)
                if new_c < cur_c:
                    h = new_h
                else:
                    temp = 0.001 * (1 - it / max_iters) + 0.0001
                    if rng.random() < np.exp(-(new_c - cur_c) / temp):
                        h = new_h
        else:
            # occasional random transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(h[i], 1 - h[j])
                if max_eps > 0:
                    eps = max_eps * 0.05 * (rng.random() * 2 - 1)
                    eps = max(-max_eps, min(eps, max_eps))
                    new_h = h.copy()
                    new_h[i] -= eps
                    new_h[j] += eps
                    new_c = compute_c5(new_h)
                    if new_c < cur_c:
                        h = new_h
    return best_h, best_c
