    start = time.time()
    n = h.shape[0]
    dx = 2.0 / n
    best = h.copy()

        cand[j] = hj - delta

import time, random, math
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ---------------------------------------------------------

TARGET_SUM_GLOBAL = None  # used by the equality constraint


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the box [0,1]^n intersected with the hyperplane sum(x)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Discrete evaluation of the Erdös overlap constant."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Log‑sum‑exp smooth approximation of max."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth version of the max‑overlap objective for SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular bump peaked at the centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    """h(x)=sin²(πx/2) on [0,2] – already integrates to 1."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation from len(init) to n points, then project onto constraints."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)


def local_refine(h: np.ndarray, target_sum: float, time_limit: float) -> np.ndarray:
    """Greedy pairwise adjustments while time permits."""
    start = time.time()
    n = h.shape[0]
    cur = h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]
        # delta keeps the total sum unchanged and respects bounds
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = cur.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        # sum is preserved, bounds are satisfied by construction
        cand_val = compute_c5(cand)
        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val
        # early exit if already excellent
        if best_val <= 0.3808:
            break
    return best


# ---------- main routine ---------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low Erdős C₅ bound."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    budget = budget_s

    # final resolution
    n_target = 250
    target_sum_target = n_target / 2.0

    best_h = None
    best_val = np.inf

    # ----- use parent construction if supplied --------------------------------
    if "initial_h_values" in globals():
        try:
            init = np.array(initial_h_values, dtype=float)
            if init.shape[0] != n_target:
                init = upsample_initial(init, n_target)
            else:
                init = project_onto_bounds_and_sum(init, target_sum_target)
            val = compute_c5(init)
            if val < best_val:
                best_val, best_h = val, init.copy()
        except Exception:
            pass

    # ----- coarse optimisation (small n) ------------------------------------
    n_coarse = 80
    target_sum_coarse = n_coarse / 2.0
    coarse_seeds = [make_triangle(n_coarse), make_sin2(n_coarse)]
    for _ in range(3):
        rnd = np.random.rand(n_coarse)
        coarse_seeds.append(project_onto_bounds_and_sum(rnd, target_sum_coarse))

    global TARGET_SUM_GLOBAL
    for seed_vec in coarse_seeds:
        if time.time() - start_time > budget - 5.0:
            break
        h = seed_vec.copy()
        TARGET_SUM_GLOBAL = target_sum_coarse
        # continuation on the smoothing parameter
        for alpha in (5.0, 10.0, 20.0, 40.0):
            res = minimize(
                objective_smooth,
                h,
                args=(2.0 / n_coarse, alpha),
                method="SLSQP",
                bounds=[(0.0, 1.0)] * n_coarse,
                constraints={"type": "eq", "fun": sum_eq_constraint},
                options={"maxiter": 400, "ftol": 1e-9, "disp": False},
            )
            h = project_onto_bounds_and_sum(res.x, target_sum_coarse)
        # upsample to target size
        h_up = upsample_initial(h, n_target)
        # local continuous refinement at full size
        remaining = budget - (time.time() - start_time)
        if remaining <= 0:
            break
        refined = local_refine(h_up, target_sum_target, time_limit=remaining * 0.6)
        val = compute_c5(refined)
        if val < best_val:
            best_val, best_h = val, refined.copy()
            if best_val <= 0.3808:
                break

    # ----- if still not good, run pure SA at full size -----------------------
    if best_val > 0.3808:
        # start from a simple smooth shape if nothing better was found
        if best_h is None:
            best_h = make_sin2(n_target)
        cur = best_h.copy()
        cur_val = compute_c5(cur)
        best = cur.copy()
        best_val = cur_val
        # temperature schedule (linear decay)
        T0 = 0.02  # typical magnitude of objective fluctuations
        it = 0
        while time.time() - start_time < budget - 1.0:
            elapsed = time.time() - start_time
            frac = elapsed / budget
            T = max(T0 * (1.0 - frac), 1e-5)
            i, j = np.random.randint(0, n_target, size=2)
            if i == j:
                continue
            hi, hj = cur[i], cur[j]
            delta_max = min(1.0 - hi, hj)
            delta_min = -min(hi, 1.0 - hj)
            if delta_max < delta_min:
                continue
            delta = np.random.uniform(delta_min, delta_max)
            if abs(delta) < 1e-12:
                continue
            cand = cur.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            cand_val = compute_c5(cand)
            if cand_val < cur_val or random.random() < math.exp(-(cand_val - cur_val) / T):
                cur, cur_val = cand, cand_val
                if cand_val < best_val:
                    best, best_val = cand.copy(), cand_val
            # occasional greedy pass to cleanse lingering peaks
            if it % 5000 == 0 and time.time() - start_time < budget - 2.0:
                best = local_refine(best, target_sum_target, time_limit=2.0)
                best_val = compute_c5(best)
            it += 1
            if best_val <= 0.3808:
                break
        best_h, best_val = best, best_val

    # final projection & sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum_target)
    final_val = compute_c5(best_h)

    return (best_h.tolist(), float(final_val), int(n_target))
