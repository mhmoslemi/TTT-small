def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, random, math
    from scipy.optimize import minimize

    # set seed
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    margin = 0.5

    n = 300  # even
    dx = 2.0 / n
    target_sum = n / 2.0
    global TARGET_SUM_GLOBAL
    TARGET_SUM_GLOBAL = target_sum

    # functions defined earlier

    # initial seeds list
    seeds = []

    # upsample parent if available
    if "initial_h_values" in globals():
        # upsample to n
        try:
            parent = upsample_initial(np.array(initial_h_values, dtype=float), n)
            seeds.append(parent)
            # binary version
            bin_parent = binarize(parent, target_sum)
            seeds.append(bin_parent.astype(float))
        except Exception:
            pass

    # deterministic continuous shapes
    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rect(n))

    # deterministic binary shapes
    seeds.append(binarize(make_triangle(n), target_sum).astype(float))
    seeds.append(binarize(make_sin2(n), target_sum).astype(float))
    seeds.append(binarize(make_rect(n), target_sum).astype(float))
    # alternating pattern
    alt = np.zeros(n)
    alt[::2][:target_sum] = 1  # fill half with 1 in alternating pattern
    # adjust to have exactly target_sum ones
    # produce alt correctly
    alt = np.zeros(n)
    alt_idx = np.arange(n)
    alt_one_idx = alt_idx[alt_idx % 2 == 0][:int(target_sum)]
    alt[alt_one_idx] = 1
    # if not enough, fill others
    if alt.sum() < target_sum:
        extra_needed = int(target_sum - alt.sum())
        remaining = np.setdiff1d(np.arange(n), alt_one_idx)
        alt[remaining[:extra_needed]] = 1
    seeds.append(alt.astype(float))

    # also random binary seeds: e.g., 5

        rnd = np.random.rand(n)
        binary = binarize(rnd, target_sum).astype(float)
        seeds.append(binary)

    # function for stagewise optimization
    def optimize_continuous(h0, time_left):
        # perform multi-stage SLSQP
        alphas = [10.0, 50.0, 200.0, 500.0]
        h = h0.copy()
        for alpha in alphas:
            # enforce time check
            if time.time() - start >= budget_s - margin:

                return objective_smooth(x, dx, alpha)

                    bounds=[(0.0, 1.0)] * n,
                    constraints={"type": "eq", "fun": sum_eq_constraint},
                    options={"maxiter": 300, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                break
            h = project_onto_bounds_and_sum(res.x, target_sum)
        return h

GLOBAL_DX = None              # step size = 2/n

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect sum(x)=target via bisection on λ."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact (discrete) C5 value for vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * GLOBAL_DX


def smooth_max(arr: np.ndarray, alpha: float) -> float:
    """Smooth approximation to max using log‑sum‑exp."""
    m = np.max(arr)
    return m + np.log(np.sum(np.exp(alpha * (arr - m)))) / alpha


def objective_smooth(x: np.ndarray, alpha: float) -> float:
    """Smooth objective used by SLSQP."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, alpha) * GLOBAL_DX


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Continuous triangle, peak at centre."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:

    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)

def hill_climb_binary(start_h: np.ndarray,
                      max_swaps: int = 5000,
                      max_no_improve: int = 2000,
                      time_limit: float = None) -> tuple:
    """
    Simple hill climbing on a binary vector.
    Returns (best_vector, best_value).
    """
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    no_improve = 0
    t0 = time.time()
    while True:
        if max_swaps is not None and max_swaps <= 0:
            break
        if max_no_improve is not None and no_improve >= max_no_improve:
            break
        if time_limit is not None and (time.time() - t0) >= time_limit:
            break
        # pick a random 1 and a random 0
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        val = compute_c5(h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
            no_improve = 0
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        if max_swaps is not None:
            max_swaps -= 1
        if best_val <= 0.3808:
            break
    return best_h, best_val


def optimize_continuous(seed_h: np.ndarray,
                        time_remaining: float) -> np.ndarray:
    """Stage‑wise SLSQP with increasing smoothing."""
    alphas = [10.0, 50.0, 200.0, 500.0]
    h = seed_h.copy()
    for alpha in alphas:
        if time.time() - run_start >= budget - 0.5:
            break
        def obj(x, a=alpha):
            return objective_smooth(x, a)
        try:
            res = minimize(
                obj,
                h,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * h.shape[0],
                constraints={'type': 'eq', 'fun': sum_eq_constraint},
                options={'maxiter': 300, 'ftol': 1e-9, 'disp': False},
            )
        except Exception:
            break
        h = project_onto_bounds_and_sum(res.x, TARGET_SUM_GLOBAL)
    return h

    n = 300                     # even for half‑mass integer
    GLOBAL_DX = 2.0 / n
    TARGET_SUM_GLOBAL = n / 2.0          # float target for continuous variables
    TARGET_SUM_INT = n // 2               # integer number of ones for binary vectors

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Build initial seeds
    seeds = []

    # parent if available
    if "initial_h_values" in globals():
        try:
            parent = upsample_initial(np.array(initial_h_values, dtype=float), n)
            seeds.append(parent)
            # binary version of parent
            seeds.append(binarize(parent, TARGET_SUM_INT).astype(float))
        except Exception:
            pass

    # deterministic continuous shapes
    seeds.append(make_triangle(n))
    seeds.append(make_sin2(n))
    seeds.append(make_rect(n))

    # deterministic binary shapes
    seeds.append(binarize(make_triangle(n), TARGET_SUM_INT).astype(float))
    seeds.append(binarize(make_sin2(n), TARGET_SUM_INT).astype(float))
    seeds.append(binarize(make_rect(n), TARGET_SUM_INT).astype(float))

    # alternating pattern (1010…)
    alt = np.zeros(n, dtype=float)
    ones_idx = np.arange(n)[np.arange(n) % 2 == 0][:TARGET_SUM_INT]
    alt[ones_idx] = 1.0
    # fill any missing ones if needed
    if alt.sum() < TARGET_SUM_INT:
        missing = TARGET_SUM_INT - int(alt.sum())
        remaining = np.setdiff1d(np.arange(n), ones_idx)[:missing]
        alt[remaining] = 1.0
    seeds.append(alt)

    # a few random binary seeds
    for _ in range(3):
        rnd = np.random.rand(n)
        seeds.append(binarize(rnd, TARGET_SUM_INT).astype(float))

    # ------------------------------------------------------------------
    # Process each seed: continuous optimisation + binary hill climb
    for seed_vec in seeds:
        if time.time() - run_start >= budget - margin:
            break

        # If the seed is already binary, skip continuous stage
        if np.all((seed_vec == 0) | (seed_vec == 1)):
            h_cont = seed_vec.copy()
        else:
            h_cont = optimize_continuous(seed_vec, budget - (time.time() - run_start))

        # Evaluate continuous solution
        val_cont = compute_c5(h_cont)
        if val_cont < best_val:
            best_val = val_cont
            best_h = h_cont.copy()
            if best_val <= 0.3808:
                break

        # Binary refinement starting from thresholded continuous solution
        bin_start = binarize(h_cont, TARGET_SUM_INT).astype(float)
        time_left = budget - (time.time() - run_start) - 0.2
        bin_h, bin_val = hill_climb_binary(
            bin_start,
            max_swaps=3000,
            max_no_improve=1500,
            time_limit=time_left,
        )
        if bin_val < best_val:
            best_val = bin_val
            best_h = bin_h.copy()
            if best_val <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Additional pure binary random searches until time runs out
    while time.time() - run_start < budget - margin:
        # generate a fresh random binary vector
        rand_bin = np.zeros(n, dtype=float)
        rand_bin[np.random.choice(n, TARGET_SUM_INT, replace=False)] = 1.0
        time_left = budget - (time.time() - run_start) - 0.1
        bin_h, bin_val = hill_climb_binary(
            rand_bin,
            max_swaps=5000,
            max_no_improve=2500,
            time_limit=time_left,
        )
        if bin_val < best_val:
            best_val = bin_val
            best_h = bin_h.copy()
            if best_val <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Final safety projection & exact evaluation
    if best_h is None:
        best_h = make_triangle(n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, TARGET_SUM_GLOBAL)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
