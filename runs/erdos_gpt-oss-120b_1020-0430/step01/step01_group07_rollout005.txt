def project_capped_simplex(v, target):
    lo = np.min(v) - 1
    hi = np.max(v) + 1
    for _ in range(50):
        lam = (lo+hi)/2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

import time
import random
import numpy as np

def _c5_metric(h):
    """C5 value for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_to_capped_simplex(v, target):
    """Project v onto {0≤x_i≤1, sum x_i = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _subgradient_descent(start_h, time_limit, step0=0.5, max_iter=20000):
    """Projected sub‑gradient descent on max_k ∫h(1‑h_shift)."""
    n = start_h.shape[0]
    target = n / 2.0
    cur_h = start_h.copy()
    best_h = cur_h.copy()
    best_c = _c5_metric(cur_h)
    start = time.time()

    for it in range(max_iter):
        if time.time() - start >= time_limit:
            break

        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx = int(np.argmax(corr))
        k = abs(idx - (n - 1))

        # move = -grad = sum of neighbours at distance k
        if k == 0:
            move = 2.0 * cur_h
        else:
            move = np.zeros_like(cur_h)
            move[: n - k] += cur_h[k:]
            move[k:] += cur_h[: n - k]

        # simple linear schedule, never zero
        progress = (it + 1) / max_iter
        step = step0 * (1.0 - progress) + 1e-5

        # tiny random jitter each few steps
        if it % 500 == 0:
            jitter = (np.random.rand(n) - 0.5) * (1.0 - cur_h) * cur_h
            move += 0.1 * step * jitter

        new_h = cur_h + step * move
        new_h = _project_to_capped_simplex(new_h, target)
        new_c = _c5_metric(new_h)

        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        cur_h = new_h

    return best_h, best_c


def _hill_climb(start_h, iterations):
    """Small random exchanges that only keep improvements."""
    n = start_h.shape[0]
    target = n / 2.0
    cur_h = start_h.copy()
    cur_c = _c5_metric(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    for _ in range(iterations):
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = random.uniform(low, high) * 0.05
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_c = _c5_metric(cur_h)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = hi
            cur_h[j] = hj
    return best_h, best_c
