def compute_c5(h):
    n = len(h)

    size = 2*n

    best_h = h.copy(); best_val = compute_c5(h)
    cur_h = h.copy(); cur_val = best_val
    t = start_temp

        i = np.random.randint(n)
        if cur_h[i] == 1:
            # need a zero j
            # choose random zero j
            zeros = np.where(cur_h == 0)[0]; j = zeros[np.random.randint(len(zeros))]
        else:
            # h[i] ==0; choose a one j
            ones = np.where(cur_h == 1)[0]; j = ones[np.random.randint(len(ones))]
        # swap i and j
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = compute_c5(cur_h)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / t):
            cur_val = new_val
            if new_val < best_val:

            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        t *= decay
        # optional early break if best below target
        if best_val < target_bound:
            break
    return best_h, best_val

import time
import numpy as np

# Optional SciPy import (used only for a final refinement step)
try:
    from scipy import optimize
except Exception:  # pragma: no cover
    optimize = None


def _c5_fft(h: np.ndarray) -> float:
    """Fast C5 evaluation using FFT (linear convolution of h with 1-h)."""
    n = h.shape[0]
    size = 2 * n
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    conv = np.fft.irfft(H * G, n=size)
    max_corr = conv.max()
    return max_corr * (2.0 / n)


def _binary_from_real(v: np.ndarray) -> np.ndarray:
    """Convert a real vector to a binary one with exactly n/2 ones."""
    n = v.size
    idx = np.argsort(-v)          # descending order
    h = np.zeros_like(v, dtype=float)
    h[idx[: n // 2]] = 1.0
    return h


def _hill_climb_binary_timed(
    h0: np.ndarray,
    max_iter: int,
    start_temp: float,
    decay: float,
    target: float,
    time_left,
):
    """Simulated‑annealing style hill‑climb on a binary vector."""
    n = h0.size
    cur = h0.astype(float).copy()
    cur_val = _c5_fft(cur)
    best = cur.copy()
    best_val = cur_val
    t = start_temp

    for it in range(max_iter):
        if time_left() < 0.5:
            break

        # propose a swap of a 1 and a 0
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:
            continue

        cur[i], cur[j] = cur[j], cur[i]
        new_val = _c5_fft(cur)
        delta = new_val - cur_val

        accept = False
        if delta < 0:
            accept = True
        else:
            if t > 0.0 and np.random.rand() < np.exp(-delta / t):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val, best = new_val, cur.copy()
                if best_val < target:
                    break
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

        t = max(t * decay, 0.0)
    return best, best_val


def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9) -> np.ndarray:
    """Clip to [0,1] and enforce the exact sum = n/2 (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """C5 using the same FFT routine (for consistency with optimisation)."""
    return _c5_fft(h)


def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    """t - max_corr >= 0."""
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    # max correlation of h with (1-h)
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return t - max_corr


def _eq_constraint(x: np.ndarray) -> float:
    """sum(h) = n/2."""
    n = x.size - 1
    return np.sum(x[:n]) - (n * 0.5)


def _solve_h(initial_h: np.ndarray, max_iter: int = 200) -> tuple:
    """One SLSQP refinement pass (if SciPy is available)."""
    if optimize is None:  # pragma: no cover
        return initial_h, _c5_from_h(initial_h)

    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)
