def project_onto_sum_box(v, target, tol=1e-9, max_iter=60):
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        tau = (low + high) / 2.0
        u = np.clip(v - tau, 0.0, 1.0)
        s = u.sum()
        if abs(s - target) <= tol:
            return u
        if s > target:
            low = tau
        else:
            high = tau
    return np.clip(v - ((low + high) / 2.0), 0.0, 1.0)

import numpy as np
import random
import math
import time

def _project_sum_box(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto the set {0<=x<=1, sum(x)=target} (Euclidean projection)."""
    # The projection is x = clip(v - tau, 0, 1) with tau chosen so that sum(x)=target.
    low = -np.max(v) - 10.0   # ensures sum > target
    high = np.max(v) + 10.0   # ensures sum < target
    for _ in range(max_iter):
        tau = (low + high) / 2.0
        u = np.clip(v - tau, 0.0, 1.0)
        s = u.sum()
        if abs(s - target) <= tol:
            return u
        if s > target:
            low = tau
        else:
            high = tau
    # final approximation
    tau = (low + high) / 2.0
    return np.clip(v - tau, 0.0, 1.0)


def _c5_of(v: np.ndarray, dx: float) -> float:
    """C5 value for a given discretised function v."""
    return float(np.correlate(v, 1.0 - v, mode="full").max() * dx)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of O_shift = sum h*(1-h_shift) with respect to h."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            g[: n - s] += 1.0 - h[s:]
            g[s:] -= h[: n - s]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[: n - s]
            g[: n - s] -= h[s:]
    return g
