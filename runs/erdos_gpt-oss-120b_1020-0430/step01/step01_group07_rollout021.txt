def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    # Compute convolution of h and reversed h: size 2n-1
    Nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, Nfft)
    Hr = np.fft.rfft(h[::-1], Nfft)
    conv = np.fft.irfft(H * Hr, Nfft)[:2*n-1]
    # product array = conv
    # Determine the minimum product (since correlation includes shift from -(n-1) to (n-1))
    # For each shift k (0..n-1) product length has same values? We need to map shift index to correct range.
    # In np.correlate full mode: shift index 0 corresponds to -(n-1) shift, i.e., k = - (n-1)
    # But min product irrespective of sign is same anyway.
    product_min = conv.min()
    # C5 = 1 - 2 * product_min / n
    c5 = 1 - (2 * product_min) / n
    return c5

import time
import random
import numpy as np

def _project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto the set {x | sum(x)=target, 0<=x_i<=1}."""
    # binary search on the shift λ such that sum(clip(v-λ,0,1)) = target
    lo = - (np.max(v) + 1.0)   # makes all entries >1 after shift → clipped to 1
    hi = np.max(v) + 1.0       # makes all entries <0 after shift → clipped to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo + hi) * 0.5), 0.0, 1.0)


def _c5_fft(h: np.ndarray) -> float:
    """Fast C5 evaluation using FFT (returns the bound, not the min product)."""
    n = h.shape[0]
    sz = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1
    H = np.fft.rfft(h, sz)
    Hr = np.fft.rfft(h[::-1], sz)
    conv = np.fft.irfft(H * Hr, sz)[: 2 * n - 1]
    prod_min = conv.min()
    # Numerical noise may produce tiny negatives; clamp.
    if prod_min < 0.0:
        prod_min = 0.0
    return 1.0 - 2.0 * prod_min / n


def _c5_full_corr(h: np.ndarray) -> float:
    """Reference implementation using np.correlate (slower)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

    # Build an initial feasible vector, using the provided construction if any.
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x, x0, h0)
        h0 = _project_box_simplex(h0, target_sum)
    else:
        # uniform start then project (produces a half‑density distribution)
        h0 = _project_box_simplex(np.random.rand(n), target_sum)

    # --------------------------------------------------------------------- #
    # Differential Evolution parameters
    pop_size = 40
    F_base = 0.6
    Cr = 0.9

    # Initialise population: first individual is the seed, rest random.
    pop = np.empty((pop_size, n), dtype=np.float64)
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        pop[i] = _project_box_simplex(np.random.rand(n), target_sum)

    # Evaluate initial population
    fitness = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        fitness[i] = _c5_fft(pop[i])

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = float(fitness[best_idx])

    # --------------------------------------------------------------------- #
    # Main DE loop
    while time.time() - start_all < budget_s - 5.0:  # keep a few seconds for polishing
        for i in range(pop_size):
            # distinct indices a,b,c ≠ i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = random.sample(idxs, 3)
            # jittered scaling factor
            Fi = 0.5 + random.random() * 0.5
            mutant = pop[a] + Fi * (pop[b] - pop[c])
            # binomial crossover
            trial = pop[i].copy()
            jrand = random.randrange(n)
            mask = np.random.rand(n) < Cr
            mask[jrand] = True
            trial[mask] = mutant[mask]
            # enforce feasibility
            trial = _project_box_simplex(trial, target_sum)
            # evaluate
            trial_f = _c5_fft(trial)
            if trial_f < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_f:
                    best_f = float(trial_f)
                    best_h = trial.copy()
        # optional early stop if we already beat the target
        if best_f < 0.3808:
            break

    # --------------------------------------------------------------------- #
    # Local refinement: tiny pairwise transfers (hill‑climbing)
    time_left = budget_s - (time.time() - start_all)
    if time_left > 1.0:
        end_refine = time.time() + time_left
        step_scale = 0.08
        while time.time() < end_refine:
            i, j = random.sample(range(n), 2)
            hi, hj = best_h[i], best_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = random.uniform(low, high) * step_scale
            best_h[i] = hi + delta
            best_h[j] = hj - delta
            new_f = _c5_fft(best_h)
            if new_f + 1e-12 < best_f:
                best_f = float(new_f)
            else:
                # revert
                best_h[i] = hi
                best_h[j] = hj

    # --------------------------------------------------------------------- #
    # Final exact‑sum repair (tiny numerical drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        best_h += diff / n
        best_h = _project_box_simplex(best_h, target_sum)

    # Compute the final bound using the provided helper if present
    try:
        final_c = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c = float(_c5_full_corr(best_h))

    return best_h, final_c, int(n)
