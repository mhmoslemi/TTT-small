def random_symmetric(n, rng):
    half = n//2
    a = rng.random(half)
    target_half = n/4.0
    a = a * (target_half / a.sum())
    a = np.clip(a, 0.0, 1.0)
    # if sum not correct, adjust by adding equal small amount to all entries and clip
    diff = target_half - a.sum()
    if abs(diff) > 1e-12:
        a += diff / half
        a = np.clip(a, 0.0, 1.0)
    # symmetrize
    h = np.concatenate([a, a[::-1]])
    return h

def optimize_start(start, n, time_deadline):
    # uses SLSQP
    # enforce eq sum constraint
    target = n/2.0
    cons = {'type':'eq','fun': lambda h: np.sum(h) - target}
    bounds = [(0.0,1.0)]*n
    # Stage 1: soft max with medium alpha
    # Use high maxiter
    res = minimize(lambda h: softmax_c5(h, alpha=10.0), start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter':500, 'ftol':1e-9})
    if not res.success:
        # maybe still works with res.x
        x = res.x
    else:
        x = res.x
    if time.time()> time_deadline: return None, None
    # Stage 2: sharper alpha
    res2 = minimize(lambda h: softmax_c5(h, alpha=50.0), x, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter':500, 'ftol':1e-9})
    if not res2.success:
        h_opt = np.clip(res2.x,0,1)
    else:
        h_opt = np.clip(res2.x,0,1)
    # enforce symmetry again maybe beneficial
    h_opt = (h_opt + h_opt[::-1]) / 2.0
    # renorm sum exactly
    cur = h_opt.sum()
    if abs(cur-target)>1e-12:
        h_opt = h_opt + (target - cur)/n
        h_opt = np.clip(h_opt,0,1)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

import time, numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def softmax_c5(h, alpha=30.0):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(alpha * conv) / alpha)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def random_symmetric(n, rng):
    half = n // 2
    a = rng.random(half)
    target_half = n / 4.0
    a = a * (target_half / a.sum())
    a = np.clip(a, 0.0, 1.0)
    diff = target_half - a.sum()
    if abs(diff) > 1e-12:
        a += diff / half
        a = np.clip(a, 0.0, 1.0)
    h = np.concatenate([a, a[::-1]])
    return h

def optimize_start(start, n, deadline):
    target = n / 2.0
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target}
    bounds = [(0.0, 1.0)] * n

    # Stage 1: moderate alpha
    res = minimize(lambda h: softmax_c5(h, alpha=10.0), start,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 400, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        # Continue anyway with whatever we got
        x = res.x
    else:
        x = res.x

    if time.time() >= deadline:
        return None, None

    # Stage 2: higher alpha
    res2 = minimize(lambda h: softmax_c5(h, alpha=60.0), x,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 400, 'ftol': 1e-9, 'disp': False})
    if not res2.success:
        x2 = res2.x
    else:
        x2 = res2.x

    # Symmetrize to keep shape nice (optional)
    h_opt = symmetrize(x2)
    # project to constraints
    cur = h_opt.sum()
    if abs(cur - target) > 1e-12:
        h_opt = h_opt + (target - cur) / n
    h_opt = np.clip(h_opt, 0.0, 1.0)

    c5 = compute_c5(h_opt)
    return h_opt, c5

def local_refine(h, n, deadline, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = 0.2
    while time.time() < deadline and step > 1e-5:
        # attempt a fixed number of moves per step size
        for _ in range(300):
            if time.time() >= deadline:
                break
            # choose i upward, j downward
            i_candidates = np.where(best_h < 1.0 - 1e-12)[0]
            j_candidates = np.where(best_h > 1e-12)[0]
            if i_candidates.size == 0 or j_candidates.size == 0:
                break
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            max_inc = min(step, 1.0 - best_h[i])
            max_dec = min(step, best_h[j])
            delta = rng.random() * min(max_inc, max_dec)
            if delta < 1e-12:
                continue
            h_new = best_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
        step *= 0.5
    return best_h, best_c5

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * conv) / alpha)


def symmetrize(x: np.ndarray) -> np.ndarray:
    """Force mirror symmetry about the centre of the interval."""
    return (x + x[::-1]) / 2.0


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
