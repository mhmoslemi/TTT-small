def smooth_gradient(h, beta):

    conv = np.correlate(h, 1 - h, mode='full')

import time
import numpy as np
from scipy.optimize import minimize  # keep import in case it's useful


def _evaluate_c5(h):
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _adjust_sum(x, target):
    """Project onto box [0,1] and hyper‑plane sum(x)=target (L2‑projection)."""
    x = np.clip(x, 0.0, 1.0)
    # binary search for lambda so that sum(clip(x-lambda,0,1)) == target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual due to numerical errors
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def _gradient_shift(h, shift):
    """Sub‑gradient of the overlap for a given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        g[shift:] += 1.0 - h[: n - shift]
        g[: n - shift] -= h[shift:]
    else:
        s = -shift
        g[: n - s] += 1.0 - h[s:]
        g[s:] -= h[: n - s]
    return g


def _max_shift_and_val(h):
    """Return shift (index‑offset) giving the largest overlap, its value and the full array."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    shift = idx - (h.shape[0] - 1)
    return shift, max_val, conv


def _subgradient_descent(h0, target, iters, rng):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _evaluate_c5(h)
    step0 = 0.7
    for t in range(iters):
        shift, _, _ = _max_shift_and_val(h)
        g = _gradient_shift(h, shift)
        step = step0 / np.sqrt(t + 1.0)
        h_new = h - step * g
        h_new = _adjust_sum(np.clip(h_new, 0.0, 1.0), target)
        c5_new = _evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        h = h_new
    return best_h, best_c5


def _hill_climb(h, c5, deadline, rng):
    """Simple greedy hill‑climbing using pairwise mass exchanges."""
    n = h.shape[0]
    target = n / 2.0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    h = _adjust_sum(np.clip(h, 0.0, 1.0), target)
    return h, float(_evaluate_c5(h))
