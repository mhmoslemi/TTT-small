def softmax_obj_and_grad(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')
    # objective = log-sum-exp / beta
    b_corr = beta * corr
    max_bc = np.max(b_corr)
    sum_exp = np.exp(b_corr - max_bc).sum()
    obj = (max_bc + np.log(sum_exp)) / beta
    w = np.exp(b_corr - max_bc) / sum_exp  # normalized weights sum to 1
    # compute gradient
    # w corresponds to lag indices from -(N-1) to N-1
    # using w_rev reversed order
    w_rev = w[::-1]
    # term1: conv of w_rev with (1 - h)
    conv1_full = np.convolve(w_rev, 1 - h, mode='full')
    # term2: conv of w with h
    conv2_full = np.convolve(w, h, mode='full')
    # extract central N entries: indices from N-1 to 2N-2 inclusive (size N)
    N = h.size
    start = N - 1
    end = start + N
    term1 = conv1_full[start:end]
    term2 = conv2_full[start:end]
    grad = term1 - term2  # shape N
    return obj, grad

import numpy as np, time

def compute_max_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_idx = np.argmax(corr)
    return max_val, max_idx, corr

def lag_from_idx(idx, n):
    return idx - (n - 1)

def gradient_worst_lag(h, L):
    n = h.size
    # compute term1: (1 - h[i+L]) for valid i.
    # term2: h[i-L]
    # Use zeros for out-of-range.
    if L >= 0:
        term1 = np.empty(n)
        term1[:n - L] = 1 - h[L:]
        term1[n - L:] = 0.0
        term2 = np.empty(n)
        term2[L:] = h[:n - L]

    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = exp_bc.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def proj_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the intersection {0 ≤ x ≤ 1, sum(x)=target}."""
    # binary search for λ such that sum(clip(v-λ,0,1)) = target
    lo = -np.max(v) - 1.0  # makes v-λ large ⇒ all ones
    hi = np.max(v) + 1.0   # makes v-λ small ⇒ all zeros
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
            np.clip(x, 0.0, 1.0, out=x)
    return x


def generate_random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform‑random point satisfying the box‑simplex constraints."""
    v = rng.random(n)
    return proj_box_simplex(v, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small ε from one coordinate to another (keeps the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Soft‑max surrogate (smooth max) and its gradient
# ----------------------------------------------------------------------


def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Log‑sum‑exp surrogate of maxₖ Σᵢ hᵢ(1‑hᵢ₊ₖ) and its gradient."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")          # length 2n‑1
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    Z = exp_bc.sum()
    obj = (m + np.log(Z)) / beta

    w = exp_bc / Z                     # normalized soft‑max weights, sum = 1
    w_rev = w[::-1]

    # convolution gives Σₗ wₗ·(1‑hᵢ₊ₗ)  and Σₗ wₗ·hᵢ₋ₗ respectively
    conv1 = np.convolve(w_rev, 1 - h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    start = n - 1
    grad = conv1[start:start + n] - conv2[start:start + n]
    return obj, grad


# ----------------------------------------------------------------------
# Gradient of a single lag (exact max‑lag)
# ----------------------------------------------------------------------


def gradient_worst_lag(h: np.ndarray, L: int) -> np.ndarray:
    """∂/∂h Σᵢ hᵢ(1‑hᵢ₊ₗ)  for a fixed lag L."""
    n = h.size
    g = np.zeros(n)
    if L >= 0:
        g[: n - L] = 1 - h[L:]          # (1‑hᵢ₊ₗ)
        g[L:] -= h[: n - L]              # –hᵢ₋ₗ
    else:
        Lp = -L
        g[Lp:] = 1 - h[: n - Lp]         # (1‑hᵢ₋ₗp)  where L = –Lp
        g[: n - Lp] -= h[Lp:]            # –hᵢ₊ₗp
    return g


# ----------------------------------------------------------------------
# Directed greedy optimisation (uses the worst‑lag gradient)
# ----------------------------------------------------------------------


def greedy_optimize(h0: np.ndarray, rng: np.random.Generator,
                    max_iters: int = 5000, init_step: float = 0.3):
    """Deterministic hill‑climb guided by the worst‑lag gradient."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step

    for it in range(max_iters):
        # locate the lag giving the current maximum
        corr_full = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr_full)
        L = max_idx - (n - 1)

        # gradient of that lag
        g = gradient_worst_lag(h, L)

        # indices we may decrease / increase
        dec_candidates = np.where((h > 1e-12) & (g > 0))[0]
        inc_candidates = np.where((h < 1 - 1e-12) & (g < 0))[0]

        if dec_candidates.size and inc_candidates.size:
            i = dec_candidates[np.argmax(g[dec_candidates])]
            j = inc_candidates[np.argmin(g[inc_candidates])]
            delta = min(step, h[i], 1 - h[j])
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # fall back to a tiny random tweak
            new_h = perturb_h(h, rng, scale=step)

        new_c = compute_c5(new_h)

        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()

        # accept only improving moves (simple descent)
        if new_c < compute_c5(h):
            h = new_h

        # gradually shrink the step size
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

# ----------------------------------------------------------------------


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Choose resolution and initialise from the supplied parent (if any)
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
    else:
        # fallback: a neutral 90‑point constant 0.5 function
        n0 = 90
        parent = np.full(n0, 0.5)

    # Target resolution: try to double the parent if the budget permits
    n = min(200, max(90, 2 * n0))

    if n == n0:
        h0 = parent.copy()
    else:
        if n % n0 == 0:                     # integer up‑sampling
            factor = n // n0
            h0 = np.repeat(parent, factor)
        else:                               # linear interpolation for odd ratios
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, parent)

    # enforce feasibility exactly
    h0 = np.clip(h0, 0.0, 1.0)
    target = n / 2.0
    h0 = proj_box_simplex(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------------
    # Phase 1 – smooth max optimisation with increasing β (L‑BFGS‑B)
    # --------------------------------------------------------------
    betas = [5.0, 20.0, 80.0, 200.0, 500.0]
    lam = 10.0                     # quadratic penalty for the sum constraint

    for beta in betas:
        if time.time() - start_time > budget_s * 0.80:
            break

        def obj_grad(x):
            o, g = softmax_obj_and_grad(x, beta)
            diff = x.sum() - target
            o += lam * diff * diff
            g = g + 2.0 * lam * diff
            return o, g

        try:
            res = minimize(
                obj_grad,
                best_h,
                method="L-BFGS-B",
                jac=True,
                bounds=[(0.0, 1.0)] * n,
                options={"maxiter": 300, "ftol": 1e-12, "disp": False},
            )
            h_new = proj_box_simplex(res.x, target)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
        except Exception:
            continue

    # --------------------------------------------------------------
    # Phase 2 – directed greedy descent (many iterations)
    # --------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    # allocate most of the rest to a long greedy run on the current best
    if remaining > 30:
        iters = int(min(20000, remaining * 15))   # ≈ 15 iters per second budget
        best_h, best_c = greedy_optimize(best_h, rng, max_iters=iters, init_step=0.5)

    # --------------------------------------------------------------
    # Phase 3 – random restarts with short greedy refinements
    # --------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.98:
        h_start = generate_random_feasible(n, rng)
        h_opt, c_opt = greedy_optimize(h_start, rng, max_iters=3000, init_step=0.4)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # --------------------------------------------------------------
    # Final feasibility clean‑up
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_simplex(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
