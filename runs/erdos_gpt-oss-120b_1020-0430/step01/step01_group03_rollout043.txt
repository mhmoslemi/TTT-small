```
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # compute convolution of h with (1-h) reversed? Actually correlation:
    # In numpy: np.correlate(h, 1-h, mode='full')
    # Equivalent to convolution with reversed second array.
    # Using FFT:
    f = np.fft.rfft(h, n*2)
    g = np.fft.rfft(1-h, n*2)
    # correlation length 2n-1; we can compute via ifft of product of f * conj(g)
    corr = np.fft.irfft(f * np.conj(g), n*2)
    # result length 2n, but we need length 2n-1 (excluding last). We'll slice up to 2n-1.
    corr = corr[:2*n-1] * dx
    return np.max(corr)
```

```
def simulated_annealing_bin(h, time_limit, start_temp=0.02, end_temp=0.0005):
    start = time.time()
    n = h.size
    weight = int(np.round(h.sum()))
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    # Precompute ones and zeros positions
    ones = np.where(cur==1)[0]
    zeros = np.where(cur==0)[0]
    iter = 0
    while time.time() - start < time_limit:
        # Temperature schedule: exponential decay based on progress
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        temp = start_temp * (end_temp/start_temp) ** fraction  # exponential decay
        # propose swap
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # apply swap
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur)
        delta = new_c - cur_c
        accept = delta < 0 or rng.random() < np.exp(-delta / temp)
        if accept:
            cur_c = new_c
            # update sets
            ones = np.where(cur==1)[0]
            zeros = np.where(cur==0)[0]
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0
        iter += 1
    return best, best_c
```

```
    low, high = 0.0, 1.0
    for _ in range(30):
        m1 = low + (high - low) / 3
        m2 = high - (high - low) / 3
        val1 = np.max(0.25 * L * (1 - m1) + m1 * corr)
```

```
def binary_hill_climb(initial_b, rng, max_iters=5000, start_temp=0.01):
    b = initial_b.copy()
    best_b = b.copy()
    best_val, best_lambda = evaluate_pattern(b)
    best_c = best_val * dx
    cur_val = best_val
    cur_lambda = best_lambda
    cur_c = best_c
    for it in range(max_iters):
        # propose swap
        one_idxs = np.where(b == 1)[0]
        zero_idxs = np.where(b == 0)[0]
        i = rng.choice(one_idxs)
        j = rng.choice(zero_idxs)
        b[i] = 0
        b[j] = 1
        # evaluate
        new_val, new_lambda = evaluate_pattern(b)
        new_c = new_val * dx
        # accept if better
        if new_c < cur_c:
            cur_c, cur_val, cur_lambda = new_c, new_val, new_lambda
            if new_c < best_c:
                best_c, best_val, best_lambda = new_c, new_val, new_lambda
                best_b = b.copy()
        else:
            # revert
            b[i] = 1
            b[j] = 0
    return best_b, best_c, best_lambda
```

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None   # sum constraint target (n/2)
CURRENT_BETA = None     # smoothing parameter for soft‑max

def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_{k} Σ_i x_i (1 - x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    """Exact C5 for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP optimisation using the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct possible drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def overlap_lengths(n):
    """Return array L_i = overlap length for each lag (full correlation)."""
    k = np.arange(-(n - 1), n)            # lag indices
    L = n - np.abs(k)                    # overlapping positions
    return L.astype(np.float64)          # shape (2n‑1,)

def evaluate_binary_pattern(b, L):
    """
    Evaluate a binary pattern b.
    Returns:
        best_val  – minimal possible max_{k} Σ_i b_i(1-b_{i+k}) after mixing,
        lambda_opt – optimal mixing coefficient (0≤λ≤1),
        corr       – raw mismatch counts for each lag.
    """
    corr = np.correlate(b, 1 - b, mode="full")          # mismatches per lag
    # ternary search on t = λ² ∈ [0,1] to minimise max_k (0.25 L (1‑t) + t·corr)
    lo, hi = 0.0, 1.0
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = np.max(0.25 * L * (1 - m1) + m1 * corr)
        v2 = np.max(0.25 * L * (1 - m2) + m2 * corr)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    best_val = np.max(0.25 * L * (1 - t_opt) + t_opt * corr)
    lambda_opt = np.sqrt(t_opt)
    return best_val, lambda_opt, corr

def binary_hill_climb(b0, L, rng, max_iters=5000):
    """
    Simple hill‑climb on binary vectors (swap a 1 with a 0).
    Uses the mixed‑λ objective defined in `evaluate_binary_pattern`.
    Returns the best binary vector, its unmixed best_val and the λ that attains it.
    """
    b = b0.copy()
    best_val, best_lambda, _ = evaluate_binary_pattern(b, L)
    cur_val, cur_lambda, _ = best_val, best_lambda, None
    best_b = b.copy()
    for _ in range(max_iters):
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        b[i] = 0
        b[j] = 1
        new_val, new_lambda, _ = evaluate_binary_pattern(b, L)
        if new_val < cur_val:
            cur_val, cur_lambda = new_val, new_lambda
            if new_val < best_val:
                best_val, best_lambda = new_val, new_lambda
                best_b = b.copy()
        else:
            # revert move
            b[i] = 1
            b[j] = 0
    return best_b, best_val, best_lambda

def legendre_binary(p, n, rng):
    """Generate a binary Legendre‑type vector of length n (weight ≈ n/2)."""
    residues = set((a * a) % p for a in range(p))
    seq = np.fromiter(((1 if i in residues else 0) for i in range(p)),
                      dtype=np.float64, count=p)
    if n > p:
        seq = np.concatenate([seq, np.zeros(n - p, dtype=np.float64)])
    else:
        seq = seq[:n]
    # enforce exact weight n/2
    target = n // 2
    cur = int(seq.sum())
    if cur > target:
        idx = np.where(seq == 1)[0][: cur - target]
        seq[idx] = 0
    elif cur < target:
        idx = np.where(seq == 0)[0][: target - cur]
        seq[idx] = 1
    return seq
