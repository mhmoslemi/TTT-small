import time
import numpy as np
from scipy import optimize

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny corrective loop (should almost never fire)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a given h (integral over [0,2] is already ensured)."""
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _gradient_of_worst_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the active (largest) correlation term."""
    n = h.size
    grad = np.ones(n)
    if k >= 0:
        if n - k > 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    else:
        kk = -k
        if n - kk > 0:
            grad[kk:] -= h[: n - kk]
            grad[: n - kk] -= h[kk:]
    return grad


def _local_optimize(
    h0: np.ndarray,
    n: int,
    max_iter: int = 2000,
    step0: float = 0.5,
    time_left=lambda: 1e9,
) -> tuple:
    """
    Simple sub‑gradient descent on the worst correlation.
    Returns the best vector encountered and its C5 value.
    """
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    for it in range(max_iter):
        if time_left() < 0.0:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        grad = _gradient_of_worst_shift(h, k)

        # linearly decaying step size
        step = step0 * (1.0 - it / max_iter)
        if step < 1e-6:
            step = 1e-6

        h = h - step * grad
        h = _project_to_feasible(h, n)

        # occasional jitter to escape plateaus
        if it % 250 == 0 and it > 0:
            h += np.random.normal(scale=0.01, size=n)
            h = _project_to_feasible(h, n)

        c5 = _c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def _solve_h(initial_h: np.ndarray, max_iter: int = 120) -> tuple:
    """One short SLSQP refinement (same formulation as the reference solution)."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    t0 = np.max(corr0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def _ineq(x):
        h = x[:n]
        t = x[-1]
        return t - np.correlate(h, 1.0 - h, mode="full")

    def _eq(x):
        return np.sum(x[:n]) - (n / 2.0)

    cons = [
        {"type": "ineq", "fun": _ineq},
        {"type": "eq", "fun": _eq},
    ]

    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    c5 = _c5_from_h(h_opt)
    return h_opt, c5
