def optimize_h(start, n, deadline):
    h = start.copy()

    for a in alphas:
        if time.time() > deadline: break
        res = minimize(lambda x: softmax_c5(x, alpha=a),

                       bounds=[(0,1)]*n,
                       constraints={'type':'eq','fun': lambda x: x.sum() - n/2},
                       options={'maxiter': 300, 'ftol':1e-9, 'disp':False})
        if not res.success:
            break
        h = np.clip(res.x, 0,1)

    h = h.astype(float)
    return h, compute_c5(h)

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max_k  ∫h(1‑h∘k) using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    # log‑sum‑exp, numerically stable
    maxv = np.max(conv)
    sumexp = np.exp(alpha * (conv - maxv)).sum()
    return float(maxv + np.log(sumexp) / alpha)


# ----------------------------------------------------------------------
# Projection onto the capped simplex {0≤x≤1, sum(x)=S}
# ----------------------------------------------------------------------
def project_to_capped_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum(x)=S}."""
    n = v.size
    lo, hi = -1.0, v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction (rare)
    diff = S - y.sum()
    if abs(diff) > 1e-10:
        y += diff / n
        y = np.clip(y, 0.0, 1.0)
    return y


# ----------------------------------------------------------------------
# Start‑vector factories
# ----------------------------------------------------------------------
def random_start_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)                     # uniform in [0,1)
    return project_to_capped_simplex(v, n / 2.0)


def generate_triangular(n: int) -> np.ndarray:
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 0.5 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def generate_cosine(n: int, amplitude: float, rng: np.random.Generator = None) -> np.ndarray:
    """h(x)=0.5+amp*cos(pi*x) on [0,2]."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 0.5 / n
    h = 0.5 + amplitude * np.cos(np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def generate_block_shape(n: int, p: float) -> np.ndarray:
    """Two linear ramps of length p·n each, flat top in the middle."""
    left = int(round(p * n))
    right = left
    mid = n - left - right
    if mid < 0:
        left = right = n // 2
        mid = n - left - right
    h = np.zeros(n)
    if left > 0:
        h[:left] = np.linspace(0.0, 1.0, left, endpoint=False)
    if mid > 0:
        h[left:left + mid] = 1.0
    if right > 0:
        h[left + mid:] = np.linspace(1.0, 0.0, right, endpoint=False)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to new length, then project."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 0.5 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 0.5 / new_n
    up = np.interp(new_x, old_x, arr)
    return project_to_capped_simplex(up, new_n / 2.0)


# ----------------------------------------------------------------------
# Local optimisation routine
# ----------------------------------------------------------------------
def local_optimize(start: np.ndarray, n: int, deadline: float) -> tuple:
    """Two‑stage SLSQP with increasing sharpness of the soft‑max objective."""
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    alphas = [5.0, 15.0, 30.0, 60.0, 120.0]   # increasingly sharp
    h = start.copy()
    for a in alphas:
        if time.time() > deadline:
            return None, None
        res = minimize(
            lambda x: softmax_c5(x, alpha=a),
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            break
        h = np.clip(res.x, 0.0, 1.0)
        # re‑project to keep the equality constraint exact (tiny drift)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    if time.time() > deadline:
        return None, None
    return h, compute_c5(h)


# ----------------------------------------------------------------------
# Main driver required by the harness
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a low C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95   # safety margin

    # ------------------------------------------------------------------
    # Determine base resolution (use parent if present)
    # ------------------------------------------------------------------
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        base_n = parent_h.shape[0]
    except NameError:
        parent_h = None
        base_n = 100

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Stage 1 – optimisation at the base resolution
    # ------------------------------------------------------------------
    n = base_n
    target_sum = n / 2.0
    starts = []

    if parent_h is not None:
        if parent_h.shape[0] == n:
            starts.append(parent_h.copy())
        else:
            starts.append(upsample(parent_h, n))

    starts.append(generate_triangular(n))
    starts.append(generate_cosine(n, amplitude=0.5, rng=rng))

    # a few random block shapes
    for _ in range(3):
        p = rng.uniform(0.1, 0.35)
        starts.append(generate_block_shape(n, p))

    # random feasible vectors
    for _ in range(5):
        starts.append(random_start_feasible(n, rng))

    # perturbed versions of the cosine start
    for _ in range(2):
        base = generate_cosine(n, amplitude=rng.uniform(0.2, 0.5), rng=rng)
        pert = base + rng.uniform(-0.15, 0.15, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = project_to_capped_simplex(pert, target_sum)
        starts.append(pert)

    # ------------------------------------------------------------------
    # Optimise each start at the base size
    # ------------------------------------------------------------------
    for s in starts:
        if time.time() > time_limit:
            break
        s = project_to_capped_simplex(s, target_sum)
        h_opt, c5_opt = local_optimize(s, n, time_limit)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

    # ------------------------------------------------------------------
    # Stage 2 – progressive refinement at larger resolutions
    # ------------------------------------------------------------------
    # Growth factors (≈ 1.5×, 2×, 2.5×, 3×)
    growths = [1.5, 2.0, 2.5, 3.0]
    for g in growths:
        if time.time() > time_limit:
            break
        new_n = int(round(base_n * g))
        if new_n <= best_n:
            continue
        target_sum = new_n / 2.0

        # Upsample the current best solution
        candidate = upsample(best_h, new_n)

        # Small random perturbations around the upsampled candidate
        starts = [candidate]
        for _ in range(3):
            pert = candidate + rng.normal(scale=0.07, size=new_n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_to_capped_simplex(pert, target_sum)
            starts.append(pert)

        # a couple of fresh random starts for diversity
        for _ in range(2):
            starts.append(random_start_feasible(new_n, rng))

        # optimise this resolution
        for s in starts:
            if time.time() > time_limit:
                break
            h_opt, c5_opt = local_optimize(s, new_n, time_limit)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, new_n

    # ------------------------------------------------------------------
    # Safety fallback (guaranteed feasible) if everything failed
    # ------------------------------------------------------------------
    if best_h is None:
        n = 80
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # ------------------------------------------------------------------
    # Final sanity adjustments
    # ------------------------------------------------------------------
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
