import time, random, numpy as np

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n ∩ {sum = target} by bisection on the Lagrange multiplier."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5_and_shift(h: np.ndarray):
    """Return (C5 value, lag) for step vector h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    val = corr[idx] * (2.0 / n)
    lag = idx - (n - 1)
    return float(val), int(lag)


def compute_c5(h: np.ndarray) -> float:
    """C5 value only."""
    return compute_c5_and_shift(h)[0]


def gradient_for_shift(h: np.ndarray, lag: int) -> np.ndarray:
    """Subgradient of the worst‑shift overlap w.r.t. h (unscaled by dx)."""
    n = h.shape[0]
    g = np.ones(n)
    if lag >= 0:
        l = lag
        if l < n:
            g[: n - l] -= h[l:]
        if l > 0:
            g[l:] -= h[: n - l]
    else:
        l = -lag
        if l < n:
            g[l:] -= h[: n - l]
        if l > 0:
            g[: n - l] -= h[l:]
    return g


def make_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(pts - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_rect(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    h = np.where(np.abs(pts - 1.0) < 0.5, 1.0, 0.0)
    return project_onto_bounds_and_sum(h.astype(float), n / 2.0)


def make_double_triangle(n: int) -> np.ndarray:
    dx = 2.0 / n
    pts = (np.arange(n) + 0.5) * dx
    t1 = np.maximum(0.0, 1 - np.abs(pts - 0.7) / 0.4)
    t2 = np.maximum(0.0, 1 - np.abs(pts - 1.3) / 0.4)
    h = np.clip(t1 + t2, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(old: np.ndarray, n: int) -> np.ndarray:
    old_n = old.shape[0]
    if old_n == n:
        return project_onto_bounds_and_sum(old, n / 2.0)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, old)
    return project_onto_bounds_and_sum(h, n / 2.0)


def optimize_via_gradient(h0: np.ndarray, target: float,
                         max_iters: int = 3000,
                         time_limit: float = None) -> (np.ndarray, float):
    """Decrease the worst‑shift overlap using subgradient steps."""
    h = h0.copy()
    best_val, lag = compute_c5_and_shift(h)
    eta = 0.2
    no_improve = 0
    start = time.time()
    for _ in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        g = gradient_for_shift(h, lag)
        improved = False
        for f in (1.0, 0.5, 0.25, 0.125):
            step = eta * f
            cand = h - step * g
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_bounds_and_sum(cand, target)
            val = compute_c5(cand)
            if val < best_val - 1e-12:
                h, best_val = cand, val
                best_val, lag = compute_c5_and_shift(h)
                improved = True
                break
        if not improved:
            eta *= 0.5
            no_improve += 1
            if eta < 1e-6 or no_improve > 30:
                break
        else:
            no_improve = 0
        if best_val <= 0.3808:
            break
    return h, best_val


def random_pairwise_refine(h0: np.ndarray, target: float,
                          max_moves: int = 20000,
                          time_limit: float = None) -> (np.ndarray, float):
    """Simple stochastic local search swapping mass between two points."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    moves = 0
    while moves < max_moves:
        if time_limit is not None and time.time() - start > time_limit:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_bounds_and_sum(cand, target)
        val = compute_c5(cand)
        if val < best_val:
            best_h, best_val = cand, val
            if best_val <= 0.3808:
                break
        moves += 1
    return best_h, best_val
