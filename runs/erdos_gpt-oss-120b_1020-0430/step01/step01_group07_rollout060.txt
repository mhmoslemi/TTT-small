def compute_c5(h):
    n = h.shape[0]
    L = 1 << (2*n-1).bit_length()   # next power of two for FFT
    H = np.fft.rfft(h, L)
    corr = np.fft.irfft(H * np.conj(H), L)[:2*n-1]
    min_corr = corr.min()

    H = np.fft.rfft(h, 2*n)  # length 2n zero-padded
    corr = np.fft.irfft(H * np.conj(H), 2*n)[:2*n -1]  # full linear conv

    c5 = ((n/2) - min_corr) * (2.0 / n)
    return float(c5)

    n = h.size
    # Ensure binary? Not required.
    current_corr = _full_corr(h)
    cur_c = _c5_from_corr(current_corr, n)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    # schedule
    T0 = 0.02  # adjust
    T_end = 1e-6
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        # choose targeted or random
        use_target = random.random() < 0.7
        if use_target:
            # find lag with minimal correlation (worst)
            idx_min = current_corr.argmin()
            k_star = idx_min - (n - 1)
            # gradient
            grad = np.zeros(n, dtype=np.float64)
            if k_star >= 0:
                k = k_star
                grad[:n - k] += h[k:]
                grad[k:] += h[:n - k]
            else:
                k = -k_star
                grad[k:] += h[:n - k]
                grad[:n - k] += h[k:]
            # find candidate indices
            # choose i with h>0 (or >1e-6) and lowest grad; j with h<1 (or <0.999) and highest grad
            # Use thresholds
            mask_one = h > 0.5
            mask_zero = h < 0.5
            if mask_one.any() and mask_zero.any():
                i_candidates = np.where(mask_one)[0]
                j_candidates = np.where(mask_zero)[0]
                i = i_candidates[np.argmin(grad[i_candidates])]
                j = j_candidates[np.argmax(grad[j_candidates])]
            else:
                # fallback random
                i = np.random.randint(n)
                j = np.random.randint(n)
                while i == j:
                    j = np.random.randint(n)
        else:
            # random swap between different values
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)

        hi, hj = h[i], h[j]
        if hi == hj:
            continue
        # swap
        h[i], h[j] = hj, hi
        new_corr = _full_corr(h)
        new_c = _c5_from_corr(new_corr, n)
        accept = False
        if new_c < cur_c - 1e-12:
            accept = True
        else:
            # probabilistic acceptance
            if T > 0 and random.random() < math.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            current_corr = new_corr
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = hi, hj
    return best_h, best_c

def _anneal_fine(start_h, max_seconds, init_c5=None):
    h = start_h.copy()
    n = h.size
    cur_c = _c5_metric(h) if init_c5 is None else init_c5
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    T0 = 0.03
    T_end = 1e-6
    max_step = 0.2
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        T = T0 * (T_end / T0) ** progress
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        # compute feasible delta range
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # restrict to moderate steps
        delta = random.uniform(low, high)
        # optionally cap max_step
        delta = max(min(delta, max_step), -max_step)
        h[i] = hi + delta
        h[j] = hj - delta

                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c

def _upscale(h, factor=2):
    n = h.size
    n_new = n * factor
    if n_new % 2:
        n_new += 1
    x_old = np.linspace(0.0,2.0,n,endpoint=False)
    x_new = np.linspace(0.0,2.0,n_new,endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    # enforce sum constraint exactly
    target = n_new / 2.0
    cur = h_up.sum()
    if cur != 0:
        h_up *= target / cur
    np.clip(h_up, 0.0, 1.0, out=h_up)
    deficit = target - h_up.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n_new):
            if 0.0 <= h_up[idx] + deficit <= 1.0:
                h_up[idx] += deficit
                break
    return h_up

import time, random, math, numpy as np

def _full_corr(h: np.ndarray) -> np.ndarray:
    """Linear (full) autocorrelation of h with itself via FFT."""
    n = h.shape[0]
    # length for FFT – next power of two ≥ 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    conv = np.fft.irfft(H * np.conj(H), n=L)[: 2 * n - 1]
    return conv.real  # discard tiny imag part


def _c5_metric(h: np.ndarray) -> float:
    """C₅ value for the sampled step function."""
    n = h.shape[0]
    corr = _full_corr(h)
    min_corr = corr.min()
    return ((n / 2.0) - min_corr) * (2.0 / n)


def _anneal_targeted(start_h: np.ndarray, max_seconds: float) -> tuple:
    """
    Simulated‑annealing / hill‑climbing that prefers swaps which
    improve the worst autocorrelation lag.
    """
    h = start_h.copy()
    n = h.shape[0]
    cur_corr = _full_corr(h)
    cur_c = ((n / 2.0) - cur_corr.min()) * (2.0 / n)
    best_h, best_c = h.copy(), cur_c

    start = time.time()
    T0, T_end = 0.04, 1e-6

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        T = T0 * (T_end / T0) ** prog

        # --- choose move ----------------------------------------------------
        use_target = random.random() < 0.7
        if use_target:
            # lag where correlation is minimal (worst shift)
            idx_min = cur_corr.argmin()
            k_star = idx_min - (n - 1)

            # gradient of the worst‑lag correlation
            grad = np.zeros(n, dtype=np.float64)
            if k_star >= 0:
                k = k_star
                grad[: n - k] += h[k:]          # forward neighbours
                grad[k:] += h[: n - k]          # backward neighbours
            else:
                k = -k_star
                grad[k:] += h[: n - k]          # i‑k term
                grad[: n - k] += h[k:]          # i+k term

            # pick a 1 with low gradient and a 0 with high gradient
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if ones.size and zeros.size:
                i = ones[np.argmin(grad[ones])]
                j = zeros[np.argmax(grad[zeros])]
            else:
                i = random.randint(0, n - 1)
                j = random.randint(0, n - 1)
                while h[i] == h[j]:
                    i = random.randint(0, n - 1)
                    j = random.randint(0, n - 1)
        else:
            # plain random swap of different values
            i = random.randint(0, n - 1)
            j = random.randint(0, n - 1)
            while h[i] == h[j]:
                i = random.randint(0, n - 1)
                j = random.randint(0, n - 1)

        hi, hj = h[i], h[j]
        if hi == hj:          # should not happen
            continue

        # perform swap
        h[i], h[j] = hj, hi
        new_corr = _full_corr(h)
        new_c = ((n / 2.0) - new_corr.min()) * (2.0 / n)

        accept = False
        if new_c < cur_c - 1e-12:
            accept = True
        else:
            if random.random() < math.exp(-(new_c - cur_c) / T):
                accept = True

        if accept:
            cur_c = new_c
            cur_corr = new_corr
            if new_c < best_c - 1e-12:
                best_c, best_h = new_c, h.copy()
        else:
            # revert swap
            h[i], h[j] = hi, hj

    return best_h, best_c


def _anneal_fine(start_h: np.ndarray, max_seconds: float, init_c: float = None) -> tuple:
    """
    Fine‑grained simulated annealing using small pairwise deltas.
    Keeps the total mass exactly.
    """
    h = start_h.copy()
    n = h.shape[0]
    cur_c = _c5_metric(h) if init_c is None else init_c
    best_h, best_c = h.copy(), cur_c

    start = time.time()
    T0, T_end = 0.06, 1e-6
    max_step = 0.2

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        prog = elapsed / max_seconds
        T = T0 * (T_end / T0) ** prog

        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        delta = random.uniform(low, high)
        # restrict step size a bit for stability
        delta = max(min(delta, max_step), -max_step)

        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = hi, hj

    return best_h, best_c


def _binary_start(n: int, source: np.ndarray = None) -> np.ndarray:
    """
    Produce a binary (or half‑binary) vector of length n whose integral equals 1.
    If *source* is supplied it is interpolated and the largest entries are taken.
    """
    target = n / 2.0
    ones = int(target)
    fractional = target - ones

    if source is not None:
        # linear interpolation onto the new grid
        x0 = np.linspace(0.0, 2.0, source.size, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        src = np.interp(x, x0, source)
        # keep the biggest entries
        idx = np.argsort(-src)[:ones]
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
        if fractional > 1e-12:
            extra_idx = np.argsort(-src)[ones]
            h[extra_idx] = fractional
        return h

    # plain random binary with exact mass
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.permutation(n)[:ones]
    h[idx] = 1.0
    if fractional > 1e-12:
        # a single fractional site
        rest = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        h[rest[0]] = fractional
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step‑function giving a low C₅ bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ---------------------------------------------------------------
    # decide the working resolution
    base_n = 200                     # modest size, fast to evaluate
    if "initial_h_values" in globals():
        base_n = max(base_n, ((len(initial_h_values) + 9) // 10) * 10)
    if base_n % 2:
        base_n += 1

    # ---------------------------------------------------------------
    # initial feasible vector (binary, derived from the supplied one if present)
    if "initial_h_values" in globals():
        h0 = _binary_start(base_n, np.asarray(initial_h_values, dtype=np.float64))
    else:
        h0 = _binary_start(base_n)

    total_start = time.time()
    total_budget = float(budget_s)

    # ------------------------ primary anneal ------------------------
    primary_time = total_budget * 0.55
    best_h, best_c = _anneal_targeted(h0, primary_time)

    # ------------------------ random restarts -----------------------
    elapsed = time.time() - total_start
    remaining = total_budget - elapsed
    restart_time = min(15.0, remaining * 0.3)   # short bursts

    while remaining > restart_time + 5.0:
        h_rand = _binary_start(base_n)
        cand_h, cand_c = _anneal_targeted(h_rand, restart_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()
        elapsed = time.time() - total_start
        remaining = total_budget - elapsed

    # ------------------------ fine anneal ---------------------------
    elapsed = time.time() - total_start
    remaining = total_budget - elapsed
    if remaining > 5.0:
        best_h, best_c = _anneal_fine(best_h, remaining, init_c=best_c)

    # ------------------------ optional up‑scale --------------------
    # if a decent chunk is still unused, double the resolution and continue
    elapsed = time.time() - total_start
    remaining = total_budget - elapsed
    if remaining > 30.0:
        n_big = best_h.size * 2
        if n_big % 2:
            n_big += 1
        # linear interpolation to the finer grid
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_big, endpoint=False)
        h_big = np.interp(x_new, x_old, best_h)

        # enforce exact integral
        target_big = n_big / 2.0
        cur = h_big.sum()
        if cur != 0:
            h_big *= target_big / cur
        np.clip(h_big, 0.0, 1.0, out=h_big)
        deficit = target_big - h_big.sum()
        if abs(deficit) > 1e-12:
            for idx in np.random.permutation(n_big):
                if 0.0 <= h_big[idx] + deficit <= 1.0:
                    h_big[idx] += deficit
                    break

        best_h, best_c = _anneal_targeted(h_big, remaining)

    # ----------------------------------------------------------------
    # final feasibility correction (tiny rounding errors)
    n = best_h.size
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # ----------------------------------------------------------------
    # compute the final bound (use helper if supplied)
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
