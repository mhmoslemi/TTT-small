def anneal(h, max_seconds):
    best = h.copy()
    best_val = c5_metric(best)
    cur = h.copy()
    cur_val = best_val
    n = len(h)
    start = time.time()
    while time.time() - start < max_seconds:
        # generate pair
        i, j = np.random.randint(0, n, 2)
        hi, hj = cur[i], cur[j]
        # compute allowable delta
        low = max(-hi, hj - 1)
        high = min(1 - hi, hj)
        if low >= high:
            continue
        # scale max step by temperature
        progress = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** progress
        # step amplitude
        max_step = 0.1 * (T / T0) + 0.001 # ensures small step at low T
        # intersect with allowed
        low_c = max(low, -max_step)
        high_c = min(high, max_step)
        if low_c < high_c:
            delta = np.random.uniform(low_c, high_c)
        else:
            delta = np.random.uniform(low, high)
        # apply
        cur[i] = hi + delta
        cur[j] = hj - delta
        new_val = c5_metric(cur)
        d = new_val - cur_val
        if d < 0 or np.random.rand() < np.exp(-d / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            cur[i] = hi
            cur[j] = hj
    return best, best_val

def enforce_sum(h):
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to adjust a random index within bounds
    idxs = np.arange(n)
    np.random.shuffle(idxs)
    for idx in idxs:
        if 0 <= h[idx] + diff <= 1:
            h[idx] += diff
            break
    else:
        # if not possible, distribute across viable indices
        # adjust with min capacity
        if diff > 0:
            caps = 1 - h
            cap_total = caps.sum()
            if cap_total > 0:
                h += diff * caps / cap_total
        else:
            caps = h
            cap_total = caps.sum()
            if cap_total > 0:
                h += diff * (-caps) / cap_total
        # clip
        np.clip(h, 0, 1, out=h)
    return h

def _c5_metric(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _make_feasible(h):
    """Clip to [0,1] and enforce sum == n/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix by adjusting a single entry
    idxs = np.random.permutation(n)
    for idx in idxs:
        lo = 0.0
        hi = 1.0
        val = h[idx] + diff
        if lo <= val <= hi:
            h[idx] = val
            diff = 0.0
            break
    if abs(diff) > 1e-12:
        # distribute diff across all non‑boundary entries
        if diff > 0:
            caps = 1.0 - h
        else:
            caps = -h
        cap_total = caps.sum()
        if cap_total > 0:

                    break
