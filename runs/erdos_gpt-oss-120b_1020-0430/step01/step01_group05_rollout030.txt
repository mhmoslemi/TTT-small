def try_pairwise_adjust(h, target_sum):
    n = len(h)
    i, j = np.random.randint(0,n,2)
    hi, hj = h[i], h[j]
    delta_max = min(1-hi, hj)  # increase hi, decrease hj
    delta_min = -min(hi, 1-hj)
    if delta_max < delta_min: return None
    delta = np.random.uniform(delta_min, delta_max)
    if abs(delta) < 1e-12: return None
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    # Already ensures within [0,1]
    # sum unchanged due to delta + (-delta)
    return cand
```

import time, random, math
import numpy as np

def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n intersect sum(x)=target via Lagrange bisection."""
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised objective."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def upsample_initial(init: np.ndarray, n: int, target: float) -> np.ndarray:
    """Linear interpolation of a parent vector to length n."""
    old_n = init.shape[0]
    if old_n == n:
        h = init.astype(float).copy()
    else:
        dx_old = 2.0 / old_n
        x_old = (np.arange(old_n) + 0.5) * dx_old
        dx_new = 2.0 / n
        x_new = (np.arange(n) + 0.5) * dx_new
        h = np.interp(x_new, x_old, init, left=0.0, right=0.0)
    return project_onto_bounds_and_sum(h, target)


def piecewise_const_h(w1: float, w2: float, n: int, target: float) -> np.ndarray:
    """Symmetric shape: height 1 for |x-1|≤w1, height a for w1<|x-1|≤w2, else 0."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    d = np.abs(x - 1.0)
    denom = max(w2 - w1, 1e-12)
    a = (0.5 - w1) / denom
    a = np.clip(a, 0.0, 1.0)
    h = np.where(d <= w1, 1.0, np.where(d <= w2, a, 0.0))
    return project_onto_bounds_and_sum(h, target)


def trapezoid_h(p: float, n: int, target: float) -> np.ndarray:
    """Symmetric trapezoid: plateau of height 1 of half‑width p, linear slopes to 0."""
    a = p          # half‑width of the flat top (0 ≤ a ≤ 0.5)
    s = 1.0 - 2.0 * a  # slope width; may be zero
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    d = np.abs(x - 1.0)
    h = np.zeros_like(x)
    if a > 0:
        h[d <= a] = 1.0
    if s > 1e-12:
        mask = (d > a) & (d <= a + s)
        h[mask] = 1.0 - (d[mask] - a) / s
    return project_onto_bounds_and_sum(h, target)

    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # problem size (keep <1000 for speed, 500 gives good resolution)
    n = 500
    target_sum = n / 2.0

    best_h = None
    best_val = float('inf')

    # --------------------------------------------------------------
    # try parent solution if supplied
    if 'initial_h_values' in globals():
        try:
            init = upsample_initial(np.array(initial_h_values, dtype=float), n, target_sum)
            val = compute_c5(init)
            best_h, best_val = init, val
        except Exception:
            pass

    # --------------------------------------------------------------

    dx = 2.0 / n

    tri = project_onto_bounds_and_sum(tri, target_sum)
    tri_val = compute_c5(tri)
    if tri_val < best_val:

    # --------------------------------------------------------------

    best_h = project_onto_bounds_and_sum(best_h, target_sum)
