    random.seed(seed)
    np.random.seed(seed)

    # define n
    n = 140
    target_sum = n / 2

            raise NameError

    # --------------------------------------------------------------

            flip = np.random.choice(cand.tolist(), int(round(-diff)), replace=False)
            best_h[flip] = 0.0

    final_c5 = c5_of(best_h)
    return (best_h.tolist(), final_c5, n)
