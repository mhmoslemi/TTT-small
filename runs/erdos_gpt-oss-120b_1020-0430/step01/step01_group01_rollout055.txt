def _corr_fft(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())
    # compute convolution of h and reversed (1-h)
    # Convolution length = 2*n-1.
    f = np.fft.rfft(h, n=N)
    g = np.fft.rfft((1 - h)[::-1], n=N)
    conv = np.fft.irfft(f * g, n=N)
    # conv length N, we need first 2*n-1 entries (full mode).
    corr = conv[:2*n-1] * dx
    return corr

def _project_onto_capped_simplex(v, target, tol=1e-9, max_iter=100):
    # Clip v to [0,1] and then adjust using bisection:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    # after loop
    proj = np.clip(v - lam, 0.0, 1.0)
    # Might need to further correct small sum error.
    # To guarantee sum exactly target, we can adjust by small amount on the entries not at bounds.
    diff = target - proj.sum()
    if abs(diff) > tol:
        # distribute diff equally among non-bound entries
        mask = (proj > 0.0) & (proj < 1.0)
        if np.any(mask):
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the capped simplex {0<=x<=1, sum(x)=target}
# ----------------------------------------------------------------------
def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                tol: float = 1e-9, max_iter: int = 80) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - lam, 0.0, 1.0)
    # tiny correction if rounding left a small error
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    return _project_onto_capped_simplex(rng.rand(n), target)


# ----------------------------------------------------------------------
# FFT based correlation and convolution utilities
# ----------------------------------------------------------------------
def _corr_fft(h: np.ndarray, dx: float) -> np.ndarray:
    """Linear correlation of h with (1‑h) using FFT (full mode)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    fh = np.fft.rfft(h, N)
    g = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(fh * g, N)
    return conv[: 2 * n - 1] * dx


def _conv_full(p: np.ndarray, h: np.ndarray) -> np.ndarray:
    """Linear convolution p * h (full length) via FFT."""
    n, m = h.shape[0], p.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    fp = np.fft.rfft(p, N)
    fh = np.fft.rfft(h, N)
    conv = np.fft.irfft(fp * fh, N)
    return conv[: n + m - 1]


def _smooth_max_and_grad(h: np.ndarray, dx: float, alpha: float):
    """Smooth max (log‑sum‑exp) of the correlation and its gradient w.r.t. h."""
    n = h.shape[0]
    corr = _corr_fft(h, dx)                # length 2n‑1
    m = np.max(corr)
    shifted = corr - m
    exp_term = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # softmax weights, sum to 1

    # convolution for gradient: 1 - conv_rev - conv_fwd
    conv_rev_full = _conv_full(w, h)
    off = n - 1
    conv_rev = conv_rev_full[off: off + n]

    w_rev = w[::-1]
    conv_fwd_full = _conv_full(w_rev, h)
    conv_fwd = conv_fwd_full[off: off + n]

    grad = 1.0 - conv_rev - conv_fwd
    smooth_val = m + np.log(sum_exp) / alpha
    return smooth_val, grad


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Exact C5 value (max correlation)."""
    return float(np.max(_corr_fft(h, dx)))

# ----------------------------------------------------------------------

    rng = np.random.RandomState(seed)
    start = time.time()

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    if parent is not None:

    # --------------------------------------------------------------
    # Optimisation parameters

            step_base = 0.45 / np.sqrt(alpha)          # heuristic step size

                m_hat = m / (1.0 - beta1 ** (it + 1))

    # final projection & evaluation (guarantee feasibility)
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), n)
