import time, random, math, numpy as np

def _c5_metric_raw(h):
    # returns raw correlation max (without dx)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def _c5_metric(h):
    n = h.shape[0]
    return _c5_metric_raw(h) * (2.0 / n)

def _build_initial(n):
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        # interpolation to n
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)
    target = n / 2.0
    # scale to match target sum
    s = target / h.sum()
    h *= s
    np.clip(h, 0.0, 1.0, out=h)
    # fix any possible deficit
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        # find index that can accept the deficit
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + deficit <= hi:
                h[idx] += deficit
                break
    return h

def _binary_start(n):
    target = n / 2.0
    int_on = int(math.floor(target))
    frac = target - int_on
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.permutation(n)
    h[idx[:int_on]] = 1.0
    if frac > 1e-12:
        h[idx[int_on]] = frac
    return h

import time
import random
import math
import numpy as np

def _c5_metric(h):
    """Maximum overlap C₅ for a sampled step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _build_initial(n):
    """Create a feasible start vector of length n."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        h = np.full(n, 0.5, dtype=np.float64)

    target = n / 2.0
    s = target / h.sum()
    if s != 0:
        h *= s
    np.clip(h, 0.0, 1.0, out=h)

    # tiny correction for rounding errors
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + deficit <= hi:
                h[idx] += deficit
                break
    return h


def _binary_start(n):
    """Binary (0/1) feasible start with exactly n/2 total mass."""
    target = n / 2.0
    whole = int(math.floor(target))
    frac = target - whole
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.permutation(n)
    h[idx[:whole]] = 1.0
    if frac > 1e-12:
        h[idx[whole]] = frac
    return h


def _anneal_focused(start_h, time_limit):
    """Simulated‑annealing using a directed move based on the worst lag."""
    n = start_h.shape[0]
    dx = 2.0 / n
    h = start_h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    cur_c_raw = corr[max_idx]
    cur_c = cur_c_raw * dx
    best_h = h.copy()
    best_c = cur_c

    T0 = 0.20               # start temperature
    T_end = 1e-5            # final temperature
    max_step_initial = 0.30

    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        progress = elapsed / time_limit
        T = T0 * (T_end / T0) ** progress
        max_step = max_step_initial * (1.0 - 0.7 * progress)  # anneal step size

        L = max_idx - (n - 1)   # current worst shift

        # ----- choose move -------------------------------------------------
        if random.random() < 0.7:                     # directed move
            if L >= 0:
                contrib = np.empty(n, dtype=np.float64)
                end = n - L
                contrib[:end] = h[:end] * (1.0 - h[L:])
                contrib[end:] = 0.0
            else:
                K = -L
                contrib = np.empty(n, dtype=np.float64)
                contrib[:K] = 0.0
                contrib[K:] = h[K:] * (1.0 - h[:n - K])

            src = int(np.argmax(contrib))
            dst = int(np.argmin(contrib))
            if src == dst:
                i, j = random.sample(range(n), 2)
                low = max(-h[i], h[j] - 1.0)
                high = min(1.0 - h[i], h[j])
                if low >= high:
                    continue
                low_cap = max(low, -max_step)
                high_cap = min(high, max_step)
                delta = random.uniform(low_cap, high_cap) if low_cap < high_cap else random.uniform(low, high)
                i_move, j_move, delta_move = i, j, delta
            else:
                max_move = min(h[src], 1.0 - h[dst], max_step)
                if max_move <= 1e-12:
                    i, j = random.sample(range(n), 2)
                    low = max(-h[i], h[j] - 1.0)
                    high = min(1.0 - h[i], h[j])
                    if low >= high:
                        continue
                    low_cap = max(low, -max_step)
                    high_cap = min(high, max_step)
                    delta = random.uniform(low_cap, high_cap) if low_cap < high_cap else random.uniform(low, high)
                    i_move, j_move, delta_move = i, j, delta
                else:
                    delta_move = -random.uniform(0.0, max_move)   # move mass from src → dst
                    i_move, j_move = src, dst
        else:                                          # random pair
            i, j = random.sample(range(n), 2)
            low = max(-h[i], h[j] - 1.0)
            high = min(1.0 - h[i], h[j])
            if low >= high:
                continue
            low_cap = max(low, -max_step)
            high_cap = min(high, max_step)
            delta_move = random.uniform(low_cap, high_cap) if low_cap < high_cap else random.uniform(low, high)
            i_move, j_move = i, j

        # ----- apply move ---------------------------------------------------
        hi_old, hj_old = h[i_move], h[j_move]
        h[i_move] = hi_old + delta_move
        h[j_move] = hj_old - delta_move

        # ----- evaluate ------------------------------------------------------
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        max_idx_new = int(np.argmax(corr_new))
        new_c_raw = corr_new[max_idx_new]
        new_c = new_c_raw * dx
        dC = new_c - cur_c

        # ----- Metropolis acceptance -----------------------------------------
        if dC < 0.0 or random.random() < math.exp(-dC / max(T, 1e-12)):
            cur_c = new_c
            corr = corr_new
            max_idx = max_idx_new
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i_move] = hi_old
            h[j_move] = hj_old

    return best_h, best_c

    random.seed(seed)

    total_budget = float(budget_s)

    # choose a modestly fine grid

        cand_h, cand_c = _anneal_focused(h_start, slice_time)
        if cand_c < best_c:

    # ----- final fine‑grained hill‑climbing --------------------------------
    n = best_h.shape[0]
    dx = 2.0 / n
    for _ in range(4):
        improved = False
        for _ in range(3000):
