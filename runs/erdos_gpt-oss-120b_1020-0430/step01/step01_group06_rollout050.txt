```
def project_bounds_sum(h, target):
    h = np.clip(h, 0, 1)
    # water-filling to achieve sum target
    # iterate:
    for _ in range(10):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        # find free indices not at bounds (within epsilon)
        free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if len(free) == 0:
            # adjust all indices proportionally
            free = np.arange(len(h))
        delta = diff / len(free)
        h[free] -= delta
        h = np.clip(h, 0, 1)
    # final small correction: adjust any remaining diff via greedy
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        # adjust greedily
        if diff > 0:
            # need to reduce sum
            idx = np.where(h > 1e-12)[0]
            for i in idx:
                dec = min(h[i], diff)
                h[i] -= dec
                diff -= dec
                if diff <= 1e-12:
                    break
        else:
            # need to increase sum
            idx = np.where(h < 1-1e-12)[0]
            for i in idx:
                inc = min(1 - h[i], -diff)
                h[i] += inc
                diff += inc
                if diff >= -1e-12:
                    break
    return h
```

```
def single_run(h):
    best_c5 = eval_c5(h)
    best_h = h.copy()
    # parameters
    beta = 2.0
    beta_max = 30.0
    beta_steps = 500   # steps for each restart 
    step0 = 0.2
    for it in range(5000):
        conv = np.correlate(h, 1 - h, mode='full')
        # compute softmax
        max_beta_conv = np.max(beta * conv)
        w = np.exp(beta * conv - max_beta_conv)
        w /= w.sum()
        # compute gradient
        grad = np.zeros_like(h)
        for shift in range(-n+1, n):
            ws = w[shift + n - 1]
            if ws == 0: continue
            if shift >= 0:
                grad[:n-shift] += ws * (1 - h[shift:])
                grad[shift:] -= ws * h[:n-shift]
            else:
                t = -shift
                grad[t:] += ws * (1 - h[:n-t])
                grad[:n-t] -= ws * h[t:]
        # step size schedule: may be step0/(1+it*0.001)
        step = step0 / np.sqrt(it+1)
        nh = h - step * grad
        nh = np.clip(nh, 0, 1)
        nh = project_bounds_sum(nh, target_sum)
        cur_c5 = eval_c5(nh)
        # Accept if better or else maybe random slight acceptance:
        if cur_c5 < best_c5 or random.random() < 0.001:
            h = nh
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = nh.copy()
        # increase beta gradually
        if beta < beta_max:
            beta += (beta_max - beta)/ (beta_steps)   # perhaps incremental
    return best_h, best_c5
```

```
def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n

    # define evaluator
    def eval_c5(v):
        return np.max(np.correlate(v, 1 - v, mode='full')) * dx

    # projection function
    def project(h):
        # clip then adjust sum
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= 1e-12:
            return h
        # lambda method...
        # Implementation as above
        lower = -1.0
        upper = 1.0
        def sum_at(lam):
            return np.clip(h - lam, 0.0, 1.0).sum()

            upper *= 2.0
        while sum_at(lower) < target:
            lower *= 2.0
        for _ in range(60):
            mid = (lower + upper) / 2.0
            if sum_at(mid) > target:
                lower = mid
            else:
                upper = mid
        return np.clip(h - (lower + upper) / 2.0, 0.0, 1.0)

        h = project(h)
        candidates.append(h)

    best_h = None
    best_c5 = float('inf')
    # time loop across candidates
    for h0 in candidates:
        # perform local optimization on each candidate
        h = h0.copy()
        # local optimizer parameters
        beta = 2.0
        beta_max = 50.0
        beta_inc = (beta_max - beta) / 2000  # for 2000 steps
        # step size base
        base_step = 0.5
        # compute initial C5
        cur_c5 = eval_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # local loop with time check
        it = 0
        while time.time() - start < budget_s * 0.95:

            max_bc = np.max(beta_conv)

            # gradient
            grad = np.zeros_like(h)
            for shift in range(-n+1, n):
                ws = w[shift + n - 1]
                if ws == 0:
                    continue
                if shift >= 0:
                    # slice lengths
                    lo = n - shift
                    grad[:lo] += ws * (1 - h[shift:])
                    grad[shift:] -= ws * h[:lo]
                else:
                    t = -shift
                    lo = n - t
                    grad[t:] += ws * (1 - h[:lo])
                    grad[:lo] -= ws * h[t:]
            # step size decays

            nh = h - step * grad

            # accept if improvement else occasional acceptance
            if new_c5 < cur_c5 or random.random() < 0.001:

                idx_max = int(np.argmax(corr))
                s = idx_max - (n - 1)
                # compute contribution per pair for this shift
                if s >= 0:
                    i_inds = np.arange(n - s)
                    p_inds = i_inds + s
                else:
                    t = -s
                    i_inds = np.arange(t, n)
                    p_inds = i_inds - t
                contrib = h[i_inds] * (1 - h[p_inds])
                k = int(np.argmax(contrib))
                i_best = i_inds[k]
                p_best = p_inds[k]
                eps_max = min(h[i_best], 1 - h[p_best])
                if eps_max > 1e-12:
                    eps = eps_max * 0.5

                    nh2[i_best] -= eps

            if it % 2000 == 0:

                base_step *= 0.9
        # after local optimization, check time for break
        if time.time() - start >= budget_s * 0.95:
            break

    # final best solution
    final_c5 = eval_c5(best_h)
    return (best_h.tolist(), float(final_c5), n)
```
