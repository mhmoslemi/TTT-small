def project_to_feasible(v, target):
    w = np.clip(v, 0, 1)
    for _ in range(10):
        s = w.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            # need to decrease
            excess = s - target
            # eligible indices with w > 0
            idx = w > 1e-12
            if not np.any(idx):
                break
            delta = excess / idx.sum()
            w[idx] -= delta
            w = np.clip(w, 0, 1)
        else:
            deficit = target - s
            idx = w < 1 - 1e-12
            if not np.any(idx):
                break
            delta = deficit / idx.sum()
            w[idx] += delta
            w = np.clip(w, 0, 1)
    return w

def grad_soft_c5(h, alpha):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    # compute softmax
    maxc = np.max(alpha * conv)  # for numeric stability
    expa = np.exp(alpha * conv - maxc)
    w = expa / expa.sum()
    grad = np.zeros_like(h)
    # offsets
    for d in range(-n+1, n):
        w_idx = d + (n-1)
        weight = w[w_idx]
        if weight == 0:
            continue
        if d >= 0:
            # term1: (1 - h[d:]) for indices 0:n-d
            grad[:n-d] += weight * (1 - h[d:])
            # term2: -h[:n-d]
            grad[d:] -= weight * h[:n-d]
        else:
            dneg = -d
            # term1: (1 - h[:n-dneg]) for indices dneg:n
            grad[dneg:] += weight * (1 - h[:n-dneg])
            # term2: -h[dneg:] for indices 0:n-dneg
            grad[:n-dneg] -= weight * h[dneg:]
    return grad

Term2: -h_{j-d} defined for j >= d (since j-d >= 0). So we subtract weight for indices d to n-1.

Similarly for d negative.

def gradient_descent(h0, target, alpha=30.0, lr=0.01, max_iter=5000, tol=1e-9, rng=None):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):

        # step
        step = -lr * g

            best_h = h_new.copy()

        if np.linalg.norm(step) < tol:
            break
        # adapt learning rate
        # Maybe reduce lr gradually
        if it % 500 == 0 and it>0:
            lr *= 0.5
        h = h_new
    return best_h, best_val

def adam_optimize(start, n, target, alphas, steps_per_alpha, time_deadline):
    h = start.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for alpha, steps in zip(alphas, steps_per_alpha):
        lr = 0.05 / np.sqrt(alpha)  # heuristic
        for _ in range(steps):
            if time.time() > time_deadline:
                break
            t += 1
            g = grad_soft_c5(h, alpha)
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            # project
            h = project_to_feasible(h, target)
    final_c5 = compute_c5(h)
    return h, final_c5

def project_to_feasible(v, target):
    w = np.clip(v, 0.0, 1.0)
    # adjust sum to target
    # Use while loop at most 10 iterations
    for _ in range(20):
        s = w.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to decrease sum
            # indices where w>0
            idx = w > 0
            if not np.any(idx):
                break
            delta = diff / idx.sum()
            w[idx] -= delta
            w = np.clip(w, 0.0, 1.0)
        else:
            idx = w < 1
            if not np.any(idx):
                break
            delta = -diff / idx.sum()
            w[idx] += delta
            w = np.clip(w, 0.0, 1.0)
    return w

def grad_soft_c5(h, alpha):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(alpha * conv)
    expa = np.exp(alpha * conv - maxc)
    w = expa / np.sum(expa)
    grad = np.zeros_like(h)
    # offset range
    for offset in range(-n+1, n):
        wi = w[offset + n - 1]
        if wi == 0:
            continue
        if offset >= 0:
            # index slice definitions
            grad[:n - offset] += wi * (1 - h[offset:])
            grad[offset:] -= wi * h[:n - offset]
        else:
            d = -offset
            grad[d:] += wi * (1 - h[:n - d])
            grad[:n - d] -= wi * h[d:]
    return grad

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
from functools import partial

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max overlap using log‑sum‑exp."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    # stabilize exponentials
    m = np.max(alpha * conv)
    expa = np.exp(alpha * conv - m)
    return float((np.log(np.sum(expa)) + m) / alpha)


def grad_soft_c5(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the softmax approximation w.r.t. h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(alpha * conv)
    expa = np.exp(alpha * conv - m)
    w = expa / expa.sum()                     # softmax weights, length 2n‑1
    grad = np.zeros_like(h)

    # offsets run from -(n‑1) … (n‑1)
    for offset in range(-n + 1, n):
        wi = w[offset + n - 1]
        if wi == 0.0:
            continue
        if offset >= 0:
            # (1‑h[j+offset]) term
            grad[: n - offset] += wi * (1.0 - h[offset:])
            # –h[j‑offset] term
            grad[offset:] -= wi * h[: n - offset]
        else:
            d = -offset
            grad[d:] += wi * (1.0 - h[: n - d])
            grad[: n - d] -= wi * h[d:]
    return grad


def sample_midpoints(n: int) -> np.ndarray:
    """Mid‑point x‑coordinates for n samples on [0,2]."""
    return np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n


def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the capped simplex 0≤h≤1, Σh = target."""
    w = np.clip(v, 0.0, 1.0)
    # iterative water‑filling / draining
    for _ in range(20):
        s = w.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:                       # need to decrease
            idx = w > 0.0
            if not np.any(idx):
                break
            delta = diff / idx.sum()
            w[idx] -= delta
            w = np.clip(w, 0.0, 1.0)
        else:                               # need to increase
            idx = w < 1.0
            if not np.any(idx):
                break
            delta = -diff / idx.sum()
            w[idx] += delta
            w = np.clip(w, 0.0, 1.0)
    # final rescaling if still off (rare)
    s = w.sum()
    if abs(s - target) > 1e-9:
        w = w * (target / s)
        w = np.clip(w, 0.0, 1.0)
    return w


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new resolution, keeping the integral."""
    old_n = arr.shape[0]
    old_x = sample_midpoints(old_n)
    new_x = sample_midpoints(new_n)
    up = np.interp(new_x, old_x, arr)
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    up = project_to_feasible(up, target)
    return up


def random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with the required sum."""
    a = rng.random(n)
    a = a * (target / a.sum())
    a = np.clip(a, 0.0, 1.0)
    a = project_to_feasible(a, target)
    return a


def build_from_func(func, n: int) -> np.ndarray:
    """Sample a continuous profile, scale to the exact integral."""
    x = sample_midpoints(n)
    y = func(x)
    y = np.clip(y, 0.0, 1.0)
    target = n / 2.0
    if y.sum() == 0.0:
        y = np.full(n, target / n)
    else:
        y = y * (target / y.sum())
    y = np.clip(y, 0.0, 1.0)
    y = project_to_feasible(y, target)
    return y


def grad_descent_opt(start: np.ndarray, target: float, deadline: float) -> np.ndarray:
    """Adam‑style optimisation of the softmax objective."""
    h = start.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    alphas = (5.0, 15.0, 30.0, 60.0)
    steps_per = (200, 200, 200, 200)          # total ≈ 800 updates
    for alpha, steps in zip(alphas, steps_per):
        lr = 0.05 / math.sqrt(alpha)          # heuristic learning‑rate schedule
        for _ in range(steps):
            if time.time() > deadline:
                return h
            t += 1
            g = grad_soft_c5(h, alpha)
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = project_to_feasible(h, target)
    return h


def slsqp_refine(h: np.ndarray, n: int, target: float, deadline: float) -> np.ndarray:
    """Two‑stage SLSQP polishing using increasingly sharp softmax."""
    def eq_con(x):
        return np.sum(x) - target

    cons = {"type": "eq", "fun": eq_con}
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    for alpha in (5.0, 30.0, 100.0):
        if time.time() > deadline:
            break
        obj = partial(softmax_c5, alpha=alpha)
        res = minimize(obj, cur, method="SLSQP", bounds=bounds,
                       constraints=cons,
                       options={"maxiter": 500, "ftol": 1e-9, "disp": False})
        if not res.success:
            break
        cur = np.clip(res.x, 0.0, 1.0)
        cur = project_to_feasible(cur, target)
    return cur
