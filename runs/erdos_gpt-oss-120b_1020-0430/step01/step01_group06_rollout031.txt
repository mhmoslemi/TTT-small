def project_to_sum_and_bounds(v, target, eps=1e-12, max_iter=100):
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)

import numpy as np
import random

def project_to_sum(v, target):
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max() * dx)

def anneal_once(h, target_sum, dx, time_limit):
    n = h.shape[0]
    cur_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5
    T = max(1e-5, cur_c5 * 0.02)
    decay = 0.9995
    it = 0
    while time.time() < time_limit:
        if it % 200 == 0:
            corr_full = np.correlate(h, 1 - h, mode='full')
            s_idx = int(np.argmax(corr_full))
            s = s_idx - (n - 1)
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:
                s_abs = -s
                i_inds = np.arange(s_abs, n)
                p_inds = i_inds + s
            contrib = h[i_inds] * (1.0 - h[p_inds])
            k = int(np.argmax(contrib))
            i_best = i_inds[k]
            p_best = p_inds[k]
            eps_max = min(h[i_best], 1.0 - h[p_best])
            if eps_max > 1e-12:
                eps = eps_max * random.random()
                nh = h.copy()
                nh[i_best] -= eps
                nh[p_best] += eps
                new_c5 = compute_c5(nh, dx)
                if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
                    h = nh
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
        i, j = random.sample(range(n), 2)

import numpy as np, random, math, time


def _proj_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] with sum(v)=target (capped simplex)."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _c5(h: np.ndarray, dx: float) -> float:
    """C₅ for a given vector h (dx = 2/n)."""
    return float(np.correlate(h, 1 - h, mode="full").max() * dx)


def _anneal_once(h: np.ndarray, target_sum: float, dx: float,
                 deadline: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing optimisation starting from h."""
    n = h.shape[0]
    cur = _c5(h, dx)
    best_h, best_val = h.copy(), cur
    T = max(1e-5, cur * 0.05)        # initial temperature
    decay = 0.9995
    it = 0
    while time.time() < deadline:
        # ----- focused move on the worst shift -----
        if it % 200 == 0:
            corr_full = np.correlate(h, 1 - h, mode="full")
            s_idx = int(np.argmax(corr_full))
            s = s_idx - (n - 1)
            if s >= 0:
                i_inds = np.arange(n - s)
                p_inds = i_inds + s
            else:
                s_abs = -s
                i_inds = np.arange(s_abs, n)
                p_inds = i_inds + s
            contrib = h[i_inds] * (1.0 - h[p_inds])
            k = int(np.argmax(contrib))
            i_best, p_best = i_inds[k], p_inds[k]
            eps_max = min(h[i_best], 1.0 - h[p_best])
            if eps_max > 1e-12:
                eps = eps_max * random.random()
                nh = h.copy()
                nh[i_best] -= eps
                nh[p_best] += eps
                new = _c5(nh, dx)
                if new < cur or random.random() < math.exp(-(new - cur) / T):
                    h, cur = nh, new
                    if new < best_val:
                        best_val, best_h = new, nh.copy()
        # ----- random two‑coordinate swap preserving the sum -----
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        max_up = min(1.0 - hi, hj)      # increase i, decrease j
        max_down = min(hi, 1.0 - hj)    # decrease i, increase j
        if max_up + max_down != 0:
            delta = (random.random() * (max_up + max_down)) - max_down
            if abs(delta) > 1e-14:
                nh = h.copy()
                nh[i] = hi + delta
                nh[j] = hj - delta
                new = _c5(nh, dx)
                if new < cur or random.random() < math.exp(-(new - cur) / T):
                    h, cur = nh, new
                    if new < best_val:
                        best_val, best_h = new, nh.copy()
        it += 1
        if it % 1000 == 0:
            T *= decay
    return best_h, best_val


def _upscale_initial(base: np.ndarray, new_n: int,
                     target_sum: float) -> np.ndarray:
    """Resize an initial vector (by linear interpolation) to length new_n."""
    m = base.shape[0]
    if m == new_n:
        up = base.copy()
    else:
        x_old = np.linspace(0, 2, m, endpoint=False)
        x_new = np.linspace(0, 2, new_n, endpoint=False)
        up = np.interp(x_new, x_old, base)
    return _proj_to_sum(up, target_sum)


def _refine(h: np.ndarray, dx: float, rounds: int = 2000) -> tuple[np.ndarray, float]:
    """Greedy post‑processing: repeatedly transfer as much mass as possible."""
    n = h.shape[0]
    best_h, best_val = h.copy(), _c5(h, dx)
    for _ in range(rounds):
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        s_idx = int(np.argmax(corr))
        s = s_idx - (n - 1)
        if s >= 0:
            i_inds = np.arange(n - s)
            p_inds = i_inds + s
        else:
            s_abs = -s
            i_inds = np.arange(s_abs, n)
            p_inds = i_inds + s
        contrib = best_h[i_inds] * (1.0 - best_h[p_inds])
        k = int(np.argmax(contrib))
        i_best, p_best = i_inds[k], p_inds[k]
        eps = min(best_h[i_best], 1.0 - best_h[p_best])
        if eps < 1e-12:
            continue
        nh = best_h.copy()
        nh[i_best] -= eps
        nh[p_best] += eps
        new = _c5(nh, dx)
        if new < best_val - 1e-12:
            best_h, best_val = nh, new
    return best_h, best_val
