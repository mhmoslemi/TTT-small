def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    ips = np.arange(n)
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # compute available increase/decrease per index
        if diff > 0:
            # need increase: indices where h < 1
            avail = 1.0 - h
            total_avail = avail.sum()
            if total_avail < 1e-12:
                break
            # distribute diff proportionally, but cap at avail
            # we can just add diff * (avail/total_avail), but may exceed per element.
            add = diff * (avail / total_avail)
            # cap each at avail (i.e., h + add <= 1)
            over = np.maximum(add - avail, 0)
            if over.any():
                # reduce over to zero by scaling diff
                # fallback to random single index method
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                h += add
        else:
            # need decrease: indices where h > 0
            avail = h
            total_avail = avail.sum()
            if total_avail < 1e-12:
                break
            dec = -diff * (avail / total_avail)
            over = np.maximum(dec - avail, 0)
            if over.any():
                i = rng.integers(0, n)
                add = max(diff, -h[i])
                h[i] += add
            else:
                h -= dec
        diff = target - h.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Project h onto {0≤h_i≤1, sum h_i = n/2}."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(5):
        if diff > 0:
            cand = np.where(h < 1 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -h[i])
            h[i] += add
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def gradient_smoothmax(h, beta):
    """Gradient of (1/β) * log Σ exp(β·overlap) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)  # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :] -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :] += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g


# ---------- worst‑shift sub‑gradient ----------
def worst_shift_gradient(h):
    """Sub‑gradient using the currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :] -= h[: n - k]
        else:
            kk = -k
            g[kk :] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g


# ---------- SLSQP smooth‑max polish ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary SA hill‑climb ----------
def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill‑climb on a binary vector."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values", None)
    base_n = len(parent) if parent is not None else 720
    parent_interp = None if parent is None else np.array(parent, dtype=float)

    # ---- multiple restarts on base resolution ----
    for restart in range(30):
        if time.time() > deadline:
            break
        n = base_n
        if restart == 0 and parent_interp is not None:
            h = parent_interp.copy()
        else:
            h = rng.random(n)
            h = h * ((n / 2.0) / h.sum())
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)

        # smooth‑max gradient descent
        betas = [10, 50, 200, 1000, 5000, 20000, 50000, 100000, 200000]
        for beta in betas:
            if time.time() > deadline:
                break
            for _ in range(30):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h, beta)
                if np.linalg.norm(g) < 1e-14:
                    continue
                step = 0.22
                best_local_h = h
                best_local_c = compute_c5(h)
                for mult in (1.0, 0.5, 0.25, 2.0):
                    cand = h - step * mult * g
                    np.clip(cand, 0.0, 1.0, out=cand)
                    adjust_sum_exact(cand, rng)
                    c = compute_c5(cand)
                    if c < best_local_c:
                        best_local_c = c
                        best_local_h = cand
                h = best_local_h

        # worst‑shift refinement (many iterations)
        step = 0.12
        for _ in range(2000):
            if time.time() > deadline:
                break
            g = worst_shift_gradient(h)
            if np.linalg.norm(g) < 1e-14:
                break
            best_local_h = h
            best_local_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0):
                cand = h - step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_local_c:
                    best_local_c = c
                    best_local_h = cand
            h = best_local_h
            step *= 0.995

        # final SLSQP polishing with very high β
        for beta in (20000, 50000, 100000, 200000):
            if time.time() > deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # binary SA hill‑climb on rounded version
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # ---- up‑sample and refine if time permits ----
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 1440
    ):
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        # small random perturbation to break exact duplication
        eps = 0.001
        h_up += eps * rng.random(n_up)
        np.clip(h_up, 0.0, 1.0, out=h_up)
        adjust_sum_exact(h_up, rng)

        betas_up = [10, 50, 200, 1000, 5000, 20000, 50000, 100000]
        for beta in betas_up:
            if time.time() > deadline:
                break
            for _ in range(15):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h_up, beta)
                if np.linalg.norm(g) < 1e-14:
                    continue
                step = 0.15
                best_local_h = h_up
                best_local_c = compute_c5(h_up)
                for mult in (1.0, 0.5, 0.25, 2.0):
                    cand = h_up - step * mult * g
                    np.clip(cand, 0.0, 1.0, out=cand)
                    adjust_sum_exact(cand, rng)
                    c = compute_c5(cand)
                    if c < best_local_c:
                        best_local_c = c
                        best_local_h = cand
                h_up = best_local_h

        # worst‑shift refinement
        step = 0.08
        for _ in range(1000):
            if time.time() > deadline:
                break
            g = worst_shift_gradient(h_up)
            if np.linalg.norm(g) < 1e-14:
                break
            best_local_h = h_up
            best_local_c = compute_c5(h_up)
            for mult in (1.0, 0.5, 0.25, 2.0):
                cand = h_up - step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_local_c:
                    best_local_c = c
                    best_local_h = cand
            h_up = best_local_h
            step *= 0.995

        for beta in (20000, 50000, 100000, 200000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

        # binary SA on up‑sampled rounded version
        h_bin = np.rint(h_up).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.15)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n_up

    # ---- final safeguard ----
    if best_h is None:
        n = base_n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    np.clip(best_h, 0.0, 1.0, out=best_h)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
