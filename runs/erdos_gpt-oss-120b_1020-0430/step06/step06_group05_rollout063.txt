def compute_c5(h):
    n = h.size
    n_fft = 1 << ((2*n - 1).bit_length())
    H = np.fft.fft(h, n_fft)
    G = np.fft.fft((1 - h)[::-1], n_fft)
    corr = np.fft.ifft(H * G, n_fft).real[:2*n - 1]
    max_corr = corr.max()
    return max_corr * (2.0 / n)

import time
import numpy as np
from scipy.optimize import minimize

# ----- global vars for SLSQP ----- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    n_fft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n_fft)
    G = np.fft.fft((1 - h)[::-1], n_fft)
    corr = np.fft.ifft(H * G, n_fft).real[: 2 * n - 1]
    return corr.max() * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    while abs(diff) > 1e-12:
        if diff > 0:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            cap = 1.0 - h[i]
            delta = min(diff, cap)
            h[i] += delta
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            cap = h[i]
            delta = min(-diff, cap)
            h[i] -= delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise h and parameters ----- #
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        target = n / 2.0
    else:
        n = 1024
        target = n / 2.0
        h = rng.random(n)
        h = enforce_sum(h, target, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # FFT size enough for all convolutions (3n‑2 length)

    beta = 100.0
    beta_max = 20000.0

    iteration = 0
