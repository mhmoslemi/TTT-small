def greedy_step(h):
    n = h.size
    g = worst_gradient(h)
    # indices for decreasing and increasing
    inc_idx = np.where(g < 0)[0]
    dec_idx = np.where(g > 0)[0]
    inc_cap = np.sum(1 - h[inc_idx])
    dec_cap = np.sum(h[dec_idx])
    M = min(inc_cap, dec_cap)
    if M <= 1e-12:
        return h.copy()
    # allocate increments
    # sort inc_idx by g ascending (most negative first)
    if inc_idx.size > 0:
        inc_order = inc_idx[np.argsort(g[inc_idx])]
    else:
        inc_order = np.array([], dtype=int)
    # sort dec_idx by g descending (most positive first)
    if dec_idx.size > 0:
        dec_order = dec_idx[np.argsort(-g[dec_idx])]
    else:
        dec_order = np.array([], dtype=int)
    d = np.zeros_like(h)
    remaining = M
    for i in inc_order:
        cap = 1 - h[i]
        if cap <= 0:
            continue
        delta = cap if cap < remaining else remaining
        d[i] = delta
        remaining -= delta
        if remaining <= 1e-12:
            break
    remaining = M
    for j in dec_order:
        cap = h[j]
        if cap <= 0:
            continue
        delta = cap if cap < remaining else remaining
        d[j] = -delta
        remaining -= delta
        if remaining <= 1e-12:
            break
    # ensure sum(d) approx 0
    # apply
    h_new = h + d
    # due to floating errors, may slightly violate bounds, correct
    np.clip(h_new, 0.0, 1.0, out=h_new)
    # enforce sum precisely
    h_new = enforce_sum(h_new, rng)   # need rng; pass as argument
    return h_new

def random_perturb(h, rng, scale):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1 - h[j]) if rng.random() < 0.5 else min(h[j], 1 - h[i])
    if max_eps <= 0:
        return h
    eps = rng.random() * max_eps * scale
    h_new = h.copy()
    if rng.random() < 0.5:
        h_new[i] -= eps
        h_new[j] += eps
    else:
        h_new[i] += eps
        h_new[j] -= eps
    # ensure bounds
    np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new
```

import time
import numpy as np

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / N)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)=N/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] = 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    return grad


def greedy_step(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move as much mass as possible from positive‑gradient to negative‑gradient entries."""
    N = h.size
    g = worst_gradient(h)
    inc_idx = np.where(g < 0)[0]
    dec_idx = np.where(g > 0)[0]
    inc_cap = np.sum(1.0 - h[inc_idx]) if inc_idx.size else 0.0
    dec_cap = np.sum(h[dec_idx]) if dec_idx.size else 0.0
    M = min(inc_cap, dec_cap)
    if M <= 1e-12:
        return h.copy()
    # order for maximal gain
    inc_order = inc_idx[np.argsort(g[inc_idx])] if inc_idx.size else np.array([], dtype=int)
    dec_order = dec_idx[np.argsort(-g[dec_idx])] if dec_idx.size else np.array([], dtype=int)
    d = np.zeros_like(h)
    rem = M
    for i in inc_order:
        cap = 1.0 - h[i]
        if cap <= 0:
            continue
        delta = cap if cap < rem else rem
        d[i] = delta
        rem -= delta
        if rem <= 1e-12:
            break
    rem = M
    for j in dec_order:
        cap = h[j]
        if cap <= 0:
            continue
        delta = cap if cap < rem else rem
        d[j] = -delta
        rem -= delta
        if rem <= 1e-12:
            break
    h_new = h + d
    np.clip(h_new, 0.0, 1.0, out=h_new)
    h_new = enforce_sum(h_new, rng)
    return h_new


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Move ε between two random coordinates (preserving sum)."""
    N = h.size
    i = rng.integers(N)
    j = rng.integers(N - 1)
    if j >= i:
        j += 1
    if rng.random() < 0.5:
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h
        eps = rng.random() * max_eps * scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
    else:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h
        eps = rng.random() * max_eps * scale
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
    np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new


def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simple continuous SA polishing on the feasible set."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps

        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

            if c_new < best_c:
