def simulated_annealing(h0, budget_s, rng):
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    c5_full = compute_c5_full(h)
    cur_max = c5_full.max()
    best_h = h.copy()
    best_max = cur_max
    K = np.arange(-n+1, n)
    pad_len = n-1
    h_pad = np.concatenate([np.zeros(pad_len), h, np.zeros(pad_len)])
    start = time.time()
    # SA parameters
    T0 = 0.01
    T = T0
    decay = 0.9995
    T_end = 1e-7
    iter_cnt = 0
    while time.time() - start < budget_s:
        iter_cnt += 1
        # select i, j
        i = rng.integers(0, n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if h[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # epsilon scaling
        scale = max(0.01, min(0.1, T / T0 * 5.0))
        epsilon = max_eps * (rng.random()**2) * scale
        if epsilon <= 0.0:
            continue
        base_i = pad_len + i
        base_j = pad_len + j
        term = (h_pad[base_i + K] - h_pad[base_j + K]) + (h_pad[base_i - K] - h_pad[base_j - K])
        new_c5_full = c5_full + epsilon * term * dx
        new_max = new_c5_full.max()
        delta = new_max - cur_max
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with prob exp(-delta/T)
            if T > 1e-12 and np.log(rng.random()) < -delta / T:
                accept = True
        if accept:
            # update state
            h[i] -= epsilon
            h[j] += epsilon
            h_pad[base_i] = h[i]
            h_pad[base_j] = h[j]
            c5_full = new_c5_full
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        # temperature decay
        T = max(T * decay, T_end)
    return best_h, best_max
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

    elapsed = time.time() - start

    sa_time = max(0.0, remaining * 0.80)

        if c_sa < best_c:

    # remaining time after SA

    remaining = budget_s - elapsed
    # subgradient descent
    if remaining > 5.0:
        sub_time = remaining * 0.70
        h_sg, c_sg = subgradient_optimize(best_h, n/2.0, sub_time, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg
    # remaining time after subgradient
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    # guided hill climb
    if remaining > 5.0:
        max_iters = int(min(5000, remaining * 300))
        h_hc, c_hc = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc
    # final sanitization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

def compute_c5_full(h: np.ndarray) -> np.ndarray:
    n = h.size
    return np.correlate(h, 1 - h, mode="full") * (2.0 / n)

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def worst_gradient(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def simulated_annealing(h0: np.ndarray, budget_s: float, rng: np.random.Generator):
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    c5_full = compute_c5_full(h)
    cur_max = c5_full.max()
    best_h = h.copy()
    best_max = cur_max
    K = np.arange(-n + 1, n)
    pad = n - 1
    h_pad = np.concatenate([np.zeros(pad), h, np.zeros(pad)])
    start = time.time()
    T0 = 0.01
    T = T0
    decay = 0.9995
    T_end = 1e-7
    while time.time() - start < budget_s:
        i = rng.integers(0, n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if h[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        scale = max(0.01, min(0.1, T / T0 * 5.0))
        eps = max_eps * (rng.random() ** 2) * scale
        if eps <= 0.0:
            continue
        base_i = pad + i
        base_j = pad + j
        term = (h_pad[base_i + K] - h_pad[base_j + K]) + (h_pad[base_i - K] - h_pad[base_j - K])
        new_c5_full = c5_full + eps * term * dx
        new_max = new_c5_full.max()
        delta = new_max - cur_max
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 1e-12 and np.log(rng.random()) < -delta / T:
                accept = True
        if accept:
            h[i] -= eps
            h[j] += eps
            h_pad[base_i] = h[i]
            h_pad[base_j] = h[j]
            c5_full = new_c5_full
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        T = max(T * decay, T_end)
    return best_h, best_max

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = random_feasible(n, rng)

    best_h, best_c = h0, compute_c5(h0)

    # Phase 1: Simulated annealing (≈80% of time)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    sa_time = max(0.0, remaining * 0.80)
    if sa_time > 5.0:
        h_sa, c_sa = simulated_annealing(best_h, sa_time, rng)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # Phase 2: Sub‑gradient descent
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        sub_time = remaining * 0.70
        h_sg, c_sg = subgradient_optimize(best_h, best_h.size / 2.0, sub_time, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # Phase 3: Guided hill‑climb
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        max_iters = int(min(5000, remaining * 300))
        h_hc, c_hc = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # Phase 4: Optional up‑sampling and final polish
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 15.0 and best_h.size * 2 <= 1024:
        h_up = upsample(best_h, rng)
        sub_time = remaining * 0.50
        h_up_sg, c_up_sg = subgradient_optimize(h_up, h_up.size / 2.0, sub_time, rng)
        if c_up_sg < best_c:
            best_h, best_c = h_up_sg, c_up_sg
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining > 5.0:
            max_iters = int(min(4000, remaining * 250))
            h_up_hc, c_up_hc = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.6)
            if c_up_hc < best_c:
                best_h, best_c = h_up_hc, c_up_hc

    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
