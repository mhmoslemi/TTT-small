def continuous_sa(h, rng, time_budget, T0=0.02, T_end=1e-7):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    N = h.size
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # propose moving mass
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        # choose direction: move from i to j or j to i: random
        if rng.random() < 0.5:
            # move delta from i to j
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * (T / T0)  # scale
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * (T / T0)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        c_new = compute_c5(h_new)
        c_curr = compute_c5(h)  # might recompute each time; expensive - we can store 
        # but N=2048 compute_c5 is cheap; we can compute each time.
        delta_c = c_new - c_curr
        if delta_c <= 0 or rng.random() < np.exp(-delta_c / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def gradient_descent_refine(h, rng, steps=50, init_step=0.02, decay=0.9):
    n = h.size
    target = n/2
    step = init_step
    for it in range(steps):
        # compute worst shift gradient
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad = 1 - 2*h
        else:
            grad[: n - k] = 1 - h[k:]
            grad[k:] -= h[: n - k]
        # gradient descent step
        h = h - step * grad
        # clip and enforce sum
        h = np.clip(h, 0.0, 1.0)
        # adjust sum: compute diff and add to random index
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # add or subtract diff to a random index with capacity
            # Loop until diff ~0
            for _ in range(1000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = target - h.sum()
        step *= decay
    return h
```

def continuous_sa(h, rng, time_budget, T0=0.02, T_end=1e-7):
    start = time.time()
    N = h.size
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        # pick two distinct indices
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        # direction
        if rng.random() < 0.5:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:

            delta = rng.random() * max_delta * (T / T0)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0.0:

            h_new = h.copy()

            h_new[j] -= delta

            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

def gradient_descent_refine(h, rng, steps=500, init_step=0.02, decay=0.95):
    n = h.size
    target = n / 2.0
    step = init_step
    for _ in range(steps):
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # compute gradient
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.zeros_like(h)
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        # gradient step
        h = h - step * grad
        # clip and enforce sum
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust diff in a few steps
            for _ in range(1000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    if add > 0:
                        h[i] += add
                else:
                    sub = min(-diff, h[i])
                    if sub > 0:
                        h[i] -= sub
                diff = target - h.sum()
        step *= decay
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff = target - h.sum()
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def continuous_sa(
    h: np.ndarray,
    rng: np.random.Generator,
    time_budget: float,
    T0: float = 0.1,
    T_end: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """Simulated annealing on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    N = h.size
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta * (T / T0)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta * (T / T0)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        c_new = compute_c5(h_new)
        dC = c_new - c_curr
        if dC <= 0.0 or rng.random() < np.exp(-dC / max(T, 1e-12)):

            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c


def gradient_descent_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    steps: int = 500,
    init_step: float = 0.02,
    decay: float = 0.95,

    """Projected gradient descent on the worst shift."""
    n = h.size

    step = init_step
