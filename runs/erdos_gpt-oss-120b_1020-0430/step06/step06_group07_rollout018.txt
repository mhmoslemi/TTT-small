    corr = np.correlate(x, 1-x, mode='full')
    bc = beta * corr
    m = bc.max()
    expb = np.exp(bc - m)
    sumexp = expb.sum()
    w = expb / sumexp
    # forward/backward via convolution
    w_rev = w[::-1]
    conv_back = np.convolve(w, x, mode='full')
    conv_for = np.convolve(w_rev, x, mode='full')
    idx = x.size - 1
    back = conv_back[idx: idx + x.size]
    forward = conv_for[idx: idx + x.size]
    grad = 1.0 - forward - back
    obj = (m + np.log(sumexp)) / beta
    return obj, grad

def slsqp_softmax_optimize(x0, beta, target, max_iter):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x)-target}
    def fun(x):
        o, g = softmax_obj_and_jac(x, beta)
        return o
    def jac(x):
        _, g = softmax_obj_and_jac(x, beta)
        return g
    res = minimize(fun, x0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = project_box_sum(x, target)
    return x

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the simplex with box constraints."""
