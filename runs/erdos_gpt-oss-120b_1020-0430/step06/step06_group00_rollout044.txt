import time, numpy as np
from scipy.optimize import minimize

# ---------- core helpers ----------
def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(20000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


# ---------- SLSQP soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1200) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                             "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- gradients ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
    return g


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- binary SA ----------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = rng.integers(len(ones_arr))
            j = rng.integers(len(zeros_arr))
            ii, jj = ones_arr[i], zeros_arr[j]
            h[ii] = 0.0
            h[jj] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i] = jj
                zeros_arr[j] = ii
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
            else:
                h[ii] = 1.0
                h[jj] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


# ---------- upsample ----------
def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ---------- continuous SA ----------
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float,
                  T0: float = 0.02, T_end: float = 1e-7, scale_coeff: float = 0.5):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = scale_coeff * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c

        c_fine = compute_c5(h_fine)
        if c_fine < best_c:
            best_h, best_c, best_n = h_fine, c_fine, 1024
        else:
            best_n = best_h.size

    # Phase 5: continuous SA on finest grid
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        # give most of remaining time to SA
        t_sa = remaining * 0.9
        h_sa, c_sa = continuous_sa(best_h, rng, t_sa,
                                   T0=0.02, T_end=1e-7, scale_coeff=0.5)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

    # Phase 6: final short gradient polish
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        h = gradient_refine(best_h, rng, step=0.015, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
