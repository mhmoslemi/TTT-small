def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None; best_c = np.inf; best_n = None

    # resolution list
    resolutions = [360, 720, 1080, 1440]

    parent = globals().get('initial_h_values', None)

    for n in resolutions:
        # check time
        if time.time() > deadline:
            break

        # Determine max restarts based on n
        if n <= 360:
            max_restarts = 30
        elif n <= 720:
            max_restarts = 20
        elif n <= 1080:
            max_restarts = 15
        else:
            max_restarts = 10

        for r in range(max_restarts):
            if time.time() > deadline:
                break

            # initialization
            if r == 0 and parent is not None and len(parent) == n:
                h = np.array(parent, copy=True)
            elif parent is not None and len(parent) > n:
                # downsample via interpolation
                xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(xt, xs, parent)
            elif parent is not None and len(parent) < n:
                # upsample via repeat
                repeat_factor = n // len(parent)
                remainder = n - repeat_factor * len(parent)
                h = np.repeat(parent, repeat_factor)
                if remainder:
                    h = np.concatenate([h, parent[:remainder]])
            else:
                # random init
                h = rng.random(n)
                h = h / h.sum() * (n / 2.0)
                np.clip(h, 0.0, 1.0, out=h)

            adjust_sum_exact(h, rng)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def binary_fix_sum(h_bin, rng):
    n = h_bin.size
    target = n // 2
    cur = int(h_bin.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h_bin == 0)
        if zeros.size:
            choose = rng.choice(zeros, size=diff, replace=False)
            h_bin[choose] = 1
    elif diff < 0:
        ones = np.flatnonzero(h_bin == 1)
        if ones.size:
            choose = rng.choice(ones, size=-diff, replace=False)
            h_bin[choose] = 0
    return h_bin

# ---------- gradients ----------
def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i:]      -= wi * h[: n - i]
        else:
            ii = -i
            g[ii:]     += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii:]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:]      -= h[: n - k]
        else:
            kk = -k
            g[kk:]     += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

# ---------- descent helpers ----------
def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_local_h = h
            best_local_c = compute_c5(h)
            for mult in (0.5, 0.75, 1.0, 1.5, 2.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_local_c:
                    best_local_c = c
                    best_local_h = cand
            h = best_local_h
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):

# ---------- SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    if not ones or not zeros:
        return cur_h, cur_c
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 20000:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    best_c = np.inf
    best_n = None

            max_restarts = 15
        else:
            max_restarts = 10

        for r in range(max_restarts):
            if time.time() > deadline:

            # ----- initialization -----
            if r == 0 and parent is not None and len(parent) == n:
                h = np.array(parent, copy=True, dtype=float)
            elif parent is not None and len(parent) > n:
                xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(xt, xs, parent)
            elif parent is not None and len(parent) < n:
                repeat = n // len(parent)
                rem = n - repeat * len(parent)
                h = np.repeat(parent, repeat)
                if rem:
                    h = np.concatenate([h, parent[:rem]])
                h = h.astype(float)
            else:
                h = rng.random(n)
                h = h / h.sum() * (n / 2.0)
                np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)

            # ----- smooth‑max gradient descent -----
            betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]

            )

            # ----- binary hill climbing -----
            h_bin = np.rint(h).astype(int)

        adjust_sum_exact(best_h, rng)
