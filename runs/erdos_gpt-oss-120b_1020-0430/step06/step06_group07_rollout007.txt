        h[j0] += eps

def compute_c5_corr_fft(h):
    n = h.shape[0]
    L = 2 * n
    h_fft = np.fft.rfft(h, n=L)
    one_minus_h_fft = np.fft.rfft(1.0 - h, n=L)

    # use first 2*n-1 entries
    corr = corr[:2*n-1]
    c5 = corr.max() * (2.0 / n)
    return c5, corr

                max_diff = d.max()
                if max_diff > 1e-12:
                    idx = int(np.argmax(d))
                    i0 = idx + s
                    j0 = idx
                    eps_max = min(h[i0], 1.0 - h[j0], max_diff * 0.5)
                    eps = eps_max * 0.9
                else:
                    i0 = rng.integers(0, n)
                    j0 = rng.integers(0, n)
                    eps_max = min(h[i0], 1.0 - h[j0])
                    eps = eps_max * 0.5

        if eps <= 0.0:
            continue

        h[i0] -= eps
        h[j0] += eps
        c5_new, corr_new = compute_c5_corr_fft(h)
        delta = c5_new - c5
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            c5 = c5_new
            corr = corr_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i0] += eps
            h[j0] -= eps
    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024  # target resolution

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = upsample_to_n(h0, n)
    else:
        h0 = random_feasible(n, rng)

    remaining = budget_s - (time.time() - start)
    if remaining < 5.0:
        final_c5 = compute_c5_corr_fft(h0)[0]
        return h0.tolist(), float(final_c5), n

    best_h, best_c5 = simulated_annealing_optimize(h0, remaining, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    final_c5 = compute_c5_corr_fft(best_h)[0]
    return best_h.tolist(), float(final_c5), int(n)
