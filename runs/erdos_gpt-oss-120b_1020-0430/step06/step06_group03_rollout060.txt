    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H * G, L)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def worst_gradient_fast(h):
    n = h.size
    # compute correlation using FFT; also get index of max
    n = h.size
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H * G, L)[:2*n - 1]
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        k_abs = abs(k)
        if k > 0:
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]
        else:
            # for negative k, same effect using absolute
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]
    return grad, corr

import time, math, numpy as np

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5_fast(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * G, L)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def worst_gradient_fast(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * G, L)[: 2 * n - 1]
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        kk = abs(k)
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return grad, corr

def run_sa(h, time_slice, rng):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    cur_h = h.copy()
    cur_c = best_c
    T0 = 0.12
    T_final = 1e-6
    start = time.time()
    while time.time() - start < time_slice:
        elapsed = time.time() - start
        t_frac = elapsed / time_slice
        T = T0 * (T_final / T0) ** t_frac
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random()
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5_fast(nh)
        if nc < cur_c or rng.random() < math.exp(-(nc - cur_c) / T):
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_c, best_h = nc, nh.copy()
    return best_h, best_c

def hill_climb(h, time_slice, rng):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    cur_h = h.copy()
    cur_c = best_c
    start = time.time()
    while time.time() - start < time_slice:
        grad, _ = worst_gradient_fast(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            # tiny random tweak to escape
            i = rng.integers(n)
            j = rng.integers(n)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0.0:
                continue
        # try a few epsilon candidates
        eps_vals = np.linspace(0, max_eps, 5)[1:]  # skip 0
        improved = False
        for eps in eps_vals:
            nh = cur_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5_fast(nh)
            if nc < cur_c:
                cur_h, cur_c = nh, nc
                improved = True
                if nc < best_c:
                    best_c, best_h = nc, nh.copy()
                break
        if not improved:
            # occasional random move to avoid stagnation
            i = rng.integers(n)
            j = rng.integers(n)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps > 0.0:
                eps = max_eps * rng.random()
                cur_h[i] -= eps
                cur_h[j] += eps
                cur_c = compute_c5_fast(cur_h)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # determine size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 512
        h0 = rng.random(n)
    target = n / 2.0
    h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5_fast(best_h)

    total_budget = budget_s
    # allocate 75% time to simulated annealing with several restarts
    sa_budget = total_budget * 0.75
    restarts = 5
    per_restart = sa_budget / restarts

    for _ in range(restarts):
        elapsed = time.time() - start
        if elapsed > total_budget * 0.9:
            break
        remaining = sa_budget - (elapsed - (total_budget - sa_budget))
        time_slice = min(per_restart, max(0.0, remaining))
        if time_slice < 1.0:
            break
        # random start for this restart
        h_start = rng.random(n)
        h_start = project_onto_box_sum(h_start, target)
        h_local, c_local = run_sa(h_start, time_slice, rng)
        if c_local < best_c:
            best_c, best_h = c_local, h_local.copy()

    # remaining time for fine hill‑climbing
    remaining_time = total_budget - (time.time() - start)
    if remaining_time > 1.0:
        h_final, c_final = hill_climb(best_h, remaining_time, rng)
        if c_final < best_c:
            best_c, best_h = c_final, h_final.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5_fast(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
