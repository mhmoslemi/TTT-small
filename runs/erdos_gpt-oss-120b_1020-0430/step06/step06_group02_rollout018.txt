def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = abs(idx)
    if k == 0:
        return 1 - 2*h
    grad = np.ones_like(h)
    # subtract h_{i+k}
    grad[: n - k] -= h[k:]
    # subtract h_{i-k}
    grad[k:] -= h[: n - k]
    return grad

def thue_morse_bits(n):
    seq = np.zeros(1, dtype=float)
    while seq.size < n:
        seq = np.concatenate([seq, 1 - seq])
    return seq[:n]

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP soft‑max
CURRENT_TARGET = None
CURRENT_BETA = None


def proj_feasible(x: np.ndarray, s: float) -> np.ndarray:
    """Project onto {0≤y≤1, sum(y)=s} using binary search on the Lagrange multiplier."""
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        total = y.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.zeros(1, dtype=float)
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_gradient_and_corr(h: np.ndarray):
    """Gradient for the worst shift and its correlation value."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, corr.max()


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 512                      # default
    target_sum = n / 2.0

    # ----- helper for projection ---------------------------------------
    def project(x):
        return proj_feasible(x, target_sum)

    # ----- build initial candidates --------------------------------------
    candidates = []

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project(h0)
        candidates.append(h0)

    tm = thue_morse_bits(n)
    candidates.append(project(tm))

    hb = np.zeros(n)
    hb[rng.choice(n, n // 2, replace=False)] = 1.0
    candidates.append(project(hb))

    ur = rng.random(n)
    candidates.append(project(ur))

    ga = rng.standard_normal(n)
    ga = (ga - ga.min()) / (ga.max() - ga.min())
    candidates.append(project(ga))

    best_h = None
    best_c = np.inf

    # ----- gradient descent with momentum --------------------------------
    time_for_gd = budget_s * 0.6
    max_iters_per = 5000
    mu = 0.8                     # momentum factor
    for h_start in candidates:
        h = h_start.copy()
        v = np.zeros_like(h)
        c5 = compute_c5(h)
        if c5 < best_c:
            best_c, best_h = c5, h.copy()
        step = 0.5
        decay = 0.995
        it = 0
        while it < max_iters_per and time.time() - start < time_for_gd:
            grad, _ = worst_gradient_and_corr(h)
            v = mu * v - step * grad
            h_new = project(h + v)
            c5_new = compute_c5(h_new)

            if c5_new < best_c:
                best_c, best_h = c5_new, h_new.copy()
            if c5_new < c5:
                h, c5 = h_new, c5_new
                # keep momentum
            else:
                # occasional momentum reset
                if rng.random() < 0.02:
                    v[:] = 0.0

            # random local tweak every 400 iterations
            if it % 400 == 0:
                i, j = rng.integers(0, n, size=2)
                eps = step * 0.3 * rng.random()
                h_t = h.copy()
                h_t[i] = np.clip(h_t[i] + eps, 0.0, 1.0)
                h_t[j] = np.clip(h_t[j] - eps, 0.0, 1.0)
                h_t = project(h_t)
                ct = compute_c5(h_t)
                if ct < best_c:
                    best_c, best_h = ct, h_t.copy()
                if ct < c5:
                    h, c5 = h_t, ct

            step *= decay
            it += 1

    # ----- SLSQP refinement with soft‑max -------------------------------
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target_sum

    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET

    def softmax_obj(x):
        N = 2 * n
        H = np.fft.fft(x, N)
        G = np.fft.fft(1.0 - x, N)
        corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
        bc = CURRENT_BETA * corr
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
        if time.time() - start > budget_s * 0.95:
            break
        CURRENT_BETA = beta
        res = minimize(
            softmax_obj,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 1500, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = proj_feasible(x, target_sum)
        c5 = compute_c5(x)
        if c5 < best_c:
            best_c, best_h = c5, x.copy()
            if best_c <= 0.3808:
                break

    # ----- final tiny‑step refinement ------------------------------------
    step = 0.03
    while time.time() - start < budget_s * 0.99:
        grad, _ = worst_gradient_and_corr(best_h)
        h_new = proj_feasible(best_h - step * grad, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c:
            best_c, best_h = c5, h_new.copy()
        step *= 0.9999
        if step < 1e-6:
            break

    best_h = proj_feasible(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
