import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient (subgradient) of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | 0≤x_i≤1, sum x_i = target}.
    Uses bisection on the Lagrange multiplier.
    """
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        idx = np.where((w > 0.0) & (w < 1.0))[0]
        if idx.size:
            i = idx[0]
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
    return w


def equalize_worst_shift(h: np.ndarray, max_iters: int,
                         time_budget: float, start_time: float) -> np.ndarray:
    """
    Repeatedly replace (h_i, h_{i+k}) by their average for the current worst shift k.
    Each averaging preserves the total sum.
    """
    n = h.size
    it = 0
    while it < max_iters and (time.time() - start_time) < time_budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            break
        avg = (h[: n - k] + h[k:]) * 0.5
        h[: n - k] = avg
        h[k:] = avg
        it += 1
    return h


def pgd_optimize(h0: np.ndarray, target_sum: float,
                 time_budget: float, start_time: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the worst‑shift subgradient.
    Stops when the allocated time is exhausted.
    Returns the best h seen and its C5 value.
    """
    n = h0.size
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h, best_c = h.copy(), c_cur
    step = 0.8
    no_improve = 0

    while (time.time() - start_time) < time_budget:
        grad = worst_gradient(h)
        h_new = h - step * grad
        h_new = project_box_sum(h_new, target_sum)
        c_new = compute_c5(h_new)

        if c_new <= c_cur + 1e-14:
            h, c_cur = h_new, c_new
            step = min(step * 1.05, 2.0)
        else:
            step = max(step * 0.5, 1e-5)

        if c_cur < best_c - 1e-12:
            best_c, best_h = c_cur, h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 300:
            # small random jitter to escape plateaus
            i = rng.integers(0, n)
            eps = (rng.random() - 0.5) * 0.2
            h[i] = np.clip(h[i] + eps, 0.0, 1.0)
            h = project_box_sum(h, target_sum)
            c_cur = compute_c5(h)
            no_improve = 0

    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using decreasing step sizes."""
    target = h.size / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # use 512 points (initial_h_values is 512 when provided)
    n = 512
    target_sum = n / 2.0

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, target_sum)
    else:
        h0 = rng.random(n)
        h0 = project_box_sum(h0, target_sum)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- Step 1: quick equalisation of the worst shift ----
    elapsed = time.time() - start
    remain = total_budget - elapsed
    eq_time = min(remain * 0.15, 120.0)          # at most 2 min
    h_eq = equalize_worst_shift(h0.copy(), max_iters=50,
                                time_budget=eq_time, start_time=start)
    c_eq = compute_c5(h_eq)
    if c_eq < best_c:
        best_h, best_c = h_eq.copy(), c_eq

    # ---- Step 2: projected gradient descent ----
    elapsed = time.time() - start
    remain = total_budget - elapsed
    pgd_time = remain * 0.75
    h_pgd, c_pgd = pgd_optimize(best_h.copy(), target_sum,
                                pgd_time, start, rng)
    if c_pgd < best_c:
        best_h, best_c = h_pgd.copy(), c_pgd

    # ---- Step 3: short gradient refinement ----
    elapsed = time.time() - start
    remain = total_budget - elapsed
    if remain > 0:
        h_ref = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=12)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref.copy(), c_ref

    # ---- Step 4: final tiny random swaps (hill‑climb style) ----
    while time.time() - start < total_budget * 0.99:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_h, best_c = nh, c_new

    # ---- Sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
