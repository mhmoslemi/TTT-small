import time
import numpy as np

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# --------------------------------------------------------------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Binary simulated annealing with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.05, 1e-6
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Iterative refinement using worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def subgradient_descent(h: np.ndarray, rng: np.random.Generator,
                        max_iter: int = 200, init_step: float = 0.02) -> np.ndarray:
    """Sub‑gradient descent with backtracking line search."""
    n = h.size
    target = n / 2.0
    c_curr = compute_c5(h)
    step = init_step
    for it in range(max_iter):
        g = worst_gradient(h)
        improved = False
        # try decreasing step sizes
        for _ in range(20):
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, rng)
            c_new = compute_c5(h_new)
            if c_new < c_curr - 1e-12:
                h, c_curr = h_new, c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # no improvement even with tiny step – stop
            break
        # optional small random shake to avoid stagnation
        if it % 30 == 0 and rng.random() < 0.2:
            eps = 0.02 * rng.random(n)
            signs = rng.choice([-1, 1], size=n)
            h = np.clip(h + signs * eps, 0.0, 1.0)
            h = enforce_sum(h, rng)
            c_curr = compute_c5(h)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    def consider(candidate):
        nonlocal best_h, best_c, best_n
        cand = np.clip(candidate, 0.0, 1.0)
        cand = enforce_sum(cand, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c, best_n = cand.copy(), c, cand.size

    # ---- start from parent if available ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        consider(h0)

    # ---- upsample to 1024 points ----
    if best_h is not None and best_n < 1024:
        consider(upsample_h(best_h, 1024))

    # ---- binary SA on 1024 ----
    remaining = budget_s - (time.time() - start)
    if remaining > 30:
        t_bin = min(remaining * 0.5, 500.0)
        h_bin, c_bin = binary_sa(1024, rng, t_bin)
        consider(h_bin)

    # ---- gradient and sub‑gradient polishing ----
    remaining = budget_s - (time.time() - start)
    if remaining > 20 and best_h is not None:
        # gradient refinement
        h = gradient_refine(best_h, rng, step=0.02, max_iter=30)
        consider(h)

    # ---- time‑boxed sub‑gradient descent with restarts ----
    end_time = start + budget_s
    while time.time() < end_time - 5:
        if best_h is None:
            break
        h_local = best_h.copy()
        # small random perturbation
        eps = 0.04 * rng.random(h_local.size)
        signs = rng.choice([-1, 1], size=h_local.size)
        h_local = np.clip(h_local + signs * eps, 0.0, 1.0)
        h_local = enforce_sum(h_local, rng)
        h_local = subgradient_descent(h_local, rng, max_iter=200, init_step=0.02)
        consider(h_local)

    # ---- fallback if nothing found ----
    if best_h is None:
        n = 512
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        best_h = enforce_sum(h, rng)
        best_c = compute_c5(best_h)

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
