def simulated_annealing(h, target, rng, budget_s, start_time):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # init temperature
    T0 = 0.02  # initial temperature
    iter = 0
    grad_interval = 200
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        progress = elapsed / budget_s
        # temperature schedule
        T = T0 * (0.001 ** progress)
        # move scale
        move_scale = max(0.01, (0.5 * (1 - progress) + 0.01))
        # propose random mass transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            continue
        epsilon = max_transfer * rng.random() * move_scale
        h_new = cur_h.copy()
        h_new[i] -= epsilon
        h_new[j] += epsilon
        # no need enforce sum
        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        # occasional gradient descent
        if iter % grad_interval == 0:
            # small gradient step
            steps = [0.03 * (1 - progress) + 0.001]

            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        iter += 1
    return best_h, best_c

import time
import numpy as np

# ---------- core utilities ----------

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function using FFT (fast)."""
    n = h.shape[0]
    m = 2 * n
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * np.conj(G), m)
    max_corr = np.max(conv[: 2 * n - 1])
    return max_corr * (2.0 / n)


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.shape[0]
    m = 2 * n
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * np.conj(G), m)
    conv_full = conv[: 2 * n - 1]
    idx = np.argmax(conv_full)
    k = idx - (n - 1)          # shift value
    grad = np.zeros_like(h)

    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:]    -= h[: n - k]
    else:                     # k < 0
        kk = -k
        grad[kk:]    += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return k, grad


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.shape[0]
    it = 0
    max_it = 10 * n
    while abs(diff) > 1e-12 and it < max_it:
        if diff > 0:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            cap = 1.0 - h[i]
            delta = min(diff, cap)
            h[i] += delta
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            cap = h[i]
            delta = min(-diff, cap)
            h[i] -= delta
        diff = target - h.sum()
        it += 1
    return np.clip(h, 0.0, 1.0)


def gradient_descent(h: np.ndarray, target: float, rng: np.random.Generator,
                    steps):
    """Apply a few gradient‑descent steps on the worst shift."""
    for s in steps:
        _, grad = worst_shift_and_grad(h)
        h = h - s * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    return h


def simulated_annealing(h: np.ndarray, target: float, rng: np.random.Generator,
                        budget_s: float, start_time: float):
    """Mass‑transfer SA with occasional gradient pulls."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    T0 = 0.02                     # initial temperature
    grad_interval = 200
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        prog = elapsed / budget_s
        T = T0 * (0.001 ** prog)                 # exponential cooling
        move_scale = max(0.005, 0.5 * (1 - prog) + 0.005)

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            it += 1
            continue
        eps = max_transfer * rng.random() * move_scale
        h_new = cur_h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

        if it % grad_interval == 0:
            step = 0.02 * (1 - prog) + 0.001
            cur_h = gradient_descent(cur_h, target, rng, [step])
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        it += 1

    return best_h, best_c


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 512
    target_sum = n / 2.0

    # ----- initial step function -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
    else:
        # Thue‑Morse binary pattern
        h0 = np.array([bin(i).count("1") % 2 for i in range(n)], dtype=float)
        h0 = enforce_sum(h0, target_sum, rng)

    # ----- early gradient polishing -----
    pre_steps = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]
    h = gradient_descent(h0, target_sum, rng, pre_steps)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing phase -----
    best_h, best_c = simulated_annealing(best_h, target_sum, rng,
                                         budget_s, start)

    # ----- final refinement -----
    final_steps = [0.02, 0.01, 0.005, 0.002, 0.001]
    best_h = gradient_descent(best_h, target_sum, rng, final_steps)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
