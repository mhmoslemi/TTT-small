import time, math, numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize

def compute_c5_fft(h):
    n = h.size
    corr = fftconvolve(h, np.flip(1.0 - h), mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target, tol=1e-12, max_iter=60):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > tol:
        mask = (x > tol) & (x < 1.0 - tol)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def softmax_grad_fft(h, beta):
    n = h.size
    # correlation
    corr = fftconvolve(h, np.flip(1.0 - h), mode='full')
    # softmax weights
    m = corr.max()
    exp_term = np.exp(beta * (corr - m))
    w = exp_term / exp_term.sum()
    w_center = w[n - 1]
    w_pos = w[n:]            # shifts +k, k=1..n-1, length n-1
    w_neg = w[:n - 1]        # shifts -k, length n-1

    # gradient start
    grad = 1.0 - 2.0 * w_center * h

    # term1: sum_{k>=1} w_pos[k-1] * h[p+k]
    conv1 = fftconvolve(w_pos, h, mode='full')
    term1 = np.concatenate([conv1[n - 1:], np.zeros(1)])  # length n
    grad -= term1

    # term2: sum_{k>=1} w_neg[k-1] * h[p-k]
    h_rev = h[::-1]
    conv2 = fftconvolve(w_neg, h_rev, mode='full')
    # conv2 length = 2n-2, we need indices 2n-3 .. n-1
    term2 = np.empty(n)
    term2[0] = 0.0
    term2[1:] = conv2[2 * n - 3 : n - 2 : -1]
    grad -= term2
    return grad

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
