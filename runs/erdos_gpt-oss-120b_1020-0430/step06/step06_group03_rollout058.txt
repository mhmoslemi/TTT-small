def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        # project onto box sum exactly
        h = project_onto_box_sum(h, target)
    else:
        n = 256
        target = n / 2.0
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, n//2, replace=False)] = 1.0
    # correlation of h with itself
    corr = np.correlate(h, h, mode='full')
    # min correlation excluding zero shift
    min_corr = min(corr[:n-1].min(), corr[n:].min())
    best_h = h.copy()
    best_min_corr = min_corr
    # Precompute
    idx = np.arange(n, dtype=np.int64)
    m = 2 * n - 1
    inc = np.zeros(m, dtype=np.float64)
    # Parameters
    T0 = 5.0
    step0 = 0.5
    # Main loop
    while time.time() - start < budget_s * 0.95:
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = max(1e-12, (1.0 - frac) * T0)
        step = max(1e-4, (1.0 - frac) * step0)
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * step
        if eps <= 0.0:
            continue
        # compute inc
        inc.fill(0.0)
        temp = eps * h
        # slice assignments using precomputed idx slices
        inc[i:i+n] -= temp[::-1]
        inc[j:j+n] += temp[::-1]
        inc[n-1-i:n-1-i+n] -= temp
        inc[n-1-j:n-1-j+n] += temp
        eps2 = eps * eps
        inc[n-1] += 2.0 * eps2
        d = i - j
        if d != 0:
            inc[n-1 + d] -= eps2
            inc[n-1 - d] -= eps2
        new_corr = corr + inc
        # compute new min correlation
        slice1 = new_corr[:n-1]
        slice2 = new_corr[n:]
        min1 = slice1.min()
        min2 = slice2.min()
        new_min_corr = min1 if min1 < min2 else min2
        diff = new_min_corr - min_corr
        if diff > 0.0 or (diff < 0.0 and rng.random() < math.exp(diff / T)):
            # accept
            corr[:] = new_corr
            min_corr = new_min_corr
            h[i] -= eps
            h[j] += eps
            if min_corr > best_min_corr:
                best_min_corr = min_corr
                best_h = h.copy()
    # final evaluation
    def compute_c5(hv):
        nloc = hv.size
        corrv = np.correlate(hv, 1.0 - hv, mode='full')
        return np.max(corrv) * (2.0 / nloc)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
