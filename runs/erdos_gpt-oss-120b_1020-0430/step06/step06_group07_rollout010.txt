# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0 ≤ x ≤ 1, sum x = target} via binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the box with sum=n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst (largest) overlap O_k."""
    n = h.size
    # cross-correlation of h with itself, keep non‑negative lags
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))  # worst shift (max overlap)
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def stochastic_subgradient_optimize(

def softmax_obj(x, beta, target):
    """Smooth approximation of max_k O_k using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_refine(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int):
    """SLSQP on the smooth max objective."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    res = minimize(
        softmax_obj,
        h0,
        args=(beta, target),
        method="SLSQP",
        bounds=bounds,
        constraints=eq_cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution on random coordinates
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

        remaining = total_budget - (time.time() - start)
        if remaining > 4.0:
            phase4b_budget = remaining * 0.20
            h_det, c_det = deterministic_subgradient_optimize(
                best_h, n_up / 2.0, phase4b_budget, rng
            )
            if c_det < best_c:
                best_c, best_h = c_det, h_det.copy()
        # Phase 4C: final SLSQP on fine grid
        remaining = total_budget - (time.time() - start)
        if remaining > 2.0:
            for beta in (200.0, 800.0, 3000.0, 10000.0):
                if time.time() - start > total_budget * 0.99:
                    break
                h_sls = slsqp_refine(best_h, beta, rng, max_iter=200)
                c_sls = compute_c5(h_sls)
                if c_sls < best_c:
                    best_c, best_h = c_sls, h_sls.copy()
                if best_c <= 0.3805:
                    break

    # ---------------------------------------------------------
    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
