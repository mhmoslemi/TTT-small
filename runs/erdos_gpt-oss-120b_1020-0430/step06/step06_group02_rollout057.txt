def project_onto_box_sum(x, target):
    # project x onto [0,1] with sum = target, using bisection on lambda.
    low = -np.max(x) - 1  # ensure low enough
    high = 1 - np.min(x) + 1
    for _ in range(50):
        lam = (low + high) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return y

    # Using bisection for lambda.
    # Since sum of clipped changes monotonic decreasing with lambda.
    lam_low = -np.max(x) - 1.0  # ensures x - lam_low >= 1

def project_sum_box(x, target):

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / h.size) * np.max(corr)


def project_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1] while enforcing sum(x)=target."""
    low, high = -2.0, 2.0

        else:
            high = lam

    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (n - 1)  # signed shift
    grad = np.zeros_like(h)
    if s == 0:
        grad = 1.0 - 2.0 * h
    else:
        if s > 0:
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] -= h[: n - s]
        else:
            s_abs = -s
            grad[s_abs:] += 1.0 - h[: n - s_abs]
            grad[: n - s_abs] -= h[s_abs:]
    return grad


def grad_softmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the softmax (log‑sum‑exp) approximation of max_k O_k."""
    n = h.size
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    bc = beta * corr
    bc_max = np.max(bc)
    w = np.exp(bc - bc_max)
    w /= w.sum()
    w_rev = w[::-1]
    conv = np.convolve(w_rev, g, mode="full")
    grad = conv[n - 1 : n - 1 + n]
    weight0 = w[n - 1]                     # weight for shift 0
    grad -= weight0 * h                    # correct the s=0 term
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = project_sum_box(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- subgradient phase (quick reduction of max overlap) ----------
    subgrad_iters = int(2000 * (n / 512))
    for i in range(subgrad_iters):
        if time.time() - start > budget_s * 0.85:
            break
        grad = worst_gradient(h)
        step = 0.2 * (0.97 ** i)
        h = project_sum_box(h - step * grad, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break

    # ----- Adam on smoothed max (log‑sum‑exp) -------------------------
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 300.0, 1200.0, 5000.0, 20000.0]
    m = np.zeros(n)
    v = np.zeros(n)
    t_iter = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.1

    for beta in beta_schedule:
        if time.time() - start > budget_s * 0.98:
            break
        # a modest number of iterations per beta; stop early if good enough
        for _ in range(800):
            if time.time() - start > budget_s * 0.99:
                break
            t_iter += 1
            grad = grad_softmax(h, beta)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            lr = base_lr / (np.sqrt(beta) + 1e-12)
            h = project_sum_box(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
                if best_c <= 0.3808:
                    break
        if best_c <= 0.3808:
            break

    # ----- final fine‑grained subgradient polishing --------------------
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        step = 0.02
        best_h = project_sum_box(best_h - step * grad, target)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c
            if best_c <= 0.3808:
                break

    # ----- return best found -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
