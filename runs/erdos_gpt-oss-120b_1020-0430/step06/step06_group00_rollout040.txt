def greedy_descent_binary(h_bin, rng, max_iters, target_max_c):
    """h_bin is uint8 array with 0/1."""
    n = h_bin.size
    best_c = compute_c5(h_bin.astype(np.float64))
    best_max_c = int(round(best_c * n / 2))  # maybe not exact but we can compute integer corr.
    # But we get max_c integer by compute_corr, not compute_c5.
    # Let's have function corr_max_int(h_bin): return np.correlate(...).max()
    # compute c_int = np.correlate(...).max()
    c_int = np.correlate(h_bin, 1 - h_bin, mode='full').max()
    max_c = c_int
    for it in range(max_iters):
        if max_c <= target_max_c:
            break
        # compute corr array and idx
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        max_c = corr.max()
        idx = np.flatnonzero(corr == max_c)[0]
        k = idx - (n - 1)
        k_abs = abs(k)
        L = n - k_abs
        # segments
        h1 = h_bin[:L]
        h2 = h_bin[k_abs:k_abs+L]
        maskA = (h1 == 1) & (h2 == 0)
        maskB = (h1 == 0) & (h2 == 1)
        # candidate strategies
        improved = False
        # 1) if maskA sum >= maskB sum, try A-zeros_out
        if maskA.sum() >= maskB.sum():
            # zeros out of overlap region
            zeros_out_indices = np.flatnonzero(h_bin[L:] == 0) + L
            if maskA.sum() > 0 and zeros_out_indices.size > 0:
                i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                j = zeros_out_indices[rng.integers(zeros_out_indices.size)]
                # swap
                h_bin[i], h_bin[j] = 0, 1
                # evaluate
                new_max_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
                if new_max_c < max_c:
                    max_c = new_max_c
                    improved = True
                else:
                    # revert
                    h_bin[i], h_bin[j] = 1, 0
        else:
            # maskB bigger
            ones_out_indices = np.flatnonzero(h_bin[L:] == 1) + L
            if maskB.sum() > 0 and ones_out_indices.size > 0:
                j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                i = ones_out_indices[rng.integers(ones_out_indices.size)]
                h_bin[i], h_bin[j] = 1, 0
                new_max_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
                if new_max_c < max_c:
                    max_c = new_max_c
                    improved = True
                else:
                    h_bin[i], h_bin[j] = 0, 1
        if not improved:
            # second strategy: A-B swap if both non-empty
            if maskA.sum() > 0 and maskB.sum() > 0:
                i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                # swapping will reduce both by 1
                h_bin[i], h_bin[j] = 0, 1
                new_max_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
                if new_max_c < max_c:
                    max_c = new_max_c
                    improved = True
                else:
                    h_bin[i], h_bin[j] = 1, 0
        if not improved:
            # fallback random swap
            # pick random i where h[i]=1, random j where h[j]=0
            i = rng.choice(np.flatnonzero(h_bin == 1))
            j = rng.choice(np.flatnonzero(h_bin == 0))
            h_bin[i], h_bin[j] = 0, 1
            new_max_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
            if new_max_c < max_c:
                max_c = new_max_c
                improved = True
            else:
                # maybe accept with small probability to escape local minima
                # temperature schedule
                T = max(0.01 * (max_iters - it) / max_iters, 1e-6)
                if rng.random() < np.exp(-(new_max_c - max_c) / T):
                    max_c = new_max_c
                    # keep swap
                    improved = True
                else:
                    h_bin[i], h_bin[j] = 1, 0
        # if not improved after fallback, perhaps we are stuck; break early or continue
    return h_bin, max_c

def anneal_binary(h_bin, rng, max_iters, target_max_c):

    n = h_bin.size
    # compute initial max_c
    max_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
    for it in range(max_iters):
        if max_c <= target_max_c:
            break
        # compute temperature
        T = 0.5 * (1 - it / max_iters) + 1e-6   # start 0.5, end near 0
        # compute worst shift
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        max_c = corr.max()
        idx = np.flatnonzero(corr == max_c)[0]
        k = idx - (n - 1)
        k_abs = abs(k)
        L = n - k_abs
        # compute masks
        h1 = h_bin[:L]; h2 = h_bin[k_abs:k_abs+L]
        maskA = (h1 == 1) & (h2 == 0)
        maskB = (h1 == 0) & (h2 == 1)
        # Attempt targeted swap with probability
        if rng.random() < 0.7:
            # Choose strategy based on which side is larger
            if maskA.sum() >= maskB.sum():
                # prefer A-zeros_out
                zeros_out = np.flatnonzero(h_bin[L:] == 0) + L

                    j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]

def anneal_binary(h_bin, rng, steps, target_max_c=None):
    n = h_bin.size
    max_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
    for it in range(steps):
        if target_max_c is not None and max_c <= target_max_c:
            break
        # temperature schedule
        T = 0.4 * (1 - it/steps) + 1e-9
        # targeted swap at probability
        if rng.random() < 0.6:
            # compute worst shift
            corr = np.correlate(h_bin, 1 - h_bin, mode='full')
            max_c = corr.max()
            idxs = np.flatnonzero(corr == max_c)
            idx = idxs[rng.integers(len(idxs))]
            k = idx - (n - 1)
            k_abs = abs(k)
            L = n - k_abs
            h1 = h_bin[:L]; h2 = h_bin[k_abs:k_abs+L]
            maskA = (h1 == 1) & (h2 == 0)
            maskB = (h1 == 0) & (h2 == 1)
            # choose swap
            swap_done = False
            if maskA.sum() >= maskB.sum():
                # try A with out zeros
                zeros_out = np.flatnonzero(h_bin[L:] == 0) + L
                if maskA.sum() > 0 and zeros_out.size > 0:
                    i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                    j = zeros_out[rng.integers(zeros_out.size)]
                    swap_done = True
                elif maskA.sum() > 0 and maskB.sum() > 0:
                    i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                    j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                    swap_done = True
            else:
                # maskB larger
                ones_out = np.flatnonzero(h_bin[L:] == 1) + L
                if maskB.sum() > 0 and ones_out.size > 0:
                    j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                    i = ones_out[rng.integers(ones_out.size)]
                    swap_done = True
                elif maskA.sum() > 0 and maskB.sum() > 0:
                    i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                    j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                    swap_done = True
            if swap_done:
                h_bin[i], h_bin[j] = 0, 1
                new_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
                delta = new_c - max_c
                if delta <= 0 or rng.random() < np.exp(-delta / T):
                    max_c = new_c
                else:
                    h_bin[i], h_bin[j] = 1, 0
                continue   # go to next iteration
        # fallback random swap
        i = rng.choice(np.flatnonzero(h_bin == 1))
        j = rng.choice(np.flatnonzero(h_bin == 0))
        h_bin[i], h_bin[j] = 0, 1
        new_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
        delta = new_c - max_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            max_c = new_c
        else:
            h_bin[i], h_bin[j] = 1, 0
    final_c = np.correlate(h_bin, 1 - h_bin, mode='full').max()
    return h_bin, final_c

import time, numpy as np

def compute_c5(h):
    """Scaled max correlation for h on [0,2]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def enforce_sum_float(h, rng):
    """Adjust h so sum(h)=len(h)/2 staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h

def binary_from_float(arr):
    """Top half values become 1, rest 0."""
    n = arr.size
    idx = np.argpartition(-arr, n // 2)[: n // 2]
    b = np.zeros(n, dtype=np.uint8)
    b[idx] = 1
    return b

def anneal_binary(h, rng, steps, target_max_c=None):
    """Simulated‑annealing on a binary vector with sum constraint."""
    n = h.size
    max_c = np.correlate(h, 1 - h, mode="full").max()
    for it in range(steps):
        if target_max_c is not None and max_c <= target_max_c:
            break
        T = 0.4 * (1 - it / steps) + 1e-9
        # try a targeted move most of the time
        if rng.random() < 0.6:
            corr = np.correlate(h, 1 - h, mode="full")
            max_c = corr.max()
            idxs = np.flatnonzero(corr == max_c)
            idx = idxs[rng.integers(len(idxs))]
            k = idx - (n - 1)
            k_abs = abs(k)
            L = n - k_abs
            h1, h2 = h[:L], h[k_abs:k_abs + L]
            maskA = (h1 == 1) & (h2 == 0)
            maskB = (h1 == 0) & (h2 == 1)
            swap_done = False
            if maskA.sum() >= maskB.sum():
                zeros_out = np.flatnonzero(h[L:] == 0) + L
                if maskA.sum() and zeros_out.size:
                    i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                    j = zeros_out[rng.integers(zeros_out.size)]
                    swap_done = True
                elif maskA.sum() and maskB.sum():
                    i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                    j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                    swap_done = True
            else:
                ones_out = np.flatnonzero(h[L:] == 1) + L
                if maskB.sum() and ones_out.size:
                    j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                    i = ones_out[rng.integers(ones_out.size)]
                    swap_done = True
                elif maskA.sum() and maskB.sum():
                    i = np.flatnonzero(maskA)[rng.integers(maskA.sum())]
                    j = np.flatnonzero(maskB)[rng.integers(maskB.sum())]
                    swap_done = True
            if swap_done:
                h[i], h[j] = 0, 1
                new_c = np.correlate(h, 1 - h, mode="full").max()
                delta = new_c - max_c

            max_c = new_c
        else:
            h[i], h[j] = 1, 0
    final_c = np.correlate(h, 1 - h, mode="full").max()

def find_best_binary(rng, time_budget):

    n = 1024
    target = int(np.floor(0.3808 * n / 2))   # ≤194
    best_h = None
    best_c = n
    start = time.time()
    # use parent if present
    if "initial_h_values" in globals():

def continuous_sa(h, rng, time_limit):
    """Simple SA on the real‑valued feasible set."""
    start = time.time()
    N = h.size
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    T0, Tmin = 0.01, 1e-7
    while time.time() - start < time_limit:
        frac = (time.time() - start) / time_limit
        T = T0 * (Tmin / T0) ** frac
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[j] -= eps
            h_new[i] += eps
        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, cur_c = h_new, new_c
            if new_c < best_c:
                best_h, best_c = h_new.copy(), new_c
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # 1) binary search (≈70% of time)
    binary_time = total * 0.70
    best_bin, best_max_c = find_best_binary(rng, binary_time)

    best_c = best_max_c * (2.0 / n)

    if remaining > 5.0:
        h_float = best_bin.astype(np.float64)
