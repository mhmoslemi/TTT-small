def delta_corr_for_move(h, i, j, d, n):
    delta_corr = np.zeros(n)
    # term A: -d * h[i + k] for k where i + k < n => k in 0..n-1-i
    k_max_i = n - i
    if k_max_i > 0:
        delta_corr[:k_max_i] -= d * h[i:i + k_max_i]
    # term B: +d * h[j + k] for k where j + k < n
    k_max_j = n - j
    if k_max_j > 0:
        delta_corr[:k_max_j] += d * h[j:j + k_max_j]
    # term C: -d * h[i - k] for k where i - k >=0 => k in 0..i
    delta_corr[:i+1] -= d * h[i::-1]  # reverse slice i to 0
    # term D: +d * h[j - k] for k where j - k >=0 => k in 0..j
    delta_corr[:j+1] += d * h[j::-1]
    # term self:
    delta_corr[0] += 2 * d * d
    diff = abs(i - j)
    if diff != 0:
        delta_corr[diff] += - d * d
    return delta_corr

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 from full correlation."""
    n = h.size
    corr = np.correlate(h, h, mode="full")[n - 1 :]
    return 1.0 - (2.0 / n) * np.min(corr)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    return project_box_sum(rng.random(n), n / 2.0)


def delta_corr_move(h: np.ndarray, i: int, j: int, d: float) -> np.ndarray:
    """Δcorr for moving mass d from i to j (i≠j)."""
    n = h.size
    dc = np.zeros(n)
    kmax_i = n - i
    if kmax_i:
        dc[:kmax_i] -= d * h[i : i + kmax_i]
    kmax_j = n - j
    if kmax_j:
        dc[:kmax_j] += d * h[j : j + kmax_j]
    dc[: i + 1] -= d * h[i :: -1]
    dc[: j + 1] += d * h[j :: -1]
    dc[0] += 2.0 * d * d
    diff = abs(i - j)
    if diff:
        dc[diff] += -d * d
    return dc


def hill_climb(
    h: np.ndarray,
    corr: np.ndarray,
    c5: float,
    dx: float,
    rng: np.random.Generator,
    max_iters: int,
    step_scale: float = 0.6,
) -> tuple:
    """Greedy hill‑climb using worst shift gradient (O(n) per step)."""
    n = h.size
    best_h, best_c, best_corr = h.copy(), c5, corr.copy()
    cur_h, cur_corr, cur_c = h, corr, c5
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        k = int(np.argmin(cur_corr))
        grad = np.ones_like(cur_h)
        if k:
            grad[: n - k] -= cur_h[k:]
            grad[k:] -= cur_h[: n - k]
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        dc = delta_corr_move(cur_h, i, j, eps)
        new_corr = cur_corr + dc
        new_c = 1.0 - dx * np.min(new_corr)
        if new_c < cur_c:
            cur_h[i] -= eps
            cur_h[j] += eps
            cur_corr, cur_c = new_corr, new_c
            if new_c < best_c:
                best_c, best_h, best_corr = new_c, cur_h.copy(), cur_corr.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.8, 1e-4)
        if best_c <= 0.3808:
            break
    return best_h, best_corr, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for final checks
    stop_time = start + budget_s - 5.0

    # ----- initialise ----------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, n / 2.0)
    else:
        n = 512
        h = random_feasible(n, rng)

    dx = 2.0 / n
    corr = np.correlate(h, h, mode="full")[n - 1 :]
    min_corr = np.min(corr)
    c5 = 1.0 - dx * min_corr
    best_h, best_c5 = h.copy(), c5
    best_corr = corr.copy()

    # ----- simulated annealing -----------------------------------------
    T = 0.005
    cooling = 0.99996
    step_scale = 0.5
    accepted = 0
    it = 0
    last_improve = 0

    while time.time() < stop_time:
        i = rng.integers(0, n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if h[j] >= 1.0 - 1e-12:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * step_scale
        dc = delta_corr_move(h, i, j, delta)
        new_min_corr = np.min(corr + dc)
        new_c5 = 1.0 - dx * new_min_corr
        if new_c5 < c5:
            accept = True
        else:
            prob = np.exp(-(new_c5 - c5) / T)
            accept = rng.random() < prob
        if accept:
            h[i] -= delta
            h[j] += delta
            corr += dc
            c5 = new_c5
            accepted += 1
            if c5 < best_c5:
                best_c5, best_h, best_corr = c5, h.copy(), corr.copy()
                last_improve = it
                if best_c5 <= 0.3808:
                    break
        T *= cooling
        it += 1
        if it % 5000 == 0:
            step_scale = max(step_scale * 0.9, 0.01)
        if accepted % 20000 == 0 and accepted:
            corr = np.correlate(h, h, mode="full")[n - 1 :]
            c5 = 1.0 - dx * np.min(corr)
        if it - last_improve > 200000:
            # small random perturbation around the best so far
            h = best_h + rng.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, n / 2.0)
            corr = np.correlate(h, h, mode="full")[n - 1 :]
            c5 = 1.0 - dx * np.min(corr)
            last_improve = it
            T = 0.005

    # ----- final greedy refinement --------------------------------------
    remaining = stop_time - time.time()
    if remaining > 1.0:
        max_iters = int(min(200000, remaining * 2000))
        best_h, best_corr, best_c5 = hill_climb(
            best_h, best_corr, best_c5, dx, rng, max_iters, step_scale=0.6
        )

    # ----- sanitise & return -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
