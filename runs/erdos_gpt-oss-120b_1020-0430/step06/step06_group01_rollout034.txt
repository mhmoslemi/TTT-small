def binary_sa_512(rng, time_budget):
    start = time.time()
    n=512
    best_h = np.zeros(n, dtype=float)
    best_c = np.inf
    # initialize at random binary (n/2 ones)
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    # store arrays for fast index selection
    ones_arr = np.copy(ones)
    zeros_arr = np.setdiff1d(np.arange(n), ones_arr, assume_unique=True)
    T0, T_end = 0.05, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end/T0)**(elapsed/time_budget)  # exponential decay
        # choose random swap
        i = ones_arr[rng.integers(len(ones_arr))]
        j = zeros_arr[rng.integers(len(zeros_arr))]
        # swap
        h[i] = 0.0; h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta/T):
                accept = True
        if accept:
            c_curr = c_new
            # swap indices in arrays
            # find positions
            pos_i = np.where(ones_arr == i)[0][0]
            pos_j = np.where(zeros_arr == j)[0][0]
            ones_arr[pos_i] = j
            zeros_arr[pos_j] = i
            if c_curr < best_c:
                best_c, best_h = c_curr, h.copy()
                # early exit if below target
                if best_c <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1.0; h[j] = 0.0
    return best_h, best_c
```

def binary_sa_incremental(n, rng, time_budget, target_c=0.3808):
    start = time.time()
    # initialize binary vector
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # compute convolution arrays
    conv_h1 = np.convolve(h, np.ones(n), mode='full')
    conv_hh = np.convolve(h, h, mode='full')
    c_curr = np.max(conv_h1 - conv_hh) * 2.0 / n

    pos_one = {i: idx for idx, i in enumerate(ones_list)}
    pos_zero = {i: idx for idx, i in enumerate(zeros_list)}
    # annealing schedule
    T0, T_end = 0.05, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)  # exponential decay

        h[j] = 1.0

def binary_sa_incremental(start_h, rng, time_budget, target_c=0.3808):
    n = start_h.size
    h = start_h.copy()
    conv_h1 = np.convolve(h, np.ones(n), mode='full')
    conv_hh = np.convolve(h, h, mode='full')
    c_curr = np.max(conv_h1 - conv_hh) * 2.0 / n
    best_h = h.copy()
    best_c = c_curr

    # Build lists
    ones_list = [i for i in range(n) if h[i] > 0.5]  # h[i] is 0 or 1
    zeros_list = [i for i in range(n) if h[i] <= 0.5]
    pos_one = {i: idx for idx, i in enumerate(ones_list)}
    pos_zero = {i: idx for idx, i in enumerate(zeros_list)}
    T0, T_end = 0.05, 1e-5
    start_time = time.time()
    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # random swap
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        # apply forward swap
        apply_swap(h, conv_h1, conv_hh, i, j, -1.0, 1.0)
        c_new = np.max(conv_h1 - conv_hh) * 2.0 / n
        delta = c_new - c_curr
        accept = (delta <= 0.0) or (rng.random() < np.exp(-delta / max(T, 1e-12)))
        if accept:
            c_curr = c_new
            # update lists
            pos_i = pos_one[i]
            pos_j = pos_zero[j]
            ones_list[pos_i] = j
            zeros_list[pos_j] = i
            pos_one[j] = pos_i
            pos_zero[i] = pos_j
            del pos_one[i]
            del pos_zero[j]
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
                if best_c <= target_c:
                    break
        else:
            # revert
            apply_swap(h, conv_h1, conv_hh, i, j, 1.0, -1.0)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities -------------------


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def binarize_to_n2(h: np.ndarray) -> np.ndarray:
    """Convert to a binary vector with exactly n/2 ones (top values)."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    bh = np.zeros_like(h)
    bh[idx] = 1.0
    return bh


# ------------------- incremental binary SA -------------------


def apply_swap(h, conv_h1, conv_hh, i, j, di, dj):
    """Apply swap with deltas di, dj (di = -1, dj = +1 for forward)."""
    n = h.shape[0]
    conv_h1[i : i + n] += di
    conv_h1[j : j + n] += dj
    conv_hh[i : i + n] += 2.0 * di * h
    conv_hh[j : j + n] += 2.0 * dj * h
    conv_hh[2 * i] += di * di
    conv_hh[2 * j] += dj * dj
    conv_hh[i + j] += 2.0 * di * dj
    h[i] += di
    h[j] += dj

            if c_curr < best_c:
                best_c, best_h = c_curr, h.copy()
                if best_c <= target_c:
                    break
        else:
            apply_swap(h, conv_h1, conv_hh, i, j, 1.0, -1.0)
    return best_h, best_c


# ------------------- continuous refinement -------------------


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def continuous_optimize(n, rng, time_budget, init_h=None):
    start = time.time()
    best_h, best_c = None, np.inf
    h = init_h.copy() if init_h is not None else rng.random(n)
    h = enforce_sum(h, rng)
    max_iters = int(min(30000, time_budget * 500))
    h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
    best_h, best_c = h, c
    beta_seq = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in beta_seq:
        if time.time() - start > time_budget * 0.96:
            break
        max_iter = 2000 if beta >= 200 else 1000
        h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    if time.time() - start < time_budget * 0.98:
        h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    while time.time() - start < time_budget * 0.995:
        if best_h is not None and rng.random() < 0.5:
            h = best_h.copy()
            for _ in range(5):
                i = rng.integers(0, n)
                delta = (rng.random() - 0.5) * 0.2
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = rng.random(n)
            h = enforce_sum(h, rng)
        max_iters = int(min(5000, (time_budget - (time.time() - start)) * 200))
        h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
        if c < best_c:
            best_h, best_c = h.copy(), c
        for beta in (20., 80., 200.):
            if time.time() - start > time_budget * 0.995:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=500)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
    return best_h, best_c


# ------------------- main entry point -------------------


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_all = time.time()
    total_budget = budget_s
    n = 512
    best_h = None
    best_c = np.inf
    best_n = None

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n
        start_bin = binarize_to_n2(h0)
    else:
        start_bin = None

    remaining = total_budget - (time.time() - start_all)
    binary_time = remaining * 0.75
    if start_bin is None:
        start_bin = np.zeros(n, dtype=float)
        start_bin[rng.choice(n, n // 2, replace=False)] = 1.0

    h_bin, c_bin = binary_sa_incremental(start_bin, rng, binary_time, target_c=0.3808)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin, c_bin, n

    if best_c <= 0.3808:
        best_h = enforce_sum(best_h, rng)
        final_c = compute_c5(best_h)
        return best_h.tolist(), float(final_c), int(best_h.size)

    remaining = total_budget - (time.time() - start_all)
    if remaining > 0:
        h_cont, c_cont = continuous_optimize(n, rng, remaining, init_h=best_h)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    while time.time() - start_all < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j and best_h[i] > 0 and best_h[j] < 1:
            eps = rng.random() * min(best_h[i], 1.0 - best_h[j]) * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
