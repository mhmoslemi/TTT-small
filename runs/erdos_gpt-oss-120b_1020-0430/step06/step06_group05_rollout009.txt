    corr = np.correlate(x, 1.0 - x, mode="full")

    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # loop until diff is tiny
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
        else:
            cap = -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.97 ** it)

        h = np.clip(h, 0.0, 1.0)

    while restarts < max_restarts and time.time() - start < budget_s * 0.70:
        h = random_binary()
        h = gradient_descent_refine(h, target_sum, rng, step=0.03, max_iter=300)
        update_best(h)
        restarts += 1

    # final SLSQP passes with increasing beta
    for beta in (500, 2000, 5000, 20000, 50000, 100000):
        if time.time() - start > budget_s * 0.85:
            break
        best_h = slsqp_optimize(best_h, beta, max_iter=800, target=target_sum, rng=rng)
        update_best(best_h)

    # simulated annealing fine‑tuning
    cur_h = best_h.copy()
    cur_c = best_c
    T0 = 0.01
    Tmin = 1e-7
    while time.time() - start < budget_s * 0.99:
        elapsed = time.time() - start
        frac = (budget_s - elapsed) / budget_s
        T = max(Tmin, T0 * (frac ** 2))
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-16)):
                cur_h, cur_c = new_h, new_c

    # final enforce and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
