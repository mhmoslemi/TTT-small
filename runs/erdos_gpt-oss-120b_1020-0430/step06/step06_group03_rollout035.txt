def compute_c5_from_lengths(a,b):
    m = len(a)
    # Build start, end arrays for 1 intervals:
    starts = np.empty(m)
    ends = np.empty(m)
    cur = 0.0
    for i in range(m):
        starts[i] = cur
        ends[i] = cur + a[i]
        cur = ends[i] + b[i]  # skip 0 interval
    # Now we'll compute I(k) minimal over k.
    # Candidate k values:
    candidates = set()
    for i in range(m):
        for j in range(m):
            candidates.add(abs(starts[i] - starts[j]))
            candidates.add(abs(starts[i] - ends[j]))
            candidates.add(abs(ends[i] - starts[j]))
            candidates.add(abs(ends[i] - ends[j]))
    # Also add zero
    candidates.add(0.0)
    # For each k, compute I(k) using O(m^2)
    min_I = float('inf')
    for k in candidates:
        I = 0.0
        # For each i,j compute overlap length
        for i in range(m):
            s_i = starts[i]
            e_i = ends[i]
            for j in range(m):
                s_j = starts[j] - k
                e_j = ends[j] - k
                # Intersection with domain [0,2] automatically satisfied because s_i,e_i in [0,2], s_j,e_j may be negative; compute overlap:
                lo = max(s_i, s_j, 0.0)
                hi = min(e_i, e_j, 2.0)
                if hi > lo:
                    I += hi - lo
        if I < min_I:
            min_I = I
    # C5 = 1 - min_I
    return 1.0 - min_I

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target, max_iter=60):
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def softmax_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exp_term = np.exp(beta * (corr - m))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    center = n - 1
    wk = w[center]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[center + k] + w[center - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def random_swap_perturb(h, rng, eps=0.1):
    up = np.where(h > eps)[0]
    dn = np.where(h < 1.0 - eps)[0]
    if up.size == 0 or dn.size == 0:
        return h
    i = rng.choice(up)
    j = rng.choice(dn)
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

            best_c, best_h = nc, nh.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
