def full_corr(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1-h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    # extract first 2*n-1 entries
    return corr[:2*n-1]

    high = 1.0

        if s < target:

            high = mid

    # assume x within small deviation from target
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute uniformly
        x = x + diff / x.size
        # clip again
        x = np.clip(x, 0.0, 1.0)
    # after second clip, sum may still off by tiny amount; we can apply small correction again
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute proportionally to distance to bounds
        # compute free capacity
        cap_up = 1.0 - x
        cap_down = x
        cap = cap_up if diff > 0 else cap_down
        total_cap = cap.sum()
        if total_cap > 0:
            # add fraction
            add = diff * (cap / total_cap)
            x = x + add
    # final clip
    return np.clip(x, 0.0, 1.0)

    if N is None:

import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _conv_fft(a, b, N=None):
    if N is None:
        N = _next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    return np.fft.irfft(A * B, n=N)[: len(a) + len(b) - 1]

def _full_corr(h):
    """Linear correlation of h with (1-h)."""
    return _conv_fft(h, 1.0 - h)

def _compute_c5(h):
    n = h.shape[0]
    corr = _full_corr(h)
    return corr.max() * (2.0 / n)

def _project_sum_box(x, target, iters=5):
    """Clip to [0,1] and enforce Σx = target by simple redistribution."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(iters):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        x += diff / x.size
        x = np.clip(x, 0.0, 1.0)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 1024                   # number of samples
    target = n / 2.0           # required sum of h

    # ---- initial h -------------------------------------------------
    if "initial_h_values" in globals():
        ih = np.array(initial_h_values, dtype=float)
        if ih.shape[0] == n:
            h = ih.copy()
        elif n % ih.shape[0] == 0:
            factor = n // ih.shape[0]
            h = np.repeat(ih, factor)
        else:
            x_old = np.linspace(0, 1, ih.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            h = np.interp(x_new, x_old, ih)
        # break perfect symmetry
        h += rng.normal(scale=0.008, size=n)
        h = _project_sum_box(h, target)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, int(target), replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_c = _compute_c5(best_h)

    # ----- pre‑computations -----------------------------------------
    N_conv = _next_pow2(3 * n - 2)   # size for convolutions with w (len=2n‑1)
    offset = n - 1                  # slice offset in conv results

    # ----- optimisation parameters ----------------------------------
    total_iters = 300000
    beta = 100.0
    lr = 0.45
    mu = 0.88
    v = np.zeros_like(h)

    # ----- main loop -----------------------------------------------
    for it in range(total_iters):
        if time.time() - start > budget_s * 0.99:
            break

        # exact C5 for monitoring
        c5 = _compute_c5(h)
        if c5 < best_c:
            best_c, best_h = c5, h.copy()
            if best_c < 0.38078:
                break

        # ----- soft‑max weights ------------------------------------
        corr = np.correlate(h, 1.0 - h, mode="full")
        w_raw = np.exp(beta * corr - np.max(beta * corr))
        w = w_raw / w_raw.sum()

        # ----- convolutions for gradient ---------------------------
        w_rev = w[::-1]

        H = np.fft.rfft(h, n=N_conv)
        W = np.fft.rfft(w, n=N_conv)
        WR = np.fft.rfft(w_rev, n=N_conv)

        conv1 = np.fft.irfft(W * H, n=N_conv)[:2 * n - 1 + n - 1]   # length 3n‑2
        conv2 = np.fft.irfft(WR * H, n=N_conv)[:2 * n - 1 + n - 1]

        sum_w_h_minus = conv1[offset:offset + n]
        sum_w_h_plus  = conv2[offset:offset + n]

        grad = 1.0 - sum_w_h_minus - sum_w_h_plus
        grad -= grad.mean()                     # keep sum invariant

        # ----- momentum step ---------------------------------------
        v = mu * v - lr * grad
        h_new = h + v
        h = _project_sum_box(h_new, target)

        # ----- schedule updates -------------------------------------
        if (it + 1) % 60000 == 0:
            beta = min(beta * 1.7, 20000.0)
            lr = max(lr * 0.72, 0.015)

    # ----- final polishing -----------------------------------------
    for _ in range(3000):
        if time.time() - start > budget_s * 0.99:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        w_raw = np.exp(20000.0 * corr - np.max(20000.0 * corr))
        w = w_raw / w_raw.sum()
        w_rev = w[::-1]

        H = np.fft.rfft(h, n=N_conv)
        W = np.fft.rfft(w, n=N_conv)
        WR = np.fft.rfft(w_rev, n=N_conv)

        conv1 = np.fft.irfft(W * H, n=N_conv)[:2 * n - 1 + n - 1]
        conv2 = np.fft.irfft(WR * H, n=N_conv)[:2 * n - 1 + n - 1]

        sum_w_h_minus = conv1[offset:offset + n]
        sum_w_h_plus  = conv2[offset:offset + n]

        grad = 1.0 - sum_w_h_minus - sum_w_h_plus
        grad -= grad.mean()
        h = _project_sum_box(h - 0.008 * grad, target)

    best_h = _project_sum_box(best_h, target)
    final_c = _compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
