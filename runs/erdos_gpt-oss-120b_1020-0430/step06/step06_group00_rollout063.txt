def compute_c5(h):
    N = h.size

    H = np.fft.fft(h, L)
    G = np.fft.fft(g, L)

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
def compute_c5_fast(h: np.ndarray) -> float:
    """C5 using FFT (full linear correlation)."""
    N = h.size
    L = 2 * N
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * N - 1]
    return np.max(corr) * (2.0 / N)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place so that sum(h) = N/2 while staying in [0,1]."""
    N = h.size
    target = N / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, N)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] = 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    N = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        target = N / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, N)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def binary_sa(N: int, rng: np.random.Generator, time_budget: float):
    """Binary SA on length N with exact N/2 ones."""
    start = time.time()
    h = np.zeros(N, dtype=np.float64)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_curr = np.max(corr) * (2.0 / N)
    best_h = h.copy()
    best_c = c_curr
    ones_list = list(ones)
    zeros_list = [i for i in range(N) if h[i] == 0.0]

    while time.time() - start < time_budget:
        i_pos = rng.integers(len(ones_list))
        j_pos = rng.integers(len(zeros_list))
        i = ones_list[i_pos]
        j = zeros_list[j_pos]

        # remove i (1 -> 0)
        one_minus_h = 1.0 - h
        corr[N - 1 - i : N - 1 - i + N] -= one_minus_h
        h[i] = 0.0

        # add j (0 -> 1)
        one_minus_h = 1.0 - h
        corr[N - 1 - j : N - 1 - j + N] += one_minus_h
        h[j] = 1.0

        # update index lists
        ones_list[i_pos] = j
        zeros_list[j_pos] = i

        c_new = np.max(corr) * (2.0 / N)

        elapsed = time.time() - start
        T0, T_end = 0.1, 1e-4
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c

            corr[N - 1 - j : N - 1 - j + N] -= one_minus_h
            h[j] = 0.0
            one_minus_h = 1.0 - h
            corr[N - 1 - i : N - 1 - i + N] += one_minus_h
            h[i] = 1.0
            ones_list[i_pos] = i
            zeros_list[j_pos] = j

    return best_h, best_c

            best_h, best_c, best_n = h_bin, c_bin, N
        if best_c <= 0.3808:
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = enforce_sum(best_h, rng)
            final_c = compute_c5_fast(best_h)
            return best_h.tolist(), float(final_c), int(best_h.size)

    # ---- Phase 1: continuous SA on current best ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h_cont, c_cont = continuous_sa(best_h.copy(), rng, remaining,
                                       T0=0.02, T_end=1e-7, scale_factor=0.3)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # ---- Phase 2: gradient tidy‑up ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h_grad = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=30)
        c_grad = compute_c5_fast(h_grad)
        if c_grad < best_c:
            best_h, best_c = h_grad, c_grad

    # ---- Final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
