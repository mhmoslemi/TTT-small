def project_to_box_and_sum(x, target):
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # compute sum diff
    diff = x.sum() - target
    if abs(diff) < 1e-12:
        return x
    # active set: indices not at bound (0 < x_i < 1)
    free = np.where((x > 0) & (x < 1))[0]
    # If free empty (unlikely), assign diff uniformly across all and clip again iteratively
    # We'll iterate a few times.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if free.size == 0:
            # fallback: adjust all uniformly
            x -= diff / x.size
            x = np.clip(x, 0.0, 1.0)
            diff = x.sum() - target
            continue
        # distribute diff equally among free vars
        delta = diff / free.size
        x[free] -= delta
        # enforce bounds again
        x = np.clip(x, 0.0, 1.0)
        diff = x.sum() - target
        free = np.where((x > 0) & (x < 1))[0]
    return x

def subgradient_average_grad(h, eps=1e-9):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k == 0:
            grad += 1 - 2*h
        else:
            shift = abs(k)
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
    grad /= max(1, idxs.size)  # avoid division by zero
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities -------------------------------------------------

TARGET_RATIO = 0.5  # sum(h) = n * TARGET_RATIO


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def project_to_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce sum(x)=target by iterative redistribution."""
    x = np.clip(x, 0.0, 1.0)
    diff = x.sum() - target
    if abs(diff) < 1e-12:
        return x
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            # adjust uniformly, then re‑clip
            x -= diff / x.size
            x = np.clip(x, 0.0, 1.0)
            diff = x.sum() - target
            continue
        delta = diff / free.size
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
        diff = x.sum() - target
    # final tiny correction if needed
    if abs(diff) > 1e-9:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size:
            i = free[0]
            x[i] -= diff
            x[i] = np.clip(x[i], 0.0, 1.0)
    return x


def subgradient_average_grad(h: np.ndarray, eps: float = 1e-9) -> np.ndarray:
    """Average gradient of O_k for all shifts whose value is within eps of the maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            shift = abs(k)
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
    if idxs.size:
        grad /= float(idxs.size)
    else:
        grad.fill(1.0)
    return grad


# ---------- SLSQP soft‑max refinement ---------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n * TARGET_RATIO
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_to_box_sum(x, CURRENT_TARGET)
    return x


# ---------- main routine ---------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initial vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

    else:
        n0 = 512
        h0 = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, n0 // 2, replace=False)
        h0[idx] = 1.0
    n = h0.size
    target = n * TARGET_RATIO
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- gradient‑descent with subgradient averaging --------------------
    def descent(start_h):
        h = start_h.copy()
        step = 0.12
        decay = 0.9987
        max_iters = 60000
        local_best_h = h.copy()
        local_best_c = compute_c5(h)
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.78:
                break
            grad = subgradient_average_grad(h, eps=1e-9)
            h_new = h - step * grad
            h_new = project_to_box_sum(h_new, target)
            c = compute_c5(h_new)
            if c < local_best_c:
                local_best_c = c
                local_best_h = h_new.copy()
            h = h_new
            step *= decay
            # occasional random transfer to diversify
            if (it % 800 == 0) and rng.random() < 0.35:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps > 0:
                        eps = rng.random() * max_eps * 0.15
                        h2 = h.copy()
                        h2[i] -= eps
                        h2[j] += eps
                        h2 = project_to_box_sum(h2, target)
                        c2 = compute_c5(h2)
                        if c2 < local_best_c:
                            local_best_c = c2
                            local_best_h = h2.copy()
        return local_best_h, local_best_c

    # ----- multiple restarts ------------------------------------------------
    restarts = 6

        if r == 0:

        else:

                i, j = rng.integers(0, n, size=2)
                if i != j:
                    max_eps = min(start_vec[i], 1.0 - start_vec[j])
                    if max_eps > 0:
                        eps = rng.random() * max_eps * 0.2

                        start_vec[j] += eps
            start_vec = project_to_box_sum(start_vec, target)

        if c_res < best_c:

        if best_c <= 0.3808:
            break

    # ----- SLSQP soft‑max refinement --------------------------------------
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):

            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=1300)
        c = compute_c5(h_opt)
        if c < best_c:
            best_c, best_h = c, h_opt.copy()
        if best_c <= 0.3808:
            break

    # ----- final random local search --------------------------------------
    while time.time() - start < budget_s * 0.998:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.06
        h2 = best_h.copy()
        h2[i] -= eps
        h2[j] += eps
        h2 = project_to_box_sum(h2, target)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            if best_c <= 0.3808:
                break

    # ----- final sanity enforcement ---------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
