import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.7,
    time_limit: float | None = None,
) -> tuple[np.ndarray, float]:
    """Gradient‑directed hill climbing with optional time limit."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    start = time.time() if time_limit is not None else None
    it = 0
    while it < max_iters:
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
        it += 1
    return best_h, best_c


def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.015,
    max_iter: int = 10,
) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ----- soft‑max SLSQP optimizer -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(
    h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 800
) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

            init_h.copy(), rng, max_iters=max_iters, step_scale=0.7
        )
        if c_hc < best_c:
            best_h, best_c, best_n = h_hc.copy(), c_hc, target_n
        init_h = h_hc.copy()

    # brief gradient refinement
    elapsed = time.time() - start
    time_left = total_budget - elapsed
    if time_left > total_budget * 0.1 and best_h is not None:
        max_refine = int(min(30, time_left * 20))
        h_ref = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=max_refine)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c, best_n = h_ref.copy(), c_ref, target_n

    # a few SLSQP passes with modest iteration limits
    elapsed = time.time() - start
    time_left = total_budget - elapsed
    beta_seq = (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0)
    for beta in beta_seq:
        if time_left < total_budget * 0.12:
            break
        max_iter = int(min(500, max(200, time_left * 100)))
        try:
            h_sl = slsqp_optimize(best_h.copy(), beta, rng, max_iter=max_iter)
        except Exception:
            continue
        c_sl = compute_c5(h_sl)
        if c_sl < best_c:
            best_h, best_c, best_n = h_sl.copy(), c_sl, target_n
        elapsed = time.time() - start
        time_left = total_budget - elapsed

    # final stochastic tweaking
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # sanitisation
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        final_c = compute_c5(best_h)
    else:
        best_h = np.zeros(target_n)
        final_c = np.inf

    return best_h.tolist(), float(final_c), int(best_h.size)
