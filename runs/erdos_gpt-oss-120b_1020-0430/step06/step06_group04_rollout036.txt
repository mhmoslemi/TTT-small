def compute_c5(h):
    n = h.size
    # compute correlation via fft
    # zero pad to length L = next power of 2 >= 2*n-1
    L = 1 << ((2*n - 2).bit_length())
    H = np.fft.rfft(h, n=L)
    Hc = np.fft.rfft(1 - h, n=L)
    corr = np.fft.irfft(H * Hc, n=L)[: 2*n - 1]
    return float(np.max(corr) * (2.0 / n))

import time, numpy as np

def correlation_fft(h):
    n = h.size
    L = 1 << ((2*n - 2).bit_length())
    H = np.fft.rfft(h, n=L)
    Hc = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * Hc, n=L)[:2*n-1]
    return corr

def compute_c5(h):
    n = h.size
    corr = correlation_fft(h)
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    m = beta * v.max()
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[:n-shift] += wi * one_minus_h[shift:]
            g[shift:] -= wi * h[:n-shift]
        else:
            s = -shift
            g[s:] += wi * one_minus_h[:n-s]
            g[:n-s] -= wi * h[s:]
    return g

def polish_smoothmax(h, rng, deadline):
    n = h.size
    betas = [10, 50, 200, 800, 2000, 5000]
    for beta in betas:
        for _ in range(5):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            step = 0.1 / (beta**0.5)  # heuristic
            new_h = h - step * g
            np.clip(new_h, 0.0, 1.0, out=new_h)
            adjust_sum_exact(new_h, rng)
            if compute_c5(new_h) < compute_c5(h):
                h = new_h
    return h

def continuous_sa(h0, rng, deadline):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    total = deadline - start
    # schedule parameters
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-7
    eps0 = 0.3
    epsf = 1e-4
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * (1 - frac) + Tf * frac
        eps = eps0 * (1 - frac) + epsf * frac
        # pick i, j
        i = rng.integers(0, n)
        if h[i] >= 1 - 1e-12:
            continue
        j = rng.integers(0, n)
        if h[j] <= 1e-12:
            continue
        max_inc = min(1 - h[i], h[j])
        if max_inc <= 0:
            continue
        delta = rng.random() * max_inc
        if delta > eps:
            delta = eps
        # apply move
        h[i] += delta
        h[j] -= delta
        new_c = compute_c5(h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # rollback
            h[i] -= delta
            h[j] += delta
        it += 1
        # occasional sum fix
        if it % 5000 == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-12:

                new_val = h[idx] + diff
                new_val = max(0.0, min(1.0, new_val))
                h[idx] = new_val

def binary_sa(h0, rng, deadline):
    cur_h = h0.copy()
    n = cur_h.size
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = [i for i, v in enumerate(cur_h) if v > 0.5]
    zeros = [i for i, v in enumerate(cur_h) if v <= 0.5]
    T = cur_c * 0.02 + 1e-6
    it = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
        T *= 0.9999
        # optionally recompute lists periodically to avoid drift
        if it % 10000 == 0:
            ones = [i for i, v in enumerate(cur_h) if v > 0.5]
            zeros = [i for i, v in enumerate(cur_h) if v <= 0.5]
    return best_h, best_c

import time
import numpy as np

def correlation_fft(h):
    n = h.size
    L = 1 << ((2 * n - 2).bit_length())
    H = np.fft.rfft(h, n=L)
    Hc = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * Hc, n=L)[: 2 * n - 1]
    return corr

def compute_c5(h):
    n = h.size
    corr = correlation_fft(h)
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = beta * v.max()
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:] -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:] += wi * one_minus_h[: n - s]
            g[: n - s] -= wi * h[s:]
    return g

def polish_smoothmax(h, rng, deadline):
    betas = [10, 50, 200, 800, 2000, 5000]
    for beta in betas:
        for _ in range(5):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            step = 0.1 / (beta**0.5)
            new_h = h - step * g
            np.clip(new_h, 0.0, 1.0, out=new_h)
            adjust_sum_exact(new_h, rng)
            if compute_c5(new_h) < compute_c5(h):
                h = new_h
    return h

def continuous_sa(h0, rng, deadline):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    total = max(deadline - start, 1e-12)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-7
    eps0 = 0.3
    epsf = 1e-4
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * (1 - frac) + Tf * frac
        eps = eps0 * (1 - frac) + epsf * frac
        i = rng.integers(0, n)
        if h[i] >= 1 - 1e-12:
            continue
        j = rng.integers(0, n)
        if h[j] <= 1e-12:
            continue
        max_inc = min(1 - h[i], h[j])
        if max_inc <= 0:
            continue
        delta = rng.random() * max_inc
        if delta > eps:
            delta = eps
        h[i] += delta
        h[j] -= delta
        new_c = compute_c5(h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        it += 1

            if abs(diff) > 1e-12:
                idx = rng.integers(0, n)
                nv = h[idx] + diff
                h[idx] = max(0.0, min(1.0, nv))
    return best_h, best_c
