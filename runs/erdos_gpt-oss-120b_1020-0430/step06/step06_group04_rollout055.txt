def gradient_smoothmax(h, beta):
    n = h.shape[0]
    v = compute_corr_vec(h)
    # softmax
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / np.sum(exp_term)
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    # Loop over all shifts
    # shift index i = 0..2n-2 -> k = i-(n-1)
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # forward shift
            # For i in [0, n-1 - k]: derivative from h[i] term is wi * (1 - h[i+k])
            # add to g[i]
            g[: n - k] += wi * one_minus_h[k:]
            # For i in [k, n-1]: derivative from h[i] term of shift -k? Actually original gradient includes negative part from i >= k: -wi * h[i - k]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            # k negative, shift left
            g[kk:] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

Now worst_shift_gradient:

def worst_shift_gradient(h):
    n = h.shape[0]
    v = compute_corr_vec(h)
    max_v = np.max(v)
    idxs = np.where(v == max_v)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1.0))
    zeros = list(np.flatnonzero(cur_h == 0.0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- correlation utilities ----------
def compute_corr_vec(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h) using FFT."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discrete h."""
    n = h.size
    max_val = np.max(compute_corr_vec(h))
    return float(max_val * (2.0 / n))


# ---------- feasibility projection ----------
def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Modify h in‑place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- gradient of smooth‑max ----------
def gradient_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log∑exp(β·overlap)."""
    n = h.size
    v = compute_corr_vec(h)
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient for the currently worst shift(s)."""
    n = h.size
    v = compute_corr_vec(h)
    max_v = np.max(v)
    idxs = np.where(v == max_v)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g


# ---------- descent helpers ----------
def gradient_descent_smoothmax(
    h: np.ndarray,
    beta_sched,
    steps_sched,
    base_step: float,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """Projected gradient descent on the smooth‑max surrogate."""
    for beta, steps in zip(beta_sched, steps_sched):
        for _ in range(steps):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = h - base_step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h


def worst_shift_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    base_step: float,
    steps: int,
    deadline: float,
) -> np.ndarray:
    """Focused sub‑gradient steps on the worst shift."""
    step = base_step
    for _ in range(steps):
        if time.time() > deadline:

        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:

# ---------- SLSQP polishing ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary simulated annealing ----------
def binary_sa_hill_climb(h_bin: np.ndarray, rng: np.random.Generator, deadline: float):
    """Swap‑based SA on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1.0))
    zeros = list(np.flatnonzero(cur_h == 0.0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
        if it > 20000:
            break
    return best_h, best_c
