    w_i = w_pad[i0 + n -1 + k_range]

import time
import numpy as np

# ---------- helpers for continuous vectors ----------
def adjust_sum_exact(h, rng, max_iter=2000):
    """Modify h in-place so that sum(h) == n/2 exactly (within floating tolerance)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_smoothmax(h, beta):
    """Gradient of (1/β) log-sum-exp of β·overlap, where overlap_k = ∑ h_i (1-h_{i+k})."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)  # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g


def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    """Projected gradient descent on smooth‑max surrogate for a schedule of β."""
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_local = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_local = cand
            h = best_local
    return h

        h_bin[i] = 1
        cur += 1
    return h_bin


def binary_sa_fast(h_bin, deadline, rng):
    """
    Simulated‑annealing hill‑climb on a binary step function.
    Returns the best binary vector seen and its C₅ bound.
    """
    n = h_bin.size
    shift0 = n - 1
    # initial state
    cur_h = h_bin.copy()
    cur_w = 1 - cur_h
    Nfft = 1 << ((2 * n - 1).bit_length())
    fft_h = np.fft.fft(cur_h, Nfft)
    fft_w = np.fft.fft(cur_w, Nfft)
    cur_v = np.fft.ifft(fft_h * fft_w).real[: 2 * n - 1]
    cur_max = cur_v.max()
    best_h = cur_h.copy()
    best_max = cur_max

    cur_h_pad = np.pad(cur_h, (shift0, shift0), mode='constant')
    cur_w_pad = np.pad(cur_w, (shift0, shift0), mode='constant')
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(cur_w))
    k_range = np.arange(-n + 1, n)
    it = 0
    T0 = cur_max * 0.1 if cur_max > 0 else 0.01

    while time.time() < deadline:
        it += 1
        T = T0 * (0.9995 ** it)
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        w_i = cur_w_pad[i + shift0 + k_range]
        w_j = cur_w_pad[j + shift0 + k_range]
        h_i = cur_h_pad[i + shift0 - k_range]
        h_j = cur_h_pad[j + shift0 - k_range]

        delta_v = -w_i + w_j + h_i - h_j
        delta_v[shift0] -= 2
        d = j - i
        if d != 0:
            delta_v[shift0 + d] += 1
            delta_v[shift0 - d] += 1

        new_v = cur_v + delta_v
        new_max = new_v.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            # update state
            cur_h[i] = 0
            cur_h[j] = 1
            cur_w[i] = 1
            cur_w[j] = 0
            cur_v = new_v
            cur_max = new_max
            cur_h_pad = np.pad(cur_h, (shift0, shift0), mode='constant')
            cur_w_pad = np.pad(cur_w, (shift0, shift0), mode='constant')
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)

            if cur_max < best_max:
                best_max = cur_max
                best_h = cur_h.copy()

        if best_max * (2.0 / n) < 0.3806:   # early stop if already very good
            break

    return best_h.astype(float), best_max * (2.0 / n)


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    # choose number of points (must be <1000)
    n = 960

    # initialise binary vector
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        parent_arr = np.array(parent, dtype=float)
        if parent_arr.size != n:
            xs = np.linspace(0.0, 2.0, parent_arr.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            parent_arr = np.interp(xt, xs, parent_arr)
        h_bin = np.rint(parent_arr).astype(int)
        h_bin = adjust_binary_sum_exact(h_bin, rng)
    else:
        h_bin = np.zeros(n, dtype=int)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1

    best_h = None
    best_c = np.inf

    # repeated SA runs until time runs out or target reached
    while time.time() < deadline:
        # allocate a slice of time for this run
        slice_dead = min(deadline, time.time() + max(5.0, (deadline - time.time()) * 0.2))
        # random restart occasionally
        if best_h is not None and rng.random() < 0.3:
            start_vec = np.rint(best_h).astype(int)
            start_vec = adjust_binary_sum_exact(start_vec, rng)
        else:
            start_vec = np.zeros(n, dtype=int)
            start_vec[rng.choice(n, n // 2, replace=False)] = 1

        h_local, c_local = binary_sa_fast(start_vec, slice_dead, rng)
        if c_local < best_c:
            best_c, best_h = c_local, h_local.copy()
        if best_c < 0.3808:
            break

    # optional fractional refinement (few gradient steps) if time allows
    if best_h is not None and time.time() < deadline:
        h_float = best_h.astype(float)
        betas = [10, 50, 200, 500, 2000, 5000]
        h_float = gradient_descent_smoothmax(
            h_float,
            betas,
            base_step=0.2,
            it_per_beta=3,
            rng=rng,
            deadline=deadline,
        )
        c_refined = compute_c5(h_float)
        if c_refined < best_c:
            best_c, best_h = c_refined, h_float.copy()

    # final safety checks
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
