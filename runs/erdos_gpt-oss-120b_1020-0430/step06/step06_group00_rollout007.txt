    def gd_max_overlap(h, rng, max_iter, eta0, decay, time_limit):
        start = time.time()
        n = h.size
        target = n/2
        best_h = h.copy()
        best_c = compute_c5(h)
        eta = eta0
        for it in range(max_iter):
            if time.time() - start > time_limit:
                break
            # correlation and worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            k = abs(max_idx - (n - 1))
            # gradient
            grad = np.zeros_like(h)
            if k == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                grad[: n - k] = 1.0 - h[k:]
                grad[k:] -= h[: n - k]
            # update
            h_new = h - eta * grad
            # clip and enforce sum
            h_new = np.clip(h_new, 0.0, 1.0)
            # sum correction
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                # distribute diff proportionally to capacity
                # choose random indices repeatedly
                attempts = 0
                n = h_new.size
                while abs(diff) > 1e-12 and attempts < 10000:
                    i = rng.integers(0, n)
                    if diff > 0:
                        add = min(diff, 1.0 - h_new[i])
                        h_new[i] += add
                        diff -= add
                    else:
                        sub = min(-diff, h_new[i])
                        h_new[i] -= sub
                        diff += sub
                    attempts += 1
            # Evaluate
            c = np.max(corr) * (2.0 / n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
                # if very good break early:
                if best_c < 0.3807: # threshold
                    # we can return early
                    return best_h, best_c
            h = h_new
            # decay eta
            eta *= decay
        return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

# Soft‑max SLSQP helper (used for final polishing)
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

                  max_iter: int, eta0: float, decay: float,

    """Sub‑gradient descent on the maximal overlap functional."""
    start = time.time()
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    eta = eta0

    for it in range(max_iter):
        if time.time() - start > time_limit:

        c = np.max(corr) * (2.0 / n)

                    diff += sub
