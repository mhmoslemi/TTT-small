    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    T = 0.5
    it = 0
    while time.time() < deadline:
        # random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose transfer epsilon
        # Determine max eps that respects bounds
        max_eps = min(best_h[i], 1.0 - best_h[j]) # move from i to j
        if max_eps <= 0:
            # perhaps move opposite direction
            max_eps = min(best_h[j], 1.0 - best_h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * T
            nh = best_h.copy()
            nh[j] -= eps
            nh[i] += eps

            eps = rng.random() * max_eps * T

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # obtain initial small h0
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n_small = 512
        h0 = np.zeros(n_small)
        h0[rng.choice(n_small, n_small // 2, replace=False)] = 1.0
        h0 = enforce_sum(h0, rng)

    n_small = h0.size
    # choose n_big
    n_big = max(1024, n_small * 2)
    # ensure n_big multiple of n_small
    if n_big % n_small != 0:
        # increase to next multiple
        factor = int(np.ceil(1024 / n_small))
        n_big = n_small * factor
    repeat_factor = n_big // n_small
    base_h = np.repeat(h0, repeat_factor)

    # best record
    best_h = base_h.copy()
    best_c = compute_c5(best_h)

    # attempt multiple trials until time out
    while time.time() < deadline - 5.0:
        # allocate portion of remaining time
        remaining = deadline - time.time()
        trial_deadline = time.time() + remaining * 0.6

        # start from base + noise
        h = base_h.copy()
        # Add random noise
        noise_scale = 0.02
        h += rng.random(h.shape) * noise_scale
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

        # hill climb
        h, c = hill_climb(h, rng, trial_deadline)

        # gradient refine
        if time.time() < trial_deadline:
            h = gradient_refine(h, rng, step=0.02, max_iter=12)
            c = compute_c5(h)

        # SLSQP passes
        beta_seq = (20.0, 200.0, 2000.0, 20000.0, 80000.0)
        for beta in beta_seq:
            if time.time() > trial_deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=400)
            c = compute_c5(h)

        # final repair
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        c = compute_c5(h)

        if c < best_c:
            best_c = c
            best_h = h.copy()

    # final repair for best_h
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

def gradient_refine(h, rng, step=0.02, max_iter=10):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust diff using random indices
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def hill_climb(h: np.ndarray, rng: np.random.Generator,
               deadline: float) -> tuple:
    """Simulated‑annealing style hill climbing until deadline."""
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    T = 0.5
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            max_eps = min(cur_h[j], 1.0 - cur_h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * T
            nh = cur_h.copy()
            nh[j] -= eps
            nh[i] += eps
        else:
            eps = rng.random() * max_eps * T
            nh = cur_h.copy()
            nh[i] -= eps
            nh[j] += eps
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = nh, nc
        it += 1
        if it % 5000 == 0:
            T *= 0.995
            T = max(T, 1e-5)
    return cur_h, cur_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    # ----- initial small vector -----

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
