def project_onto_sum_and_box(v, target, eps=1e-10, max_iter=50):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1 - 2 * h
    elif k > 0:
        # k positive: O_k = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
        grad[:n - k] += 1 - h[k:]  # derivative w.r.t h[i] in first factor
        grad[k:] -= h[:n - k]      # derivative w.r.t h[i] in second factor (as -h[i-k])
    else:  # k < 0
        k = -k
        # O_{-k} = sum_{i=0}^{n-1-k} h[i + k] * (1 - h[i])
        # treat i' = i where first factor is h[i + k], second is (1 - h[i])
        # gradient w.r.t h[j]:
        # if j >= k: derivative from first factor: (1 - h[j - k])
        # if j <= n-1-k: derivative from second factor: -h[j + k]
        grad[k:] += 1 - h[:n - k]      # j >= k: + (1 - h[j - k])
        grad[:n - k] -= h[k:]          # j <= n-1-k: - h[j + k]
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with equally spaced points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the shift attaining the maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift giving the worst overlap
    k = idx
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        # O_k = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
        grad[: n - k] += 1.0 - h[k:]        # derivative w.r.t. first factor
        grad[k:]       -= h[: n - k]        # derivative w.r.t. second factor
    else:  # k < 0, work with -k
        k = -k
        # O_{-k} = sum_{i=0}^{n-1-k} h[i+k] * (1 - h[i])
        grad[k:]       += 1.0 - h[: n - k]  # first factor contribution
        grad[: n - k]  -= h[k:]             # second factor contribution
    return grad


# ----------------------------------------------------------------------
# Projection onto the hyperplane sum(h)=target intersected with [0,1]^n
# ----------------------------------------------------------------------
def project_onto_sum_and_box(v: np.ndarray, target: float, eps: float = 1e-12,
                             max_iter: int = 60) -> np.ndarray:
    """Find x = clip(v + λ, 0, 1) with Σ x_i = target via binary search on λ."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------------------
# Projected subgradient (gradient descent) optimizer
# ----------------------------------------------------------------------
def gradient_descent_best(h0: np.ndarray, target: float,
                         allotted_time: float, start_time: float):
    """Projected subgradient descent with exact projection."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.5
    while time.time() - start_time < allotted_time:
        g = worst_gradient(h)
        h_candidate = h - step * g
        h_candidate = project_onto_sum_and_box(h_candidate, target)
        c_candidate = compute_c5(h_candidate)

        if c_candidate < best_c - 1e-12:
            best_c = c_candidate
            best_h = h_candidate.copy()
            h = h_candidate
            step *= 1.02  # slight acceleration after improvement
        else:
            step *= 0.97  # back‑off when no improvement

        if step < 1e-8:
            step = 1e-8
        if step > 1.0:
            step = 1.0
        if best_c <= 0.3808:
            break
    return best_h, best_c


# ----------------------------------------------------------------------
# Soft‑max SLSQP refinement (same as previous implementation, but with
# exact projection instead of random fixing)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, numerically stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 1500):
    """Run SLSQP on the smooth max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = h0.size / 2.0
    CURRENT_BETA = beta
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_sum_and_box(x, CURRENT_TARGET)
    return x
