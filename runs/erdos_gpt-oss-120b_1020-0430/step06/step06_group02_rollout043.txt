def compute_c5_fft(h):
    n = h.size
    # pad to length 2n
    size = 2 * n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1 - h, size)
    # convolution (full) = ifft(H * conj(G))[0:size]
    conv = np.fft.irfft(H * np.conj(G), size)
    # The conv array of length size contains cross-correlation values at indices 0..size-1, where shift k corresponds to index (n-1 + k)
    # For our correlation using np.correlate, full mode returns length 2*n-1: values for k= -(n-1) .. (n-1)
    # conv length 2n includes an extra zero at the end (since padded). So we take conv[:2*n-1].
    corr = conv[:2*n - 1]
    return np.max(corr) * (2.0 / n)
```

def worst_gradient(h):
    n = h.size
    # compute corr to find worst shift (max O)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1 - 2 * h
    else:
        # grad = 1 - shift_plus - shift_minus
        # compute shift_plus: h shifted left by k (i.e., h_i+k)
        # shift_minus: h shifted right by k (i.e., h_{i-k})
        # For indices where missing, treat as 0.
        grad[:n - k] -= h[k:]        # i from 0 to n-k-1, subtract h[i+k]
        grad[k:] -= h[:n - k]        # i from k to n-1, subtract h[i - k]
    return grad
```

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # While diff not close to zero, adjust random indices.
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    # Ensure within bounds
    h = np.clip(h, 0.0, 1.0)
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]    -= h[: n - k]
    return grad

def slsqp_optimize(h0, beta, rng, max_iter=1000):
    n = h0.size
    target = n / 2.0
    # define closure variables as mutable via dictionary
    global CURRENT_BETA, CURRENT_TARGET
    CURRENT_BETA = beta
    CURRENT_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_BETA = None
CURRENT_TARGET = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (full correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the shift where the overlap is maximal."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 1000) -> np.ndarray:
    """SLSQP on the smooth soft‑max approximation for a given β."""
    global CURRENT_BETA, CURRENT_TARGET
    n = h0.size
    CURRENT_BETA = beta
    CURRENT_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # log‑sum‑exp trick
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x
