import time
import numpy as np

# ------------------ basic utilities ------------------
def compute_c5(h):
    """Maximum overlap integral (discrete)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def proj_box_sum(y, S, iters=60):
    """Projection of y onto {x : 0<=x<=1, sum x = S} via bisection on lambda."""
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)


def smoothmax_objective(h, beta):
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    return (m + np.log(np.sum(np.exp(beta * v - m)))) / beta


def gradient_smoothmax(h, beta):
    """Gradient of smooth‑max surrogate."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g


def worst_shift_gradient(h):
    """Sub‑gradient using the worst‑shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g


# ------------------ optimization steps ------------------
def grad_descent_proj(h, beta, step0, iters, rng, deadline):
    """Projected gradient descent on smooth‑max with backtracking line search."""
    S = h.size / 2.0
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = gradient_smoothmax(h, beta)
        f0 = smoothmax_objective(h, beta)
        step = step0
        for _ in range(10):
            cand = proj_box_sum(h - step * g, S)
            f1 = smoothmax_objective(cand, beta)
            if f1 < f0:
                h = cand
                break
            step *= 0.5
    return h


def refine_worst_shift(h, step0, max_iter, rng, deadline):
    """Sub‑gradient refinement focused on the current worst shift."""
    S = h.size / 2.0
    step = step0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        f0 = compute_c5(h)
        cand = proj_box_sum(h - step * g, S)
        f1 = compute_c5(cand)
        if f1 < f0:
            h = cand
            step *= 1.02
        else:
            step *= 0.7
    return h


def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill‑climb on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c


# ------------------ main run ------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # Choose base resolution
    base_n = 1440
    parent = globals().get("initial_h_values", None)
    if parent is not None and len(parent) != base_n:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
        parent = np.interp(xt, xs, parent)

    # Schedules
    betas = [5, 20, 80, 200, 500, 2000, 5000, 20000, 50000, 200000]
    steps_per_beta = [50, 50, 50, 50, 50, 80, 80, 80, 100, 120]

    restarts = 5
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent is not None:
            h = parent.copy()
        else:
            h = rng.random(n)
            h = proj_box_sum(h, n / 2.0)
        # smooth‑max descent
        for beta, nit in zip(betas, steps_per_beta):
            if time.time() > deadline:
                break
            h = grad_descent_proj(h, beta, step0=0.2, iters=nit,
                                 rng=rng, deadline=deadline)
        # worst‑shift refinement
        h = refine_worst_shift(h, step0=0.12, max_iter=30,
                               rng=rng, deadline=deadline)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # binary SA polish
        h_bin = np.rint(h).astype(float)
        h_bin = proj_box_sum(h_bin, n / 2.0)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # Optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 2880:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        h_up = proj_box_sum(h_up, n_up / 2.0)
        # short descent on up‑sampled version
        for beta, nit in zip(betas[:6], [30, 30, 30, 30, 30, 50]):
            if time.time() > deadline:
                break
            h_up = grad_descent_proj(h_up, beta, step0=0.15, iters=nit,
                                     rng=rng, deadline=deadline)
        h_up = refine_worst_shift(h_up, step0=0.09, max_iter=20,
                                  rng=rng, deadline=deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up
        # binary polish on up‑sampled
        h_bin = np.rint(h_up).astype(float)
        h_bin = proj_box_sum(h_bin, n_up / 2.0)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n_up

    # final safety
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = proj_box_sum(best_h, n / 2.0)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
