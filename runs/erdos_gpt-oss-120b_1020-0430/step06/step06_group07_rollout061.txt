def worst_shift_fft(h):

    # compute autocorrelation using FFT
    m = 2 * n - 1

import time
import numpy as np

# --------------------------------------------------------------
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum x = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the box with given sum."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def compute_c5_fft(h: np.ndarray) -> float:
    """C5 from autocorrelation (FFT, no full correlation)."""
    n = h.shape[0]
    m = 2 * n - 1
    H = np.fft.rfft(h, n=m)
    power = H * np.conj(H)
    corr_full = np.fft.irfft(power, n=m)
    cross = corr_full[n - 1 :]          # length n, cross[k] = Σ h_i h_{i+k}
    min_cross = cross.min()
    return 1.0 - (2.0 * min_cross) / n


def worst_shift_fft(h: np.ndarray):
    """Return worst shift index, its gradient for O_k, and the full cross array."""
    n = h.shape[0]
    m = 2 * n - 1
    H = np.fft.rfft(h, n=m)
    power = H * np.conj(H)
    corr_full = np.fft.irfft(power, n=m)
    cross = corr_full[n - 1 :]
    k = int(np.argmin(cross))
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad, cross


def hill_climb_adaptive(h: np.ndarray, rng: np.random.Generator,
                        iters: int, eps_start: float = 0.5,
                        decay: float = 0.999) -> tuple:
    """Hill‑climb that reduces max O_k (i.e. raises min overlap)."""
    n = h.size
    eps = eps_start
    k, grad, cross = worst_shift_fft(h)
    best_h = h.copy()
    best_c = 1.0 - (2.0 * cross.min()) / n

    for _ in range(iters):
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            # pick any admissible pair
            pos = np.where(h > 1e-12)[0]
            neg = np.where(h < 1.0 - 1e-12)[0]
            if pos.size == 0 or neg.size == 0:
                break
            i = int(rng.choice(pos))
            j = int(rng.choice(neg))
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                break

        eps_val = rng.random() * max_eps * eps
        h[i] -= eps_val
        h[j] += eps_val

        # recompute worst shift for next iteration
        k, grad, cross = worst_shift_fft(h)
        c = 1.0 - (2.0 * cross.min()) / n
        if c < best_c:
            best_c, best_h = c, h.copy()
        eps *= decay

    return best_h, best_c


def multi_start(rng: np.random.Generator, time_limit: float,
                init_best_h=None, init_best_c=np.inf) -> tuple:
    """Many short restarts with cheap hill‑climb."""
    best_h = init_best_h
    best_c = init_best_c
    n = 512
    while time.time() < time_limit:
        h = random_feasible(n, rng)
        h, c = hill_climb_adaptive(h, rng, iters=800,
                                   eps_start=0.5, decay=0.997)
        if c < best_c:
            best_h, best_c = h, c
    return best_h, best_c


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    end_time = start + budget_s

    # initial best (maybe provided)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, h0.size / 2.0)
        best_h = h0
        best_c = compute_c5_fft(best_h)
    else:
        best_h = None
        best_c = np.inf

    # ----------------------------------------------------------
    # Phase 1: many quick restarts
    phase1_end = start + budget_s * 0.70
    best_h, best_c = multi_start(rng, phase1_end, best_h, best_c)

    # ----------------------------------------------------------
    # Phase 2: longer local refinement on the current best
    remaining = end_time - time.time()
    if remaining > 5.0:
        iters = int(min(40000, remaining * 4000))
        h_ref, c_ref = hill_climb_adaptive(best_h.copy(), rng,
                                           iters=iters,
                                           eps_start=0.4, decay=0.999)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----------------------------------------------------------
    # Phase 3: up‑sample and fine‑tune (if size permits)
    remaining = end_time - time.time()
    if remaining > 5.0 and best_h.size * 2 <= 1024:
        n2 = best_h.size * 2
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_box_sum(h_up, n2 / 2.0)

        iters = int(min(60000, remaining * 5000))
        h_up_ref, c_up_ref = hill_climb_adaptive(h_up, rng,
                                                 iters=iters,
                                                 eps_start=0.3,
                                                 decay=0.9995)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref

    # ----------------------------------------------------------
    # final sanitisation
    n = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(n)
