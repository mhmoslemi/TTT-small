        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial feasible h -------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        if n0 < 1024:
            factor = 1024 // n0
            h = np.repeat(base, factor)
        else:
            h = base.copy()
    else:
        n0 = 1024
        h = np.zeros(n0)
        t0 = n0 // 2
        h[rng.choice(n0, t0, replace=False)] = 1.0

    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _proj_box_sum(h, target)

    cur_h = h.copy()
    cur_c = _c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # ---- simulated annealing ------------------------------------------------
    T0, T_end = 0.5, 1e-4
    max_iters = 5_000_000
    it = 0
    while it < max_iters:
        if time.time() - start > budget_s * 0.92:
            break
        t = T0 * (T_end / T0) ** (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue

        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            it += 1
            continue

        eps = (rng.random() * 2.0 - 1.0) * max_eps * t
        if abs(eps) < 1e-15:
            it += 1
            continue

        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = _c5_fft(nh)

        if nc < cur_c or rng.random() < math.exp(-(nc - cur_c) / t):
            cur_h, cur_c = nh, nc

        if nc < best_c:
            best_h, best_c = nh.copy(), nc

        # periodic re‑projection to eliminate drift
        if (it & 1023) == 0:
            cur_h = np.clip(cur_h, 0.0, 1.0)
            if abs(cur_h.sum() - target) > 1e-9:
                cur_h = _proj_box_sum(cur_h, target)

        it += 1

    # ---- gradient polishing -------------------------------------------------
    for r in range(2000):
        if time.time() - start > budget_s * 0.98:
            break
        grad = _worst_gradient(best_h)
        step = 0.005 / (r + 1)
        v = best_h - step * grad
        best_h = _proj_box_sum(v, target)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj_box_sum(best_h, target)

    final_c = _c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
