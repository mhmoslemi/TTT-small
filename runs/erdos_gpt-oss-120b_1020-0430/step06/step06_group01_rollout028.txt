def enforce_sum(h, rng=None):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = step_scale * max_eps
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

# globals for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple random redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    c_curr = best_c
    ones_arr = np.copy(ones)
    zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_idx = rng.integers(len(ones_arr))
        j_idx = rng.integers(len(zeros_arr))
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]
        # swap
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
        if accept:
            c_curr = c_new
            # update index arrays
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = step_scale * max_eps
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    c_curr = best_c
    ones_arr = np.copy(ones)
    zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_idx = rng.integers(len(ones_arr))
        j_idx = rng.integers(len(zeros_arr))
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
        if accept:
            c_curr = c_new
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

def continuous_optimize(n: int, rng: np.random.Generator,
                       time_budget: float, init_h: np.ndarray = None):
    start = time.time()
    if init_h is not None:
        h = init_h.copy()
    else:
        h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    remaining = time_budget - (time.time() - start)
    max_iters = int(min(40000, remaining * 400))
    h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
    best_h, best_c = h, c

    beta_seq = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in beta_seq:
        if time.time() - start > time_budget * 0.94:
            break
        h = slsqp_optimize(best_h, beta, rng, max_iter=500)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    if time.time() - start < time_budget * 0.96:
        h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    while time.time() - start < time_budget * 0.99:
        h = best_h.copy()
        for _ in range(8):
            i = rng.integers(0, n)
            delta = (rng.random() - 0.5) * 0.3
            h[i] = np.clip(h[i] + delta, 0.0, 1.0)
        h = enforce_sum(h, rng)
        sub_budget = (time_budget - (time.time() - start)) * 0.2
        if sub_budget <= 0:
            break
        max_iters = int(min(8000, sub_budget * 400))
        h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
        if c < best_c:

                c2 = compute_c5(h2)
                if c2 < best_c:

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
