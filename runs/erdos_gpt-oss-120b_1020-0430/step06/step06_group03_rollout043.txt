import time, math, numpy as np

def _project(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _c5(v):
    n = v.size
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * (2.0 / n)

def _softmax_grad(x, beta):
    n = x.size
    O = np.correlate(x, 1.0 - x, mode='full')
    Omax = O.max()
    w = np.exp(beta * (O - Omax))
    w /= w.sum()
    grad = np.zeros_like(x)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * x)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * x[k:]
        grad[k:] -= wk * x[: n - k]
    return grad

def _worst_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    g = np.ones_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    return g

def _guided_perturb(h, g, rng, scale=0.5):
    i = int(np.argmax(g))
    j = int(np.argmin(g))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def _guided_hill(h, rng, iters=2000, step=0.5):
    best_h = h.copy()
    best_c = _c5(h)
    decay = max(1, iters // 5)
    s = step
    for t in range(iters):
        g = _worst_grad(h)
        nh = _guided_perturb(h, g, rng, scale=s)
        c = _c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (t + 1) % decay == 0:
            s = max(s * 0.7, 1e-4)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h = rng.random(n)
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _project(h, target)

    best_h = h.copy()
    best_c = _c5(best_h)

    # ----- annealing over beta ---------------------------------------
    beta_list = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    time_opt = budget_s * 0.85
    for beta in beta_list:
        if time.time() - start > time_opt:
            break
        # iterations grow with smoothness; 7500 for n=512 is a good baseline
        iters = int(7500 * 256 / n)
        step0 = 0.4
        h_cur = best_h.copy()
        for it in range(iters):
            if time.time() - start > time_opt:
                break
            grad = _softmax_grad(h_cur, beta)
            ng = np.linalg.norm(grad)
            if ng == 0:
                step = 0.0
            else:
                step = step0 * (1.0 - it / iters) / ng
            v = h_cur - step * grad
            h_cur = _project(v, target)
            if (it + 1) % max(1, iters // 10) == 0:
                c = _c5(h_cur)
                if c < best_c:
                    best_c, best_h = c, h_cur.copy()
        c = _c5(h_cur)
        if c < best_c:
            best_c, best_h = c, h_cur.copy()

        # short hill climb after each beta stage
        if time.time() - start < budget_s * 0.90:
            hc, cc = _guided_hill(best_h, rng, iters=800, step=0.5)
            if cc < best_c:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, target)
    final_c = _c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
