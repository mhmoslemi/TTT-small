def binary_sa_optimize(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    # initial temperature as a fraction of current c
    T0 = cur_c * 0.02 + 1e-8
    T = T0
    decay = 0.9996
    it = 0
    while time.time() < deadline and it < 15000:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            # accept
            cur_c = c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                # reset temperature maybe?
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        T = T * decay
        if T < 1e-12:
            T = 1e-12
    return best_h, best_c

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    n = h.size
    best_h = h
    best_c = compute_c5(h)
    for beta in betas:
        for it in range(it_per_beta):
            if time.time() > deadline:
                return best_h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:

            for mult in (1.0, 0.5, 0.25, 0.75, 1.5):

                np.clip(cand, 0.0, 1.0, out=cand)

                if c < best_c:
                    best_c = c
                    best_h = cand.copy()
                    h = cand
                    # maybe break early?
            # optional: if candidate improved, continue from improved h
    return best_h

import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    w = np.exp(beta * v - m)
    w /= w.sum()
    g = np.zeros_like(h)
    onemh = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * onemh[k:]
            g[k:]    -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:]   += wi * onemh[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

# ---------- gradient descent ----------
def gradient_descent_smoothmax(h, betas, step0, it_per_beta, rng, deadline):
    n = h.size
    best_c = compute_c5(h)
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            improved = False
            for mult in (1.0, 0.5, 0.25, 0.75, 1.5):
                step = step0 * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, h = c, cand
                    improved = True
            if not improved:
                # tiny random jitter to escape stagnation
                eps = 0.005
                mask = rng.random(n) < eps
                cand = h.copy()
                cand[mask] += rng.uniform(-eps, eps, size=mask.sum())
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, h = c, cand
    return h

# ---------- binary simulated annealing ----------
def binary_sa_optimize(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    T = max(1e-8, cur_c * 0.02)
    decay = 0.9996
    it = 0
    while time.time() < deadline and it < 15000:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        c = compute_c5(cur_h)
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
        it += 1
        T = max(T * decay, 1e-12)
    return best_h, best_c

# ---------- SLSQP polish ----------
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    overall_deadline = start + budget_s * 0.97
    # small and large resolutions
    n_small, n_large = 360, 720
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
        parent_small = np.interp(xt_small, xs, parent)
    else:
        parent_small = None

    best_small_h, best_small_c = None, np.inf
    # ---------- Stage 1: explore at small resolution ----------
    while time.time() < overall_deadline - 80:  # reserve time for stage 2
        # initialise
        if best_small_h is None and parent_small is not None:
            h = parent_small.copy()
        else:
            h = rng.random(n_small)
            h *= (n_small / 2) / h.sum()
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)
        # gradient descent
        betas = [5, 20, 80, 200, 500, 2000, 5000, 20000]
        h = gradient_descent_smoothmax(h, betas, step0=0.2,
                                       it_per_beta=30, rng=rng,
                                       deadline=overall_deadline)
        # binary SA
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        h_bin, c_bin = binary_sa_optimize(h_bin, rng, overall_deadline)
        # choose better of fractional / binary
        c_frac = compute_c5(h)
        if c_bin < c_frac:
            h, c_frac = h_bin, c_bin
        # SLSQP polishing
        for beta in (20000, 50000):
            if time.time() > overall_deadline - 80:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=600)
        c = compute_c5(h)
        if c < best_small_c:
            best_small_c, best_small_h = c, h.copy()
    # ---------- Stage 2: upsample and refine ----------
    if best_small_h is not None and time.time() < overall_deadline:
        # upsample by repetition (large is multiple of small)
        factor = n_large // n_small
        h = np.repeat(best_small_h, factor)
        adjust_sum_exact(h, rng)
        # gradient descent on large
        betas = [5, 20, 80, 200, 500, 2000, 5000, 20000]
        h = gradient_descent_smoothmax(h, betas, step0=0.2,
                                       it_per_beta=40, rng=rng,
                                       deadline=overall_deadline)
        # binary SA on large
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        h_bin, c_bin = binary_sa_optimize(h_bin, rng, overall_deadline)
        c_frac = compute_c5(h)
        if c_bin < c_frac:
            h, c_frac = h_bin, c_bin
        # final SLSQP polish
        for beta in (20000, 50000, 100000):
            if time.time() > overall_deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800)
        best_h, final_c = h, compute_c5(h)
        best_n = n_large
    else:
        # fallback simple half‑ones
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        final_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
