def worst_gradient(h):
    n = h.shape[0]
    # compute autocorrelation cross: sum h_i * h_{i+k} for k >= 0
    cross = np.correlate(h, h, mode="full")[n-1:] # len n
    k = int(np.argmin(cross))
    # gradient: 1 - h_{i+k} - h_{i-k}, missing neighbours treated as 0
    grad = np.ones(n)
    # subtract h_{i+k} where i+k < n
    grad[:n-k] -= h[k:] # shift +k
    # subtract h_{i-k} where i-k >=0 -> i >= k
    grad[k:] -= h[:n-k] # shift -k
    return grad, k

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def upsample(h, factor=2):
    n = h.size
    n2 = n * factor
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    return project_box_sum(h2, n2 / 2.0)

def worst_gradient(h):
    n = h.size
    cross = np.correlate(h, h, mode='full')[n - 1:]  # len n
    k = int(np.argmin(cross))
    grad = np.ones(n)
    if k != 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    return grad, k

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    scale = step_scale
    for it in range(max_iters):
        grad, _ = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            break
        eps = max_eps * rng.random() * scale
        cur_h[i] -= eps
        cur_h[j] += eps
        c_new = compute_c5(cur_h)
        if c_new < best_c:
            best_c, best_h = c_new, cur_h.copy()
        if c_new <= cur_c:
            cur_c = c_new
        else:
            cur_h[i] += eps
            cur_h[j] -= eps
        if (it + 1) % 500 == 0:
            scale = max(scale * 0.9, 0.001)
    return best_h, best_c

def sa_optimize(h0, target, time_budget, rng):
    n = h0.size
    cur_h = h0.copy()

    best_h, best_c = cur_h.copy(), cur_c
    # temperature proportional to current best
    temp = max(cur_c * 0.1, 1e-3)
    move_scale = 0.5
    start = time.time()
    last_improve = 0
    it = 0
    while time.time() - start < time_budget:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * move_scale
        cur_h[i] -= eps
        cur_h[j] += eps
        c_new = compute_c5(cur_h)
        delta = c_new - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = cur_h.copy()
                last_improve = it
                # adapt temperature to new best if lower
                temp = max(best_c * 0.1, 1e-4)
        else:

            cur_h[j] -= eps

        if it - last_improve > 200000:
            cur_h = random_feasible(n, rng)
            cur_c = compute_c5(cur_h)
            last_improve = it
    return best_h, best_c

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} using binary search on the Lagrange multiplier."""
    lo, hi = -1.0, 1.0

            hi = lam
