def hill_climb_refine(h, rng, max_iters, time_budget):
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() - start >= time_budget:
            break
        # random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # compute max delta that can be moved from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        # sample delta scaled
        delta = rng.random() * max_delta
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # compute new C5
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
            # If good enough, break early
            if best_c5 <= 0.3808:
                break
    return best_h, best_c5

    N = 2 * n
    H = np.fft.rfft(h, n=N)
    Hr = np.fft.rfft(h[::-1], n=N)
    conv = np.fft.irfft(H * Hr, n=N)
    return conv[n - 1 : n - 1 + n]  # length n


def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx = 1 - 2*min_c / n."""
    n = h.shape[0]
    min_c = np.min(linear_autocorr(h))
    return 1.0 - 2.0 * min_c / n


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by binary search on a shift λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


# --------------------------------------------------------------
def subgrad_maxmin_corr(
    h: np.ndarray, target_sum: float, time_budget: float, rng: np.random.Generator
):
    """Projected sub‑gradient ascent on min_k c(k). Returns (best_h, best_min_c)."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_min_c = np.min(linear_autocorr(h))
    t = 0
    eps = 1e-12

    while time.time() - start < time_budget:
        t += 1
        cross = linear_autocorr(h)
        min_c = np.min(cross)
        ks = np.where(cross <= min_c + eps)[0]  # include k=0 if it is minimal
        grad = np.zeros_like(h)
        for k in ks:
            grad += np.roll(h, k) + np.roll(h, -k)
        grad /= len(ks)

        alpha = 2.0 / np.sqrt(t + 10.0)
        h_new = project_box_sum(h + alpha * grad, target_sum)

        new_min_c = np.min(linear_autocorr(h_new))
        if new_min_c > best_min_c:
            best_min_c = new_min_c
            best_h = h_new.copy()
            # early exit if we already beat the target bound
            if (1.0 - 2.0 * best_min_c / n) <= 0.3808:
                h = h_new
                break
        h = h_new

    return best_h, best_min_c


# --------------------------------------------------------------
def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation to double the number of points, then re‑project."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)


def hill_climb_refine(
    h: np.ndarray, rng: np.random.Generator, max_iters: int, time_budget: float
):
    """Simple random mass‑transfer hill climbing."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(max_iters):
        if time.time() - start >= time_budget:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
            if best_c5 <= 0.3808:
                break

    return best_h, best_c5


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ---- initialise -------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = random_feasible(n, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Phase 1: multiple sub‑gradient restarts -------------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        sub_total = remaining * 0.6
        # aim for ~20 s per restart, cap at 10 restarts
        num_restarts = max(1, int(sub_total // 20.0))
        num_restarts = min(num_restarts, 10)
        per_budget = sub_total / num_restarts

        for _ in range(num_restarts):
            # random start
            h_start = random_feasible(n, rng)
            h_opt, min_c = subgrad_maxmin_corr(
                h_start, n / 2.0, per_budget, rng
            )
            c5 = 1.0 - 2.0 * min_c / n
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt.copy()

            # slight refinement from current best (if time permits)
            if time.time() - start < total_budget * 0.95:
                h_opt2, min_c2 = subgrad_maxmin_corr(
                    best_h, n / 2.0, per_budget, rng
                )
                c5_2 = 1.0 - 2.0 * min_c2 / n
                if c5_2 < best_c5:
                    best_c5, best_h = c5_2, h_opt2.copy()
            if time.time() - start >= total_budget * 0.95:
                break

    # ---- Phase 2: up‑sampling and fine‑graining -------------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 8.0 and n <= 512:
        h_up = upsample(best_h, rng)
        n_up = h_up.size

        sub_budget = remaining * 0.6
        h_up_opt, min_c_up = subgrad_maxmin_corr(
            h_up, n_up / 2.0, sub_budget, rng
        )
        c5_up = 1.0 - 2.0 * min_c_up / n_up
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_up_opt.copy()

        # final hill‑climb on the refined grid
        remaining = total_budget - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(1e5)
            h_ref, c5_ref = hill_climb_refine(
                best_h, rng, max_iters, remaining
            )
            if c5_ref < best_c5:
                best_c5, best_h = c5_ref, h_ref.copy()

    # ---- final sanitisation -----------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
