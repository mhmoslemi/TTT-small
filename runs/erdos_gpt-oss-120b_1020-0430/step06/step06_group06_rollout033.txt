def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum_float(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)=n/2 while staying in [0,1]."""
    n = h.size

        # shifted version (right by k_abs, zero‑padded)
        hk = np.empty(n, dtype=h.dtype)
        hk[: n - k_abs] = h[k_abs:]
        hk[n - k_abs :] = 0
        A = np.where((h == 1) & (hk == 0))[0]
        B = np.where((h == 0) & (hk == 1))[0]

        if A.size and B.size:
            m = min(A.size, B.size)
            i_sel = rng.choice(A, size=m, replace=False)
            j_sel = rng.choice(B, size=m, replace=False)
            h[i_sel], h[j_sel] = h[j_sel], h[i_sel]  # batch swap
            new_c = compute_c5(h)
            if new_c <= cur_c:
                cur_c = new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h.copy()
                if best_c <= 0.3808:
                    break
            else:
                # revert batch
                h[i_sel], h[j_sel] = h[j_sel], h[i_sel]
        else:
            # random opposite‑value swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if ones_idx.size and zeros_idx.size:
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                h[i], h[j] = h[j], h[i]
                new_c = compute_c5(h)
                if new_c <= cur_c:
                    cur_c = new_c
                    if new_c < best_c:
                        best_c, best_h = new_c, h.copy()
                else:
                    h[i], h[j] = h[j], h[i]

    # ---- SLSQP refinement (remaining time) ----------------------------------
    if time.time() < deadline - 5:
        # start from the best binary solution, allow fractional improvement
        h_float = best_h.astype(float)
        beta_seq = (20.0, 200.0, 2000.0, 20000.0, 100000.0)
        for beta in beta_seq:
            if time.time() > deadline - 5:
                break
            h_float = slsqp_optimize(h_float, beta, rng, max_iter=400)
            cur_c = compute_c5(h_float)
            if cur_c < best_c:
                best_c, best_h = cur_c, h_float.copy()

    # ---- final repair & return ---------------------------------------------
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = enforce_sum_float(final_h, rng)
    final_c = compute_c5(final_h)
    return final_h.tolist(), float(final_c), int(n)
