def project_onto_box_simplex(v, target):
    # v is ndarray
    # We find lambda such that sum(clip(v+lam, 0,1)) = target
    # monotonic in lam, can binary search.
    lo = -1.0
    hi = 1.0
    for _ in range(30):
        lam = (lo+hi)/2
        y = np.clip(v + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)
```

def slsqp_refine(h0, target, rng, beta=5000.0, max_iter=300):
    n = h0.size
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def eq(x):
        return np.sum(x) - target
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': eq}
    res = minimize(obj, np.clip(h0,0.0,1.0), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # enforce sum
    x = proj_box_simplex(x, target)
    return x

import time
import numpy as np

def compute_c5_and_k(h: np.ndarray):
    """Return (C5 value, shift k attaining the maximum) for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    c = corr[idx] * (2.0 / h.size)
    k = abs(idx - (h.size - 1))
    return float(c), int(k)

def compute_c5(h: np.ndarray) -> float:
    """C5 only (used for final checks)."""
    c, _ = compute_c5_and_k(h)
    return c

def proj_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the hyperplane sum x = target with box constraints 0 ≤ x ≤ 1."""
    lo = -np.max(v)            # makes all entries ≤ 0 after adding lo
    hi = 1.0 - np.min(v)       # makes all entries ≥ 1 after adding hi
    for _ in range(40):
        lam = (lo + hi) * 0.5
        y = np.clip(v + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def projected_gradient_descent(h0: np.ndarray, target: float,
                               rng: np.random.Generator, max_iter: int = 200):
    """Iteratively decrease the worst‑shift overlap using projected gradient steps."""
    h = h0.copy()
    c, k = compute_c5_and_k(h)
    n = h.size
    for _ in range(max_iter):
        if k == 0 or c < 0.38079:
            break
        grad = np.ones(n)
        if k > 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        step = 1.0
        improved = False
        for _ in range(20):
            v = h - step * grad
            h_new = proj_box_simplex(v, target)
            c_new, k_new = compute_c5_and_k(h_new)
            if c_new < c - 1e-12:
                h, c, k = h_new, c_new, k_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # small random transfer as a fallback
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                eps = min(h[i], 1.0 - h[j]) * 0.1
                if eps > 1e-15:
                    h_tmp = h.copy()
                    h_tmp[i] -= eps
                    h_tmp[j] += eps
                    h_tmp = proj_box_simplex(h_tmp, target)
                    c_new, k_new = compute_c5_and_k(h_tmp)
                    if c_new < c - 1e-12:
                        h, c, k = h_tmp, c_new, k_new
        # loop continues
    return h, c

def slsqp_refine(h0: np.ndarray, target: float,
                 rng: np.random.Generator, beta: float = 5000.0,
                 max_iter: int = 300) -> np.ndarray:
    """One SLSQP pass on a smooth max (soft‑max) approximation."""
    n = h0.size

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def eq(x):
        return np.sum(x) - target

    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq}
    start = np.clip(h0, 0.0, 1.0)
    res = minimize(obj, start, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = proj_box_simplex(x, target)
    return x
