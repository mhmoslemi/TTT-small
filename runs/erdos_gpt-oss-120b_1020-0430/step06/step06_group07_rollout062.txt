def compute_c5_fft(h):
    n = h.size
    H = np.fft.rfft(h, n=2*n)
    G = np.fft.rfft(1.0 - h, n=2*n)
    conv = np.fft.irfft(H * G, n=2*n)[:2*n-1]
    return conv.max() * (2.0 / n)

def simulated_annealing(n, rng, start_time, total_budget):
    # initial binary h
    h = random_binary_feasible(n, rng)
    curr_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = curr_c
    # schedule parameters
    # total steps based on time
    max_steps = int(2e5)  # cap
    # compute steps allowed by time
    remaining = total_budget - (time.time() - start_time)
    # approximate time per step: use a quick measurement?
    # We'll just limit by max_steps
    # temperature schedule: T = T0 * (1 - step / max_steps)
    T0 = 0.05
    for step in range(max_steps):
        # check time
        if time.time() - start_time > total_budget * 0.95:
            break
        # propose swap
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h <= 0.5)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5_fft(h)
        delta = new_c - curr_c  # positive if worse
        T = T0 * (1 - step / max_steps)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

import time
import numpy as np

# ----------------------------------------------------------------------
def compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 using FFT (h on [0,2] with step size 2/n)."""
    n = h.size
    n2 = 2 * n
    H = np.fft.rfft(h, n=n2)
    G = np.fft.rfft(1.0 - h, n=n2)
    conv = np.fft.irfft(H * G, n=n2)[: n2 - 1]
    return conv.max() * (2.0 / n)


def random_binary_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    ones = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[ones] = 1.0
    return h


def shift_array(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift with zero padding (positive k shifts to the right)."""
    n = arr.size
    if k >= 0:
        res = np.empty_like(arr)
        if k:
            res[:k] = 0.0
        res[k:] = arr[: n - k]
    else:
        s = -k
        res = np.empty_like(arr)
        if s:
            res[-s:] = 0.0
        res[: n - s] = arr[s:]
    return res


def greedy_improve(
    h: np.ndarray,
    max_iters: int,
    rng: np.random.Generator,
    start_time: float,
    total_budget: float,
) -> tuple[np.ndarray, float]:
    """Iteratively improve worst shift by moving a 1 to a 0."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    for it in range(max_iters):
        if time.time() - start_time > total_budget * 0.95:
            break

        # autocorrelation Q_k
        n2 = 2 * n
        H = np.fft.rfft(h, n=n2)
        Q = np.fft.irfft(H * np.conj(H), n=n2)[: n2 - 1]

        S = n / 2.0
        O = S - Q
        k_idx = int(np.argmax(O))
        shift = k_idx - (n - 1)

        a = shift_array(h, shift)
        b = shift_array(h, -shift)
        s = a + b

        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h <= 0.5)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break

        i = ones_idx[np.argmin(s[ones_idx])]
        j = zeros_idx[np.argmax(s[zeros_idx])]
        delta = s[j] - s[i]

        if delta <= 1e-12:
            if rng.random() < 0.01:
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
            else:
                break

        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5_fft(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c


def simulated_annealing(
    n: int,
    rng: np.random.Generator,
    start_time: float,
    total_budget: float,
    *,
    init_h: np.ndarray | None = None,
    max_steps: int = 200_000,
    T0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Binary simulated annealing on C5."""
    if init_h is not None:
        h = np.where(init_h > 0.5, 1.0, 0.0).copy()
    else:
        h = random_binary_feasible(n, rng)

    curr_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = curr_c

    for step in range(max_steps):
        if time.time() - start_time > total_budget * 0.95:
            break

        # propose a random 1↔0 swap
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5_fft(h)
        delta = new_c - curr_c
        T = T0 * (1.0 - step / max_steps)

        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h on [0,2] with a small C5 bound."""
    start = time.time()
    total_budget = budget_s
    rng = np.random.default_rng(seed)

    # work with 1024 samples (still < 1k points requirement)
    n = 1024
    best_h = None
    best_c = np.inf

    # ---------- Phase 1: long simulated annealing ----------
    remaining = total_budget - (time.time() - start)
    if remaining > 20.0:
        steps = int(min(800_000, remaining * 2500))
        h1, c1 = simulated_annealing(
            n, rng, start, total_budget, max_steps=steps, T0=0.05
        )
        best_h, best_c = h1, c1

    # ---------- Phase 2: refinement annealing ----------
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0 and best_h is not None:
        steps = int(min(500_000, remaining * 2500))
        h2, c2 = simulated_annealing(
            n,
            rng,
            start,
            total_budget,
            init_h=best_h,
            max_steps=steps,
            T0=0.01,
        )
        if c2 < best_c:
            best_h, best_c = h2, c2

    # ---------- Phase 3: greedy hill‑climb ----------
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        iters = int(min(3000, remaining * 250))
        h3, c3 = greedy_improve(best_h.copy(), iters, rng, start, total_budget)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # ---------- Final sanitisation ----------
    if best_h is None:
        best_h = random_binary_feasible(n, rng)

    best_h = np.clip(best_h, 0.0, 1.0)

    # enforce exact sum = n/2 (binary ensures it, but fix possible drift)
    cur_sum = best_h.sum()
    target = n / 2.0
    if abs(cur_sum - target) > 1e-9:
        diff = int(round(target - cur_sum))
        if diff > 0:
            zeros = np.where(best_h < 0.5)[0]
            pick = rng.choice(zeros, diff, replace=False)
            best_h[pick] = 1.0
        elif diff < 0:
            ones = np.where(best_h > 0.5)[0]
            pick = rng.choice(ones, -diff, replace=False)
            best_h[pick] = 0.0

    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(n)
