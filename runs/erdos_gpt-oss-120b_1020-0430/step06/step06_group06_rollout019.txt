function project_h(h):
    n = h.size
    target = n/2
    # clip
    h = np.clip(h, 0.0, 1.0)
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) < 1e-12:
        return h
    # if diff > 0, need to add diff to some indices not at 1.
    # We'll use water-filling:
    if diff > 0:
        # capacities to increase
        caps = 1.0 - h
        # sort indices

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP
CUR_TARGET = None
CUR_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_h(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum = N/2 using random transfers."""
    n = h.size
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # random water‑filling
    for _ in range(5000):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def gradient_step(h: np.ndarray, beta: float, step: float,
                  rng: np.random.Generator) -> np.ndarray:
    """One projected gradient step on the smoothed max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    exp_vals = np.exp(beta * (corr - maxc))
    w = exp_vals / exp_vals.sum()
    w_rev = w[::-1]

    L = 1 << ((3 * n - 2).bit_length())          # power of two ≥ 3n‑2
    wr = np.fft.rfft(w, n=L)
    wr_rev = np.fft.rfft(w_rev, n=L)
    hr = np.fft.rfft(h, n=L)

    conv_full = np.fft.irfft(wr * hr, n=L)[: 3 * n - 2]
    conv_rev = np.fft.irfft(wr_rev * hr, n=L)[: 3 * n - 2]

    off = 2 * n - 2
    sum_minus = conv_full[off : off + n]   # Σ w_k h[i‑k]
    sum_plus = conv_rev[off : off + n]    # Σ w_k h[i+k]

    grad = 1.0 - (sum_plus + sum_minus)
    h_new = h - step * grad
    return project_h(h_new, rng)


def random_move(h: np.ndarray, rng: np.random.Generator,
                scale: float) -> np.ndarray:
    """Transfer a random amount between two positions while preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    nh = h.copy()
    max_eps = min(nh[i], 1.0 - nh[j])
    if max_eps <= 0.0:
        max_eps = min(nh[j], 1.0 - nh[i])
        if max_eps <= 0.0:
            nh[i], nh[j] = nh[j], nh[i]
            return nh
        eps = rng.random() * max_eps * scale
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_feasible(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a random vector in [0,1]^n with exact sum = n/2."""
    h = rng.random(n)
    return project_h(h, rng)


# --- SLSQP helpers ----------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CUR_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CUR_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CUR_BETA


def slsqp_optimize(h: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:

        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_h(x, rng)
