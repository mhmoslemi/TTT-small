def smooth_min_descent(h, target, time_budget, rng,
                       step0=0.5, beta0=5.0, beta_factor=2.0,
                       beta_step=20000, check_interval=1000):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    t = 0
    beta = beta0
    # precompute fft length for each iteration maybe constant given n
    n_fft = 1 << ((3*n-2).bit_length())
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        # compute cross
        cross = np.correlate(h, h, mode='full')[n-1:]
        # compute w
        w = np.exp(-beta * cross)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute w_sym
        w_sym = np.r_[w[:0:-1], w]  # length 2n-1
        # convolution via FFT
        H = np.fft.rfft(h, n=n_fft)
        W = np.fft.rfft(w_sym, n=n_fft)
        conv = np.fft.irfft(H * W, n=n_fft)
        T = conv[n-1:n-1+n]
        grad = 1.0 - T
        # step size
        t += 1
        alpha = step0 / np.sqrt(t)
        h_new = h - alpha * grad
        # clip
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # sum correction via fast redistribution
        diff = target - h_new.sum()
        # try up to 10 attempts (should be enough)
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 10:
            i = rng.integers(n)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            if add != 0:
                h_new[i] += add
                diff = target - h_new.sum()
            attempts += 1
        # if still diff, fallback to binary search
        if abs(diff) > 1e-12:
            h_new = project_box_sum(h_new, target)
        h = h_new
        # beta schedule
        if t % beta_step == 0:
            beta = min(beta * beta_factor, 1000.0)
        # periodic evaluation
        if t % check_interval == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                if best_c <= 0.3808:
                    # early exit?
                    # continue till time limit maybe
                    pass
    return best_h, best_c

def upsample(h, rng):
    n = h.size
    n2 = n * 2
    # define original x positions equally spaced over [0,2), exclude endpoint.
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    # clip and project to sum
    np.clip(h2, 0.0, 1.0, out=h2)
    # ensure sum = n2/2
    target = n2 / 2.0
    diff = target - h2.sum()
    # use fast redistribution similar to above
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10:
        i = rng.integers(n2)
        if diff > 0:
            add = min(diff, 1.0 - h2[i])
        else:
            add = max(diff, -h2[i])
        if add != 0:
            h2[i] += add
            diff = target - h2.sum()
        attempts += 1
    if abs(diff) > 1e-12:
        h2 = project_box_sum(h2, target)
    return h2
```

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo+hi)*0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)*0.5, 0.0, 1.0)
```

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities ------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0 ≤ x ≤ 1, sum x = target} via binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the max‑overlap objective (C5) before projection."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Simple hill‑climb using the worst‑gradient direction."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h, h = c, h_new.copy(), h_new
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# --------------------------------------------------------------
# Main optimisation routine (smooth‑min descent)

def smooth_min_descent(h: np.ndarray, target: float, time_budget: float,
                       rng: np.random.Generator,
                       step0: float = 0.5, beta0: float = 5.0,
                       beta_factor: float = 2.0, beta_step: int = 20000,
                       beta_max: float = 2000.0,
                       check_interval: int = 1000):
    """Projected gradient descent on a smooth approximation of min‑cross."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    t = 0
    beta = beta0
    n_fft = 1 << ((3 * n - 2).bit_length())  # power‑of‑two >= 3n‑2

    while True:
        if time.time() - start >= time_budget:
            break

        # autocorrelation (linear)
        cross = np.correlate(h, h, mode="full")[n - 1 :]

        # smooth‑min weights (numerically stable)
        cmin = cross.min()
        w = np.exp(-beta * (cross - cmin))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # symmetric weight vector for convolution
        w_sym = np.r_[w[:0:-1], w]               # length 2n‑1

        # convolution via FFT: T_i = Σ_k w_k (h_{i+k}+h_{i‑k})
        H = np.fft.rfft(h, n=n_fft)
        W = np.fft.rfft(w_sym, n=n_fft)
        conv = np.fft.irfft(H * W, n=n_fft)
        T = conv[n - 1 : n - 1 + n]               # central n entries

        grad = 1.0 - T

        t += 1
        alpha = step0 / np.sqrt(t)
        h_new = h - alpha * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)

        # fast sum‑correction (few random adjustments)
        diff = target - h_new.sum()
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 10:
            i = rng.integers(n)
            if diff > 0.0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            if add != 0.0:
                h_new[i] += add
                diff = target - h_new.sum()
            attempts += 1
        if abs(diff) > 1e-12:
            h_new = project_box_sum(h_new, target)

        h = h_new

        # beta schedule
        if t % beta_step == 0:
            beta = min(beta * beta_factor, beta_max)

        # periodic evaluation
        if t % check_interval == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
                # early exit if already good enough
                if best_c <= 0.3808:
                    # keep improving until time runs out, but do not break
                    pass

    return best_h, best_c

# Upsampling helper

def upsample(h: np.ndarray, rng: np.random.Generator):
    """Linear interpolation to double the resolution, preserving constraints."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    np.clip(h2, 0.0, 1.0, out=h2)
    target = n2 / 2.0
    diff = target - h2.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10:
        i = rng.integers(n2)
        if diff > 0.0:
            add = min(diff, 1.0 - h2[i])
        else:
            add = max(diff, -h2[i])
        if add != 0.0:
            h2[i] += add
            diff = target - h2.sum()
        attempts += 1
    if abs(diff) > 1e-12:
        h2 = project_box_sum(h2, target)
    return h2

    elapsed = time.time() - start_total
