    # third - eps * rev_old to conv_new[i : i+n]

    # fourth + eps * rev_old to conv_new[j : j+n]
    conv_new[j : j+n] += eps * rev_old
    # epsilon^2 terms
    d = j - i
    conv_new[n-1 + d] -= eps * eps
    conv_new[n-1 - d] -= eps * eps
    return h_new, conv_new

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum x = target} via binary search on lambda."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the box with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def compute_c5_from_conv(conv: np.ndarray, dx: float) -> float:
    """C5 from the self‑correlation array (only non‑negative shifts)."""
    # conv length = 2n-1, index n-1 corresponds to shift 0
    return 0.5 - conv[len(conv) // 2 :].min() * dx


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of O_k for the shift k that minimizes the overlap."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n, shift 0..n-1
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def subgradient_descent(
    h0: np.ndarray, target: float, time_budget: float, rng: np.random.Generator
):
    """Short projected sub‑gradient descent (used for quick polishing)."""
    start = time.time()
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h, len(h))
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h, len(h))
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c


def compute_c5(h: np.ndarray, n_points: int) -> float:
    """Discrete C5 for a step function on [0,2] (fallback)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n_points)


# ----------------------------------------------------------------------
# Incremental transfer + correlation update (used in annealing)
# ----------------------------------------------------------------------
def transfer_and_update(h: np.ndarray, conv: np.ndarray, i: int, j: int, eps: float):
    """
    Return new h and new conv after moving 'eps' from position i to j.
    Both h and conv are NOT modified in‑place.
    """
    old_h = h  # alias for clarity
    h_new = h.copy()
    conv_new = conv.copy()
    n = h.size

    # modify h
    h_new[i] -= eps
    h_new[j] += eps

    # slices for the linear terms
    # -eps * old_h[i:] contributes to conv[n-1 : n-1 + (n-i)]
    conv_new[n - 1 : n - 1 + (n - i)] -= eps * old_h[i:]
    # +eps * old_h[j:] contributes to conv[n-1 : n-1 + (n-j)]
    conv_new[n - 1 : n - 1 + (n - j)] += eps * old_h[j:]

    # reversed vector for the second term
    rev_old = old_h[::-1]
    conv_new[i : i + n] -= eps * rev_old
    conv_new[j : j + n] += eps * rev_old

    # quadratic ε² term
    d = j - i
    conv_new[n - 1 + d] -= eps * eps
    conv_new[n - 1 - d] -= eps * eps

    return h_new, conv_new

# Main routine
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # --------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible(512, rng)

    n = h0.size
    dx = 2.0 / n
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_box_sum(h0, n / 2.0)
    conv0 = np.correlate(h0, h0, mode="full")
    best_h, best_conv = h0.copy(), conv0.copy()
    best_c = compute_c5_from_conv(best_conv, dx)

    # --------------------------------------------------------------
    # Phase 0 – brief sub‑gradient polishing (≈5% of time)
    elapsed = time.time() - start
    remain = total_budget - elapsed
    if remain > 5.0:
        sub_time = min(0.05 * total_budget, remain * 0.10)
        h_sg, c_sg = subgradient_descent(
            best_h, n / 2.0, sub_time, rng
        )
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg
            best_conv = np.correlate(best_h, best_h, mode="full")
            dx = 2.0 / n

    # --------------------------------------------------------------
    # Phase 1 – Simulated annealing on the current grid
    elapsed = time.time() - start
    remain = total_budget - elapsed
    if remain > 5.0:
        anneal_start = time.time()
        anneal_end = start + total_budget
        anneal_total = anneal_end - anneal_start

        # temperature and step‑size schedules
        T0 = max(0.001, best_c * 0.02)
        while time.time() < anneal_end:
            # progress fraction
            prog = (time.time() - anneal_start) / anneal_total
            T = max(1e-9, T0 * (1.0 - prog))
            scale = 0.5 * (1.0 - prog) + 0.01

            # pick source i (with some mass) and sink j (with room)
            i = rng.integers(0, n)
            if best_h[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if best_h[j] >= 1.0 or i == j:
                continue

            eps_max = min(best_h[i], 1.0 - best_h[j])
            if eps_max <= 1e-12:
                continue
            eps = eps_max * rng.random() * scale

            # propose move
            h_new, conv_new = transfer_and_update(best_h, best_conv, i, j, eps)
            c_new = compute_c5_from_conv(conv_new, dx)

            delta = c_new - best_c
            if delta < 0.0 or rng.random() < np.exp(-delta / T):
                # accept
                best_h, best_conv, best_c = h_new, conv_new, c_new
                if best_c < best_c:  # dummy to keep syntax tidy
                    pass

    # --------------------------------------------------------------
    # Phase 2 – optional up‑sampling and final short anneal
    elapsed = time.time() - start
    remain = total_budget - elapsed
    if remain > 8.0 and n * 2 <= 1024:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = project_box_sum(h2, n2 / 2.0)
        conv2 = np.correlate(h2, h2, mode="full")
        c2 = compute_c5_from_conv(conv2, 2.0 / n2)
        if c2 < best_c:
            best_h, best_conv, best_c, n, dx = h2, conv2, c2, n2, 2.0 / n2

        # short anneal on the finer grid
        anneal_start = time.time()
        anneal_end = start + total_budget
        anneal_total = anneal_end - anneal_start
        T0 = max(0.001, best_c * 0.015)
        while time.time() < anneal_end:
            prog = (time.time() - anneal_start) / anneal_total
            T = max(1e-9, T0 * (1.0 - prog))
            scale = 0.4 * (1.0 - prog) + 0.005

            i = rng.integers(0, n)
            if best_h[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if best_h[j] >= 1.0 or i == j:
                continue
            eps_max = min(best_h[i], 1.0 - best_h[j])
            if eps_max <= 1e-12:
                continue
            eps = eps_max * rng.random() * scale

            h_new, conv_new = transfer_and_update(best_h, best_conv, i, j, eps)
            c_new = compute_c5_from_conv(conv_new, dx)
            delta = c_new - best_c
            if delta < 0.0 or rng.random() < np.exp(-delta / T):
                best_h, best_conv, best_c = h_new, conv_new, c_new

    # --------------------------------------------------------------
    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h, best_h.size)

    return best_h.tolist(), float(final_c), int(best_h.size)
