def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # Use while loop but deterministic, each iteration distributes diff uniformly among eligible indices.
    while abs(diff) > 1e-12:
        if diff > 0:
            eligible = np.where(h < 1.0)[0]
            if eligible.size == 0:
                break
            inc = diff / eligible.size
            # clip inc to not exceed 1
            inc = np.minimum(inc, 1.0 - h[eligible])
            # May have varying inc per eligible; but we can distribute equally then clip.
            # Simpler: loop through eligible:
            for i in eligible:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) <= 1e-12:
                    break
        else:
            # diff < 0
            eligible = np.where(h > 0.0)[0]
            if eligible.size == 0:
                break
            for i in eligible:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff = target - h.sum()
                if abs(diff) <= 1e-12:
                    break
    return h

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.005, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # choose two distinct indices i,j
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        # decide direction random
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def project_sum(h: np.ndarray, target: float, max_iter: int = 10000) -> np.ndarray:
    """Deterministic projection onto {0≤h≤1, sum h = target}."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            for i in range(n):
                if diff <= 0:
                    break
                if h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff = target - h.sum()
        else:
            for i in range(n):
                if diff >= 0:
                    break
                if h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff = target - h.sum()
        it += 1
    return h


def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated annealing on feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.008, 1e-8
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = T / T0
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = 0

    # ----- initial solution if provided -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = project_sum(h0, h0.size / 2)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ----- Phase 1: Differential Evolution on 256 points -----
    remaining = budget_s - (time.time() - start)
    if remaining > 10:
        N = 256
        target = N / 2.0
        maxiter = 400  # 400 generations
        popsize = 12
        t_de = min(remaining * 0.2, 200.0)

        def de_obj(x):
            h = project_sum(x, target)
            return compute_c5(h)

        def de_cb(xk, convergence):
            if time.time() - start > budget_s * 0.94:
                return True
            return False

        result = differential_evolution(
            de_obj,
            [(0.0, 1.0)] * N,
            strategy="best1bin",
            maxiter=maxiter,
            popsize=popsize,
            polish=False,
            init="latinhypercube",
            updating="deferred",
            workers=1,
            callback=de_cb,
            seed=seed,
        )
        h_de = project_sum(result.x, target)
        c_de = compute_c5(h_de)
        if c_de < best_c:
            best_h, best_c, best_n = h_de, c_de, N

    # ----- Phase 2: Upsample to 512 if needed -----
    remaining = budget_s - (time.time() - start)
    if remaining > 5 and best_h is not None and best_n < 512:
        N2 = 512
        h_mid = upsample_h(best_h, N2)
        h_mid = project_sum(h_mid, N2 / 2.0)
        c_mid = compute_c5(h_mid)
        if c_mid < best_c:
            best_h, best_c, best_n = h_mid, c_mid, N2

    # ----- Phase 3: Differential Evolution on 512 points -----
    remaining = budget_s - (time.time() - start)
    if remaining > 20 and best_h is not None:
        N = 512
        target = N / 2.0
        maxiter = 800
        popsize = 12
        t_de2 = min(remaining * 0.35, 300.0)

        def de_obj2(x):
            h = project_sum(x, target)
            return compute_c5(h)

        def de_cb2(xk, convergence):
            if time.time() - start > budget_s * 0.96:
                return True
            return False

        result2 = differential_evolution(
            de_obj2,
            [(0.0, 1.0)] * N,
            strategy="best1bin",
            maxiter=maxiter,
            popsize=popsize,
            polish=False,
            init="latinhypercube",
            updating="deferred",
            workers=1,
            callback=de_cb2,
            seed=seed + 1,
        )
        h_de2 = project_sum(result2.x, target)
        c_de2 = compute_c5(h_de2)
        if c_de2 < best_c:
            best_h, best_c, best_n = h_de2, c_de2, N

    # ----- Phase 4: Gradient refinement on current best -----
    remaining = budget_s - (time.time() - start)
    if remaining > 5 and best_h is not None:
        h = best_h.copy()
        n = h.size
        target = n / 2.0
        total_iters = 100
        for it in range(total_iters):
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(corr) - (n - 1)
            k = abs(idx)
            if k == 0:
                grad = 1.0 - 2.0 * h
            else:
                grad = np.zeros_like(h)
                grad[: n - k] = 1.0 - h[k:]
                grad[k:] -= h[: n - k]
            step = 0.02 * (0.9 ** (it // 10))
            h = h - step * grad
            h = project_sum(np.clip(h, 0.0, 1.0), target)
            if it % 5 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c = h.copy(), c
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
        best_n = best_h.size

    # ----- Phase 5: Upsample to 1024 and Continuous SA -----
    remaining = budget_s - (time.time() - start)
    if remaining > 5 and best_h is not None:
        if best_n < 1024:
            h_fine = upsample_h(best_h, 1024)
            h_fine = project_sum(h_fine, 1024 / 2.0)
            best_h, best_n = h_fine, 1024
        # allocate remaining time to SA
        h_sa, c_sa = continuous_sa(best_h, rng, remaining)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa
            best_n = best_h.size

    # ----- Final sanitisation -----
    best_h = project_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)

    # fallback in the improbable case best_h is None
    if best_h is None:
        N = 256
        best_h = np.zeros(N, dtype=float)
        best_h[: N // 2] = 1.0
        final_c = compute_c5(best_h)
        best_n = N

    return best_h.tolist(), float(final_c), int(best_n)
