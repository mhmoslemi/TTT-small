def project(v, target):

    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + hi, 0.0, 1.0)
    # due to rounding errors may not exactly sum target, adjust small diff
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust by adding diff to entries not at bounds
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt > 0:
            # distribute equally
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)  # ensure within bounds
    return x

def annealing_phase(h, target, max_time, start_time, rng):
    best_h = h.copy()
    best_c = compute_c5(h)
    current_h = h.copy()
    current_c = best_c
    T0 = 0.01  # initial temperature
    T_end = 0.000001
    total_iters = 0
    while time.time() - start_time < max_time:
        # compute temperature decaying linearly with time
        elapsed = time.time() - start_time
        fraction = elapsed / max_time
        T = T0 * (1 - fraction) + T_end * fraction
        # propose move
        i, j = rng.integers(0, h.size, size=2)
        if i == j: continue
        # transfer epsilon from i to j
        max_eps = min(current_h[i], 1.0 - current_h[j])
        if max_eps <= 0: continue
        eps = rng.random() * max_eps * 0.1  # up to 10% of max
        h_new = current_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # projection not needed as sum preserved and bounds kept
        c_new = compute_c5(h_new)
        # acceptance
        if c_new < current_c or rng.random() < np.exp(-(c_new - current_c) / T):
            current_h = h_new
            current_c = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
        total_iters += 1
        if best_c <= 0.3808:
            break
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=1500):
    n = h0.size
    target = n / 2.0
    # Use global variables to be accessible in constraint and objective.
    global _SL_SQP_TARGET, _SL_SQP_BETA
    _SL_SQP_TARGET = target
    _SL_SQP_BETA = beta
    def sum_eq(x):
        return np.sum(x) - _SL_SQP_TARGET
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = _SL_SQP_BETA * corr
        m = np.max(bc)
        # use log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / _SL_SQP_BETA
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum if small violation
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust via lambda approach
        x = project_onto_sum_and_bounds(x, target)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities -------------------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst overlap O_k w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift of worst O_k
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]               # -h[i+k]
        grad[k:]    -= h[: n - k]            # -h[i-k]
    return grad


def project_onto_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x∈[0,1]^n : sum x = target}.
    Finds λ such that sum clip(v+λ,0,1)=target via binary search.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + hi, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

# ------------------- SLSQP refinement -------------------
_SL_SQP_TARGET = None
_SL_SQP_BETA = None


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global _SL_SQP_TARGET, _SL_SQP_BETA
    n = h0.size
    _SL_SQP_TARGET = n / 2.0
    _SL_SQP_BETA = beta

    def sum_eq(x):
        return np.sum(x) - _SL_SQP_TARGET

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = _SL_SQP_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / _SL_SQP_BETA

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(_SL_SQP_TARGET - x.sum()) > 1e-12:
        x = project_onto_sum_and_bounds(x, _SL_SQP_TARGET)
    return x

# ------------------- gradient descent -------------------
def gradient_descent_phase(h0: np.ndarray, target: float,
                          max_time: float, start: float,
                          rng: np.random.Generator,
                          bound: float = 0.3808) -> tuple:
    """Projected gradient descent with momentum on the worst shift."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    v = np.zeros_like(h)               # momentum term
    step = 0.5
    mu = 0.9
    it = 0
    while time.time() - start < max_time:
        grad = worst_gradient(h)
        v = mu * v - step * grad
        h_new = project_onto_sum_and_bounds(h + v, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= bound:
                break
        h = h_new
        step *= 0.9995
        it += 1
        # occasional small Gaussian jitter
        if it % 500 == 0:
            noise = rng.normal(scale=0.01, size=h.shape)
            h_noise = project_onto_sum_and_bounds(h + noise, target)
            c_noise = compute_c5(h_noise)
            if c_noise < best_c:
                best_c, best_h = c_noise, h_noise.copy()
            if c_noise < c or rng.random() < 0.1:
                h = h_noise
    return best_h, best_c


# ------------------- simulated annealing -------------------
def annealing_phase(h0: np.ndarray, target: float,
                    max_time: float, start: float,
                    rng: np.random.Generator,
                    bound: float = 0.3808) -> tuple:
    """Simple annealing that swaps small amounts of mass."""
    current_h = h0.copy()
    current_c = compute_c5(current_h)
    best_h, best_c = current_h.copy(), current_c
    T0, T1 = 0.01, 1e-6
    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = T0 * (1 - frac) + T1 * frac
        i, j = rng.integers(0, h0.size, size=2)
        if i == j:
            continue
        max_eps = min(current_h[i], 1.0 - current_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.1
        h_new = current_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        if c_new < current_c or rng.random() < np.exp(-(c_new - current_c) / T):
            current_h, current_c = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                if best_c <= bound:
                    break
    return best_h, best_c
