def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution, usually converges in few steps
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

    return h

    """Transfer mass between two positions while respecting bounds."""

    h = enforce_sum(h, rng)
    c = compute_c5(h)
    return h, c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    best_c = float("inf")
    best_h = None
    best_n = None
    best_small = []  # list of (c, h) for N=256, length ≤5

    # ---------- Phase 1 : cheap 256‑point runs ----------
    N_small = 256
    phase1_deadline = start + budget_s * 0.30
    while time.time() < phase1_deadline:
        h, c = optimize_at_N(N_small, rng)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), N_small
        # maintain top‑5 small solutions
        if len(best_small) < 5 or c < best_small[-1][0]:
            inserted = False
            for idx, (c0, _) in enumerate(best_small):
                if c < c0:
                    best_small.insert(idx, (c, h.copy()))
                    inserted = True
                    break
            if not inserted:
                best_small.append((c, h.copy()))
            if len(best_small) > 5:
                best_small.pop()

    # ---------- Phase 2 : upscale to 512 / 1024 ----------
    large_Ns = [512, 1024]
    while time.time() < start + budget_s * 0.95:
        N = rng.choice(large_Ns)
        use_upscaled = best_small and rng.random() < 0.7
        if use_upscaled:
            _, h_small = rng.choice(best_small)
            h, c = optimize_at_N(N, rng, start_h=h_small,
                                 max_iter_slsqp=100, beta=3000)
        else:
            h, c = optimize_at_N(N, rng,
                                 max_iter_slsqp=80, beta=2000)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), N

    # ---------- Phase 3 : final high‑beta polishing ----------
    if best_h is not None:
        for beta in (5000, 20000, 50000):
            if time.time() > deadline - 2.0:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
            best_h = enforce_sum(best_h, rng)
            best_c = compute_c5(best_h)

    # final safety repair
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
