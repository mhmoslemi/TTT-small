def adam_optimize(h0, steps, base_lr, target):
    n = h0.size
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, steps+1):
        # compute worst gradient
        grad = worst_gradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = base_lr * (1.0 / np.sqrt(t))  # decaying
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, target)
    return h

def worst_gradient(h):
    n = h.size
    # compute corr and idx
    a = np.concatenate([h, np.zeros(n)])
    b = np.concatenate([1 - h, np.zeros(n)])
    A = np.fft.rfft(a)
    B = np.fft.rfft(b)
    corr = np.fft.irfft(A * np.conj(B), n=2*n)
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1 - 2 * h
    else:
        # grad[i] = 1 - h[i+k] - h[i-k] inside bounds
        # start with ones and subtract contributions
        # For i from 0 to n-k-1, subtract h[i+k]
        grad[: n - k] -= h[k:]
        # For i from k to n-1, subtract h[i - k]
        grad[k:] -= h[: n - k]
    return grad

def simulated_annealing(h, rng, time_limit, start_temp, end_temp):
    start_time = time.time()
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = h.size
    # precompute list of indices with non-zero and not full to speed? We'll just pick random i,j where move possible each iteration.
    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        T = start_temp * ((end_temp/start_temp) ** (elapsed / time_limit))
        # Pick random i,j distinct
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # Determine max epsilon can move from i to j without violating box.
        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        if eps_max <= 0: continue
        # Propose epsilon scaled by T? Not necessarily; we sample eps uniformly up to eps_max.
        eps = rng.random() * eps_max
        # Apply move
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

    i = int(np.argmax(grad))

    if i == j:

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = step_scale

    n = h0.size
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)

import time
import numpy as np

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.size
    a = np.concatenate([h, np.zeros(n)])
    b = np.concatenate([1.0 - h, np.zeros(n)])
    A = np.fft.rfft(a)
    B = np.fft.rfft(b)
    corr = np.fft.irfft(A * np.conj(B), n=2 * n)
    return corr.max() * (2.0 / n)

def compute_c5_and_idx(h):
    n = h.size
    a = np.concatenate([h, np.zeros(n)])
    b = np.concatenate([1.0 - h, np.zeros(n)])
    A = np.fft.rfft(a)
    B = np.fft.rfft(b)
    corr = np.fft.irfft(A * np.conj(B), n=2 * n)
    idx = np.argmax(corr) - (n - 1)
    return corr.max() * (2.0 / n), idx

def worst_gradient(h):
    n = h.size
    # obtain worst shift index
    _, idx = compute_c5_and_idx(h)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def adam_optimize(h0, steps, lr, target):
    n = h0.size
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        grad = worst_gradient(h)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / np.sqrt(t)
        h = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, target)
    return h

def simulated_annealing(h, rng, time_limit, start_temp=0.1, end_temp=1e-4):
    start = time.time()
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = h.size
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = start_temp * ((end_temp / start_temp) ** (elapsed / time_limit))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        if eps_max <= 0.0:
            continue
        eps = rng.random() * eps_max
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    end_time = start + budget_s

    # initialise h
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h0 = rng.random(n)
    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Adam from current best
    if time.time() < end_time - 0.1 * budget_s:
        steps = int(3000 * 256 / n)
        h_adam = adam_optimize(best_h, steps, lr=0.6, target=target)
        c_adam = compute_c5(h_adam)
        if c_adam < best_c:
            best_h, best_c = h_adam, c_adam

    # Phase 2: many random restarts with Adam
    while time.time() < end_time - 0.4 * budget_s:
        h_rand = rng.random(n)
        h_rand = project_onto_box_sum(h_rand, target)
        steps = int(1500 * 256 / n)
        h_opt = adam_optimize(h_rand, steps, lr=0.5, target=target)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 3: Guided hill climbing
    if time.time() < end_time - 0.2 * budget_s:
        hc, cc = guided_hill_climb(best_h, rng,
                                   max_iters=int(2500 * 256 / n),
                                   step_scale=0.5)
        if cc < best_c:
            best_h, best_c = hc, cc

    # Phase 4: Simulated annealing on remaining time
    remaining = end_time - time.time()
    if remaining > 0.05 * budget_s:
        hc, cc = simulated_annealing(best_h, rng,
                                     time_limit=remaining,
                                     start_temp=0.1,
                                     end_temp=1e-4)
        if cc < best_c:
            best_h, best_c = hc, cc

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
