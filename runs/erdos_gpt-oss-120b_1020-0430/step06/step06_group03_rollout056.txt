import time, math, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        N = h.size
        h = np.clip(h, 0.0, 1.0)
        target = N / 2.0
        if abs(h.sum() - target) > 1e-12:
            h = project_onto_box_sum(h, target)
    else:
        N = 256
        target = N / 2.0
        h = rng.random(N)
        h = project_onto_box_sum(h, target)

    # product of h with itself for all shifts
    prod = np.correlate(h, h, mode='full')
    cur_c5 = 1.0 - (2.0 / N) * prod.max()
    best_h = h.copy()
    best_c5 = cur_c5

    # ----- parameters -------------------------------------------------
    recompute_interval = N * 4
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.99:
            break
        frac = elapsed / budget_s

        # temperature for simulated annealing
        T = cur_c5 * 0.10 * (1.0 - frac) + 1e-12
        # step‑size scale (larger early, tiny later)
        step_scale = max(0.02, 0.5 * (1.0 - frac))

        # ----- choose move --------------------------------------------
        if rng.random() < 0.05:
            i_from, i_to = rng.choice(N, size=2, replace=False)
        else:
            # worst shift (min product)
            k_idx = prod.argmin()
            k = abs(k_idx - (N - 1))
            if k == 0:
                grad = 1.0 - 2.0 * h
            else:
                grad = np.ones(N)
                grad[:N - k] -= h[k:]
                grad[k:] -= h[:N - k]
            i_from = int(np.argmax(grad))
            i_to = int(np.argmin(grad))
            if i_from == i_to:
                i_from, i_to = rng.choice(N, size=2, replace=False)

        max_delta = min(h[i_from], 1.0 - h[i_to])
        if max_delta <= 1e-12:
            it += 1
            continue
        d = max_delta * rng.random() * step_scale

        # ----- update product in place --------------------------------
        rev = h[::-1]  # view, no copy
        # subtract contribution of i_from
        prod[N - 1 - i_from : 2 * N - 1 - i_from] -= d * h
        prod[i_from : N + i_from] -= d * rev
        # add contribution of i_to
        prod[N - 1 - i_to : 2 * N - 1 - i_to] += d * h
        prod[i_to : N + i_to] += d * rev

        new_max_prod = prod.max()
        new_c5 = 1.0 - (2.0 / N) * new_max_prod
        delta_c5 = new_c5 - cur_c5

        # ----- acceptance --------------------------------------------
        accept = False
        if delta_c5 <= 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta_c5 / T):
                accept = True

        if accept:
            h[i_from] -= d
            h[i_to] += d
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert product update
            prod[N - 1 - i_from : 2 * N - 1 - i_from] += d * h
            prod[i_from : N + i_from] += d * rev
            prod[N - 1 - i_to : 2 * N - 1 - i_to] -= d * h
            prod[i_to : N + i_to] -= d * rev

        it += 1
        if it % recompute_interval == 0:
            prod = np.correlate(h, h, mode='full')
            cur_c5 = 1.0 - (2.0 / N) * prod.max()
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # ----- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, N / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(N)
