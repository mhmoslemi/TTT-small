def binary_sa(N, rng, time_budget, start_h=None):
    start_time = time.time()
    if start_h is not None:
        h = start_h.copy()
        # ensure binary
        h = np.where(h >= 0.5, 1.0, 0.0)
        # adjust sum to N//2:
        current_ones = int(h.sum())
        target_ones = N // 2
        if current_ones > target_ones:
            # turn some ones to zero
            ones_idx = np.where(h == 1)[0]
            to_zero = rng.choice(ones_idx, current_ones - target_ones, replace=False)
            h[to_zero] = 0.0
        elif current_ones < target_ones:
            zeros_idx = np.where(h == 0)[0]
            to_one = rng.choice(zeros_idx, target_ones - current_ones, replace=False)
            h[to_one] = 1.0
    else:
        h = np.zeros(N, dtype=float)
        ones = rng.choice(N, N//2, replace=False)
        h[ones] = 1.0

    # initialize arrays of indices:
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]

    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    last_improve = start_time

    T0 = 0.01
    T_end = 1e-6

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= time_budget:
            break
        # optional restart if stagnant:
        if now - last_improve > 30.0:
            # restart fresh
            h = np.zeros(N, dtype=float)
            ones = rng.choice(N, N//2, replace=False)
            h[ones] = 1.0
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            c_curr = compute_c5(h)
            # don't update best
            last_improve = now  # reset stagnation timer
            continue

        # pick random 1 and 0
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]

        # swap
        h[i] = 0.0
        h[j] = 1.0

        c_new = compute_c5(h)
        delta = c_new - c_curr
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # acceptance
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            c_curr = c_new
            # update index arrays
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                last_improve = now
                # early exit if below target
                if best_c <= TARGET_C5:
                    return best_h, best_c
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad

def gradient_refine(h, rng, step=0.015, max_iter=15):
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff randomly
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = target - h.sum()
    return h

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        elapsed = now - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # decide direction random
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / N)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, max_iter: int = 10000) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        it += 1
    return h

# Gradient based tidy‑up
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 20) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = target - h.sum()
    return h

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float) -> tuple[np.ndarray, float]:
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        elapsed = now - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c
