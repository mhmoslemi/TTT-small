def project_to_feasible(v, target, rng):
    # Clip to [0,1]
    u = np.clip(v, 0.0, 1.0)
    # adjust sum to target
    diff = target - u.sum()
    # water-filling iterative
    # We'll adjust distribution over free indices (not at bounds)
    # Use while loop with tolerance.
    # At each iteration:
    #   free_mask = (u > 0) & (u < 1)
    #   free_count = free_mask.sum()
    #   if free_count ==0: break (can't adjust, just break)
    #   delta = diff / free_count
    #   u[free_mask] += delta
    #   # clip again for any crossing bounds
    #   u = np.clip(u, 0, 1)
    #   diff = target - u.sum()
    #   loop.
    # For speed we can limit to few loops.

    diff = target - u.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free_mask = (u > 0) & (u < 1)
        free_count = free_mask.sum()
        if free_count == 0:
            # we have to adjust by perturbation to free bounds
            # add small epsilon to random free index
            idx = rng.integers(0, len(u))
            # try to adjust
            add = min(diff, 1.0 - u[idx]) if diff > 0 else max(diff, -u[idx])
            u[idx] += add
            diff = target - u.sum()
            continue
        delta = diff / free_count
        # update
        u[free_mask] += delta
        # clip any that may exceed bounds now
        u = np.clip(u, 0.0, 1.0)
        diff = target - u.sum()
    # final clip
    u = np.clip(u, 0.0, 1.0)
    return u

def worst_shift_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')

    grad[: n - k] -= h[k:]
    # subtract h[i-k] for i >= k
    grad[k:] -= h[: n - k]
    return grad

def adam_optimize(h0, time_limit, rng):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    # Hyperparameters
    lr = 0.1  # initial learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h_local = h.copy()
    best_c_local = compute_c5(h)
    while time.time() < time_limit:
        t += 1
        grad = worst_shift_gradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # update
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # project
        h = project_to_feasible(h, target, rng)
        # evaluate
        c_val = compute_c5(h)
        if c_val < best_c_local:
            best_c_local = c_val
            best_h_local = h.copy()
    return best_h_local, best_c_local

def multi_gradient(h, M=3):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # get top M indices (including potentially negative shift)
    if M >= len(corr):
        idxs = np.arange(len(corr))
    else:
        # argpartition
        idxs = np.argpartition(corr, -M)[-M:]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = abs(idx - (n - 1))
        if k == 0:
            grad += (1.0 - 2.0 * h)
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
            grad += g
    grad /= len(idxs)
    return grad

def gradient_refine(h, rng, step=0.02, max_iter=10):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        for _ in range(500):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

import time
import numpy as np

# ----- basic utilities -------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust sum to `target` preserving bounds (water‑filling)."""
    u = np.clip(v, 0.0, 1.0)
    diff = target - u.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (u > 0.0) & (u < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            # pick a random index that can still move
            i = rng.integers(0, u.size)
            add = min(diff, 1.0 - u[i]) if diff > 0 else max(diff, -u[i])
            u[i] += add
        else:
            delta = diff / nfree
            u[free] += delta
            u = np.clip(u, 0.0, 1.0)
        diff = target - u.sum()
    return np.clip(u, 0.0, 1.0)


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.ones_like(h)
    g[: n - k] -= h[k:]
    g[k:] -= h[: n - k]
    return g


def multi_gradient(h: np.ndarray, M: int = 5) -> np.ndarray:
    """Average gradient of the M worst shifts (softened worst‑case)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    if M >= corr.size:
        idxs = np.arange(corr.size)
    else:
        idxs = np.argpartition(corr, -M)[-M:]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = abs(idx - (n - 1))
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
            grad += g
    return grad / len(idxs)


# ----- Adam optimizer on the worst‑case C5 ---------------------------

def adam_optimize(h0: np.ndarray, deadline: float, rng: np.random.Generator, M: int = 5):
    """Adam on the (averaged) worst–shift gradient, with projection."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)

    while time.time() < deadline:
        t += 1
        grad = multi_gradient(h, M)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_to_feasible(h, target, rng)

        c_val = compute_c5(h)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c


# ----- simple gradient polishing -------------------------------------

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Diminishing‑step gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        for _ in range(300):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


# ----- very cheap local swap search ----------------------------------

def local_swap_search(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """Random pairwise swaps (sum preserved) accepting improvements."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        nh = best_h.copy()
        nh[i], nh[j] = nh[j], nh[i]      # sum unchanged, bounds stay valid
        nc = compute_c5(nh)
        if nc < best_c:
            best_c = nc
            best_h = nh
    return best_h, best_c


# ----- core optimizer for a given resolution -------------------------

def optimize_n(n: int, time_budget: float, rng: np.random.Generator,
               h_init: np.ndarray | None = None):
    """Return a good h for `n` points within `time_budget` seconds."""
    target = n / 2.0
    if h_init is not None:
        h0 = project_to_feasible(np.clip(h_init, 0.0, 1.0), target, rng)
    else:
        h0 = rng.random(n)
        h0 = h0 / h0.sum() * target
        h0 = project_to_feasible(h0, target, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)
    start = time.time()
    # main loop: many short Adam runs with perturbed starts
    while time.time() - start < time_budget * 0.85:
        remain = start + time_budget * 0.85 - time.time()
        if remain < 6.0:
            break
        # perturb current best a bit
        noise = 0.18
        h_start = best_h + rng.normal(scale=noise, size=n)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = project_to_feasible(h_start, target, rng)

        deadline = time.time() + min(22.0, remain - 5.0)
        h_opt, c_opt = adam_optimize(h_start, deadline, rng, M=5)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # final polishing
    if time.time() - start < time_budget * 0.95:
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=14)
        best_c = compute_c5(best_h)

    # very cheap swap search with remaining slice of time
    if time.time() - start < time_budget:
        best_h, best_c = local_swap_search(best_h, rng, time.time() + (time_budget - (time.time() - start)))

    best_h = project_to_feasible(best_h, target, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c


# ----- helper to downsample / upsample --------------------------------

def downsample_h(h_big: np.ndarray, n_small: int) -> np.ndarray:
    factor = h_big.size // n_small
    return h_big.reshape(n_small, factor).mean(axis=1)


def upsample_h(h_small: np.ndarray, n_big: int) -> np.ndarray:
    x_big = np.linspace(0.0, 2.0, n_big, endpoint=False)
    x_small = np.linspace(0.0, 2.0, h_small.size, endpoint=False)
    return np.interp(x_big, x_small, h_small)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    # initial data (if supplied by the harness)

    else:

    # upsample to full size and give it a quick polish
