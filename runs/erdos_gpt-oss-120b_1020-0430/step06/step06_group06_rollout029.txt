def compute_c5_fft(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)
    # linear convolution length = 2n-1
    conv = conv[:2*n-1]
    max_val = conv.max()
    return max_val * (2.0/n) # since dx = 2/n
```

def hill_climb(h, rng, max_iter, temp_schedule):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    cur_h = h.copy()
    cur_c = best_c
    T0 = 0.1
    for it in range(max_iter):
        # choose random swap: pick pos_i where cur_h[i]==1, pos_j where cur_h[j]==0
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cur_h[i], cur_h[j] = 0, 1
        new_c = compute_c5_fft(cur_h)
        delta = new_c - cur_c
        T = T0 * (0.999**it)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1, 0
    return best_h, best_c
```

def correlation_raw_fft(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    # compute length m as next power of two >= 2*n-1
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)
    return conv[:2*n-1]
```

def slsqp_optimize(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if abs(add) < 1e-12:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- FFT correlation ----------
def _fft_raw_float(h: np.ndarray) -> np.ndarray:
    """Convolution h * (1‑h) (full) as float array."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)
    return conv[: 2 * n - 1]


def _fft_raw_int(h: np.ndarray) -> np.ndarray:
    """Convolution h * (1‑h) (full) as integer array (h must be 0/1)."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)
    return np.rint(conv[: 2 * n - 1]).astype(np.int64)


def compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n points."""
    n = h.shape[0]
    raw = _fft_raw_float(h)
    return float(raw.max() * (2.0 / n))


# ---------- Constraint handling ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-12:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


# ---------- Local optimisation (swap‑gradient) ----------
def local_optimize_binary(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Iteratively swap a 1 from a negative‑gradient index with a 0 from a
    positive‑gradient index until no such pair exists."""
    n = h.shape[0]
    while True:
        raw = _fft_raw_int(h)               # integer convolution
        idx = int(np.argmax(raw))
        k = idx - (n - 1)
        grad = np.zeros(n, dtype=np.int8)
        if k == 0:
            grad = 1 - 2 * h
        else:
            grad[: n - k] = 1 - h[k:]
            grad[k:] = -h[: n - k]
        pos_one = np.where((grad == 1) & (h == 1))[0]
        neg_zero = np.where((grad == -1) & (h == 0))[0]
        if pos_one.size == 0 or neg_zero.size == 0:
            break
        i = rng.choice(pos_one)
        j = rng.choice(neg_zero)
        h[i] = 0
        h[j] = 1
    return h


def perturb_binary(h: np.ndarray, rng: np.random.Generator, num_swaps: int) -> np.ndarray:
    """Perform a few random 1↔0 swaps."""
    n = h.shape[0]
    h2 = h.copy()
    for _ in range(num_swaps):
        ones = np.where(h2 == 1)[0]
        zeros = np.where(h2 == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h2[i] = 0
        h2[j] = 1
    return h2


# ---------- Soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = _fft_raw_float(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    """Smooth max‑overlap refinement via SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        if abs(add) > 1e-12:
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
        it += 1
    return x

        cand = local_optimize_binary(cand, rng)

        if c < best_c:

        # occasional fresh random start

    # ----- final smooth refinement with SLSQP (if time permits) -----
    if time.time() - start < budget_s - 20.0:
