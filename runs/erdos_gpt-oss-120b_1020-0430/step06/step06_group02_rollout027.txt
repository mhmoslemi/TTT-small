import time
import math
import numpy as np
from scipy.optimize import minimize

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- initial data ----------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    n = h0.size
    target_sum = n / 2.0

    # enforce exact sum for continuous vector
    def enforce_sum_continuous(h):
        diff = target_sum - h.sum()
        it = 0
        while abs(diff) > 1e-12 and it < 2000:
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if add != 0.0:
                h[i] += add
                diff = target_sum - h.sum()
            it += 1
        return h

    h0 = enforce_sum_continuous(h0)

    # ---------- helpers ----------
    def compute_c5(h):
        """C5 using FFT convolution."""
        inv = 1.0 - h
        L = 1 << ((2 * n - 1).bit_length())
        H = np.fft.rfft(h, n=L)
        I = np.fft.rfft(inv, n=L)
        conv = np.fft.irfft(H * I, n=L)[: 2 * n - 1]
        return conv.max() * (2.0 / n)

    def compute_corr(h):
        """Full convolution (FFT) – used in iterative loops."""
        inv = 1.0 - h
        L = 1 << ((2 * n - 1).bit_length())
        H = np.fft.rfft(h, n=L)
        I = np.fft.rfft(inv, n=L)
        return np.fft.irfft(H * I, n=L)[: 2 * n - 1]

    # ---------- initial best ----------
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_corr = compute_corr(best_h)

    # binary version of the start (balanced)
    def enforce_sum_binary(h):
        diff = int(target_sum - h.sum())
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if zeros.size >= diff:
                choose = rng.choice(zeros, diff, replace=False)
                h[choose] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            diff = -diff
            if ones.size >= diff:
                choose = rng.choice(ones, diff, replace=False)
                h[choose] = 0.0
        return h

    h_bin = (h0 >= 0.5).astype(np.float64)
    h_bin = enforce_sum_binary(h_bin)

    # consider binary start as a candidate
    c_bin = compute_c5(h_bin)
    if c_bin < best_c:
        best_h = h_bin.copy()
        best_c = c_bin
        best_corr = compute_corr(best_h)

    # ---------- hill climbing on binary ----------
    hill_time = budget_s * 0.7
    while time.time() - start < hill_time:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        corr_new = compute_corr(h_new)
        c_new = corr_new.max() * (2.0 / n)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new
            best_corr = corr_new

    # ---------- simulated annealing ----------
    sa_time = budget_s * 0.2
    curr_h = best_h.copy()
    curr_corr = best_corr.copy()
    curr_c = best_c
    temp0 = best_c * 0.001
    sa_start = time.time()
    while time.time() - sa_start < sa_time:
        frac = (time.time() - sa_start) / sa_time
        temp = temp0 * (1.0 - frac)
        ones = np.where(curr_h == 1)[0]
        zeros = np.where(curr_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = curr_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        corr_new = compute_corr(h_new)
        c_new = corr_new.max() * (2.0 / n)
        delta = c_new - curr_c
        if delta < 0.0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            curr_h = h_new
            curr_corr = corr_new
            curr_c = c_new
            if c_new < best_c:
                best_h = h_new
                best_c = c_new
                best_corr = corr_new

    # ---------- SLSQP refinement ----------
    # soft‑max objective (beta will be set in the loop)
    def softmax_obj(x):
        inv = 1.0 - x
        L = 1 << ((2 * n - 1).bit_length())
        H = np.fft.rfft(x, n=L)
        I = np.fft.rfft(inv, n=L)
        conv = np.fft.irfft(H * I, n=L)[: 2 * n - 1]
        bc = CURRENT_BETA * conv
        m = bc.max()
        # stable log‑sum‑exp
        return (m + np.log(np.exp(bc - m).sum())) / CURRENT_BETA

    def sum_eq(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}

    # start point for SLSQP
    x0 = best_h.copy()
    x0 = np.clip(x0, 0.0, 1.0)
    x0 = enforce_sum_continuous(x0)

    beta_vals = [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0, 200000.0]
    slsqp_time = budget_s * 0.09
    while time.time() - start < budget_s * 0.99 and (time.time() - start) < (budget_s - slsqp_time):
        # leave a small margin for final checks
        break
    for beta in beta_vals:
        if time.time() - start > budget_s * 0.99:
            break
        CURRENT_BETA = beta
        res = minimize(
            softmax_obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 2000, "ftol": 1e-12, "disp": False},
        )
        x_opt = np.clip(res.x, 0.0, 1.0)
        x_opt = enforce_sum_continuous(x_opt)
        c_val = compute_c5(x_opt)
        if c_val < best_c:
            best_c = c_val
            best_h = x_opt.copy()
        x0 = x_opt.copy()  # warm‑start next beta

    # ---------- final sanity ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_continuous(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
