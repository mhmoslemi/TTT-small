    best_c = compute_c5(best)
    for _ in range(max_iters):
        # random donor and receiver
        donors = np.where(best > 1e-12)[0]

        nh[j] += eps

def project_onto_box_sum(v, target, max_iter=50):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        lam = 0.5 * (lo + hi)
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - 0.5 * (lo+hi), 0.0, 1.0)
    # adjust small leftover
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff over entries not at bounds
        mask = (x > 1e-12) & (x < 1-1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
    return x
```

def worst_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

    best_h = h.copy()

```
def softmax_gradient(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    Omax = corr.max()
    exp_term = np.exp(beta * (corr - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad
```

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target, max_iter=30):
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = 0.5 * (lo + hi)
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > target:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = abs(np.argmax(corr) - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    n = h.shape[0]
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    best_h = project_onto_box_sum(best_h, target)
    return best_h, best_c

def softmax_gradient(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    Omax = corr.max()
    w = np.exp(beta * (corr - Omax))
    w /= w.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def smooth_gradient_descent(h0, beta, target, max_iter, step0):
    h = h0.copy()
    vel = np.zeros_like(h)
    for it in range(max_iter):
        grad = softmax_gradient(h, beta)
        vel = 0.9 * vel - (step0 / np.sqrt(it + 1.0)) * grad
        h = project_onto_box_sum(h + vel, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    budget = budget_s

    if 'initial_h_values' in globals():
