def enforce_sum(h, rng):
    target = h.size/2
    diff = target - h.sum()
    # small threshold
    if abs(diff) < 1e-12:
        return h
    # More efficient: find indices where we can add or subtract.
    if diff > 0:
        # need to increase sum
        inds = np.where(h < 1)[0]
        while diff > 1e-12 and inds.size > 0:
            i = rng.choice(inds)
            max_inc = min(1.0 - h[i], diff)
            h[i] += max_inc
            diff -= max_inc
            if h[i] >= 1.0 - 1e-12:
                inds = np.delete(inds, np.where(inds == i))
    else:
        # need to decrease sum
        inds = np.where(h > 0)[0]
        while diff < -1e-12 and inds.size > 0:
            i = rng.choice(inds)
            max_dec = min(h[i], -diff)
            h[i] -= max_dec
            diff += max_dec
            if h[i] <= 1e-12:
                inds = np.delete(inds, np.where(inds == i))
    return h

def compute_c5_and_gradient(h):
    n = h.size
    size = 2*n
    f = h
    g = 1 - h
    g_rev = g[::-1]
    fft_f = np.fft.fft(f, size)
    fft_g = np.fft.fft(g_rev, size)
    corr = np.fft.ifft(fft_f * fft_g).real
    max_idx = int(np.argmax(corr))
    max_corr = corr[max_idx]
    k = abs(max_idx - (n - 1))
    grad = np.zeros(n)
    if k == 0:
        grad = 1 - 2*h
    else:
        grad[:n - k] += 1 - h[k:]
        grad[k:] -= h[:n - k]
    return max_corr * (2.0 / n), grad

def softmax_obj(x):

    fft_f = np.fft.fft(x, size)
    fft_g = np.fft.fft((1 - x)[::-1], size)

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def guided_hill_climb(h, rng, time_budget, step_scale_start=1.0, decay=0.99):
    start = time.time()
    best_h = h.copy()
    best_c, _ = compute_c5_and_gradient(h)
    step_scale = step_scale_start
    it = 0
    while time.time() - start < time_budget:
        c, grad = compute_c5_and_gradient(h)
        # find i,j
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
                # Ensure sum unchanged (we moved mass)
                # Might be slight floating errors: we can enforce sum at intervals
        # Update best
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # Decay step_scale occasionally
        it += 1
        if it % 1000 == 0:
            step_scale = max(step_scale * decay, 1e-4)
        # Occasionally enforce sum to correct drift
        if it % 5000 == 0:
            h = enforce_sum(h, rng)
    # final enforce sum
    h = enforce_sum(h, rng)
    # recompute best_c for best_h in case
    best_c, _ = compute_c5_and_gradient(best_h)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=10):

def slsqp_optimize(h0, beta, rng, max_iter=2000):
    global CURRENT_TARGET, CURRENT_BETA
    n = len(h0)
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    # we need softmax_obj defined
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - np.sum(x)
    if abs(diff) > 1e-12:
        # simple redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - np.sum(x)
    return x

def continuous_optimize(n, rng, time_budget, init_h):
    start = time.time()
    # hill climb
    hill_time = time_budget * 0.5
    best_h, best_c = guided_hill_climb(init_h.copy(), rng, hill_time, step_scale_start=0.5)
    elapsed = time.time() - start
    remaining = time_budget - elapsed
    if remaining <= 0:
        return best_h, best_c
    # SLSQP phases
    beta_seq = [20., 80., 200., 800., 3000., 8000., 20000.]
    for beta in beta_seq:
        if remaining <= 0:
            break
        max_iter = 1500 if beta >= 200 else 1000
        h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
        c, _ = compute_c5_and_gradient(h)
        if c < best_c:
            best_h, best_c = h, c
        elapsed = time.time() - start
        remaining = time_budget - elapsed
    # gradient refine
    if remaining > 0:
        h = gradient_refine(best_h, rng, step=0.015, max_iter=6)
        c, _ = compute_c5_and_gradient(h)
        if c < best_c:
            best_h, best_c = h, c
    # random restarts while time left
    while time.time() - start < time_budget * 0.98:
        # small perturbation of best
        h = best_h.copy()
        # random small swaps of mass
        for _ in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            eps = rng.random() * max_eps * 0.1
            h[i] -= eps
            h[j] += eps
        h = enforce_sum(h, rng)
        # short hill climb
        short_time = min(remaining * 0.2, 5.0)
        h2, c2 = guided_hill_climb(h, rng, short_time, step_scale_start=0.3)
        if c2 < best_c:
            best_h, best_c = h2, c2
        remaining = time_budget - (time.time() - start)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        inds = np.where(h < 1.0)[0]
        while diff > 1e-12 and inds.size:
            i = rng.choice(inds)
            inc = min(1.0 - h[i], diff)
            h[i] += inc
            diff -= inc
            if h[i] >= 1.0 - 1e-12:
                inds = np.delete(inds, np.where(inds == i))
    else:
        inds = np.where(h > 0.0)[0]
        while diff < -1e-12 and inds.size:
            i = rng.choice(inds)
            dec = min(h[i], -diff)
            h[i] -= dec
            diff += dec
            if h[i] <= 1e-12:
                inds = np.delete(inds, np.where(inds == i))
    return h


def compute_c5_and_gradient(h: np.ndarray):
    """Return C5 (scaled) and gradient of the worst shift."""
    n = h.size
    size = 2 * n
    fft_f = np.fft.fft(h, size)
    fft_g = np.fft.fft((1.0 - h)[::-1], size)
    corr = np.fft.ifft(fft_f * fft_g).real
    idx = int(np.argmax(corr))
    max_corr = corr[idx]
    k = abs(idx - (n - 1))
    grad = np.zeros(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return max_corr * (2.0 / n), grad


def compute_c5(h: np.ndarray) -> float:
    """Scaled C5 value only."""
    n = h.size
    size = 2 * n
    fft_f = np.fft.fft(h, size)
    fft_g = np.fft.fft((1.0 - h)[::-1], size)
    corr = np.fft.ifft(fft_f * fft_g).real
    return np.max(corr) * (2.0 / n)


def softmax_obj(x):
    n = x.size
    size = 2 * n
    fft_f = np.fft.fft(x, size)
    fft_g = np.fft.fft((1.0 - x)[::-1], size)
    corr = np.fft.ifft(fft_f * fft_g).real
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - np.sum(x)
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - np.sum(x)
    return x


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      time_budget: float, step_scale_start: float = 0.5,
                      decay: float = 0.99):
    start = time.time()
    best_h = h.copy()
    best_c, _ = compute_c5_and_gradient(best_h)
    step = step_scale_start
    it = 0
    while time.time() - start < time_budget:
        c, grad = compute_c5_and_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step
                h[i] -= eps
                h[j] += eps
        if c < best_c:
            best_c, best_h = c, h.copy()
        it += 1
        if it % 1000 == 0:
            step = max(step * decay, 1e-4)
        if it % 5000 == 0:
            h = enforce_sum(h, rng)
    h = enforce_sum(h, rng)
    best_c, _ = compute_c5_and_gradient(best_h)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    for it in range(max_iter):
        _, grad = compute_c5_and_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def continuous_optimize(n: int, rng: np.random.Generator,
                        time_budget: float, init_h: np.ndarray):
    start = time.time()
    best_h, best_c = guided_hill_climb(init_h.copy(), rng,
                                       time_budget * 0.5, step_scale_start=0.5)
    elapsed = time.time() - start
    remaining = time_budget - elapsed
    if remaining <= 0:
        return best_h, best_c
    beta_seq = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in beta_seq:
        if remaining <= 0:
            break
        max_iter = 1500 if beta >= 200 else 1000
        h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
        c, _ = compute_c5_and_gradient(h)
        if c < best_c:
            best_h, best_c = h, c
        elapsed = time.time() - start
        remaining = time_budget - elapsed
    if remaining > 0:
        h = gradient_refine(best_h, rng, step=0.015, max_iter=6)
        c, _ = compute_c5_and_gradient(h)
        if c < best_c:
            best_h, best_c = h, c
    while time.time() - start < time_budget * 0.98:
        h = best_h.copy()
        for _ in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            eps = rng.random() * max_eps * 0.1
            h[i] -= eps
            h[j] += eps
        h = enforce_sum(h, rng)
        short = min(5.0, (time_budget - (time.time() - start)) * 0.2)
        h2, c2 = guided_hill_climb(h, rng, short, step_scale_start=0.3)
        if c2 < best_c:
            best_h, best_c = h2, c2
    return best_h, best_c


def final_tweaks(h: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = h.copy()
    best_c, _ = compute_c5_and_gradient(best_h)
    while time.time() - start < time_budget:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nh = enforce_sum(nh, rng)
            nc, _ = compute_c5_and_gradient(nh)
            if nc < best_c:
                best_h, best_c = nh, nc
        # occasional random perturbation
        if rng.random() < 0.07:
            a, b = rng.integers(0, h.size, size=2)
            if a != b:
                max_eps = min(best_h[a], 1.0 - best_h[b])
                eps = rng.random() * max_eps * 0.05
                nh = best_h.copy()
                nh[a] -= eps
                nh[b] += eps
                nh = enforce_sum(nh, rng)
                nc, _ = compute_c5_and_gradient(nh)
                if nc < best_c:
                    best_h, best_c = nh, nc
    return best_h, best_c


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (same as in compute_c5_and_gradient)."""
    n = h.size
    size = 2 * n
    fft_f = np.fft.fft(h, size)
    fft_g = np.fft.fft((1.0 - h)[::-1], size)
    corr = np.fft.ifft(fft_f * fft_g).real
    idx = int(np.argmax(corr))
    k = abs(idx - (n - 1))
    grad = np.zeros(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    n = 768  # < 1k samples, enough flexibility

    if 'initial_h_values' in globals():
        old_h = np.array(initial_h_values, dtype=float)
        old_n = old_h.size

        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, old_h)
        h0 = np.clip(h0, 0.0, 1.0)

        h0 = enforce_sum(h0, rng)

    # Main optimization
    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        cont_time = remaining * 0.85

    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        best_h, best_c = final_tweaks(best_h, rng, remaining)

    best_h = np.clip(best_h, 0.0, 1.0)

    final_c, _ = compute_c5_and_gradient(best_h)
