import time
import numpy as np

# ----------------------------------------------------------------------
# Constraint handling
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h

# ----------------------------------------------------------------------
# C5 evaluation (linear correlation, matches original np.correlate)
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    N = h.size
    M = 2 * N - 1
    H = np.fft.fft(h, n=M)
    G = np.fft.fft(1.0 - h, n=M)
    corr = np.fft.ifft(H * np.conj(G)).real
    return float(corr.max() * (2.0 / N))

# ----------------------------------------------------------------------
# Gradient of the worst shift (subgradient of max‑overlap)
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    N = h.size
    M = 2 * N - 1
    H = np.fft.fft(h, n=M)
    G = np.fft.fft(1.0 - h, n=M)
    corr = np.fft.ifft(H * np.conj(G)).real
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] = 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    return grad

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift subgradient."""
    N = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # exact sum correction (quick redistribution)
        diff = N / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, N)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = N / 2.0 - h.sum()
    return h
