CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h, target, rng, beta=5000.0, max_iter=1000):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    # Option: supply initial guess h
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

def random_perturb(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

import time
import numpy as np

# ---------- utilities ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    diff = target - h.sum()
    n = h.shape[0]
    while abs(diff) > 1e-12:
        if diff > 0.0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            delta = min(1.0 - h[i], diff)
            h[i] += delta
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            delta = min(h[i], -diff)
            h[i] -= delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    ak = abs(k)
    grad = np.ones_like(h)
    if ak == 0:
        grad -= 2.0 * h
    else:
        grad[: n - ak] -= h[ak:]
        grad[ak:] -= h[: n - ak]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float = 5000.0, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


# ---------- local optimisation ----------
def local_opt(h0: np.ndarray, target: float, rng: np.random.Generator,
              max_iter: int, start: float, budget_s: float):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    alphas = (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625)
    for it in range(max_iter):
        if time.time() - start > budget_s * 0.96:
            break
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        _, grad = worst_shift_and_grad(h)
        improved = False
        for a in alphas:
            h_new = mass_transfer_move(h, grad, a, target, rng)
            c_new = compute_c5(h_new)
            if c_new < c - 1e-12:
                h = h_new
                improved = True
                break
        if not improved:
            if rng.random() < 0.07:
                h = random_perturb(h, rng, scale=0.3)
    return best_h, best_c


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Choose resolution (power of two, fast correlation)
    N = 1024
    target = N / 2.0

    best_h = rng.random(N)
    best_h = enforce_sum(best_h, target, rng)
    best_c = compute_c5(best_h)

    # occasional use of provided parent solution if compatible

        cand = np.array(initial_h_values, dtype=float)

            cand = enforce_sum(cand, target, rng)
            c0 = compute_c5(cand)
            if c0 < best_c:
                best_c, best_h = c0, cand.copy()

    # main multi‑restart loop
    while time.time() - start < budget_s * 0.94:
        h0 = rng.random(N)
        h0 = enforce_sum(h0, target, rng)
        h_opt, c_opt = local_opt(h0, target, rng, max_iter=2500,
                                 start=start, budget_s=budget_s)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
            if best_c < 0.38079:
                break

    # final polishing with SLSQP if time permits
    if time.time() - start < budget_s * 0.98:
        try:
            best_h = slsqp_refine(best_h, target, rng,
                                  beta=8000.0, max_iter=1200)
        except Exception:
            pass
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(N)
