def compute_c5(h):
    n = h.size
    L = 2*n - 1
    hf = np.fft.fft(h, L)
    one_m_h = 1.0 - h
    g = np.fft.fft(one_m_h, L)
    corr = np.fft.ifft(hf * np.conj(g)).real

    max_corr = np.max(corr)

def binary_sa(n, rng, time_budget, early_target=None):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.05, 1e-4  # larger initial temperature for exploration
    while time.time() - start < time_budget:
        # start from random binary vector
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        # create arrays of positions
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        # inner SA loop with time check
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            t_frac = elapsed / time_budget
            T = T0 * (T_end / T0) ** t_frac
            # select swap
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                # swap entries in arrays
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if early_target is not None and best_c <= early_target:
                        return best_h, best_c
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

def run(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start=time.time()
    total_budget=budget_s

    # initial h if available
    best_h = None
    best_c = np.inf
    best_n = None

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0,0.0,1.0)
        h0 = enforce_sum(h0,rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0,c0,h0.size

    # Phase 0: binary SA on 512
    remaining = total_budget - (time.time()-start)
    if remaining>0:
        t_bin = min(remaining*0.25, 300.0)  # up to 300 sec
        h_bin,c_bin=binary_sa(512,rng,t_bin,early_target=0.38079)
        if c_bin < best_c:

    if remaining>0 and best_h is not None:

            h_fine = upsample_h(best_h,1024)

            c_fine = compute_c5(h_fine)
            if c_fine < best_c:

            best_h = enforce_sum(best_h,rng)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using FFT correlation (linear, not circular)."""
    n = h.size
    L = 2 * n - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

# Soft‑max SLSQP optimisation
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 50) -> np.ndarray:
    """Iterative refinement using the worst‑shift gradient."""
    n = h.size
    L = 2 * n - 1
    target = n / 2.0
    for it in range(max_iter):
        H = np.fft.fft(h, L)
        G = np.fft.fft(1.0 - h, L)
        corr = np.fft.ifft(H * np.conj(G)).real
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        s = target - h.sum()
        if abs(s) > 1e-12:
            for _ in range(500):
                if abs(s) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(s, 1.0 - h[i]) if s > 0 else max(s, -h[i])
                h[i] += add
                s = target - h.sum()
    return h


def subgradient_descent(h: np.ndarray, rng: np.random.Generator,
                        max_iters: int = 200, init_step: float = 0.05) -> np.ndarray:
    """Sub‑gradient descent on the max‑overlap objective with back‑tracking."""
    n = h.size
    L = 2 * n - 1
    target = n / 2.0
    step = init_step
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    for it in range(1, max_iters + 1):
        H = np.fft.fft(h, L)
        G = np.fft.fft(1.0 - h, L)
        corr = np.fft.ifft(H * np.conj(G)).real
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        # back‑tracking line search
        for _ in range(10):
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                h_new = enforce_sum(h_new, rng)
            c_new = compute_c5(h_new)
            if c_new <= c_curr:
                h = h_new
                c_curr = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h_new.copy()
                break
            step *= 0.5
        step = init_step / np.sqrt(it + 1.0)
    return best_h

# Binary simulated annealing
def binary_sa(n: int, rng: np.random.Generator,
              time_budget: float, early_target: float = None):
    """Binary SA with exact n/2 ones and periodic restarts."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.05, 1e-4
    while time.time() - start < time_budget:
        # fresh start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        swap_cnt = 0
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            frac = elapsed / time_budget
            T = T0 * (T_end / T0) ** frac
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if early_target is not None and best_c <= early_target:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
            swap_cnt += 1
            if swap_cnt % 5000 == 0:
                # restart to escape possible traps
                break
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
    return best_h, best_c

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c

        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, 512

    # ---- Phase 1: Upsample to 1024 ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        if best_n < 1024:
            h_up = upsample_h(best_h, 1024)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, 1024
        else:
            best_h = enforce_sum(best_h, rng)
            best_c = compute_c5(best_h)
            best_n = best_h.size

    # ---- Phase 2: Soft‑max SLSQP refinement on 1024 ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None and best_n == 1024:
        h = best_h.copy()
        betas = [20., 80., 200., 800., 3000., 8000., 20000., 50000., 100000.]
        for beta in betas:
            if time.time() - start > total_budget * 0.95:
                break
            iters = 800 if beta < 2000 else 1000
            h = slsqp_optimize(h, beta, rng, max_iter=iters)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
        best_n = h.size
        best_h = h

    # ---- Phase 3: Gradient tidy‑up ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        h = gradient_refine(best_h, rng, step=0.015, max_iter=50)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- Phase 4: Sub‑gradient descent for extra polish ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        h = subgradient_descent(best_h, rng, max_iters=200, init_step=0.05)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- Phase 5: Continuous SA on the finest grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        h, c = continuous_sa(best_h, rng, remaining)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- Final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
