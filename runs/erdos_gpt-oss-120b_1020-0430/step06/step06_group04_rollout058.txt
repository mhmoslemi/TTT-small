def grad_descent_smoothmax_sym(free, betas, base_step, it_per_beta, rng, deadline):
    n = free.size * 2
    half = free.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline: return free
            # compute full h
            h = np.concatenate([free, 1 - free[::-1]])
            # compute gradient for full, then free gradient
            grad_full = compute_full_smoothed_gradient(h, beta) # returns full gradient
            grad = grad_full[:half] - grad_full[half:][::-1]
            # step size line search
            best_free = free
            best_c = compute_c5(h)  # evaluate current
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand_free = free - step * grad
                np.clip(cand_free, 0.0, 1.0, out=cand_free)
                cand_h = np.concatenate([cand_free, 1 - cand_free[::-1]])

                if c < best_c:
                    best_c = c
                    best_free = cand_free
            free = best_free
    return free

Implement compute_full_smoothed_gradient:

def compute_full_smoothed_gradient(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def worst_shift_gradient_sym(free):
    half = free.size
    h = np.concatenate([free, 1 - free[::-1]])
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros(n)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g = g / max(1, len(idxs))
    # free gradient
    free_grad = g[:half] - g[half:][::-1]
    return free_grad

def slsqp_optimize_sym(free0, beta, rng, max_iter=500):
    # using L-BFGS-B (faster)
    n_full = free0.size * 2
    half = free0.size
    target_sum = n_full / 2.0  # automatically satisfied

    def obj(free):
        h = np.concatenate([free, 1 - free[::-1]])
        v = np.correlate(h, 1.0 - h, mode="full")
        bc = beta * v
        m = np.max(bc)
        val = (m + np.log(np.sum(np.exp(bc - m)))) / beta
        return val

    def grad(free):
        h = np.concatenate([free, 1 - free[::-1]])
        g_full = compute_full_smoothed_gradient(h, beta)
        # free gradient: g_i - g_{n-1-i}
        return g_full[:half] - g_full[half:][::-1]

    bounds = [(0.0, 1.0)] * half
    res = minimize(obj, free0, method='L-BFGS-B', jac=grad, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    free_opt = np.clip(res.x, 0.0, 1.0)
    return free_opt

def binary_sa_sym(free_bin, rng, deadline):
    # free_bin is array of ints 0/1, length half_n
    cur = free_bin.copy()
    cur_h = np.concatenate([cur, 1 - cur[::-1]])
    cur_c = compute_c5(cur_h)
    best = cur.copy()
    best_c = cur_c
    # Temperature schedule
    init_T = max(1e-6, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.integers(0, half)
        # flip i
        cur[i] = 1 - cur[i]
        # compute new full h and C5
        cur_h = np.concatenate([cur, 1 - cur[::-1]])
        c = compute_c5(cur_h)
        # acceptance
        if c < cur_c:
            accept = True
        else:
            T = init_T * (0.9997 ** it) + 1e-12
            accept = rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            # update best
            if c < best_c:
                best_c = c
                best = cur.copy()
                it = 0  # reset stagnation
        else:
            # revert
            cur[i] = 1 - cur[i]
        it += 1
        # early break if no improvement for a while
        if it > 50000:
            break
    # return best full vector and best_c
    best_h = np.concatenate([best, 1 - best[::-1]])
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# Core evaluation
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


# ------------------------------------------------------------
# Smooth‑max gradient (full vector)
def _full_smoothed_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)          # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g


# ------------------------------------------------------------
# Gradient descent on smooth‑max (symmetric representation)
def grad_descent_smoothmax_sym(free: np.ndarray,
                               betas,
                               base_step: float,
                               it_per_beta: int,
                               rng,
                               deadline: float) -> np.ndarray:
    half = free.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return free
            # full vector
            h = np.concatenate([free, 1.0 - free[::-1]])
            g_full = _full_smoothed_gradient(h, beta)
            grad = g_full[:half] - g_full[half:][::-1]
            if np.linalg.norm(grad) < 1e-14:
                continue
            # current value
            best_c = compute_c5(h)
            best_free = free
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = free - step * grad
                np.clip(cand, 0.0, 1.0, out=cand)
                cand_h = np.concatenate([cand, 1.0 - cand[::-1]])
                c = compute_c5(cand_h)
                if c < best_c:
                    best_c = c
                    best_free = cand
            free = best_free
    return free


# ------------------------------------------------------------
# Worst‑shift sub‑gradient (symmetric)
def _worst_shift_gradient_sym(free: np.ndarray) -> np.ndarray:
    half = free.size
    h = np.concatenate([free, 1.0 - free[::-1]])
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros(n)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g[:half] - g[half:][::-1]


def worst_shift_refine_sym(free: np.ndarray,
                           rng,
                           base_step: float,
                           iters: int,
                           deadline: float) -> np.ndarray:
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        grad = _worst_shift_gradient_sym(free)
        if np.linalg.norm(grad) < 1e-14:
            break
        # current value
        h = np.concatenate([free, 1.0 - free[::-1]])
        best_c = compute_c5(h)
        best_free = free
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = free - step * mult * grad
            np.clip(cand, 0.0, 1.0, out=cand)
            cand_h = np.concatenate([cand, 1.0 - cand[::-1]])
            c = compute_c5(cand_h)
            if c < best_c:
                best_c = c
                best_free = cand
        free = best_free
        step *= 0.95
    return free


# ------------------------------------------------------------
# Simulated‑annealing on binary symmetric vectors
def binary_sa_sym(free_bin: np.ndarray,
                  rng,
                  deadline: float,
                  max_iters: int = 8000) -> (np.ndarray, float):
    """Binary SA on the free half; returns full binary h and its C5."""
    half = free_bin.size
    cur = free_bin.copy()
    cur_h = np.concatenate([cur, 1.0 - cur[::-1]])
    cur_c = compute_c5(cur_h)
    best = cur.copy()
    best_c = cur_c
    init_T = max(1e-6, cur_c * 0.02)
    it = 0
    while time.time() < deadline and it < max_iters:
        i = rng.integers(0, half)
        cur[i] = 1 - cur[i]                     # flip pair
        cur_h = np.concatenate([cur, 1.0 - cur[::-1]])
        c = compute_c5(cur_h)
        if c < cur_c:
            accept = True
        else:
            T = init_T * (0.9997 ** it) + 1e-12
            accept = rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                it = 0
        else:
            cur[i] = 1 - cur[i]                 # revert
        it += 1
        if it > 5000 and best_c < cur_c * 0.999:
            break
    best_h = np.concatenate([best, 1.0 - best[::-1]])
    return best_h, best_c

    # Binary simulated annealing on the best (maybe up‑sampled) solution
    if best_free is not None and time.time() < deadline:
        h_bin, c_bin = binary_sa_sym(best_free, rng, deadline, max_iters=8000)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), best_h.shape[0]

    # --------------------------------------------------------
    # Final safety fallback
    if best_h is None:
        n = 540
        half = n // 2
        free = np.ones(half)
        best_h = np.concatenate([free, 1.0 - free[::-1]])
        best_n = n
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
