def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                              'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)

            x[i] += add
            diff = CURRENT_TARGET - x.sum()

import time, math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


# ------------------- soft‑max SLSQP helpers --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Run SLSQP on the soft‑max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- choose size -------------------------------------------------
    n = 1024                     # finer grid
    half = n // 2

    # ----- initialise binary h ----------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.size
        # balanced binary from the values (largest half become 1)
        order = np.argsort(-init)
        bin0 = np.zeros(n0, dtype=np.uint8)
        bin0[order[: n0 // 2]] = 1
        # upscale to n (must be integer multiple)
        if n % n0 == 0:
            factor = n // n0
            h_bin = np.repeat(bin0, factor)
        else:
            # otherwise fill the remainder randomly
            factor = n // n0
            h_bin = np.repeat(bin0, factor)
            extra = n - h_bin.size
            extra_bits = np.zeros(extra, dtype=np.uint8)
            extra_bits[: extra // 2] = 1
            rng.shuffle(extra_bits)
            h_bin = np.concatenate([h_bin, extra_bits])
    else:
        h_bin = np.zeros(n, dtype=np.uint8)
        h_bin[rng.choice(n, half, replace=False)] = 1

    # ----- data structures for fast swaps ------------------------------
    ones_arr = np.where(h_bin == 1)[0].astype(np.int32)   # size half
    zeros_arr = np.where(h_bin == 0)[0].astype(np.int32)  # size half

    pos_one = np.full(n, -1, dtype=np.int32)
    pos_zero = np.full(n, -1, dtype=np.int32)
    pos_one[ones_arr] = np.arange(half, dtype=np.int32)
    pos_zero[zeros_arr] = np.arange(half, dtype=np.int32)

    # ----- initial autocorrelation S_k ---------------------------------
    S = np.correlate(h_bin, h_bin, mode="full")
    S = S[n - 1 :].astype(np.int64)          # S[0] .. S[n-1]

    # ----- current C5 and best so far ----------------------------------
    min_S = S[1:].min()
    curr_C5 = 1.0 - (2.0 / n) * min_S
    best_C5 = curr_C5
    best_h_bin = h_bin.copy()

    # ----- simulated annealing on binary space -------------------------
    T0 = 1e-3
    decay = 0.9999
    it = 0
    while time.time() - start < budget_s * 0.78:
        # pick a 1 and a 0
        i_idx = rng.integers(half)
        j_idx = rng.integers(half)
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]

        # distances from i and j to all other ones (excluding i)
        mask = ones_arr != i
        p_others = ones_arr[mask]

        di = np.abs(i - p_others)
        dj = np.abs(j - p_others)

        # histograms of distances
        hist_i = np.bincount(di, minlength=n)
        hist_j = np.bincount(dj, minlength=n)

        # propose update
        S -= hist_i
        S += hist_j

        new_min_S = S[1:].min()
        new_C5 = 1.0 - (2.0 / n) * new_min_S

        # acceptance test
        if new_C5 < curr_C5:
            accept = True
        else:
            T = T0 * (decay ** it)
            if T < 1e-12:
                T = 1e-12
            prob = math.exp(-(new_C5 - curr_C5) / T)
            accept = rng.random() < prob

        if accept:
            # perform the swap in the index arrays
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
            pos_one[j] = i_idx
            pos_zero[i] = j_idx
            pos_one[i] = -1
            pos_zero[j] = -1

            curr_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h_bin = np.zeros(n, dtype=np.uint8)
                best_h_bin[ones_arr] = 1
                if best_C5 <= 0.3808:
                    break
        else:
            # revert autocorrelation
            S += hist_i
            S -= hist_j

        it += 1

    # ----- optionally improve with SLSQP on real values -----------------
    h_ref = best_h_bin.astype(float)
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0):
        if time.time() - start > budget_s * 0.95:
            break
        h_opt = slsqp_optimize(h_ref, beta, rng, max_iter=500)
        c_opt = compute_c5(h_opt)
        if c_opt < best_C5:
            best_C5 = c_opt
            h_ref = h_opt
            if best_C5 <= 0.3808:
                break

    # ----- final sanity and return ------------------------------------
    final_h = enforce_sum(h_ref, rng)
    final_c5 = compute_c5(final_h)
    return final_h.tolist(), float(final_c5), int(final_h.size)
