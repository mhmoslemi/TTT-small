def subgradient_optimize(h0, rng, start_time, end_time):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    # For schedule: step = 0.5 / sqrt(it+1)
    while time.time() < end_time:
        # compute correlation with 1-h to get O_k
        corr = np.correlate(h, 1 - h, mode='full')
        O = corr * (2.0 / h.size)  # note compute_c5 does max(corr)*(dx)
        # find worst shift index (max O)
        idx = np.argmax(O) - (h.size - 1)
        k = abs(idx)
        # compute gradient
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: h.size - k] -= h[k:]
            grad[k:] -= h[: h.size - k]
        # step size schedule
        step = 0.5 / np.sqrt(it + 1)
        # update
        h = h - step * grad
        # clip to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # enforce sum = n/2 exactly
        h = enforce_sum(h, rng)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        it += 1
    return best_h, best_c

def enforce_sum_fast(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total_cap = cap.sum()
        if total_cap <= 0:
            return h
        h = h + diff * cap / total_cap
    else:
        cap = h
        total_cap = cap.sum()
        if total_cap <= 0:
            return h
        h = h + diff * cap / total_cap  # diff is negative
    # clip for safety
    h = np.clip(h, 0.0, 1.0)
    # after clipping, fix residual diff maybe
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # small random adjustment
        n = h.size
        # For a few attempts
        for _ in range(10):
            i = np.random.randint(0, n)
            max_eps = min(residual, 1.0 - h[i]) if residual > 0 else max(residual, -h[i])
            if abs(max_eps) > 1e-16:
                h[i] += max_eps
                residual = target - h.sum()
                if abs(residual) < 1e-12:
                    break
    return h

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum_fast(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h so that sum(h)=target using proportional redistribution."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
    else:
        cap = h
    total = cap.sum()
    if total > 0:
        h = h + diff * cap / total
    h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust a single element deterministically
        i = np.argmin(h) if residual > 0 else np.argmax(h)
        h[i] = np.clip(h[i] + residual, 0.0, 1.0)
    return h


def enforce_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Exact repair of sum using random adjustments."""
    diff = target - h.sum()
    n = h.size
    for _ in range(n * 5):
        if abs(diff) < 1e-12:

        i = rng.integers(0, n)

    return h


def subgradient_optimize(h0: np.ndarray, rng: np.random.Generator,
                         start_time: float, end_time: float):
    """Sub‑gradient descent on max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    it = 0
    while time.time() < end_time:
        # correlation with 1‑h gives O_k up to a factor 2/n
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # gradient of the worst shift
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        step = 0.6 / np.sqrt(it + 1)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_fast(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        it += 1
    return best_h, best_c, it


def slsqp_optimize(h: np.ndarray, target: float, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(obj, h, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum_fast(x, target)
    return x

    target = n / 2.0
    h0 = enforce_sum_fast(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- multiple sub‑gradient runs (≈60% of budget) -----
    sub_end_total = start + budget_s * 0.6
    while time.time() < sub_end_total:
        # pick a start: occasionally reuse the parent construction
        if rng.random() < 0.25 and "initial_h_values" in globals():
            h_start = np.array(initial_h_values, dtype=float)
            h_start = np.clip(h_start, 0.0, 1.0)
        else:
            h_start = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            h_start[idx] = 1.0
        h_start = enforce_sum_fast(h_start, target)
        # allocate a slice of remaining sub‑gradient time
        now = time.time()
        slice_end = min(sub_end_total, now + budget_s * 0.05)
        h_opt, c_opt, _ = subgradient_optimize(h_start, rng, now, slice_end)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- gradient polishing (≈10% of budget) -----
    pol_end = start + budget_s * 0.7
    if time.time() < pol_end:
        # a short sub‑gradient refinement with finer steps
        h_opt, c_opt, _ = subgradient_optimize(best_h, rng, time.time(), pol_end)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- SLSQP polishing (≈20% of budget) -----
    slsqp_end = start + budget_s * 0.95
    if time.time() < slsqp_end:
        beta_seq = (50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0)
        for beta in beta_seq:
            if time.time() > slsqp_end:
                break
            best_h = slsqp_optimize(best_h, target, beta, rng, max_iter=400)
            best_c = compute_c5(best_h)

    # ----- final exact repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_exact(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
