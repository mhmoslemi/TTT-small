import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for SLSQP constraints
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Simple random redistribution loop; converges quickly because diff shrinks.
    while abs(diff) > 1e-12:
        idx = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
        else:
            add = max(diff, -h[idx])
        h[idx] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Either swap two entries or transfer a random amount while preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.2:  # pure swap
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    # transfer mass
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum after each iteration (quick random fix)
        diff = (n / 2.0) - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff to a random feasible index
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                idx = rng.integers(0, n)
                add = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, -h[idx])
                h[idx] += add
                diff = (n / 2.0) - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    """Smooth‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise -----
    n = 1024
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # up‑sample to n points (repeat if integer factor, else linear interp)
        if h0.size != n:
            factor = n // h0.size
            if factor * h0.size == n:
                h0 = np.repeat(h0, factor)
            else:
                x_old = np.linspace(0, 2, h0.size, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = h0.copy()
        best_c = compute_c5(best_h)

    # ----- simulated annealing -----
    current_h, current_c = best_h.copy(), best_c
    SA_deadline = start + budget_s * 0.55
    T = best_c * 0.05 if best_c > 0 else 0.001
    eps_scale = 0.5
    it = 0
    update_interval = 1000
    while time.time() < SA_deadline:
        if rng.random() < 0.4:
            grad = worst_gradient(current_h)
            nh = guided_perturb(current_h, grad, rng, scale=eps_scale)
        else:
            nh = random_move(current_h, rng, scale=eps_scale)
        nc = compute_c5(nh)
        delta = nc - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_h, current_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        it += 1
        if it % update_interval == 0:
            T *= 0.996
            eps_scale *= 0.996
            T = max(T, 1e-10)
            eps_scale = max(eps_scale, 1e-6)

    # ----- gradient polishing -----
    if time.time() < start + budget_s * 0.8:
        refined = gradient_refine(best_h, rng, step=0.02, max_iter=12)
        rc = compute_c5(refined)
        if rc < best_c:
            best_h, best_c = refined.copy(), rc

    # ----- soft‑max SLSQP passes -----
    # use remaining time but leave a small safety margin
    while time.time() < start + budget_s * 0.96:
        # adapt beta based on current C5 scale
        beta = max(100.0, min(200000.0, 2000.0 / max(best_c, 1e-6)))
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
        best_c = compute_c5(best_h)

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
