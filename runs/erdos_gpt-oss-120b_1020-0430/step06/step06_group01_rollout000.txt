    H = np.fft.rfft(h, n2)

def binary_sa_fft(n, rng, time_budget, initial_h=None):
    start = time.time()
    # initialize binary vector
    if initial_h is not None:
        # initial_h may be fractional; convert to binary with same sum
        # Choose top n/2 indices as 1
        indices = np.argsort(-initial_h)[:n//2]
        h = np.zeros(n, dtype=float)
        h[indices] = 1.0
    else:
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=float)
        h[perm[:n//2]] = 1.0
    # initial ones/zeros arrays
    ones_arr = np.empty(n//2, dtype=np.int64)
    zeros_arr = np.empty(n//2, dtype=np.int64)
    pos = np.empty(n, dtype=np.int64)
    is_one = np.zeros(n, dtype=bool)
    # fill
    ones_arr[:] = np.where(h == 1)[0]
    zeros_arr[:] = np.where(h == 0)[0]
    is_one[ones_arr] = True
    pos[ones_arr] = np.arange(n//2)
    pos[zeros_arr] = np.arange(n//2)

    cur_c = max_overlap_fft(h)
    best_h = h.copy()
    best_c = cur_c
    # cooling

    T_end = 1e-8
    # iteration loop
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        fraction = elapsed / time_budget
        T = T0 * (T_end / T0) ** fraction
        # random swap
        i = ones_arr[rng.integers(n//2)]
        j = zeros_arr[rng.integers(n//2)]
        # swap in copy? We'll modify h in-place and revert if not accepted

# ------------------- basic utilities -------------------

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution, limited iterations
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using direct correlation (exact)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def max_overlap_fft(h: np.ndarray) -> float:
    """Fast C5 evaluation via FFT (same scaling as compute_c5)."""
    n = h.size
    n2 = n * 2
    H = np.fft.rfft(h, n2)
    G = np.fft.rfft(1.0 - h, n2)
    c = np.fft.irfft(H * G, n2)
    return np.max(c[:2 * n - 1]) * (2.0 / n)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

                      max_iters: int = 20000, step_scale: float = 0.7):
    """Gradient‑directed hill climbing."""
    best_h = h.copy()
    best_c = compute_c5(h)

        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ------------------- soft‑max SLSQP -------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ------------------- binary SA with FFT -------------------

def binary_sa_fft(n: int, rng: np.random.Generator,
                 time_budget: float, init_h: np.ndarray | None = None):
    """Binary simulated annealing using fast FFT evaluation."""
    start = time.time()
    # initialise binary vector
    if init_h is not None:
        # turn fractional init into binary with correct sum
        idx = np.argsort(-init_h)[: n // 2]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=float)
        h[perm[: n // 2]] = 1.0

    # structures for O(1) swaps
    ones_arr = np.empty(n // 2, dtype=np.int64)
    zeros_arr = np.empty(n // 2, dtype=np.int64)
    pos = np.empty(n, dtype=np.int64)
    is_one = np.zeros(n, dtype=bool)

    ones_arr[:] = np.where(h == 1)[0]
    zeros_arr[:] = np.where(h == 0)[0]
    is_one[ones_arr] = True
    pos[ones_arr] = np.arange(n // 2)
    pos[zeros_arr] = np.arange(n // 2)

    cur_c = max_overlap_fft(h)
    best_h = h.copy()
    best_c = cur_c

    T0 = 0.002
    T_end = 1e-8

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        f = elapsed / time_budget
        T = T0 * (T_end / T0) ** f

        i = ones_arr[rng.integers(n // 2)]
        j = zeros_arr[rng.integers(n // 2)]

        h[i] = 0.0
        h[j] = 1.0
        new_c = max_overlap_fft(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_c = new_c
            i_pos = pos[i]
            j_pos = pos[j]
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            is_one[i] = False
            is_one[j] = True
            pos[i] = j_pos
            pos[j] = i_pos
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            if best_c <= 0.3808:
                break
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c


# ------------------- continuous refinement -------------------

def continuous_optimize(h0: np.ndarray, rng: np.random.Generator,
                       time_budget: float):
    """Run hill‑climb and a short SLSQP schedule."""
    start = time.time()
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # hill‑climb phase
    remaining = time_budget - (time.time() - start)
    if remaining > 0.1:
        max_iters = int(min(60000, remaining * 800))
        h, c = guided_hill_climb(best_h.copy(), rng,
                                 max_iters=max_iters, step_scale=0.5)
        if c < best_c:
            best_h, best_c = h, c

    # short SLSQP passes
    beta_seq = (50.0, 200.0, 800.0, 2000.0)
    for beta in beta_seq:
        if time.time() - start > time_budget * 0.96:
            break
        h = slsqp_optimize(best_h, beta, rng, max_iter=600)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c

    # tiny random tweaks
    while time.time() - start < time_budget * 0.998:
        i = rng.integers(0, best_h.size)
        j = rng.integers(0, best_h.size)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = enforce_sum(nh, rng)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    return best_h, best_c


# ------------------- main entry point -------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # initial solution if provided
    best_h = None
    best_c = np.inf
    n = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        n = h0.size
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, n = h0, c0, n

    # target size
    target_n = 512
    if n is None:
        n = target_n

    # binary SA (≈75% of time)
    binary_time = total * 0.75
    h_bin, c_bin = binary_sa_fft(n, rng, binary_time,
                                 init_h=best_h if best_h is not None else None)
    if c_bin < best_c:
        best_h, best_c, n = h_bin.copy(), c_bin, n

    # continuous refinement (remaining time)
    remaining = total - (time.time() - start)
    if remaining > 2.0:
        # ensure float array for refinement
        h_float = best_h.astype(float)
        h_float = enforce_sum(h_float, rng)
        h_cont, c_cont = continuous_optimize(h_float, rng, remaining)
        if c_cont < best_c:
            best_h, best_c, n = h_cont.copy(), c_cont, n

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
