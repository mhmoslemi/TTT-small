def gradient_J(h, lam):
    n = len(h)
    O = np.empty(n)
    # compute O
    for k in range(n):
        i_end = n - k
        O[k] = np.dot(h[:i_end], 1 - h[k:k+i_end])
    O_max = O.max()
    O_mean = O.mean()
    c = 2 * lam * (O - O_mean)
    # add 1 to coefficient for max k
    max_k = np.argmax(O)
    c[max_k] += 1.0
    # compute grad
    grad = np.zeros(n)

        # contributions from i where i >= k:
        grad[k:] -= coeff * h[:i_end]
    return grad

def anneal(h, rng, max_moves, init_T, final_T):
    n = h.size
    target = n/2
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    T_start = init_T
    T_end = final_T
    for move in range(max_moves):
        t = move / max_moves
        T = T_start * (1 - t) + T_end * t
        # propose transfer
        i, j = rng.choice(n, 2, replace=False)
        # compute max transfer epsilon
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        # apply
        cur_h[i] -= eps
        cur_h[j] += eps
        # compute c
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            # accept
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps
    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    target = n / 2.0
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 512
        target = n / 2.0
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # step sizes for each beta
    step_map = {
        20: 0.5, 100: 0.45, 500: 0.35,
        2000: 0.25, 8000: 0.18,
        20000: 0.12, 50000: 0.09, 100000: 0.07
    }
    beta_choices = [20, 100, 500, 2000, 8000, 20000, 50000, 100000]

    # Main iterative search
    while time.time() - start < budget_s * 0.94:
        # occasional random jitter to escape plateaus
        if rng.random() < 0.4:
            noise = rng.normal(scale=0.01, size=n)
            cur_h = best_h + noise
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = project_onto_box_sum(cur_h, target)
        else:
            cur_h = best_h.copy()

        if rng.random() < 0.7:                     # softmax gradient descent
            beta = int(rng.choice(beta_choices))
            step0 = step_map.get(beta, 0.06)
            if beta >= 20000:
                steps = int(3000 * 256 / n)
            else:
                steps = int(2000 * 256 / n)
            for it in range(steps):
                if time.time() - start > budget_s * 0.94:
                    break
                grad = softmax_gradient(cur_h, beta)
                norm = np.linalg.norm(grad)
                if norm > 0:
                    step = step0 * (1.0 - it / steps) / norm
                else:
                    step = 0.0
                cur_h = project_onto_box_sum(cur_h - step * grad, target)
            c = compute_c5(cur_h)
            if c < best_c:
                best_h, best_c = cur_h.copy(), c
        else:                                      # guided hill climbing
            hc, cc = guided_hill_climb(
                cur_h, rng,
                max_iters=int(2000 * 256 / n),
                step_scale=0.3
            )
            if cc < best_c:
                best_h, best_c = hc.copy(), cc

    # Final refinement with tiny perturbations
    while time.time() - start < budget_s * 0.99:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
