def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.shape[0]
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
        else:
            cap = -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h
```

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad
```

def gradient_refine(h, target, rng, step=0.01, max_iter=30):
    n = h.size
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
```

def slsqp_refine(h, target, rng, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                              'ftol': 1e-12,
                                              'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

import time
import numpy as np
from scipy.optimize import linprog

# ---------- helper functions ----------

def enforce_sum(h, target, rng):
    """Adjust h to have exact sum=target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
        else:
            cap = -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_shift_and_grad(h):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def gradient_refine(h, target, rng, step=0.01, max_iter=30):
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


# ----- soft‑max and SLSQP refinement -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h, target, rng, beta, max_iter=300):
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024
    target = n / 2.0

    # ---- initialise h ----
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n and n % init.size == 0:
            h = np.repeat(init, n // init.size)
        else:
            h = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0
    else:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    h = enforce_sum(h, target, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    max_iters = 100
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.94:
            break

        # raw correlation (no scaling)
        raw_corr = np.correlate(h, 1.0 - h, mode="full")
        O_raw = raw_corr[n - 1 :]  # length n, non‑negative shifts

        # build gradient matrix
        grads = np.empty((n, n), dtype=np.float64)
        for s in range(n):
            grad = np.ones(n, dtype=np.float64)
            if s == 0:
                grad -= 2.0 * h
            else:
                grad[: n - s] -= h[s:]
                grad[s:] -= h[: n - s]
            grads[s] = grad

        # linear program: minimize t subject to O_raw + grads·d <= t
        A_ub = np.hstack([grads, -np.ones((n, 1), dtype=np.float64)])
        b_ub = -O_raw.astype(np.float64)
        A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
        b_eq = np.array([0.0], dtype=np.float64)
        bounds = [(-h_i, 1.0 - h_i) for h_i in h] + [(None, None)]

        res = linprog(
            c=np.concatenate([np.zeros(n), [1.0]]),
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )

        if res.success:
            d = res.x[:n]
            h_new = h + d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target, rng)
            new_c = compute_c5(h_new)
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h_new.copy()
                h = h_new
                # small gradient polishing
                h_ref = gradient_refine(h, target, rng, step=0.01, max_iter=5)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c = c_ref
                    best_h = h_ref.copy()
                    h = h_ref
                continue

        # occasional random mass transfer when LP offers no gain
        if rng.random() < 0.2:
            eps = rng.random(n) * 0.02
            mask = rng.random(n) < 0.5
            h2 = h.copy()
            h2[mask] = np.clip(h2[mask] + eps[mask], 0.0, 1.0)
            h2[~mask] = np.clip(h2[~mask] - eps[~mask], 0.0, 1.0)
            h2 = enforce_sum(h2, target, rng)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                h = h2

    # ----- final polishing ----- #
    h_fin = gradient_refine(best_h, target, rng, step=0.008, max_iter=15)
    c_fin = compute_c5(h_fin)
    if c_fin < best_c:
        best_c = c_fin
        best_h = h_fin.copy()

    # ----- optional SLSQP refinement if time permits ----- #
    if time.time() - start < budget_s * 0.98:
        for beta in (2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            h_s = slsqp_refine(best_h, target, rng, beta=beta, max_iter=200)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # ----- local random swaps in remaining time ----- #
    while time.time() - start < budget_s * 0.99:
        idx = rng.integers(0, n, size=5)
        jdx = rng.integers(0, n, size=5)
        h2 = best_h.copy()
        for a, b in zip(idx, jdx):
            delta = min(h2[a], 1.0 - h2[b]) * rng.random() * 0.05
            h2[a] -= delta
            h2[b] += delta
        h2 = enforce_sum(h2, target, rng)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
