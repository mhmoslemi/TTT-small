def sa_step(h, current_c, T, scale):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # random direction: move from i to j
    # choose direction based on a coin
    if rng.random() < 0.5:
        # move from i to j
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 0:
            return h, current_c, False
        eps = rng.random() * eps_max * scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
    else:
        # move from j to i
        eps_max = min(h[j], 1.0 - h[i])
        if eps_max <= 0:
            return h, current_c, False
        eps = rng.random() * eps_max * scale
        h_new = h.copy()
        h_new[j] -= eps
        h_new[i] += eps

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} using binary search on the offset."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Double the resolution using linear interpolation and re‑project."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)


def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap (via worst shift)."""
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        # worst shift and gradient
        cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
        k = int(np.argmin(cross))
        grad = np.ones_like(h)
        if k != 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        # step size
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    total_budget = budget_s

    # ----------------------------------------------------------------------

    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c

    sa_budget = total_budget * 0.70
    sa_start = time.time()
    # temperature schedule
    T0, Tf = 0.02, 1e-6
    # epsilon scale schedule
    scale_max, scale_min = 0.5, 0.02

    it = 0
    while time.time() - sa_start < sa_budget:
        elapsed = time.time() - sa_start
        frac = elapsed / sa_budget
        T = T0 * (1 - frac) + Tf * frac
        scale = max(scale_min, scale_max * (1 - frac))

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # random direction: i -> j or j -> i
        if rng.random() < 0.5:
            # move from i to j
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:

            eps = rng.random() * max_eps * scale
            h[i] -= eps
            h[j] += eps
        else:

            if max_eps <= 0.0:

            eps = rng.random() * max_eps * scale
            h[j] -= eps
            h[i] += eps
