def compute_c5_and_gradient(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())  # next power of 2
    # zero pad h to length m
    h_pad = np.zeros(m, dtype=float)
    h_pad[:n] = h
    # FFT
    H = np.fft.rfft(h_pad)
    conv = np.fft.irfft(H * H, n=m)
    conv = conv[:2*n-1]  # linear conv length
    pref = np.concatenate(([0.0], np.cumsum(h)))  # length n+1
    k_arr = np.arange(n)
    # O_pos: pref[n - k] - conv[k]
    O_pos = pref[n - k_arr] - conv[k_arr]
    # O_neg: pref[n] - pref[k] - conv[k]
    sum_h = pref[n]
    O_neg = sum_h - pref[k_arr] - conv[k_arr]
    O_all = np.maximum(O_pos, O_neg)
    idx = np.argmax(O_all)
    k = idx
    # Determine sign
    if O_pos[k] >= O_neg[k]:
        s = int(k)  # shift >=0
    else:
        s = -int(k)
    # gradient
    grad = np.zeros(n, dtype=float)

        grad[s:] += -h[:n-s]
    elif s < 0:
        ks = -s
        grad[ks:] += 1 - h[:n-ks]
        grad[:n-ks] += -h[ks:]
    else:
        grad = 1 - 2 * h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- FFT based overlap and gradient ----------

def compute_c5_and_gradient(h: np.ndarray):
    """
    Returns (c5, grad, shift) where
    c5  : overlap constant (integral form)
    grad: gradient of the worst shift overlap
    shift: signed integer shift that attains the worst overlap
    """
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two >= 2n-1
    hp = np.zeros(m, dtype=float)
    hp[:n] = h
    H = np.fft.rfft(hp)
    conv = np.fft.irfft(H * H, n=m)[: 2 * n - 1]   # linear convolution h*h
    pref = np.concatenate(([0.0], np.cumsum(h)))   # length n+1
    k = np.arange(n)
    # O_pos for shift +k, O_neg for shift -k
    O_pos = pref[n - k] - conv[k]
    sum_h = pref[n]
    O_neg = sum_h - pref[k] - conv[k]
    O_all = np.maximum(O_pos, O_neg)
    idx = int(np.argmax(O_all))
    # decide sign of the worst shift
    shift = idx if O_pos[idx] >= O_neg[idx] else -idx

    # gradient
    grad = np.zeros(n, dtype=float)
    if shift > 0:
        s = shift
        grad[: n - s] += 1.0 - h[s:]
        grad[s:] += -h[: n - s]
    elif shift < 0:
        s = -shift
        grad[s:] += 1.0 - h[: n - s]
        grad[: n - s] += -h[s:]
    else:  # shift == 0
        grad = 1.0 - 2.0 * h

    c5 = O_all[idx] * (2.0 / n)
    return c5, grad, shift


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 computation using FFT (no gradient)."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    hp = np.zeros(m, dtype=float)
    hp[:n] = h
    H = np.fft.rfft(hp)
    conv = np.fft.irfft(H * H, n=m)[: 2 * n - 1]
    pref = np.concatenate(([0.0], np.cumsum(h)))
    k = np.arange(n)
    O_pos = pref[n - k] - conv[k]
    sum_h = pref[n]
    O_neg = sum_h - pref[k] - conv[k]
    O_all = np.maximum(O_pos, O_neg)
    return np.max(O_all) * (2.0 / n)

def enforce_sum_exact(h: np.ndarray, rng: np.random.Generator):
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - np.sum(h)
    # simple random redistribution; loop limited to avoid infinite
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


# ---------- SLSQP with soft‑max smoothing ----------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """SLSQP step with soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick random redistribution)
    diff = CURRENT_TARGET - np.sum(x)
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return x

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Simple SA on binary vectors with exact half‑ones."""
    start = time.time()
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr

    while time.time() - start < time_budget:
        # temperature decays exponentially from 0.02 to 1e-4
        elapsed = time.time() - start
        T = 0.02 * (1e-4 / 0.02) ** (elapsed / time_budget)

        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        while h[i] == 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1.0:
            j = rng.integers(0, n)

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
        if accept:
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:  # early exit if very good
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c


# ---------- gradient‑guided hill climbing with SA acceptance ----------

def hill_climb(h: np.ndarray, rng: np.random.Generator,
               max_iters: int, step_scale: float):
    """Gradient guided SA hill‑climb on continuous values."""
    n = h.size
    best_h = h.copy()
    best_c, _, _ = compute_c5_and_gradient(best_h)
    cur_c = best_c

    for it in range(max_iters):
        # current gradient and worst shift
        cur_c, grad, _ = compute_c5_and_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        # propose transfer
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0.0:
            eps = rng.random() * max_eps * step_scale
            h[i] -= eps
            h[j] += eps
        else:
            eps = 0.0

        # new overlap
        new_c = compute_c5_fft(h)
        # SA acceptance
        T = max(1e-5, 0.01 * (0.5 ** (it / (max_iters / 10 + 1))))
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
        if not accept and eps > 0.0:
            # revert
            h[i] += eps
            h[j] -= eps
            new_c = cur_c  # stay at old value
        else:
            cur_c = new_c

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # decay step scale periodically
        if (it + 1) % max(1, max_iters // 10) == 0:
            step_scale = max(step_scale * 0.85, 1e-5)

    return best_h, best_c

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5_fft(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # ----- sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_exact(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
