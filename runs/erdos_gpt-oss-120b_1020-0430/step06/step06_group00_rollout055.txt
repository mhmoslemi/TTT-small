```python
def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # If diff small do nothing
    if abs(diff) < 1e-12:
        return h
    # We'll adjust random positions
    # For diff>0, we need to add; for diff<0 subtract.
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0: 
                continue
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
            diff += sub
    return h
```

```python
def binary_sa(n, rng, time_budget):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.1, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        # inner loop: limited number of steps per restart?
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                # accept
                c_curr = c_new
                # update arrays
                ones_arr[i_idx] = j
                zeros_arr[j_idx] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n)
    return best_h, best_c
```

```python
def upsample_h(h, new_n):
```

```python
_current_target = None
_current_beta = None

def _sum_eq_constraint(x):
    return np.sum(x) - _current_target

def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = _current_beta * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / _current_beta

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global _current_target, _current_beta
    n = h0.size
    _current_target = n / 2.0
    _current_beta = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = _current_target - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = _current_target - x.sum()
    return x
```

```python
def gradient_refine(h, rng, step=0.015, max_iter=30):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h
```

```python
def continuous_sa(h0, rng, time_budget, T0=0.2, T_end=1e-6):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # scale epsilon proportionally to temperature
        scale = 0.5 * (T / T0)  # factor
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0.0:
                continue
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            if sub <= 0.0:
                continue
            h[i] -= sub
            diff += sub
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# --------------------------------------------------------------
# Binary simulated annealing (0/1 vectors)
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Binary SA with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.1, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

# Soft‑max SLSQP optimisation
_current_target = None
_current_beta = None


def _sum_eq_constraint(x):
    return np.sum(x) - _current_target


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = _current_beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / _current_beta


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global _current_target, _current_beta
    n = h0.size
    _current_target = n / 2.0
    _current_beta = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = _current_target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = _current_target - x.sum()
    return x

# Gradient refinement of the worst shift
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 30) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

# Continuous simulated annealing (real‑valued)
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float,
                  T0: float = 0.5, T_end: float = 1e-6):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.5 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c
