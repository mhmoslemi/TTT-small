def adjust_sum_exact_fast(h):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # diff may be positive or negative
    # we can select indices using argsort of (1-h) or h
    if diff > 0:
        # need to increase sum by diff
        # find indices where h < 1
        idx = np.where(h < 1)[0]
        # sort by (1 - h) descending (largest capacity)
        cap = 1 - h[idx]
        order = idx[np.argsort(-cap)]
        i = 0
        while diff > 1e-12 and i < len(order):
            add = min(diff, 1 - h[order[i]])
            h[order[i]] += add
            diff -= add
            i += 1
    else:
        # need to decrease sum
        idx = np.where(h > 0)[0]
        # sort by h descending (largest values)
        order = idx[np.argsort(-h[idx])]
        i = 0
        while diff < -1e-12 and i < len(order):
            sub = min(-diff, h[order[i]])
            h[order[i]] -= sub
            diff += sub
            i += 1
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities -------------------
def compute_c5(h):
    """max overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact_fast(h):
    """Force sum(h) == n/2 with box constraints, no RNG."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        caps = 1.0 - h[idx]
        order = idx[np.argsort(-caps)]
        for i in order:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        idx = np.where(h > 0.0)[0]
        order = idx[np.argsort(-h[idx])]
        for i in order:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h

def project_h(h):
    """Clip to [0,1] and enforce the exact sum constraint."""
    np.clip(h, 0.0, 1.0, out=h)
    adjust_sum_exact_fast(h)
    return h

# ------------------- smooth‑max gradient -------------------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)log∑exp(β·overlap) wrt h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")        # overlaps for all integer shifts
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()                   # soft‑max weights
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)                     # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

# ------------------- worst‑shift refinement -------------------
def worst_shift_gradient(h):
    """Sub‑gradient using the currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

def worst_shift_refine(h, rng, base_step, iters, deadline):
    """Projected sub‑gradient steps focused on the current worst shift."""
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_local_h = h
        best_local_c = compute_c5(h)
        for factor in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * factor * g
            project_h(cand)
            c = compute_c5(cand)
            if c < best_local_c:
                best_local_c = c
                best_local_h = cand
        h = best_local_h
        step *= 0.95
    return h

# ------------------- SLSQP polishing -------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    project_h(x)
    return x

# ------------------- main optimisation -------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96   # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # base resolution
    base_n = 720
    parent = globals().get("initial_h_values", None)

    # schedule of smooth‑max parameters
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000,
             10000, 20000, 50000]
    def steps_for_beta(beta):
        return 80 if beta <= 2000 else 30

    restarts = 6
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent is not None:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(xt, xs, parent)
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
        project_h(h)

        # smooth‑max gradient descent
        for beta in betas:
            if time.time() > deadline:
                break
            iters = steps_for_beta(beta)
            step_base = 0.3 / (np.sqrt(beta) + 1e-12)
            for _ in range(iters):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h, beta)
                if np.linalg.norm(g) < 1e-14:
                    continue
                best_local_h = h
                best_local_c = compute_c5(h)
                for factor in (1.0, 0.5, 0.25, 2.0, 4.0):
                    cand = h - step_base * factor * g
                    project_h(cand)
                    c = compute_c5(cand)
                    if c < best_local_c:
                        best_local_c = c
                        best_local_h = cand
                h = best_local_h

        # focused refinement
        h = worst_shift_refine(h, rng, base_step=0.12, iters=25,
                               deadline=deadline)

        # final SLSQP polish (high β, modest iterations)
        for beta in (20000, 50000):
            if time.time() > deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=300)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

    # optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        project_h(h_up)
        for beta in betas[:6]:
            if time.time() > deadline:
                break
            iters = 20
            step_base = 0.25 / (np.sqrt(beta) + 1e-12)
            for _ in range(iters):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h_up, beta)
                best_local_h = h_up
                best_local_c = compute_c5(h_up)
                for factor in (1.0, 0.5, 0.25, 2.0):
                    cand = h_up - step_base * factor * g
                    project_h(cand)
                    c = compute_c5(cand)
                    if c < best_local_c:
                        best_local_c = c
                        best_local_h = cand
                h_up = best_local_h
        for beta in (20000, 50000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=200)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # fallback safety net
    if best_h is None:
        n = base_n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        project_h(best_h)
        best_c = compute_c5(best_h)
        best_n = n

    project_h(best_h)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
