def project_onto_sum_and_box(x, target):
    # Ensure box by clipping
    x = np.clip(x, 0.0, 1.0)
    # If sum is already target, done
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    # Need to adjust values not at bounds
    # Use water-filling: find lambda such that sum(clip(x - lambda, 0, 1)) = target
    # Solve lambda via bisection.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        sy = y.sum()
        if sy > target:
            lo = lam
        else:
            hi = lam
    return y
```

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sum_box(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
