def projected_subgradient_descent(h0, rng, time_budget, max_iter=100000):

    start = time.time()
    h = h0.copy()
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    a0 = 0.5 # initial step size maybe relative to dx? We'll tune.

    iter = 0
    while time.time() - start < time_budget and iter < max_iter:
        iter += 1
        # compute gradient of worst shift
        L = 2*N
        # compute correlation via FFT for speed
        h_fft = np.fft.rfft(np.concatenate([h, np.zeros(N)]))
        o_fft = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(N)]))
        corr = np.fft.irfft(h_fft * np.conj(o_fft), n=L)
        idx = np.argmax(corr)
        shift = idx - (N - 1)
        # compute gradient
        grad = np.zeros(N)
        if shift == 0:
            grad[:] = 1.0 - 2.0 * h
        elif shift > 0:
            k = shift
            grad[:N - k] = 1.0 - h[k:]
            grad[k:] -= h[:N - k]
        else:
            k = -shift
            # t < k: grad = 1.0
            grad[:k] = 1.0
            # t >= k: grad = 1 - h[t - k]
            grad[k:] = 1.0 - h[:N - k]
            # subtract h[t + k] for t in 0..N - k -1
            grad[:N - k] -= h[k:]
        # step size
        alpha = a0 / np.sqrt(iter)
        h = h - alpha * grad
        # project onto [0,1]
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        diff = target - h.sum()
        # Adjust diff using random selections
        if abs(diff) > 1e-12:
            # Determine number of attempts based on diff magnitude
            # We'll perform up to 1000 attempts
            attempts = 0
            while abs(diff) > 1e-12 and attempts < 2000:
                i = rng.integers(N)
                if diff > 0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                else:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
                attempts += 1
        # maybe evaluate best periodically
        if iter % 100 == 0:
            c = np.max(corr) * (2.0 / N)  # use current corr
            if c < best_c:
                best_c = c
                best_h = h.copy()
    # final compute
    final_c = compute_c5_fft(best_h)
    return best_h, final_c

def continuous_sa(h0, rng, time_budget):

    start = time.time()
    h = h0.copy()
    N = h.size
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    T0, T_end = 0.002, 1e-7

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i: j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0: continue
            eps = rng.random() * max_eps * scale
            if eps <= 0: continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0: continue
            eps = rng.random() * max_eps * scale
            if eps <= 0: continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    N = h.size
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

import numpy as np
import time

# ------------------------------------------------------------
def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT (linear correlation)."""
    N = h.size
    L = 2 * N
    h_fft = np.fft.rfft(h, n=L)
    o_fft = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(h_fft * np.conj(o_fft), n=L)
    return np.max(corr) * (2.0 / N)


def worst_shift_and_gradient(h: np.ndarray):
    """Return worst shift, its gradient and the max correlation value."""
    N = h.size
    L = 2 * N
    h_fft = np.fft.rfft(h, n=L)
    o_fft = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(h_fft * np.conj(o_fft), n=L)
    idx = np.argmax(corr)
    shift = idx - (N - 1)
    grad = np.zeros(N)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        k = shift
        grad[:N - k] = 1.0 - h[k:]
        grad[k:] -= h[:N - k]
    else:
        k = -shift
        grad[:k] = 1.0
        grad[k:] = 1.0 - h[:N - k]
        grad[:N - k] -= h[k:]
    return shift, grad, corr[idx]


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto [0,1] with exact sum = len(h)/2."""
    target = h.size / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / h.size
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = min(diff, np.sum(1.0 - h[free]))
            h[free] += inc / free.size
        else:
            over = np.where(h > 0.0)[0]
            if over.size == 0:
                break
            dec = min(-diff, np.sum(h[over]))
            h[over] -= dec / over.size
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
    attempts = 0
    N = h.size
    while abs(diff) > 1e-12 and attempts < 500:
        i = rng.integers(N)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated annealing on the continuous feasible set."""
    start = time.time()
    h = h0.copy()
    N = h.size
    c_curr = compute_c5_fft(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c


def projected_subgradient_descent(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    start = time.time()
    N = h0.size
    target = N / 2.0
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5_fft(h)
    it = 0
    while time.time() - start < time_budget:
        it += 1
        _, grad, _ = worst_shift_and_gradient(h)
        alpha = 0.5 / np.sqrt(it)
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        if it % 200 == 0:
            c = compute_c5_fft(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    final_c = compute_c5_fft(h)
    if final_c < best_c:
        best_h, best_c = h, final_c
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.015, max_iter: int = 15):
    """Polish using the explicit worst‑shift gradient."""
    N = h.size
    for it in range(max_iter):
        _, grad, _ = worst_shift_and_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    N = 2048

    # ---- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h = upsample_h(h0, N)
    else:
        h = np.zeros(N, dtype=float)
        h[rng.choice(N, N // 2, replace=False)] = 1.0
    h = enforce_sum(h, rng)
    best_h, best_c = h.copy(), compute_c5_fft(h)

    # ---- Phase A: continuous SA -----------------------------------------
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 30:
        t_sa = min(remaining * 0.35, 400)
        h_sa, c_sa = continuous_sa(best_h, rng, t_sa)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ---- Phase B: projected sub‑gradient descent -------------------------
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 10:
        h_pgd, c_pgd = projected_subgradient_descent(best_h, rng, remaining)
        if c_pgd < best_c:
            best_h, best_c = h_pgd, c_pgd

    # ---- Phase C: final gradient refinement -------------------------------
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 5:
        h_ref = gradient_refine(best_h, rng, step=0.015, max_iter=20)
        c_ref = compute_c5_fft(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ---- Sanitisation ----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
