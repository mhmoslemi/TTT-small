import time
import numpy as np
from scipy.optimize import minimize

# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_uniform(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1] and enforce sum(x)=target by uniform redistribution."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    max_iter = 500
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            inc = np.minimum(inc, 1.0 - x[idx])
            x[idx] += inc
        else:
            idx = np.where(x > 1e-12)[0]
            if idx.size == 0:
                break
            dec = diff / idx.size
            dec = np.maximum(dec, -x[idx])
            x[idx] += dec
        diff = target - np.sum(x)
    return x


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

# Soft‑max SLSQP optimisation (global helpers)
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum projection
    x = project_sum_uniform(x, CURRENT_TARGET)
    return x

# Gradient of the worst shift and fine‑tuning
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        g = worst_gradient(h)
        alpha = step * (0.8 ** it)
        h = h - alpha * g
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_uniform(h, target)
    return h

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Binary SA with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.1, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

        scores = np.array([compute_c5(ind) for ind in pop])
        cur_best_idx = np.argmin(scores)
        if scores[cur_best_idx] < best_c:
            best_c = scores[cur_best_idx]
            best_h = pop[cur_best_idx].copy()
        gen += 1
    return best_h, best_c

# Continuous simulated annealing
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.005, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.25 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None
    target = None

    # ---- initial solution (if any) ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        target = n0 / 2.0
        h0 = project_sum_uniform(h0, target)
        c0 = compute_c5(h0)
        best_h, best_c, best_n, target = h0, c0, n0, target

    # ---- Phase 0: binary SA on coarse grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 30.0:
        t_bin = min(remaining * 0.15, 200.0)
        h_bin, c_bin = binary_sa(256, rng, t_bin)
        if c_bin < best_c:
            best_h, best_c, best_n, target = h_bin, c_bin, 256, 128.0

    # ---- Phase 1: real‑valued GA on coarse grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 60.0:
        t_ga = min(remaining * 0.25, 300.0)
        ga_budget = t_ga
        ga_target = 128.0
        h_ga, c_ga = genetic_algorithm_real(
            N=256,
            pop_size=30,
            generations=60,
            rng=rng,
            target=ga_target,
            time_budget=ga_budget,
        )
        if c_ga < best_c:
            best_h, best_c, best_n, target = h_ga, c_ga, 256, ga_target

    # ---- Phase 2: upsample to medium resolution (512) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0 and best_h is not None and best_n < 512:
        h_mid = upsample_h(best_h, 512)
        target = 512 / 2.0
        h_mid = project_sum_uniform(h_mid, target)
        c_mid = compute_c5(h_mid)
        if c_mid < best_c:
            best_h, best_c, best_n = h_mid, c_mid, 512

    # ---- Phase 3: Soft‑max SLSQP refinement on 512 points ----
    remaining = total_budget - (time.time() - start)
    if remaining > 30.0 and best_h is not None and best_n == 512:
        h = best_h.copy()
        beta_sequence = (10., 40., 120., 360., 1080., 3240., 9720., 29160.)
        for beta in beta_sequence:
            if time.time() - start > total_budget * 0.95:
                break
            iters = 800 if beta >= 1080 else 500
            h = slsqp_optimize(h, beta, rng, max_iter=iters)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
        best_n = h.size
        best_h = h

    # ---- Phase 4: Gradient tidy‑up ----
    remaining = total_budget - (time.time() - start)
    if remaining > 15.0 and best_h is not None:
        h = gradient_refine(best_h, rng, step=0.02, max_iter=30)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- Phase 5: Upsample to fine resolution (1024) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None and best_n < 1024:
        h_fine = upsample_h(best_h, 1024)
        target = 1024 / 2.0
        h_fine = project_sum_uniform(h_fine, target)
        c_fine = compute_c5(h_fine)
        if c_fine < best_c:
            best_h, best_c, best_n = h_fine, c_fine, 1024

    # ---- Phase 6: Continuous SA on the finest grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h, c = continuous_sa(best_h, rng, remaining)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- Final sanitisation ----
    if best_h is None:
        # fallback to uniform half‑filled vector of size 512
        n_fallback = 512
        best_h = np.zeros(n_fallback, dtype=float)
        best_h[: n_fallback // 2] = 1.0
        best_c = compute_c5(best_h)
        best_n = n_fallback
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        target = best_h.size / 2.0
        best_h = project_sum_uniform(best_h, target)
        best_c = compute_c5(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c), int(best_n)
