    n = h.size

    if k >= 0:

        i = slice(0, n - k)

import time
import numpy as np

def _project_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {x: 0<=x<=1, sum(x)=target} using bisection on lambda."""
    low = np.min(y) - 1.0
    high = np.max(y) + 1.0
    for _ in range(40):
        mid = (low + high) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(y - lam, 0.0, 1.0)


def _corr_and_C(h: np.ndarray, dx: float):
    """Linear correlation of h with (1-h) and corresponding max C."""
    n = h.size
    L = 2 * n
    H = np.fft.fft(h, L)
    B = np.fft.fft(1.0 - h, L)
    corr = np.real(np.fft.ifft(H * np.conj(B)))[: 2 * n - 1] * dx
    return corr, np.max(corr)


def _worst_shift_gradient(h: np.ndarray, k: int):
    """Gradient of the overlap for the worst shift k."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        i = slice(0, n - k)
        j = slice(k, n)
        g[i] = 1.0 - h[j]
        g[j] = -h[i]
    else:
        l = -k
        i = slice(l, n)
        j = slice(0, n - l)
        g[i] = 1.0 - h[j]
        g[j] = -h[i]
    return g


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline_main = start + budget_s * 0.9  # reserve margin for final steps

    # ---------- initial construction ----------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 1024
        h = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, n0 // 2, replace=False)
        h[idx] = 1.0

    # Upsample to at least 1024 points (helps lower C)
    while h.size < 1024:
        h = np.repeat(h, 2)

    # Small noise to break symmetry after upsampling
    h += rng.normal(scale=1e-3, size=h.size)

    # Feasibility projection
    n = h.size
    target_sum = n / 2.0
    h = _project_box_sum(np.clip(h, 0.0, 1.0), target_sum)

    dx = 2.0 / n
    _, C = _corr_and_C(h, dx)
    best_h = h.copy()
    best_C = C

    # ---------- parameters ----------
    step_base = 0.08
    temp = 0.001
    temp_decay = 0.9995

    # ---------- main optimisation loop ----------
    while time.time() < deadline_main:
        corr, C = _corr_and_C(h, dx)
        if C < best_C:
            best_h = h.copy()
            best_C = C

        # worst shift and its gradient
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        grad = _worst_shift_gradient(h, k)

        # line search (only accept strict improvements)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
