def compute_delta_corr(h, i, j):
    N = h.size
    delta = np.zeros(N-1, dtype=np.int32)  # since sums of bits
    # i contributions (remove)
    # positive k: i+k < N
    k_pos = N - 1 - i
    if k_pos > 0:
        delta[:k_pos] -= h[i+1:i+1+k_pos]
    # negative k: i-k >=0 => k <= i
    if i > 0:
        # k from 1..i, delta index = i - k
        # Corresponds to reversed slice of h[:i]
        delta[i-1::-1] -= h[:i]
    # j contributions (add)
    k_pos_j = N - 1 - j
    if k_pos_j > 0:
        delta[:k_pos_j] += h[j+1:j+1+k_pos_j]
    if j > 0:
        delta[j-1::-1] += h[:j]
    return delta
```

import time
import numpy as np

# --------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# --------------------------------------------------------------
# Binary SA helpers (exact sum, {0,1} vectors)
def _adjust_binary_sum(h_bin: np.ndarray, rng: np.random.Generator) -> None:
    """Force exactly N/2 ones by flipping bits."""
    N = h_bin.size
    target = N // 2
    cur = int(h_bin.sum())
    if cur == target:
        return
    if cur > target:
        # turn some ones into zeros
        ones = np.flatnonzero(h_bin)
        to_flip = rng.choice(ones, cur - target, replace=False)
        h_bin[to_flip] = 0
    else:
        zeros = np.flatnonzero(h_bin == 0)
        to_flip = rng.choice(zeros, target - cur, replace=False)
        h_bin[to_flip] = 1


def _delta_corr_binary(h: np.ndarray, i: int, j: int, delta: np.ndarray) -> None:
    """Compute delta for correlation of binary vector when swapping i (1) with j (0)."""
    N = h.size
    delta.fill(0)
    # remove contributions of i (was 1)
    kp = N - 1 - i
    if kp > 0:
        delta[:kp] -= h[i + 1 : i + 1 + kp]
    if i > 0:
        delta[i - 1 :: -1] -= h[:i]
    # add contributions of j (becomes 1)
    kp_j = N - 1 - j
    if kp_j > 0:
        delta[:kp_j] += h[j + 1 : j + 1 + kp_j]
    if j > 0:
        delta[j - 1 :: -1] += h[:j]


def binary_sa_fixed(N: int, rng: np.random.Generator, time_budget: float,
                    init_h: np.ndarray | None = None) -> tuple[np.ndarray, int]:
    """Binary simulated annealing on length N, returns best float vector and best Omax."""
    start = time.time()
    # initialise binary vector
    if init_h is not None:
        h_bin = (init_h >= 0.5).astype(np.int8)
        _adjust_binary_sum(h_bin, rng)
    else:
        perm = rng.permutation(N)
        h_bin = np.zeros(N, dtype=np.int8)
        h_bin[perm[: N // 2]] = 1

    # correlation for positive shifts k=1..N-1
    corr_full = np.correlate(h_bin, h_bin, mode="full")
    corr_pos = corr_full[N:]  # length N-1
    Omax = N // 2 - int(corr_pos.min())

    # bookkeeping for fast swaps
    inds = np.arange(N)
    ones_list = list(inds[h_bin == 1])
    zeros_list = list(inds[h_bin == 0])
    pos_one = np.full(N, -1, dtype=np.int32)
    pos_zero = np.full(N, -1, dtype=np.int32)
    for idx, v in enumerate(ones_list):
        pos_one[v] = idx
    for idx, v in enumerate(zeros_list):
        pos_zero[v] = idx

    best_Omax = Omax
    best_h = h_bin.astype(float).copy()

    # cooling schedule
    T0 = max(1.0, Omax * 0.5)
    T_end = 1e-4
    delta = np.empty(N - 1, dtype=np.int32)

    target_Omax = int(np.floor(0.3808 * N / 2.0 + 1e-9))

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        # pick swap candidates
        idx_i = rng.integers(len(ones_list))
        idx_j = rng.integers(len(zeros_list))
        i = ones_list[idx_i]
        j = zeros_list[idx_j]

        _delta_corr_binary(h_bin, i, j, delta)
        cand_min = np.min(corr_pos + delta)
        cand_Omax = N // 2 - int(cand_min)

        accept = False
        if cand_Omax <= Omax:
            accept = True
        else:
            prob = np.exp(-(cand_Omax - Omax) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # apply delta
            corr_pos += delta
            Omax = cand_Omax
            # swap in the binary vector
            h_bin[i] = 0
            h_bin[j] = 1
            # update lists and position maps
            ones_list[idx_i] = j
            zeros_list[idx_j] = i
            pos_one[j] = idx_i
            pos_zero[i] = idx_j
            pos_one[i] = -1
            pos_zero[j] = -1

            if Omax < best_Omax:
                best_Omax = Omax
                best_h = h_bin.astype(float).copy()
                if best_Omax <= target_Omax:
                    break

    return best_h, best_Omax


# --------------------------------------------------------------
# Continuous SA (preserves sum)
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float) -> tuple[np.ndarray, float]:
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.005, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.3 * (T / T0)  # shrinking step scale
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # decide direction
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c


# --------------------------------------------------------------
# Gradient tidy‑up (same as parent)
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- initial solution from parent (if present) ----

    # ---------------------------------------------------------
    # Phase A: binary SA on a fine grid (N = 1024)
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0:  # need some time
        N = 1024
        # upsample parent if available for a warm start
        init_bin = None
        if best_h is not None and best_n == 512:

        t_bin = min(remaining * 0.60, 600.0)
        h_bin_best, Omax_best = binary_sa_fixed(N, rng, t_bin, init_bin)
        c_bin_best = (2.0 / N) * Omax_best
        if c_bin_best < best_c:
            best_h, best_c, best_n = h_bin_best, c_bin_best, N

    # ---------------------------------------------------------
    # Phase B: continuous SA on the best vector so far
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        t_cont = remaining * 0.70
        h_cont, c_cont = continuous_sa(best_h, rng, t_cont)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # ---------------------------------------------------------
    # Phase C: gradient refinement
    remaining = total_budget - (time.time() - start)
    if remaining > 2.0 and best_h is not None:

        c_grad = compute_c5(h_grad)
        if c_grad < best_c:
            best_h, best_c = h_grad, c_grad

    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
