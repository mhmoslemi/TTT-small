def compute_c5(h: np.ndarray) -> float:
    """C5 using direct correlation (exact)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_fast(h: np.ndarray) -> float:
    """C5 via FFT (linear convolution)."""
    N = h.size
    rev = (1 - h)[::-1]
    conv = np.fft.ifft(np.fft.fft(h, n=2 * N) * np.fft.fft(rev, n=2 * N)).real
    return conv.max() * (2.0 / N)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by a binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst shift (max O_k)."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

    return best_h, best_c


def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, start_time: float,
                         total_budget: float, rng: np.random.Generator):
    deadline = start_time + time_budget
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int,
                   rng: np.random.Generator) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    N = h0.size
    CURRENT_TARGET = N / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * N
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, N)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    N = h.size
    N2 = N * 2
    x_old = np.linspace(0.0, 2.0, N, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, N2 / 2.0)


def obj_y(y):
    """DE objective – uses globals DE_BASE_H, DE_B, DE_TARGET_SUM."""
    cand = DE_BASE_H + DE_B @ y
    h = project_box_sum(cand, DE_TARGET_SUM)
    return compute_c5_fast(h)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initial feasible h -------------------------------------------------
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        N = base_h.size
        base_h = np.clip(base_h, 0.0, 1.0)
        base_h = project_box_sum(base_h, N / 2.0)
    else:
        N = 512
        base_h = np.full(N, 0.5)

    best_h = base_h.copy()
    best_c = compute_c5(best_h)

    # --- Differential Evolution (global search) ----------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 20.0:
        d = 20
        global DE_BASE_H, DE_B, DE_TARGET_SUM
        DE_BASE_H = base_h
        DE_B = rng.standard_normal((N, d))
        DE_TARGET_SUM = N / 2.0
        bounds = [(-2.0, 2.0)] * d
        try:
            result = differential_evolution(obj_y, bounds, seed=seed,
                                            maxiter=200, popsize=5,
                                            polish=False, disp=False,
                                            recombination=0.7,
                                            mutation=(0.5, 1))
            if result.fun < best_c:
                best_c = result.fun
                best_h = project_box_sum(base_h + DE_B @ result.x, DE_TARGET_SUM)
        except Exception:
            pass

    # --- Subgradient descent ------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        sub_h, sub_c = subgradient_optimize(best_h, best_h.size / 2.0,
                                            remaining * 0.7, start, budget_s, rng)
        if sub_c < best_c:
            best_h, best_c = sub_h, sub_c

    # --- Guided hill‑climb -------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        max_iters = int(min(5000, remaining * 300))
        hc_h, hc_c = guided_hill_climb(best_h, rng, max_iters, 0.7)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

    # --- Soft‑max SLSQP passes ---------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        for beta in (20., 80., 200., 800., 3000., 8000.):
            if time.time() - start > budget_s * 0.99:
                break
            max_iter = min(500, int(remaining * 0.3))
            h_opt = slsqp_optimize(best_h, beta, max_iter, rng)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # --- Optional up‑sampling and fine polishing ---------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 8.0 and best_h.size * 2 <= 1024:
        up_h = upsample(best_h, rng)
        sub_budget = remaining * 0.4
        fine_h, fine_c = subgradient_optimize(up_h, up_h.size / 2.0,
                                              sub_budget, start, budget_s, rng)
        if fine_c < best_c:
            best_h, best_c = fine_h, fine_c
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(min(4000, remaining * 250))
            hc_fine_h, hc_fine_c = guided_hill_climb(best_h, rng,
                                                     max_iters, 0.6)
            if hc_fine_c < best_c:
                best_h, best_c = hc_fine_h, hc_fine_c

    # --- final sanitisation -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
