def simulated_annealing(h, S_full, rng, total_iters, T_start, T_end):
    best_h = h.copy()
    best_C = compute_c5_from_S(S_full)
    cur_h = h
    cur_S = S_full
    cur_C = best_C
    for t in range(total_iters):
        T = T_start * ((T_end/T_start) ** (t / total_iters))  # exponential decay
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        if eps_max <= 0:
            continue
        eps = eps_max * rng.random() * step_scale  # step_scale maybe 0.5
        if eps == 0:
            continue
        # compute ΔS array
        contrib_i = compute_contrib(i, cur_h, offsets)
        contrib_j = compute_contrib(j, cur_h, offsets)
        delta_S = eps * (contrib_j - contrib_i)
        new_S = cur_S + delta_S
        new_min_S = new_S.min()
        new_Omax = (N/2) - new_min_S
        new_C = new_Omax * (2.0 / N)
        dC = new_C - cur_C
        if dC < 0 or rng.random() < np.exp(-dC / T):
            # accept
            cur_h[i] -= eps
            cur_h[j] += eps
            cur_S = new_S
            cur_C = new_C
            if new_C < best_C:
                best_C = new_C
                best_h = cur_h.copy()
        # else reject (do nothing)
        # optionally periodic recompute cur_S from scratch to avoid numeric error accumulation.
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} using binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n, k≥0
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, float(best_c)


def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap (via worst shift)."""
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, float(best_c)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def upsample(h: np.ndarray, rng: np.random.Generator):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # --------------------------------------------------------------
    # initialise candidates
    N = 512
    candidates = []

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != N:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, N / 2.0)
        candidates.append(h0)

    # a few deterministic seeds
    x = np.linspace(0.0, 2.0, N, endpoint=False)

    for _ in range(2):

    best_h = None
    best_c = np.inf
    for h in candidates:

        if c < best_c:
            best_c, best_h = c, h.copy()

    # --------------------------------------------------------------
    # Phase 1: sub‑gradient descent (≈65% of time)
    remaining = total_budget - (time.time() - start)
    sub_budget = remaining * 0.65
    if sub_budget > 5.0:

        if c_sg < best_c:

    remaining = total_budget - (time.time() - start)

            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    # Phase 4: optional up‑sampling and final polish
    remaining = total_budget - (time.time() - start)
    if remaining > 8.0 and N * 2 <= 1024:
        h_up = upsample(best_h, rng)
        sub_budget = remaining * 0.4
        if sub_budget > 5.0:
            h_up_sg, c_up_sg = subgradient_optimize(h_up, h_up.size / 2.0,
                                                    sub_budget, rng)
            if c_up_sg < best_c:
                best_c, best_h = c_up_sg, h_up_sg
        remaining = total_budget - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(min(6000, remaining * 300))
            h_up_hc, c_up_hc = guided_hill_climb(best_h, rng,

    return best_h.tolist(), float(final_c), int(best_h.size)
