def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h
```

```
def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h
```

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
```

Define slsqp_refine:

```
def slsqp_refine(h, target, rng, beta, max_iter):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                              'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024
    target = n // 2

    best_h = None
    best_c = np.inf

    # initial from parent if available
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        init = np.clip(init, 0.0, 1.0)
        # adjust sum to target
        init = enforce_sum(init, target, rng)
        c_init = compute_c5(init)
        best_h = init.copy()
        best_c = c_init

        # also consider binary version
        h_bin = (init > 0.5).astype(float)
        # fix sum
        sum_bin = int(h_bin.sum())
        while sum_bin > target:
            idx = rng.choice(np.flatnonzero(h_bin == 1))
            h_bin[idx] = 0.0
            sum_bin -= 1
        while sum_bin < target:
            idx = rng.choice(np.flatnonzero(h_bin == 0))
            h_bin[idx] = 1.0
            sum_bin += 1
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    else:
        # no parent, start with random binary
        perm = rng.permutation(n)
        h_bin = np.zeros(n, dtype=float)
        h_bin[perm[:target]] = 1.0
        best_h = h_bin.copy()
        best_c = compute_c5(best_h)

    # Main random-restart hill climbing loop
    # We'll limit each local search to a modest number of swaps
    max_local_swaps = 2500
    while time.time() - start < budget_s * 0.94:
        # generate new random binary candidate
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=float)
        h[perm[:target]] = 1.0
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # local greedy improvement
        for it in range(max_local_swaps):
            if time.time() - start >= budget_s * 0.94:
                break
            # indices of 1s and 0s
            idx1 = np.flatnonzero(h)
            idx0 = np.flatnonzero(1 - h)
            i1 = rng.choice(idx1)
            i0 = rng.choice(idx0)
            # perform swap
            h[i1] = 0.0
            h[i0] = 1.0
            c_new = compute_c5(h)
            if c_new < c - 1e-12:
                c = c_new
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                # revert
                h[i1] = 1.0
                h[i0] = 0.0
        # optional early exit if very good
        if best_c < 0.3807:
            break

    # final refinement using SLSQP if time permits
    if time.time() - start < budget_s * 0.92:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() - start >= budget_s * 0.92:
                break
            refined = slsqp_refine(best_h, target, rng, beta=beta, max_iter=300)
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_c = c_ref
                best_h = refined.copy()

    # final enforcement of constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

# ----- global vars used by the SLSQP soft‑max -----
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in‑place so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def compute_c5_and_shift(h: np.ndarray):
    """Return (C5, shift) for the current h. Shift is signed."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    c = corr[idx] * (2.0 / h.size)
    shift = idx - (h.size - 1)
    return c, shift


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h, target, rng, beta, max_iter):
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x
