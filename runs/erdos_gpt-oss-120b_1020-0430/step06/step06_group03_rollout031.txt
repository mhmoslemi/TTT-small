import time, numpy as np

def project_onto_box_sum(v, target):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(v - lam))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def compute_corr_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.rfft(h, n=L)
    fg = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(fh * fg, n=L)
    return conv

def compute_c5(h):
    n = h.shape[0]
    conv = compute_corr_fft(h)
    return conv.max() * (2.0 / n)

import time, numpy as np

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(v - lam))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def compute_corr_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.rfft(h, n=L)
    fg = np.fft.rfft(1.0 - h, n=L)
    return np.fft.irfft(fh * fg, n=L)

def compute_c5(h):
    n = h.shape[0]
    return compute_corr_fft(h).max() * (2.0 / n)
