def local_optimize(h_start, time_budget):
   h = h_start.copy()
   c = compute_c5(h)
   best_h_local = h.copy()
   best_c_local = c
   step0 = 0.15
   decay = 0.995
   it = 0
   while True:
       elapsed = time.time() - start
       if elapsed >= time_budget:
           break
       step = step0 * (decay ** it)
       if step < 1e-9:
           break
       # compute worst gradient
       grad = worst_gradient(h)
       # gradient descent candidate
       nh = h - step * grad
       nh = np.clip(nh, 0.0, 1.0)
       nh = enforce_sum(nh, rng)
       nc = compute_c5(nh)
       # accept if better
       if nc < c:
           h, c = nh, nc
           if nc < best_c_local:
               best_c_local, best_h_local = nc, nh.copy()
       else:
           # maybe try backtracking (reduce step) a few times
           # simple: reduce step by half up to 3 times
           temp_step = step
           improved = False
           for _ in range(3):
               temp_step *= 0.5
               nh2 = h - temp_step * grad
               nh2 = np.clip(nh2, 0.0, 1.0)
               nh2 = enforce_sum(nh2, rng)
               nc2 = compute_c5(nh2)
               if nc2 < c:
                   h, c = nh2, nc2
                   if nc2 < best_c_local:
                       best_c_local, best_h_local = nc2, nh2.copy()
                   improved = True
                   break
           # if no improvement, maybe random perturb
           if not improved and it % 200 == 0:
               # random transfer from high gradient to low
               i = np.argmax(grad)
               j = np.argmin(grad)
               max_eps = min(h[i], 1.0 - h[j])

       it += 1
   return best_h_local, best_c_local

def slsqp_optimize(h0, beta, rng, max_iter=1500):
   global CURRENT_TARGET, CURRENT_BETA
   n = h0.size
   CURRENT_TARGET = n/2.0
   CURRENT_BETA = beta
   bounds = [(0.0,1.0)]*n
   cons = {"type": "eq", "fun": sum_eq_constraint}
   res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
   x = np.clip(res.x, 0.0, 1.0)
   diff = CURRENT_TARGET - x.sum()
   if abs(diff) > 1e-12:
       # adjust via enforce_sum
       x = enforce_sum(x, rng)
   return x

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    max_iter = 5000
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    max_iter = 5000
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        km = -k
        grad[: n - km] -= h[km:]
        grad[km:] -= h[: n - km]
    return grad


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    scaled = corr * (2.0 / x.size)
    bc = CURRENT_BETA * scaled
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    """Run SLSQP on the soft‑max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, rng)
    return x


def local_optimize(h_start: np.ndarray, max_iters: int,
                   rng: np.random.Generator, start_time: float,
                   budget_s: float) -> tuple:
    """Gradient descent on the worst shift with simple back‑tracking."""
    h = h_start.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step0 = 0.1
    decay = 0.995
    it = 0
    while it < max_iters and (time.time() - start_time) < budget_s:
        step = step0 * (decay ** it)
        if step < 1e-12:
            break
        grad = worst_gradient(h)
        nh = h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, rng)
        nc = compute_c5(nh)
        if nc < c:
            h, c = nh, nc
            if nc < best_c:
                best_c, best_h = nc, nh.copy()
        else:
            # back‑track a few times
            improved = False
            tmp_step = step
            for _ in range(3):
                tmp_step *= 0.5
                nh2 = h - tmp_step * grad
                nh2 = np.clip(nh2, 0.0, 1.0)
                nh2 = enforce_sum(nh2, rng)
                nc2 = compute_c5(nh2)
                if nc2 < c:
                    h, c = nh2, nc2
                    if nc2 < best_c:
                        best_c, best_h = nc2, nh2.copy()
                    improved = True
                    break
            if not improved and it % 200 == 0:
                i = int(np.argmax(grad))
                j = int(np.argmin(grad))
                if i != j:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps > 0:
                        eps = rng.random() * max_eps * 0.1
                        nh3 = h.copy()
                        nh3[i] -= eps
                        nh3[j] += eps
                        nh3 = np.clip(nh3, 0.0, 1.0)
                        nh3 = enforce_sum(nh3, rng)
                        nc3 = compute_c5(nh3)
                        if nc3 < c:
                            h, c = nh3, nc3
                            if nc3 < best_c:
                                best_c, best_h = nc3, nh3.copy()
        it += 1
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- parameters -----
    n_target = 1024  # resolution
    # ----- build initial candidates -----
    candidates = []

    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        init_h = np.clip(init_h, 0.0, 1.0)
        # linear interpolation to finer grid
        x_old = np.linspace(0.0, 2.0, init_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        up_h = np.interp(x_new, x_old, init_h)
        up_h = np.clip(up_h, 0.0, 1.0)
        up_h = enforce_sum(up_h, rng)
        candidates.append(up_h)

    # Thue–Morse (continuous version)
    tm = thue_morse_bits(n_target).astype(float)
    candidates.append(tm)

    # Random balanced binary
    rand_bin = np.zeros(n_target)
    rand_bin[rng.choice(n_target, n_target // 2, replace=False)] = 1.0
    candidates.append(rand_bin)

    # Random continuous with exact sum
    rand_cont = rng.random(n_target)
    rand_cont = rand_cont / rand_cont.sum() * (n_target / 2.0)
    rand_cont = np.clip(rand_cont, 0.0, 1.0)
    rand_cont = enforce_sum(rand_cont, rng)
    candidates.append(rand_cont)

    # ----- evaluate initial candidates -----
    best_h = None
    best_c = np.inf
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- local gradient descent on each candidate -----
    remaining = budget_s - (time.time() - start_time)
    # allocate ~70% of remaining time for these passes
    local_budget = remaining * 0.7
    per_candidate_budget = local_budget / len(candidates)
    max_iters = 5000
    for h0 in candidates:
        h_opt, c_opt = local_optimize(
            h0, max_iters, rng, start_time, per_candidate_budget
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
            if best_c <= 0.3808:
                break

    # ----- restarts with jitter -----
    while (time.time() - start_time) < budget_s * 0.85:
        # create a jittered copy of the best solution so far
        jitter = rng.normal(scale=0.08, size=n_target)
        h0 = best_h + jitter
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        # a short run of gradient descent
        h_opt, c_opt = local_optimize(
            h0, 2000, rng, start_time, budget_s * 0.05
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
            if best_c <= 0.3808:
                break

    # ----- SLSQP refinement (soft‑max) -----
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
        if (time.time() - start_time) > budget_s * 0.95:
            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=1500)
        c = compute_c5(h_opt)
        if c < best_c:
            best_c, best_h = c, h_opt.copy()
            if best_c <= 0.3808:
                break

    # ----- final random tweaking -----
    while (time.time() - start_time) < budget_s * 0.99:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.02
                nh = best_h.copy()
                nh[i] -= eps
                nh[j] += eps
                nh = np.clip(nh, 0.0, 1.0)
                nh = enforce_sum(nh, rng)
                c = compute_c5(nh)
                if c < best_c:
                    best_c, best_h = c, nh.copy()

    # ----- final sanity enforcement -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
