def enforce_sum(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    max_iter = 10000
    i = 0
    # Greedy approach: while diff not zero, adjust random idx.
    while abs(diff) > 1e-12 and i < max_iter:
        idx = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        i += 1
    # If still diff, distribute uniformly (should be negligible)
    if abs(diff) > 1e-12:
        # Add diff/n to all, then clip, then adjust again.
        h += diff / n
        # Clip and adjust iteratively until sum correct (but diff small).
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # Last resort: adjust first element
            h[0] += diff
    return h

def worst_gradient(h):
    n = h.size
    # compute correlation via FFT
    rev = (1.0 - h)[::-1]
    conv = np.fft.ifft(np.fft.fft(h, 2*n) * np.fft.fft(rev, 2*n)).real[:2*n-1]
    idx = np.argmax(conv)
    lag = idx - (n-1)
    d = lag
    grad = np.zeros_like(h)
    if d >= 0:
        grad[d:] += 1 - h[:n-d]
        grad[:n-d] -= h[d:]
    else:
        d_abs = -d
        grad[:n-d_abs] += 1 - h[d_abs:]
        grad[d_abs:] -= h[:n-d_abs]
    return grad

def softmax_grad_descent(h, rng, beta, step, iters):
    n = h.size
    for i in range(iters):
        # Compute conv for c
        rev = (1.0 - h)[::-1]
        c = np.fft.ifft(np.fft.fft(h, 2*n) * np.fft.fft(rev, 2*n)).real[:2*n-1]
        maxc = c.max()
        w = np.exp(beta * (c - maxc))
        Z = w.sum()
        w /= Z
        # w_shifted for convolution
        w_shifted = np.zeros(2*n)
        w_shifted[:2*n-1] = w
        B = 1.0 - h
        # conv1 = w * B
        conv1 = np.fft.ifft(np.fft.fft(w_shifted, 2*n) * np.fft.fft(B, 2*n)).real
        conv2 = np.fft.ifft(np.fft.fft(w_shifted, 2*n) * np.fft.fft(h, 2*n)).real
        grad = conv1[n-1:n-1+n] - conv2[n-1:n-1+n]
        # gradient step
        h = h - step * grad
        # Project
        h = np.clip(h, 0.0, 1.0)
        # Enforce sum
        h = enforce_sum(h, rng)
    return h

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # scale epsilon proportionally to temperature
        scale = T / T0
        # pick two distinct indices
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # decide direction
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT convolution (full correlation)."""
    n = h.size
    rev = (1.0 - h)[::-1]
    conv = np.fft.ifft(np.fft.fft(h, 2 * n) * np.fft.fft(rev, 2 * n)).real[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        attempts += 1
    if abs(diff) > 1e-12:  # fallback uniform tweak
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h[0] += diff
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(h) to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining maximal overlap."""
    n = h.size
    rev = (1.0 - h)[::-1]
    conv = np.fft.ifft(np.fft.fft(h, 2 * n) * np.fft.fft(rev, 2 * n)).real[: 2 * n - 1]

        grad[: n - d] -= h[d:]
    else:
        d = -d
        grad[: n - d] += 1.0 - h[d:]

    return grad

                         beta: float, step: float, iters: int) -> np.ndarray:

    n = h.size

        c = np.fft.ifft(np.fft.fft(h, 2 * n) * np.fft.fft(rev, 2 * n)).real[: 2 * n - 1]
        m = c.max()

        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c


# -------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    TARGET = 0.3808

    # Choose resolution (larger N gives finer discretisation)
    N = 4096

    # Initial h
    if 'initial_h_values' in globals():
        h0 = upsample_h(np.array(initial_h_values, dtype=float), N)
    else:
        h0 = rng.random(N)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5_fft(best_h)

    # ---------------------------------------------------------
    # Phase 1 – soft‑max gradient descent across increasing beta
    beta_list = [5, 20, 80, 300, 1200, 4800, 20000, 80000]
    step = 0.2
    for beta in beta_list:
        if time.time() - start > budget_s * 0.90:
            break
        best_h = softmax_grad_descent(best_h, rng, beta, step, iters=2000)
        cur_c = compute_c5_fft(best_h)
        if cur_c < best_c:
            best_c, best_h = cur_c, best_h.copy()
        if best_c <= TARGET:
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = enforce_sum(best_h, rng)
            return best_h.tolist(), float(best_c), int(best_h.size)
        step *= 0.5

    # ---------------------------------------------------------
    # Phase 2 – exact worst‑shift gradient tidy‑up
    if time.time() - start < budget_s * 0.95:
        best_h = gradient_refine(best_h, rng, step=0.015, max_iter=30)
        cur_c = compute_c5_fft(best_h)
        if cur_c < best_c:
            best_c, best_h = cur_c, best_h.copy()
        if best_c <= TARGET:
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = enforce_sum(best_h, rng)
            return best_h.tolist(), float(best_c), int(best_h.size)

    # ---------------------------------------------------------
    # Phase 3 – continuous simulated annealing on remaining time
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        h_sa, c_sa = continuous_sa(best_h, rng, remaining)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        if best_c <= TARGET:
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = enforce_sum(best_h, rng)
            return best_h.tolist(), float(best_c), int(best_h.size)

    # ---------------------------------------------------------
    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
