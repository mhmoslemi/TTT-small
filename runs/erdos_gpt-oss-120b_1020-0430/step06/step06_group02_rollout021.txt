import time
import numpy as np
from scipy.optimize import minimize

# ------------------ projection onto [0,1] & sum constraint ------------------
def _project_sum_bounds(x: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------ correlation & C5 ---------------------------------------
def _corr_vec(v: np.ndarray) -> np.ndarray:
    n = v.size
    L = 2 * n
    H = np.fft.rfft(v, L)
    R = np.fft.rfft(np.flip(1.0 - v), L)
    conv = np.fft.irfft(H * R, L)
    return conv[: 2 * n - 1]


def _c5(v: np.ndarray) -> float:
    return _corr_vec(v).max() * (2.0 / v.size)


# ------------------ worst‑shift gradient ----------------------------------
def _worst_grad(v: np.ndarray) -> np.ndarray:
    n = v.size
    corr = _corr_vec(v)
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * v
    grad = np.ones_like(v)
    grad[: n - k] -= v[k:]
    grad[k:] -= v[: n - k]
    return grad


# ------------------ SLSQP with soft‑max ---------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x: np.ndarray) -> float:
    corr = _corr_vec(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x


# ------------------ main optimisation routine -----------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- problem size -------------------------------------------------
    n = 1024
    target = n / 2.0

    # ---- initial h ----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size < n:
            rep = int(np.ceil(n / h0.size))
            h0 = np.repeat(h0, rep)[:n]
        else:
            h0 = h0[:n]
    else:
        h0 = rng.random(n)

    h = np.clip(h0, 0.0, 1.0)
    h = _project_sum_bounds(h, target)

    best_h = h.copy()
    best_c = _c5(best_h)

    # ---- Adam on worst‑shift gradient ---------------------------------
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    init_lr = 0.05
    decay = 1e-5
    it = 0
    while True:
        if time.time() - start > budget_s * 0.55:
            break
        corr = _corr_vec(h)
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        lr = init_lr / (1.0 + decay * it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = _project_sum_bounds(h, target)

        c = corr.max() * (2.0 / n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        it += 1

    # ---- SLSQP refinement ---------------------------------------------
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
        if time.time() - start > budget_s * 0.90:
            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=1200)
        c = _c5(h_opt)
        if c < best_c:
            best_c, best_h = c, h_opt.copy()
            if best_c <= 0.3808:
                break

    # ---- fine‑grained worst‑gradient descent ---------------------------
    for step in (0.05, 0.02, 0.01, 0.005):
        if time.time() - start > budget_s * 0.98:
            break
        for _ in range(300):
            grad = _worst_grad(best_h)
            nh = best_h - step * grad
            nh = np.clip(nh, 0.0, 1.0)
            nh = _project_sum_bounds(nh, target)
            c = _c5(nh)
            if c < best_c:
                best_c, best_h = c, nh.copy()
                if best_c <= 0.3808:
                    break

    # ---- random tweaks until time runs out ----------------------------
    while time.time() - start < budget_s * 0.999:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = np.clip(nh, 0.0, 1.0)
        nh = _project_sum_bounds(nh, target)
        c = _c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()

    # ---- final sanity check -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_bounds(best_h, target)
    final_c = _c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
