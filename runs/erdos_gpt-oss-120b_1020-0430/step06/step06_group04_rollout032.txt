import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """max overlap integral for discrete h."""
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def fix_sum(x, target, rng, max_iter=2000):
    """Adjust x in-place so that sum(x) == target."""
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, x.size)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP on smooth‑max surrogate."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        v = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * v
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return fix_sum(x, target, rng)
