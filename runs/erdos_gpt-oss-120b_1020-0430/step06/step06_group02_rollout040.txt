    n = h.shape[0]
    corr = np.empty(n)
    grad = np.zeros_like(h)
    # compute corr_k
    # For k from 0..n-1
    for k in range(n):

        corr_k = np.dot(h[:n-k], 1.0 - h[k:])
        corr[k] = corr_k

    # compute gradient as sum_k w_k * grad_corr_k

    for k in range(n):

    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * (2.0 / h.size)

    def project_onto_bound_simplex(x):
        lo = -1.0
        hi = 1.0
        for _ in range(40):
            lam = (lo + hi) * 0.5
            y = np.clip(x + lam, 0.0, 1.0)
            s = y.sum()
            if s < target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(x + (lo+hi)*0.5, 0.0, 1.0)

    def softmax_obj_and_grad(h, beta):
        n = h.size
        corr = np.empty(n)
        grad = np.zeros_like(h)
        # compute corr_k via dot products
        # use vectorized: for k in range(n): corr[k] = np.dot(h[:n-k], 1 - h[k:])
        for k in range(n):
            corr[k] = np.dot(h[:n - k], 1.0 - h[k:])
        # softmax weights
        bc = beta * corr
        maxb = bc.max()
        exp_shift = np.exp(bc - maxb)
        w = exp_shift / exp_shift.sum()
        # gradient accumulation
        for k in range(n):
            wk = w[k]
            if wk == 0.0:
                continue
            if k == 0:
                # grad of corr_0 = 1 - 2h
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[:n - k] += wk * (1.0 - h[k:])
                grad[k:] -= wk * h[:n - k]
        obj = (maxb + np.log(exp_shift.sum())) / beta
        return obj, grad

    # initial candidates list
    best_h = None
    best_c = np.inf

    # generate some initial seeds
    init_seeds = []
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        init = np.clip(init, 0.0, 1.0)
        # enforce sum via projection
        init = project_onto_bound_simplex(init)
        init_seeds.append(init)
    # random continuous
    for _ in range(3):
        x = rng.random(n)
        # scale to achieve target sum using projection
        x = project_onto_bound_simplex(x)
        init_seeds.append(x)

    # Optimization phases
    betas = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0]
    step0 = 0.2
    for init_h in init_seeds:
        h = init_h.copy()

            max_iters = int(4000 * (0.5**np.log10(beta+1)))  # fewer iters for high beta
            if time.time() - start > budget_s * 0.90:

                step = step0 * (0.99 ** it)
                if step < 1e-8:

                h = project_onto_bound_simplex(h - step * grad)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            if best_c <= 0.3808:
                break

    # final refinement with SLSQP similar to previous code (optional)
    # we can reuse slsqp_optimize but need to define it again.

    import scipy.optimize as opt
    def slsqp(h0, beta):
        n = h0.size
        target = n / 2.0
        def obj(x):
            # softmax objective
            # compute corr via dot
            corr = np.empty(n)
            for k in range(n):
                corr[k] = np.dot(x[:n - k], 1.0 - x[k:])
            bc = beta * corr
            maxb = bc.max()

        res = opt.minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": 500, "ftol":1e-9, "disp":False})
        x = np.clip(res.x,0,1)
        # repair sum if needed
        diff = target - x.sum()
        if abs(diff) > 1e-12:

                i = rng.integers(0,n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
                if abs(diff) < 1e-12:

    best_h = project_onto_bound_simplex(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)

import time, numpy as np, scipy.optimize as opt

def _proj_simplex(x, target):
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def _softmax_obj_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")[n - 1 :]
    bc = beta * corr
    m = bc.max()
    w = np.exp(bc - m)
    w /= w.sum()
    grad = np.zeros_like(h)
    for k, wk in enumerate(w):
        if wk == 0.0:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[: n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[: n - k]
    obj = (m + np.log(w.sum())) / beta
    return obj, grad

def _c5(h):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def _slsqp_refine(h0, beta, rng):
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")[n - 1 :]
        bc = beta * corr
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
        options={"maxiter": 500, "ftol": 1e-9, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(500):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x
