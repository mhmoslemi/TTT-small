def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    c = np.fft.irfft(C, N)
    return c[:n]

def project_onto_sum_box(y, target):
    # Find lambda such that sum(min(1, max(0, y + λ))) = target
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    # Ensure f(lo) <= target <= f(hi)
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo+hi)/2, 0.0, 1.0)

import time
import numpy as np

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via FFT (real inputs). Returns length len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    c = np.fft.irfft(C, N)
    return c[:n]

def _project_sum_box(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function (using FFT for speed)."""
    n = h.size
    corr = _fft_convolve(h, (1.0 - h)[::-1])  # correlation h * (1-h)
    return corr.max() * (2.0 / n)

def _gradient_step(h: np.ndarray, beta: float, lr: float, target: float) -> np.ndarray:
    """One projected gradient step on the soft‑max approximation of max overlap."""
    n = h.size
    # soft‑max weights
    corr = _fft_convolve(h, (1.0 - h)[::-1])
    max_corr = corr.max()
    w = np.exp(beta * (corr - max_corr))
    w /= w.sum()
    # gradient = conv(w,1-h) - conv(w,h_rev)
    conv1 = _fft_convolve(w, 1.0 - h)
    conv2 = _fft_convolve(w, h[::-1])
    g = conv1[n - 1 : n - 1 + n] - conv2[n - 1 : n - 1 + n]
    y = h - lr * g
    return _project_sum_box(y, target)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------- problem size ----------
    n = 4096  # higher resolution than the parent (512 -> 4096)

    # ----------- initial vector ----------
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        if base.size < n:
            factor = n // base.size
            remainder = n - factor * base.size
            h = np.tile(base, factor)
            if remainder:
                h = np.concatenate([h, base[:remainder]])
        else:
            h = base[:n].copy()
    else:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    h = np.clip(h, 0.0, 1.0)
    target_sum = n * 0.5
    h = _project_sum_box(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----------- parameters ----------
    beta = 5.0
    max_beta = 1e5
    beta_mult = 1.35
    base_lr = 0.8
    inner_steps = 2000
    c5_check_interval = 20

    # ----------- main optimisation loop ----------
    while time.time() - start < budget_s * 0.93 and beta < max_beta:
        lr = base_lr / np.sqrt(beta)
        for step in range(inner_steps):
            if time.time() - start >= budget_s * 0.93:
                break
            h = _gradient_step(h, beta, lr, target_sum)
            if step % c5_check_interval == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        beta = min(max_beta, beta * beta_mult)
        # occasional gentle noise to escape shallow plateaus
        if time.time() - start < budget_s * 0.90:
            noise = rng.normal(scale=0.015 / np.sqrt(beta), size=n)
            h = _project_sum_box(np.clip(h + noise, 0.0, 1.0), target_sum)

    # ----------- final fine‑graining (very high beta, few steps) ----------
    if time.time() - start < budget_s * 0.97:
        beta = max_beta * 5
        lr = base_lr / np.sqrt(beta) * 0.5
        for _ in range(500):
            if time.time() - start >= budget_s * 0.97:
                break
            h = _gradient_step(h, beta, lr, target_sum)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
