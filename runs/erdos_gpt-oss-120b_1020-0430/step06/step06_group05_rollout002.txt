import time
import numpy as np

# ---- helper functions -------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_shift_and_gradient(h: np.ndarray):
    """Return worst shift k (>=0) and gradient of O_k = Σ h_i(1-h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    g = np.ones(n, dtype=float)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    return k, g


def project_onto_sum_box(z: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of z onto [0,1]^n ∩ {sum = target}."""
    low = -np.max(z) - 2.0
    high = 2.0 - np.min(z)
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(z + mid, 0.0, 1.0)
        if x.sum() < target:
            low = mid
        else:
            high = mid
    x = np.clip(z + (low + high) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x += diff / x.size
        x = np.clip(x, 0.0, 1.0)
    return x


# ---- main entry point -------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target = n / 2.0

    # ----- initialise h ------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_sum_box(h, target)
    else:
        h = rng.random(n)
        h = project_onto_sum_box(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- optimisation loop -------------------------------------------
    alpha = 1.0
    min_alpha = 1e-6
    iter_cnt = 0
    while time.time() - start < budget_s * 0.985:
        iter_cnt += 1

        # current worst shift and gradient
        k, g = worst_shift_and_gradient(h)
        cur_c = compute_c5(h)

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            if best_c < 0.38078:
                break

        # line search along -g with projection
        improved = False
        a = alpha
        while a > min_alpha:
            z = h - a * g
            h_new = project_onto_sum_box(z, target)
            c_new = compute_c5(h_new)
            if c_new < cur_c - 1e-12:
                h = h_new
                # modestly increase step size for next iteration
                alpha = min(a * 1.2, 2.0)

                break
            a *= 0.5

        if not improved:
