def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        # ensure even n
        if n % 2 != 0:
            # pad with zero to make even? We'll just reduce n by 1.
            n -= 1
    else:
        n = 512
    target = n // 2
    # Build initial binary h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values[:n], dtype=float)
        # convert to binary: top target values set to 1
        idx_sorted = np.argsort(h0)[::-1]
        h = np.zeros(n, dtype=np.int8)
        h[idx_sorted[:target]] = 1
    else:
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, target, replace=False)
        h[ones] = 1
    # Compute autocorrelation
    corr = np.correlate(h, h, mode='full')
    autoc = corr[n-1:].astype(np.int32)  # length n
    # current C5
    cur_min_autoc = autoc[1:].min()
    cur_c = target - cur_min_autoc
    best_c = cur_c
    best_h = h.copy()
    # Preallocate delta arrays
    delta_i = np.zeros(n, dtype=np.int32)
    delta_j = np.zeros(n, dtype=np.int32)
    # set temperature initial
    T0 = cur_c if cur_c > 0 else 1.0
    # main loop
    it = 0
    max_no_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.99:
            break
        # fraction for temperature schedule
        frac = it / (budget_s * 200)  # approximate total iterations 200 per second * budget
        # just avoid zero division
        temp = max(T0 * (1 - frac), 1e-8)
        # find worst lag
        k_min = int(np.argmin(autoc[1:])) + 1
        n_k = n - k_min
        # target swap indices
        if n_k > 0:
            h_base = h[:n_k]
            h_shift = h[k_min:]
            mask_i = (h_base == 1) & (h_shift == 0)
            mask_j = (h_base == 0) & (h_shift == 1)
            idx_i = np.flatnonzero(mask_i)
            idx_j = np.flatnonzero(mask_j)
        else:
            idx_i = np.empty(0, dtype=int)
            idx_j = np.empty(0, dtype=int)
        if idx_i.size and idx_j.size:
            i = int(rng.choice(idx_i))
            j = int(rng.choice(idx_j) + k_min)
        else:
            # random swap
            ones_idx = np.flatnonzero(h)
            zeros_idx = np.flatnonzero(1 - h)
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = int(rng.choice(ones_idx))
            j = int(rng.choice(zeros_idx))
        # Apply removal of i
        delta_i.fill(0)
        if i < n - 1:
            delta_i[1:n-i] -= h[i+1:n]
        if i > 0:
            delta_i[1:i+1] -= h[i-1::-1]
        autoc += delta_i
        h[i] = 0
        # Apply addition of j
        delta_j.fill(0)
        if j < n - 1:
            delta_j[1:n-j] += h[j+1:n]
        if j > 0:
            delta_j[1:j+1] += h[j-1::-1]
        autoc += delta_j
        h[j] = 1
        # evaluate new c5
        new_min_autoc = autoc[1:].min()
        new_c = target - new_min_autoc
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # acceptance probability
            if temp > 0.0:
                prob = math.exp(-(new_c - cur_c) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:

            if cur_c < best_c:

                best_h = h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    # -------------------------------------------------------------
    # Helper: compute the final C5 value using the original definition
    # -------------------------------------------------------------
    def compute_c5(arr):
        """C5 for a binary (or float) array."""
        arr = arr.astype(float)
        corr = np.correlate(arr, 1.0 - arr, mode="full")
        return np.max(corr) * (2.0 / arr.size)

    start = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # Determine problem size and initialise a binary candidate
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        if n % 2:
            n -= 1                     # must be even
        target = n // 2
        h0 = np.array(initial_h_values[:n], dtype=float)
        # binaryisation: keep the largest `target` values as 1
        idx = np.argsort(h0)[::-1]
        h = np.zeros(n, dtype=np.int8)
        h[idx[:target]] = 1
    else:
        n = 512
        target = n // 2
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, target, replace=False)
        h[ones] = 1

    # -------------------------------------------------------------
    # Autocorrelation (integer) and current C5
    # -------------------------------------------------------------
    corr = np.correlate(h, h, mode="full")          # length 2n‑1
    autoc = corr[n - 1 :].astype(np.int32)          # autoc[k] for k>=0
    cur_min_autoc = autoc[1:].min()
    cur_c = target - cur_min_autoc
    best_c = cur_c
    best_h = h.copy()

    # -------------------------------------------------------------
    # Pre‑allocate delta arrays for incremental updates
    # -------------------------------------------------------------
    delta_i = np.zeros(n, dtype=np.int32)
    delta_j = np.zeros(n, dtype=np.int32)

    # temperature schedule – start at current C5, cool linearly
    T0 = float(cur_c) if cur_c > 0 else 1.0
    it = 0
    max_no_improve = 0

    while True:
        # ---- time check -------------------------------------------------
        if time.time() - start > budget_s * 0.99:
            break

        # ---- temperature -------------------------------------------------
        elapsed = time.time() - start
        temp = max(0.0, T0 * (1.0 - elapsed / (budget_s * 0.98)))

        # ---- choose a swap ------------------------------------------------
        # k where autoc is smallest (worst overlap)
        k_min = int(np.argmin(autoc[1:])) + 1
        n_k = n - k_min
        if n_k > 0:
            base = h[:n_k]
            shifted = h[k_min:]
            mask_i = (base == 1) & (shifted == 0)          # 1 that could become 0
            mask_j = (base == 0) & (shifted == 1)          # 0 that could become 1
            idx_i = np.flatnonzero(mask_i)
            idx_j = np.flatnonzero(mask_j)
        else:
            idx_i = idx_j = np.empty(0, dtype=int)

        if idx_i.size and idx_j.size:
            i = int(rng.choice(idx_i))
            j = int(rng.choice(idx_j) + k_min)            # real index of j
        else:
            # fallback: completely random 1↔0 swap
            ones_idx = np.flatnonzero(h)
            zeros_idx = np.flatnonzero(1 - h)
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = int(rng.choice(ones_idx))
            j = int(rng.choice(zeros_idx))

        # ---- apply removal of i (1 → 0) ---------------------------------
        delta_i.fill(0)
        if i < n - 1:
            delta_i[1 : n - i] -= h[i + 1 : n]
        if i > 0:
            delta_i[1 : i + 1] -= h[i - 1 :: -1]
        autoc += delta_i
        h[i] = 0

        # ---- apply addition of j (0 → 1) ---------------------------------
        delta_j.fill(0)
        if j < n - 1:
            delta_j[1 : n - j] += h[j + 1 : n]
        if j > 0:
            delta_j[1 : j + 1] += h[j - 1 :: -1]
        autoc += delta_j
        h[j] = 1

        # ---- evaluate new C5 --------------------------------------------
        new_min_autoc = autoc[1:].min()
        new_c = target - new_min_autoc

        # ---- acceptance --------------------------------------------------
        accept = False
        if new_c < cur_c:
            accept = True
        elif temp > 0.0:
            prob = math.exp(-(new_c - cur_c) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_c = new_c
            cur_min_autoc = new_min_autoc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                max_no_improve = 0
            else:
                max_no_improve += 1
        else:
            # revert the whole swap
            h[j] = 0
            autoc -= delta_j
            h[i] = 1
            autoc -= delta_i
            max_no_improve += 1

        # ---- occasional clean recomputation (stability) ------------------
        if (it & 1023) == 0:
            corr = np.correlate(h, h, mode="full")
            autoc = corr[n - 1 :].astype(np.int32)

        # ---- stagnation handling -----------------------------------------
        if max_no_improve > 20000:
            # restart from the best observed configuration, perturb a bit
            h = best_h.copy()
            # a short burst of cheap random swaps (no Metropolis check)
            for _ in range(500):
                ones_idx = np.flatnonzero(h)
                zeros_idx = np.flatnonzero(1 - h)
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i2 = int(rng.choice(ones_idx))
                j2 = int(rng.choice(zeros_idx))
                delta_i.fill(0)
                if i2 < n - 1:
                    delta_i[1 : n - i2] -= h[i2 + 1 : n]
                if i2 > 0:
                    delta_i[1 : i2 + 1] -= h[i2 - 1 :: -1]
                autoc += delta_i
                h[i2] = 0
                delta_j.fill(0)
                if j2 < n - 1:
                    delta_j[1 : n - j2] += h[j2 + 1 : n]
                if j2 > 0:
                    delta_j[1 : j2 + 1] += h[j2 - 1 :: -1]
                autoc += delta_j
                h[j2] = 1
            # recompute everything cleanly
            corr = np.correlate(h, h, mode="full")
            autoc = corr[n - 1 :].astype(np.int32)
            cur_min_autoc = autoc[1:].min()
            cur_c = target - cur_min_autoc
            max_no_improve = 0

        it += 1

    # -------------------------------------------------------------
    # Final evaluation of the best found solution
    # -------------------------------------------------------------
    final_c = compute_c5(best_h.astype(float))
    return best_h.astype(float).tolist(), float(final_c), int(n)
