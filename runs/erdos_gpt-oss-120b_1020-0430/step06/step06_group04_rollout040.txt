def binary_sa_long(h_bin, rng, deadline):
    n = h_bin.size
    cur_c = compute_c5(h_bin)
    best_c = cur_c
    best_h = h_bin.copy()
    # temperature schedule
    T0 = cur_c * 0.02 + 1e-6
    decay = 0.99995  # produce many steps before T small
    T = T0
    it = 0
    while time.time() < deadline:
        # pick swap
        i = rng.integers(0, n)
        while h_bin[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h_bin[j] == 1:
            j = rng.integers(0, n)
        # perform swap
        h_bin[i] = 0
        h_bin[j] = 1
        new_c = compute_c5(h_bin)
        # acceptance
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            # revert
            h_bin[i] = 1
            h_bin[j] = 0
        it += 1
        T *= decay
        # occasionally reset T when too low to allow escapes?
        if T < 1e-12:
            T = T0 * (0.5 ** (it // 100000))  # maybe not needed
    return best_h, best_c

def apply_swap(v, h, i, j):
    n = h.shape[0]
    rev_h = h[::-1]

def binary_sa_incremental(h_bin, rng, deadline):
    n = h_bin.size
    dx = 2.0 / n
    v = np.correlate(h_bin, 1 - h_bin, mode='full')
    cur_c = v.max() * dx
    best_c = cur_c
    best_h = h_bin.copy()
    best_v = v.copy()
    # initial temperature
    T = cur_c * 0.02 + 1e-8
    decay = 0.99995
    while time.time() < deadline:
        # choose i (ones) and j (zeros)
        i = rng.integers(0, n)
        while h_bin[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h_bin[j] == 1:
            j = rng.integers(0, n)
        old_v = v.copy()
        old_hi = 1  # h_bin[i] before swap
        old_hj = 0
        apply_swap(v, h_bin, i, j)
        new_c = v.max() * dx
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):

        if accept:

            h_bin[i] = old_hi

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000):
    """Modify h in‑place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    """Gradient of (1/β)·log∑exp(β·overlap)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = shift_idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k :]    -= wi * h[: n - k]
        else:
            kk = -k
            g[kk :]   += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk :]
    return g


def gradient_descent_smoothmax(h, betas, base_step, steps_per_beta, rng, deadline):
    """Projected gradient descent on the smooth‑max surrogate."""
    n = h.size
    for beta in betas:
        step = base_step
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 0.125, 2.0):
                cand = h - step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
            step *= 0.97
    return h

def _apply_swap(v, h, i, j):
    """Update full correlation v and binary vector h for swapping i (1) ↔ j (0)."""
    n = h.shape[0]
    rev_h = h[::-1]
    # i contributions (hi=1 → 0)
    v[(n - 1 - i) : (2 * n - 1 - i)] += (h - 1)
    v[i : i + n] += rev_h
    # j contributions (hj=0 → 1)
    v[(n - 1 - j) : (2 * n - 1 - j)] += (1 - h)
    v[j : j + n] -= rev_h
    # d·d term: add +1 at shifts i‑j and j‑i
    v[i - j + (n - 1)] += 1
    v[j - i + (n - 1)] += 1
    # shift 0 must be zero
    v[n - 1] = 0
    # perform swap
    h[i] = 0
    h[j] = 1


def binary_sa_incremental(h_bin, rng, deadline):
    """Simulated‑annealing on binary vectors using O(n) incremental updates."""
    n = h_bin.size
    dx = 2.0 / n
    v = np.correlate(h_bin, 1 - h_bin, mode="full")
    cur_c = v.max() * dx
    best_c = cur_c
    best_h = h_bin.copy()
    T = cur_c * 0.02 + 1e-12

    while time.time() < deadline:

            j = rng.integers(0, n)

        _apply_swap(v, h_bin, i, j)

    # --- smooth‑max gradient descent (if time permits) -------------------
    if time.time() < deadline:
        betas = [5, 20, 80, 200, 800, 2000, 5000, 20000]
        h = gradient_descent_smoothmax(h, betas, base_step=0.25,
                                       steps_per_beta=30, rng=rng,
                                       deadline=deadline)

    # --- convert to binary ------------------------------------------------
    h_bin = np.rint(h).astype(np.int64)
    adjust_sum_exact(h_bin, rng)

    # --- binary simulated annealing on base size -------------------------
    if time.time() < deadline:
        h_bin, c_bin = binary_sa_incremental(h_bin, rng, deadline)
    else:
        c_bin = compute_c5(h_bin)

    best_h = h_bin
    best_c = c_bin
    best_n = n

    # --- optional up‑sampling ------------------------------------------------
    if time.time() < deadline and best_n * 2 <= 1440:
        h_up = np.repeat(best_h, 2)
        n_up = best_n * 2
        # binary SA on up‑sampled version
        if time.time() < deadline:
            h_up, c_up = binary_sa_incremental(h_up, rng, deadline)
            if c_up < best_c:
                best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # --- final safety adjustments -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
