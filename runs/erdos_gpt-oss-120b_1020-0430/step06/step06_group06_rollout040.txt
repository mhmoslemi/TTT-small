import time
import math
import numpy as np

def _random_one_zero_swap(h, rng):
    """Return (i,j) with h[i]==1, h[j]==0."""
    n = h.shape[0]
    # expect roughly half ones
    while True:
        i = rng.integers(n)
        if h[i]:
            break
    while True:
        j = rng.integers(n)
        if not h[j]:
            break
    return int(i), int(j)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise h ----------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        base = (base > 0.5).astype(np.uint8)          # binarise just in case
        h = np.repeat(base, 2)                       # duplicate → double size
    else:
        n = 1024
        h = np.zeros(n, dtype=np.uint8)
        h[: n // 2] = 1
        rng.shuffle(h)

    N = h.shape[0]
    dx = 2.0 / N

    # correlation S_k = sum_i h_i h_{i+k}
    S = np.correlate(h, h, mode="full").astype(np.int32)
    min_S = S.min()
    current_O = 1.0 - dx * min_S
    best_h = h.copy()
    best_O = current_O

    # temporary arrays for updates
    delta_S = np.zeros_like(S, dtype=np.int32)
    temp_S = np.empty_like(S, dtype=np.int32)

    # annealing schedule
    total_time = start + budget_s * 0.96   # reserve a bit for final check
    T0 = 0.02
    T_end = 1e-6
    prob_grad = 0.7

    # ----- main SA loop ----------------------------------------------------
    while True:
        now = time.time()
        if now >= total_time:
            break
        # temperature (exponential cooling)
        frac = (now - start) / (total_time - start)
        T = T0 * (T_end / T0) ** frac

        # choose move
        if rng.random() < prob_grad:
            # gradient‑guided swap based on current worst shift
            idx_min = S.argmin()
            shift = abs(idx_min - (N - 1))
            if shift == 0:
                g = 1 - 2 * h
            else:
                g = np.ones(N, dtype=np.int16)
                g[: N - shift] -= h[shift:]
                g[shift:] -= h[: N - shift]

            masked_on = np.where(h == 1, g, -32768)
            i = int(masked_on.argmax())
            masked_off = np.where(h == 0, g, 32767)
            j = int(masked_off.argmin())
            if g[i] <= g[j]:
                i, j = _random_one_zero_swap(h, rng)
        else:
            i, j = _random_one_zero_swap(h, rng)

        # ----- build delta_S for the swap ---------------------------------
        delta_S.fill(0)

        # removal of i (1 → 0)
        delta_S[N - 1] -= 1
        if i > 0:
            delta_S[N - 1 - i : N - 1] -= h[i - 1 :: -1]
        if i < N - 1:
            delta_S[N : N + (N - 1 - i)] -= h[i + 1 :]

        # addition of j (0 → 1)
        delta_S[N - 1] += 1
        if j > 0:
            delta_S[N - 1 - j : N - 1] += h[j - 1 :: -1]
        if j < N - 1:
            delta_S[N : N + (N - 1 - j)] += h[j + 1 :]

        # ----- evaluate new solution ---------------------------------------
        np.add(S, delta_S, out=temp_S)
        new_min_S = temp_S.min()
        delta_O = -dx * (new_min_S - min_S)   # >0 means worse

        # ----- accept / reject ---------------------------------------------
        if delta_O <= 0.0 or rng.random() < math.exp(-delta_O / T):
            # accept
            S, temp_S = temp_S, S        # swap references
            min_S = new_min_S
            current_O = 1.0 - dx * min_S
            h[i] = 0
            h[j] = 1
            if current_O < best_O:
                best_O = current_O
                best_h = h.copy()
        # else: reject (no changes)

    # ----- final verification (re‑compute from scratch) --------------------
    final_S = np.correlate(best_h, best_h, mode="full")
    final_O = 1.0 - dx * final_S.min()
    best_O = float(final_O)

    return best_h.tolist(), best_O, int(N)
