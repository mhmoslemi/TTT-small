```
def compute_c5(h):
    n = h.shape[0]
    b = 1.0 - h
    # reverse b
    b_rev = b[::-1]
    size = 1 << (2*n - 1).bit_length()
    # FFT
    fa = np.fft.rfft(h, n=size)
    fb = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:2*n-1]
    return np.max(conv) * (2.0 / n)
```

def worst_shift_and_grad(h):
    n = h.shape[0]
    b = 1.0 - h
    b_rev = b[::-1]
    size = 1 << (2*n - 1).bit_length()

def mass_transfer_move(h, grad, alpha, target, rng):
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])

def random_perturb(h, rng, scale=0.2):
    n = h.shape[0]
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i==j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2
```

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    max_iter = 1000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            it += 1
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
        it += 1
    # If still diff, adjust first element forcibly
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  using FFT."""
    n = h.shape[0]
    b = 1.0 - h
    b_rev = b[::-1]
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(h, n=size)
    fb = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < 1000:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            it += 1
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Shift k attaining max overlap integral and its gradient."""
    n = h.shape[0]
    b = 1.0 - h
    b_rev = b[::-1]
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(h, n=size)
    fb = np.fft.rfft(b_rev, n=size)
    corr = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        kp = -k
        grad[kp:] -= h[: n - kp]
        grad[:kp] -= h[kp:]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Transfer mass from positive‑gradient to negative‑gradient positions."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    h_new[pos] = h[pos] * (1.0 - eps / total_out)
    h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h

        else:
            h = h0.copy()
    else:

        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    h = enforce_sum(h, target_sum, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    cur_h = h.copy()
    cur_c = best_c

                                 step=0.01, max_iter=20)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
