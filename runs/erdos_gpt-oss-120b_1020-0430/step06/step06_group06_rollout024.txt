def project_onto_capsimplex(v, s):
    x = np.clip(v, 0.0, 1.0)
    total = x.sum()
    if abs(total - s) < 1e-12:
        return x
    # Adjust delta
    if total < s:
        # Need to add amount = s - total
        deficit = s - total
        # available capacity per element = 1 - x_i
        while deficit > 1e-12:
            caps = 1 - x
            total_cap = caps.sum()
            if total_cap == 0:
                break
            add = deficit * caps / total_cap
            x += add
            # Clip again to handle overshoot
            x = np.clip(x, 0.0, 1.0)
            deficit = s - x.sum()
    else:
        # need to subtract
        surplus = total - s
        while surplus > 1e-12:
            caps = x
            total_cap = caps.sum()
            if total_cap == 0:
                break
            sub = surplus * caps / total_cap
            x -= sub
            x = np.clip(x, 0.0, 1.0)
            surplus = x.sum() - s
    return x
```

def project_to_sum_cap(v, s):
    # Clip to [0,1] first
    x = np.clip(v, 0.0, 1.0)
    total = x.sum()
    if abs(total - s) < 1e-12:
        return x
    # If total < s, need to add mass to entries less than 1.
    # If total > s, need to remove mass from entries greater than 0.
    # Use while loop with tolerance.
    max_iter = 1000
    for _ in range(max_iter):
        if total < s:
            deficit = s - total
            # capacity to add
            cap = 1.0 - x
            cap_sum = cap.sum()
            if cap_sum == 0:
                break
            # Distribute deficit proportionally to capacity
            add = deficit * cap / cap_sum
            x += add
            x = np.clip(x, 0.0, 1.0)
        else:
            surplus = total - s
            cap = x
            cap_sum = cap.sum()
            if cap_sum == 0:
                break
            sub = surplus * cap / cap_sum
            x -= sub
            x = np.clip(x, 0.0, 1.0)
        total = x.sum()
        if abs(total - s) < 1e-12:
            break
    return x

        it += 1
        if it % 1000 == 0:
            beta = min(beta * beta_mult, max_beta)
            step *= step_decay
            step = max(step, 1e-4)

        if it % pert_interval == 0:
            i, j = rng.integers(0, n, size=2)
            h[i], h[j] = h[j], h[i]
            eps = rng.random() * 0.08
            h[i] = np.clip(h[i] + eps, 0.0, 1.0)
            h[j] = np.clip(h[j] - eps, 0.0, 1.0)
            h = project_to_sum_cap(h, n / 2.0)

    # ----- final SLSQP refinement -----
    global CURRENT_TARGET, CURRENT_BETA

    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    CURRENT_TARGET = n / 2.0
    h_cur = best_h.copy()
    for beta in (50.0, 500.0, 2000.0, 8000.0, 30000.0):
        if time.time() > deadline:
            break
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        res = minimize(
            softmax_obj,
            h_cur,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 200, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = project_to_sum_cap(x, n / 2.0)
        h_cur = x
        cur_c = compute_c5(h_cur)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_cur.copy()

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
