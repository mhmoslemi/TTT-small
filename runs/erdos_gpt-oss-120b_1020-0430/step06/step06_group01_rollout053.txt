def enforce_sum(h, rng):

    target = n/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # We'll adjust via random indices; vectorized: get mask of where h<1 if diff>0, else h>0.
    # We'll loop while diff not zero but each iteration can adjust many elements.
    # But diff magnitude could be up to n/2 = 256. We'll iterate maybe up to 256.
    # Use while.
    while abs(diff) > 1e-9:
        if diff > 0:
            idx = rng.choice(np.where(h < 1)[0], size=1)[0]
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.choice(np.where(h > 0)[0], size=1)[0]
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    k = abs(shift)
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        # According to earlier code: gradient is vector where change at i changes overlap for shift k:
        # For shift k>0, overlap = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]).
        # The derivative w.r.t h[i] where i < n-k: (1 - h[i+k]) (increase in h[i] increases overlap)
        # The derivative w.r.t h[i+k] where i < n-k: -h[i] (increase in h[i+k] reduces overlap)
        # So gradient[i] = (1 - h[i+k]) for i in [0, n-k-1]; gradient[i] = -h[i - k] for i in [k, n-1]; others 0.
        # Combine.
        grad[:n - k] = 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    return grad

def hill_climb(h, rng, max_iters, init_step=0.7, step_decay=0.9995, min_step=1e-5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        grad = worst_gradient(h)
        # find i with largest positive gradient (increase leads to larger C5) and j with most negative gradient (decrease C5)
        # But we want to reduce C5, so we should move mass from i (positive gradient) to j (negative gradient).
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] <= 0:  # no positive gradient left, can't improve?
            # maybe gradient small: break
            break
        # max epsilon we can transfer
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            # cannot transfer; maybe try another pair? we could randomize.
            # just skip.
            continue
        eps = rng.random() * max_eps * step
        if eps <= 0:
            continue
        h[i] -= eps
        h[j] += eps
        # enforce sum (should remain same - but we transferred so sum unchanged)

        if (it % 1000) == 0:

import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.flatnonzero(h < 1.0)
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            idx = np.flatnonzero(h > 0.0)
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    k = abs(shift)
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- gradient‑directed hill climb ----------
def hill_climb(h0, rng, max_iters, init_step=0.7, step_decay=0.9999, min_step=1e-5):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] <= 0:
            break
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-14:
            continue
        eps = rng.random() * max_eps * step
        h[i] -= eps
        h[j] += eps
        if (it & 1023) == 0:  # every 1024 steps
            h = enforce_sum(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        step = max(step * step_decay, min_step)
    best_h = enforce_sum(best_h, rng)
    return best_h, compute_c5(best_h)


# ---------- binary simulated annealing ----------
def binary_sa(n, rng, time_budget):
    start = time.time()
    best_h, best_c = None, np.inf
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        # inner SA
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T0, T_end = 0.1, 1e-4
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = rng.choice(np.flatnonzero(h))
            j = rng.choice(np.flatnonzero(1.0 - h))
            h[i], h[j] = 0.0, 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i], h[j] = 1.0, 0.0
        if c_curr < best_c:
            best_c, best_h = c_curr, h.copy()
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=1000):

        h0,
        method="SLSQP",
        bounds=bounds,

    )
    x = np.clip(res.x, 0.0, 1.0)
