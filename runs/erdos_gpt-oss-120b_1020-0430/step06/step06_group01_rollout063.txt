def project_to_capped_simplex(z, M, cap=1.0, tol=1e-9, max_iter=100):
    # caps all values at [0, cap] after translation.
    # Find tau via bisection.
    lo = np.min(z) - cap
    hi = np.max(z)
    for _ in range(50):
        tau = (lo + hi) / 2.0
        x = np.clip(z - tau, 0, cap)
        s = x.sum()
        if s > M:
            lo = tau
        else:
            hi = tau
    x = np.clip(z - (lo+hi)/2.0, 0, cap)
    # after bisection, sum may be slightly off, adjust by distributing residual
    residual = M - x.sum()
    if abs(residual) > 1e-8:
        # adjust by adding residual to indices not at bound, proportional to 1.
        # choose indices where 0 < x[i] < cap.
        free = (x > 0) & (x < cap)
        if np.any(free):
            incr = residual / np.count_nonzero(free)
            x[free] += incr
        else:
            # fallback: add to arbitrary index within bounds
            i = np.argmin(cap - x)  # index with largest margin
            x[i] += residual
        # clip again
        x = np.clip(x, 0, cap)
    return x
```

def smooth_descent(h0, rng, time_budget):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    n = h.shape[0]
    M = n / 2.0
    # Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    # beta schedule
    beta_vals = np.geomspace(10.0, 20000.0, num=30)
    for beta in beta_vals:
        # compute how many iterations we can afford for this beta
        remaining = time_budget - (time.time() - start)
        if remaining <= 0:
            break
        # allocate proportion of remaining to this beta based on log spacing maybe:
        # just set iters = max(1, int(remaining * 200)) (200 iters per sec)
        iters = int(min(5000, remaining * 200))
        for _ in range(iters):
            t_iter += 1
            # correlation
            c = np.correlate(h, 1.0 - h, mode='full')
            # softmax weights
            bc = beta * c
            max_bc = np.max(bc)
            exp_bc = np.exp(bc - max_bc)
            w = exp_bc / exp_bc.sum()
            # compute gradient using convolution
            w_pos = w[n-1:]
            u = 1.0 - h
            w_rev = w_pos[::-1]
            A = np.convolve(w_rev, u, mode='full')[n-1:n-1 + n]
            B = np.convolve(w_pos, h, mode='full')[:n]
            grad = A - B
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t_iter)
            v_hat = v / (1 - beta2**t_iter)
            step = 0.2 / (np.sqrt(v_hat) + eps)  # base step size
            h_new = h - step * m_hat
            # project onto capped simplex

            # evaluate

            if c_val < best_c:

                best_h = h_new.copy()
            h = h_new
            if time.time() - start > time_budget * 0.99:

        # reset Adam moments after each beta to avoid stale
        m.fill(0)
        v.fill(0)
        if time.time() - start > time_budget * 0.99:
            break
    return best_h, best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function on [0,2] with n samples."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_capped_simplex(z: np.ndarray, M: float, cap: float = 1.0, tol: float = 1e-9,
                          max_iter: int = 80) -> np.ndarray:
    """Euclidean projection of z onto {x: 0<=x_i<=cap, sum x_i = M}."""
    lo = np.min(z) - cap
    hi = np.max(z)
    for _ in range(max_iter):
        tau = (lo + hi) * 0.5
        x = np.clip(z - tau, 0.0, cap)
        s = x.sum()
        if s > M:
            lo = tau
        else:
            hi = tau
    x = np.clip(z - (lo + hi) * 0.5, 0.0, cap)
    diff = M - x.sum()
    if abs(diff) > tol:
        free = (x > 0.0) & (x < cap)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, cap)
        else:
            i = np.argmin(x) if diff > 0 else np.argmax(x)
            x[i] = np.clip(x[i] + diff, 0.0, cap)
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‐shift overlap (max corr)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def hill_climb(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Guided hill‑climb using the worst‑shift gradient."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    step_scale = 0.5
    decay_interval = 2000
    it = 0
    while time.time() - start < time_budget:
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        if it % 10 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
        if (it + 1) % decay_interval == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
        it += 1
    c = compute_c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c


def smooth_descent(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Projected Adam on the soft‑max approximation of max corr."""
    h = h0.copy()
    n = h.size
    M = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0
    beta_vals = np.geomspace(10.0, 20000.0, num=30)
    for beta in beta_vals:
        if time.time() - start >= time_budget:
            break
        remaining = time_budget - (time.time() - start)
        iters = int(min(5000, remaining * 200))
        for _ in range(iters):
            t_iter += 1
            c = np.correlate(h, 1.0 - h, mode="full")
            bc = beta * c
            max_bc = np.max(bc)
            w = np.exp(bc - max_bc)
            w /= w.sum()
            w_pos = w[n - 1 :]                     # shifts 0 .. n-1
            u = 1.0 - h
            w_rev = w_pos[::-1]
            A = np.convolve(w_rev, u, mode="full")[n - 1 : n - 1 + n]
            B = np.convolve(w_pos, h, mode="full")[:n]
            grad = A - B
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)
            step = 0.2 / (np.sqrt(v_hat) + eps)
            h_new = h - step * m_hat
            h_new = project_capped_simplex(h_new, M=M, cap=1.0)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
            h = h_new
            if time.time() - start >= time_budget * 0.99:
                break
        m.fill(0.0)
        v.fill(0.0)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    n = 512
    M = n / 2.0

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_capped_simplex(h0, M=M, cap=1.0)
    else:
        h0 = np.full(n, 0.5)
        h0 += (rng.random(n) - 0.5) * 0.02
        h0 = project_capped_simplex(h0, M=M, cap=1.0)

    best_h, best_c = h0.copy(), compute_c5(h0)
    start = time.time()

    # ----- stage 1: hill‑climb -----
    t1 = budget_s * 0.30
    h1, c1 = hill_climb(best_h, rng, t1)
    if c1 < best_c:
        best_h, best_c = h1, c1
        if best_c <= 0.3808:
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = project_capped_simplex(best_h, M=M, cap=1.0)
            return best_h.tolist(), float(best_c), int(n)

    # ----- stage 2: smooth descent -----
    remaining = budget_s - (time.time() - start)
    if remaining > 0:
        h2, c2 = smooth_descent(best_h, rng, remaining * 0.90)
        if c2 < best_c:
            best_h, best_c = h2, c2
            if best_c <= 0.3808:
                best_h = np.clip(best_h, 0.0, 1.0)
                best_h = project_capped_simplex(best_h, M=M, cap=1.0)
                return best_h.tolist(), float(best_c), int(n)

    # ----- final tiny stochastic tweaks -----
    while time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * 0.005
                h_new = best_h.copy()
                h_new[i] -= eps
                h_new[j] += eps
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_h, best_c = h_new, c_new

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_capped_simplex(best_h, M=M, cap=1.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
