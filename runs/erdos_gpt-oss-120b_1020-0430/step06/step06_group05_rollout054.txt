def global_smoothing_move(h, alpha, rng):
    # transfer from top fraction to bottom fraction
    n = h.size
    # determine indices sorted by h value
    sorted_idx = np.argsort(h)
    # bottom half (lower values)
    low = sorted_idx[:n//4]  # add to low
    # top half (higher values)
    high = sorted_idx[-n//4:]  # subtract from high
    # compute total_out = sum(h[high])
    # total_in = sum(1 - h[low])
    total_out = h[high].sum()
    total_in = (1 - h[low]).sum()
    eps = min(total_out, total_in) * alpha
    if total_out > 0:
        h[high] = h[high] * (1 - eps / total_out)
    if total_in > 0:
        h[low] = h[low] + (1 - h[low]) * (eps / total_in)
    return h

def targeted_move(h, k, alpha, rng):
    n = h.size
    if k == 0:
        # gradient move: move from high to low
        grad = 1.0 - 2.0 * h
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            return h
        i_pos = pos[np.argmax(grad[pos])]
        i_neg = neg[np.argmin(grad[neg])]
        delta_max = min(h[i_neg], 1.0 - h[i_pos])
        delta = min(alpha * delta_max, delta_max)
        h_new = h.copy()
        h_new[i_neg] -= delta
        h_new[i_pos] += delta
        return h_new
    else:
        i0 = np.arange(0, n - k)
        b = (1.0 - h[i0 + k]) + h[i0]
        delta_max = np.minimum(h[i0], 1.0 - h[i0 + k])
        mask = delta_max > 1e-12
        if not np.any(mask):
            return h
        i_candidates = i0[mask]

        i_best = i_candidates[np.argmax(b_masked)]

def local_refine(h, target, rng, time_end):
    n = h.size
    i_range = np.arange(n)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    h_curr = h.copy()
    c_curr = best_c
    start = time.time()
    # initial temperature
    T0 = max(0.001, (best_c - 0.35) * 0.5)  # guess
    iter = 0
    while time.time() < time_end:
        # compute worst shift
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # propose new h
        alpha = 0.5 * (0.99 ** iter)  # decreasing
        # 80% targeted move, 20% global move
        if rng.random() < 0.8:
            # targeted move
            if k == 0:
                # gradient move as described
                grad = 1.0 - 2.0 * h_curr
                pos = np.where(grad > 1e-12)[0]
                neg = np.where(grad < -1e-12)[0]
                if pos.size > 0 and neg.size > 0:
                    i_pos = pos[np.argmax(grad[pos])]
                    i_neg = neg[np.argmin(grad[neg])]
                    delta_max = min(h_curr[i_neg], 1.0 - h_curr[i_pos])
                    delta = min(alpha * delta_max, delta_max)
                    h_new = h_curr.copy()
                    h_new[i_neg] -= delta
                    h_new[i_pos] += delta
                else:
                    h_new = random_swap_move(h_curr, rng)
            else:
                i0 = i_range[: n - k]
                b = (1.0 - h_curr[i0 + k]) + h_curr[i0]
                delta_max = np.minimum(h_curr[i0], 1.0 - h_curr[i0 + k])
                mask = delta_max > 1e-12
                if np.any(mask):
                    i_candidates = i0[mask]
                    b_masked = b[mask]
                    i_best = i_candidates[np.argmax(b_masked)]
                    max_delta = delta_max[i_best]
                    delta = min(alpha * max_delta, max_delta)
                    h_new = h_curr.copy()
                    h_new[i_best] -= delta
                    h_new[i_best + k] += delta

                    h_new = random_swap_move(h_curr, rng)
        else:

import time, math, numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def random_binary_h(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, int(target), replace=False)
    h[idx] = 1.0
    return h


def random_swap_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap a 0 and a 1, preserving the total sum."""
    zeros = np.where(h < 1e-12)[0]
    ones = np.where(h > 1.0 - 1e-12)[0]
    if zeros.size == 0 or ones.size == 0:
        return h
    i, j = rng.choice(zeros), rng.choice(ones)
    h2 = h.copy()
    h2[i] = 1.0
    h2[j] = 0.0
    return h2


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


# ---------- local SA refinement ----------
def local_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 time_end: float) -> tuple:
    """Simulated‑annealing style refinement for C5."""
    n = h.size
    i_range = np.arange(n)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    start = time.time()
    T0 = max(0.001, (best_c - 0.35) * 0.5)  # initial temperature heuristic
    it = 0

    while time.time() < time_end:
        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        lag = np.argmax(corr) - (n - 1)
        k = abs(lag)

        # move proposal
        alpha = 0.5 * (0.99 ** it)          # decreasing step size
        if rng.random() < 0.8:              # 80% targeted
            if k == 0:
                grad = 1.0 - 2.0 * cur_h
                pos = np.where(grad > 1e-12)[0]
                neg = np.where(grad < -1e-12)[0]
                if pos.size and neg.size:
                    i_pos = pos[np.argmax(grad[pos])]
                    i_neg = neg[np.argmin(grad[neg])]
                    dmax = min(cur_h[i_neg], 1.0 - cur_h[i_pos])
                    delta = min(alpha * dmax, dmax)
                    h_new = cur_h.copy()
                    h_new[i_neg] -= delta
                    h_new[i_pos] += delta
                else:
                    h_new = random_swap_move(cur_h, rng)
            else:
                i0 = i_range[: n - k]
                b = (1.0 - cur_h[i0 + k]) + cur_h[i0]
                dmax = np.minimum(cur_h[i0], 1.0 - cur_h[i0 + k])
                mask = dmax > 1e-12
                if mask.any():
                    cand = i0[mask]
                    bmask = b[mask]
                    i_best = cand[np.argmax(bmask)]
                    d = dmax[i_best]
                    delta = min(alpha * d, d)
                    h_new = cur_h.copy()
                    h_new[i_best] -= delta
                    h_new[i_best + k] += delta
                else:
                    h_new = random_swap_move(cur_h, rng)
        else:                                 # 20% plain random swap
            h_new = random_swap_move(cur_h, rng)

        h_new = np.clip(h_new, 0.0, 1.0)
        c_new = compute_c5(h_new)

        # temperature based acceptance
        elapsed = time.time() - start
        frac = elapsed / (time_end - start)
        T = T0 * max(0.001, 1.0 - frac)

        if c_new < cur_c - 1e-12 or rng.random() < math.exp(-(c_new - cur_c) / T):
            cur_h, cur_c = h_new, c_new
            if cur_c < best_c - 1e-12:
                best_c, best_h = cur_c, cur_h.copy()
        it += 1
    return best_h, best_c


# ---------- gradient polish ----------
def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                   steps: int = 15) -> np.ndarray:
    """Projected gradient descent on the worst‑shift gradient."""
    for s in range(steps):
        _, grad = worst_shift_and_grad(h)
        gamma = 0.015 * (0.5 ** s)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


# ---------- SLSQP soft‑max polish ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s
    # split time roughly in half for n=256 then n=512
    mid = start + total * 0.5
    final_deadline = start + total * 0.99

    # ---- phase 1: n=256 ----
    n1 = 256
    target1 = n1 / 2.0
    # initialise (ignore initial_h_values if length mismatches)
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size == n1:
            h = np.clip(init, 0.0, 1.0)
            h = enforce_sum(h, target1, rng)
        else:
            h = random_binary_h(n1, target1, rng)
    else:
        h = random_binary_h(n1, target1, rng)

    best_h1, best_c1 = local_refine(h, target1, rng, mid)

    # ---- phase 2: upscale to n=512 ----
    n2 = 512
    target2 = n2 / 2.0
    h_up = np.repeat(best_h1, 2)
    h_up = h_up.astype(float)
    # enforce exact sum (should already hold)
    h_up = enforce_sum(h_up, target2, rng)

    best_h2, best_c2 = local_refine(h_up, target2, rng, final_deadline)

    # ---- final polish with gradient and SLSQP ----
    best_h2 = gradient_refine(best_h2, target2, rng, steps=20)

    now = time.time()
    if now < final_deadline - 20:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() > final_deadline - 5:
                break
            h_s = slsqp_refine(best_h2, target2, rng, beta=beta, max_iter=300)
            c_s = compute_c5(h_s)
            if c_s < best_c2:
                best_c2, best_h2 = c_s, h_s.copy()

    # final enforcement and result
    best_h2 = np.clip(best_h2, 0.0, 1.0)
    best_h2 = enforce_sum(best_h2, target2, rng)
    final_c = compute_c5(best_h2)

    return best_h2.tolist(), float(final_c), int(best_h2.size)
