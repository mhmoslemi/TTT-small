def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # Setup n
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h0 = np.zeros(n)
        h0[rng.choice(n, n//2, replace=False)] = 1.0
    h0 = np.clip(h0, 0.0, 1.0)
    target = n / 2.0
    # projection function defined later
    h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Define optimizer parameters
    betas = [10, 50, 200, 800, 3200, 12800]
    # total iteration budget: maybe 3000 to 5000 iterations across all restarts.
    # We'll allocate per beta: steps_per_beta = 300
    steps_per_beta = 300

    # Prepare function to run optimization from a starting h
    def optimize_from(h_start):
        nonlocal best_h, best_c
        h = h_start.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        # For each beta
        for beta in betas:
            for _ in range(steps_per_beta):
                if time.time() - start > budget_s * 0.94:
                    return
                t += 1
                grad = grad_softmax(h, beta)  # returns array size n
                # Adam update
                beta1 = 0.9
                beta2 = 0.999
                eps = 1e-8
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad * grad)
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                # learning rate schedule: base_lr * (1/(1+beta*something)) maybe
                # Let's set base_lr = 0.2
                base_lr = 0.2
                lr = base_lr / (np.sqrt(v_hat) + eps)
                step = lr * m_hat
                # step might be large; we can scale down by factor
                # maybe apply elementwise scaling? Instead just apply step.
                h_new = h - step
                h_new = project_onto_box_sum(h_new, target)
                h = h_new
        # after finishing betas
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        return

    # Run a few restarts
    num_restarts = 5
    for r in range(num_restarts):
        if time.time() - start > budget_s * 0.94:
            break
        # random start: random values then project
        h_rand = rng.random(n)
        h_rand = project_onto_box_sum(h_rand, target)
        optimize_from(h_rand)

    # Also run optimization from best_h (might have already done)
    if time.time() - start < budget_s * 0.94:
        optimize_from(best_h)

    # final local search (small perturbations)
    while time.time() - start < budget_s * 0.985:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.01)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h = nh.copy()
            best_c = nc

    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions

def project_onto_box_sum(v, target, max_iter=60):
    """Project v onto [0,1]^n while fixing sum(v)=target."""
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def compute_c5(h):
    """Exact C5 using FFT (max overlap * dx)."""
    n = h.size
    target = n / 2.0
    N = 1 << ((3 * n).bit_length())
    H = np.fft.rfft(h, n=N)
    H_rev = np.fft.rfft(h[::-1], n=N)
    conv = np.fft.irfft(H * H_rev, n=N)[: 2 * n - 1]
    O = target - conv
    return O.max() * (2.0 / n)


def worst_gradient(h):
    """Gradient of the worst (max) overlap shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Move a little mass from a large‑gradient index to a small‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def grad_softmax(h, beta):
    """Gradient of the smooth (log‑sum‑exp) overlap objective."""
    n = h.shape[0]
    target = n / 2.0
    # FFT size covering 3n elements
    N = 1 << ((3 * n).bit_length())

    # ----- correlation h * h_rev -----
    H = np.fft.rfft(h, n=N)
    H_rev = np.fft.rfft(h[::-1], n=N)
    conv_hh = np.fft.irfft(H * H_rev, n=N)[: 2 * n - 1]

    O = target - conv_hh  # overlap values for all shifts
    Omax = O.max()
    w = np.exp(beta * (O - Omax))
    w /= w.sum()  # softmax weights, length 2n-1

    # ----- convolutions with w -----
    L = 2 * n - 1
    W = np.fft.rfft(w, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)

    conv_w_h = np.fft.irfft(W * H, n=N)[: L + n - 1]        # length 3n-2
    conv_w_rev_h = np.fft.irfft(W_rev * H, n=N)[: L + n - 1]

    offset = n - 1
    A = conv_w_rev_h[offset : offset + n]   # sum_k w_k * h_{i+k}
    B = conv_w_h[offset : offset + n]      # sum_k w_k * h_{i-k}
    grad = 1.0 - A - B
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    h0 = np.clip(h0, 0.0, 1.0)
    target = n / 2.0
    h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- optimisation parameters ----------------------------------------
    betas = [10, 50, 200, 800, 3200, 12800]
    steps_per_beta = 1000          # iterations per beta stage
    lr = 0.05                      # Adam base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # ------------------------------------------------------------
    def optimize_from(h_start):
        nonlocal best_h, best_c
        h = h_start.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        for beta in betas:
            for _ in range(steps_per_beta):
                if time.time() - start > budget_s * 0.94:
                    return
                t += 1
                grad = grad_softmax(h, beta)
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad * grad)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                h = project_onto_box_sum(h - step, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # ------------------------------------------------------------

    # ---- several random restarts ---------------------------------------
    restarts = 5
    for _ in range(restarts):
        if time.time() - start > budget_s * 0.94:
            break
        h_rand = rng.random(n)
        h_rand = project_onto_box_sum(h_rand, target)
        optimize_from(h_rand)

    # ---- polish the current best ---------------------------------------
    if time.time() - start < budget_s * 0.94:
        optimize_from(best_h)

    # ---- final local search with guided perturbations -------------------
    while time.time() - start < budget_s * 0.99:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.01)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---- final projection and result ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
