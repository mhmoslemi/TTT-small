def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.zeros_like(h)
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        # step size decreasing
        h = h - (step * (0.5 ** it)) * grad
        # clip and enforce sum
        np.clip(h, 0.0, 1.0, out=h)
        # enforce sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust random idx
            idx = rng.integers(n)
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
    return h

# ---------- utility ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        idx = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            if add <= 0:
                continue
            h[idx] += add
        else:
            sub = min(-diff, h[idx])
            if sub <= 0:
                continue
            h[idx] -= sub
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 via FFT (full cross‑correlation)."""
    n = h.shape[0]
    size = 2 * n - 1
    H = np.fft.rfft(h, size)
    B = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(B), size)
    return float(corr.max() * (2.0 / n))


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap term."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.zeros_like(h)
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = rng.integers(n)
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
    return h


# ---------- continuous simulated annealing ----------
def continuous_sa(h0: np.ndarray, rng: np.random.Generator,
                 time_budget: float, target_c: float = 0.3808) -> tuple:
    """Temperature‑controlled SA on the continuous domain."""
    n = h0.size
    h = h0.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    start = time.time()
    T0, T_end = 0.02, 1e-7
    step0, step_end = 1.0, 1e-5
    it = 0
    while time.time() - start < time_budget:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + T_end * frac
        step_scale = step0 * (1 - frac) + step_end * frac

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        if eps <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        np.clip(h_new, 0.0, 1.0, out=h_new)
        c_new = compute_c5(h_new)

        if c_new < c or rng.random() < np.exp(-(c_new - c) / max(T, 1e-12)):
            h, c = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
                if best_c <= target_c:
                    return best_h, best_c

        if it % 5000 == 0:
            diff = (n / 2.0) - h.sum()
            if abs(diff) > 1e-12:
                idx = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c = h.copy(), c
    return best_h, best_c


# ---------- entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----- initial vector -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n = 512
        h0 = rng.random(n)
        h0 = enforce_sum(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- allocate time for SA -----
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > total_budget * 0.10:
        sa_time = remaining * 0.85
        h_sa, c_sa = continuous_sa(best_h, rng, sa_time, target_c=0.3808)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ----- final gradient tidy‑up (skip if already good enough) -----
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 2.0 and best_c > 0.3808:
        h_ref = gradient_refine(best_h.copy(), rng, step=0.01, max_iter=8)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----- sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
