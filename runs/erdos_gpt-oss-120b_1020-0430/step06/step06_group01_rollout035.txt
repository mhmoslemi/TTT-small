import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- gradient of worst shift ----------
def worst_shift_and_gradient(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    s = abs(idx)
    if s == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - s] -= h[s:]
        grad[s:] -= h[: n - s]
    return s, grad, corr

    """Aggressive hill‑climb, returns best (h, c)."""

    best_h = h0.copy()
    best_c = compute_c5(best_h)
    start = time.time()
    # parameters
    max_local_iters = 20000
    while time.time() - start < time_budget:
        # start from current best or a small perturbation
        if rng.random() < 0.7:
            h = best_h.copy()
        else:
            h = best_h.copy()
            for _ in range(3):
                i = rng.integers(n)
                delta = (rng.random() - 0.5) * 0.1
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)

        cur_c = compute_c5(h)
        it = 0
        while time.time() - start < time_budget and it < max_local_iters:
            s, grad, _ = worst_shift_and_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                break
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps < 1e-12:
                break
            # try full move
            h_new = h.copy()
            h_new[i] -= max_eps
            h_new[j] += max_eps
            new_c = compute_c5(h_new)
            if new_c <= cur_c:
                h, cur_c = h_new, new_c
                if new_c < best_c:
                    best_c, best_h = new_c, h_new.copy()
            else:
                # try half move
                eps = max_eps * 0.5
                h_new[i] = h[i] - eps
                h_new[j] = h[j] + eps
                new_c = compute_c5(h_new)
                if new_c <= cur_c:
                    h, cur_c = h_new, new_c
                    if new_c < best_c:
                        best_c, best_h = new_c, h_new.copy()
            it += 1
            # occasional random shake to escape plateaus
            if it % 500 == 0 and rng.random() < 0.2:
                for _ in range(2):
                    k = rng.integers(n)
                    delta = (rng.random() - 0.5) * 0.05
                    h[k] = np.clip(h[k] + delta, 0.0, 1.0)
                h = enforce_sum(h, rng)
        # end inner climb
    return best_h, best_c


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=2000):
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x


# ---------- fine gradient tidy‑up ----------
def gradient_refine(h, rng, step=0.015, max_iter=10):
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        s, grad, _ = worst_shift_and_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target size (even)
    n = 1024

    # ----- initial vector -----
    if "initial_h_values" in globals():
        init_old = np.array(initial_h_values, dtype=float)
        old_n = init_old.size
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init_old)
    else:
        h0 = rng.random(n)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------- aggressive hill‑climb ----------
    hc_time = total_budget * 0.68
    h_hc, c_hc = hill_climb(best_h, rng, hc_time - (time.time() - start))
    if c_hc < best_c:
        best_h, best_c = h_hc, c_hc

    # ---------- SLSQP refinement ----------
    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        beta_seq = (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0, 50000.0)
        for beta in beta_seq:
            if time.time() - start > total_budget * 0.98:
                break
            # allocate iterations proportionally to remaining time
            max_iter = int(min(3000, remaining * 1500 / (len(beta_seq) * (beta_seq[-1] / beta + 1))))
            max_iter = max(500, max_iter)
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
            remaining = total_budget - (time.time() - start)

    # ---------- final tidy‑up ----------
    if time.time() - start < total_budget * 0.995:
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        best_c = compute_c5(best_h)

    # tiny stochastic tweaks
    while time.time() - start < total_budget * 0.999:
        s, grad, _ = worst_shift_and_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # ---------- sanitise ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
