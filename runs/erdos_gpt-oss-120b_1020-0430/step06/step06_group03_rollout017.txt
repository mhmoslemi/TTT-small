import time, math, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.2):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def bound_from_S(S, n):
    return 1.0 - 2.0 * S.min() / n

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initial h -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 256
        h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    h = project_onto_box_sum(h, target)

    # ----- initial overlap matrix S ----------------------------------
    S = np.correlate(h, h, mode='full')[n - 1 :]  # S[k] = sum_i h[i]*h[i+k]
    cur_bound = bound_from_S(S, n)
    best_h = h.copy()
    best_bound = cur_bound

    # ----- simulated annealing ---------------------------------------
    T0 = 0.02
    T_final = 1e-6
    while time.time() - start < budget_s * 0.90:
        # linear cooling
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.90)
        T = max(T_final, T0 * (1.0 - frac))

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            continue
        # bias towards larger moves
        eps = max_eps * (rng.random() ** 0.5)

        # copy S and apply move
        S_new = S.copy()
        di = -eps
        dj = eps

        # i contribution
        S_new[: n - i] += di * h[i:]
        S_new[: i + 1] += di * h[i::-1]
        # j contribution
        S_new[: n - j] += dj * h[j:]
        S_new[: j + 1] += dj * h[j::-1]
        # cross term
        d = j - i
        if d > 0:
            S_new[d] += di * dj
        elif d < 0:
            S_new[-d] += di * dj
        # squares at k=0
        S_new[0] += di * di + dj * dj

        new_bound = bound_from_S(S_new, n)

        # acceptance test
        accept = False
        if new_bound < cur_bound:
            accept = True
        else:
            if T > 1e-12:
                delta = new_bound - cur_bound
                if rng.random() < math.exp(-delta / T):
                    accept = True
        if accept:
            h[i] -= eps
            h[j] += eps
            S = S_new
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best_h = h.copy()

    # ----- local refinement -------------------------------------------
    if time.time() - start < budget_s * 0.97:
        hc, cc = guided_hill_climb(best_h, rng,
                                   max_iters=3000, step_scale=0.1)
        if cc < best_bound:
            best_h, best_bound = hc.copy(), cc

    # final projection and exact evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
