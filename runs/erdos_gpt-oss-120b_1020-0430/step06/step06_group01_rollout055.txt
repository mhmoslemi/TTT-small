def line_search_hill_climb(h, rng, max_iters, eps_factors=[0.25,0.5,0.75,1.0]):
   n = h.size
   best_c = compute_c5(h)
   best_h = h.copy()
   for it in range(max_iters):
       grad = worst_gradient(h)
       i = int(np.argmax(grad))
       j = int(np.argmin(grad))
       if i == j:
           continue
       max_eps = min(h[i], 1.0 - h[j])
       if max_eps < 1e-12:
           continue
       # try candidates
       eps_list = [max_eps * f for f in eps_factors]
       improved = False
       for eps in eps_list:
           nh = h.copy()
           nh[i] -= eps
           nh[j] += eps
           c = compute_c5(nh)
           if c < best_c - 1e-12:
               best_c, best_h = c, nh.copy()
               h = nh
               improved = True
               break  # accept first improvement? maybe keep best of all candidates
       if not improved:
           # maybe do a random small transfer to explore
           rand_i = rng.integers(n)
           rand_j = rng.integers(n)
           if rand_i != rand_j:
               max_eps2 = min(h[rand_i], 1.0 - h[rand_j])
               if max_eps2 > 0:
                   eps2 = max_eps2 * rng.random() * 0.05
                   h[rand_i] -= eps2
                   h[rand_j] += eps2
       # optional early stopping
   return best_h, best_c

def binary_sa(n, rng, time_budget):
    start = time.time()
    # initial random binary vector with n/2 ones
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    ones_set = set(ones.tolist())
    zeros_set = set(range(n)) - ones_set
    # compute current c
    c_curr = compute_c5(h)
    best_c = c_curr
    best_h = h.copy()
    # annealing schedule: linear cooling Or exponential as prior.
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick random swap
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            # accept
            c_curr = c_new
            # update sets
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def line_search_hill_climb(h, rng, max_iters):
    n = h.size
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            continue
        # candidates: half and full
        eps_candidates = [max_eps * 0.5, max_eps]
        new_h = None
        new_c = None
        for eps in eps_candidates:
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            c = compute_c5(nh)
            if c < current_c - 1e-12:
                new_h = nh
                new_c = c
                break  # accept first improving candidate
        if new_h is not None:
            h = new_h
            current_c = new_c
            if current_c < best_c:
                best_c, best_h = current_c, h.copy()
        else:
            # small random perturbation to escape
            i2 = rng.integers(n)
            j2 = rng.integers(n)
            if i2 != j2:
                max_eps2 = min(h[i2], 1.0 - h[j2])
                if max_eps2 > 0:

                    nh[i2] -= eps2
                    nh[j2] += eps2
                    c2 = compute_c5(nh)
                    if c2 < current_c:
                        h = nh
                        current_c = c2
                        if c2 < best_c:
                            best_c, best_h = c2, h.copy()
    return best_h, best_c

            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- line‑search hill climbing ----------
def line_search_hill_climb(h: np.ndarray, rng: np.random.Generator,
                          max_iters: int) -> tuple[np.ndarray, float]:
    """Gradient‑directed hill climbing with simple line search."""
    n = h.size
    current_c = compute_c5(h)
    best_h, best_c = h.copy(), current_c

    for _ in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            continue

        # try half‑step then full‑step
        improved = False
        for eps in (max_eps * 0.5, max_eps):
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            c = compute_c5(nh)
            if c < current_c - 1e-12:
                h, current_c = nh, c
                if c < best_c:
                    best_c, best_h = c, nh.copy()
                improved = True
                break
        if not improved:
            # tiny random move to escape plateaus
            i2 = rng.integers(n)
            j2 = rng.integers(n)
            if i2 != j2:
                max_eps2 = min(h[i2], 1.0 - h[j2])
                if max_eps2 > 0:
                    eps2 = max_eps2 * rng.random() * 0.05
                    nh = h.copy()
                    nh[i2] -= eps2
                    nh[j2] += eps2
                    c2 = compute_c5(nh)
                    if c2 < current_c:
                        h, current_c = nh, c2
                        if c2 < best_c:
                            best_c, best_h = c2, nh.copy()
    return best_h, best_c


# ---------- binary simulated annealing ----------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float) -> tuple[np.ndarray, float]:
    """SA on binary (0/1) vectors with exactly n/2 ones."""
    start = time.time()
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    ones_arr = ones.copy()
    zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.02, 1e-4

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = ones_arr[rng.integers(len(ones_arr))]
        j = zeros_arr[rng.integers(len(zeros_arr))]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            # update index arrays
            idx_i = np.where(ones_arr == i)[0][0]
            idx_j = np.where(zeros_arr == j)[0][0]
            ones_arr[idx_i] = j
            zeros_arr[idx_j] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:   # early exit if already good enough
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target resolution
    n = 1024

    # ----- initialise from parent if present -----
    if 'initial_h_values' in globals():
        small = np.array(initial_h_values, dtype=float)
        x_small = np.linspace(0.0, 2.0, small.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_small, small)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        h0 = rng.random(n)
        h0 = enforce_sum(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------- binary simulated annealing (≈35% time) ----------
    binary_time = total_budget * 0.35
    h_bin, c_bin = binary_sa(n, rng, binary_time)
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin

    # ---------- continuous refinement (remaining time) ----------
    remaining = total_budget - (time.time() - start)
    if remaining > 0.1:
        # ensure best_h has correct size (it already does)
        while time.time() - start < total_budget * 0.99:
            time_left = total_budget - (time.time() - start)
            # number of iterations for this inner run
            max_iters = int(min(20000, max(500, time_left * 2000)))
            # starting point: either a perturbed best or fresh random
            if rng.random() < 0.5:
                h_start = best_h.copy()
                for _ in range(5):
                    i = rng.integers(n)
                    delta = (rng.random() - 0.5) * 0.2
                    h_start[i] = np.clip(h_start[i] + delta, 0.0, 1.0)
                h_start = enforce_sum(h_start, rng)
            else:
                h_start = rng.random(n)
                h_start = enforce_sum(h_start, rng)

            h_loc, c_loc = line_search_hill_climb(h_start, rng, max_iters)
            if c_loc < best_c:
                best_h, best_c = h_loc.copy(), c_loc

    # ---------- final tiny stochastic tweaks ----------
    final_time = total_budget - (time.time() - start)
    if final_time > 0.5:
        end_time = time.time() + final_time * 0.9
        while time.time() < end_time:
            grad = worst_gradient(best_h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 1e-12:
                eps = max_eps * rng.random() * 0.01
                nh = best_h.copy()
                nh[i] -= eps
                nh[j] += eps
                c_new = compute_c5(nh)
                if c_new < best_c:
                    best_h, best_c = nh, c_new

    # ---------- sanitisation ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
