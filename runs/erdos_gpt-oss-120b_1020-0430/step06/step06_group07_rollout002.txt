import time, numpy as np, math
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v, target):
    # Water-filling using binary search (like before)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n/2.0)

def worst_gradient(h):
    n = h.size
    # autocorrelation length n, starting at lag 0
    cross = np.correlate(h, h, mode='full')[n-1:]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[:n-k] -= h[k:]
        grad[k:] -= h[:n-k]
    return grad, k

def subgradient_momentum(h0, target, time_budget, rng):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros_like(h)
    t = 0
    mu = 0.9
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.6 / math.sqrt(t + 10)  # step size
        v = mu * v - alpha * grad
        h_new = project_box_sum(h + v, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            # optional reset momentum?
            v = np.zeros_like(h)
        h = h_new
        # early break if near target
        if best_c <= 0.3808:
            break
    return best_h, best_c

def guided_hill_climb(h0, rng, max_iters=20000, step_scale=0.6):
    # same algorithm as previous but longer
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        # greedy transfer
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # optional random restart if stagnation
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c

def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = beta * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize(h0, beta, rng, max_iter=1500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    cons = {"type": "eq", "fun": cons_fun}
    def obj(x):
        return softmax_obj(x, beta)
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def upsample(h, rng):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # Determine base resolution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n0 / 2.0)
    else:
        n0 = 256
        h0 = random_feasible(n0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase A: random restarts with short subgradient
    time_elapsed = time.time() - start
    remaining = total_budget - time_elapsed
    restart_budget = remaining * 0.20
    if restart_budget > 10.0:
        per_restart = 60.0  # each restart tries ~60 sec subgradient
        while time.time() - start < start + restart_budget - per_restart:
            hi = random_feasible(n0, rng)
            hi_best, ci = subgradient_momentum(hi, n0/2.0, per_restart, rng)
            if ci < best_c:
                best_h, best_c = hi_best, ci

    # Phase B: long subgradient with momentum on best
    time_elapsed = time.time() - start
    remaining = total_budget - time_elapsed
    if remaining > 50.0:
        sub_budget = remaining * 0.45
        h_sg, c_sg = subgradient_momentum(best_h, best_h.size/2.0, sub_budget, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # Phase C: hill climb
    time_elapsed = time.time() - start
    remaining = total_budget - time_elapsed
    if remaining > 30.0:
        max_iters = int(min(20000, remaining * 500))
        h_hc, c_hc = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # Phase D: SLSQP with increasing beta
    time_elapsed = time.time() - start
    remaining = total_budget - time_elapsed
    if remaining > 20.0:
        # sequence of betas; start low, end high
        betas = [10., 30., 80., 200., 800., 3000., 8000., 20000.]
        for beta in betas:
            if time.time() - start > total_budget * 0.98:
                break
            # allocate per beta proportionally to remaining
            per_beta = max(5.0, remaining * 0.08)
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=int(per_beta*10))
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
            # update remaining
            remaining = total_budget - (time.time() - start)
            if remaining <= 5.0:
                break

    # Phase E: upsample to 1024 and final polish
    time_elapsed = time.time() - start
    remaining = total_budget - time_elapsed
    if remaining > 15.0 and best_h.size * 2 <= 1024:
        h_up = upsample(best_h, rng)
        # subgradient on finer grid
        sub_budget = remaining * 0.5
        h_up_sg, c_up_sg = subgradient_momentum(h_up, h_up.size/2.0, sub_budget, rng)
        if c_up_sg < best_c:
            best_h, best_c = h_up_sg, c_up_sg
        time_elapsed = time.time() - start
        remaining = total_budget - time_elapsed
        if remaining > 8.0:
            max_iters = int(min(30000, remaining * 400))
            h_up_hc, c_up_hc = guided_hill_climb(best_h, rng,
                                                 max_iters=max_iters,
                                                 step_scale=0.6)
            if c_up_hc < best_c:
                best_h, best_c = h_up_hc, c_up_hc

    # Final project
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size/2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time, math, numpy as np
from scipy.optimize import minimize, LinearConstraint

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def worst_gradient(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def subgradient_momentum(h0: np.ndarray, target: float, budget: float,
                         rng: np.random.Generator):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros_like(h)
    mu = 0.9
    t = 0
    while time.time() - start < budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.6 / math.sqrt(t + 10)
        v = mu * v - alpha * grad
        h_new = project_box_sum(h + v, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            v.fill(0.0)
        h = h_new
        if best_c <= 0.3808:
            break
    return best_h, best_c

def guided_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 20000, step_scale: float = 0.7):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c

def slsqp_obj(x: np.ndarray, beta: float) -> float:
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    lin_con = LinearConstraint(np.ones((1, n)), [target], [target])
    res = minimize(slsqp_obj, h0, args=(beta,), method="SLSQP",
                   bounds=bounds, constraints=[lin_con],
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)
