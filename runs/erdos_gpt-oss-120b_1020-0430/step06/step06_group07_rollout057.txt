def compute_c5_fft(h):
    n = h.size
    L = 2 * n
    h_pad = np.concatenate([h, np.zeros(n)])
    ih_pad = np.concatenate([1 - h, np.zeros(n)])
    # FFT
    H = np.fft.rfft(h_pad)
    I = np.fft.rfft(ih_pad)
    corr = np.fft.irfft(H * I, n=L)  # real result length L
    # only first 2*n-1 values are relevant; last is zero due to padding
    max_val = np.max(corr[:2*n-1])
    return max_val * (2.0 / n)

def hill_climb(h0, target, rng, time_budget, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    # compute worst shift index each iteration using same cross method
    n = h.size
    while True:
        if time.time() - start_time > time_budget:
            break
        # get worst shift gradient
        cross = np.correlate(h, h, mode="full")[n - 1:]
        k = int(np.argmin(cross))
        grad = np.ones_like(h)
        if k != 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        # propose perturb guided by gradient:
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        # ensure i != j and possible transfer
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0:
            eps = rng.random() * max_eps * 0.5
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            h_new = h.copy()
        c_new = compute_c5_fft(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # Determine n: if initial_h_values exists and len *2 <= 1024, upsample to that. Else n=1024.
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        n_target = min(1024, n0 * 2)  # upsample to at most 1024
        if n0 != n_target:
            # linear upsample
            x_old = np.linspace(0., 2., n0, endpoint=False)
            x_new = np.linspace(0., 2., n_target, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        # Clip & project to sum target
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0., 1.)
        h0 = project_box_sum(h0, target)

        target = n / 2.0
        h0 = random_feasible(n, rng)  # uses proj
    best_h = h0
    best_c = compute_c5_fft(best_h)

    # Phase 1: subgradient
    remaining = budget_s - (time.time() - start_time)

def compute_c5_fft(h):
    n = h.size
    L = 2 * n
    h_pad = np.concatenate([h, np.zeros(n)])
    ih_rev = np.concatenate([(1 - h)[::-1], np.zeros(n)])
    H = np.fft.rfft(h_pad)
    I = np.fft.rfft(ih_rev)
    corr = np.fft.irfft(H * I, n=L)
    # length L = 2*n, first 2*n-1 are valid
    max_val = np.max(corr[:2*n-1])
    return max_val * (2.0 / n)

def subgradient_optimize(h0, target, time_budget, rng, start_time):
    n = h0.size
    h = h0.copy()
    # initial C5
    cross0 = np.correlate(h, h, mode="full")[n - 1:]
    best_k = int(np.argmin(cross0))
    best_c = 1.0 - cross0[best_k] * (2.0 / n)
    best_h = h.copy()
    t = 0
    gamma = 1.0
    no_improve = 0
    while True:
        if time.time() - start_time > time_budget:
            break
        t += 1
        cross = np.correlate(h, h, mode="full")[n - 1:]
        k = int(np.argmin(cross))
        c = 1.0 - cross[k] * (2.0 / n)
        # update best if needed
        if c < best_c:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # Gradient
        grad = np.ones_like(h)
        if k != 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        alpha = gamma / np.sqrt(t + 10)
        h = project_box_sum(h - alpha * grad, target)
        # occasional random perturb
        if t % 5000 == 0:
            h = project_box_sum(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
        # early exit if reached target bound
        if best_c <= 0.3808:
            break
        # restart if stuck
        if no_improve > 50000:
            h = project_box_sum(best_h + rng.normal(scale=0.05, size=n), target)
            no_improve = 0
    return best_h, best_c

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + ((lo + hi) * 0.5), 0.0, 1.0)

def upsample_h(h, n_target):
    n0 = h.size
    if n0 == n_target:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

    L = 2 * n
    # reverse (1-h) for linear correlation
    rev = (1.0 - h)[::-1]
    h_pad = np.concatenate([h, np.zeros(n)])
    rev_pad = np.concatenate([rev, np.zeros(n)])
    H = np.fft.rfft(h_pad)
    R = np.fft.rfft(rev_pad)
    corr = np.fft.irfft(H * R, n=L)
    max_val = np.max(corr[: 2 * n - 1])
    return max_val * (2.0 / n)


# ----------------------------------------------------------------------
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


# ----------------------------------------------------------------------
def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


# ----------------------------------------------------------------------
def upsample_h(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear up‑sampling of a step function on [0,2]."""
    n0 = h.size
    if n0 == n_target:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_new, x_old, h)


# ----------------------------------------------------------------------
def subgradient_optimize(
    h0: np.ndarray, target: float, end_time: float, rng: np.random.Generator
):
    """Projected sub‑gradient descent on max‑overlap."""
    n = h0.size
    h = h0.copy()
    # initial C5 from worst shift
    cross0 = np.correlate(h, h, mode="full")[n - 1 :]
    best_k = int(np.argmin(cross0))
    best_c = 1.0 - cross0[best_k] * (2.0 / n)
    best_h = h.copy()

    t = 0
    gamma = 1.0
    no_improve = 0
    while True:
        if time.time() > end_time:
            break
        t += 1

        cross = np.correlate(h, h, mode="full")[n - 1 :]
        k = int(np.argmin(cross))
        c = 1.0 - cross[k] * (2.0 / n)

        if c < best_c:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradient of the worst‑shift overlap
        grad = np.ones_like(h)
        if k != 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        alpha = gamma / np.sqrt(t + 10)
        h = project_box_sum(h - alpha * grad, target)

        # occasional random jitter to escape stagnation
        if t % 5000 == 0:
            h = project_box_sum(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

        if best_c <= 0.3808:
            break

        if no_improve > 50000:
            h = project_box_sum(best_h + rng.normal(scale=0.05, size=n), target)
            no_improve = 0

    return best_h, best_c


# ----------------------------------------------------------------------
def hill_climb_optimize(
    h0: np.ndarray, target: float, end_time: float, rng: np.random.Generator
):
    """Guided hill‑climb using the worst‑shift gradient."""
    n = h0.size
    h = h0.copy()
    cross0 = np.correlate(h, h, mode="full")[n - 1 :]
    best_k = int(np.argmin(cross0))
    best_c = 1.0 - cross0[best_k] * (2.0 / n)
    best_h = h.copy()
    no_improve = 0

    while True:
        if time.time() > end_time:
            break

        cross = np.correlate(h, h, mode="full")[n - 1 :]
        k = int(np.argmin(cross))
        grad = np.ones_like(h)
        if k != 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0:
            eps = rng.random() * max_eps * 0.5
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            h_new = h.copy()

        cross_new = np.correlate(h_new, h_new, mode="full")[n - 1 :]
        k_new = int(np.argmin(cross_new))
        c_new = 1.0 - cross_new[k_new] * (2.0 / n)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1

        if best_c <= 0.3808:
            break

        if no_improve > 20000:
            h = project_box_sum(best_h + rng.normal(scale=0.05, size=n), target)
            no_improve = 0

    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        n_target = min(1024, n0 * 2)
        h0 = upsample_h(h0, n_target)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, target)
    else:
        n = 1024
        target = n / 2.0
        h0 = random_feasible(n, rng)

    best_h = h0
    best_c = compute_c5_fft(best_h)

    # ------------------------------------------------------------------
    # Phase 1: long projected sub‑gradient descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15.0:
        sub_end = start_time + remaining * 0.70
        h_sg, c_sg = subgradient_optimize(best_h, target, sub_end, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # ------------------------------------------------------------------
    # Phase 2: guided hill‑climb for final polishing
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        hc_end = start_time + remaining
        h_hc, c_hc = hill_climb_optimize(best_h, target, hc_end, rng)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ------------------------------------------------------------------
    # final sanitisation and exact C5 evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
