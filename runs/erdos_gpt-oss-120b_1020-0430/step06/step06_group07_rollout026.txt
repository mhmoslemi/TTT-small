def simulated_annealing_binary(h0, time_budget, rng, target_c=None):
    start = time.time()
    n = h0.size
    cur_h = h0.astype(np.float64)
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # temperature schedule: choose initial temperature as cur_c (or a fraction of it). We'll compute steps based on budget.
    # We'll compute steps = max(10, int((time_budget) * 200)) perhaps.
    steps_total = int(time_budget * 200)  # approx 200 iter per second
    if steps_total < 10:
        steps_total = 10

    # linear cooling from T0 = cur_c to T_end = cur_c*1e-4 maybe.
    T0 = cur_c
    T_end = cur_c * 1e-4 + 1e-8

    for step in range(steps_total):
        # early exit if time
        if time.time() - start >= time_budget:
            break
        t = step / steps_total
        T = T0 * (1 - t) + T_end * t
        # propose swap
        # pick i where cur_h[i]==1
        # naive loop
        i = rng.integers(n)
        while cur_h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] > 0.5:
            j = rng.integers(n)
        # swap temporarily
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5_fft(cur_h)
        # decide acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
                if target_c is not None and best_c <= target_c:
                    break
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c

def softmax_obj(x):
    n = x.size
    size = n * 2
    H = np.fft.rfft(x, size)
    G = np.fft.rfft(1 - x, size)
    corr = np.fft.irfft(H * G, size)[:2 * n - 1]
    bc = CURRENT_BETA * corr
    m = bc.max()
    # compute log-sum-exp in a stable way:
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    # adjust via simple loop
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# ---------- C5 evaluation ----------
def compute_c5_fft(h: np.ndarray) -> float:
    n = h.shape[0]
    size = n * 2
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    conv = np.fft.irfft(H * G, size)[: 2 * n - 1]
    return conv.max() * (2.0 / n)

# ---------- simulated annealing for binary vectors ----------
def simulated_annealing_binary(h0: np.ndarray, time_budget: float,
                               rng: np.random.Generator,
                               target_c: float = None) -> tuple:
    start = time.time()
    n = h0.size
    cur_h = h0.astype(np.float64)
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # number of iterations, roughly 500 per second of budget
    steps_total = max(10, int(time_budget * 500))

    T0 = cur_c
    T_end = cur_c * 1e-4 + 1e-8

    for step in range(steps_total):
        if time.time() - start >= time_budget:
            break
        t = step / steps_total
        T = T0 * (1 - t) + T_end * t

        # propose a swap 1<->0
        i = rng.integers(n)
        while cur_h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] > 0.5:
            j = rng.integers(n)

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5_fft(cur_h)

        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
                if target_c is not None and best_c <= target_c:
                    break
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_c

# ---------- SLSQP soft‑max optimizer ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    n = x.size
    size = n * 2
    H = np.fft.rfft(x, size)
    G = np.fft.rfft(1.0 - x, size)
    corr = np.fft.irfft(H * G, size)[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = bc.max()
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s
    target_c = 0.3808

    # grid size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        if n % 2:
            n -= 1
    else:
        n = 512

    # ----- initial binary vector -----
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=np.float64)
        idx = np.argsort(-init_arr)
        h_bin0 = np.zeros(n, dtype=np.float64)
        h_bin0[idx[: n // 2]] = 1.0
    else:
        perm = rng.permutation(n)
        h_bin0 = np.zeros(n, dtype=np.float64)
        h_bin0[perm[: n // 2]] = 1.0

    best_h, best_c = h_bin0, compute_c5_fft(h_bin0)

    # ----- phase 1: binary simulated annealing -----
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 5.0:
        sa_time = remaining * 0.60
        h_sa, c_sa = simulated_annealing_binary(best_h, sa_time, rng, target_c=target_c)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ----- phase 2: SLSQP refinement (if time) -----
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 5.0 and best_c > target_c:
        beta_seq = [100.0, 500.0, 2000.0, 8000.0]
        for beta in beta_seq:
            if time.time() - start >= total_budget * 0.99:
                break
            max_iter = min(200, int(remaining * 5))
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            c_opt = compute_c5_fft(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                if best_c <= target_c:
                    break
            elapsed = time.time() - start
            remaining = total_budget - elapsed

    # ----- phase 3: up‑sample and repeat SA on finer grid -----
    if best_h is not None and n * 2 <= 1024:
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        if remaining > 5.0:
            h_up = np.repeat(best_h, 2)
            n_up = h_up.size
            h_up = project_box_sum(h_up, n_up / 2.0)
            h_up_sa, c_up_sa = simulated_annealing_binary(
                h_up, remaining, rng, target_c=target_c
            )
            if c_up_sa < best_c:
                best_h, best_c = h_up_sa, c_up_sa

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
