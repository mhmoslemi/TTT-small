Pseudo:

def softmax_gradient(h, beta):
    n = h.size
    corr = np.convolve(h, (1-h)[::-1], mode='full')
    Omax = corr.max()
    exp_term = np.exp(beta * (corr - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk_center = w[n-1]
    grad += wk_center * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n-1 + k] + w[n-1 - k]
        if wk == 0.0:
            continue
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

Note that np.convolve returns length 2n-1, same as correlate.
