def repair_balance(h):
    s = h.sum()
    target = N // 2
    if s > target:
        # need to turn off s - target ones
        idx = np.where(h == 1)[0]
        # random choice
        flip_idx = np.random.choice(idx, size=s-target, replace=False)
        h[flip_idx] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        flip_idx = np.random.choice(idx, size=target - s, replace=False)
        h[flip_idx] = 1
    return h

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP soft‑max
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with N points."""
    ha = np.asarray(h, dtype=float)
    corr = np.correlate(ha, 1.0 - ha, mode="full")
    return np.max(corr) * (2.0 / ha.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def repair_balance(arr: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    """Flip the minimal number of bits to make sum(arr) == target."""
    s = int(arr.sum())
    diff = target - s
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        idx = rng.choice(zeros, size=diff, replace=False)
        arr[idx] = 1
    elif diff < 0:
        ones = np.where(arr == 1)[0]
        idx = rng.choice(ones, size=-diff, replace=False)
        arr[idx] = 0
    return arr


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------- problem size -------------------
    N = 1024                     # number of sample points
    target = N // 2

    # ------------------- initial population -------------------
    popsize = 150
    population = np.empty((popsize, N), dtype=np.uint8)

    for i in range(popsize):
        arr = np.zeros(N, dtype=np.uint8)
        arr[:target] = 1
        rng.shuffle(arr)
        population[i] = arr

    # possibility: seed with up‑sampled parent if provided
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != N:
            # linear interpolation to new grid
            x_old = np.linspace(0, 1, parent.size, endpoint=False)
            x_new = np.linspace(0, 1, N, endpoint=False)
            parent = np.interp(x_new, x_old, parent)
        parent_bin = (parent > 0.5).astype(np.uint8)
        parent_bin = repair_balance(parent_bin, target, rng)
        population[rng.integers(0, popsize)] = parent_bin

    # ------------------- evaluation of initial pop -------------------
    c5_vals = np.empty(popsize)
    best_idx = 0
    best_c5 = np.inf
    for i in range(popsize):
        c = compute_c5(population[i])
        c5_vals[i] = c
        if c < best_c5:
            best_c5 = c
            best_idx = i
    best_h = population[best_idx].copy()

    # ------------------- GA parameters -------------------
    tournament_size = 5
    crossover_prob = 0.7
    mutation_rate = 0.02

    # leave time for final refinement
    final_refine_time = 45.0

    # ------------------- GA main loop -------------------
    while time.time() - start < budget_s - final_refine_time:
        new_pop = np.empty_like(population)
        # elitism – keep current best
        new_pop[0] = best_h
        for i in range(1, popsize):
            # tournament selection for parent
            cand = rng.choice(popsize, size=tournament_size, replace=False)
            parent_idx = cand[np.argmin(c5_vals[cand])]
            parent = population[parent_idx]
            child = parent.copy()

            # crossover
            if rng.random() < crossover_prob:
                other_cand = rng.choice(popsize, size=tournament_size, replace=False)
                other_idx = other_cand[np.argmin(c5_vals[other_cand])]
                other = population[other_idx]
                mask = rng.random(N) < 0.5
                child[mask] = other[mask]

            # mutation
            mut_mask = rng.random(N) < mutation_rate
            child[mut_mask] = 1 - child[mut_mask]

            # repair to keep exactly target ones
            child = repair_balance(child, target, rng)

            new_pop[i] = child

        # evaluate new population
        new_c5 = np.empty(popsize)
        for i in range(popsize):
            c = compute_c5(new_pop[i])
            new_c5[i] = c
            if c < best_c5:
                best_c5 = c
                best_h = new_pop[i].copy()

        population = new_pop
        c5_vals = new_c5

    # ------------------- final SLSQP refinement -------------------
    best_h_float = best_h.astype(float)
    beta_seq = (5, 25, 100, 500, 2000, 5000, 20000, 50000)
    for beta in beta_seq:
        if time.time() - start > budget_s - 5.0:
            break
        best_h_float = slsqp_optimize(best_h_float, beta, rng, max_iter=400)
        best_c5 = compute_c5(best_h_float)

    # final repair and output
    best_h_float = np.clip(best_h_float, 0.0, 1.0)
    best_h_float = repair_balance(best_h_float.astype(np.uint8), target, rng).astype(float)
    final_c5 = compute_c5(best_h_float)

    return best_h_float.tolist(), float(final_c5), int(N)
