def softmax_gradient(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1-h, mode='full')
    # stable weights
    maxc = np.max(beta * corr)
    expw = np.exp(beta * corr - maxc)
    w = expw / np.sum(expw)
    # compute weighted sum of grad_r
    A = np.zeros_like(h)
    B = np.zeros_like(h)
    # r from -(n-1) to (n-1)
    # index in corr/w: r_idx = r + (n-1)
    for r_idx, r in enumerate(range(-(n-1), n)):
        weight = w[r_idx]
        if r >= 0:
            # shift right: h[i+r] for i in 0..n-1-r
            # slice h[r:] -> prep for A[0:n-r] add weight* h[r:]
            A[:n-r] += weight * h[r:]
            # shift left: h[i-r] == shift right with -r
            B[r:] += weight * h[:n-r]
        else:
            # r negative: shift right: h[i+r] meaning h[i - |r|]
            # for i >= |r|
            A[-r:] += weight * h[:n+r]  # since r negative: -r = |r|
            # shift left: h[i-r] = h[i + |r|]
            B[:n+r] += weight * h[-r:]
    # gradient of f_beta = (2/n) * (1 - A - B)
    grad = (2.0 / n) * (1.0 - A - B)
    return grad, corr
```

- shift_right(h,r) => A contribution obviously: we add weight * shift_right to A[i] for i=0..n-r-1.

import time

# ----------------------------------------------------------------------
# helper functions
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def mass_transfer_move(
    h: np.ndarray,
    grad: np.ndarray,
    alpha: float,
    target: float,
    rng: np.random.Generator,
) -> np.ndarray:

def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def softmax_gradient(h: np.ndarray, beta: float):
    """
    Gradient of the smooth max: f_beta(h) = (1/beta) log sum_k exp(beta*corr_k)
    scaled so that the final C5 = (2/n)*max(corr).
    Returns (grad, corr) where grad has the same scaling as compute_c5.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    maxv = np.max(scaled)
    expw = np.exp(scaled - maxv)
    w = expw / np.sum(expw)

    A = np.zeros_like(h)
    B = np.zeros_like(h)

    # iterate over shifts r = -(n-1) .. (n-1)
    for idx, r in enumerate(range(-(n - 1), n)):
        weight = w[idx]
        if r >= 0:
            A[: n - r] += weight * h[r:]
            B[r:] += weight * h[: n - r]
        else:
            d = -r
            A[d:] += weight * h[: n - d]
            B[: n - d] += weight * h[d:]

    grad = (2.0 / n) * (1.0 - A - B)
    return grad, corr

            c_new = compute_c5(h_new)
            if c_new < c - 1e-12:
                h, c = h_new, c_new
                if c < best_c:
                    best_c, best_h = c, h.copy()
                improved = True
                break
        if not improved:
            h2 = random_perturb(h, rng, scale=0.2)
            c2 = compute_c5(h2)
            if c2 < c - 1e-12:
                h, c = h2, c2
                if c < best_c:
                    best_c, best_h = c, h.copy()
        refine_iters += 1
        if best_c < 0.38078:
            break

    # ------------------------------------------------------------------
    # final feasibility enforcement and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
