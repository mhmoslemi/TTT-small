def gradient_refine(h, rng, step=0.01, max_iter=5):
    N = h.size
    for it in range(max_iter):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr) - (N - 1)

import time
import numpy as np
from scipy.optimize import minimize

# global state for current problem size and beta
CURRENT_N = 0
CURRENT_L = 0
CURRENT_BETA = 0.0


def set_current(N, beta):
    """Set globals for current dimension and beta, and pre‑compute FFT length."""
    global CURRENT_N, CURRENT_L, CURRENT_BETA
    CURRENT_N = N
    CURRENT_BETA = beta
    # length for linear convolution (next power of two >= 2N-1)
    CURRENT_L = 1 << ((2 * N - 1).bit_length())


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    N = h.size
    target = N / 2.0
    diff = target - h.sum()
    # simple random walk repair
    while abs(diff) > 1e-12:
        i = rng.integers(0, N)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear cross‑correlation of h with (1‑h) using FFT."""
    A = np.fft.fft(h, n=CURRENT_L)
    B = np.fft.fft(1.0 - h, n=CURRENT_L)
    corr = np.fft.ifft(A * np.conj(B)).real[: 2 * CURRENT_N - 1]
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 value for vector h."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint sum(x) = N/2."""
    return np.sum(x) - CURRENT_N / 2.0


def softmax_obj(x: np.ndarray) -> float:
    """Soft‑max approximation of max correlation (objective for SLSQP)."""
    beta = CURRENT_BETA
    corr = compute_corr_fft(x)
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    w = np.exp(shifted)
    wsum = np.sum(w)
    return max_corr + np.log(wsum) / beta


def softmax_grad(x: np.ndarray) -> np.ndarray:
    """Exact gradient of the soft‑max objective."""
    beta = CURRENT_BETA
    N = CURRENT_N
    L = CURRENT_L

    corr = compute_corr_fft(x)
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    w = np.exp(shifted)
    wsum = np.sum(w)
    w_norm = w / wsum

    # convolution w_norm * x
    A = np.fft.fft(w_norm, n=L)
    B = np.fft.fft(x, n=L)
    conv = np.fft.ifft(A * B).real[: 2 * N - 1]

    # convolution of reversed w_norm with x
    w_rev = w_norm[::-1]
    A_rev = np.fft.fft(w_rev, n=L)
    conv_rev = np.fft.ifft(A_rev * B).real[: 2 * N - 1]

    start = N - 1
    term3 = conv[start : start + N]       # Σ w_k * h_{j‑k}
    term2 = conv_rev[start : start + N]   # Σ w_k * h_{j+k}

    grad = 1.0 - term2 - term3
    return grad


def slsqp_optimize_grad(h0: np.ndarray, beta: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000) -> np.ndarray:
    """SLSQP minimization of soft‑max C5 with analytic gradient."""
    set_current(h0.size, beta)
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        jac=softmax_grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst–shift overlap O_k."""
    N = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.zeros_like(h)
        grad[: N - k] = 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 15) -> np.ndarray:
    """Fine‑tune using the exact worst‑shift gradient."""
    N = h.size
    for it in range(max_iter):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr) - (N - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: N - k] = 1.0 - h[k:]
            g[k:] -= h[: N - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = N / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, N)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise ----- (prefer a doubled version of the parent if present)

        h_small = enforce_sum(h_small, rng)

        N = h_small.size * repeat
        h0 = np.repeat(h_small, repeat)
    else:
        N = 1024
        h0 = np.zeros(N)
        idx = rng.choice(N, N // 2, replace=False)
        h0[idx] = 1.0
    h0 = enforce_sum(h0, rng)

    set_current(N, 0.0)          # initialise FFT parameters
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- SLSQP passes with increasing beta -----
    beta_seq = (10.0, 100.0, 1000.0, 5000.0, 20000.0, 100000.0)
    for beta in beta_seq:
        if time.time() - start > budget_s * 0.90:
            break
        best_h = slsqp_optimize_grad(best_h, beta, rng, max_iter=2000)
        best_c = compute_c5(best_h)

    # ----- gradient polishing -----
    if time.time() - start < budget_s * 0.95:
        refined = gradient_refine(best_h, rng, step=0.02, max_iter=15)
        rc = compute_c5(refined)
        if rc < best_c:
            best_h, best_c = refined, rc

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    set_current(N, 0.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
