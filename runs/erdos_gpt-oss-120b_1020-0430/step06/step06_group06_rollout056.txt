def compute_c5_fft(h):
    N = h.size
    L = 2*N - 1
    H = np.fft.fft(h, L)
    Q = np.fft.fft(1.0 - h, L)
    C = np.fft.ifft(H * np.conj(Q)).real
    max_corr = C.max()

def proj_sum_bounds(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # if sum too high, reduce excess from non-zero entries
    # Use water-filling:
    # Shift = (target - sum(v)) / N
    # adjust iteratively
    for _ in range(10):
        diff = target - v.sum()
        if abs(diff) < 1e-10:
            break
        # adjust free variables (those not at bounds)
        free = (v > 0.0 + 1e-12) & (v < 1.0 - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, cannot adjust -> just clip and break
            break
        delta = diff / n_free
        v[free] += delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
    # final adjustment for residual diff: add to first free variable
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # find index not at bound
        for i in range(len(v)):
            if v[i] > 0.0 + 1e-12 and v[i] < 1.0 - 1e-12:
                v[i] += diff
                break
    # ensure bounds
    v = np.clip(v, 0.0, 1.0)
    return v

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------

def proj_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤v≤1, sum(v)=target} by iterative water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        free = (v > 1e-12) & (v < 1.0 - 1e-12)
        n_free = int(free.sum())
        if n_free == 0:
            break
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # fix any tiny residual on a non‑bound entry
        for i in range(v.size):
            if 0.0 < v[i] < 1.0:
                v[i] = np.clip(v[i] + diff, 0.0, 1.0)
                break
    return v


def corr_full(h: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear correlation of h with (1‑h) using zero‑padded FFT."""
    N = h.shape[0]
    H = np.fft.fft(h, fft_len)
    Q = np.fft.fft(1.0 - h, fft_len)
    C = np.fft.ifft(H * np.conj(Q)).real[: 2 * N - 1]
    return C


def c5_value(h: np.ndarray, fft_len: int) -> float:
    """Exact C₅ for a step function."""
    N = h.shape[0]
    C = corr_full(h, fft_len)
    return C.max() * (2.0 / N)


def softmax_weights(C: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max of correlation values."""
    m = C.max()
    exp = np.exp(beta * (C - m))
    w = exp / exp.sum()
    return w


def softmax_gradient(h: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of the soft‑max objective (vectorised via FFT)."""
    N = h.shape[0]
    w_rev = w[::-1]
    conv1 = np.fft.ifft(
        np.fft.fft(w_rev, fft_len) * np.fft.fft(1.0 - h, fft_len)
    ).real[:N]
    conv2 = np.fft.ifft(
        np.fft.fft(w, fft_len) * np.fft.fft(h, fft_len)
    ).real[:N]
    return conv1 - conv2


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (sub‑gradient)."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    g = np.ones(N)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        g[: N - k] -= h[k:]
        g[k:] -= h[: N - k]
    return g


def refine_worst(h: np.ndarray, target: float, steps: int = 30, step_size: float = 0.02) -> np.ndarray:
    """Polish a solution using the worst‑shift sub‑gradient."""
    for _ in range(steps):
        g = worst_shift_gradient(h)
        h = h - step_size * g
        h = proj_sum_bounds(h, target)
    return h


def single_restart(rng: np.random.Generator, N: int, target: float, fft_len: int,
                  best_h: np.ndarray, best_c: float, deadline: float):
    """One stochastic optimisation run; returns possibly improved best."""
    # random start
    h = proj_sum_bounds(rng.random(N), target)

    # continuation schedule (β increasing, step decreasing)
    beta_seq = [5, 30, 150, 800, 4000, 20000]
    step_seq = [0.8, 0.5, 0.3, 0.15, 0.08, 0.04]

    for beta, step in zip(beta_seq, step_seq):
        # a few iterations per β; more when β is large
        its = 50 if beta <= 800 else 80
        for _ in range(its):
            if time.time() > deadline:
                return best_h, best_c, True
            C = corr_full(h, fft_len)
            w = softmax_weights(C, beta)
            g = softmax_gradient(h, w, fft_len)
            h = h - step * g
            h = proj_sum_bounds(h, target)

            # occasional exact evaluation
            if _ % 7 == 0:
                c_val = C.max() * (2.0 / N)
                if c_val < best_c:
                    best_c = c_val
                    best_h = h.copy()
    # final evaluation for this run
    C = corr_full(h, fft_len)
    c_val = C.max() * (2.0 / N)
    if c_val < best_c:
        best_c = c_val
        best_h = h.copy()
    return best_h, best_c, False


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # problem size – 1024 gives finer resolution while staying fast
    N = 1024
    target_sum = N / 2.0
    fft_len = 1 << ((3 * N - 2).bit_length())      # >= 3N‑2 for all convolutions

    # start from parent if available
    best_h = None
    best_c = float("inf")
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != N:
            rep = N // init.size
            extra = N % init.size
            init = np.repeat(init, rep)
            if extra:
                init = np.concatenate([init, init[:extra]])
        init = proj_sum_bounds(np.clip(init, 0.0, 1.0), target_sum)
        best_h = init
        best_c = c5_value(init, fft_len)

    # main stochastic search (many cheap restarts)
    deadline = start + budget_s * 0.97
    while time.time() < deadline:
        best_h, best_c, stopped = single_restart(
            rng, N, target_sum, fft_len, best_h, best_c, deadline
        )
        if stopped:
            break

    # final polishing with the true worst‑shift gradient
    if best_h is None:
        # fallback deterministic construction
        best_h = np.zeros(N)
        best_h[: N // 2] = 1.0
        best_h = proj_sum_bounds(best_h, target_sum)

    best_h = refine_worst(best_h, target_sum, steps=40, step_size=0.02)
    final_c = c5_value(best_h, fft_len)

    # final feasibility safeguard
    best_h = proj_sum_bounds(best_h, target_sum)

    return best_h.tolist(), float(final_c), int(N)
