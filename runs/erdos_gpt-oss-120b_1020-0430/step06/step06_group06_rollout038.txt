```python
def compute_c5_fft(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    rev = (1 - h)[::-1]
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)
    # conv length L, full conv first 2*n-1 entries
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

def full_corr_fft(h):
    n = h.size
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(H * R, n=L)
    return conv[:2*n-1]

def gradient_refine(h, rng, step=0.015, max_iter=5):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        k, _, _ = worst_shift_corr(h)
        # compute gradient for O_k
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            if k > 0:
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
            else:
                k_abs = -k
                # O_{-k}
                g[k_abs:] += 1.0 - h[: n - k_abs]
                g[: n - k_abs] -= h[k_abs:]
        # update
        h = h - (step * (0.5 ** it)) * g
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # enforce sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility ----------

def next_pow_two(x: int) -> int:
    return 1 << (x - 1).bit_length()

def full_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.size
    L = next_pow_two(2 * n - 1)
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * R, n=L)
    return conv[: 2 * n - 1]

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    n = h.size
    return np.max(full_corr_fft(h)) * (2.0 / n)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- correlation helpers ----------

def worst_shift_corr(h: np.ndarray):
    """Return (k, correlation array, index_of_max)."""
    corr = full_corr_fft(h)
    idx = int(np.argmax(corr))
    k = idx - (h.size - 1)
    return k, corr, idx

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the current worst–shift overlap."""
    n = h.size
    k, _, _ = worst_shift_corr(h)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros_like(h)

        g[k:] -= h[: n - k]

        k = -k
        g[k:] += 1.0 - h[: n - k]
        g[: n - k] -= h[k:]
    return g

# ---------- SLSQP with soft‑max ----------

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = full_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
