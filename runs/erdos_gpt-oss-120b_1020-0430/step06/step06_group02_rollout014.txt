def project_sum(h, target):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling
    for _ in range(10):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Adjust uniformly
        h += (target - s) / len(h)
        # Clip again
        h = np.clip(h, 0.0, 1.0)
    # final fix: adjust any residual diff by distributing to entries not at bounds
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute to entries with room
        mask = ( (diff > 0) & (h < 1.0) ) | ( (diff < 0) & (h > 0.0) )
        idx = np.where(mask)[0]
        if idx.size > 0:
            # add diff equally among idx (maybe fractional)
            h[idx] += diff / idx.size
            h = np.clip(h, 0.0, 1.0)
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

# global variables used by the optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(h) == target."""
    h = np.clip(h, 0.0, 1.0)
    # iterative uniform correction
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final small correction on free variables
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0)
        idx = np.where(mask)[0]
        if idx.size:
            h[idx] += diff / idx.size
            h = np.clip(h, 0.0, 1.0)
    return h


def sum_con(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def sum_con_jac(x: np.ndarray) -> np.ndarray:
    return np.ones_like(x)


def softmax_obj_and_grad(x: np.ndarray):
    """Objective = smooth max of overlap, with analytic gradient."""
    n = x.size
    O = np.correlate(x, 1.0 - x, mode="full")
    beta = CURRENT_BETA
    betaO = beta * O
    m = np.max(betaO)
    expb = np.exp(betaO - m)
    sumexpb = np.sum(expb)
    obj = (m + np.log(sumexpb)) / beta
    w = expb / sumexpb
    conv = np.convolve(w, 1.0 - 2.0 * x, mode="full")
    grad = conv[n - 1 : n - 1 + n]
    return obj, grad


def slsqp_optimize_softmax(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500):
    """SLSQP on the soft‑max objective with analytic gradient."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_con, "jac": sum_con_jac}
    res = minimize(
        softmax_obj_and_grad,
        h0,
        method="SLSQP",
        jac=True,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(x.sum() - CURRENT_TARGET) > 1e-12:
        x = project_sum(x, CURRENT_TARGET)
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def c_key(pair):
    return pair[1]

    """Continuation on a geometric β schedule."""
    best_h = h_start.copy()
    best_c = compute_c5(best_h)

            best_c, best_h = c_new, h_new.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c

        grad = worst_gradient(best_h)
        nh = best_h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = project_sum(nh, best_n / 2.0)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
            step = 0.2
        else:
            step *= 0.97
        it += 1

    # final sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
