def worst_shift_and_grad(h):
    n = len(h)
    cross = np.correlate(h, h, mode='full')[n-1:]  # shape n
    k = int(np.argmin(cross))
    # compute gradient elements
    grad = np.ones_like(h)  # start with 1
    if k > 0:
        # shift +k: positions i where i+k < n, subtract h_{i+k}
        grad[:n-k] -= h[k:]
        # shift -k: positions i where i-k >=0, subtract h_{i-k}
        grad[k:] -= h[:n-k]
    # else k=0: grad = 1 - 2h (do below)
    else:
        grad -= 2 * h
    return grad, k, cross

def upsample_to_n(h, n):
    n_old = h.size
    x_old = np.linspace(0,2, n_old, endpoint=False)
    x_new = np.linspace(0,2, n, endpoint=False)

def guided_hill_climb_once(h, rng):
    # compute grad and worst shift
    grad, k = worst_grad_and_k(h)
    # find i where grad is max (positive) and j where grad is min (negative)
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = max_eps * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    dx = 2.0 / n
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    O = 1.0 - cross * dx
    return float(np.max(O))


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum x = target} using bisection on the dual."""
    lo, hi = -2.0, 2.0
    for _ in range(30):  # 30 iterations give ~1e-9 tolerance
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample_to_n(h: np.ndarray, n: int) -> np.ndarray:
    """Linear up‑sampling from len(h) to n points, then projection onto the feasible set."""
    n0 = h.size
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(np.clip(h_new, 0.0, 1.0), n / 2.0)


def subgradient_step(h: np.ndarray, target: float, alpha: float):
    """One projected sub‑gradient step for max‑overlap."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # len n
    k = int(np.argmin(cross))                     # worst shift (max O_k)

    # gradient of O_k w.r.t. h : 1 - h_{i+k} - h_{i-k}
    grad = np.ones_like(h)
    if k == 0:
        grad -= 2 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]

    h_new = project_box_sum(h - alpha * grad, target)

    dx = 2.0 / n
    O = 1.0 - cross * dx
    c = float(np.max(O))
    return h_new, c, k


def subgradient_optimize(h0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap with time limit."""
    start = time.time()
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    # initial C5
    cross = np.correlate(h, h, mode="full")[n - 1 :]
    best_c = float(np.max(1.0 - cross * (2.0 / n)))
    best_h = h.copy()
    t = 0
    while time.time() - start < time_budget:
        t += 1
        alpha = 0.8 / np.sqrt(t + 10.0)
        h, c, _ = subgradient_step(h, target, alpha)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c


def softmax_c5(x: np.ndarray, beta: float) -> float:
    """Smooth approximation of max_k O_k."""
    n = x.size
    dx = 2.0 / n
    cross = np.correlate(x, x, mode="full")[n - 1 :]
    O = 1.0 - cross * dx
    m = np.max(O)
    e = np.exp(beta * (O - m))
    return float(m + np.log(np.sum(e)) / beta)


def slsqp_refine(h: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 150):
    """Short SLSQP polish using the soft‑max approximation."""
    n = h.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        lambda x: softmax_c5(x, beta),
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (project once more)
    x = project_box_sum(x, target)
    return x
