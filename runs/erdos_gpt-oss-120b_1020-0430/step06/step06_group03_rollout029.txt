import time, math, numpy as np

def compute_c5(h):
    """official evaluation: max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_corr_and_min_corr(h):
    """autocorrelation of h with itself and the minimum value for non‑zero lag."""
    corr = np.correlate(h, h, mode='full')
    n = h.size
    zero = n - 1                     # index of lag 0
    left = corr[:zero]
    right = corr[zero + 1:]
    # left/right may be empty only for n=1 which never occurs here
    min_corr = left.min() if left.size else np.inf
    if right.size:
        rc = right.min()
        if rc < min_corr:
            min_corr = rc
    return corr, min_corr


def propose_targeted_swap(h, n, zero, rng, corr, min_corr):
    """
    Try to improve the worst autocorrelation lag.
    Returns a pair (i, j) to swap (i becomes 1, j becomes 0) or None.
    """
    worst = np.where(corr == min_corr)[0]
    # discard the zero‑lag entry
    mask = worst != zero
    worst = worst[mask]
    if worst.size == 0:
        return None
    lag_idx = worst[rng.integers(worst.size)]
    lag = lag_idx - zero
    k = abs(lag)
    if k == 0:
        return None
    # region where the lag k contributes
    region_end = n - k                     # indices 0 … region_end‑1
    # i must be 0 inside the region with partner 1
    hi = h[0:region_end]
    hi_k = h[k:n]
    A_mask = (hi == 0) & (hi_k == 1)
    A = np.where(A_mask)[0]                # i‑candidates (relative to start)
    # j must be outside the region (its removal does not affect lag k)
    B_mask = (h[region_end:] == 1)
    B = np.where(B_mask)[0] + region_end   # absolute indices
    if A.size and B.size:
        i = A[rng.integers(A.size)]
        j = B[rng.integers(B.size)]
        return int(i), int(j)
    return None


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------------ #
    # initialise a balanced binary sequence (use the provided one if any)
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        arr = np.array(initial_h_values, dtype=float)
        h = np.zeros(n, dtype=float)
        # take the n/2 largest entries as 1
        h[np.argsort(-arr)[:n // 2]] = 1.0
    else:
        n = 512
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    # ------------------------------------------------------------------ #
    corr, min_corr = compute_corr_and_min_corr(h)
    zero = n - 1
    best_h = h.copy()
    best_min_corr = min_corr
    best_c5 = 1.0 - (2.0 / n) * min_corr

    total_budget = budget_s
    T0 = 3.0                               # initial temperature
    p_target = 0.7                         # probability of trying a targeted move
    max_no_improve = 150000                # restart when stuck
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > total_budget * 0.99:
            break

        # linear cooling
        T = T0 * max(1e-12, 1.0 - elapsed / (total_budget * 0.99))

        # --------------------------------------------------------------
        # propose a swap
        if rng.random() < p_target:
            tgt = propose_targeted_swap(h, n, zero, rng, corr, min_corr)
            if tgt is not None:

                new_h = h.copy()
                new_h[i] = 1.0
                new_h[j] = 0.0
            else:

                new_h[j] = 0.0
        else:
            zeros = np.where(h == 0)[0]

    # final evaluation with the official routine
    best_h = np.clip(best_h, 0.0, 1.0)

    cur_sum = best_h.sum()
    target_sum = n / 2.0

        if diff > 0:

            idx = rng.choice(zeros, diff, replace=False)
            best_h[idx] = 1.0
        elif diff < 0:
            ones = np.where(best_h > 0.5)[0]
            idx = rng.choice(ones, -diff, replace=False)
            best_h[idx] = 0.0
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(n)
