def project_onto_capped_simplex(v, target):
    # v arbitrary real vector.
    # Clip to [0,1] after adjusting lambda.
    # Equivalent to find lambda such that sum_i clip(v_i - lambda, 0, 1) = target.
    # Use bisection on lambda.
    lo = -1.0  # function at lo > target? Let's compute sum_i clip(v_i - lo, 0, 1) monotonic decreasing in lo.
    hi = 2.0  # because v_i - hi will be <= -1 => clip to 0; sum = 0 (< target). At lo large negative, clip =1 all => sum = n (> target). So we can bisection.
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0.0, 1.0)

import time
import numpy as np

# ---------- utilities ----------
def compute_c5_autocorr(h: np.ndarray) -> float:
    """C5 from autocorrelation: C5 = 1 - (2/n)*min_k sum_i h_i h_{i+k}."""
    n = h.size
    full = np.correlate(h, h, mode="full")
    C = full[len(full) // 2 :]  # k = 0..n-1
    return 1.0 - (2.0 / n) * np.min(C)


def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x : 0<=x_i<=1, sum x_i = target}.
    Uses bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 2.0  # bounds for lambda
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512  # default size
    target = n / 2.0

    # ---- initialise h ----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        if h.size != n:
            x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, h)
    else:
        h = rng.random(n)

    h = np.clip(h, 0.0, 1.0)
    h = project_onto_capped_simplex(h, target)

    best_h = h.copy()
    best_c = compute_c5_autocorr(h)

    # ---- parameters for the projected‑gradient loop ----
    base_step = 1.0 / n                # roughly 0.002 for n=512
    beta = 1.0
    max_beta = 20000.0
    beta_step_iters = 2000
    it = 0
    no_improve = 0
    prev_c = best_c

    # ---- main optimisation loop ----
    while time.time() - start < budget_s:
        # autocorrelation and weights
        full = np.correlate(h, h, mode="full")
        C = full[len(full) // 2 :]          # C_k for k>=0
        C_min = C.min()
        w = np.exp(beta * (C_min - C))      # w_k >=0, max at min C_k

        # directional gradient (sum of weighted right/left shifts)
        shift_r = np.convolve(w, h, mode="full")[:n]
        shift_l = np.convolve(w[::-1], h, mode="full")[n - 1 : 2 * n - 1]
        direction = shift_r + shift_l

        # propose a step and project onto feasible set
        lr = base_step / (beta + 1.0)
        h_candidate = project_onto_capped_simplex(h + lr * direction, target)
        c_candidate = compute_c5_autocorr(h_candidate)

        if c_candidate < prev_c - 1e-12:
            h = h_candidate
            prev_c = c_candidate
            no_improve = 0
            if c_candidate < best_c:
                best_c = c_candidate
                best_h = h_candidate.copy()
                if best_c <= 0.3808:
                    break
        else:
            no_improve += 1
            if no_improve >= 5000:
                # random shake to escape a plateau
                h = project_onto_capped_simplex(
                    h + rng.normal(scale=0.01, size=n), target
                )
                prev_c = compute_c5_autocorr(h)
                no_improve = 0

        # increase beta periodically
        if (it % beta_step_iters) == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    # ---- final polishing with very large beta ----
    for _ in range(3):
        if time.time() - start >= budget_s:
            break
        full = np.correlate(best_h, best_h, mode="full")
        C = full[len(full) // 2 :]
        C_min = C.min()
        w = np.exp(max_beta * (C_min - C))
        shift_r = np.convolve(w, best_h, mode="full")[:n]
        shift_l = np.convolve(w[::-1], best_h, mode="full")[n - 1 : 2 * n - 1]
        direction = shift_r + shift_l
        lr = base_step * 0.01 / (max_beta + 1.0)
        cand = project_onto_capped_simplex(best_h + lr * direction, target)
        cand_c = compute_c5_autocorr(cand)
        if cand_c < best_c:
            best_c = cand_c
            best_h = cand.copy()
            if best_c <= 0.3808:
                break

    # ---- sanitise and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target)
    final_c = compute_c5_autocorr(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
