def grad_smooth_max(h, beta):
    n = h.size
    dx = 2.0 / n
    oc = np.correlate(h, 1 - h, mode="full") * dx  # size 2n-1
    bc = beta * oc
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp
    neighbor_sum = np.zeros(n)
    # iterate over shift range
    # using python loops may cost O(n^2) anyway.
    for shift in range(-(n-1), n):
        wi = w[shift + (n-1)]
        if wi == 0.0:
            continue
        if shift >= 0:
            neighbor_sum[: n - shift] += wi * h[shift:]
        else:
            neighbor_sum[-shift:] += wi * h[: n + shift]
    w0 = w[n-1]
    grad = dx * (1.0 - neighbor_sum - w0 * h)
    return grad

import time
import numpy as np

# ----------------------------------------------------------------------
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by binary search on the shift λ."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample_initial_to_n(n: int, rng: np.random.Generator) -> np.ndarray:
    if "initial_h_values" not in globals():
        return random_feasible(n, rng)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = h0.size
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x_old, h0)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_sum(h_new, n / 2.0)


def compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 using FFT (max over k of ∫h(1‑h∘k))."""
    n = h.size
    dx = 2.0 / n
    M = 1 << (int(np.ceil(np.log2(2 * n - 1))))
    H = np.fft.rfft(h, n=M)
    U = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(U), n=M)
    oc = corr[: 2 * n - 1] * dx
    return float(oc.max())


def grad_smooth_max(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max of O_k = ∫h(1‑h∘k) with parameter β."""
    n = h.size
    dx = 2.0 / n

    # --- correlation O_k via FFT ---------------------------------------
    M1 = 1 << (int(np.ceil(np.log2(2 * n - 1))))
    H1 = np.fft.rfft(h, n=M1)
    U1 = np.fft.rfft(1.0 - h, n=M1)
    corr = np.fft.irfft(H1 * np.conj(U1), n=M1)
    oc = corr[: 2 * n - 1] * dx

    # --- soft‑max weights ----------------------------------------------
    bc = beta * oc
    m = bc.max()
    exp_bc = np.exp(bc - m)
    w = exp_bc / exp_bc.sum()                   # length 2n‑1
    w0 = w[n - 1]                               # weight of shift 0

    # --- neighbor sum = Σ_k w_k h_{i+k} via convolution -----------------
    L = w.size
    M2 = 1 << (int(np.ceil(np.log2(n + L - 1))))
    H2 = np.fft.rfft(h, n=M2)
    Wr = np.fft.rfft(w[::-1], n=M2)
    conv = np.fft.irfft(H2 * Wr, n=M2)
    neighbor = conv[L - 1 : L - 1 + n]          # Σ_k w_k h_{i+k}

    # --- gradient ------------------------------------------------------
    grad = dx * (1.0 - neighbor - w0 * h)
    return grad


def adam_optimize(
    h0: np.ndarray,
    beta: float,
    target: float,
    num_iters: int,
    step_size: float = 1.0,
) -> np.ndarray:
    """Adam with projection onto the box‑sum polytope."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    for t in range(1, num_iters + 1):
        g = grad_smooth_max(h, beta)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h = h - step_size * m_hat / (np.sqrt(v_hat) + eps)
        h = project_box_sum(h, target)
    return h


def anneal_optimize(
    h0: np.ndarray,
    target: float,
    start_time: float,
    budget_deadline: float,
) -> tuple[np.ndarray, float]:
    """Gradual increase of β with Adam; stops when time is low or target met."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    beta = 5.0
    max_beta = 20000.0
    while time.time() < budget_deadline - 5.0 and beta <= max_beta:
        time_left = budget_deadline - time.time()
        if time_left < 1.0:
            break
        # allocate a modest number of iterations (200‑5000) based on remaining time

            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break
        beta = min(beta * 2.0, max_beta)
    return best_h, best_c


def final_hill_climb(
    h0: np.ndarray, rng: np.random.Generator, max_iters: int = 2000
) -> tuple[np.ndarray, float]:
    """Pairwise transfer guided by the (large‑β) gradient."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    beta_big = 20000.0
    for _ in range(max_iters):
        grad = grad_smooth_max(h, beta_big)
        i = int(np.argmax(grad))
