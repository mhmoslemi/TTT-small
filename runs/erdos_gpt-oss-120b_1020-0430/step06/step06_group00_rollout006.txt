def project_onto_simplex(v, target):
    # v is numpy array
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # final x
    return np.clip(v - (lo + hi)/2.0, 0.0, 1.0)
```

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | 0<=x_i<=1, sum x_i = target}.
    Uses bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 2.0  # guarantees sum(lo) = N, sum(hi) = 0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def pgd_optimize(h0: np.ndarray, max_iters: int, base_step: float,
                 rng: np.random.Generator):
    """Projected gradient descent on the worst‑shift objective."""
    h = h0.copy()
    target = h.size / 2.0
    best_c = compute_c5(h)
    step = base_step
    for it in range(max_iters):
        g = worst_gradient(h)
        # line‑search with projection
        for _ in range(10):
            h_new = project_onto_simplex(h - step * g, target)
            c_new = compute_c5(h_new)
            if c_new <= best_c + 1e-14:
                h = h_new
                best_c = c_new
                step *= 1.02
                break
            step *= 0.5
        else:
            # stagnation: small random perturbation
            i, j = rng.choice(h.size, size=2, replace=False)
            eps = rng.random() * 0.01
            h[i] = min(1.0, max(0.0, h[i] + eps))
            h[j] = min(1.0, max(0.0, h[j] - eps))
            h = project_onto_simplex(h, target)
            best_c = compute_c5(h)
            step = base_step
    return h, best_c


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def now(): return time.time() - start

    # ---- initial feasible point ---------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_simplex(h0, h0.size / 2.0)
    else:
        n0 = 256
        h0 = rng.random(n0)
        h0 = project_onto_simplex(h0, n0 / 2.0)

    best_h = h0
    best_c = compute_c5(best_h)
    best_n = best_h.size

    # ---- low‑resolution multiple restarts (256) -----------------
    low_n = 256
    restarts = 5
    steps_low = 2000
    for _ in range(restarts):
        if now() > budget_s * 0.5:
            break
        h_rand = rng.random(low_n)
        h_rand = project_onto_simplex(h_rand, low_n / 2.0)
        h_opt, c_opt = pgd_optimize(h_rand, steps_low, 0.5, rng)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, low_n

    # ---- upsample to medium resolution (512) -------------------
    if now() < budget_s * 0.8:
        mid_n = 512
        if best_n != mid_n:
            best_h = upsample_h(best_h, mid_n)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = project_onto_simplex(best_h, mid_n / 2.0)
            best_c = compute_c5(best_h)
            best_n = mid_n
        steps_mid = 1500
        h_opt, c_opt = pgd_optimize(best_h, steps_mid, 0.5, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- upsample to fine resolution (1024) --------------------
    if now() < budget_s * 0.95:
        fine_n = 1024
        if best_n != fine_n:
            best_h = upsample_h(best_h, fine_n)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = project_onto_simplex(best_h, fine_n / 2.0)
            best_c = compute_c5(best_h)
            best_n = fine_n
        steps_fine = 800
        h_opt, c_opt = pgd_optimize(best_h, steps_fine, 0.5, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- tiny continuous SA refinement if time permits ----------
    if now() < budget_s:
        h_curr = best_h.copy()
        c_curr = best_c
        best_local_h = h_curr.copy()
        best_local_c = c_curr
        remaining = budget_s - now()
        # estimate iterations (each ~0.5 ms)
        max_iters = max(1, int(remaining / 0.0005))
        T0, T1 = 0.01, 1e-6
        for it in range(max_iters):
            t = it / max_iters
            T = T0 * (T1 / T0) ** t
            i = rng.integers(fine_n)
            j = rng.integers(fine_n - 1)
            if j >= i:
                j += 1
            max_eps = min(h_curr[i], 1.0 - h_curr[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * 0.1
            h_new = h_curr.copy()
            h_new[i] -= eps
            h_new[j] += eps
            c_new = compute_c5(h_new)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
                h_curr, c_curr = h_new, c_new
                if c_new < best_local_c:
                    best_local_c, best_local_h = c_new, h_new.copy()
        if best_local_c < best_c:
            best_h, best_c = best_local_h, best_local_c

    # ---- final sanitisation ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
