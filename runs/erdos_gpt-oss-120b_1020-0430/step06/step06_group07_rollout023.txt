def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

import time

# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete max overlap C5 using FFT (full correlation)."""
    n = h.shape[0]
    pad = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n
    H = np.fft.rfft(h, pad)
    rev = np.flip(1.0 - h)
    G = np.fft.rfft(rev, pad)
    corr = np.fft.irfft(H * G, pad)
    return corr.max() * (2.0 / n)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via a simple bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the box with exact sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def downsample_to_n(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear down‑sample to n_target points and re‑project."""
    n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_sum(h_new, n_target / 2.0)

    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

# Simple simulated annealing
def simulated_annealing(h0: np.ndarray, time_budget: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """SA on the max‑overlap objective with sum‑preserving swaps."""
    n = h0.shape[0]
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c

    start = time.time()
    T_start, T_end = 0.04, 1e-7                 # cooling schedule bounds
    last_improve = start

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        T = T_start * (1 - t_frac) + T_end * t_frac

        # propose a sum‑preserving swap
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # direction: increase i, decrease j
        delta_max = min(1.0 - h[i], h[j])
        if delta_max <= 0:
            # try opposite direction
            delta_max = min(1.0 - h[j], h[i])
            if delta_max <= 0:
                continue
            i, j = j, i

        delta = rng.random() * delta_max
        if rng.random() < 0.5:                 # random sign
            delta = -delta
            i, j = j, i

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c = compute_c5(h_new)

        # record best
        if new_c < best_c:
            best_c, best_h = new_c, h_new.copy()
            last_improve = time.time()

        # Metropolis acceptance
        dC = new_c - cur_c
        if dC <= 0 or rng.random() < np.exp(-dC / (T + 1e-12)):
            h, cur_c = h_new, new_c

        # occasional random restart if stalled
        if time.time() - last_improve > max(5.0, time_budget * 0.07):
            h = random_feasible(n, rng)
            cur_c = compute_c5(h)
            last_improve = time.time()

        # early exit if we already beat the target
        if best_c <= 0.3808:
            break

    return best_h, best_c

def worst_gradient(h: np.ndarray) -> tuple[np.ndarray, int]:
    """Gradient of the worst‑shift overlap O_k (non‑negative shifts only)."""
    n = h.shape[0]
    cross = np.correlate(h, h, mode='full')[n - 1:]   # length n, lag 0..n‑1
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6) -> tuple[np.ndarray, float]:
    """Local hill‑climb using the worst‑gradient direction."""
    best_h, best_c = h.copy(), compute_c5(h)
    cur_h, cur_c = h.copy(), best_c
    decay = max(1, max_iters // 5)

    for it in range(max_iters):
        grad, _ = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        if eps_max <= 0:
            continue
        eps = rng.random() * eps_max * step_scale
        h_new = cur_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)

        if new_c < best_c:
            best_c, best_h, cur_h, cur_c = new_c, h_new.copy(), h_new.copy(), new_c
        else:
            if new_c < cur_c:
                cur_h, cur_c = h_new, new_c

        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)

    return best_h, best_c

    else:
        n_init = 512
        h_init = random_feasible(n_init, rng)

    # Target resolution (keep ≤512)
    final_n = min(512, n_init)

    # Fast resolution for intensive SA
    n_fast = 256
    if n_init > n_fast:
        h_fast = downsample_to_n(h_init, n_fast)
    else:
        h_fast = project_box_sum(h_init, n_fast / 2.0)

    # ----------------------------------------------------------------
    # Build a small portfolio of 3 seeds for the fast SA phase
    candidates = [h_fast]

    # Sinusoidal seed (amplitude 0.5)
    x = np.linspace(0.0, 2.0, n_fast, endpoint=False)
    h_sin = 0.5 + 0.5 * np.sin(np.pi * x)
    h_sin = np.clip(h_sin, 0.0, 1.0)
    h_sin = project_box_sum(h_sin, n_fast / 2.0)
    candidates.append(h_sin)

    # Pure random seed
    candidates.append(random_feasible(n_fast, rng))

    # ----------------------------------------------------------------
    # Allocate time: ~60 % for fast SA, remainder for up‑sampling & polishing
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    fast_budget = remaining * 0.60
    fast_budget = max(fast_budget, 30.0)                     # guarantee a minimum

    per_candidate = fast_budget / len(candidates)

    best_h_fast, best_c_fast = None, np.inf
    for h0 in candidates:
        h_opt, c_opt = simulated_annealing(h0, per_candidate, rng)
        if c_opt < best_c_fast:
            best_h_fast, best_c_fast = h_opt, c_opt

    # ----------------------------------------------------------------
    # Upsample to final resolution (≤512) and polish further
    if final_n > n_fast:
        h_up = upsample_double(best_h_fast)
    else:
        h_up = best_h_fast

    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 6.0:
        # Fine‑grained SA on the full grid
        fine_sa_budget = remaining * 0.80
        h_fine, c_fine = simulated_annealing(h_up, fine_sa_budget, rng)
    else:
        h_fine, c_fine = h_up, compute_c5(h_up)

    # ----------------------------------------------------------------
    # Final short gradient hill‑climb (if time permits)
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 5.0:
        h_final, c_final = guided_hill_climb(h_fine, rng,
                                             max_iters=int(min(4000, remaining * 250)),
                                             step_scale=0.6)
    else:
        h_final, c_final = h_fine, c_fine

    # ----------------------------------------------------------------
    # Sanitize and return
    h_final = np.clip(h_final, 0.0, 1.0)
    h_final = project_box_sum(h_final, h_final.size / 2.0)
    final_c = compute_c5(h_final)

    return h_final.tolist(), float(final_c), int(h_final.size)
