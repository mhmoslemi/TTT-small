def weighted_gradient(a, beta):
    n = a.size
    b = 1.0 - a
    # compute correlation
    corr = np.correlate(a, b, mode='full')
    shifts = np.arange(-(n-1), n)
    # weights
    exp_beta_corr = np.exp(beta * corr - beta_max)   # substract max for numerical stability
    sum_w = np.sum(exp_beta_corr)
    # compute gradient
    grad = np.zeros(n)
    # we need to sum w_j * g_j across j
    # use loop over shifts
    for idx, shift in enumerate(shifts):
        w = exp_beta_corr[idx]
        if w == 0: continue
        if shift >= 0:
            # i in [0, n-shift)
            grad[:n-shift] += w * b[shift:]
            # i in [shift, n)
            grad[shift:] -= w * a[:n-shift]
        else:
            # shift negative: equivalent to shift = -s
            s = -shift
            # i in [s, n): from a_i * b_{i-s}
            grad[s:] += w * b[:n-s]
            # i in [0, n-s): subtract a[i+s]
            grad[:n-s] -= w * a[s:]
    # divide by sum_w to get weighted average
    grad = grad / sum_w
    return (np.log(sum_w) / beta, grad)
```

def softmax_obj_grad(h, beta):
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    # compute softmax weights safely
    max_val = beta * np.max(corr)
    exp_terms = np.exp(beta * corr - max_val)
    sum_w = np.sum(exp_terms)
    f = (max_val + np.log(sum_w)) / beta  # because log(sum(exp(beta*c))) = max_val + log(sum_w)
    # compute gradient
    grad = np.zeros(n)
    # shifts = np.arange(-(n-1), n)
    # Loop over idx and shift
    for idx, shift in enumerate(range(-(n-1), n)):
        w = exp_terms[idx]
        if w == 0.0:
            continue
        if shift >= 0:
            # first part
            grad[:n-shift] += w * b[shift:]
            # second part
            grad[shift:] -= w * h[:n-shift]
        else:
            s = -shift
            grad[s:] += w * b[:n-s]
            grad[:n-s] -= w * h[s:]
    grad = grad / sum_w
    return f, grad
```

def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        # find suitable indices
        if diff > 0:
            # indices where h < 1
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1 - h[i])
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while keeping 0 ≤ h ≤ 1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk to fix the sum
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1 - h[i])
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift (used in sub‑gradient steps)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


# ---------- soft‑max objective with analytic gradient ----------
def softmax_obj_grad(h: np.ndarray, beta: float):
    """Return (f, grad) for the smooth approximation of max_k O_k."""
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")               # length 2n‑1
    max_val = beta * np.max(corr)                       # for numerical stability
    exp_terms = np.exp(beta * corr - max_val)           # safe exponents
    sum_w = np.sum(exp_terms)
    f = (max_val + np.log(sum_w)) / beta                # log‑sum‑exp

    grad = np.zeros(n)
    # shifts run from -(n‑1) … n‑1
    shift = -(n - 1)
    for w in exp_terms:
        if w != 0.0:
            if shift >= 0:
                grad[: n - shift] += w * b[shift:]
                grad[shift:] -= w * h[: n - shift]
            else:
                s = -shift
                grad[s:] += w * b[: n - s]
                grad[: n - s] -= w * h[s:]
        shift += 1
    grad /= sum_w
    return f, grad


# ---------- optimization per resolution ----------
def optimize_n(h0: np.ndarray, n: int, rng: np.random.Generator,
               time_limit: float, start_time: float):
    """Run multi‑β SLSQP with analytic gradient on length‑n vector."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    betas = [10.0, 50.0, 200.0, 1_000.0, 5_000.0,
             20_000.0, 80_000.0, 200_000.0]

    for beta in betas:
        if time.time() - start_time >= time_limit:
            break
        # remaining time for this β
        remain = time_limit - (time.time() - start_time)
        maxiter = int(min(2500, max(200, remain * 30)))   # crude estimate
        bounds = [(0.0, 1.0)] * n
        target = n / 2.0

        cons = [{
            "type": "eq",
            "fun": lambda x, t=target: np.sum(x) - t,
            "jac": lambda x, t=target: np.ones_like(x),
        }]

        # wrappers for objective & Jacobian
        def fun(x):
            f, _ = softmax_obj_grad(x, beta)
            return f

        def jac(x):
            _, g = softmax_obj_grad(x, beta)
            return g

        try:
            res = minimize(fun, best_h, method="SLSQP", jac=jac,
                           bounds=bounds, constraints=cons,
                           options={"maxiter": maxiter,
                                    "ftol": 1e-12,
                                    "disp": False})
        except Exception:
            res = minimize(fun, best_h, method="SLSQP",
                           bounds=bounds, constraints=cons,
                           options={"maxiter": maxiter,
                                    "ftol": 1e-12,
                                    "disp": False})
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x.copy()

        # small random jitter to escape potential plateaus
        if rng.random() < 0.5:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = rng.random() * min(best_h[i], 1 - best_h[j]) * 0.1
                best_h[i] -= eps
                best_h[j] += eps
                best_h = np.clip(best_h, 0.0, 1.0)
                best_h = enforce_sum(best_h, rng)

    return best_h, best_c

                      start_time: float):

    best_h = h.copy()
    best_c = compute_c5(best_h)

        nh = np.clip(nh, 0.0, 1.0)

        c = compute_c5(nh)

            best_c, best_h = c, nh.copy()
            it = 0
        else:
            it += 1
        if it > 2000:

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # base 512‑point construction
    if "initial_h_values" in globals():
        init_full = np.array(initial_h_values, dtype=float)
        init_full = np.clip(init_full, 0.0, 1.0)
        init_full = enforce_sum(init_full, rng)
    else:
        init_full = thue_morse_bits(512).astype(float)
        init_full = enforce_sum(init_full, rng)

    best_h = None
    best_c = np.inf
    best_n = None
    remaining_budget = budget_s

    # work on three resolutions (more points → tighter bound)
    for n, fraction in [(128, 0.15), (256, 0.25), (512, 0.5)]:
        alloc = budget_s * fraction
        time_allowed = min(alloc, remaining_budget)
        if time_allowed < 5.0:
            continue

        # resize initial vector to current resolution
        if n == 512:
            h0 = init_full.copy()
        else:
            x_old = np.linspace(0, 2, init_full.size, endpoint=False)

            h0 = np.clip(h0, 0.0, 1.0)

        h_opt, c_opt = optimize_n(h0, n, rng, time_allowed, start)
        if c_opt < best_c:

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
