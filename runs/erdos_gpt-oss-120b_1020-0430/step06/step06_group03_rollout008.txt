    N = 1 << ((3*n - 2).bit_length())

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    # max amount we can move: limited by h[i] (can't go negative) and (1 - h[j]) (can't exceed 1)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:

    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import numpy as np

def correlation_fft(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(H * np.conj(G)).real
    return corr

def compute_c5_fast(h):
    n = h.size
    corr = correlation_fft(h)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = correlation_fft(h)
    idx = np.argmax(corr) - (n - 1)
    k = idx
    grad = np.empty_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[:] = 1.0
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        k = -k
        grad[:] = 0.0
        grad[k:] += (1.0 - h[: n - k])
        grad[: n - k] -= h[k:]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=5000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5_fast(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def grad_logsumexp(h, beta):
    n = h.shape[0]
    # correlation to obtain O
    N_corr = 2 * n
    Hc = np.fft.fft(h, N_corr)
    Gc = np.fft.fft(1.0 - h, N_corr)
    O_full = np.fft.ifft(Hc * np.conj(Gc)).real
    O = O_full[: 2 * n - 1]
    m = O.max()
    exp_term = np.exp(beta * (O - m))
    w = exp_term / exp_term.sum()
    w_rev = w[::-1]

    # convolution size for gradient
    N = 1 << ((3 * n - 2).bit_length())
    W = np.fft.fft(w, N)
    W_rev = np.fft.fft(w_rev, N)
    H = np.fft.fft(h, N)

    conv_rev_a = np.fft.ifft(W_rev * (N - H)).real
    conv_h = np.fft.ifft(W * H).real
    start = n - 1
    g = conv_rev_a[start:start + n] - conv_h[start:start + n]
    return g, O, w

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024
    target = n / 2.0

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            h0 = np.repeat(h0, n // h0.size)
        h = np.clip(h0, 0.0, 1.0)
        h = project_onto_box_sum(h, target)
    else:
        raw = rng.random(n)
        raw = raw / raw.sum() * target
        h = np.clip(raw, 0.0, 1.0)
        h = project_onto_box_sum(h, target)

    best_h = h.copy()
    best_c = compute_c5_fast(best_h)

    # beta schedule with increasing precision
    beta_schedule = [5, 20, 80, 320, 1280, 5120, 20480]
    iters_per_beta = {5: 8000, 20: 12000, 80: 18000,
                      320: 24000, 1280: 30000,
                      5120: 36000, 20480: 42000}
    for beta in beta_schedule:
        if time.time() - start > budget_s * 0.85:
            break
        max_iters = iters_per_beta[beta]
        # Adam parameters
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.12 / (beta ** 0.25 + 1.0)

        for t in range(1, max_iters + 1):
            g, O, w = grad_logsumexp(h, beta)
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_sum(h, target)

            if (t % 2000) == 0:
                c = compute_c5_fast(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()

            if time.time() - start > budget_s * 0.85:
                break

    # Local refinement with guided hill climbing
    if time.time() - start < budget_s * 0.93:
        hc, cc = guided_hill_climb(best_h, rng, max_iters=8000, step_scale=0.5)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    # Final stochastic improvement until time runs out
    while time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5_fast(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5_fast(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
