                diff2 -= delta
            elif diff2 < 0 and h[i] > 0.0:
                delta = min(-diff2, h[i])
                h[i] -= delta

def compute_corr_full(h):
    n = h.shape[0]
    N = 2 * n
    f = np.fft.rfft(h, N)
    g = np.fft.rfft(np.flip(1.0 - h), N)
    conv = np.fft.irfft(f * g, N)[: 2 * n - 1]
    return conv

    n = h.shape[0]
    corr = compute_corr_full(h)
    maxc = np.max(corr)

            grad -= 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    return idxs, grad

            for beta in (1000.0, 3000.0, 10000.0):
                if time.time() - start >= budget_s * 0.95:
                    break
                h_s = slsqp_refine(pop[0], target, rng, beta=beta, max_iter=200)
                c_s = compute_c5(h_s)
                if c_s < c_vals[0]:
                    pop[0] = h_s
                    c_vals[0] = c_s
                    if c_s < global_best_c:
                        global_best_c = c_s
                        global_best_h = h_s.copy()

    # final SLSQP polish if time permits
    if time.time() - start < budget_s * 0.96:
        for beta in (2000.0, 8000.0, 20000.0):
            if time.time() - start >= budget_s * 0.98:
                break
            h_s = slsqp_refine(global_best_h, target, rng, beta=beta, max_iter=400)
            c_s = compute_c5(h_s)
            if c_s < global_best_c:
                global_best_c = c_s
                global_best_h = h_s.copy()

    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = project_sum_and_bounds(global_best_h, target, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(global_best_h.size)
