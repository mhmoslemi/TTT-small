def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = abs(idx)
    grad = np.ones_like(h)

        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h[i+k] term
        grad[k:]    -= h[: n - k]       # subtract h[i-k] term
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# --------- core helpers -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def thue_morse_bits(n: int) -> np.ndarray:
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def random_balanced_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n)
    a[rng.choice(n, n // 2, replace=False)] = 1.0
    return a


def resize_h(h: np.ndarray, n: int) -> np.ndarray:
    if h.size == n:
        return h.copy()
    x_old = np.linspace(0, 2, h.size, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    return np.interp(x_new, x_old, h)


# --------- SLSQP with soft‑max -----------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1200) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x


# --------- main routine -------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = resize_h(h, n)
    else:
        h = thue_morse_bits(n).astype(float)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    current_h = h.copy()
    current_c = compute_c5(current_h)
    best_h = current_h.copy()
    best_c = current_c

    # ----- annealing parameters -----------------------------------------
    temp0 = 5.0
    anneal_end = start + budget_s * 0.78
    no_improve = 0

    while time.time() < anneal_end:
        elapsed = time.time() - start
        temp = temp0 * np.exp(-6.0 * elapsed / budget_s)
        grad = worst_gradient(current_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i, j = rng.choice(n, size=2, replace=False)
        max_eps = min(current_h[i], 1.0 - current_h[j])
        # if chosen direction impossible, try reverse
        if max_eps <= 0:
            max_eps = min(current_h[j], 1.0 - current_h[i])
            if max_eps > 0:
                i, j = j, i
        if max_eps <= 0:
            i, j = rng.choice(n, size=2, replace=False)
            max_eps = min(current_h[i], 1.0 - current_h[j])
            if max_eps <= 0:
                continue
        eps = max_eps * rng.random() * max(0.1, temp / 5.0)
        if eps <= 0:
            continue
        new_h = current_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - current_c
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / (temp + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            current_h, current_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
                no_improve = 0
                if best_c <= 0.3808:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random restart if stuck
        if no_improve > 50000:
            current_h = random_balanced_binary(n, rng)
            current_c = compute_c5(current_h)
            no_improve = 0

    # ----- final SLSQP refinement (if time permits) --------------------
    if time.time() < start + budget_s * 0.97:
        for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
            if time.time() - start > budget_s * 0.98:
                break
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=1200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
                if best_c <= 0.3808:
                    break

    # ----- ensure feasibility and return --------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
