def upsample(h, new_n):
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, new_n/2.0)
```

def simulated_annealing(h0, budget, rng):
    n = h0.size
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    start = time.time()
    T_start = 0.005
    T_end = 1e-7
    eps_start = 0.5
    eps_end = 0.01
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget:
            break
        p = elapsed / budget
        T = T_start * (T_end / T_start) ** p
        eps_scale = eps_start * (eps_end / eps_start) ** p
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * eps_scale
        if eps <= 0:
            continue
        # apply move
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        # decide
        if new_c < c or rng.random() < np.exp((c - new_c) / T):
            c = new_c
            if c < best_c:
                best_c = c

                # optional early exit if below target
                if best_c <= 0.3808:
                    # keep moving? can break early
                    # but we may still have time to improve; but at target fine.
                    pass
        else:
            # revert
            h[i] += eps
            h[j] -= eps
    return best_h, best_c

import time
import numpy as np

# ----------------------------------------------------------------------
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0 <= x <= 1, sum x = target} by binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid, then projection to feasibility."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, new_n / 2.0)


def compute_c5(h: np.ndarray) -> float:
    """C5 = 1 - min autocorrelation (scaled by dx)."""
    n = h.size
    r = np.correlate(h, h, mode="full")
    min_r = r.min()
    return 1.0 - min_r * (2.0 / n)


# ----------------------------------------------------------------------
def simulated_annealing(h0: np.ndarray, budget: float,
                        rng: np.random.Generator,
                        target: float = 0.3808) -> tuple[np.ndarray, float]:
    """Projected SA on the max‑overlap objective."""
    n = h0.size
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    start = time.time()
    # temperature schedule
    T_start, T_end = 0.02, 1e-7
    eps_start, eps_end = 0.5, 0.01

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget:
            break
        p = elapsed / budget
        T = T_start * (T_end / T_start) ** p
        eps_scale = eps_start * (eps_end / eps_start) ** p

        # pick distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * eps_scale
        if eps <= 0.0:
            continue

        # apply move
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)

        if new_c < c or rng.random() < np.exp((c - new_c) / T):
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
                if best_c <= target:
                    # continue a bit more to possibly improve further
                    pass
        else:
            # revert
            h[i] += eps
            h[j] -= eps

    return best_h, best_c
