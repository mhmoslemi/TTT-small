def compute_c5_fft(h):
    n = h.size
    m = 2*n
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    # The linear correlation values correspond to first (2n-1) indices (0..2n-2)
    # After ifft, corr[0] corresponds to shift - (n-1)? Wait careful:
    # The correlation computed this way yields circular convolution of length m.
    # For linear correlation, we want zero padding, so the part from 0 to 2n-2 yields correct.
    # Let's slice corr[:2*n-1].
    corr = corr[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

import time
import numpy as np

# --------------------------- basic utilities ---------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using direct correlation (exact as the evaluator)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 via FFT (used during search)."""
    n = h.size
    m = 2 * n
    revcomp = (1.0 - h)[::-1]
    H = np.fft.fft(h, m)
    R = np.fft.fft(revcomp, m)
    conv = np.fft.ifft(H * R).real[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to enforce sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple greedy redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift (max overlap)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient (few iterations)."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# --------------------------- number theory ---------------------------

def generate_primes(limit: int) -> np.ndarray:
    """Simple sieve returning array of primes up to limit."""
    sieve = np.ones(limit + 1, dtype=bool)
    sieve[:2] = False
    for p in range(2, int(limit ** 0.5) + 1):
        if sieve[p]:
            sieve[p * p :: p] = False
    return np.flatnonzero(sieve)


def legendre_binary(p: int) -> np.ndarray:
    """Binary Legendre sequence of length p (0 at residues, 1 at non‑residues)."""
    seq = np.zeros(p, dtype=np.float64)
    for i in range(1, p):
        ls = pow(i, (p - 1) // 2, p)
        if ls == p - 1:  # non‑residue
            seq[i] = 1.0
    return seq


# --------------------------- binary simulated annealing ---------------------------

def binary_sa_optimize(h_start: np.ndarray, rng: np.random.Generator,
                       time_budget: float, start_time: float,
                       target: float = 0.3808) -> tuple[np.ndarray, float]:
    """Fast binary SA that respects the exact half‑ones constraint."""
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    cur_c = best_c

    # pre‑compute index arrays
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    T0, T_end = 0.2, 1e-5
    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac

        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5_fft(h)
        delta = new_c - cur_c

        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            cur_c = new_c
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
                if best_c <= target:
                    break
        else:
            # reject swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_budget = budget_s

    # ---------- initial candidates ----------
    best_h = None
    best_c = np.inf
    best_n = None

    # consider supplied initial_h_values if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5_fft(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # scan double‑prime constructions (n = 2p, p prime ≤ 500)
    primes = generate_primes(500)
    for p in primes:
        if p % 2 == 0:
            continue
        n = 2 * p
        seq = legendre_binary(p)
        h = np.concatenate([seq, seq])
        # adjust sum to exactly n/2 (add a single 1 if needed)
        if h.sum() < n / 2:
            h[0] = 1.0
        # ensure feasibility
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        c = compute_c5_fft(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ---------- binary simulated annealing ----------
    remaining = time_budget - (time.time() - start_time)
    if remaining > 5.0 and best_h is not None:
        # allocate ~95% of remaining to SA, keep a little for final tweaks
        sa_time = remaining * 0.95
        best_h, best_c = binary_sa_optimize(
            best_h, rng, sa_time, start_time, target=0.3808
        )

    # ---------- continuous fine‑tuning ----------
    elapsed = time.time() - start_time
    if elapsed < time_budget * 0.98 and best_h is not None:
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=8)

    # ---------- final sanitisation ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
