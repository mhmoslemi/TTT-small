   - compute gradient g of worst shift.

   - step size η_i decreasing: e.g., η_i = η0 / sqrt(i+1) or use line search.

   - h_new = h - η_i * g.

   - Clip h_new to [0,1].

   - Adjust sum: if sum != target, we can project onto hyperplane sum = target while respecting bounds, which is like "water filling" algorithm. This is also known as projection onto simplex with bounds [0,1] and sum = target. The sum constraint plus bounds can be solved efficiently.

Implement "project_to_feasible" function that takes h (already clipped) and target sum, and projects onto the set {x ∈ [0,1]^n, Σ x = target}. This is similar to bounded simplex projection. Implementation can be done via iterative algorithm akin to "solve for lambda" using sorting.

Algorithm:

- Input h ∈ [0,1] already.

- We need to find x = min(max(h - λ, 0), 1) for some λ such that Σ x = target (since projection onto hyperplane with box constraints). This is "Euclidean projection onto the capped simplex". We can find λ via bisection.

Implementation:

def project_onto_capped_simplex(v, target, max_iter=100, tol=1e-10):
    lo = -np.max(v) - 1.0  # low bound for λ
    hi = np.max(1 - v) + 1.0 # high bound
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) <= tol:
            break
    return x

We need initial lo/hi. We can set lo = -1, hi = 1, but we need ensure they bracket. Considering v in [0,1], v-λ changes sum monotonic decreasing with λ (larger λ reduces x). When λ very large negative, v - λ is large positive, clipped to 1 => sum = n. For λ very large positive, v - λ negative => sum = 0. So we can set lo = -1e3; hi = 1e3, safe.

def worst_shift_and_grad(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad = np.zeros_like(h)
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        pos = -k
        grad = np.zeros_like(h)
        grad[pos:] = 1.0 - h[: n - pos]
        grad[: n - pos] -= h[pos:]
    return k, grad

def solve_lp(pool_h, target_sum, dx):
    # pool_h is list of arrays shape (n,)
    import cvxpy as cp
    M = len(pool_h)
    n = pool_h[0].shape[0]
    A = np.empty((M, 2*n - 1), dtype=np.float64)
    for i, h in enumerate(pool_h):
        # self-correlation (h with h) not with 1-h
        A[i, :] = np.correlate(h, h, mode='full')
    # T corresponds to min correlation
    w = cp.Variable(M, nonneg=True)
    T = cp.Variable()
    constraints = [cp.sum(w) == 1, A.T @ w >= T]
    prob = cp.Problem(cp.Maximize(T), constraints)
    # Solve with ECOS
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        prob.solve(solver=cp.SCS, max_iters=5000)
    # In case not solved, set default
    if w.value is None:
        return None, None
    w_opt = np.array(w.value).flatten()
    T_opt = T.value
    h_comb = np.dot(w_opt, np.array(pool_h))
    return h_comb, T_opt

import time
import numpy as np
import cvxpy as cp

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_shift_and_grad(h: np.ndarray):
    """Return (shift k, gradient) of the worst (max) overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad = np.zeros_like(h)
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        pos = -k
        grad = np.zeros_like(h)
        grad[pos:] = 1.0 - h[: n - pos]
        grad[: n - pos] -= h[pos:]
    return k, grad


def project_onto_capped_simplex(v: np.ndarray, target: float, max_iter=30, tol=1e-9):
    """Euclidean projection of v onto {0≤x≤1, Σx=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) <= tol:
            break
    return x


def adam_optimize(h0, target, end_time):
    """Adam‐style projected gradient descent on the worst–shift objective."""
    n = h0.size
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.1
    t = 0
    pool = []
    snap_interval = 50
    while time.time() < end_time:
        t += 1
        _, grad = worst_shift_and_grad(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / (1 + 0.001 * t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_capped_simplex(h, target)
        if t % snap_interval == 0:
            pool.append(h.copy())
    return h, pool


def solve_lp(pool_h, target, dx, time_left):
    """Maximize the minimum self‑correlation over the convex hull of pool_h."""
    M = len(pool_h)
    n = pool_h[0].size
    K = 2 * n - 1
    A = np.empty((M, K), dtype=np.float64)
    for i, h in enumerate(pool_h):
        A[i, :] = np.correlate(h, h, mode="full")
    w = cp.Variable(M, nonneg=True)
    T = cp.Variable()
    constraints = [cp.sum(w) == 1, A.T @ w >= T]
    prob = cp.Problem(cp.Maximize(T), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=4000, abstol=1e-8, reltol=1e-6)
    except Exception:
        prob.solve(solver=cp.SCS, max_iters=4000, verbose=False)
    if w.value is None:
        return None, None
    w_opt = np.array(w.value).ravel()
    T_opt = T.value
    h_comb = np.dot(w_opt, np.vstack(pool_h))
    h_comb = np.clip(h_comb, 0.0, 1.0)
    h_comb = project_onto_capped_simplex(h_comb, target)
    return h_comb, T_opt


def final_adam(h_start, target, end_time):
    """Short Adam run for polishing."""
    n = h_start.size
    h = h_start.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05
    t = 0
    while time.time() < end_time:
        t += 1
        _, grad = worst_shift_and_grad(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / (1 + 0.001 * t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_capped_simplex(h, target)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0
    dx = 2.0 / n

    # ---- initial vector ----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_capped_simplex(h0, target_sum)
    else:
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- Adam exploration --------------------------------------------------
    adam_budget = budget_s * 0.6
    h_adam, snapshots = adam_optimize(h0, target_sum, start + adam_budget)

    pool = [h0] + snapshots + [h_adam]

    # add a few random binary vectors for hull diversity
    for _ in range(30):
        b = np.zeros(n)
        b[rng.choice(n, n // 2, replace=False)] = 1.0
        pool.append(b)

    # ---- Linear programming refinement --------------------------------------
    elapsed = time.time() - start
    time_left = budget_s - elapsed - 1.0
    if time_left > 20.0:
        h_lp, T_opt = solve_lp(pool, target_sum, dx, time_left)
        if h_lp is not None:
            c_lp = compute_c5(h_lp)
            if c_lp < best_c:
                best_c = c_lp
                best_h = h_lp.copy()

    # ---- Final polishing ---------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed - 0.5
    if remaining > 5.0:
        best_h = final_adam(best_h, target_sum, time.time() + remaining)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
