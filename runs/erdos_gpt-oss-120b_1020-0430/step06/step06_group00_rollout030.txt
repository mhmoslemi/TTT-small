        # random binary start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        # store positions
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        # inner SA loop with decreasing T
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            # linear cooling
            T = max(0.001, 0.02 * (1 - elapsed / time_budget))
            # propose swap
            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / T):
                c_curr = c_new
                # accept swap
                ones_arr[i_idx] = j
                zeros_arr[j_idx] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0

def adam_optimize(h0, rng, time_budget, beta_schedule=None):
    start = time.time()
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0

    # Define schedule function
    def beta_for_step(s):
        if s < 500:
            return 1.0
        elif s < 2000:
            return 5.0
        elif s < 8000:
            return 20.0
        elif s < 20000:
            return 80.0
        elif s < 50000:
            return 200.0
        elif s < 120000:
            return 800.0
        elif s < 250000:
            return 3000.0
        elif s < 400000:
            return 8000.0
        else:
            return 20000.0

    # base learning rate
    base_lr = 0.05

    while time.time() - start < time_budget:
        beta = beta_for_step(step)
        # correlation
        corr = np.correlate(h, h, mode='full')
        r_max = np.max(corr)
        # softmax weights
        exp_term = np.exp(beta * (corr - r_max))
        sum_exp = np.sum(exp_term)
        p = exp_term / sum_exp
        # gradient via convolution
        conv = np.convolve(p, h, mode='full')
        grad = 2.0 * conv[n-1:n-1+n]
        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        lr = base_lr / np.sqrt(beta)  # scale learning rate with beta
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Project onto box and sum constraint
        h = project_sum_box(h, n/2)
        # Periodically evaluate
        if step % 500 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                # Early exit if good enough
                if best_c <= 0.3808:
                    return best_h, best_c
        step += 1
    # final best
    return best_h, best_c

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h, best_c = h.copy(), compute_c5(h)

    def beta_for_step(s):
        if s < 500:
            return 1.0
        if s < 2000:
            return 5.0
        if s < 8000:
            return 20.0
        if s < 20000:
            return 80.0
        if s < 50000:
            return 200.0
        if s < 120000:
            return 800.0
        if s < 250000:
            return 3000.0
        if s < 400000:
            return 8000.0
        return 20000.0

    step = 0
    base_lr = 0.05
    while time.time() - start < time_budget:
        beta = beta_for_step(step)
        corr = np.correlate(h, h, mode="full")
        r_max = np.max(corr)
        exp_term = np.exp(beta * (corr - r_max))
        sum_exp = np.sum(exp_term)
        p = exp_term / sum_exp
        conv = np.convolve(p, h, mode="full")
        grad = 2.0 * conv[n - 1:n - 1 + n]
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        lr = base_lr / np.sqrt(beta)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum_box(h, n / 2, rng)
        if step % 500 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        step += 1
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum_box(h0, h0.size / 2, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # Phase 0: binary SA on 256 points
    rem = total - (time.time() - start)
    if rem > 0:
        t_bin = min(rem * 0.15, 200.0)
        h_bin, c_bin = binary_sa(256, rng, t_bin)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, 256

    # upsample to 512
    rem = total - (time.time() - start)
    if rem > 0 and best_h is not None and best_n < 512:
        h_mid = upsample_h(best_h, 512)
        h_mid = np.clip(h_mid, 0.0, 1.0)
        h_mid = project_sum_box(h_mid, 512 / 2, rng)
        c_mid = compute_c5(h_mid)
        if c_mid < best_c:
            best_h, best_c, best_n = h_mid, c_mid, 512

    # Adam on 512 (if we have it)
    rem = total - (time.time() - start)
    if rem > 0 and best_h is not None and best_n == 512:
        h_opt, c_opt = adam_optimize(best_h, rng, rem * 0.5)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # upsample to 1024
    rem = total - (time.time() - start)
    if rem > 0 and best_h is not None and best_n < 1024:
        h_fine = upsample_h(best_h, 1024)
        h_fine = np.clip(h_fine, 0.0, 1.0)
        h_fine = project_sum_box(h_fine, 1024 / 2, rng)
        c_fine = compute_c5(h_fine)
        if c_fine < best_c:
            best_h, best_c, best_n = h_fine, c_fine, 1024

    # final Adam on 1024
    rem = total - (time.time() - start)
    if rem > 0 and best_h is not None:
        h_opt2, c_opt2 = adam_optimize(best_h, rng, rem * 0.9)
        if c_opt2 < best_c:
            best_h, best_c = h_opt2, c_opt2

    # final gradient tidy‑up
    rem = total - (time.time() - start)
    if rem > 0 and best_h is not None:
        h_final = gradient_refine(best_h, rng, step=0.015, max_iter=30)
        c_final = compute_c5(h_final)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, best_h.size / 2, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
