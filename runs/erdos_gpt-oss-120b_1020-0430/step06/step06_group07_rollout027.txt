def smooth_obj(h):
    O = 1 - np.correlate(h, h, mode='full') * dx  # O_k as array of length 2*n-1
    # often O is near 0.38.
    # compute softmax:
    M = np.max(beta * O)
    exp_beta_O = np.exp(beta * O - M)
    s = exp_beta_O.sum()
    w = exp_beta_O / s
    # compute gradient
    # w as shape 2n-1, indices k from -(n-1) to n-1
    w_arr = w
    conv_back = np.convolve(w_arr, h, mode='full')[n-1:n-1+n]
    w_rev = w_arr[::-1]
    conv_fwd = np.convolve(w_rev, h, mode='full')[n-1:n-1+n]
    grad = -(conv_fwd + conv_back)  # gradient of softmax obj
    # penalty
    sum_h = h.sum()
    diff = sum_h - target
    grad += 2 * penalty * diff * np.ones_like(h)
    # objective value
    obj = (np.log(s) + M) / beta + penalty * diff * diff
    return obj, grad

def adam_optimize(h, beta, penalty, step_size, max_iters, rng):
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1=0.9; beta2=0.999; eps=1e-8
    for t in range(1, max_iters+1):
        obj, grad = smooth_obj(h, beta, penalty)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        h_new = h - step_size * m_hat / (np.sqrt(v_hat)+eps)
        h_new = project_box_sum(h_new, target)
        c = compute_c5(h_new)
        if c < best_c: best_c = c; best_h = h_new.copy()
        h = h_new
        # maybe reduce step_size gradually
        if time.time() - start > budget: break
    return best_h, best_c

def smooth_obj(h, beta, penalty, target):
    n = h.size
    dx = 2.0 / n
    # autocorrelation
    corr = np.correlate(h, h, mode='full')
    O = 1.0 - corr * dx
    # softmax
    # to avoid overflow, subtract max beta*O
    # Actually compute maxBetaO = beta * O_max
    max_beta_O = beta * O.max()
    exp_beta_O = np.exp(beta * O - max_beta_O)
    sum_exp = exp_beta_O.sum()
    w = exp_beta_O / sum_exp
    # gradient of softmax objective
    # compute convolution of w with h
    # w array length 2n-1, indexing from -(n-1) to n-1
    conv_back = np.convolve(w, h, mode='full')[n-1:n-1+n]
    w_rev = w[::-1]
    conv_fwd = np.convolve(w_rev, h, mode='full')[n-1:n-1+n]
    grad = -(conv_fwd + conv_back)
    # penalty term
    diff = h.sum() - target
    grad += 2 * penalty * diff * np.ones_like(h)
    # objective value
    obj = (np.log(sum_exp) + max_beta_O) / beta + penalty * diff * diff
    return obj, grad

def compute_c5(h: np.ndarray) -> float:

    lo, hi = -1.0, 1.0

        else:

    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

# Gradient for worst shift (used by hill‑climb)
def worst_gradient(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, h, mode="full")
    k = int(np.argmin(corr[n - 1 :]))  # 0 <= k < n
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def upsample(h: np.ndarray, rng: np.random.Generator):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)

# Smooth max objective and its gradient
def smooth_obj(h: np.ndarray, beta: float, penalty: float, target: float):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, h, mode="full")
    O = 1.0 - corr * dx                     # O_k for all shifts
    max_beta_O = beta * O.max()
    exp_beta_O = np.exp(beta * O - max_beta_O)
    sum_exp = exp_beta_O.sum()
    w = exp_beta_O / sum_exp                # softmax weights

    # gradient of (1/beta) log sum exp(beta O)
    conv_back = np.convolve(w, h, mode="full")[n - 1 : n - 1 + n]
    w_rev = w[::-1]
    conv_fwd = np.convolve(w_rev, h, mode="full")[n - 1 : n - 1 + n]

    diff = h.sum() - target
    grad += 2.0 * penalty * diff * np.ones_like(h)

    obj = (np.log(sum_exp) + max_beta_O) / beta + penalty * diff * diff
    return obj, grad
