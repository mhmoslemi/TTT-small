def ternary_search(f, lo, hi, iters=30):
    for _ in range(iters):
        m1 = lo + (hi-lo)/3
        m2 = hi - (hi-lo)/3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # if diff >0, need to add mass; else remove.
    # We'll random redistribute per iteration as previous.
    # But could be O(n^2) diff loops if diff large. Typically diff will be small (<1e-12) after modifications due to exact sum changes. So loop fine.
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 1024                     # number of discretisation points
    dx = 2.0 / n

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        if h.size != n:
            # linear interpolation to required size
            x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, h)
    else:
        h = rng.random(n)

    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    # ----- first evaluation ---------------------------------------------
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr * dx
    best_c = O.max()
    best_h = h.copy()
    no_improve = 0
    MIN_EPS = 1e-12

    # ----- main optimisation loop ---------------------------------------
    while time.time() - start_time < budget_s * 0.999:
        # recompute correlation and overlaps
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr * dx

        # worst shift (index in full correlation)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        k_abs = abs(shift)

        # gradient of the worst shift
        if k_abs == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_e = float(min(h[i], 1.0 - h[j]))
        if max_e < MIN_EPS:
            # look for any feasible pair
            tries = 0
            while max_e < MIN_EPS and tries < 10:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                max_e = float(min(h[i], 1.0 - h[j]))
                tries += 1
            if max_e < MIN_EPS:
                break

        # --- construct linear coefficients a_k for O_k(e) = O_k + a_k*e*dx ---
        a = np.zeros_like(corr)          # length 2n-1

        # term from (1 - h) at i+k (negative contribution)
        start1 = n - 1 - i
        a[start1:start1 + n] -= (1.0 - h)

        # term from (1 - h) at j+k (positive contribution)
        start2 = n - 1 - j
        a[start2:start2 + n] += (1.0 - h)

        # reversed h for i-k and j-k terms
        h_rev = h[::-1]                  # view, no copy
        a[i:i + n] += h_rev              # +h_{i-k}
        a[j:j + n] -= h_rev              # -h_{j-k}

        # --- ternary search for optimal epsilon -------------------------
        lo, hi = 0.0, max_e
        if hi <= 0.0:
            continue

        O_loc = O
        a_loc = a
        dx_loc = dx

        def f(e):
            return np.max(O_loc + a_loc * dx_loc * e)

        for _ in range(25):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if f(m1) < f(m2):
                hi = m2
            else:
                lo = m1
        e_opt = (lo + hi) / 2.0

        if e_opt < MIN_EPS:
            # tiny move – try a modest fraction of max_e
            e_opt = max_e * 0.02
            if f(e_opt) >= O.max():
                no_improve += 1
                if no_improve > 5000:
                    # random jitter to escape plateau
                    h = best_h.copy()
                    delta = (rng.random(n) - 0.5) * 0.04
                    h += delta
                    h = np.clip(h, 0.0, 1.0)
                    h = enforce_sum(h, rng)
                    no_improve = 0
                continue

        # apply the transfer
        h[i] -= e_opt
        h[j] += e_opt
        # enforce sum (tiny correction for rounding)
        h = enforce_sum(h, rng)

        # new overlap value
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr * dx
        cur_c = O.max()

        if cur_c + 1e-12 < best_c:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                h = best_h.copy()
                delta = (rng.random(n) - 0.5) * 0.04
                h += delta
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
                no_improve = 0

    # ----- final sanitisation -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
