    return idx, k, corr

    return grad

def guided_perturb(h, grad, rng, scale):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))

    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

        norm = np.linalg.norm(grad) + 1e-12
        step = step0 * (1.0 - it / iters) / norm
        v = h - step * grad
        h = project_onto_box_sum(v, target)
    return h

def guided_hill_climb(h0, rng, iters=3000, step_scale=0.4):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    n = h.size
    decay = max(1, iters // 5)
    step = step_scale
    for it in range(iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0:
            eps = rng.random() * max_eps * step
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
        else:
            nh = h.copy()
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def build_lp(n):
    ny = n * (n + 1) // 2
    total_cols = n + ny + 1          # h vars, y vars, alpha
    alpha_idx = total_cols - 1

    rows, cols, data = [], [], []
    b_ub = []

    # mapping (i,k) -> idx
    y_idx = {}
    counter = 0
    for i in range(n):
        for k in range(n - i):
            idx = n + counter
            y_idx[(i, k)] = idx
            # y - h_i <= 0
            rows.append(3 * counter)
            cols.append(idx); data.append(1.0)
            rows.append(3 * counter)
            cols.append(i); data.append(-1.0)
            b_ub.append(0.0)
            # y - h_{i+k} <= 0
            rows.append(3 * counter + 1)
            cols.append(idx); data.append(1.0)
            rows.append(3 * counter + 1)
            cols.append(i + k); data.append(-1.0)
            b_ub.append(0.0)
            # -y + h_i + h_{i+k} <= 1
            rows.append(3 * counter + 2)
            cols.append(idx); data.append(-1.0)
            rows.append(3 * counter + 2)
            cols.append(i); data.append(1.0)
            rows.append(3 * counter + 2)
            cols.append(i + k); data.append(1.0)
            b_ub.append(1.0)
            counter += 1

    # shift constraints: -sum y_{i,k} + alpha <= 0
    base = 3 * ny
    for k in range(n):
        rows.append(base + k)
        cols.append(alpha_idx); data.append(1.0)
        b_ub.append(0.0)
        for i in range(n - k):
            rows.append(base + k)
            cols.append(y_idx[(i, k)])
            data.append(-1.0)

    A_ub = sparse.csc_matrix((data, (rows, cols)), shape=(3 * ny + n, total_cols))

    # equality: sum h_i = n/2
    eq_data = np.ones(n)
    eq_rows = np.zeros(n, dtype=int)
    eq_cols = np.arange(n, dtype=int)
    A_eq = sparse.csc_matrix((eq_data, (eq_rows, eq_cols)), shape=(1, total_cols))

    # bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * ny + [(None, None)]

    # objective: maximize alpha -> minimize -alpha
    c = np.zeros(total_cols)
    c[alpha_idx] = -1.0
    return c, A_ub, np.array(b_ub), A_eq, np.array([n / 2.0]), bounds

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 200
    target = n / 2.0

    # 1) linear programming relaxation
    c, A_ub, b_ub, A_eq, b_eq, bounds = build_lp(n)
    lp_res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': max(1, int(budget_s * 0.2))})
    if not lp_res.success:
        # fallback: random binary start
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        h0 = lp_res.x[:n]
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # 2) intensive local optimisation
    # a) subgradient descent
    h = subgradient_descent(best_h, rng, iters=5000, step0=0.6)
    # b) guided hill climb
    h, c = guided_hill_climb(h, rng, iters=8000, step_scale=0.35)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # 3) occasional random perturbation phase
    while time.time() - start < budget_s * 0.95:
        # small random swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.05
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
