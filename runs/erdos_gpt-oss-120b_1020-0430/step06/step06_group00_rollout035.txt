def compute_c5_fft(h):
    n = len(h)
    # convolution h * (1-h) reversed? Actually correlation h * (1-h) (not reversed) as in np.correlate. For correlation, we can compute FFT of h and FFT of (1-h) and multiply conj?
    # np.correlate(h, g, mode='full') = convolution of h with reversed g, i.e., (h * g_rev)[...] .
    # To compute correlation using FFT: compute FFT of h, FFT of g, then multiply h_fft * conj(g_fft) (if correlation of h with g). Since g is real, g_rev's FFT is conj(g_fft). So correlation = ifft( h_fft * conj(g_fft) ).
    # Thus:
    g = 1.0 - h
    H = np.fft.rfft(h, n*2-1)
    G = np.fft.rfft(g, n*2-1)
    corr = np.fft.irfft(H * np.conj(G), n*2-1)
    # Should be same as np.correlate.
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def enforce_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to add mass
        free = 1.0 - h
        cap = free.sum()
        if cap <= 0:
            return h
        inc = diff * free / cap

def binary_sa_fft(n, rng, time_budget):
    start = time.time()
    best_h = None
    best_c = np.inf

    # initial random binary vector
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    curr_c = compute_c5(h)

    # store ones_set and zeros_set arrays for quick swap
    ones_arr = np.sort(ones)
    zeros_arr = np.setdiff1d(np.arange(n), ones_arr)

    # parameters
    T0 = 0.05
    T_end = 1e-5

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * ((T_end / T0) ** (elapsed / time_budget))
        # pick random one and zero
        i = rng.choice(ones_arr)
        j = rng.choice(zeros_arr)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            curr_c = new_c
            # update arrays
            # replace i with j in ones_arr, j with i in zeros_arr
            # find positions
            pos_i = np.searchsorted(ones_arr, i)
            # pos_i is index of i in sorted ones_arr
            # replace at pos_i
            ones_arr[pos_i] = j
            # sort maybe after each swap? Instead, we can just maintain unsorted arrays and use random choice; we don't need sorted for performance.
            # Simpler: keep unsorted arrays
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        # track best
        if curr_c < best_c:
            best_c = curr_c
            best_h = h.copy()
            if best_c <= 0.3808:
                break

    return best_h, best_c

def gradient_descent(h, rng, time_budget, init_step=0.05, decay=0.99):
    start = time.time()
    step = init_step

    best_c = compute_c5(h)

        # So update h = h - step*grad

        if c_new <= best_c or rng.random() < np.exp(-(c_new - best_c) / max(step, 1e-12)):
            h = h_new
        # decay step
        step *= decay

### project_sum_target

def project_sum_target(h, target):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Use water-filling
    for _ in range(10):
        if diff > 0:
            free = 1.0 - h
            cap = free.sum()
            if cap <= 0:
                break
            inc = diff * free / cap
            h += inc
        else:
            avail = h
            cap = avail.sum()
            if cap <= 0:
                break
            dec = (-diff) * avail / cap
            h -= dec
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def binary_sa_fft(n, rng, time_budget):
    start = time.time()
    # initial binary vector
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    # parameters
    T0 = 0.05
    T_end = 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * ((T_end / T0) ** (elapsed / time_budget))
        # pick random i,j distinct
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j = j + 1
        # ensure swapping different values
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

import time
import numpy as np
import scipy.optimize as opt

# ---------- Core utilities ----------
def compute_corr_full(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    size = 2 * n - 1
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    n = h.shape[0]
    max_corr = np.max(compute_corr_full(h))
    return max_corr * (2.0 / n)


def project_sum_target(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(10):
        if diff > 0:
            free = 1.0 - h
            cap = free.sum()
            if cap <= 0:
                break
            inc = diff * free / cap
            h += inc
        else:
            avail = h
            cap = avail.sum()
            if cap <= 0:
                break
            dec = (-diff) * avail / cap
            h -= dec
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h


def worst_gradient(h: np.ndarray):
    """Gradient of the shift attaining the maximal overlap."""
    n = h.shape[0]
    corr = compute_corr_full(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        k_abs = abs(k)
        grad[: n - k_abs] = 1.0 - h[k_abs:]
        grad[k_abs:] -= h[: n - k_abs]
    return grad, np.max(corr)


# ---------- Global optimisation ----------
def gradient_descent(h, rng, time_budget, init_step=0.02, decay=0.995):
    """Projected gradient descent on the worst‑shift subgradient."""
    start = time.time()
    step = init_step
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    curr_h = h.copy()
    curr_c = best_c
    while time.time() - start < time_budget:
        grad, _ = worst_gradient(curr_h)
        new_h = curr_h - step * grad
        new_h = project_sum_target(new_h, target)
        new_c = compute_c5(new_h)
        # Metropolis‑type acceptance to allow occasional uphill moves
        if new_c < curr_c or rng.random() < np.exp(-(new_c - curr_c) / max(step, 1e-12)):
            curr_h, curr_c = new_h, new_c
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()
        step *= decay
    return best_h, best_c


def hill_climb_small(h, rng, time_budget, eps=0.0005):
    """Tiny pairwise mass moves."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(best_h)
    while time.time() - start < time_budget:
        i = rng.integers(n)
        if best_h[i] < eps:
            continue
        j = rng.integers(n)
        if best_h[j] > 1.0 - eps:
            continue
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h
    return best_h, best_c


def binary_sa_fft(n, rng, time_budget):
    """Simulated annealing on binary vectors, using FFT for fast C5."""
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    T0, T_end = 0.05, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * ((T_end / T0) ** (elapsed / time_budget))
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]          # swap
        new_c = compute_c5(h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
                if best_c <= 0.3808:
                    break
        else:
            h[i], h[j] = h[j], h[i]      # revert
    return best_h, best_c


def de_optimize(m, N, rng, time_budget):
    """Low‑dimensional differential‑evolution on a block‑constant ansatz."""
    de_start = time.time()
    block_size = N // m
    target = N / 2.0

    def build_h(u):
        hm = m / 2.0 - np.sum(u)
        blocks = np.concatenate([u, [hm]])
        return np.repeat(blocks, block_size)

    def obj(u):
        hm = m / 2.0 - np.sum(u)
        if hm < 0.0 or hm > 1.0:
            return 10.0
        h = np.repeat(np.concatenate([u, [hm]]), block_size)
        return compute_c5(h)

    bounds = [(0.0, 1.0)] * (m - 1)
    callback = lambda x, convergence: (time.time() - de_start) > time_budget
    result = opt.differential_evolution(
        obj,
        bounds,
        maxiter=2000,
        tol=1e-8,
        polish=False,
        callback=callback,
        init='random',
        seed=int(rng.integers(2**32)),
    )
    best_h = build_h(result.x)
    best_c = compute_c5(best_h)
    return best_h, best_c
