import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility functions
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# --------------------------------------------------------------
# Soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Gradient refinement (worst shift)
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.025, max_iter: int = 30) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

        h[ones] = 1.0

        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            # pick a random 1 and 0 to swap
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

# Continuous simulated annealing
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----------------------------------------------------------
    # Base resolution
    N = 1024
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size < N:
            h0 = upsample_h(h0, N)
    else:
        h0 = rng.random(N)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # ----------------------------------------------------------
    # Phase 0: binary SA on 512 points (optional)
    remaining = total_budget - (time.time() - start)
    if remaining > 130.0:
        t_bin = min(remaining * 0.12, 130.0)
        h_bin, c_bin = binary_sa(512, rng, t_bin)
        if c_bin < best_c:
            # promote binary to continuous (float) and upsample
            h_bin_f = h_bin.astype(float)
            if h_bin_f.size < N:
                h_bin_f = upsample_h(h_bin_f, N)
            h_bin_f = enforce_sum(h_bin_f, rng)
            best_h, best_c = h_bin_f, c_bin

    # ----------------------------------------------------------
    # Phase 1: SLSQP restarts on N points
    betas = [20., 80., 200., 800., 3000., 8000., 20000.]
    while True:
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        if remaining < 8.0:
            break
        # small random perturbation of best_h
        h = best_h.copy()
        h += rng.normal(scale=0.02, size=N)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        # progressive soft‑max SLSQP
        for beta in betas:
            iters = 800 if beta >= 2000 else 500
            h = slsqp_optimize(h, beta, rng, max_iter=iters)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c
            if best_c <= 0.3808:
                break

    # ----------------------------------------------------------
    # Phase 2: Gradient tidy‑up (if time permits)
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 4.0 and best_c > 0.3808:
        h = gradient_refine(best_h, rng, step=0.025, max_iter=30)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c

    # ----------------------------------------------------------
    # Phase 3: Continuous SA on remaining time
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 2.0 and best_c > 0.3808:
        h, c = continuous_sa(best_h, rng, remaining)
        if c < best_c:
            best_h, best_c = h, c

    # ----------------------------------------------------------
    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
