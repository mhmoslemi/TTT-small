Implementation:

    n = h.size
    b = 1.0 - h
    # linear correlation: for k from -(n-1) to n-1:
    # we can compute convolution of h and reversed b.
    b_rev = b[::-1]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    B = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(H * B, L)[:2*n-1]
    # conv[k] corresponds to shift index = k - (n-1)
    # max overlap
    max_overlap = conv.max()
    dx = 2.0 / n
    return max_overlap * dx

# ---------- correlation helpers ----------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def compute_corr(h: np.ndarray) -> np.ndarray:
    """Linear correlation v_k = Σ_i h_i (1-h_{i+k}) for k=-(n-1)..(n-1)."""
    n = h.size
    b = 1.0 - h
    b_rev = b[::-1]
    L = _next_pow2(2 * n - 1)
    H = np.fft.rfft(h, L)
    B = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(H * B, L)[: 2 * n - 1]
    return conv

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for discretised h."""
    n = h.size
    return float(compute_corr(h).max() * (2.0 / n))

def compute_c5_and_corr(h: np.ndarray):
    """Return both C5 and the full correlation vector."""
    n = h.size
    v = compute_corr(h)

# ---------- exact sum adjustment ----------
def adjust_sum_exact(x: np.ndarray, rng: np.random.Generator, max_iter: int = 2000):
    """Modify x in‑place so that Σ x = len(x)/2 exactly (x∈[0,1])."""
    n = x.size
    target = n / 2.0
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

# ---------- smooth‑max gradient ----------
def gradient_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    n = h.size
    v = compute_corr(h)
    m = np.max(beta * v)
    w = np.exp(beta * v - m)
    w /= w.sum()
    g = np.zeros_like(h)
    omh = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:

            g[i:] -= wi * h[: n - i]
        else:
            ii = -i
            g[ii:] += wi * omh[: n - ii]
            g[: n - ii] -= wi * h[ii:]
    return g

                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best = c, cand
            h = best
    return h

# ---------- worst‑shift sub‑gradient ----------
def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    v = compute_corr(h)
    mx = v.max()
    idxs = np.where(v == mx)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def worst_shift_refine(h: np.ndarray, rng: np.random.Generator,
                       base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best = c, cand
        h = best
        step *= 0.95
    return h

# ---------- SLSQP on smooth‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    v = compute_corr(x)
    bc = CURRENT_BETA * v
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum repair
    adjust_sum_exact(x, rng)
    return x

# ---------- binary simulated‑annealing ----------
def binary_sa_optimize(h0: np.ndarray, rng: np.random.Generator,
                       deadline) -> tuple:
    """SA on binary h; returns best (h, C5) seen before deadline."""
    h = h0.copy()
    cur_c, cur_v = compute_c5_and_corr(h)
    best_h = h.copy()
    best_c = cur_c

    # Main loop
    T = cur_c * 0.04 + 1e-12
    decay = 0.99996
    while time.time() < deadline - 0.1 and T > 1e-14:
        n = h.size
        # worst shift info (already have cur_v)
        idx = cur_v.argmax()
        s = idx - (n - 1)

        # candidate i (a 1 that contributes to the worst shift, if any)
        if s >= 0:
            mask = (h[: n - s] == 1) & (h[s:] == 0)
            i_cands = np.flatnonzero(mask)
        else:
            i_cands = np.flatnonzero(h)  # fallback
        if i_cands.size:
            i = rng.choice(i_cands)
        else:
            i = rng.choice(np.flatnonzero(h))

        # candidate j (a zero, any)
        j = rng.choice(np.flatnonzero(h == 0))

        # perform swap
        h[i] = 0
        h[j] = 1
        new_c, new_v = compute_c5_and_corr(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c, cur_v = new_c, new_v
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        T *= decay
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    # grid sizes to explore
    sizes = [720, 960, 1080]

    for n in sizes:
        if time.time() > deadline:
            break
        ones_target = n // 2

        # build initial candidates
        candidates = []

        parent = globals().get("initial_h_values", None)
        if parent is not None:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            h_bin = np.rint(interp).astype(int)
            cur = h_bin.sum()
            diff = ones_target - cur
            if diff > 0:
                zero_idxs = np.flatnonzero(h_bin == 0)
                choose = rng.choice(zero_idxs, diff, replace=False)
                h_bin[choose] = 1
            elif diff < 0:
                one_idxs = np.flatnonzero(h_bin == 1)
                choose = rng.choice(one_idxs, -diff, replace=False)
                h_bin[choose] = 0
            candidates.append(h_bin)

        # a few random binaries
        for _ in range(2):
            hb = np.zeros(n, dtype=int)
            hb[:ones_target] = 1
            rng.shuffle(hb)
            candidates.append(hb)

        # process each candidate
        for h0 in candidates:
            if time.time() > deadline:
                break

            # ---------- binary SA ----------
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
            h_sa, c_sa = binary_sa_optimize(h0, rng, sub_dead)
            if c_sa < best_c:
                best_c, best_h, best_n = c_sa, h_sa.copy(), n

            # ---------- continuous polishing ----------
            # turn binary into float and run a light gradient + SLSQP pass
            h_float = h_sa.astype(float)

            betas = [5, 20, 80, 200, 500, 2000, 5000, 20000]
            it_per_beta = 6
            h_float = gradient_descent_smoothmax(h_float, betas, 0.2,
                                                 it_per_beta, rng, deadline)

            h_float = worst_shift_refine(h_float, rng, 0.12, 12, deadline)

            for beta in (20000, 50000):
                if time.time() > deadline:
                    break
                h_float = slsqp_optimize(h_float, beta, rng, max_iter=400)

            c_float = compute_c5(h_float)
            if c_float < best_c:
                best_c, best_h, best_n = c_float, h_float.copy(), n

    # ---------- optional up‑sampling ----------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n2 = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)

        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        # brief SA on up‑sampled version
        h_up, c_up = binary_sa_optimize(h_up, rng, sub_dead)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n2

        # final polish
        h_up = h_up.astype(float)
        betas = [5, 20, 80, 200, 500, 2000, 5000, 20000]
        h_up = gradient_descent_smoothmax(h_up, betas, 0.2, 5, rng, deadline)
        h_up = worst_shift_refine(h_up, rng, 0.12, 8, deadline)
        for beta in (20000, 50000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n2

    # ---------- final safety ----------
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
