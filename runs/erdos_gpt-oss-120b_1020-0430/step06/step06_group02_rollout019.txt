def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def anneal_optimize(h_init: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simulated‑annealing / subgradient search on a fixed grid."""
    n = h_init.size
    h = h_init.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    T0 = 0.03               # initial temperature
    T = T0
    cooling = 0.9995        # exponential cooling
    it = 0
    last_improve = 0
    max_no_improve = 20000

    while time.time() < time_limit:
        # decide move type
        if rng.random() < 0.15:                      # random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * (T / T0)
            i_move, j_move = i, j
        else:                                        # gradient‑guided move
            grad = worst_gradient(h)
            i_move = int(np.argmax(grad))
            j_move = int(np.argmin(grad))
            if i_move == j_move:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * (T / T0)
                i_move, j_move = i, j
            else:
                max_eps = min(h[i_move], 1.0 - h[j_move])
                if max_eps <= 0:
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i == j:
                        continue
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps <= 0:
                        continue
                    eps = max_eps * rng.random() * (T / T0)
                    i_move, j_move = i, j
                else:
                    eps = max_eps * rng.random() * (T / T0)

        # propose move
        h_new = h.copy()
        h_new[i_move] -= eps
        h_new[j_move] += eps
        c_new = compute_c5(h_new)
        delta = c_new - cur_c

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                last_improve = it
                if best_c <= 0.3808:
                    break

        # cooling
        T = max(T * cooling, 1e-6)
        it += 1

        # restart if stagnant
        if it - last_improve > max_no_improve:
            h = best_h.copy()
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * 0.02
                h[i] -= eps
                h[j] += eps
            cur_c = compute_c5(h)
            T = T0
            last_improve = it

    return best_h, best_c


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence as floats 0/1."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # --- collect a few straightforward candidates ------------------------
    candidates = []

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        h0 = np.clip(h0, 0.0, 1.0)
        target0 = h0.size / 2.0
        diff = target0 - h0.sum()
        while abs(diff) > 1e-12:
            if diff > 0:
                idx = np.where(h0 < 1.0)[0]
                if idx.size == 0:
                    break
                i = rng.choice(idx)
                add = min(diff, 1.0 - h0[i])
            else:
                idx = np.where(h0 > 0.0)[0]
                if idx.size == 0:
                    break
                i = rng.choice(idx)
                add = max(diff, -h0[i])
            h0[i] += add
            diff = target0 - h0.sum()
        candidates.append((h0, compute_c5(h0)))

    tm = thue_morse_bits(512)
    candidates.append((tm, compute_c5(tm)))

    n_bal = 512
    bal = np.zeros(n_bal)
    bal[rng.choice(n_bal, n_bal // 2, replace=False)] = 1.0
    candidates.append((bal, compute_c5(bal)))

    # --- coarse optimisation (256 points) ------------------------------
    coarse_n = 256
    coarse_start = np.zeros(coarse_n)
    coarse_start[rng.choice(coarse_n, coarse_n // 2, replace=False)] = 1.0
    coarse_deadline = start + total_budget * 0.4
    coarse_best_h, coarse_best_c = anneal_optimize(coarse_start, coarse_deadline, rng)
    candidates.append((coarse_best_h, coarse_best_c))

    # --- upscale to full resolution ------------------------------------
    factor = 512 // coarse_n
    fine_start = np.repeat(coarse_best_h, factor)

    # --- fine optimisation (512 points) --------------------------------
    fine_deadline = start + total_budget
    fine_best_h, fine_best_c = anneal_optimize(fine_start, fine_deadline, rng)
    candidates.append((fine_best_h, fine_best_c))

    # --- choose the best found -----------------------------------------
    best_h, best_c = min(candidates, key=lambda t: t[1])

    # final tiny correction to enforce the exact integral
    n = best_h.size
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(best_h < 1.0)[0]
            if idx.size:
                best_h[rng.choice(idx)] += diff
        else:
            idx = np.where(best_h > 0.0)[0]
            if idx.size:
                best_h[rng.choice(idx)] += diff
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
