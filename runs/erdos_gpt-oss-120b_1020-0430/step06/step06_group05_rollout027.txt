Implement gradient_softmax(h, beta):
- Compute corr = np.correlate(h, 1 - h, mode='full')
- bc = beta * corr
- m = bc.max()
- exp_bc = np.exp(bc - m)
- w = exp_bc / exp_bc.sum()
- Now compute gradient = np.zeros_like(h)
- For each shift idx (0 .. 2n-2):
    - shift = idx - (n-1) (so shift can be negative)
    - weight = w[idx]
    - if shift >=0:
        - grad[:n-shift] += weight * (1 - h[shift:])   # (1 - h_{i+shift}) for i=0..n-1-shift
        - grad[shift:]  -= weight * h[:n-shift]       # -h_i for i+shift = j => h_i part
    - else:
        - s = -shift
        - grad[s:] += weight * (1 - h[:n-s])          # (1 - h_{i-s}) for i=s..n-1 (since shift negative moves other way)
        - grad[:n-s] -= weight * h[s:]                # -h_{i+s}
- return gradient.

# ---------- projection onto capped simplex ----------
def project_onto_capped_simplex(v: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum x = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    pos = grad > 1e-12
    neg = grad < -1e-12

        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])

    if total_in > 0.0:

            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                if best_c < 0.38079:
                    break

    # ----------- final polishing with SLSQP ----------
    if time.time() - start < budget_s * 0.95:
        for beta in (2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.97:
                break
            h_s = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=300)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # ----------- mass‑transfer refinement ----------
    for _ in range(5):
        _, g = worst_shift_and_grad(best_h)
        h_new = mass_transfer_move(best_h, g, alpha=0.05, target=target_sum, rng=rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # final enforcement and compute final C5
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target_sum, max_iter=30)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
