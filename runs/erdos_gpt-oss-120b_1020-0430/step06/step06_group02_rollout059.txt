import time
import numpy as np

# ----- global placeholders (set in run) ---------------------------------
G_n = None          # number of points
G_N = None          # 2 * G_n
G_fft_ones = None   # rfft of all‑ones vector of length G_N
G_h_pad = None      # h padded with zeros, length G_N

# ----------------------------------------------------------------------
def compute_c5():
    """C5 using pre‑computed padded array and FFT (fast linear correlation)."""
    global G_n, G_N, G_fft_ones, G_h_pad
    H = np.fft.rfft(G_h_pad)
    B = G_fft_ones - H                # (1‑h) in frequency domain
    corr = np.fft.irfft(H * B, n=G_N)[: 2 * G_n - 1]
    return np.max(corr) * (2.0 / G_n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)=len(h)/2 while 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 5000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (maximal) overlap shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)      # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # term h[i+k]
        grad[k:]    -= h[: n - k]       # term h[i‑k]
    return grad


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    global G_n, G_N, G_fft_ones, G_h_pad
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        n = h.size
    else:
        n = 512
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    # setup globals needed by compute_c5
    G_n = n
    G_N = 2 * n
    G_fft_ones = np.fft.rfft(np.ones(G_N, dtype=float))
    G_h_pad = np.concatenate([h, np.zeros(n, dtype=float)])

    cur_c = compute_c5()
    best_h = h.copy()
    best_c = cur_c

    # ----- index lists for fast swaps ----------------------------------
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] <= 0.5]

    # ----- simulated annealing -----------------------------------------
    T0 = 0.02                     # initial temperature
    T_end = 1e-8
    it = 0
    max_it_ann = 3_000_000       # safety cap
    while it < max_it_ann and time.time() - start < budget_s * 0.78:
        # geometric cooling
        T = max(T_end, T0 * (0.9995 ** it))

        # pick a random 1‑position and a random 0‑position
        pi = rng.integers(len(ones))
        pj = rng.integers(len(zeros))
        i = ones[pi]
        j = zeros[pj]

        # perform swap (maintaining padded copy)
        h[i] = 0.0
        h[j] = 1.0
        G_h_pad[i] = 0.0
        G_h_pad[j] = 1.0
        ones[pi] = j
        zeros[pj] = i

        c_new = compute_c5()
        d = c_new - cur_c
        if d < 0.0 or rng.random() < np.exp(-d / T):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            G_h_pad[i] = 1.0
            G_h_pad[j] = 0.0
            ones[pi] = i
            zeros[pj] = j

                j = zeros[pj]

                zeros[pj] = i
            c_new2 = compute_c5()
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h.copy()
                cur_c = c_new2
            else:

        it += 1

    # ----- bring best solution into working buffers ---------------------
    h = best_h.copy()
    G_h_pad[:n] = h
    cur_c = best_c

            best_h = nh.copy()
        h = nh
        cur_c = c_val

    # ----- final sanity enforcement -------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    G_h_pad[:n] = best_h
    final_c = compute_c5()

    return best_h.tolist(), float(final_c), int(best_h.size)
