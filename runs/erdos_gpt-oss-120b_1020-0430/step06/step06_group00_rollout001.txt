def enforce_sum(v, rng):
    target = v.size / 2.0
    diff = target - v.sum()
    n = v.size
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(n)
        if diff > 0:
            inc = min(diff, 1.0 - v[i])
            if inc > 0:
                v[i] += inc
                diff -= inc
        else:
            dec = min(-diff, v[i])
            if dec > 0:
                v[i] -= dec
                diff += dec
        it += 1
    # After loop, small diff may remain due to precision
    # Distribute evenly
    if abs(diff) > 1e-12:
        v += diff / n
        # Clip again
        v = np.clip(v, 0.0, 1.0)
        # Recompute diff and if still there, adjust small amounts
        # but this is unlikely
    return v

def _sum_eq_constraint(x):

    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun':_sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,

    diff = CURRENT_TARGET - x.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust v so that sum(v) == len(v)/2 while staying in [0,1]."""
    target = v.size / 2.0
    diff = target - v.sum()
    n = v.size
    max_iter = 20000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(n)
        if diff > 0:
            inc = min(diff, 1.0 - v[i])
            if inc > 0:
                v[i] += inc
                diff -= inc
        else:
            dec = min(-diff, v[i])
            if dec > 0:
                v[i] -= dec
                diff += dec
        it += 1
    if abs(diff) > 1e-12:
        # distribute remaining diff uniformly, then clip
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from length len(h) to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ----------------------------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """SLSQP on the soft‑max approximation of max‑overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, rng)
    return x


# ----------------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    """Iterative refinement using the gradient of the worst shift."""
    n = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated annealing on the continuous feasible set."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    N = h.size
    T0, T_end = 0.01, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - compute_c5(h)
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def remaining():
        return max(0.0, budget_s - (time.time() - start))

    # ---- initial solution -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n0 = 512
        h = rng.random(n0)
        h = enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- upsample to 1024 points -----------------------------------------
    if best_h.size < 1024:
        best_h = upsample_h(best_h, 1024)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)

    # ---- soft‑max SLSQP refinement -----------------------------------------
    betas = [20., 80., 200., 800., 3000., 8000., 20000., 50000.]
    for beta in betas:
        if remaining() < 5.0:
            break
        iters = 500 if beta < 8000 else 300
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=iters)
        best_c = compute_c5(best_h)
        if best_c <= 0.3808:
            break

    # ---- gradient tidy‑up --------------------------------------------------
    if remaining() > 5.0:
        best_h = gradient_refine(best_h, rng, step=0.015, max_iter=12)
        best_c = compute_c5(best_h)

    # ---- continuous SA on the finest grid ---------------------------------
    if remaining() > 5.0:
        best_h, best_c = continuous_sa(best_h, rng, remaining())

    # ---- final sanitisation ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
