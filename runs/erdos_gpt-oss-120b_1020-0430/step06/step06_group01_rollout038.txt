def hill_climb_diversified(h, rng, max_iters, step_scale=0.7, k_top=3):
    n = h.size
    target = n/2
    best_h = h.copy()
    best_c = np.max(np.correlate(h, 1-h, mode='full')) * (2.0 / n)
    dx = 2.0 / n
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        c = np.max(corr) * dx
        # choose shift among top k
        if k_top >= 1:
            top_idxs = np.argpartition(-corr, k_top)[:k_top]
            idx = int(rng.choice(top_idxs))
        else:
            idx = int(np.argmax(corr))
        k = abs(idx - (n - 1))
        # gradient for shift k
        grad = np.ones(n)
        if k == 0:
            grad = 1 - 2 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # step decay
        if (it+1) % (max_iters // 10 + 1) == 0:
            step_scale = max(step_scale * 0.8, 1e-5)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s
    best_h = None
    best_c = np.inf
    best_n = None

    # get initial solution if provided
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # target n
    target_n = 512
    if best_h is not None and best_n != target_n:
        # interpolate to target_n
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        best_h = np.clip(h_interp, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = target_n

    # Phase 1: binary SA
    time_left = lambda: total_budget - (time.time() - start)
    binary_time = total_budget * 0.20
    if time_left() > binary_time * 0.9:
        h_bin, c_bin = binary_sa(target_n, rng, binary_time)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_bin, target_n

    # Phase 2: diversified hill climb with restarts
    while time_left() > 10.0:
        # determine max_iters based on remaining budget
        # assume each iteration ~0.0005 sec? Let's set max_iters up to 20000
        remaining = time_left()
        max_iters = int(min(20000, remaining * 180))  # approx 180 it per sec
        # start from best_h with small random perturbation
        h_start = best_h.copy()
        # apply small random perturbation to maybe escape
        for _ in range(3):
            i = rng.integers(target_n)
            delta = (rng.random() - 0.5) * 0.2
            h_start[i] = np.clip(h_start[i] + delta, 0.0, 1.0)
        h_start = enforce_sum(h_start, rng)
        h_new, c_new = hill_climb_diversified(h_start, rng, max_iters, step_scale=0.7, k_top=3)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
            if best_c <= 0.3808:
                break

    # Final refinement
    if time_left() > 5.0:
        # small gradient refine
        for it in range(6):
            # using worst gradient only
            grad = worst_gradient(best_h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))

                    best_h[j] += eps

    final_c = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    return best_h.tolist(), float(final_c), int(target_n)

import time
import numpy as np

# -------------------- core utilities --------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (max C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# -------------------- binary simulated annealing --------------------

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = np.array(ones, dtype=int)
        zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = ones_arr[rng.integers(len(ones_arr))]
            j = zeros_arr[rng.integers(len(zeros_arr))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                idx_i = np.where(ones_arr == i)[0][0]
                idx_j = np.where(zeros_arr == j)[0][0]
                ones_arr[idx_i] = j
                zeros_arr[idx_j] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


# -------------------- diversified hill climbing --------------------

def hill_climb_diversified(h, rng, max_iters, step_scale=0.7, k_top=3):
    """Gradient‑directed climb using a random top‑k worst shift."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        c = np.max(corr) * dx
        if k_top > 1:
            top_idxs = np.argpartition(-corr, k_top)[:k_top]
            idx = int(rng.choice(top_idxs))
        else:
            idx = int(np.argmax(corr))
        k = abs(idx - (n - 1))
        grad = np.ones(n)
        if k == 0:
            grad = 1 - 2 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        if c < best_c:
            best_c, best_h = c, h.copy()
        # decay step scale periodically
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step_scale = max(step_scale * 0.8, 1e-5)
    return best_h, best_c

    # ---------- binary simulated annealing (≈20% time) ----------

    binary_time = total_budget * 0.20
