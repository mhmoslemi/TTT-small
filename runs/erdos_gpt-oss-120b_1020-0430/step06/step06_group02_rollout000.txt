import cvxpy as cp

def lp_step(h):
    n = h.shape[0]
    O, grads = compute_O_and_grads(h)
    d = cp.Variable(n)
    gamma = cp.Variable()
    constraints = [d >= -h, d <= 1 - h, cp.sum(d) == 0]
    for s in range(n):
        constraints.append(O[s] + grads[s] @ d <= gamma)
    prob = cp.Problem(cp.Minimize(gamma), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if d.value is None:
        return None, None, None
    return np.array(d.value), prob.value, O

import time, numpy as np, scipy.optimize as opt, cvxpy as cp

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 10000
    iter = 0
    while abs(diff) > 1e-12 and iter < max_iter:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        iter += 1
    # clamp final
    h[:] = np.clip(h, 0.0, 1.0)
    return h

global variables for constraints etc.

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = opt.minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # small fix
        for _ in range(2000):
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x

import time
import numpy as np
import scipy.optimize as opt
import cvxpy as cp

# -------------------------- utilities ---------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale h to satisfy sum(h)=len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    # final clamp
    h[:] = np.clip(h, 0.0, 1.0)
    return h


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


# -------------------------- SLSQP (soft‑max) ---------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """SLSQP on the smooth max for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x


# -------------------------- LP descent -------------------------------

def compute_O_and_grads(h: np.ndarray):
    """Return O_s = total - Q_s and gradients g_s for s=0..n-1."""
    n = h.size
    total = h.sum()
    O = np.empty(n, dtype=float)
    grads = np.empty((n, n), dtype=float)
    # s = 0
    O[0] = total - np.dot(h, h)
    grads[0] = -2.0 * h
    # s > 0
    for s in range(1, n):
        prod = np.dot(h[: n - s], h[s:])
        O[s] = total - prod
        g = np.zeros(n, dtype=float)
        g[: n - s] -= h[s:]
        g[s:] -= h[: n - s]
        grads[s] = g
    return O, grads


def lp_step(h: np.ndarray):
    """Linearized reduction of max O_s via an LP."""
    O, grads = compute_O_and_grads(h)
    n = h.shape[0]
    d = cp.Variable(n)
    gamma = cp.Variable()
    constraints = [d >= -h, d <= 1 - h, cp.sum(d) == 0]
    for s in range(n):
        constraints.append(O[s] + grads[s] @ d <= gamma)
    prob = cp.Problem(cp.Minimize(gamma), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:
        prob.solve(solver=cp.ECOS, warm_start=False)
    if d.value is None:
        return None, None, None
    return np.asarray(d.value), prob.value, O


# -------------------------- main driver ------------------------------
