def compute_c5(h):
    n = h.shape[0]
    L = 2*n
    fft_h = np.fft.rfft(h, n=L)
    fft_one_h = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_one_h), n=L)
    # corr has length L, but linear correlation for full mode length 2n-1; ignore last element (zero).
    # However due to rounding, slight negatives may appear.
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- constraints for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- problem specific helpers ----------
def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """C5 using FFT‑based linear correlation."""
    n = h.shape[0]
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_one = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_one), n=L)
    # linear correlation length = 2n‑1
    return np.max(corr[: 2 * n - 1]) * (2.0 / n)


def worst_shift_and_grad(h: np.ndarray):
    """Shift (k) attaining max overlap and its gradient."""
    n = h.shape[0]
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_one = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_one), n=L)
    idx = np.argmax(corr[: 2 * n - 1])
    shift = idx - (n - 1)
    k = abs(shift)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Transfer mass from positive‑gradient to negative‑gradient entries."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_move(h: np.ndarray, rng: np.random.Generator, eps_scale: float) -> np.ndarray:
    """Transfer a small random amount preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        return h
    eps = min(max_transfer, eps_scale * rng.random())
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 20) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        target_sum = n / 2.0
        h = np.zeros(n)
        idx = rng.choice(n, int(target_sum), replace=False)
        h[idx] = 1.0

    n = h.size
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target_sum, rng)

    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # ----- annealing loop -----
    # temperature schedule (exponential)
    T0 = max(1e-3, cur_c * 0.5)
    T_end = max(1e-8, cur_c * 0.001)
    tau = 1.0 / np.log(T0 / T_end)  # will be multiplied by step count later

    # gradient‑move probability schedule
    p_start, p_end = 0.9, 0.2

    step = 0
    while True:
        if time.time() - start > budget_s * 0.99:
            break

        # dynamic temperature
        T = T0 * np.exp(-tau * step)
        # dynamic probability of gradient move
        p_grad = p_start + (p_end - p_start) * (step / (step + 1.0))

        if rng.random() < p_grad:
            # gradient‑guided mass transfer
            _, grad = worst_shift_and_grad(h)
            alpha = min(0.5, max(1e-4, T / (cur_c + 1e-12)))
            h_new = mass_transfer_move(h, grad, alpha, target_sum, rng)
        else:
            # small random transfer
            eps_scale = min(0.3, T)
            h_new = random_move(h, rng, eps_scale)

        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            h = h_new
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

        step += 1

    # ----- local gradient polish -----
    h = gradient_refine(best_h, target_sum, rng, step=0.02, max_iter=30)
    cur_c = compute_c5(h)
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()

    # ----- final SLSQP polish (time‑aware) -----
    if time.time() - start < budget_s * 0.95:
        for beta in (1000.0, 3000.0, 8000.0):
            if time.time() - start > budget_s * 0.99:
                break
            h_s = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=300)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # ----- final cleanup -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
