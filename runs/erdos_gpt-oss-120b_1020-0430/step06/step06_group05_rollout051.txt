import time
import numpy as np

# ---------- core utilities ----------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of v onto {0 <= x_i <= 1, sum x_i = target}.
    Uses a binary search on the Lagrange multiplier.
    """
    n = v.size
    low = np.min(v) - 1.0          # ensures all entries become 1
    high = np.max(v)                # ensures all entries become 0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def worst_shift_and_grad_corr(h: np.ndarray, corr: np.ndarray):
    """Given correlation array, return worst shift k and its gradient."""
    n = h.size
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def random_initial(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector projected onto the feasible set."""
    v = rng.random(n)
    return project_to_capped_simplex(v, target)


def symmetrize_and_project(h: np.ndarray, target: float) -> np.ndarray:
    """Enforce mirror symmetry and re‑project to the feasible set."""
    h_sym = (h + h[::-1]) * 0.5
    return project_to_capped_simplex(h_sym, target)


# ---------- local optimisation ----------

def pgd(
    h: np.ndarray,
    target: float,
    rng: np.random.Generator,
    max_iter: int,
    step0: float,
) -> np.ndarray:
    """Projected gradient descent on the worst‑shift objective."""
    n = h.size
    scale = 2.0 / n
    step = step0
    best_h = h.copy()
    best_c = compute_c5(h)

    for it in range(max_iter):
        # correlation and current C5
        corr = np.correlate(h, 1.0 - h, mode="full")
        c_cur = corr.max() * scale
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()

        # gradient of the worst shift
        _, grad = worst_shift_and_grad_corr(h, corr)

        # primary move
        h_new = project_to_capped_simplex(h - step * grad, target)
        c_new = compute_c5(h_new)

        improved = False
        if c_new < c_cur - 1e-12:
            h = h_new
            step = min(step * 1.1, step0)
            improved = True
        else:
            # try smaller steps
            for factor in (0.5, 0.25, 0.125):
                h_cand = project_to_capped_simplex(h - step * factor * grad, target)
                c_cand = compute_c5(h_cand)
                if c_cand < c_cur - 1e-12:
                    h = h_cand
                    step = step * factor
                    improved = True
                    break

        if not improved:
            # random noise perturbation
            noise = rng.standard_normal(n) * 0.02
            h_cand = project_to_capped_simplex(h + noise, target)
            c_cand = compute_c5(h_cand)
            if c_cand < c_cur - 1e-12:
                h = h_cand
                improved = True
            else:
                step *= 0.5

        if step < 1e-6:
            break
    return best_h


def gradient_refine(
    h: np.ndarray,
    target: float,
    rng: np.random.Generator,
    step: float = 0.015,
    max_iter: int = 10,
) -> np.ndarray:
    """Fine‑tuned projected gradient steps with a decaying step size."""
    n = h.size
    scale = 2.0 / n
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        _, grad = worst_shift_and_grad_corr(h, corr)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) > 1e-12:
            h = project_to_capped_simplex(h, target)
    return h


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = start + budget_s * 0.96  # reserve a little for final polishing

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_capped_simplex(h0, target)
    else:
        n = 1024
        target = n / 2.0
        h0 = random_initial(n, target, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- stochastic multi‑restart PGD ----
    while time.time() < time_limit:
        # choose a start point
        if rng.random() < 0.6:
            # perturb current best slightly
            h_start = best_h.copy()
            noise = rng.standard_normal(n) * 0.01
            h_start = project_to_capped_simplex(h_start + noise, target)
        else:
            h_start = random_initial(n, target, rng)

        # allocate iterations proportionally to remaining time
        remaining = time_limit - time.time()
        max_it = int(max(30, min(300, remaining * 50)))
        h_opt = pgd(h_start, target, rng, max_it, step0=0.7)
        c_opt = compute_c5(h_opt)

        if c_opt < best_c - 1e-12:
            best_c = c_opt
            best_h = h_opt.copy()
            if best_c < 0.38079:      # early exit if we already beat the target
                break

    # ---- final polishing ----
    best_h = gradient_refine(best_h, target, rng, step=0.02, max_iter=30)
    best_h = symmetrize_and_project(best_h, target)
    best_h = gradient_refine(best_h, target, rng, step=0.01, max_iter=20)

    # enforce constraints one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_capped_simplex(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
