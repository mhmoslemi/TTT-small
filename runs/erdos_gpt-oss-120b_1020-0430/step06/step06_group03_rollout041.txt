def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    N = 256  # number of samples
    # initialize h
    h = np.zeros(N, dtype=np.int8)
    h[rng.choice(N, N // 2, replace=False)] = 1

    def compute_corr(vec):
        return np.correlate(vec.astype(np.int32), (1 - vec).astype(np.int32), mode='full')
    f = compute_corr(h)
    offset = N - 1
    max_f = int(f.max())
    best_max_f = max_f
    best_h = h.copy()
    best_c = best_max_f * 2.0 / N

    # stop early if already good
    if best_c <= 0.3808:
        return best_h.astype(float).tolist(), float(best_c), int(N)

    # parameters
    no_improve_cnt = 0
    # main loop
    while time.time() - start < budget_s * 0.95:
        # find worst shift
        idx_max = int(f.argmax())
        k = idx_max - offset

        # get mismatched indices for this shift
        if k >= 0:
            i_range = np.arange(N - k)
            hi = h[i_range]
            hj = h[i_range + k]
        else:
            k_abs = -k
            i_range = np.arange(k_abs, N)
            hi = h[i_range]
            hj = h[i_range - k_abs]
        A = i_range[(hi == 1) & (hj == 0)]
        B = i_range[(hi == 0) & (hj == 1)]

        if A.size and B.size:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
        else:
            # random opposite bits
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                break
            i = int(rng.choice(zeros))
            j = int(rng.choice(ones))

        # perform swap
        h[i], h[j] = h[j], h[i]
        f_new = compute_corr(h)
        new_max_f = int(f_new.max())

        # SA acceptance
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.95)
        # initial temperature proportional to current max_f
        T0 = max_f * 0.5 if max_f > 0 else 0.1
        T = max(0.001, T0 * (1.0 - frac))
        if new_max_f <= max_f or rng.random() < math.exp(-(new_max_f - max_f) / T):
            # accept
            f = f_new
            if new_max_f < max_f:
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1
            max_f = new_max_f
            # update best if improved
            if new_max_f < best_max_f:
                best_max_f = new_max_f
                best_h = h.copy()
                best_c = best_max_f * 2.0 / N
                if best_c <= 0.3808:
                    # early exit
                    break
        else:
            # reject, revert swap
            h[i], h[j] = h[j], h[i]
            no_improve_cnt += 1

        # occasional random restart if stagnating
        if no_improve_cnt >= 50000:
            # restart
            h = np.zeros(N, dtype=np.int8)
            h[rng.choice(N, N // 2, replace=False)] = 1
            f = compute_corr(h)
            max_f = int(f.max())
            no_improve_cnt = 0

    # final best
    best_h = np.clip(best_h, 0, 1).astype(float)
    best_c = best_max_f * 2.0 / N
    return best_h.tolist(), float(best_c), int(N)

import time, math, numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # target number of points (<=1000)
    N = 256
    # initialize h, possibly using initial_h_values if present
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=np.int8)
        # down‑sample / truncate to length N
        if parent.size >= N:
            h = parent[:N].copy()
        else:
            h = np.zeros(N, dtype=np.int8)
            h[:parent.size] = parent
        # enforce exactly N/2 ones
        target_ones = N // 2
        cur = int(h.sum())
        if cur > target_ones:
            idx = np.where(h == 1)[0]
            rm = rng.choice(idx, cur - target_ones, replace=False)
            h[rm] = 0
        elif cur < target_ones:
            idx = np.where(h == 0)[0]
            add = rng.choice(idx, target_ones - cur, replace=False)
            h[add] = 1
    else:
        h = np.zeros(N, dtype=np.int8)
        h[rng.choice(N, N // 2, replace=False)] = 1

    def compute_corr(v):
        return np.correlate(v.astype(np.int32), (1 - v).astype(np.int32), mode='full')
    f = compute_corr(h)
    offset = N - 1
    max_f = int(f.max())
    best_max_f = max_f
    best_h = h.copy()
    best_c = best_max_f * 2.0 / N

    # early exit if already good enough
    if best_c <= 0.3808:
        return best_h.astype(float).tolist(), float(best_c), int(N)

    no_improve = 0
    # main SA loop
    while time.time() - start < budget_s * 0.95:
        # worst shift
        idx_max = int(f.argmax())
        k = idx_max - offset

        # mismatched index sets for this shift
        if k >= 0:
            i_range = np.arange(N - k)
            hi = h[i_range]
            hj = h[i_range + k]
        else:
            kk = -k
            i_range = np.arange(kk, N)
            hi = h[i_range]
            hj = h[i_range - kk]
        A = i_range[(hi == 1) & (hj == 0)]
        B = i_range[(hi == 0) & (hj == 1)]

        if A.size and B.size:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
        else:
            # fallback random opposite bits
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                break
            i = int(rng.choice(zeros))
            j = int(rng.choice(ones))

        # try swap
        h[i], h[j] = h[j], h[i]
        f_new = compute_corr(h)
        new_max_f = int(f_new.max())

        # temperature schedule (linear decay)
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.95)
        T0 = max_f * 0.5 if max_f > 0 else 0.1
        T = max(0.001, T0 * (1.0 - frac))

        if new_max_f <= max_f or rng.random() < math.exp(-(new_max_f - max_f) / T):
            # accept
            f = f_new
            max_f = new_max_f
            if new_max_f < best_max_f:
                best_max_f = new_max_f
                best_h = h.copy()
                best_c = best_max_f * 2.0 / N
                if best_c <= 0.3808:
                    break
            no_improve = 0 if new_max_f < max_f else no_improve + 1
        else:
            # reject, revert swap
            h[i], h[j] = h[j], h[i]
            no_improve += 1

        # random restart if stuck
        if no_improve >= 50000:
            h = np.zeros(N, dtype=np.int8)
            h[rng.choice(N, N // 2, replace=False)] = 1
            f = compute_corr(h)
            max_f = int(f.max())
            no_improve = 0

    best_h = best_h.astype(float).clip(0.0, 1.0)
    best_c = best_max_f * 2.0 / N
    return best_h.tolist(), float(best_c), int(N)
