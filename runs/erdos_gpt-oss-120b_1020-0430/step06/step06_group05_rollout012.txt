def compute_c5(h):
    n = h.size
    a = h.astype(np.float64)
    b = 1 - a
    # compute convolution of a with reversed b to get correlation
    # correlation(k) = sum_i a[i] * b[i+k] with k shift (positive shift means b index offset)
    # Equivalent to convolution of a with flipped b:
    b_rev = b[::-1]
    # length = 2*n-1
    N = 2*n - 1
    # use next power of 2 for speed
    M = 1 << (N-1).bit_length()
    fa = np.fft.rfft(a, M)
    fb = np.fft.rfft(b_rev, M)
    conv = np.fft.irfft(fa*fb, M)[:N]
    # Now conv[idx] corresponds to sum_i a[i] * b_rev[idx-i] = sum_i a[i] * b[(n-1)-(idx-i)] = sum_i a[i] * b[i + (n-1-idx)].
    # Let k = (n-1) - idx, then conv[idx] = sum_i a[i] * b[i + k] = O_k.
    # So O_k for shift k = (n-1) - idx.
    # Max over k corresponds to max over idx.
    max_corr = conv.max()
    return max_corr * (2.0 / n)

def run(seed, budget_s=1100):

    start = time.time()
    n = 512
    target_sum = n/2
    best_h = None
    best_c = np.inf

    # function to init candidate
    def init_candidate():
        # use random binary
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
        return h

    # if initial_h_values global:

    else:

    # tracking
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c

    stagnant = 0
    iter_cnt = 0
    max_iters = 200000 # maybe large but time will stop earlier

    while True:
        if time.time() - start > budget_s * 0.98:
            break
        if iter_cnt >= max_iters:
            break

        # compute gradient
        _, grad = worst_shift_and_grad(h)

        improved = False
        # iterate over scale alphas, maybe adaptively extend if needed.
        for alpha in alpha_schedule(iter_cnt, max_iters):

import time
import numpy as np

# ----------------- helper functions ----------------- #
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple iterative transfer
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def project_onto_capped_simplex(v: np.ndarray, target: float,
                               rng: np.random.Generator) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum x = target}."""
    lo, hi = -1e5, 1e5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        w = enforce_sum(w, target, rng)
    return w


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def alpha_schedule(it: int, max_it: int):
    """Piecewise decreasing list of α values."""
    frac = it / max_it
    if frac < 0.3:
        return (1.0, 0.5, 0.25, 0.125, 0.0625)
    if frac < 0.6:
        return (0.5, 0.25, 0.125, 0.0625, 0.03125)
    if frac < 0.9:
        return (0.25, 0.125, 0.0625, 0.03125, 0.015625)
    return (0.125, 0.0625, 0.03125, 0.015625, 0.0078125)


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


# variables for SLSQP -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize  # local import

    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 512
    target_sum = n / 2.0

    # ---- initialise h ---- #
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target_sum, rng)
    else:
        # random fractional start via projection
        h = project_onto_capped_simplex(rng.random(n), target_sum, rng)

    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c

    stagnant = 0
    it = 0
    max_it = 500000  # safety cap; real stop by time

    while True:
        # time guard
        if time.time() - start > budget_s * 0.98:
            break
        if it >= max_it:
            break

        # gradient for worst shift
        _, grad = worst_shift_and_grad(h)

        improved = False
        # try decreasing alphas
        for alpha in alpha_schedule(it, max_it):
            h_new = mass_transfer_move(h, grad, alpha, target_sum, rng)
            c_new = compute_c5(h_new)
            if c_new < current_c - 1e-12:
                h = h_new
                current_c = c_new
                improved = True
                stagnant = 0
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break

        if not improved:
            # random perturbation
            h2 = random_perturb(h, rng, scale=0.2)
            c2 = compute_c5(h2)
            if c2 < current_c - 1e-12:
                h = h2
                current_c = c2
                stagnant = 0
                if c2 < best_c:
                    best_c = c2
                    best_h = h.copy()
                improved = True
            else:
                stagnant += 1

        # restart if stuck too long
        if stagnant > 5000:
            # new random fractional candidate
            h = project_onto_capped_simplex(rng.random(n), target_sum, rng)
            current_c = compute_c5(h)
            stagnant = 0

        # early exit if we already beat the target bound
        if best_c < 0.38079:
            break

        it += 1

    # ----- polish the best solution ----- #
    if time.time() - start < budget_s * 0.96:
        best_h = gradient_refine(best_h, target_sum, rng,
                                 step=0.020, max_iter=20)
        best_c = compute_c5(best_h)

    # ----- final SLSQP passes if time permits ----- #
    if time.time() - start < budget_s * 0.94:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.94:
                break
            h_s = slsqp_refine(best_h, target_sum, rng,
                               beta=beta, max_iter=400)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
