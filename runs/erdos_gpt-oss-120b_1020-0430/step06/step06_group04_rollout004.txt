import time
import numpy as np
from scipy.optimize import minimize

# ---------- correlation helpers ----------
def linear_corr(a, b):
    n = a.size
    m = 2 * n
    return np.fft.irfft(np.fft.rfft(a, n=m) * np.fft.rfft(b, n=m), n=m)[: 2 * n - 1]

def compute_c5(h):
    """max overlap integral for discrete h (using FFT)."""
    n = h.size
    return float(np.max(linear_corr(h, 1.0 - h)) * (2.0 / n))

# ---------- projection onto bounded simplex ----------
def proj_bounds_simplex(y, target, eps=1e-12):
    """Projection of y onto {x: sum x = target, 0 <= x <= 1}."""
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    # binary search for lambda
    while hi - lo > eps:
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(y + lam, 0.0, 1.0)

# ---------- smooth‑max gradient ----------
def gradient_smoothmax(h, beta):
    n = h.size
    v = linear_corr(h, 1.0 - h)                 # overlaps for all integer shifts
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:]      -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:]          += wi * one_minus_h[: n - s]
            g[: n - s]     -= wi * h[s:]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    target = h.size / 2.0
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h, best_c = h, compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = h - base_step * mult * g
                cand = proj_bounds_simplex(cand, target)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h

# ---------- worst‑shift refinement ----------
def worst_shift_gradient(h):
    n = h.size
    v = linear_corr(h, 1.0 - h)
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]      -= h[: n - k]
        else:
            kk = -k
            g[kk :]     += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

def worst_shift_refine(h, rng, base_step, iters, deadline):
    target = h.size / 2.0
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h, best_c = h, compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            cand = proj_bounds_simplex(cand, target)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.95
    return h

# ---------- SLSQP on smooth‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = linear_corr(x, 1.0 - x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum fix (projected simplex)
    x = proj_bounds_simplex(x, CURRENT_TARGET)
    return x

# ---------- binary simulated‑annealing ----------
def binary_sa_long(h_bin, rng, deadline):
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = cur_h.size
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    T0 = max(1e-6, cur_c * 0.02)
    T = T0
    decay = 0.99995
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        T *= decay
        if T < 1e-8:
            T = T0
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    # --- Phase 1: continuous optimisation on base size ---
    n_base = 720
    parent = globals().get("initial_h_values", None)
    parent_interp = None
    if parent is not None:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt = np.linspace(0.0, 2.0, n_base, endpoint=False)
        parent_interp = np.interp(xt, xs, parent)

    phase1_dead = start + (deadline - start) * 0.35
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 20000, 50000]

    for restart in range(6):
        if time.time() > phase1_dead:
            break
        if restart == 0 and parent_interp is not None:
            h = parent_interp.copy()
        else:
            h = rng.random(n_base)
            h = proj_bounds_simplex(h, n_base / 2.0)
        # smooth‑max gradient descent
        h = gradient_descent_smoothmax(
            h,
            betas,
            base_step=0.25,
            it_per_beta=30,
            rng=rng,
            deadline=phase1_dead,
        )
        # SLSQP polish with increasing β
        for beta in (2000, 5000, 20000, 50000):
            if time.time() > phase1_dead:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n_base

    # --- Phase 2: up‑sampling & further refinement (if time) ---
    if best_h is not None and time.time() < deadline:
        n_up = best_n * 2
        if n_up <= 1440:
            h_up = np.repeat(best_h, 2)
            h_up = proj_bounds_simplex(h_up, n_up / 2.0)
            h_up = gradient_descent_smoothmax(
                h_up,
                betas[:6],
                base_step=0.20,
                it_per_beta=20,
                rng=rng,
                deadline=deadline,
            )
            h_up = worst_shift_refine(h_up, rng, base_step=0.12, iters=20, deadline=deadline)
            for beta in (2000, 5000, 20000):
                if time.time() > deadline:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # --- Phase 3: binary simulated annealing ---
    if best_h is None:
        # fallback simple half‑ones
        n_fallback = 180
        best_h = np.zeros(n_fallback)
        best_h[: n_fallback // 2] = 1.0
        best_h = proj_bounds_simplex(best_h, n_fallback / 2.0)
        best_c = compute_c5(best_h)
        best_n = n_fallback

    # construct binary start from continuous best_h
    idx_desc = np.argsort(-best_h)
    h_bin = np.zeros(best_n)
    h_bin[idx_desc[: best_n // 2]] = 1.0

    # binary SA for the remaining time
    binary_dead = deadline
    h_bin_opt, c_bin = binary_sa_long(h_bin, rng, binary_dead)
    if c_bin < best_c:
        best_c, best_h, best_n = c_bin, h_bin_opt.copy(), best_n

    # final projection to guarantee feasibility
    best_h = proj_bounds_simplex(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
