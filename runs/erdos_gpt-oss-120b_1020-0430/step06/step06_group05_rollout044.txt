def project_to_sum_target(h, target):
    # ensure within [0,1]
    h = np.clip(h,0,1)
    # iterative water-filling
    while True:
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = free.sum()
        if nfree == 0:
            # cannot adjust further
            break
        delta = diff / nfree
        h[free] += delta
        # clip again
        h = np.clip(h,0,1)
    return h
```

def slsqp_refine(h, target, rng, beta, max_iter=300):
    # using global variables for sum constraint.
    global SL_TARGET, SL_BETA
    SL_TARGET = target
    SL_BETA = beta
    def sum_eq(x):
        return np.sum(x) - SL_TARGET
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = SL_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / SL_BETA
    bounds = [(0.0,1.0)]*len(h)
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds,
                   constraints={'type':'eq','fun':sum_eq},
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0,1)
    # ensure sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = project_to_sum_target(x, target)
    return x
```

import time
import numpy as np

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Discrete C5 for step function on [0,2] using FFT."""
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    rev = (1.0 - h)[::-1]
    corr = np.fft.ifft(np.fft.fft(h, size) * np.fft.fft(rev, size)).real[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def worst_shift_and_grad(h):
    """Return worst shift k and gradient of O_k."""
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    rev = (1.0 - h)[::-1]
    corr = np.fft.ifft(np.fft.fft(h, size) * np.fft.fft(rev, size)).real[: 2 * n - 1]
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def project_to_sum_target(h, target, tol=1e-12, max_iter=200):
    """Project h onto [0,1]^n with sum = target (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h


def pgd_refine(h, target, rng, max_iters=300, step0=0.5):
    """Projected gradient descent using worst‑shift subgradient."""
    h = h.copy()
    for it in range(max_iters):
        _, grad = worst_shift_and_grad(h)
        step = step0 * (0.95 ** it)
        h -= step * grad
        h = project_to_sum_target(h, target)
    return h


def random_perturb(h, rng, scale=0.2):
    """Transfer a small amount of mass between two coordinates."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    rev = (1.0 - x)[::-1]
    corr = np.fft.ifft(np.fft.fft(x, size) * np.fft.fft(rev, size)).real[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h, target, rng, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize  # local import
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_to_sum_target(x, target)
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = budget_s * 0.99

    n = len(initial_h_values) if "initial_h_values" in globals() else 512
    target = n / 2.0

    best_h = None
    best_c = np.inf

    top_k = 20
    top = []

    def consider(h, c):
        nonlocal top
        if len(top) < top_k:
            top.append((c, h.copy()))
            top.sort(key=lambda t: t[0])
        elif c < top[-1][0]:
            top[-1] = (c, h.copy())
            top.sort(key=lambda t: t[0])

    # ---------- Phase 1: random sampling ----------
    max_samples = 3000

        if time.time() - start > budget * 0.45:

        h = rng.random(n)

    for c0, h0 in top:
