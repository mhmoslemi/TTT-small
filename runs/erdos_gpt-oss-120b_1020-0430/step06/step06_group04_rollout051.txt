global CURRENT_TARGET, CURRENT_BETA, CURRENT_L

def softmax_obj(x):
    h = np.repeat(x, CURRENT_L)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def slsqp_optimize(x0, L, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA, CURRENT_L
    n = x0.size
    CURRENT_L = L
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, x0, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    # exact sum adjust
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def block_gradient_descent_smoothmax(v, L, betas, base_step, it_per_beta, rng, deadline):
    n = v.size
    for beta, it in zip(betas, it_per_beta):
        for _ in range(it):
            if time.time() > deadline:
                return v
            g = gradient_v_smoothmax(v, L, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_v = v
            best_c = compute_c5(np.repeat(v, L))
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = v - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(np.repeat(cand, L))
                if c < best_c:
                    best_c = c
                    best_v = cand
            v = best_v
    return v

import time
import numpy as np
from scipy.optimize import minimize

# ------------------ utilities ------------------
def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(x, rng, max_iter=2000):
    """Scale x in-place so that sum(x) == len(x)/2 exactly."""
    n = x.size
    target = n / 2.0
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

# ------------------ smooth‑max gradient ------------------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)·log ∑exp(β·overlap)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    omh = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * omh[shift:]
            g[shift:] -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:] += wi * omh[: n - s]
            g[: n - s] -= wi * h[s:]
    return g

def gradient_v_smoothmax(v, L, beta):
    """Gradient w.r.t. block vector v (h = repeat(v, L))."""
    h = np.repeat(v, L)
    g_h = gradient_smoothmax(h, beta)
    return g_h.reshape(-1, L).sum(axis=1)

# ------------------ block gradient descent ------------------
def block_gradient_descent_smoothmax(v, L, betas, iters, base_step, rng, deadline):
    for beta, it in zip(betas, iters):
        for _ in range(it):
            if time.time() > deadline:
                return v
            g = gradient_v_smoothmax(v, L, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_v = v
            best_c = compute_c5(np.repeat(v, L))
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = v - base_step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(np.repeat(cand, L))
                if c < best_c:
                    best_c, best_v = c, cand
            v = best_v
    return v

    for beta, it in zip(betas, iters):
        for _ in range(it):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):

                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h

# ------------------ worst‑shift refinement ------------------
def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.95
    return h

# ------------------ SLSQP polishing ------------------
CURRENT_TARGET = None
CURRENT_BETA = None
CURRENT_L = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    h = np.repeat(x, CURRENT_L)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(x0, L, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA, CURRENT_L
    n = x0.size
    CURRENT_L = L
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c
