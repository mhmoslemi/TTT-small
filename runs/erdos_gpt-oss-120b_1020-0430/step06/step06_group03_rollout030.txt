def corr_and_grad(h, beta):
    n = h.shape[0]
    # compute correlation via FFT
    N = 2*n
    f_h = np.fft.rfft(h, N)
    f_one_minus_h = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(f_h * np.conj(f_one_minus_h), N)
    corr = conv[:2*n-1]  # length 2n-1, shift indices - (n-1) .. (n-1)
    # compute softmax weights
    maxc = np.max(corr)
    expb = np.exp(beta * (corr - maxc))
    w = expb / expb.sum()
    # compute gradient: 1 - sum_k w_k * (h shifted forward + backward)
    # Let's compute conv_h_w = convolution of h with w (full)
    # similarly conv_h_w_rev = convolution of h with w_rev
    # we need slice n-1:n-1+n
    # For FFT convolution we need length >= n + (2n-1) -1 = 3n-2. Choose M = 1<<ceil_pow2(3*n)
    M = 1
    while M < n + (2*n-1) - 1:
        M <<= 1
    # FFTs
    f_h = np.fft.rfft(h, M)
    f_w = np.fft.rfft(w, M)
    conv_hw = np.fft.irfft(f_h * f_w, M)
    # slice
    start = n-1
    slice_hw = conv_hw[start:start+n]
    # reversed w
    w_rev = w[::-1]
    f_w_rev = np.fft.rfft(w_rev, M)
    conv_hw_rev = np.fft.irfft(f_h * f_w_rev, M)
    slice_hw_rev = conv_hw_rev[start:start+n]
    grad = 1.0 - (slice_hw + slice_hw_rev)
    return corr, grad

def project_onto_box_sum(v, target):
    lo = -1.0   # lower bound for lambda
    hi = 1.0    # upper bound
    # ensure that sum(clipped(v - lo)) >= target and sum(clipped(v - hi)) <= target.
    # we can expand bounds until condition met.
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # small correction for numerical errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff among free variables (0<x<1)
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            # reclip
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback adjust largest element
            i = np.argmin(np.abs(v - (lo+hi)/2.0))
            x[i] += diff
            x = np.clip(x, 0.0, 1.0)

def corr_and_grad(h, beta):

    N = n * 2
    # correlation
    f_h = np.fft.rfft(h, N)
    f_g = np.fft.rfft(1.0 - h, N)

    expb = np.exp(beta * (corr - maxc))

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        cnt = mask.sum()
        if cnt:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.shape[0]
    N = n * 2
    f_h = np.fft.rfft(h, N)
    f_g = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(f_h * np.conj(f_g), N)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def corr_and_grad(h, beta):
    n = h.shape[0]
    # correlation
    N = n * 2
    f_h = np.fft.rfft(h, N)
    f_g = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(f_h * np.conj(f_g), N)[:2 * n - 1]

    m = corr.max()

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad
