def project_capped_simplex(v, S):
    # ensure 0 <= v <= 1 after projection
    # binary search tau in [low, high]
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(50):
        tau = (low + high) / 2.0
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > S:
            low = tau
        else:
            high = tau
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # due to rounding may not exactly equal S, adjust by adding/subtracting small amounts
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        # adjust using indices not at bounds
        # If diff >0, increase some x under 1; else decrease some x >0
        if diff > 0:
            idx = (x < 1 - 1e-12)
            # distribute equally
            inc = diff / np.count_nonzero(idx)
            x[idx] += inc
        else:
            idx = (x > 1e-12)
            dec = -diff / np.count_nonzero(idx)
            x[idx] -= dec
        # clamp again
        x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def _enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that sum(v) == target."""
    diff = target - v.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, v.size)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v

def _compute_c5_full(h):
    """C5 using the original linear correlation definition."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)
