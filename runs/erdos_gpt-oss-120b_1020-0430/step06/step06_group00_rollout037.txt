def project_onto_box_and_simplex(v, target, eps=1e-12):
    # monotonic f(theta) = sum_i clip(v_i + theta, 0, 1)
    low = -np.max(v) - 1.0  # ensures v_i+theta <=0 for all i => f(low)=0
    high = 1.0 - np.min(v) + 1.0  # ensures v_i+theta >=1 => f(high)=n
    for _ in range(50):
        mid = (low + high) * 0.5
        p = np.clip(v + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            low = mid
        else:
            high = mid
    p = np.clip(v + high, 0.0, 1.0)
    return p

def project_onto_box_and_simplex(v, target, eps=1e-12):
    """Project vector v onto the set {x in [0,1]^n, sum x = target}."""
    n = v.shape[0]
    # monotone function f(theta) = sum_i clip(v_i + theta, 0, 1)
    # low ensures all clipped to 0: v_i + low <= 0 for all i.
    low = -np.max(v) - 1.0
    # high ensures all clipped to 1: v_i + high >= 1 for all i.
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        p = np.clip(v + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            low = mid
        else:
            high = mid
    p = np.clip(v + high, 0.0, 1.0)
    # Due to numerical errors, adjust small diff by distributing among non-tight components
    diff = target - p.sum()
    if abs(diff) > eps:
        # distribute diff uniformly over indices where 0<p_i<1
        free = (p > eps) & (p < 1 - eps)
        if free.any():
            p[free] += diff / free.sum()
        else:
            # fallback: adjust first element
            p[0] += diff
        # clip again
        p = np.clip(p, 0.0, 1.0)
    return p

def project_onto_box_and_simplex(v, target, eps=1e-12):
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        p = np.clip(v + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            low = mid
        else:
            high = mid
    p = np.clip(v + high, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > eps:
        free = (p > eps) & (p < 1 - eps)
        if free.any():
            p[free] += diff / free.sum()
        else:
            p[0] += diff
        p = np.clip(p, 0.0, 1.0)
    return p

def random_feasible_vector(rng, n, target):
    v = rng.uniform(0.0, 1.0, size=n)
    return project_onto_box_and_simplex(v, target)

def gradient_descent_one_restart(h, target, max_iters, rng, deadline=None):
    n = h.shape[0]
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(cur_h)
        if k == 0:
            g[:] = 1.0 - 2.0 * cur_h
        else:
            g[: n - k] = 1.0 - cur_h[k:]
            g[k:] -= cur_h[: n - k]
        alpha = 1.0
        improved = False
        for _ in range(10):
            h_new = project_onto_box_and_simplex(cur_h - alpha * g, target)
            c_new = compute_c5(h_new)
            if c_new < cur_c - 1e-12:
                cur_h = h_new
                cur_c = c_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return cur_h, cur_c
