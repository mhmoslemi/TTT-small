function compute_c5_fft(h, L=None):

def compute_c5_fft(h, L=None):
    n = h.shape[0]
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    # linear correlation of length L; we need first 2n-1 values:
    corr = corr[:2*n - 1]
    return corr.max() * (2.0 / n)
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5_fft(h: np.ndarray, L: int = None) -> float:
    n = h.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[: 2 * n - 1]
    return corr.max() * (2.0 / n)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def subgradient_descent(h: np.ndarray, max_iters: int, step0: float,
                       rng: np.random.Generator, L: int) -> tuple:

    cur_h = h.copy()
    cur_c = compute_c5_fft(cur_h, L)
    best_h, best_c = cur_h.copy(), cur_c

        nh = enforce_sum(nh, rng)

        cur_h = nh
        if cur_c < best_c:
            best_c, best_h = cur_c, nh.copy()
    return best_h, best_c

CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    H = np.fft.rfft(x, n=L_global)
    G = np.fft.rfft(1.0 - x, n=L_global)
    corr = np.fft.irfft(H * np.conj(G), n=L_global)[: 2 * x.size - 1]
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA, L_global
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    L_global = 1 << ((2 * n - 1).bit_length())
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h, L)
    return best_h.tolist(), float(final_c), int(best_n)
