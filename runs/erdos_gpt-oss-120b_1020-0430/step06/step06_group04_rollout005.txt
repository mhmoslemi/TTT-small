import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- smooth‑max gradient ----------
def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

# ---------- descent helpers ----------
def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h, best_c = h, compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = h - base_step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h, best_c = h, compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.95
    return h

# ---------- SLSQP smooth‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- binary hill‑climb ----------
def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c

# ---------- main routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    base_n = 720
    # parent handling
    parent_arr = None
    parent_raw = globals().get("initial_h_values", None)
    if parent_raw is not None:
        parent_raw = np.array(parent_raw, dtype=float)
        if parent_raw.size != base_n:
            xs = np.linspace(0.0, 2.0, parent_raw.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            parent_arr = np.interp(xt, xs, parent_raw)
        else:
            parent_arr = parent_raw.copy()

    best_c = np.inf
    best_h = None
    best_n = base_n

    # periods dividing base_n
    periods = [12,15,18,20,24,30,36,40,45,60,72,90,120,144,180,240,360,720]
    periods = [p for p in periods if base_n % p == 0]

    for p in periods:
        if time.time() > deadline:
            break
        m = base_n // p
        if parent_arr is not None:
            q = parent_arr.reshape(m, p).mean(axis=0)
        else:
            q = rng.random(p)
        adjust_sum_exact(q, rng)

        def eval_q(arr):
            return compute_c5(np.tile(arr, m))

        cur_c = eval_q(q)
        betas = [5,20,80,200,500,2000,5000,20000]
        step = 0.25
        for beta in betas:
            for _ in range(5):
                if time.time() > deadline:
                    break
                h = np.tile(q, m)
                g = gradient_smoothmax(h, beta)
                g_q = g.reshape(m, p).sum(axis=0)
                cand = q - step * g_q
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = eval_q(cand)
                if c < cur_c:
                    cur_c, q = c, cand
            step *= 0.9
        h_final = np.tile(q, m)
        c_final = compute_c5(h_final)
        if c_final < best_c:
            best_c, best_h, best_n = c_final, h_final.copy(), base_n

    if best_h is None:
        best_h = np.zeros(base_n)
        best_h[: base_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # ------- further refinement on the best full‑size candidate -------
    if time.time() < deadline:
        betas = [5,20,80,200,500,1000,2000,5000,20000]
        best_h = gradient_descent_smoothmax(best_h, betas,
                                            base_step=0.2, it_per_beta=6,
                                            rng=rng, deadline=deadline)
    if time.time() < deadline:
        best_h = worst_shift_refine(best_h, rng,
                                    base_step=0.12, iters=20,
                                    deadline=deadline)
    if time.time() < deadline:
        for beta in (2000,5000,20000,50000):
            if time.time() > deadline:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=600)

    # binary polishing (only if improves)
    if time.time() < deadline:
        h_bin = np.rint(best_h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
