def worst_shift_and_grad(h):
    n = h.size

    else:
        grad[: n - k] -= h[k:]

def mass_transfer_move(h, grad, alpha, target, rng):
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0 or total_in == 0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    # enforce bounds
    h_new = np.clip(h_new, 0.0, 1.0)
    # correct sum
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new
```

import time
import numpy as np

# ---------- utilities ----------
def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """Correlation based C5 using FFT (fast)."""
    n = h.shape[0]
    m = n * 2
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)


def correlation_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    m = n * 2
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    return corr[: 2 * n - 1]


def worst_shift_and_grad(h: np.ndarray):
    n = h.shape[0]
    corr = correlation_fft(h)
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    h_new[pos] = h[pos] * (1.0 - eps / total_out)
    h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int = 300) -> np.ndarray:
    global _CURRENT_TARGET, _CURRENT_BETA
    _CURRENT_TARGET = target
    _CURRENT_BETA = beta

    def sum_eq(x):
        return np.sum(x) - _CURRENT_TARGET

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = _CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / _CURRENT_BETA

    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0

    # ---- initialise h ----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target_sum, rng)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_c = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c = best_c

    # parameters for simulated annealing
    max_time_main = budget_s * 0.94
    no_improve = 0
    while time.time() - start < max_time_main:
        elapsed = time.time() - start
        t_frac = elapsed / max_time_main
        # temperature decreasing linearly, never below 1e-4
        T = max(1e-4, 0.02 * (1.0 - t_frac))
        # step size for mass transfer, decreasing
        alpha = max(0.01, 0.5 * (1.0 - t_frac))

        r = rng.random()
        if r < 0.55:
            # gradient‑guided mass transfer
            _, grad = worst_shift_and_grad(cur_h)
            new_h = mass_transfer_move(cur_h, grad, alpha, target_sum, rng)
        elif r < 0.85:
            new_h = random_perturb(cur_h, rng, scale=0.25)
        else:
            i, j = rng.integers(0, n, size=2, endpoint=False)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]

        new_c = compute_c5(new_h)

        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
                no_improve = 0
                if best_c < 0.38078:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= 200000:
            # restart from best with a perturbation
            cur_h = best_h.copy()
            cur_h = random_perturb(cur_h, rng, scale=0.5)
            cur_c = compute_c5(cur_h)
            no_improve = 0

    # ----- polish with gradient descent ----- #
    best_h = gradient_refine(best_h, target_sum, rng, step=0.015, max_iter=12)
    best_c = compute_c5(best_h)

    # ----- final SLSQP passes if time permits ----- #
    if time.time() - start < budget_s * 0.98:
        for beta in (2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            h_s = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=250)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
