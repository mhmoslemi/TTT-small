def objective_with_grad(x, beta, target, penalty):
   # compute corr
   corr = np.correlate(x, 1.0 - x, mode='full')
   # compute smooth max
   bc = beta * corr
   max_bc = bc.max()
   sum_exp = np.exp(bc - max_bc).sum()
   f_smooth = (max_bc + np.log(sum_exp)) / beta
   # penalty term
   s = x.sum()
   diff = s - target
   f = f_smooth + penalty * diff * diff
   # gradient of smooth part
   # compute softmax weights
   w = np.exp(bc - max_bc) / sum_exp  # shape len(corr)
   # compute gradient
   n = x.size
   grad = np.zeros_like(x)
   # central term k=0
   wk = w[n-1]
   grad += wk * (1.0 - 2.0 * x)
   # other shifts
   for k in range(1, n):
       wk = w[n-1 + k] + w[n-1 - k]
       if wk != 0.0:
           # subtract contributions
           grad[: n - k] -= wk * x[k:]
           grad[k:] -= wk * x[: n - k]
   # add penalty gradient
   grad += 2.0 * penalty * diff
   return f, grad

import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def objective_and_grad(x, beta, target, penalty):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = beta * corr
    m = bc.max()
    w = np.exp(bc - m)
    sum_exp = w.sum()
    f_smooth = (m + np.log(sum_exp)) / beta
    diff = x.sum() - target
    f = f_smooth + penalty * diff * diff
    w /= sum_exp
    n = x.size
    grad = np.zeros_like(x)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * x)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk != 0.0:
            grad[: n - k] -= wk * x[k:]
            grad[k:]     -= wk * x[: n - k]
    grad += 2.0 * penalty * diff
    return f, grad
