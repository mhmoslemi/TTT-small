import time, numpy as np, math
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    omh = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        sh = idx - (n - 1)
        if sh >= 0:
            g[: n - sh] += wi * omh[sh:]
            g[sh:]    -= wi * h[: n - sh]
        else:
            sh = -sh
            g[sh:]   += wi * omh[: n - sh]
            g[: n - sh] -= wi * h[sh:]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-15:
                continue
            best_h, best_c = h, compute_c5(h)
            for mul in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = h - base_step * mul * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    maxv = v.max()
    idxs = np.where(v == maxv)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:]    -= h[: n - k]
        else:
            k = -k
            g[k:]   += 1.0 - h[: n - k]
            g[: n - k] -= h[k:]
    if len(idxs) > 1:
        g /= len(idxs)
    return g

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-15:
            break
        best_h, best_c = h, compute_c5(h)
        for mul in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mul * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.97
    return h

CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

    best = cur.copy()

        h_up = worst_shift_refine(h_up, rng, base_step=0.1, iters=150,
                                  deadline=deadline)
        for beta in betas_slsqp[:3]:
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        h_up = worst_shift_refine(h_up, rng, base_step=0.08, iters=150,
                                  deadline=deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up
        # binary polish on up‑sampled
        h_bin = np.rint(h_up).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.15)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n_up

    # final safety
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
