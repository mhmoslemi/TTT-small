def project_box_sum(v, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=50):
    # Bisection for lambda.
    # Define function f(lambda) = sum(clip(v_i - lambda, lo, hi)) - target
    # lambda in [min(v- hi), max(v- lo)]
    lo_lambda = np.min(v - hi)
    hi_lambda = np.max(v - lo)
    for _ in range(max_iter):
        lam = (lo_lambda + hi_lambda) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo_lambda = lam
        else:
            hi_lambda = lam
    return np.clip(v - (lo_lambda+hi_lambda)/2.0, lo, hi)

import time
import numpy as np
from scipy.optimize import minimize

# ----- global helpers for soft‑max SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None
CURRENT_N = None


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.size
    m = 2 * n
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    R = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(H * R, n=m)
    return conv[: 2 * n - 1]


def project_box_sum(v: np.ndarray, target: float, lo: float = 0.0, hi: float = 1.0,
                    tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Euclidean projection of v onto { lo≤x≤hi, sum(x)=target }."""
    lo_lam = np.min(v - hi)
    hi_lam = np.max(v - lo)
    for _ in range(max_iter):
        lam = (lo_lam + hi_lam) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo_lam = lam
        else:
            hi_lam = lam
    return np.clip(v - lam, lo, hi)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = compute_corr_fft(x)
    O = corr * (2.0 / CURRENT_N)
    m = np.max(O)
    # stable log‑sum‑exp with current beta
    return (m + np.log(np.sum(np.exp(CURRENT_BETA * (O - m))))) / CURRENT_BETA

    end = start + budget_s

    # ----- initialise h -----

        init = np.array(initial_h_values, dtype=float)
        n0 = init.size

        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, init)
    else:
        n = 1024
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    target = n / 2.0
    h = project_box_sum(h, target)

    # ----- current best -----
    corr = compute_corr_fft(h)
    best_c = np.max(corr) * (2.0 / n)
    best_h = h.copy()

    step = 0.5
    no_improve = 0

    # ----- main subgradient loop -----
    while time.time() < end:
        corr = compute_corr_fft(h)
        c5 = np.max(corr) * (2.0 / n)

        if c5 + 1e-12 < best_c:
            best_c = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # restart from best with a small perturbation if stuck
        if no_improve > 20000:
            h = best_h.copy()
            noise = rng.normal(scale=0.02, size=n)

            h = project_box_sum(h, target)
            step = 0.5
            no_improve = 0

        k = worst_idx - (n - 1)
