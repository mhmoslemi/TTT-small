But for i where i+k doesn't exist (i > n-1-k), the plus term is 0; for i < k, minus term is 0; correct.

However for k > n/2, there will be overlapping region but shift will be zero for some indices, but still correct. The product sum_i h_i h_{i+k} only sums where both indices within range, i.e., i up to n-1-k. The gradient for i > n-1-k includes -h_{i-k} (minus term), which is in shift_minus term. So the formula works: gradient_i = - (h_{i+k} if i+k < n else 0) - (h_{i - k} if i - k >= 0 else 0). That's consistent.

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # water‑filling style adjustments
    for _ in range(30):
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            inc = np.minimum(add, 1.0 - h[idx])
            h[idx] += inc
            diff = target - h.sum()
            if diff <= 1e-12:
                break
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            sub = (-diff) / idx.size
            dec = np.minimum(sub, h[idx])

            diff = target - h.sum()

                    diff = target - h.sum()

                break
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the shift achieving the maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = abs(idx - (n - 1))
    if k == 0:

        return 1.0 - 2.0 * h
    # shift plus (i+k) padded with zeros

    # shift minus (i-k) padded with zeros
    minus = np.empty(n, dtype=h.dtype)
    minus[k:] = h[: n - k]
    minus[:k] = 0.0
    return -(plus + minus)
