def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase sum
        cap = 1.0 - h
        idx = np.where(cap > 1e-12)[0]
        # order random to avoid bias
        rng.shuffle(idx)
        for i in idx:
            inc = min(cap[i], diff)
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to decrease sum
        cap = h
        idx = np.where(cap > 1e-12)[0]
        rng.shuffle(idx)
        diff = -diff
        for i in idx:
            dec = min(cap[i], diff)
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return h

def binary_sa(n, rng, time_budget):
    start = time.time()
    half = n // 2
    # initial binary vector
    ones_idx = rng.choice(n, half, replace=False)
    ones_idx.sort()
    zeros_idx = np.setdiff1d(np.arange(n), ones_idx, assume_unique=True)
    # position maps
    pos_one = -np.ones(n, dtype=int)
    pos_zero = -np.ones(n, dtype=int)
    pos_one[ones_idx] = np.arange(half)
    pos_zero[zeros_idx] = np.arange(half)
    h = np.zeros(n, dtype=np.float64)
    h[ones_idx] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    # temperature schedule
    T0 = 0.1
    T_end = 1e-5
    # approximate max iterations based on time budget
    # We'll loop until time out
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        # linear or exponential decay of temperature
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick random ones and zeros
        i_pos = rng.integers(half)
        j_pos = rng.integers(half)
        i = ones_idx[i_pos]
        j = zeros_idx[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # update index arrays and maps
        ones_idx[i_pos] = j
        zeros_idx[j_pos] = i
        pos_one[j] = i_pos
        pos_one[i] = -1
        pos_zero[i] = j_pos
        pos_zero[j] = -1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones_idx[i_pos] = i
            zeros_idx[j_pos] = j
            pos_one[i] = i_pos
            pos_one[j] = -1
            pos_zero[j] = j_pos
            pos_zero[i] = -1
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- core utilities --------------------

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase sum
        caps = 1.0 - h
        idx = np.where(caps > 1e-12)[0]
        rng.shuffle(idx)
        for i in idx:
            inc = min(caps[i], diff)
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to decrease sum
        caps = h
        idx = np.where(caps > 1e-12)[0]
        rng.shuffle(idx)
        diff = -diff
        for i in idx:
            dec = min(caps[i], diff)
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return h


def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function using FFT (fast)."""
    n = h.size
    n2 = 2 * n
    H = np.fft.rfft(h, n=n2)
    G = np.fft.rfft(1.0 - h, n=n2)
    corr = np.fft.irfft(H * np.conj(G), n=n2)
    max_val = np.max(corr[: 2 * n - 1])
    return max_val * (2.0 / n)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 20000,
    step_scale: float = 0.7,
):
    """Gradient‑directed hill climbing."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                cur_h[i] -= eps
                cur_h[j] += eps
        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_c, best_h, cur_h = cur_c, cur_h.copy(), cur_h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.015,
    max_iter: int = 10,
) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# -------------------- SLSQP with soft‑max --------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(
    h0: np.ndarray,
    beta: float,
    rng: np.random.Generator,
    max_iter: int = 800,
) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # quick random redistribution
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# -------------------- binary simulated annealing --------------------

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Binary SA targeting max overlap (C5)."""
    start = time.time()
    half = n // 2
    # initial random binary vector
    ones_idx = rng.choice(n, half, replace=False)
    zeros_idx = np.setdiff1d(np.arange(n, dtype=int), ones_idx, assume_unique=True)
    h = np.zeros(n, dtype=np.float64)
    h[ones_idx] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

    T0, T_end = 0.1, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_pos = rng.integers(half)
        j_pos = rng.integers(half)
        i = ones_idx[i_pos]
        j = zeros_idx[j_pos]
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        ones_idx[i_pos] = j
        zeros_idx[j_pos] = i
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones_idx[i_pos] = i
            zeros_idx[j_pos] = j
    return best_h, best_c


# -------------------- main entry point --------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s
    n = 512

    best_h = None
    best_c = np.inf

    # ------------------------------------------------------------------
    # 1) start from provided construction (if any)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c = h0, c0

    # ------------------------------------------------------------------
    # 2) binary simulated annealing (multiple short runs)
    binary_budget = total_budget * 0.40
    bin_start = time.time()
    best_bin_h, best_bin_c = None, np.inf
    while time.time() - bin_start < binary_budget:
        run_time = min(30.0, binary_budget - (time.time() - bin_start))
        h_bin, c_bin = binary_sa(n, rng, run_time)
        if c_bin < best_bin_c:
            best_bin_h, best_bin_c = h_bin, c_bin
            if best_bin_c <= 0.3808:
                break
    if best_bin_h is not None and best_bin_c < best_c:
        best_h, best_c = best_bin_h.astype(np.float64), best_bin_c

    # ------------------------------------------------------------------
    # 3) continuous refinement from the best binary/continuous candidate
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        # small random perturbation to break strict binary structure
        h_start = best_h + (rng.random(n) - 0.5) * 0.01
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = enforce_sum(h_start, rng)

        # guided hill‑climb
        max_iters = int(min(30000, remaining * 500))
        h_hc, c_hc = guided_hill_climb(h_start, rng, max_iters=max_iters, step_scale=0.6)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

        # soft‑max SLSQP passes (high β)
        time_for_slsqp = total_budget - (time.time() - start) - 0.1 * total_budget
        if time_for_slsqp > 5.0:
            beta_seq = (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0, 50000.0)
            max_iter_per = max(200, int(time_for_slsqp * 0.5 / len(beta_seq)))
            h_slsqp = best_h.copy()
            for beta in beta_seq:
                if time.time() - start > total_budget * 0.95:
                    break
                h_slsqp = slsqp_optimize(h_slsqp, beta, rng, max_iter=max_iter_per)
                c_s = compute_c5(h_slsqp)
                if c_s < best_c:
                    best_h, best_c = h_slsqp.copy(), c_s

        # final gradient tidy‑up
        if time.time() - start < total_budget * 0.99:
            h_ref = gradient_refine(best_h, rng, step=0.01, max_iter=10)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

        # occasional tiny random tweaks
        while time.time() - start < total_budget * 0.999:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            eps = rng.random() * 0.02
            hi, hj = best_h[i], best_h[j]
            delta = min(eps, hi, 1.0 - hj)
            if delta <= 0.0:
                continue
            nh = best_h.copy()
            nh[i] -= delta
            nh[j] += delta
            nh = enforce_sum(nh, rng)
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # ------------------------------------------------------------------
    # 4) final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
