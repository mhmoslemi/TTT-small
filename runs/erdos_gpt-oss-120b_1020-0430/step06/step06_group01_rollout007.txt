def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # while diff not near zero
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h
```

import time
import numpy as np

# ----- helpers ---------------------------------------------------------

def enforce_sum(h: np.ndarray, rng: np.random.Generator, max_iter: int = 10000) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:  # last resort
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def compute_c5(h: np.ndarray) -> float:
    """Fast C5 using FFT (linear convolution)."""
    n = h.size
    dx = 2.0 / n
    b_rev = (1.0 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, size)
    fb = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(fa * fb, size)[: 2 * n - 1]
    return np.max(conv) * dx


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.01, max_iter: int = 10) -> np.ndarray:
    """Small gradient descent on the worst shift."""

        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h
