def softmax_gradient(h, beta):
    n = h.shape[0]
    corr = full_corr(h, 1.0 - h)
    Omax = corr.max()
    exp_term = np.exp(beta * (corr - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk_center = w[n-1]
    grad += wk_center * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n-1 + k] + w[n-1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(40):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # adjust small residual
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - (lo+hi)*0.5))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

import time
import numpy as np

# ---------- core utilities ----------
def full_corr(h, g):
    """Linear correlation of real vectors using FFT (full length)."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(g, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    return corr[: 2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    corr = full_corr(h, 1.0 - h)
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    """Projection of v onto [0,1]^n with sum equal to target."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(40):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def softmax_gradient(h, beta):
    n = h.shape[0]
    corr = full_corr(h, 1.0 - h)
    Omax = corr.max()
    exp_term = np.exp(beta * (corr - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

    idx = np.argmax(corr) - (n - 1)
