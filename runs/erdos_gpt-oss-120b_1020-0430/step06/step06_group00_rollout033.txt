def projected_subgradient_descent(h0, rng, time_budget):

    start = time.time()
    h = h0.copy()
    target_sum = h.size / 2.0
    iter = 0
    while time.time() - start < time_budget:
        grad = worst_gradient(h)  # defined earlier using full correlation (but use FFT for identifying worst shift)
        # step size: alpha = 0.01 / sqrt(iter+1) maybe.
        alpha = 0.01 / np.sqrt(iter+1)
        h_new = h - alpha * grad
        # project: clip to 0-1
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce sum:
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            # adjust by adding diff to random positions respecting bounds
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                idx = rng.integers(h_new.size)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[idx])
                else:
                    add = max(diff, -h_new[idx])
                h_new[idx] += add
                diff = target_sum - h_new.sum()
        h = h_new
        iter += 1

    return h

    ones_arr = ones.copy()

    best_h = h.copy()

    # main loop
    while time.time() - start < time_budget:

        frac = elapsed / time_budget
        T = T0 * ((T_end / T0) ** frac)
        # random swap
        i_pos = rng.integers(len(ones_arr))

def worst_gradient_and_c5(h):
    N = h.size
    n_fft = 1 << ((2*N-1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    rev = np.fft.rfft(np.flip(1.0 - h), n=n_fft)
    conv = np.fft.irfft(H * rev, n=n_fft)[:2*N-1]
    idx = np.argmax(conv) - (N-1)
    k = abs(idx)
    # gradient
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] = 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    c5 = conv.max() * (2.0 / N)
    return grad, c5

import time
import numpy as np

def compute_c5_fft(h: np.ndarray) -> float:
    N = h.size
    n_fft = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    rev = np.fft.rfft(np.flip(1.0 - h), n=n_fft)
    conv = np.fft.irfft(H * rev, n=n_fft)[: 2 * N - 1]
    return conv.max() * (2.0 / N)

def binary_sa_fft(N: int, rng: np.random.Generator, time_budget: float,
                  init_h: np.ndarray = None):
    start = time.time()
    if init_h is not None:
        h = init_h.copy()
        ones_arr = np.where(h > 0.5)[0]
        zeros_arr = np.where(h <= 0.5)[0]
        # ensure exactly N/2 ones
        if ones_arr.size > N // 2:
            # turn off excess ones
            excess = ones_arr.size - N // 2
            drop = rng.choice(ones_arr, excess, replace=False)
            h[drop] = 0.0
            ones_arr = np.where(h > 0.5)[0]
        elif ones_arr.size < N // 2:
            missing = N // 2 - ones_arr.size
            add = rng.choice(zeros_arr, missing, replace=False)
            h[add] = 1.0
            ones_arr = np.where(h > 0.5)[0]
        zeros_arr = np.where(h <= 0.5)[0]
    else:
        h = np.zeros(N, dtype=float)
        ones_arr = rng.choice(N, N // 2, replace=False)
        h[ones_arr] = 1.0
        zeros_arr = np.setdiff1d(np.arange(N), ones_arr)
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    T0 = 0.05
    T_end = 1e-6
    iter_since_best = 0
    max_no_improve = 5000
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * ((T_end / T0) ** frac)
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        ones_arr[i_pos] = j
        zeros_arr[j_pos] = i
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                iter_since_best = 0
                if best_c <= 0.3808:
                    return best_h, best_c
            else:
                iter_since_best += 1
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            ones_arr[i_pos] = i
            zeros_arr[j_pos] = j
            iter_since_best += 1
        if iter_since_best >= max_no_improve:
            # restart
            h = np.zeros(N, dtype=float)
            ones_arr = rng.choice(N, N // 2, replace=False)
            h[ones_arr] = 1.0
            zeros_arr = np.setdiff1d(np.arange(N), ones_arr)
            c_curr = compute_c5_fft(h)
            iter_since_best = 0
    return best_h, best_c

def worst_gradient_and_c5(h: np.ndarray):
    N = h.size
    n_fft = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    rev = np.fft.rfft(np.flip(1.0 - h), n=n_fft)
    conv = np.fft.irfft(H * rev, n=n_fft)[: 2 * N - 1]
    idx = np.argmax(conv) - (N - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] = 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    c5 = conv.max() * (2.0 / N)
    return grad, c5

def projected_subgradient_descent(h0: np.ndarray, rng: np.random.Generator,
                                 time_budget: float):
    start = time.time()
    h = h0.copy()
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    _, best_c = worst_gradient_and_c5(h)
    it = 0
    while time.time() - start < time_budget:
        grad, c5 = worst_gradient_and_c5(h)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        alpha = 0.02 / np.sqrt(it + 1)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                idx = rng.integers(N)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[idx])
                else:
                    add = max(diff, -h_new[idx])
                h_new[idx] += add
                diff = target - h_new.sum()
        h = h_new
        it += 1
    # final evaluation
    _, final_c = worst_gradient_and_c5(best_h)
    return best_h, final_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s
    N = 1024
    best_h = None
    best_c = np.inf

    # initial from parent if present
    init_bin = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # upsample to N
        x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h_up = np.interp(x_new, x_old, h0)
        # threshold to get exactly N/2 ones
        thresh = np.partition(h_up, -N // 2)[-N // 2]
        bin_start = (h_up > thresh).astype(float)
        # adjust if count not exact
        while bin_start.sum() > N // 2:
            excess = int(bin_start.sum() - N // 2)
            idxs = np.where(bin_start == 1)[0]
            drop = rng.choice(idxs, excess, replace=False)
            bin_start[drop] = 0.0
        while bin_start.sum() < N // 2:
            deficit = int(N // 2 - bin_start.sum())
            idxs = np.where(bin_start == 0)[0]
            add = rng.choice(idxs, deficit, replace=False)
            bin_start[add] = 1.0
        init_bin = bin_start

    # Phase 0: binary SA
    remaining = total - (time.time() - start)
    if remaining > 10.0:
        t_bin = remaining * 0.80
        best_h_bin, c_bin = binary_sa_fft(N, rng, t_bin, init_h=init_bin)
        best_h = best_h_bin.copy()
        best_c = c_bin

    # Phase 1: projected subgradient descent
    remaining = total - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h0 = best_h.astype(float)
        h_ref, c_ref = projected_subgradient_descent(h0, rng, remaining)
        if c_ref < best_c:
            best_h = h_ref.copy()
            best_c = c_ref

    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    target = N / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        while abs(diff) > 1e-12:
            idx = rng.integers(N)
            if diff > 0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
            else:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
            diff = target - best_h.sum()
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(N)

import time
import numpy as np

def compute_c5_fft(h: np.ndarray) -> float:
    N = h.size
    n_fft = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    rev = np.fft.rfft(np.flip(1.0 - h), n=n_fft)
    conv = np.fft.irfft(H * rev, n=n_fft)[: 2 * N - 1]
    return conv.max() * (2.0 / N)


def binary_sa_fft(N: int, rng: np.random.Generator, time_budget: float,
                  init_h: np.ndarray = None):
    start = time.time()
    if init_h is not None:
        h = init_h.copy()
        # force exact half‑ones
        ones_arr = np.where(h > 0.5)[0]
        if ones_arr.size > N // 2:
            excess = ones_arr.size - N // 2
            drop = rng.choice(ones_arr, excess, replace=False)
            h[drop] = 0.0
        elif ones_arr.size < N // 2:
            deficit = N // 2 - ones_arr.size
            zeros = np.where(h <= 0.5)[0]
            add = rng.choice(zeros, deficit, replace=False)
            h[add] = 1.0
        ones_arr = np.where(h > 0.5)[0]
        zeros_arr = np.where(h <= 0.5)[0]
    else:
        h = np.zeros(N, dtype=float)
        ones_arr = rng.choice(N, N // 2, replace=False)
        h[ones_arr] = 1.0
        zeros_arr = np.setdiff1d(np.arange(N), ones_arr)

    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    T0, T_end = 0.05, 1e-6
    iter_since_best = 0
    max_no_improve = 5000

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * ((T_end / T0) ** frac)

        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        ones_arr[i_pos] = j
        zeros_arr[j_pos] = i

        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                iter_since_best = 0
                if best_c <= 0.3808:
                    return best_h, best_c
            else:
                iter_since_best += 1
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones_arr[i_pos] = i
            zeros_arr[j_pos] = j
            iter_since_best += 1

        if iter_since_best >= max_no_improve:
            # restart from fresh random binary vector
            h = np.zeros(N, dtype=float)
            ones_arr = rng.choice(N, N // 2, replace=False)
            h[ones_arr] = 1.0
            zeros_arr = np.setdiff1d(np.arange(N), ones_arr)
            c_curr = compute_c5_fft(h)
            iter_since_best = 0

    return best_h, best_c


def worst_gradient_and_c5(h: np.ndarray):
    N = h.size
    n_fft = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    rev = np.fft.rfft(np.flip(1.0 - h), n=n_fft)
    conv = np.fft.irfft(H * rev, n=n_fft)[: 2 * N - 1]
    idx = np.argmax(conv) - (N - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] = 1.0 - h[k:]
        grad[k:] -= h[: N - k]
    c5 = conv.max() * (2.0 / N)
    return grad, c5


def projected_subgradient_descent(h0: np.ndarray, rng: np.random.Generator,
                                 time_budget: float):
    start = time.time()
    h = h0.copy()
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    _, best_c = worst_gradient_and_c5(h)
    it = 0
    while time.time() - start < time_budget:
        grad, c5 = worst_gradient_and_c5(h)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        alpha = 0.05 / np.sqrt(it + 1)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                idx = rng.integers(N)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[idx])
                else:
                    add = max(diff, -h_new[idx])
                h_new[idx] += add
                diff = target - h_new.sum()
        h = h_new
        it += 1
    # final check
    _, final_c = worst_gradient_and_c5(best_h)
    return best_h, final_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s
    N = 1024
    best_h = None
    best_c = np.inf

    # start from parent if available
    init_bin = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)

        h_up = np.interp(x_new, x_old, h0)
        thresh = np.partition(h_up, -N // 2)[-N // 2]
        bin_start = (h_up > thresh).astype(float)
        while bin_start.sum() > N // 2:
            excess = int(bin_start.sum() - N // 2)
            idxs = np.where(bin_start == 1)[0]
            drop = rng.choice(idxs, excess, replace=False)
            bin_start[drop] = 0.0
        while bin_start.sum() < N // 2:
            deficit = int(N // 2 - bin_start.sum())
            idxs = np.where(bin_start == 0)[0]
            add = rng.choice(idxs, deficit, replace=False)
            bin_start[add] = 1.0
        init_bin = bin_start

    # Phase 0: binary simulated annealing
    remaining = total - (time.time() - start)
    if remaining > 10.0:
        t_bin = remaining * 0.80
        best_h_bin, c_bin = binary_sa_fft(N, rng, t_bin, init_h=init_bin)
        best_h = best_h_bin.copy()
        best_c = c_bin

    # Phase 1: subgradient refinement
    remaining = total - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h0 = best_h.astype(float)
        h_ref, c_ref = projected_subgradient_descent(h0, rng, remaining)
        if c_ref < best_c:
            best_h = h_ref.copy()
            best_c = c_ref

    # final feasibility fix
    best_h = np.clip(best_h, 0.0, 1.0)
    target = N / 2.0
    diff = target - best_h.sum()

                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
            else:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
            diff = target - best_h.sum()

    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(N)
