import time
import numpy as np

# --------------------------------------------------------------
# Utility projections
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

# --------------------------------------------------------------
# Shift helper (zero‑padding)
def shift(h: np.ndarray, offset: int) -> np.ndarray:
    n = h.size
    out = np.zeros_like(h)
    if offset > 0:
        out[: n - offset] = h[offset:]
    elif offset < 0:
        out[-offset :] = h[: n + offset]
    else:
        out[:] = h
    return out

# --------------------------------------------------------------
# Correlation / C5 computation via FFT (linear convolution)
def compute_O_array(h: np.ndarray) -> np.ndarray:
    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * G, L)
    dx = 2.0 / n
    return corr[: 2 * n - 1] * dx

def compute_c5(h: np.ndarray) -> float:
    return float(np.max(compute_O_array(h)))

# --------------------------------------------------------------
# Subgradient descent (projected) with occasional random jitter
def subgradient_descent(h0: np.ndarray, target: float,
                        time_budget: float, rng: np.random.Generator):
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        O = compute_O_array(h)
        c = float(O.max())
        idx = int(O.argmax())
        k = idx - (n - 1)
        grad = 1.0 - (shift(h, k) + shift(h, -k))
        t += 1
        alpha = 0.5 / np.sqrt(t + 10.0)
        h = project_box_sum(h - alpha * grad, target)
        if t % 250 == 0:
            h = project_box_sum(h + (rng.random(n) - 0.5) * 0.02, target)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3805:        # early stop if very good
                break
    return best_h, best_c

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.6):

    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        O = compute_O_array(h)
        idx = int(O.argmax())
        k = idx - (n - 1)
        grad = 1.0 - (shift(h, k) + shift(h, -k))
        i = int(np.argmax(grad))

        if i != j:

            if max_eps > 0.0:
