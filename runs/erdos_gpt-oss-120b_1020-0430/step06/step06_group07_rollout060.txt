from scipy.optimize import minimize, LinearConstraint, Bounds

def smooth_max(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def optimize_smooth(h0, beta, max_iter):
    n = h0.size
    sum_target = n/2.0
    bounds = Bounds(0.0, 1.0)
    linear_constraint = LinearConstraint(np.ones((1, n)), sum_target, sum_target)
    res = minimize(smooth_max, h0, args=(beta,), method='trust-constr',
                   jac=None, hess=None,
                   constraints=[linear_constraint], bounds=bounds,
                   options={'maxiter': max_iter, 'gtol':1e-12, 'verbose':0})
    return res.x

def subgradient_optimize_time(h0, target, time_budget, rng):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fast(h)

        corr_full = np.correlate(h, h, mode='full')
        cross = corr_full[n-1:]

        # gradient for O_k
        if k == 0:

        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]

        # step size schedule

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
def compute_c5_fast(h: np.ndarray) -> float:
    """Discrete C5 using correlation of h with itself."""
    n = h.size
    corr_full = np.correlate(h, h, mode="full")
    min_corr = np.min(corr_full[n - 1 :])
    return 1.0 - min_corr * (2.0 / n)


# alias used by helpers that expect a function named compute_c5
compute_c5 = compute_c5_fast


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via binary search on Lagrange multiplier."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum=n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample_to_n2(h: np.ndarray, n2: int) -> np.ndarray:
    """Linear up‑sample h to n2 points and enforce the sum constraint."""
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def worst_shift_and_gradient(h: np.ndarray):
    """Return worst shift index k (max O_k) and gradient of O_k w.r.t. h."""
    n = h.size
    corr_full = np.correlate(h, h, mode="full")
    cross = corr_full[n - 1 :]  # length n, cross[k] = Σ h_i h_{i+k}
    k = int(np.argmin(cross))  # minimal correlation → maximal O_k
    if k == 0:
        grad = 1.0 - 2 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

def worst_gradient(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):

        if (it + 1) % decay == 0:

    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )

    # exact sum correction (quick redistribution)

    if abs(diff) > 1e-12:

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
