def proj_to_feasible(v, target):
    # binary search for lambda
    lo = np.min(v - 1)  # ensures v_i - lo >= 1 for all i
    hi = np.max(v)      # ensures v_i - hi <= 0 for all i
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # due to rounding, adjust sum difference
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust a single element within bounds
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size == 0:
            # all at bounds, adjust one with capacity
            if diff > 0:
                idx = np.where(x < 1)[0]
            else:
                idx = np.where(x > 0)[0]
        # Clip diff to capacity
        idx = idx[0]
        cap = (1.0 - x[idx]) if diff > 0 else -x[idx]
        delta = np.clip(diff, -cap, cap)
        x[idx] += delta
    return x

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024
    target = n / 2.0

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Ensure feasible
        # Clip then project
        h = np.clip(h, 0.0, 1.0)
        # if sum not equal target, project
        if abs(h.sum() - target) > 1e-12:
            # project onto feasible set
            h = proj_to_feasible(h, target)
    else:
        # random uniform
        h = rng.random(n)
        h = proj_to_feasible(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Parameters
    step0 = 0.8  # initial step size
    no_improve = 0
    max_no_improve = 5000  # after which we apply random perturb

    iter = 0
    while time.time() - start < budget_s * 0.97:
        # Compute correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute c5
        c5 = corr.max() * (2.0 / n)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
            no_improve = 0
            # early exit if very good
            if best_c < 0.38079:
                break
        else:
            no_improve += 1

        # Compute gradient for worst shift
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # compute gradient
        grad = np.ones_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        # step size
        step = step0 / np.sqrt(iter + 1.0)

        # gradient step
        y = h - step * grad
        # project
        h = proj_to_feasible(y, target)

        # Occasionally random perturb if stuck
        if no_improve >= max_no_improve:
            # random swap of two positions (or random perturb of many)
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            # swap values
            h[i], h[j] = h[j], h[i]
            no_improve = 0

        iter += 1

    # Final projection to ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target) > 1e-12:
        best_h = proj_to_feasible(best_h, target)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np

# --------------------------------------------------------------
# Helper: compute the discrete C5 value for a step function h.
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Helper: projection onto the capped simplex {x|0≤x≤1, sum(x)=target}.
# --------------------------------------------------------------
def proj_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(x)=target. Uses a simple bisection on the Lagrange
    multiplier λ solving sum_i clip(v_i-λ,0,1)=target.
    """
    lo = np.min(v) - 1.0          # ensures v_i-lo ≥ 1 ⇒ clip = 1
    hi = np.max(v)                # ensures v_i-hi ≤ 0 ⇒ clip = 0
    # 30 iterations give ~1e-9 accuracy, enough for our integer target
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)

    # tiny correction to hit target exactly (within 1e-12)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # try to adjust a non‑boundary coordinate; if none, adjust any feasible one
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size == 0:
            idx = np.where(x < 1.0)[0] if diff > 0 else np.where(x > 0.0)[0]
        i = idx[0]
        cap = (1.0 - x[i]) if diff > 0 else -x[i]
        delta = np.clip(diff, -cap, cap)
        x[i] += delta
    return x


# --------------------------------------------------------------
# Helper: gradient refinement using the worst‑shift subgradient.
# --------------------------------------------------------------
def gradient_refine(h: np.ndarray, target: float, step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune h by repeated subgradient steps with shrinking step size."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.ones_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum
        if abs(h.sum() - target) > 1e-12:
            h = proj_to_feasible(h, target)
    return h


# --------------------------------------------------------------
# Main entry point.
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024                      # resolution (larger → finer bound)
    target = n / 2.0

    # ------------------------------------------------------------------
    # Global best solution (updated throughout the search)
    # ------------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    init_used = False

    # ------------------------------------------------------------------
    # Parameters for the stochastic sub‑runs
    # ------------------------------------------------------------------
    max_time_factor = 0.96        # reserve a margin for final checks
    sub_max_iters = 50000        # iterations per sub‑run
    stagnation_limit = 5000      # when to apply a random swap
    step0 = 0.9                   # initial step size for sub‑run

    while time.time() - start < budget_s * max_time_factor:
        # ---- initialise a feasible h ----------------------------------
        if not init_used and "initial_h_values" in globals():
            h = np.array(initial_h_values, dtype=float)
            h = np.clip(h, 0.0, 1.0)
            if abs(h.sum() - target) > 1e-12:
                h = proj_to_feasible(h, target)
            init_used = True
        else:
            h = rng.random(n)
            h = proj_to_feasible(h, target)

        # ---- run a projected subgradient descent --------------------
        no_improve = 0
        for it in range(sub_max_iters):
            # time guard
            if time.time() - start > budget_s * max_time_factor:
                break

            # correlation and current C5
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = corr.max() * (2.0 / n)

            # update global best
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
                no_improve = 0
                # early exit if we beat the target bound
                if best_c < 0.38079:
                    break
            else:
                no_improve += 1

            # worst shift determines subgradient
            idx = np.argmax(corr) - (n - 1)
            k = abs(idx)
            grad = np.ones_like(h)

            else:

    if abs(best_h.sum() - target) > 1e-12:
        best_h = proj_to_feasible(best_h, target)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
