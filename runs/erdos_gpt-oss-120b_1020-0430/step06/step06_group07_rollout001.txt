def simulated_annealing(h0, target, time_budget, rng):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

    # Determine total steps roughly based on time budget: e.g., steps = int( (time_budget / (time_per_iter approx)) )
    # We'll just run while time left > 0.05 seconds

    # Set temperature schedule parameters:
    # Use T0 = 0.005 (maybe) ; T_end = 1e-5.
    # We'll compute fraction = (time.time() - start) / time_budget; T = T0 * (1 - fraction) + T_end * fraction

    while time.time() - start < time_budget * 0.95:
        # compute gradient of worst shift
        grad, k = worst_gradient(h)
        # choose i, j indices
        i_candidates = np.where(grad > 1e-12)[0]
        j_candidates = np.where(grad < -1e-12)[0]
        if i_candidates.size == 0 or j_candidates.size == 0:
            # random pair
            i = rng.integers(0, h.size)
            j = rng.integers(0, h.size)
        else:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)

        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            continue
        # epsilon scaling
        # Determine epsilon as random fraction * max_eps
        eps = rng.random() * max_eps * 0.5  # scale 0.5

        # Apply move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # Ensure sum remains same (should)
        # Compute cost
        new_c = compute_c5(h_new)

        # Acceptance
        # compute temperature based on elapsed fraction
        fraction = (time.time() - start) / time_budget
        T = 0.01 * (1 - fraction) + 0.000001 * fraction
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            h = h_new
            cur_c = new_c
            # update best
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        # else reject
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global values used by the SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via a binary search on the Lagrange multiplier."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def downsample_h(h: np.ndarray, n_target: int) -> np.ndarray:
    """Interpolate h to n_target points and project onto the feasible set."""
    x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_sum(h_new, n_target / 2.0)

def upsample_h(h: np.ndarray, n_target: int) -> np.ndarray:
    """Interpolate h to a finer grid and project onto the feasible set."""
    return downsample_h(h, n_target)

def worst_gradient(h: np.ndarray):
    """Gradient of the overlap for the worst shift (the shift giving max C5)."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))  # smallest auto‑correlation → largest overlap
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 400) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    # soft‑max of the correlation
    def obj(x):
        c = np.correlate(x, 1 - x, mode="full")
        bc = CURRENT_BETA * c
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (redistribute remaining mass)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def simulated_annealing(h0: np.ndarray, target: float,
                        time_budget: float, rng: np.random.Generator):
    """Simple simulated‑annealing on the mass‑transfer neighbourhood."""
    start = time.time()
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    step = 0
    reproject_interval = 5000
    while time.time() - start < time_budget * 0.95:
        grad, k = worst_gradient(h)
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        # temperature schedule
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = 0.01 * (1.0 - frac) + 1e-6
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            h = h_new
            cur_c = new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        step += 1
        if step % reproject_interval == 0:
            h = project_box_sum(h, target)
    return best_h, best_c

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 3000, step_scale: float = 0.6):
    """Greedy hill‑climb using the worst‑gradient direction."""
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            break
        eps = rng.random() * max_eps * step_scale
        cur_h[i] -= eps
        cur_h[j] += eps
        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_c, best_h = cur_c, cur_h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # --------------------------------------------------------------
    # Initialise on a modest grid (256 points)
    n_start = 256
    if "initial_h_values" in globals():
        h0 = downsample_h(np.array(initial_h_values, dtype=float), n_start)
    else:
        h0 = random_feasible(n_start, rng)

    best_h, best_c = h0, compute_c5(h0)

    # --------------------------------------------------------------
    # Phase 1 – Simulated annealing (majority of time)
    remaining = total_budget - (time.time() - start)
    if remaining > 30.0:
        anneal_time = remaining * 0.55
        h_sa, c_sa = simulated_annealing(best_h, n_start / 2.0,
                                         anneal_time, rng)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # --------------------------------------------------------------
    # Phase 2 – Short guided hill‑climb for polishing
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0:
        max_iters = int(min(4000, remaining * 250))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                       max_iters=max_iters,
                                       step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # --------------------------------------------------------------
    # Phase 3 – Upsample to 512 points and apply SLSQP
    remaining = total_budget - (time.time() - start)
    if remaining > 15.0 and n_start * 2 <= 1024:
        h_up = upsample_h(best_h, n_start * 2)
        cur_c = compute_c5(h_up)
        if cur_c < best_c:
            best_h, best_c = h_up, cur_c
        # SLSQP passes with increasing β
        remaining = total_budget - (time.time() - start)
        betas = (20., 80., 200., 800., 3000., 8000.)
        for beta in betas:
            if time.time() - start > total_budget * 0.97:
                break
            # allocate a small budget per beta
            beta_budget = remaining * 0.12
            max_iter = max(200, int(beta_budget * 4))  # very modest
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
            remaining = total_budget - (time.time() - start)

    # --------------------------------------------------------------
    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
