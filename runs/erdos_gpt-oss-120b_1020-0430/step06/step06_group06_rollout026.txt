    # Zero-pad to length 2n-1 for linear convolution
    N = 2*n - 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * G, N)[:2*n-1]  # linear convolution

            diff -= 1

import time
import numpy as np

def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (no scaling)."""
    n = h.shape[0]
    N = 2 * n - 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * G, N)
    return corr

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return float(np.max(corr) * (2.0 / n))

def enforce_weight(h: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    """Adjust binary vector h so that sum(h) == target (preserving 0/1)."""
    cur = int(h.sum())
    diff = target - cur
    while diff != 0:
        if diff > 0:
            zero_idx = np.where(h == 0)[0]
            i = int(rng.choice(zero_idx))
            h[i] = 1
            diff -= 1
        else:
            one_idx = np.where(h == 1)[0]
            i = int(rng.choice(one_idx))
            h[i] = 0
            diff += 1
    return h

def upsample_initial(init: np.ndarray, n_target: int, rng: np.random.Generator) -> np.ndarray:
    """Interpolate a parent vector to a larger size and enforce exact weight."""
    n0 = init.shape[0]
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_target, endpoint=False)
    interp = np.interp(x1, x0, init)
    bin_vec = (interp >= 0.5).astype(np.uint8)
    target = n_target // 2
    bin_vec = enforce_weight(bin_vec, target, rng)
    return bin_vec

def continuous_refine(h_bin: np.ndarray, rng: np.random.Generator,
                     beta: float, step: float, iters: int) -> np.ndarray:
    """Few steps of projected gradient descent on the smooth max objective."""
    n = h_bin.shape[0]
    target = n // 2
    h = h_bin.astype(np.float64)
    for _ in range(iters):
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        exps = np.exp(beta * (corr - max_corr))
        w = exps / np.sum(exps)
        w_rev = w[::-1]
        M = w.shape[0]               # 2n-1
        Nfft = 1 << ((M + n - 2).bit_length())   # next power of two >= M+n-1
        # term1 = (w_rev * (1-h))_{center}
        A = np.fft.rfft(w_rev, Nfft)
        B = np.fft.rfft(1.0 - h, Nfft)
        conv1 = np.fft.irfft(A * B, Nfft)[: M + n - 1]
        term1 = conv1[n - 1:n - 1 + n]
        # term2 = (w * h)_{center}
        A2 = np.fft.rfft(w, Nfft)
        B2 = np.fft.rfft(h, Nfft)
        conv2 = np.fft.irfft(A2 * B2, Nfft)[: M + n - 1]
        term2 = conv2[n - 1:n - 1 + n]
        grad = term1 - term2
        h -= step * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-9:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        step *= 0.92
    h_bin_new = (h >= 0.5).astype(np.uint8)
    h_bin_new = enforce_weight(h_bin_new, target, rng)
    return h_bin_new

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # target size (under 1000)
    n_target = 960
    target = n_target // 2

    # initialise best solution
    if "initial_h_values" in globals():
        h0 = upsample_initial(np.array(initial_h_values, dtype=float), n_target, rng)
    else:
        h0 = np.zeros(n_target, dtype=np.uint8)
        h0[rng.choice(n_target, target, replace=False)] = 1
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # main optimisation loop (restarts + refinements)
    while time.time() < start + budget_s * 0.95:
        time_left = start + budget_s * 0.95 - time.time()
        # allocate at most 30 s per attempt
        attempt_time = min(time_left * 0.5, 30.0)
        end_attempt = time.time() + attempt_time

        # hill‑climb on current best (or on a fresh random start)
        if rng.random() < 0.4 and best_h is not None:
            h_start = best_h.copy()
        else:
            h_start = np.zeros(n_target, dtype=np.uint8)
            h_start[rng.choice(n_target, target, replace=False)] = 1
        best_h, best_c = binary_hill_climb(h_start, rng, end_attempt, best_h, best_c)

        # a short continuous polishing phase
        if time.time() < start + budget_s * 0.90:
            # sequence of increasing β, modest steps
            for beta, step, it in ((50.0, 0.04, 4),

                best_h = continuous_refine(best_h, rng, beta, step, it)
                best_c = compute_c5(best_h)

    # final repair and return
    best_h = np.clip(best_h, 0, 1).astype(np.uint8)
    best_h = enforce_weight(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
