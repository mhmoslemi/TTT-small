import time
import numpy as np

# ---------- correlation utilities ----------
def _compute_corr(h: np.ndarray):
    """
    Compute the linear correlation Corr_k = sum_{i=0}^{n-1-k} h_i * (1 - h_{i+k})
    for a binary (or [0,1]) vector h of length n.
    Returns the full Corr array (size n) and the raw overlap O_k = sum h_i h_{i+k}.
    """
    n = h.shape[0]
    hf = h.astype(np.float64)

    # linear correlation h with reversed h via FFT (size = next power of two >= 2n)
    m = 1 << ((2 * n - 1).bit_length())
    rev = hf[::-1]
    fft_h = np.fft.rfft(hf, n=m)
    fft_rev = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(fft_h * fft_rev, n=m)

    # O_k = sum_i h_i * h_{i+k}, k = 0..n-1
    O = conv[n - 1 : n - 1 + n]

    # sum_h_k = sum_{i=0}^{n-1-k} h_i
    prefix = np.cumsum(hf)
    sum_h_k = prefix[n - 1 - np.arange(n)]

    Corr = sum_h_k - O
    return Corr, O


def _c5_from_h(h: np.ndarray) -> float:
    """Discrete C5 for a step function represented by h."""
    Corr, _ = _compute_corr(h)
    return np.max(Corr) * (2.0 / h.size)


# ---------- greedy improvement ----------
def _greedy_improve(h: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
    """
    Repeatedly reduce the worst shift by swapping a 1/0 pair that directly
    improves the current maximal overlap. Stops when no improving swap exists
    or the deadline is reached.
    """
    n = h.size
    while time.time() < deadline:
        Corr, _ = _compute_corr(h)
        k_w = int(np.argmax(Corr))
        if Corr[k_w] <= 0:
            break  # already optimal
        length = n - k_w
        if length <= 0:
            break

        front = h[:length]
        back = h[k_w:k_w + length]

        pos_idx = np.where((front == 1) & (back == 0))[0]
        neg_idx = np.where((front == 0) & (back == 1))[0]

        if pos_idx.size == 0 or neg_idx.size == 0:
            break  # cannot improve this shift

        i = int(rng.choice(pos_idx))
        j = int(rng.choice(neg_idx))
        # swap the bits
        h[i], h[j] = h[j], h[i]
    return h

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a length‑n binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.uint8)
    h[rng.permutation(n)[: n // 2]] = 1
    return h


def _init_from_parent(n: int, rng: np.random.Generator) -> np.ndarray:

            h = np.repeat(parent_bin, factor)
        else:

        h = _random_binary(n, rng)

        if best_c < 0.3808:
            return best_h.astype(float).tolist(), float(best_c), int(N)

    # subsequent random restarts
    while time.time() < overall_deadline:
        time_left = overall_deadline - time.time()
        if time_left < 5.0:
            break

        # start from a fresh random binary vector
        h = _random_binary(N, rng)
        # give each restart a few seconds for greedy refinement
        pass_deadline = time.time() + min(5.0, time_left * 0.2)
        h = _greedy_improve(h, rng, pass_deadline)

        c = _c5_from_h(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c < 0.3808:
                break

    # final verification
    final_c = _c5_from_h(best_h)
    return best_h.astype(float).tolist(), float(final_c), int(N)
