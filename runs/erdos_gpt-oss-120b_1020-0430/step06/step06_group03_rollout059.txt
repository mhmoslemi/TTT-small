def anneal(b_start, time_limit):
    b = b_start.copy()
    o = 1 - b
    n = len(b)
    c = np.correlate(b, o, mode='full').astype(np.int32)

    delta = np.empty_like(c, dtype=np.int32)
    idx = np.arange(n)
    T_start = 1.0
    T_end = 0.001
    start = time.time()

        elapsed = time.time() - start
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * (1 - progress) + T_end * progress
        # find worst shift

def anneal(b_start, rng, time_limit, K, idx):
    n = b_start.size
    b = b_start.copy()
    o = 1 - b
    c = np.correlate(b, o, mode='full').astype(np.int32)
    cur_max_int = c.max()
    best_b = b.copy()
    best_max_int = cur_max_int
    delta = np.empty_like(c, dtype=np.int32)
    start = time.time()
    while time.time() - start < time_limit:
        # progress for temperature
        elapsed = time.time() - start
        progress = min(elapsed / time_limit, 1.0)
        T = 1.0 * (1 - progress) + 0.001 * progress
        # worst shift
        k_max = int(c.argmax() - (n - 1))
        # candidate selection
        if rng.random() < 0.8:
            if k_max >= 0:
                i_range = idx[:n - k_max]
                i_cand = i_range[(b[i_range] == 1) & (b[i_range + k_max] == 0)]
                j_range = idx[k_max:]
                j_cand = j_range[(b[j_range] == 0) & (b[j_range - k_max] == 0)]
            else:
                k = -k_max
                i_range = idx[k:]
                i_cand = i_range[(b[i_range] == 1) & (b[i_range - k] == 0)]
                j_range = idx[:n - k]
                j_cand = j_range[(b[j_range] == 0) & (b[j_range + k] == 0)]
            if i_cand.size > 0:
                i = rng.choice(i_cand)
            else:
                i = rng.choice(np.where(b == 1)[0])
            if j_cand.size > 0:
                j = rng.choice(j_cand)
            else:
                j = rng.choice(np.where(b == 0)[0])
        else:
            i = rng.choice(np.where(b == 1)[0])
            j = rng.choice(np.where(b == 0)[0])
        # compute delta
        compute_delta_corr(b, o, i, j, K, delta)
        new_max_int = np.max(c + delta)
        if new_max_int < cur_max_int:
            c += delta
            cur_max_int = new_max_int
            b[i] = 0
            b[j] = 1
            o[i] = 1
            o[j] = 0
            if cur_max_int < best_max_int:
                best_max_int = cur_max_int
                best_b = b.copy()
        else:
            # acceptance probability
            delta_c5 = (new_max_int - cur_max_int) * (2.0 / n)
            if T > 0 and rng.random() < math.exp(-delta_c5 / T):
                c += delta
                cur_max_int = new_max_int
                b[i] = 0
                b[j] = 1
                o[i] = 1
                o[j] = 0
    best_c5 = best_max_int * (2.0 / n)
    return best_b, best_c5

    if -n + 1 <= ki_j <= n - 1:
        delta[n - 1 + ki_j] += 1

def anneal(b_start, rng, time_limit, K, idx):
    """Simulated annealing on a binary vector."""
    n = b_start.size
    b = b_start.copy()
    o = 1 - b
    c = np.correlate(b, o, mode='full').astype(np.int32)
    cur_max = c.max()
    best_b = b.copy()
    best_max = cur_max
    delta = np.empty_like(c, dtype=np.int32)
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = min(elapsed / time_limit, 1.0)
        T = 1.0 * (1 - prog) + 0.001 * prog
        k_max = int(c.argmax() - (n - 1))
        # choose indices
        if rng.random() < 0.8:
            if k_max >= 0:
                i_range = idx[:n - k_max]
                i_cand = i_range[(b[i_range] == 1) & (b[i_range + k_max] == 0)]
                j_range = idx[k_max:]
                j_cand = j_range[(b[j_range] == 0) & (b[j_range - k_max] == 0)]
            else:
                k = -k_max
                i_range = idx[k:]
                i_cand = i_range[(b[i_range] == 1) & (b[i_range - k] == 0)]
                j_range = idx[:n - k]
                j_cand = j_range[(b[j_range] == 0) & (b[j_range + k] == 0)]
            i = rng.choice(i_cand) if i_cand.size else rng.choice(np.where(b == 1)[0])
            j = rng.choice(j_cand) if j_cand.size else rng.choice(np.where(b == 0)[0])
        else:
            i = rng.choice(np.where(b == 1)[0])
            j = rng.choice(np.where(b == 0)[0])
        compute_delta_corr(b, o, i, j, K, delta)
        new_max = np.max(c + delta)
        if new_max < cur_max:
            c += delta
            cur_max = new_max
            b[i] = 0
            b[j] = 1
            o[i] = 1
            o[j] = 0
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
        else:
            d_c5 = (new_max - cur_max) * (2.0 / n)
            if T > 0 and rng.random() < math.exp(-d_c5 / T):
                c += delta
                cur_max = new_max
                b[i] = 0
                b[j] = 1
                o[i] = 1
                o[j] = 0
    return best_b, best_max * (2.0 / n)

def improve(b_start, rng, time_limit, K, idx):
    """Greedy improvement (only accepts better moves)."""
    n = b_start.size
    b = b_start.copy()
    o = 1 - b
    c = np.correlate(b, o, mode='full').astype(np.int32)
    cur_max = c.max()
    delta = np.empty_like(c, dtype=np.int32)
    start = time.time()
    while time.time() - start < time_limit:
        k_max = int(c.argmax() - (n - 1))
        if k_max >= 0:
            i_range = idx[:n - k_max]
            i_cand = i_range[(b[i_range] == 1) & (b[i_range + k_max] == 0)]
            j_range = idx[k_max:]
            j_cand = j_range[(b[j_range] == 0) & (b[j_range - k_max] == 0)]
        else:
            k = -k_max
            i_range = idx[k:]
            i_cand = i_range[(b[i_range] == 1) & (b[i_range - k] == 0)]
            j_range = idx[:n - k]
            j_cand = j_range[(b[j_range] == 0) & (b[j_range + k] == 0)]
        i = rng.choice(i_cand) if i_cand.size else rng.choice(np.where(b == 1)[0])
        j = rng.choice(j_cand) if j_cand.size else rng.choice(np.where(b == 0)[0])
        compute_delta_corr(b, o, i, j, K, delta)
        new_max = np.max(c + delta)
        if new_max < cur_max:
            c += delta
            cur_max = new_max
            b[i] = 0
            b[j] = 1
            o[i] = 1
            o[j] = 0
    return b, cur_max * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    rng = np.random.default_rng(seed)
    start = time.time()
    n = len(initial_h_values) if 'initial_h_values' in globals() else 512
    target = n // 2
    idx = np.arange(n)
    K = np.arange(-n + 1, n, dtype=np.int32)

    # binary start from any provided continuous initial vector
    if 'initial_h_values' in globals():
        vals = np.array(initial_h_values, dtype=np.float64)
        order = np.argsort(-vals)                     # descending
        b0 = np.zeros(n, dtype=np.uint8)
        b0[order[:target]] = 1
    else:
        b0 = np.zeros(n, dtype=np.uint8)
        b0[rng.choice(n, target, replace=False)] = 1

    best_b = None
    best_c5 = np.inf

    # repeatedly anneal from fresh starts
    while True:
        elapsed = time.time() - start
        remain = budget_s - elapsed
        if remain < 8.0:
            break
        anneal_time = min(remain * 0.5, 38.0)   # cap per run
        b_loc, c5_loc = anneal(b0, rng, anneal_time, K, idx)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_b = b_loc.copy()
        # new random start for next round
        b0 = np.zeros(n, dtype=np.uint8)
        b0[rng.choice(n, target, replace=False)] = 1

    # final deterministic improvement with whatever time is left
    final_elapsed = time.time() - start
    remain = budget_s - final_elapsed
    if remain > 2.0 and best_b is not None:
        b_imp, c5_imp = improve(best_b, rng, remain, K, idx)
        if c5_imp < best_c5:
            best_c5 = c5_imp
            best_b = b_imp.copy()

    if best_b is None:
        best_b = np.zeros(n, dtype=np.uint8)
        best_b[rng.choice(n, target, replace=False)] = 1
        best_c5 = np.max(np.correlate(best_b, 1 - best_b, mode='full')) * (2.0 / n)

    h_vals = best_b.astype(np.float64).tolist()
    return h_vals, float(best_c5), int(n)
