def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')

def guided_hill_climb(h, rng, max_iters=20000, step_scale=0.5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
                # ensure bounds
                # no need to clip if eps limited by max_eps
        # after change, enforce sum? The sum might change: we moved eps from i to j, sum unchanged.
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # small tolerance threshold
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
    return h
```

import time, math
import numpy as np

# ---------- core evaluation ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- helpers for continuous vectors ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (sub‑gradient)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # lag giving the max
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 20000,
    step_scale: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Mass‑preserving hill climbing using the worst‑shift gradient."""
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- binary simulated annealing ----------
def binary_sa(
    n: int,
    rng: np.random.Generator,
    time_budget: float,
    init_h: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """SA on binary vectors with O(1) swap handling."""
    start = time.time()
    half = n // 2

    if init_h is not None:
        cur_h = init_h.copy()
    else:
        cur_h = np.zeros(n, dtype=np.int8)
        cur_h[rng.choice(n, half, replace=False)] = 1

    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)
    cur_c = best_c

    ones_list = np.flatnonzero(cur_h).tolist()
    zeros_list = np.flatnonzero(1 - cur_h).tolist()

    T0, T_end = 0.2, 1e-6
    it = 0
    last_improve = 0
    # estimate total iterations for temperature schedule
    total_iters_est = max(1, int(time_budget * 3000))

    while True:
        if time.time() - start > time_budget * 0.99:
            break

        # temperature (exponential cooling)
        T = T0 * (T_end / T0) ** (it / total_iters_est)

        # propose a random swap 1<->0
        idx_i = rng.integers(half)
        idx_j = rng.integers(half)
        i = ones_list[idx_i]
        j = zeros_list[idx_j]

        # execute swap
        cur_h[i] = 0
        cur_h[j] = 1
        c_new = compute_c5(cur_h)
        delta = c_new - cur_c
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_c = c_new
            ones_list[idx_i] = j
            zeros_list[idx_j] = i
            if c_new < best_c:
                best_c, best_h, last_improve = c_new, cur_h.copy(), it
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0

        it += 1

        # periodic stagnation restart
        if it % 5000 == 0 and it - last_improve > total_iters_est // 10:
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[rng.choice(n, half, replace=False)] = 1
            cur_c = compute_c5(cur_h)
            ones_list = np.flatnonzero(cur_h).tolist()
            zeros_list = np.flatnonzero(1 - cur_h).tolist()
            last_improve = it

    return best_h, best_c


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target discretisation size
    n = 512
    half = n // 2

    # ----- obtain a continuous starter (if supplied) -----
    init_cont = None
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        # resample to our n if needed
        if init_arr.size != n:
            x_old = np.linspace(0.0, 2.0, init_arr.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init_cont = np.interp(x_new, x_old, init_arr)
        else:
            init_cont = init_arr.copy()
        init_cont = np.clip(init_cont, 0.0, 1.0)
        init_cont = enforce_sum(init_cont, rng)

    # ----- binary SA (≈60% of the budget) -----
    binary_time = total_budget * 0.60
    # build a binary start from the continuous one if present
    init_bin = None
    if init_cont is not None:
        top_idx = np.argpartition(init_cont, -half)[-half:]
        init_bin = np.zeros(n, dtype=np.int8)
        init_bin[top_idx] = 1
    best_bin, best_c = binary_sa(n, rng, binary_time, init_h=init_bin)

    # ----- continuous refinement with guided hill climbing (remaining time) -----
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 0.2:
        h_float = best_bin.astype(float)
        # number of iterations roughly proportional to remaining seconds
        max_iters = max(1000, int(remaining * 2000))
        h_float, c_float = guided_hill_climb(h_float, rng, max_iters=max_iters, step_scale=0.5)
        if c_float < best_c:
            best_c = c_float
            best_bin = (h_float > 0.5).astype(np.int8)  # preserve a binary answer for safety

    # ----- final tiny stochastic tweaks until time runs out -----
    while time.time() - start < total_budget * 0.999:
        h = best_bin.astype(float)
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            eps = rng.random() * max_eps * 0.02
            h[i] -= eps
            h[j] += eps
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_bin = (h > 0.5).astype(np.int8)

    # ----- output -----
    final_h = best_bin.astype(float)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = enforce_sum(final_h, rng)
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(final_h.size)
