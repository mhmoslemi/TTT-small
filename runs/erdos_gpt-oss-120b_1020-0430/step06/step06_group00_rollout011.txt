def gradient_refine(h, rng, step=0.01, max_iter=30):
    N = h.size
    for it in range(max_iter):
        idx = worst_shift_index(h)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1 - 2*h
        else:
            grad[:N - k] = 1 - h[k:]
            grad[k:] -= h[:N - k]
        # Use diminishing step size
        alpha = step * (0.8 ** it)
        h = h - alpha * grad
        # clip
        np.clip(h, 0, 1, out=h)
        # enforce sum
        h = enforce_sum(h, rng)
    return h

def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp for max approximation
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum precisely
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

# ---------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / N)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < max_iter:
        i = rng.integers(h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    # final tiny correction on first element
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ---------------------------------------------------------------
# Gradient of worst shift and refinement
def worst_shift_index(h: np.ndarray) -> int:
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.argmax(corr) - (N - 1)


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 25) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    N = h.size
    for it in range(max_iter):
        idx = worst_shift_index(h)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: N - k] = 1.0 - h[k:]
            grad[k:] -= h[: N - k]
        alpha = step * (0.85 ** it)
        h = h - alpha * grad
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, rng)
    return h

# Soft‑max SLSQP optimisation (convexified max)
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set (swap ε from one index to another)."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    c_curr = best_c
    N = h.size
    T0, T_end = 0.02, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        rev = False
        if max_eps <= 0:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            i, j = j, i
            rev = True
        eps = rng.random() * max_eps * (T / T0)
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:

                best_h = h_new.copy()

    def elapsed():

    # -----------------------------------------------------------------

    final_left = budget_s - elapsed()
    if final_left > 0.5:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
