def sa_optimize(h0, time_budget, rng):
    h = h0.copy()
    n = h.size
    target = n/2.0
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    # temperature parameters: start high based on c maybe.
    T = 0.02 * c  # scale
    decay = 0.9995
    iter = 0
    start = time.time()
    while time.time() - start < time_budget:
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j+=1
        # choose direction
        if rng.random() < 0.5:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            h[i] -= delta
            h[j] += delta
            # moved from i to j
            move = (i, j, delta, 1)
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            h[j] -= delta
            h[i] += delta
            move = (i, j, delta, -1)  # moved from j to i
        # compute new c
        new_c = compute_c5(h)
        # acceptance
        if new_c < c:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_c - c) / T)
        if accept:
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert
            i, j, delta, dir_flag = move
            if dir_flag == 1:
                # originally i->j, revert
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta
        # temperature decay
        T *= decay
        # occasional sum correction if drift
        if abs(h.sum() - target) > 1e-9:
            # adjust diff using random index
            diff = target - h.sum()
            # apply to a random index not at bound
            # try up to few attempts
            for _ in range(5):
                idx = rng.integers(0,n)
                if diff > 0 and h[idx] < 1.0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12: break
                elif diff < 0 and h[idx] > 0.0:
                    add = max(diff, -h[idx])
                    h[idx] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12: break
        iter += 1
        # optional gradient descent step every 2000 iterations
        if (iter % 2000) == 0:
            # do small gradient move
            grad = worst_gradient(h)
            step = 0.001  # small
            # compute delta to maintain sum: mean of grad
            delta = step * np.mean(grad)
            h2 = np.clip(h - step * grad + delta, 0.0, 1.0)
            # repair sum
            if abs(h2.sum() - target) > 1e-12:
                # simple global adjustment
                diff = target - h2.sum()
                # distribute to non-boundary idxs
                for _ in range(10):
                    idx = rng.integers(0, n)
                    if diff > 0 and h2[idx] < 1.0:
                        add = min(diff, 1.0 - h2[idx])
                        h2[idx] += add
                        diff = target - h2.sum()
                    elif diff < 0 and h2[idx] > 0.0:
                        add = max(diff, -h2[idx])
                        h2[idx] += add
                        diff = target - h2.sum()
                    if abs(diff) < 1e-12:
                        break
            new_c2 = compute_c5(h2)
            if new_c2 < c:
                h = h2
                c = new_c2
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    return best_h, best_c

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- utilities -----------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]    -= h[: n - k]
    return grad


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> None:
    """Adjust h in‑place so that sum(h)==target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1


# ---------- global state for SLSQP ---------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- simulated annealing -----------------------------------------

def sa_optimize(h0: np.ndarray, time_budget: float, rng: np.random.Generator):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    T = max(1e-8, c * 0.05)          # initial temperature
    decay = 0.9995
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # propose a transfer of mass between two positions
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h[i] -= delta
            h[j] += delta
            moved = (i, j, delta, 1)       # i → j
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h[j] -= delta
            h[i] += delta
            moved = (i, j, delta, -1)      # j → i

        new_c = compute_c5(h)

        if new_c < c:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_c - c) / T)

        if accept:
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert the move
            i, j, delta, d = moved
            if d == 1:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta

        # occasional sum‑drift correction
        if abs(h.sum() - target) > 1e-9:
            diff = target - h.sum()
            for _ in range(5):
                idx = rng.integers(0, n)
                if diff > 0 and h[idx] < 1.0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff = target - h.sum()
                elif diff < 0 and h[idx] > 0.0:
                    add = max(diff, -h[idx])
                    h[idx] += add
                    diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break

        # small deterministic gradient step every few thousand iterations
        if (it % 2000) == 0 and it > 0:
            grad = worst_gradient(h)
            step = 0.001
            delta = step * np.mean(grad)
            h2 = np.clip(h - step * grad + delta, 0.0, 1.0)
            enforce_sum(h2, target, rng)
            new_c2 = compute_c5(h2)
            if new_c2 < c:
                h = h2
                c = new_c2
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        T *= decay
        it += 1
    return best_h, best_c


# ---------- main entry point ---------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = float("inf")

    # ----- initial base vector -------------------------------------------
    if "initial_h_values" in globals():
        h_base = np.array(initial_h_values, dtype=float)
        h_base = np.clip(h_base, 0.0, 1.0)
        enforce_sum(h_base, h_base.size / 2.0, rng)
    else:
        n0 = 512
        h_base = np.zeros(n0)
        h_base[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    base_n = h_base.size
    sizes = [base_n]
    if base_n * 2 <= 1024:
        sizes.append(base_n * 2)

    # ----- time management -------------------------------------------------
    def remaining():
        return max(0.0, budget_s - (time.time() - start))

    # ----- optimisation per size -----------------------------------------
    for n in sizes:
        if remaining() <= 0.0:
            break
        # allocate roughly half of the remaining budget to the first size
        alloc = remaining() * (0.5 if n == sizes[0] else 1.0)
        restarts = 3
        per_restart = max(alloc / restarts, 1.0)

        for _ in range(restarts):
            if remaining() <= 0.0:
                break

            # ----- construct a start vector ---------------------------------
            if n == base_n:
                # first restart uses the provided base vector
                if best_h is None:
                    h_start = h_base.copy()
                else:
                    # random balanced binary
                    h_start = np.zeros(n)
                    h_start[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                # interpolate the base vector to double resolution
                xs = np.linspace(0, 2, n, endpoint=False)
                xs0 = np.linspace(0, 2, base_n, endpoint=False)
                h_start = np.interp(xs, xs0, h_base)
                h_start = np.clip(h_start, 0.0, 1.0)
                enforce_sum(h_start, n / 2.0, rng)

            # ----- simulated annealing ------------------------------------
            t_budget = min(per_restart, remaining() * 0.9)
            h_loc, c_loc = sa_optimize(h_start, t_budget, rng)
            if c_loc < best_c:
                best_c, best_h = c_loc, h_loc.copy()
            if best_c <= 0.3808:
                break
        if best_c <= 0.3808:
            break

    # ----- final SLSQP refinement -----------------------------------------
    if best_h is not None and remaining() > 5.0:
        global CURRENT_TARGET, CURRENT_BETA
        n = best_h.size
        CURRENT_TARGET = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        for beta in (5e1, 2e2, 1e3, 5e3, 2e4, 8e4, 2e5):
            if remaining() <= 0.5:
                break
            CURRENT_BETA = beta
            res = minimize(
                softmax_obj,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 500, "ftol": 1e-12, "disp": False},
            )
            x = np.clip(res.x, 0.0, 1.0)
            enforce_sum(x, n / 2.0, rng)
            c = compute_c5(x)
            if c < best_c:
                best_c, best_h = c, x.copy()
                if best_c <= 0.3808:
                    break

    # ----- final sanity check --------------------------------------------
    if best_h is None:
        best_h = h_base.copy()
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
