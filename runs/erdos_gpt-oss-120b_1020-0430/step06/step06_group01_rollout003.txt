import time
import numpy as np

# ------------------ basic helpers ------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (full correlation)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h


# ------------------ optimal transfer step ------------------

def optimal_transfer(h, O, i, j, dx, rng):
    """Move mass from i to j with epsilon that minimizes max(O+eps*d)."""
    n = h.size
    # if no room, skip
    lo = -h[i]
    hi = 1.0 - h[j]
    if lo >= hi:
        return h, O, 0.0

    # vector of shifts
    idxs = np.arange(-n + 1, n)
    i_plus = (i + idxs) % n
    i_minus = (i - idxs) % n
    j_plus = (j + idxs) % n
    j_minus = (j - idxs) % n
    d = ((h[i_plus] + h[i_minus]) - (h[j_plus] + h[j_minus])) * dx

    # binary search on epsilon
    for _ in range(30):
        mid = (lo + hi) * 0.5
        vals = O + mid * d
        k_active = np.argmax(vals)
        b = d[k_active]
        if b > 0:
            hi = mid
        else:
            lo = mid
    eps = (lo + hi) * 0.5
    # apply
    h[i] -= eps
    h[j] += eps
    O = O + eps * d
    return h, O, eps


# ------------------ main search ------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h, best_c, best_n = None, np.inf, None

    # if a parent construction is provided, start from it
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # allocate remaining time
    remaining = total_budget - (time.time() - start)
    if remaining <= 0:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        return best_h.tolist(), float(compute_c5(best_h)), int(best_h.size)

    # try two discretizations: 256 and 512 points
    n_candidates = [256, 512]
    time_per_n = remaining / len(n_candidates)

    for n in n_candidates:
        n_time_budget = time_per_n
        dx = 2.0 / n
        start_n = time.time()

        while time.time() - start_n < n_time_budget:
            # --- random binary start (exactly n/2 ones) ---
            h = np.zeros(n, dtype=float)
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0

            # initial correlation
            O = np.correlate(h, 1.0 - h, mode="full") * dx

            # hill‑climb with optimal transfers
            max_iters = 1500
            for _ in range(max_iters):
                # worst shift
                idx = int(np.argmax(O))
                k = idx - (n - 1)  # signed shift

                # gradient for that shift
                idxs = np.arange(n)
                g = (1.0 - h[(idxs + k) % n]) - h[(idxs - k) % n]
                i_max = int(np.argmax(g))
                i_min = int(np.argmin(g))
                if i_max == i_min:
                    break
                if h[i_max] <= 1e-12 or h[i_min] >= 1.0 - 1e-12:
                    break

                # optimal epsilon transfer
                h, O, _ = optimal_transfer(h, O, i_max, i_min, dx, rng)

                # early exit if already worse than the best known
                if O.max() >= best_c:
                    break

            c = O.max()
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n
                # stop early if we beat the target
                if best_c <= 0.3808:
                    best_h = np.clip(best_h, 0.0, 1.0)
                    best_h = enforce_sum(best_h, rng)
                    return best_h.tolist(), float(compute_c5(best_h)), int(best_h.size)

    # -------- final fine‑tuning on the best vector --------
    if best_h is not None:
        n = best_n
        dx = 2.0 / n
        O = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        # a few extra optimal‑transfer sweeps

    else:

        best_h = np.zeros(256, dtype=float)
        final_c = compute_c5(best_h)

    # sanitise output
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
