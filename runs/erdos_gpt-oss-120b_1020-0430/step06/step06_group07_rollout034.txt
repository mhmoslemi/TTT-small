    n = h_float.size
    dx = 2.0 / n
    total = n/2  # sum of ones always
    # prefix sum
    pref = np.concatenate(([0.0], np.cumsum(h_float)))  # length n+1
    # FFT cross-correlation
    # Use rfft for real speed:
    # compute convolution with reversed sequence:
    # h_rev = h_float[::-1]
    # conv = np.fft.irfft(np.fft.rfft(h_float, 2*n) * np.fft.rfft(h_float[::-1], 2*n), 2*n)
    # conv length 2*n, but we only need first 2*n-1 values
    # cross = conv[:2*n-1]

    h_rev_fft = np.fft.rfft(h_float[::-1], 2*n)

    cross = conv[:2*n-1]  # length 2*n-1

    sum_h = np.where(idx >= 0, pref[n-idx], total - pref[-idx])
    cross_lags = cross[n-1 + idx]
    # O(s)
    o = dx * (sum_h - cross_lags)
    return np.max(o)

# Fast C5 evaluation using FFT
def compute_c5_fast(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (FFT implementation)."""
    n = h.size
    dx = 2.0 / n
    total = h.sum()
    # prefix sums for edge overlaps
    pref = np.concatenate(([0.0], np.cumsum(h)))  # length n+1
    # cross‑correlation h with h via FFT (same as np.correlate(h, h, 'full'))
    h_fft = np.fft.rfft(h, 2 * n)
    rev_fft = np.fft.rfft(h[::-1], 2 * n)
    conv = np.fft.irfft(h_fft * rev_fft, 2 * n)  # length 2n
    cross = conv[: 2 * n - 1]                     # length 2n-1
    idx = np.arange(-n + 1, n)                   # lags
    # overlapping sums of h (sum_h_overlap for each lag)
    sum_h = np.where(idx >= 0, pref[n - idx], total - pref[-idx])
    # cross‑correlation values for the same lags
    cross_lags = cross[n - 1 + idx]
    # O(k) = dx * (sum_h - cross)
    o = dx * (sum_h - cross_lags)
    return np.max(o)

# Projection onto {0≤x≤1, sum x = target}
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# Soft‑max SLSQP optimizer (same as previous solution)
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----- initial best ------------------------------------------------
    best_h = None
    best_c = np.inf
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        best_h = h0.copy()
        best_c = compute_c5_fast(best_h)

    # ----- problem size ------------------------------------------------
    n = best_h.size if best_h is not None else 512
    half_n = n // 2

    # ----- start SA runs -----------------------------------------------
    while time.time() - start < total_budget * 0.95:
        remaining = total_budget - (time.time() - start)
        if remaining < 8.0:
            break
        # allocate a slice (max 30 s or 20 % of remaining)
        slice_time = min(30.0, remaining * 0.2)
        # estimate iterations (≈70 µs per iteration)
        max_iters = max(1, int(slice_time / 7e-5))

        # random binary start
        ones = rng.choice(n, half_n, replace=False)
        mask = np.ones(n, dtype=bool)
        mask[ones] = False
        zeros = np.nonzero(mask)[0]
        h = np.zeros(n, dtype=float)
        h[ones] = 1.0
        ones_list = ones.copy()
        zeros_list = zeros.copy()
        cur_c = compute_c5_fast(h)

        # estimate a sensible initial temperature
        deltas = []
        for _ in range(30):
            i1 = rng.integers(0, half_n)
            i0 = rng.integers(0, half_n)
            a, b = ones_list[i1], zeros_list[i0]
            h[a] = 0.0
            h[b] = 1.0
            ones_list[i1], zeros_list[i0] = b, a
            new_c = compute_c5_fast(h)
            deltas.append(abs(new_c - cur_c))
            # revert
            h[a] = 1.0
            h[b] = 0.0
            ones_list[i1], zeros_list[i0] = a, b
        avg_delta = np.mean(deltas) if deltas else 0.0
        T0 = max(1e-4, avg_delta * 2.0)

        run_best_h = h.copy()
        run_best_c = cur_c

        # ----- SA loop -------------------------------------------------
        for it in range(max_iters):
            T = max(T0 * (1.0 - it / max_iters), 1e-12)
            i1 = rng.integers(0, half_n)
            i0 = rng.integers(0, half_n)
            a, b = ones_list[i1], zeros_list[i0]

            # perform swap
            h[a] = 0.0
            h[b] = 1.0
            ones_list[i1], zeros_list[i0] = b, a

            new_c = compute_c5_fast(h)
            delta = new_c - cur_c
            if delta < 0.0 or rng.random() < np.exp(-delta / T):
                cur_c = new_c
                if new_c < run_best_c:
                    run_best_c = new_c
                    run_best_h = h.copy()
            else:
                # revert swap
                h[a] = 1.0
                h[b] = 0.0
                ones_list[i1], zeros_list[i0] = a, b

        # update global best
        if run_best_c < best_c:
            best_c = run_best_c
            best_h = run_best_h.copy()
        if best_c <= 0.3808:
            break

    # ----- optional SLSQP refinement ------------------------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 12.0 and best_h is not None:
        for beta in (200.0, 800.0, 3000.0):
            if time.time() - start > total_budget * 0.99:
                break
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=250)
            c_opt = compute_c5_fast(h_opt)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
            if best_c <= 0.3808:
                break

    # ----- final sanitisation -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact sum (numerical safety)
    best_h = project_box_sum(best_h, half_n)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
