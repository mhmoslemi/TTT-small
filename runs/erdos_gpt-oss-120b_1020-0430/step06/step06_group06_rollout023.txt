def propose_move(h, rng, max_step):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i: j += 1  # ensure j != i
    # compute max epsilon allowable
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()  # cannot move
        eps = rng.random() * max_eps * max_step
        # move from j to i (since h[j] >0 and 1-h[i]>0)
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * max_step
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place so that sum(h) == n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # tiny differences are fixed by a few random adjustments
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            # try another index
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k ∫h(1‑h_k)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Refine h0 with SLSQP on the soft‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if add == 0.0:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def gradient_polish(h: np.ndarray, rng: np.random.Generator,
                   steps: int = 30, step0: float = 0.02) -> np.ndarray:
    """Subgradient descent using the worst‑shift gradient."""
    n = h.size
    for it in range(steps):
        grad = worst_gradient(h)
        pos = np.where(grad > 0)[0]
        neg = np.where(grad < 0)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            # try opposite direction
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = step0 * (0.9 ** it) * max_eps
            nh = h.copy()
            nh[j] -= eps
            nh[i] += eps
        else:
            eps = step0 * (0.9 ** it) * max_eps
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
        h = np.clip(nh, 0.0, 1.0)
    enforce_sum(h, rng)
    return h


def upscale_h(low: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of low‑resolution h to a higher resolution."""
    n_low = low.size
    x_low = np.linspace(0.0, 1.0, n_low, endpoint=False)
    x_target = np.linspace(0.0, 1.0, n_target, endpoint=False)
    return np.interp(x_target, x_low, low)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0_low = np.array(initial_h_values, dtype=float)
    else:
        # fallback: half ones, half zeros shuffled
        n0 = 512
        h0_low = np.zeros(n0, dtype=float)
        h0_low[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    # aim for double resolution, but keep under 1200 points
    n_target = min(1024, h0_low.size * 2)
    h0 = upscale_h(h0_low, n_target)
    enforce_sum(h0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing ----------------------------------------
    current_h = best_h.copy()
    current_c = best_c
    T = 0.05                       # initial temperature
    alpha = 0.9999                 # cooling factor per iteration
    scale = 0.5                    # move magnitude factor
    it = 0
    SA_deadline = start + budget_s * 0.60
    n = n_target
    while time.time() < SA_deadline:
        # pick move type
        r = rng.random()
        if r < 0.15:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            nh = current_h.copy()
            nh[i], nh[j] = nh[j], nh[i]
        elif r < 0.45:
            # transfer mass between two positions
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_eps = min(current_h[i], 1.0 - current_h[j])
            if max_eps <= 0.0:
                max_eps = min(current_h[j], 1.0 - current_h[i])
                if max_eps <= 0.0:
                    continue
                eps = rng.random() * max_eps * scale
                nh = current_h.copy()
                nh[j] -= eps
                nh[i] += eps
            else:
                eps = rng.random() * max_eps * scale
                nh = current_h.copy()
                nh[i] -= eps
                nh[j] += eps
        else:

            grad = worst_gradient(current_h)
            i = int(np.argmax(grad))

            if i == j:

                nh[j] -= eps

                nh = current_h.copy()
                nh[i] -= eps
                nh[j] += eps
        nc = compute_c5(nh)
        delta = nc - current_c
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

    # ----- final repair ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
