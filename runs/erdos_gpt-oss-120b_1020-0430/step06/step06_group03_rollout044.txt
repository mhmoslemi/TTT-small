import time, math, numpy as np

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def correlation_fft(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    prod = H * np.conj(G)
    corr = np.fft.irfft(prod, n=L)
    return corr[:2 * n - 1]

def fft_convolve(a, b):
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)
    return conv[:2 * n - 1]

def compute_c5_fft(h, dx):
    O = correlation_fft(h)
    return O.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 1024
        h = np.zeros(n)

    target = n / 2.0
    h = project_onto_box_sum(h, target)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5_fft(h, dx)

    beta_list = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 60000]
    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for beta in beta_list:
        if time.time() - start > budget_s * 0.98:
            break
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(20000, remaining * 15))
        if max_iters < 5:
            break
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        # base step size inversely proportional to sqrt(beta)
        base_lr = 0.15 / math.sqrt(beta)
        for it in range(max_iters):
            t += 1
            # correlation O (raw sums)
            O = correlation_fft(h)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
