import time
import math
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Basic evaluation utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 using np.correlate (linear correlation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 using FFT (same result as compute_c5 up to rounding)."""
    n = h.size
    N = 2 * n - 1
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    return np.max(corr) * (2.0 / n)


# ------------------------------------------------------------
# Simple sum‑enforcement for continuous vectors
# ------------------------------------------------------------
def enforce_sum_fast(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == target by touching a single element."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase sum
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0:
            # fallback: distribute evenly
            h += diff / h.size
        else:
            i = rng.choice(cand)
            h[i] += diff
    else:
        # need to decrease sum
        cand = np.where(h > 1e-12)[0]
        if cand.size == 0:
            h += diff / h.size
        else:
            i = rng.choice(cand)
            h[i] += diff
    np.clip(h, 0.0, 1.0, out=h)
    return h


# ------------------------------------------------------------
# Gradient for the worst shift (used in local continuous search)
# ------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the max‑overlap term w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift giving the worst overlap
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ------------------------------------------------------------
# Soft‑max objective and SLSQP refinement
# ------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny repair if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x

# Thue–Morse binary sequence generator

def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]
