def weighted_gradient(h, w):
    N = h.size
    G = np.zeros(N, dtype=float)
    # positive shifts
    for s in range(N):
        w_s = w[N-1 + s]
        if w_s == 0:
            continue
        # a_s: for j in 0..N-s-1: add w_s * (1 - h[j+s])
        G[:N-s] += w_s * (1.0 - h[s:])   # vectorized
        # b_s: for j in s..N-1: subtract w_s * h[j-s]
        G[s:] -= w_s * h[:N-s]   # vectorized
    # negative shifts
    for p in range(1, N):
        w_s = w[N-1 - p]   # shift -p
        if w_s == 0:
            continue
        # contribute: for j from p..N-1: + w_s * (1 - h[j-p])
        G[p:] += w_s * (1.0 - h[:N-p])
        # for j from 0..N-p-1: - w_s * h[j+p]
        G[:N-p] -= w_s * h[p:]
    return G
```

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def weighted_gradient(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Return Σ w_s ∇corr_s for all shifts."""
    N = h.size
    G = np.zeros(N, dtype=float)
    off = N - 1
    # positive shifts including 0
    for s in range(N):
        ws = w[off + s]
        if ws == 0.0:
            continue
        if s < N:
            G[: N - s] += ws * (1.0 - h[s:])      # a_s part
            G[s:] -= ws * h[: N - s]             # b_s part
    # negative shifts
    for p in range(1, N):
        ws = w[off - p]
        if ws == 0.0:
            continue
        G[p:] += ws * (1.0 - h[: N - p])        # (1 - h[j-p])
        G[: N - p] -= ws * h[p:]                # -h[j+p]
    return G


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def pgd_optimize(h0: np.ndarray, rng: np.random.Generator,
                 time_limit: float) -> tuple:
    """Projected gradient descent with soft‑max smoothing."""
    N = h0.size
    target = N / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    beta_seq = [5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]
    base_lr = 0.5
    for beta in beta_seq:
        lr = base_lr / (beta + 1.0)
        max_iters = 3000
        for _ in range(max_iters):
            if time.time() - start > time_limit:
                return best_h, best_c
            # correlation and soft‑max weights
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_corr = np.max(corr)
            w = np.exp(beta * (corr - max_corr))  # stable
            S = w.sum()
            if S == 0.0:  # numerical safety
                S = 1.0
            logS = np.log(S)
            factor = 1.0 - beta * max_corr - logS  # = 1 - beta*O
            G = weighted_gradient(h, w)
            grad = factor * G / S
            h = h - lr * grad
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

            best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)

    best_h = enforce_sum(best_h, rng)
