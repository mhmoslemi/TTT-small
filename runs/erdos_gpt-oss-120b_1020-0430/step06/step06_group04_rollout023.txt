def greedy_binary_improve(h_bin, rng, deadline):
    n = len(h_bin)
    # ensure exact sum
    adjust_sum_exact(h_bin, rng)
    while time.time() < deadline:
        v = np.correlate(h_bin, 1.0 - h_bin, mode="full")
        max_v = v.max()
        # identify worst shift(s)
        worst_shift_idxs = np.where(v == max_v)[0]  # these correspond to shift k = idx - (n - 1)
        shifts = worst_shift_idxs - (n - 1)
        # compute total neighbor counts
        total_neighbor = np.zeros(n, dtype=int)
        for k in shifts:
            if k >= 0:
                plus = np.concatenate([h_bin[k:], np.zeros(k, dtype=int)])
                minus = np.concatenate([np.zeros(k, dtype=int), h_bin[:n-k]])
            else:
                kk = -k
                plus = np.concatenate([np.zeros(kk, dtype=int), h_bin[:n-kk]])
                minus = np.concatenate([h_bin[kk:], np.zeros(kk, dtype=int)])
            total_neighbor += plus + minus
        # positions of ones and zeros
        ones = np.flatnonzero(h_bin == 1)
        zeros = np.flatnonzero(h_bin == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # compute neighbor counts for ones and zeros
        neighbor_ones = total_neighbor[ones]
        neighbor_zeros = total_neighbor[zeros]
        # pick i (one) with minimal neighbor, j (zero) with maximal neighbor
        i_min_idx = neighbor_ones.argmin()
        j_max_idx = neighbor_zeros.argmax()
        i = ones[i_min_idx]
        j = zeros[j_max_idx]
        if total_neighbor[i] >= total_neighbor[j]:
            # no improvement possible
            break
        # perform swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        # continue loop
    return h_bin

import time
import numpy as np

# --------------------------- utilities ---------------------------

def compute_c5(h):
    """Maximum overlap integral (discrete)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Patch h so that sum(h) == n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    """Gradient of smooth‑max of overlaps."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")        # length 2n‑1
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()                   # soft‑max weights
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)                     # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

# --------------------------- gradient descent ---------------------------

def gradient_descent(h, rng, deadline,
                    betas=None, it_per_beta=12, base_step=0.30):
    """Projected gradient descent on smooth‑max."""
    if betas is None:
        betas = [5, 20, 80, 200, 800, 2000,
                 8000, 20000, 80000, 200000]
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = cand
            h = best_h
    return h

# --------------------------- greedy binary improvement ---------------------------

def greedy_binary_improve(h_bin, rng, deadline):
    """Improve a binary vector by swapping 1↔0 using worst shifts."""
    n = h_bin.shape[0]
    # ensure exact feasibility
    adjust_sum_exact(h_bin, rng)
    h_int = (h_bin > 0.5).astype(np.int8)

    while time.time() < deadline:
        v = np.correlate(h_int, 1 - h_int, mode="full")
        max_v = v.max()
        worst_idxs = np.where(v == max_v)[0]
        shifts = worst_idxs - (n - 1)

        total_neighbor = np.zeros(n, dtype=np.int32)
        for k in shifts:
            if k >= 0:
                forward = np.concatenate((h_int[k:], np.zeros(k, dtype=np.int8)))
                backward = np.concatenate((np.zeros(k, dtype=np.int8), h_int[:n - k]))
            else:
                kk = -k
                forward = np.concatenate((np.zeros(kk, dtype=np.int8), h_int[:n - kk]))
                backward = np.concatenate((h_int[kk:], np.zeros(kk, dtype=np.int8)))
            total_neighbor += forward + backward

        ones = np.flatnonzero(h_int)
        zeros = np.flatnonzero(1 - h_int)
        if ones.size == 0 or zeros.size == 0:
            break

        neighbor_ones = total_neighbor[ones]
        neighbor_zeros = total_neighbor[zeros]

        i = ones[neighbor_ones.argmin()]
        j = zeros[neighbor_zeros.argmax()]

        if total_neighbor[i] >= total_neighbor[j]:
            break

        h_int[i] = 0
        h_int[j] = 1

    h_bin[:] = h_int.astype(float)
    return h_bin

# --------------------------- main driver ---------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97   # safety margin

    # ---- initial vector -------------------------------------------------
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        h0 = np.array(parent, dtype=float)
        n = h0.size
    else:
        n = 720
        h0 = rng.random(n)
    h0 = h0 / h0.sum() * (n / 2.0)
    np.clip(h0, 0.0, 1.0, out=h0)
    adjust_sum_exact(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_n = n

    # ---- primary optimisation (gradient + greedy) -----------------------
    def optimise_one(init_h):
        """Run full pipeline on a given starting vector."""
        h = init_h.copy()
        # gradient descent (full schedule)
        h = gradient_descent(h, rng, deadline)
        c = compute_c5(h)
        nonlocal best_h, best_c
        if c < best_c:
            best_c, best_h = c, h.copy()
        # binary rounding + greedy refinement
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        h_bin = greedy_binary_improve(h_bin, rng, deadline)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin.copy()
        return h_bin

    # first use the parent (or the constructed h0)
    optimise_one(h0)

    # ---- additional random restarts ------------------------------------
    restarts = 6
    for r in range(restarts):
        if time.time() > deadline:
            break
        h_rand = rng.random(n)
        h_rand = h_rand / h_rand.sum() * (n / 2.0)
        np.clip(h_rand, 0.0, 1.0, out=h_rand)
        adjust_sum_exact(h_rand, rng)
        optimise_one(h_rand)

    # ---- perturb‑and‑refine loop (escape local optima) -----------------
    perturb_trials = 5
    for _ in range(perturb_trials):
        if time.time() > deadline:
            break
        cur = best_h.copy()
        # perform a few random swaps
        for __ in range(3):
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur[i] = 0
            cur[j] = 1
        cur = greedy_binary_improve(cur, rng, deadline)
        c_cur = compute_c5(cur)
        if c_cur < best_c:
            best_c, best_h = c_cur, cur.copy()

    # ---- optional up‑sampling if time permits -------------------------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        # shorter gradient schedule for the larger array
        short_betas = [5, 20, 80, 200, 800, 2000]
        h_up = gradient_descent(h_up, rng, deadline,
                                betas=short_betas, it_per_beta=6,
                                base_step=0.20)
        h_up_bin = np.rint(h_up).astype(float)
        adjust_sum_exact(h_up_bin, rng)
        h_up_bin = greedy_binary_improve(h_up_bin, rng, deadline)
        c_up = compute_c5(h_up_bin)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up_bin.copy(), n_up

    # ---- final safety check --------------------------------------------
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
