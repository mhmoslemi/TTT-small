    delta_i = -eps
    delta_j = eps
    n = len(h)
    # shift range k from -(n-1) to n-1, index offset = n-1
    # Compute contributions for each k using vectorization.

    # First linear contributions:
    # For each k, term1 = delta_i * h[i + k] if 0<=i+k<n else 0
    # term2 = delta_j * h[j + k] if 0<=j+k<n else 0
    # term3 = delta_i * h[i - k] if 0<=i - k<n else 0
    # term4 = delta_j * h[j - k] if 0<=j - k<n else 0

    j_minus = j - ks
    mask_j_minus = (j_minus >= 0) & (j_minus < n)

    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst (largest) overlap shift."""
    n = h.size
    cross = np.correlate(h, h, mode="full")
    k = int(np.argmin(cross[n - 1 :]))  # shift ≥0 with minimal autocorrelation
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator,
                   scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h0: np.ndarray, max_iters: int,
                        rng: np.random.Generator,
                        T0: float, eps_max: float = 1.0):
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    for it in range(max_iters):
        T = T0 * max(0.0, 1.0 - it / max_iters)  # linear cooling
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # step size shrinks with temperature, never below 1 % of max_eps
        scale = max(T / T0, 0.01)
        eps = rng.random() * max_eps * scale
        if eps <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h, cur_c = h_new, c_new
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = random_feasible(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # main simulated‑annealing loop
    no_improve = 0
    while True:
        elapsed = time.time() - start
        remain = total - elapsed
        if remain < 2.0:
            break
        # iterations budget (≈6000 it/s)
        iters = int(min(60000, remain * 6000))
        T0 = max(0.001, (best_c - 0.38) * 0.8)  # larger if we are far from target
        sa_h, sa_c = simulated_annealing(best_h, iters, rng, T0, eps_max=1.0)
        if sa_c < best_c - 1e-9:
            best_h, best_c = sa_h, sa_c
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                # random restart to escape stagnation
                best_h = random_feasible(n, rng)
                best_c = compute_c5(best_h)
                no_improve = 0

    # final hill‑climb refinement
    elapsed = time.time() - start
    remain = total - elapsed
    if remain > 1.0:
        max_it = int(min(5000, remain * 4000))
        hc_h, hc_c = guided_hill_climb(best_h, rng, max_iters=max_it,
                                       step_scale=0.7)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c

    # sanitise and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
