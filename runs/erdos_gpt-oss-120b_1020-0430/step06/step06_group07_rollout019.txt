    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_refine(h, dx, time_end):
    n = h.size
    O_raw = np.correlate(h, 1 - h, mode='full')
    cur_c = O_raw.max() * dx
    best_h = h.copy()

        t += 1

        k_star = k_idx - (n - 1)
        # compute gradient: g_i = 1 - h[i + k_star] if in range else 1
        if k_star >=0:
            g = np.ones_like(h)
            g[: n - k_star] -= h[k_star:]

            g = np.ones_like(h)

            best_h = h_new.copy()

def upsample(h, rng):

    """C5 for a step function on [0,2]."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def deterministic_reduce(h, O_raw, dx, deadline):
    """Greedy reduction of the worst shift."""
    n = h.size
    hpad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
    best_h, best_c = h.copy(), O_raw.max() * dx
    no_imp = 0
    while time.time() < deadline:
        k_idx = np.argmax(O_raw)
        cur_c = O_raw[k_idx] * dx
        k_star = k_idx - (n - 1)
        if k_star >= 0:
            diff_i = np.concatenate([h[k_star:], np.zeros(k_star, dtype=float)])
        else:
            diff_i = np.concatenate([np.zeros(-k_star, dtype=float), h[: n + k_star]])
        zero = diff_i <= 1e-9
        one = diff_i >= 1.0 - 1e-9
        src_cand = np.where(zero & (h > 1e-12))[0]
        dst_cand = np.where(one & (h < 1.0 - 1e-12))[0]
        if src_cand.size == 0 or dst_cand.size == 0:
            break
        src = src_cand[np.argmax(h[src_cand])]
        dst = dst_cand[np.argmin(h[dst_cand])]
        delta = min(h[src], 1.0 - h[dst])
        if delta <= 1e-12:
            break
        diff_raw = hpad[src: src + 2 * n - 1] - hpad[dst: dst + 2 * n - 1]
        O_raw += delta * diff_raw
        h[src] -= delta
        h[dst] += delta
        hpad[src + n - 1] = h[src]
        hpad[dst + n - 1] = h[dst]
        new_c = O_raw.max() * dx
        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= 2000:
            break
    return best_h, best_c, O_raw, hpad

def simulated_annealing(h, O_raw, dx, deadline, cur_best_c):
    """Simple SA with a linear cooling schedule."""
    n = h.size
    hpad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
    cur_max = O_raw.max()
    cur_c = cur_max * dx
    best_h, best_c = h.copy(), cur_c
    T_start = cur_max * 0.3
    T_end = max(1e-12, 1e-9 / dx)
    start = time.time()
    total = max(1e-12, deadline - start)
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T_start * (T_end / T_start) ** frac
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        src, dst = i, j
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0:
            src, dst = dst, src
            max_delta = min(h[src], 1.0 - h[dst])
            if max_delta <= 0:
                continue
        scale = 1.0 - 0.9 * frac
        delta = np.random.random() * max_delta * scale
        diff_raw = hpad[src: src + 2 * n - 1] - hpad[dst: dst + 2 * n - 1]
        new_max = np.max(O_raw + delta * diff_raw)
        if new_max < cur_max or (T > 0 and np.random.random() < np.exp(-(new_max - cur_max) / T)):
            O_raw += delta * diff_raw
            cur_max = new_max
            cur_c = cur_max * dx
            h[src] -= delta
            h[dst] += delta
            hpad[src + n - 1] = h[src]
            hpad[dst + n - 1] = h[dst]
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c

def subgradient_refine(h, dx, deadline):
    """Projected sub‑gradient descent on the max‑overlap."""
    n = h.size
    O_raw = np.correlate(h, 1 - h, mode="full")
    cur_c = O_raw.max() * dx
    best_h, best_c = h.copy(), cur_c
    t = 0
    while time.time() < deadline:
        t += 1
        k_idx = np.argmax(O_raw)
        k_star = k_idx - (n - 1)
        if k_star >= 0:
            g = np.ones_like(h)
            g[: n - k_star] -= h[k_star:]
        else:
            g = np.ones_like(h)
            g[-k_star:] -= h[: n + k_star]
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * g, n / 2.0)
        O_raw = np.correlate(h_new, 1 - h_new, mode="full")
        new_c = O_raw.max() * dx
        if new_c < best_c:
            best_c, best_h = new_c, h_new.copy()
        h = h_new
        if best_c <= 0.3808:
            break
    return best_h, best_c

def upsample(h, rng):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # initialise
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, n / 2.0)
    else:
        n = 512
        v = rng.random(n)
        h = project_box_sum(v, n / 2.0)

    dx = 2.0 / n
    O_raw = np.correlate(h, 1 - h, mode="full")
    cur_c = O_raw.max() * dx
    best_h, best_c = h.copy(), cur_c

    # Phase 1: deterministic reduction (≈30% time)
    remaining = total - (time.time() - start)
    deadline = start + remaining * 0.30
    if deadline > time.time():
        best_h, best_c, O_raw, _ = deterministic_reduce(best_h, O_raw, dx, deadline)

    # Phase 2: simulated annealing (majority of remaining time)
    remaining = total - (time.time() - start)
    deadline = start + total - max(5.0, remaining * 0.05)
    O_raw = np.correlate(best_h, 1 - best_h, mode="full")
    best_h, best_c = simulated_annealing(best_h, O_raw, dx, deadline, best_c)

    # Phase 3: sub‑gradient polish (final few seconds)
    remaining = total - (time.time() - start)
    deadline = start + total - 2.0
    if deadline > time.time():
        best_h, best_c = subgradient_refine(best_h, dx, deadline)

    # Phase 4: optional up‑sampling if time permits
    remaining = total - (time.time() - start)
    if remaining > 8.0 and n * 2 <= 1024:
        h_up = upsample(best_h, rng)
        n_up = h_up.size
        dx_up = 2.0 / n_up
        O_up = np.correlate(h_up, 1 - h_up, mode="full")
        deadline = start + total - 2.0
        if deadline > time.time():
            h_up, c_up, O_up, _ = deterministic_reduce(h_up, O_up, dx_up, deadline)
            if time.time() < deadline:
                h_up, c_up = subgradient_refine(h_up, dx_up, deadline)
            if c_up < best_c:
                best_h, best_c = h_up, c_up
                n, dx = n_up, dx_up

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
