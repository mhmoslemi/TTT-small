def compute_c5(h):
    n = h.size
    L = 2*n - 1
    # compute linear correlation h * (1-h) reversed
    ha = np.fft.fft(h, L)
    hb = np.fft.fft((1.0 - h)[::-1], L)
    corr = np.fft.ifft(ha * hb).real
    return np.max(corr) * (2.0 / n)
```

def guided_hill_climb(h, rng, max_iters, step_scale):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = np.argmax(grad)
        j = np.argmin(grad)
        if i == j:
            # random perturb by small epsilon

            j = rng.integers(n)

                j = rng.integers(n)

            h[j] += eps

            h[i] += eps
            h[j] -= eps
        it += 1
        if it % 4000 == 0:
            diff = n / 2.0 - h.sum()
            if abs(diff) > 1e-9:
                idxs = np.where((h > 0) & (h < 1))[0]
                if idxs.size:
                    idx = rng.choice(idxs)
                    h[idx] = np.clip(h[idx] + diff, 0, 1)
                else:
                    h[0] = np.clip(h[0] + diff, 0, 1)
    best_h = enforce_sum(best_h, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # choose resolution
    n = 1024 if total_budget > 600 else 512

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0, 2, h0.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        h0 = rng.random(n)
        h0 = enforce_sum(h0, rng)

    best_h, best_c = h0, compute_c5(h0)

    while time.time() - start < total_budget * 0.93:
        remaining = total_budget - (time.time() - start)
        chunk = min(5.0, remaining * 0.4)
        h_new, c_new = hill_climb_chunk(best_h.copy(), rng, chunk)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        # occasional SLSQP refinement
        if time.time() - start < total_budget * 0.90:
            for beta in (50.0, 200.0, 800.0):
                if time.time() - start >= total_budget * 0.90:
                    break
                h_s = slsqp_optimize(best_h.copy(), beta, rng, max_iter=800)
                c_s = compute_c5(h_s)
                if c_s < best_c:
                    best_h, best_c = h_s, c_s
        # small random perturbation to escape
        if time.time() - start < total_budget * 0.92:
            h_noise = best_h.copy()
            h_noise += (rng.random(n) - 0.5) * 0.02
            h_noise = np.clip(h_noise, 0.0, 1.0)
            h_noise = enforce_sum(h_noise, rng)
            c_noise = compute_c5(h_noise)
            if c_noise < best_c:
                best_h, best_c = h_noise, c_noise

    remaining = total_budget - (time.time() - start)
    if remaining > 1.0:
        h_fin, c_fin = hill_climb_chunk(best_h.copy(), rng, remaining * 0.5)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin
        h_ref = gradient_refine(best_h.copy(), rng, step=0.01, max_iter=8)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
