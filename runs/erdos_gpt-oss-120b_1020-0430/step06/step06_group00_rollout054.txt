def refine_h(h, rng, time_budget):
    start = time.time()
    n = h.size
    target_sum = n / 2.0
    best_local_h = h.copy()
    best_local_c = compute_c5_fft(h)
    curr_h = h.copy()
    curr_c = best_local_c
    # set initial step size
    step0 = 0.4  # can adjust based on n
    # temperature schedule
    T0 = 0.02
    T_end = 1e-6
    # total iterations limit maybe set high (like 100000)
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # compute progress ratio
        progress = elapsed / time_budget
        step = step0 * (1.0 - progress) + 1e-5 * progress
        T = T0 * (T_end / T0) ** progress

        # compute worst gradient
        g = worst_gradient(curr_h)  # O(N)
        # candidate
        h_new = curr_h - step * g
        # clip
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # enforce sum
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[i])
                    if add <= 0:
                        continue
                    h_new[i] += add
                    diff -= add
                else:
                    sub = min(-diff, h_new[i])
                    if sub <= 0:
                        continue
                    h_new[i] -= sub
                    diff += sub
        # evaluate
        c_new = compute_c5_fft(h_new)

        # accept condition
        if c_new < curr_c or rng.random() < np.exp(-(c_new - curr_c) / max(T, 1e-12)):
            curr_h = h_new
            curr_c = c_new

        # track best
        if curr_c < best_local_c:
            best_local_c = curr_c
            best_local_h = curr_h.copy()
        # optional random move every some steps

import time
import numpy as np
import math

def compute_c5_fft(h: np.ndarray) -> float:
    n = h.size
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    H2 = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * H2, n=L)
    return np.max(conv) * (2.0 / n)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.fft.irfft(np.fft.rfft(h) * np.conj(np.fft.rfft(1.0 - h)), n=2 * n - 1)
    idx = int(np.argmax(corr) - (n - 1))
    k = abs(idx)
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
    return g

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                attempts += 1
                continue
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            if sub <= 0:
                attempts += 1
                continue
            h[i] -= sub
            diff += sub
        attempts += 1
    return h

def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def refine_h(h: np.ndarray, rng: np.random.Generator, time_budget: float) -> tuple:
    start = time.time()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    cur_h = h.copy()
    cur_c = best_c
    step0 = 0.4
    T0, T_end = 0.02, 1e-6
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        prog = elapsed / time_budget
        step = step0 * (1.0 - prog) + 1e-5 * prog
        T = T0 * (T_end / T0) ** prog
        g = worst_gradient(cur_h)
        cand = cur_h - step * g
        np.clip(cand, 0.0, 1.0, out=cand)
        diff = target - cand.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - cand[i])
                    if add <= 0:
                        continue
                    cand[i] += add
                    diff -= add
                else:
                    sub = min(-diff, cand[i])
                    if sub <= 0:
                        continue
                    cand[i] -= sub
                    diff += sub
        c_new = compute_c5_fft(cand)
        if c_new < cur_c or rng.random() < math.exp(-(c_new - cur_c) / max(T, 1e-12)):
            cur_h, cur_c = cand, c_new
            if c_new < best_c:
                best_c, best_h = c_new, cand.copy()
        # occasional random transfer
        if int(elapsed * 10) % 7 == 0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.2
                if eps > 0:
                    h2 = cur_h.copy()
                    h2[i] -= eps
                    h2[j] += eps
                    c2 = compute_c5_fft(h2)
                    if c2 < cur_c or rng.random() < math.exp(-(c2 - cur_c) / max(T, 1e-12)):
                        cur_h, cur_c = h2, c2
                        if c2 < best_c:
                            best_c, best_h = c2, h2.copy()
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n0 = 256
        h0 = np.zeros(n0)
        idx = rng.choice(n0, n0 // 2, replace=False)
        h0[idx] = 1.0

    best_h = h0.copy()
    best_c = compute_c5_fft(best_h)
    best_n = best_h.size

    phases = [
        (256, 0.15),
        (512, 0.30),
        (1024, 0.35),
        (1024, 0.20),
    ]
    cum = 0.0
    for n_target, frac in phases:
        cum += frac
        target_time = start + total * cum
        if time.time() >= target_time:
            continue
        if best_h.size != n_target:
            h = upsample_h(best_h, n_target)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = best_h.copy()
        phase_time = target_time - time.time()
        if phase_time <= 0:
            continue
        h_ref, c_ref = refine_h(h, rng, phase_time)
        if c_ref < best_c:
            best_h, best_c, best_n = h_ref, c_ref, n_target
        best_h = h_ref

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
