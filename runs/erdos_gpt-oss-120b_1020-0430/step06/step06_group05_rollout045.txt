def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n/2

    # initialize population
    Npop = 32
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce sum exactly
        h0 = enforce_sum(h0, target_sum, rng)
        population.append(h0)
    while len(population) < Npop:
        # random binary
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
        population.append(h)
    # evaluate each
    scores = np.array([compute_c5(h) for h in population])
    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # main loop
    while time.time() - start < budget_s * 0.97:
        # parent selection via tournament
        # tournament size 4
        idxs = rng.choice(Npop, size=4, replace=False)
        tournament_scores = scores[idxs]
        parent1_idx = idxs[np.argmin(tournament_scores)]
        # second parent: from remaining tournament participants (excluding parent1)
        remaining = idxs[idxs != parent1_idx]
        parent2_idx = remaining[np.argmin(scores[remaining])]
        p1 = population[parent1_idx]
        p2 = population[parent2_idx]

        # crossover
        cut = rng.integers(1, n)  # cut point not at edges?
        child = np.concatenate((p1[:cut], p2[cut:]))
        # fix sum
        cur_sum = child.sum()
        diff = target_sum - cur_sum
        if diff != 0:
            # need to flip diff entries
            if diff > 0:
                zeros = np.flatnonzero(child == 0)
                if zeros.size > 0:
                    chosen = rng.choice(zeros, size=int(diff), replace=False)
                    child[chosen] = 1.0
            else:
                ones = np.flatnonzero(child == 1)
                if ones.size > 0:
                    chosen = rng.choice(ones, size=int(-diff), replace=False)
                    child[chosen] = 0.0
        # now child sum is target

        # mutation: swap bits with probability p_mut
        p_mut = 0.3
        if rng.random() < p_mut:
            # determine number of swaps 1 or 2
            n_swaps = rng.integers(1, 4)  # 1-3 swaps
            for _ in range(n_swaps):
                ones_idx = np.flatnonzero(child == 1)
                zeros_idx = np.flatnonzero(child == 0)
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                child[i] = 0.0
                child[j] = 1.0

        # optionally apply small gradient refinement
        # We'll have remaining time; apply refinement if enough time left
        if time.time() - start < budget_s * 0.95:
            child = gradient_refine(child, target_sum, rng, step=0.02, max_iter=6)

        child_score = compute_c5(child)

        # replace worst if better
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            # update best
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()
        # else option to keep best unchanged

    # final polish: more gradient steps + SLSQP if time
    if time.time() - start < budget_s * 0.9:

import time
import numpy as np

# ----- basic helpers ----- #
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple iterative transfer
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 10) -> np.ndarray:
    """Sub‑gradient descent using the worst‑shift direction."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient to negative‑gradient indices."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


# ----- SLSQP (soft‑max) ----- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

def crossover_fix(p1: np.ndarray, p2: np.ndarray, target: float,
                  rng: np.random.Generator) -> np.ndarray:
    n = p1.size
    cut = rng.integers(1, n)
    child = np.concatenate((p1[:cut], p2[cut:]))
    cur = child.sum()
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(child == 0)
        if zeros.size:
            chosen = rng.choice(zeros, size=int(diff), replace=False)
            child[chosen] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(child == 1)
        if ones.size:
            chosen = rng.choice(ones, size=int(-diff), replace=False)
            child[chosen] = 0.0
    return child


def mutate_swap(h: np.ndarray, rng: np.random.Generator, p_mut: float = 0.3):
    if rng.random() >= p_mut:
        return h
    n_swaps = rng.integers(1, 4)  # 1–3 swaps
    for _ in range(n_swaps):
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
    return h


def tournament_select(scores: np.ndarray, rng: np.random.Generator, k: int = 4):
    idxs = rng.choice(scores.size, size=k, replace=False)
    best = idxs[np.argmin(scores[idxs])]
    return best


# ----- main entry point ----- #
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0

    # ---- initialise population ---- #
    pop_size = 32
    population = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
        population.append(h0)
    while len(population) < pop_size:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        population.append(h)

    scores = np.array([compute_c5(h) for h in population])
    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # ---- evolutionary loop ---- #
    while time.time() - start < budget_s * 0.97:
        # parent selection
        p1_idx = tournament_select(scores, rng, k=4)
        # ensure second parent is different
        while True:
            p2_idx = tournament_select(scores, rng, k=4)
            if p2_idx != p1_idx:
                break
        p1 = population[p1_idx]
        p2 = population[p2_idx]

        # crossover and fix sum
        child = crossover_fix(p1, p2, target_sum, rng)

        # mutation
        child = mutate_swap(child, rng, p_mut=0.3)

        # local gradient refinement (light)
        if time.time() - start < budget_s * 0.95:
            child = gradient_refine(child, target_sum, rng,
                                   step=0.02, max_iter=6)

        child_score = compute_c5(child)

        # replace worst if better
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()

        # occasional direct improvement of the current best
        if time.time() - start < budget_s * 0.94:
            _, grad_best = worst_shift_and_grad(best_h)
            cand = mass_transfer_move(best_h, grad_best, alpha=0.07,
                                     target=target_sum, rng=rng)
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
                # possibly insert into population
                worst_idx = np.argmax(scores)
                if cand_score < scores[worst_idx]:
                    population[worst_idx] = cand
                    scores[worst_idx] = cand_score

    # ---- final polishing ---- #
    # deeper gradient refinement
    if time.time() - start < budget_s * 0.90:
        best_h = gradient_refine(best_h, target_sum, rng,
                                 step=0.01, max_iter=15)

    # one SLSQP pass if time permits
    if time.time() - start < budget_s * 0.85:
        best_h = slsqp_refine(best_h, target_sum, rng,
                              beta=20000.0, max_iter=400)
        best_score = compute_c5(best_h)

    # ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    best_score = compute_c5(best_h)

    return best_h.tolist(), float(best_score), int(n)
