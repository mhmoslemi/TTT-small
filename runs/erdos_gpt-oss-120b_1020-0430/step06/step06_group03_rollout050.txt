import time, math, numpy as np

def _init_autocorr(h):
    """autocorrelation C[k] = sum_i h[i] * h[i+k] for k>=0."""
    n = h.size
    # np.correlate uses convolution; the central element (lag 0) is at index n-1
    full = np.correlate(h, h, mode='full')
    return full[n - 1 :].astype(float)


def _swap_update(h, C, i, j):
    """Swap a 1 at i with a 0 at j, updating autocorrelation C in‑place."""
    n = h.shape[0]

    # --- remove contribution of i (which is 1) ---
    C[0] -= 1.0
    if i > 0:
        C[1 : i + 1] -= h[i - 1 :: -1]          # h[i-1], h[i-2], ...
    if i < n - 1:
        C[1 : n - i] -= h[i + 1 :]              # h[i+1], ...
    h[i] = 0.0

    # --- add contribution of j (which becomes 1) ---
    C[0] += 1.0
    if j > 0:
        C[1 : j + 1] += h[j - 1 :: -1]
    if j < n - 1:
        C[1 : n - j] += h[j + 1 :]
    h[j] = 1.0


def _c5_from_corr(C, total_h, n):
    """C5 = (total_h - min_autocorr) * 2/n."""
    return (total_h - C.min()) * (2.0 / n)


def _full_c5(h):
    """Exact C5 using the definition (for the final output)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # -------------------------------------------------
    # parameters
    n = 1000                     # < 1k samples, even
    total_h = n // 2            # exactly n/2 ones
    best_h = None
    best_c = np.inf

    # iteration control
    max_no_improve = 15000      # restart after this many non‑improving steps
    T0 = 0.01                    # initial temperature for simulated annealing
    decay = 0.99995             # temperature decay per iteration

    # -------------------------------------------------
    while time.time() - start < budget_s * 0.97:
        # ---- random binary start ----
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, total_h, replace=False)
        h[ones] = 1.0
        C = _init_autocorr(h)

        cur_c = _c5_from_corr(C, total_h, n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        no_improve = 0
        T = T0

        # ---- local search loop ----
        while (time.time() - start < budget_s * 0.97) and no_improve < max_no_improve:
            # worst (minimal) autocorrelation shift, ignore k=0
            k_min = int(np.argmin(C[1:])) + 1  # shift where C is smallest

            # try a guided swap that creates a new pair at distance k_min
            idx = np.arange(n)
            i_cand = np.where(
                (h == 1) & (idx + k_min < n) & (h[idx + k_min] == 0)
            )[0]
            j_cand = np.where(
                (h == 0) & (idx + k_min < n) & (h[idx + k_min] == 1)
            )[0]

            if i_cand.size and j_cand.size:
                i = int(rng.choice(i_cand))
                j = int(rng.choice(j_cand))
                # ensure i≠j (they are, by construction)
            else:
                # fallback: random 1↔0 swap
                i = int(rng.integers(n))
                while h[i] == 0:
                    i = int(rng.integers(n))
                j = int(rng.integers(n))
                while h[j] == 1:
                    j = int(rng.integers(n))

            # perform swap
            _swap_update(h, C, i, j)
            new_c = _c5_from_corr(C, total_h, n)

            # acceptance (simulated annealing)
            accept = False
            if new_c < cur_c:
                accept = True
            else:
                prob = math.exp(-(new_c - cur_c) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_c = new_c
                no_improve = 0
                T = max(T * decay, 1e-8)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                # revert the swap
                _swap_update(h, C, j, i)
                no_improve += 1
                T = max(T * decay, 1e-8)

        # end of one restart loop; continue with a fresh start

    # -------------------------------------------------
    # ensure we have a solution
    if best_h is None:
        # very unlikely, but fall back to a simple half‑ones pattern
        best_h = np.zeros(n, dtype=float)
        best_h[: total_h] = 1.0
        best_c = _full_c5(best_h)

    # final exact evaluation
    final_c = _full_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
