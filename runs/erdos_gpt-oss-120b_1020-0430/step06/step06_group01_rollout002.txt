def binary_targeted_sa(n, rng, time_budget):
    start = time.time()
    # Start with random binary.
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T0, T_end = 0.2, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # compute worst shift
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        # Determine overlapping indices
        if k >= 0:
            i_vals = np.arange(n - k)
            ipk = i_vals + k
        else:
            i_vals = np.arange(-k, n)
            ipk = i_vals + k
        # Identify A and B sets
        hi = h[i_vals]
        hipk = h[ipk]
        A_mask = (hi == 1) & (hipk == 0)
        B_mask = (hi == 0) & (hipk == 1)
        if A_mask.any() and B_mask.any():
            A_idx = i_vals[A_mask]
            B_idx = i_vals[B_mask]
            i = rng.choice(A_idx)
            j = rng.choice(B_idx)
        else:
            # fallback random swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        # occasional random restart if stagnation?
        # Not needed perhaps.
    return best_h, best_c

def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    # while loop moving diff
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def binary_targeted_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    T0, T_end = 0.2, 1e-5
    max_iters = int(min(200000, time_budget * 3000))
    for it in range(max_iters):
        if time.time() - start >= time_budget:
            break
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        if k >= 0:
            i_vals = np.arange(n - k)
            ipk = i_vals + k
        else:
            i_vals = np.arange(-k, n)
            ipk = i_vals + k
        hi = h[i_vals]
        hipk = h[ipk]
        A_mask = (hi == 1) & (hipk == 0)
        B_mask = (hi == 0) & (hipk == 1)
        if A_mask.any() and B_mask.any():
            i = rng.choice(i_vals[A_mask])
            j = rng.choice(i_vals[B_mask])
        else:
            i = rng.choice(np.where(h == 1)[0])
            j = rng.choice(np.where(h == 0)[0])
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # Determine dimension
    n = 512
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
    else:
        init = None

    # Binary targeted SA (≈40% time)
    bin_time = total * 0.4
    best_h, best_c = binary_targeted_sa(n, rng, bin_time)

    # Continuous refinement with remaining time
    remaining = total - (time.time() - start)
    if remaining > 0:
        h_cont = best_h.astype(float)
        max_iters = int(min(20000, remaining * 1500))
        h_cont, c_cont = guided_hill_climb(h_cont, rng, max_iters=max_iters, step_scale=0.5)
        h_cont = gradient_refine(h_cont, rng, step=0.01, max_iter=5)
        c_cont = compute_c5(h_cont)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # Final tiny tweaks
    while time.time() - start < total * 0.995:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.01
            if eps > 1e-12:
                nh = best_h.copy()
                nh[i] -= eps
                nh[j] += eps
                nc = compute_c5(nh)
                if nc < best_c:
                    best_h, best_c = nh, nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
