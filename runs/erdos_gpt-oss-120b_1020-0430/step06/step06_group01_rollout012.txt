        # inner SA loop
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)

        if c_cont < best_c:

        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))

import time
import numpy as np
import scipy.optimize as opt

# ------------------- basic helpers -------------------


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 20000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    if idx >= 0:  # shift to the right
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # shift to the left
        s = k
        grad[s:] -= h[: n - s]
        grad[: n - s] -= h[s:]
    return grad


# ------------------- binary SA -------------------


def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Simulated annealing on binary vectors with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4

    # pre‑allocate structures for O(1) swaps
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones_arr = rng.choice(n, n // 2, replace=False)
        zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)

        pos_one = np.full(n, -1, dtype=int)
        pos_zero = np.full(n, -1, dtype=int)
        pos_one[ones_arr] = np.arange(n // 2)
        pos_zero[zeros_arr] = np.arange(n // 2)

        h[ones_arr] = 1.0
        c_curr = compute_c5(h)

        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)

            i = ones_arr[rng.integers(n // 2)]
            j = zeros_arr[rng.integers(n // 2)]

            # propose swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                # update bookkeeping
                idx_i = pos_one[i]
                idx_j = pos_zero[j]
                ones_arr[idx_i] = j
                zeros_arr[idx_j] = i
                pos_one[j] = idx_i
                pos_zero[i] = idx_j
                pos_one[i] = -1
                pos_zero[j] = -1

                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


# ------------------- soft‑max objective -------------------


class SoftmaxC5Objective:
    """Callable for L‑BFGS‑B: soft‑max approximation of max overlap + sum penalty."""

    def __init__(self, beta: float, target: float, penalty: float, n: int):
        self.beta = beta
        self.target = target
        self.penalty = penalty
        self.n = n
        self.start_idx = n - 1

    def __call__(self, x: np.ndarray):
        beta = self.beta
        n = self.n
        target = self.target
        penalty = self.penalty
        start_idx = self.start_idx

        corr = np.correlate(x, 1.0 - x, mode="full")
        maxc = np.max(corr)
        exp_beta = np.exp(beta * (corr - maxc))
        sumexp = np.sum(exp_beta)

        f = (maxc + np.log(sumexp)) / beta
        diff = np.sum(x) - target
        f += penalty * diff * diff

        w = exp_beta / sumexp
        w_seq = w
        w_rev = w_seq[::-1]

        conv1 = np.convolve(w_rev, 1.0 - x, mode="full")
        conv2 = np.convolve(w_seq, x, mode="full")
        grad_f = conv1[start_idx:start_idx + n] - conv2[start_idx:start_idx + n]

        grad = grad_f + 2.0 * penalty * diff
        return f, grad

                                    "ftol": 1e-12, "disp": False})

            h = enforce_sum(h, rng)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c


# ------------------- main entry point -------------------


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # start from provided construction if any
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # -------- binary simulated annealing (≈30% of time) --------
    binary_time = total_budget * 0.30
    h_bin, c_bin = binary_sa(512, rng, binary_time)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin, c_bin, 512

    # -------- continuous refinement (remaining time) --------
    remaining = total_budget - (time.time() - start)
    if remaining > 0.0:
        if best_h is not None and best_n == 512:
            init_h = best_h.copy()
        elif best_h is not None:
            # interpolate to size 512
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, 512, endpoint=False)
            init_h = np.interp(x_new, x_old, best_h)
            init_h = np.clip(init_h, 0.0, 1.0)
            init_h = enforce_sum(init_h, rng)
        else:
            init_h = rng.random(512)
            init_h = enforce_sum(init_h, rng)

        h_cont, c_cont = lbfgsb_optimize(init_h, rng, remaining)
        if c_cont < best_c:
            best_h, best_c, best_n = h_cont.copy(), c_cont, 512

    # -------- final tiny stochastic tweaking --------
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # -------- sanitisation --------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
