    idx_max = np.argmax(corr)

import time
import numpy as np
from scipy.optimize import minimize

# ----- globals for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None


def get_corr(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.size
    hp = np.concatenate([h, np.zeros(n)])
    qp = np.concatenate([1.0 - h, np.zeros(n)])
    H = np.fft.fft(hp)
    Q = np.fft.fft(qp)
    corr = np.fft.ifft(H * np.conj(Q)).real
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    max_corr = np.max(get_corr(h))
    return max_corr * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    # fallback uniform scaling
    if abs(diff) > 1e-8:
        h *= target / h.sum()
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = get_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the current worst‑shift overlap."""
    n = h.size
    corr = get_corr(h)
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    k = abs(shift)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Swap two entries or transfer a random amount while preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.2:  # pure swap
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = get_corr(h)
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        n = n0 * 2
        h0 = np.repeat(h0, 2)  # simple up‑sampling
    else:
        n = 1024
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing (≈50% budget) -----
    SA_deadline = start + budget_s * 0.5
    current_h = best_h.copy()
    current_c = best_c
    T = 1.0
    eps_scale = 0.5
    it = 0
    while time.time() < SA_deadline:
        if rng.random() < 0.5:
            nh = guided_perturb(current_h, worst_gradient(current_h), rng,
                                scale=eps_scale)
        else:
            nh = random_move(current_h, rng, scale=eps_scale)
        nc = compute_c5(nh)
        delta = nc - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_h, current_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        it += 1
        if it % 1000 == 0:
            T *= 0.995
            eps_scale *= 0.995
            T = max(T, 1e-6)
            eps_scale = max(eps_scale, 1e-5)

    # ----- gradient polishing -----
    refined = gradient_refine(best_h, rng, step=0.02, max_iter=12)
    rc = compute_c5(refined)
    if rc < best_c:
        best_h, best_c = refined.copy(), rc

    # ----- soft‑max SLSQP passes -----
    beta_seq = (20.0, 200.0, 2000.0, 10000.0, 50000.0)
    for beta in beta_seq:
        if time.time() > start + budget_s * 0.95:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
        best_c = compute_c5(best_h)

    # ----- smoothing & final polishing -----
    if time.time() < start + budget_s * 0.98:
        sigma = 1.0
        w = int(4 * sigma + 0.5)
        xs = np.arange(-w, w + 1)
        kernel = np.exp(-0.5 * (xs / sigma) ** 2)
        kernel /= kernel.sum()
        h_s = np.convolve(best_h, kernel, mode="same")
        h_s = np.clip(h_s, 0.0, 1.0)
        h_s = enforce_sum(h_s, rng)
        h_s = slsqp_optimize(h_s, 5000.0, rng, max_iter=100)
        sc = compute_c5(h_s)
        if sc < best_c:
            best_h, best_c = h_s.copy(), sc

    # ----- tiny hill‑climb for last seconds -----
    while time.time() < start + budget_s * 0.99:
        nh = random_move(best_h, rng, scale=0.001)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
