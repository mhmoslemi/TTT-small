import time
import numpy as np

# ---------- core evaluation ----------
def max_corr_fft(h: np.ndarray) -> float:
    n = h.size
    a = h.astype(np.float64)
    b = (1.0 - h).astype(np.float64)[::-1]
    L = 2 * n
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(b, L)
    c = np.fft.ifft(fa * fb).real
    return np.max(c[:2 * n - 1])

def c5_fft(h: np.ndarray) -> float:
    return max_corr_fft(h) * (2.0 / h.size)

# ---------- gradient for worst shift ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

# ---------- simple sum repair ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def enforce_sum_binary(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n // 2
    diff = int(target - h.sum())
    if diff > 0:
        zero = np.flatnonzero(h == 0)
        choose = rng.choice(zero, diff, replace=False)
        h[choose] = 1.0
    elif diff < 0:
        one = np.flatnonzero(h == 1)
        choose = rng.choice(one, -diff, replace=False)
        h[choose] = 0.0
    return h

# ---------- hill climbing on continuous values ----------
def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int, step_scale: float):
    best_h = h.copy()
    best_c = c5_fft(h)
    n = h.size
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step_scale
        h[i] -= eps
        h[j] += eps
        c = c5_fft(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

    zeros = np.setdiff1d(np.arange(n, dtype=int), ones, assume_unique=True)
    pos_in_zeros[zeros] = np.arange(n // 2)
    ones_arr = ones.copy()
    zeros_arr = zeros.copy()

    c_curr = max_corr_fft(h)
    if c_curr < best_corr:
        best_corr, best_h = c_curr, h.copy()
    T0, T_end = 5.0, 0.001

    while time.time() - start < sa_time:
        elapsed = time.time() - start
        frac = (elapsed) / sa_time
        T = T0 * (T_end / T0) ** frac

        i = ones_arr[rng.integers(len(ones_arr))]
        j = zeros_arr[rng.integers(len(zeros_arr))]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = max_corr_fft(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            c_curr = c_new
            idx_i = pos_in_ones[i]
            idx_j = pos_in_zeros[j]
            ones_arr[idx_i] = j
            zeros_arr[idx_j] = i
            pos_in_ones[i] = -1
            pos_in_zeros[j] = -1
            pos_in_ones[j] = idx_i
            pos_in_zeros[i] = idx_j
            if c_new < best_corr:
                best_corr, best_h = c_new, h.copy()
                if best_corr <= target_corr:
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

    # ----- continuous refinement (guided hill climb) -----
    remaining = total - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        max_iters = int(min(30000, remaining * 800))
        best_h, best_c = guided_hill_climb(best_h.copy(), rng,
                                          max_iters=max_iters, step_scale=0.7)
        best_corr = max_corr_fft(best_h)
        remaining = total - (time.time() - start)

    # ----- final stochastic tweaks -----
    while time.time() - start < total * 0.99 and best_h is not None:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            break
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = c5_fft(nh)
        if nc < c5_fft(best_h):
            best_h = nh
            best_corr = max_corr_fft(best_h)

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
