def gradient_descent_max_overlap(h0, rng, max_iter=200, step0=0.02, step_decay=0.5):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for i in range(max_iter):
        # compute worst shift gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        step = step0 * (step_decay ** i)
        h_new = h - step * grad
        # clip and enforce sum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target = 0.3808
    best_h = None
    best_c = np.inf
    best_n = None

    # initial if supplied
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size
        if best_c <= target:
            return best_h.tolist(), float(best_c), int(best_n)

    # Phase 0: many binary starts on 256 points
    remaining = budget_s - (time.time() - start)
    t_coarse = min(remaining * 0.45, 300.0)
    n_coarse = 256
    while time.time() - start < start + t_coarse:
        h = np.zeros(n_coarse, dtype=float)
        ones = rng.choice(n_coarse, n_coarse // 2, replace=False)
        h[ones] = 1.0
        h_opt, c_opt = gradient_descent_max_overlap(h, rng, max_iter=200, step0=0.03)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, n_coarse
        if best_c <= target:
            break

    # Phase 1: upsample to 512 if needed
    remaining = budget_s - (time.time() - start)
    if best_h is not None and best_n < 512 and remaining > 5:
        h_mid = upsample_h(best_h, 512)
        h_mid = np.clip(h_mid, 0.0, 1.0)
        h_mid = enforce_sum(h_mid, rng)
        c_mid = compute_c5(h_mid)
        if c_mid < best_c:
            best_h, best_c, best_n = h_mid, c_mid, 512

    # Phase 2: descent on 512
    remaining = budget_s - (time.time() - start)
    if best_h is not None and best_n == 512 and remaining > 5:
        t_mid = min(remaining * 0.4, 200.0)
        h_opt, c_opt = gradient_descent_max_overlap(best_h, rng, max_iter=500, step0=0.02)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        while time.time() - start < start + t_mid:
            h = best_h.copy()
            eps = 0.01
            idxs = rng.choice(512, size=5, replace=False)
            for idx in idxs:
                delta = (rng.random() - 0.5) * eps
                h[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)
            h_opt, c_opt = gradient_descent_max_overlap(h, rng, max_iter=200, step0=0.02)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
            if best_c <= target:
                break

    # Phase 3: upsample to 1024
    remaining = budget_s - (time.time() - start)
    if best_h is not None and best_n < 1024 and remaining > 5:
        h_fine = upsample_h(best_h, 1024)
        h_fine = np.clip(h_fine, 0.0, 1.0)
        h_fine = enforce_sum(h_fine, rng)
        c_fine = compute_c5(h_fine)
        if c_fine < best_c:
            best_h, best_c, best_n = h_fine, c_fine, 1024

    # Phase 4: refinement on 1024
    remaining = budget_s - (time.time() - start)
    if best_h is not None and best_n == 1024 and remaining > 5:
        max_iter = min(800, int(remaining // 0.001))
        h_opt, c_opt = gradient_descent_max_overlap(best_h, rng, max_iter=max_iter, step0=0.015)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        remaining = budget_s - (time.time() - start)
        if remaining > 5 and best_c > target:
            h_sa, c_sa = continuous_sa(best_h, rng, remaining)
            if c_sa < best_c:
                best_h, best_c = h_sa, c_sa

    # final sanitisation
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5, dtype=float)
        best_c = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c), int(best_n)
