def hill_climb_targeted(h, rng, time_limit):
    start = time.time()
    best_h = h.copy()

    while time.time() - start < time_limit:
        corr = compute_corr(h)
        max_corr = np.max(corr)
        # early exit if no better feasible moves
        order = np.argsort(-corr)  # descending
        moved = False
        for idx in order:
            shift = idx - (len(h)-1)
            # define i_range,j_range based on shift
            if shift >= 0:
                i_range = np.arange(shift, len(h))
                j_range = i_range - shift
            else:
                i_range = np.arange(0, len(h) + shift)
                j_range = i_range - shift
            eps_max = np.minimum(h[i_range], 1.0 - h[j_range])
            mask = eps_max > 1e-12
            if not np.any(mask):
                continue

            max_b_idx = np.argmax(masked_benefit)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def hill_climb_targeted(h0: np.ndarray, rng: np.random.Generator, time_limit: float):
    """Deterministic hill‑climb that transfers mass from i to j to reduce the worst shift."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # pre‑compute index pairs for every possible shift
    shift_dict = {}
    for shift in range(-(n - 1), n):
        if shift >= 0:
            i_arr = np.arange(shift, n, dtype=int)
        else:
            i_arr = np.arange(0, n + shift, dtype=int)
        j_arr = i_arr - shift
        shift_dict[shift] = (i_arr, j_arr)

    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        if max_corr * (2.0 / n) < best_c5:
            best_c5 = max_corr * (2.0 / n)
            best_h = h.copy()

        # shifts ordered by descending overlap
        order = np.argsort(-corr)
        moved = False
        for idx in order:
            shift = idx - (n - 1)
            i_arr, j_arr = shift_dict[shift]

            eps_max = np.minimum(h[i_arr], 1.0 - h[j_arr])
            mask = eps_max > 1e-12
            if not np.any(mask):
                continue

            benefit = (1.0 - h[j_arr]) + h[i_arr]
            b_vals = benefit[mask]
            i_sel = i_arr[mask][np.argmax(b_vals)]
            j_sel = j_arr[mask][np.argmax(b_vals)]
            max_eps = eps_max[mask][np.argmax(b_vals)]

            # try decreasing fractions of the maximal transferable mass
            for factor in (1.0, 0.5, 0.25, 0.125):
                eps = max_eps * factor
                if eps <= 0:
                    continue
                h[i_sel] -= eps
                h[j_sel] += eps
                new_corr = np.correlate(h, 1.0 - h, mode="full")
                new_max = np.max(new_corr)
                if new_max < max_corr - 1e-12:
                    moved = True
                    max_corr = new_max
                    if new_max * (2.0 / n) < best_c5:
                        best_c5 = new_max * (2.0 / n)
                        best_h = h.copy()
                    break
                # revert
                h[i_sel] += eps
                h[j_sel] -= eps
            if moved:
                break

        if not moved:
            # random transfer to escape stagnation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = max_eps * rng.random() * 0.3
                    h[i] -= eps
                    h[j] += eps
    return best_h, best_c5
