def compute_c5_fft(h):
    n = h.size
    L = 1 << (2*n - 1).bit_length()
    # However we can pad to next power of two for speed: choose L = 2**int(np.ceil(np.log2(2*n)))
    L = 1 << ((2*n - 1).bit_length())
    # but we can use np.fft.rfft which autocomputes length? We'll just use np.fft.ifft with length L.
    h_pad = np.zeros(L, dtype=np.float64)
    one_minus_h_pad = np.zeros(L, dtype=np.float64)
    h_pad[:n] = h
    one_minus_h_pad[:n] = 1.0 - h
    # compute convolution (full) via FFT
    H = np.fft.rfft(h_pad)
    O = np.fft.rfft(one_minus_h_pad)
    conv = np.fft.irfft(H * O, n=L)
    # conv length L, we need first 2n-1 entries corresponding to full correlation.
    corr = conv[: 2*n-1]
    # Compute max * (2/n)
    return float(np.max(corr) * (2.0 / n))

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    # compute softmax weights w = exp(beta*v) / sum(exp(beta*v))
    # To avoid overflow, subtract max
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    # shift index: idx - (n-1) = k
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # h[:n-k] interacts with (1-h)[k:]
            g[: n - k] += wi * one_minus_h[k:]
            g[k :]    -= wi * h[: n - k]
        else:
            kk = -k
            # h[kk:] interacts with (1-h)[:n-kk]
            g[kk :]   += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk :]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# -----------------------------------------------------------
# basic utilities
# -----------------------------------------------------------
def adjust_sum_exact(x, rng, max_iter=2000):
    """Modify x in-place so that sum(x) == len(x)/2 exactly."""
    n = x.size
    target = n / 2.0
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x


def compute_c5(h):
    """Exact C5 for a (possibly fractional) vector."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


# -----------------------------------------------------------
# smooth‑max gradient utilities
# -----------------------------------------------------------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)log∑exp(β·C_k) where C_k are overlaps."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k :] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk :] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk :]
    return g


def worst_shift_gradient(h):
    """Sub‑gradient using the currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :] -= h[: n - k]
        else:
            kk = -k
            g[kk :] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g


def worst_shift_refine(h, rng, base_step, iters, deadline):
    """Projected sub‑gradient steps focusing on the worst shift."""
    n = h.size
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        cur_c = compute_c5(h)
        best_h = h
        best_c = cur_c
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.95
    return h

# SLSQP polish on smooth‑max surrogate
# -----------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP on the smooth‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# -----------------------------------------------------------
# fast FFT‑based C5 evaluation (used in binary search)
# -----------------------------------------------------------
def compute_corr_fft_array(h):
    """Return the full linear correlation array of h with (1‑h) using FFT."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    h_pad = np.zeros(L, dtype=np.float64)
    oh_pad = np.zeros(L, dtype=np.float64)
    h_pad[:n] = h
    oh_pad[:n] = 1.0 - h
    H = np.fft.rfft(h_pad)
    O = np.fft.rfft(oh_pad)
    conv = np.fft.irfft(H * O, n=L)
    return conv[: 2 * n - 1]


def compute_c5_fft(h):
    n = h.size
    corr = compute_corr_fft_array(h)
    return float(np.max(corr) * (2.0 / n))

# binary targeted simulated‑annealing search
# -----------------------------------------------------------
def binary_targeted_search(h_bin, rng, deadline):
    """SA on binary vectors, using fast FFT evaluation and
    occasional moves aimed at the current worst shift."""
    n = h_bin.size
    cur_c = compute_c5_fft(h_bin)
    best_h = h_bin.copy()
    best_c = cur_c
    it = 0
    T0 = cur_c * 0.02 + 1e-12
    decay = 0.9995
    while time.time() < deadline:
        T = T0 * (decay ** it) + 1e-12
        v = compute_corr_fft_array(h_bin)
        v_max = v.max()
        # decide move type
        if rng.random() < 0.7:  # targeted
            max_idxs = np.where(v == v_max)[0]
            idx = rng.choice(max_idxs)
            k = idx - (n - 1)
            if k >= 0:
                i_range = np.arange(0, n - k)
                i_mask = (h_bin[i_range] == 1) & (h_bin[i_range + k] == 0)
                i_cands = i_range[i_mask]
                j_range = np.arange(0, n)
                j_mask = (h_bin == 0) & (
                    ((j_range + k) >= n) | (h_bin[j_range + k] == 1)
                )
                j_cands = j_range[j_mask]
            else:
                k_abs = -k
                i_range = np.arange(k_abs, n)
                i_mask = (h_bin[i_range] == 1) & (h_bin[i_range - k_abs] == 0)
                i_cands = i_range[i_mask]
                j_range = np.arange(0, n)
                j_mask = (h_bin == 0) & (
                    ((j_range + k) < 0) | (h_bin[j_range + k] == 1)
                )
                j_cands = j_range[j_mask]
            if i_cands.size and j_cands.size:
                i = rng.choice(i_cands)
                j = rng.choice(j_cands)
            else:
                ones = np.flatnonzero(h_bin == 1)
                zeros = np.flatnonzero(h_bin == 0)
                i = rng.choice(ones)
                j = rng.choice(zeros)
        else:  # random swap
            ones = np.flatnonzero(h_bin == 1)
            zeros = np.flatnonzero(h_bin == 0)
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # apply swap
        h_bin[i] = 0
        h_bin[j] = 1
        c = compute_c5_fft(h_bin)

        # acceptance
        if c <= cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c - 1e-12:
                best_c = c
                best_h = h_bin.copy()
        else:
            # revert
            h_bin[i] = 1
            h_bin[j] = 0

        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    base_n = 720
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
        parent_interp = np.interp(xt, xs, parent)
    else:
        parent_interp = None

    restarts = 3
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        # initial vector
        if r == 0 and parent_interp is not None:
            h = parent_interp.copy()
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # ----- smooth‑max gradient descent -----
        betas = [5, 20, 80, 200, 500, 2000, 5000, 20000, 50000, 100000]
        it_per_beta = 12
        base_step = 0.25
        for beta in betas:
            for _ in range(it_per_beta):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h, beta)
                if np.linalg.norm(g) < 1e-14:
                    continue
                cur_c = compute_c5(h)
                best_local_h = h

                for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                    step = base_step * mult

                h = best_local_h

        adjust_sum_exact(h_bin, rng)  # enforce exact half‑ones

        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)

        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
