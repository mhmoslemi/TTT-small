def linear_conv(a, b):
    n = a.shape[0]
    m = b.shape[0]

def linear_conv(a, b):
    n = a.shape[0]
    m = b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    return conv[: n + m - 1]

def linear_corr(a, b):
    # correlation a ⋆ b (linear) for lags 0..len(b)-1 (assuming len(a)=len(b))
    # using conv of reversed a with b
    a_rev = a[::-1]
    conv = linear_conv(a_rev, b)
    start = a.shape[0] - 1
    return conv[start:start + b.shape[0]]

    if k != 0:
        grad[: n - k] -= h[k:]

    return grad, k

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto {0≤x≤1, sum x = target}
def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)


def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


# --------------------------------------------------------------
# Linear convolution / correlation via FFT (real input)
def linear_conv(a, b):
    n, m = a.shape[0], b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    return conv.real[: n + m - 1]


def linear_corr(a, b):
    a_rev = a[::-1]
    conv = linear_conv(a_rev, b)
    start = a.shape[0] - 1
    return conv[start:start + b.shape[0]]


# --------------------------------------------------------------
# C5 evaluation (fast, exact for the discretisation)
def compute_c5_fft(h):
    n = h.shape[0]
    conv = linear_corr(h, h)          # h * h, length n
    min_conv = conv.min()
    return 1.0 - (2.0 / n) * min_conv


# --------------------------------------------------------------
# Smoothed max objective (soft‑max) and its gradient
def smoothed_obj_and_grad(h, beta):
    n = h.shape[0]
    conv_hh = linear_corr(h, h)               # h⋆h
    min_conv = conv_hh.min()
    O = 1.0 - conv_hh                         # O_k values
    max_beta_O = np.max(beta * O)
    exp_term = np.exp(beta * O - max_beta_O)
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                    # soft‑max weights

    w_corr_h = linear_corr(w, h)              # w⋆h
    w_conv_h = linear_conv(w, h)[:n]          # first n entries of w*h
    grad = - (w_corr_h + w_conv_h)            # ∇ of smoothed max

    f = (max_beta_O + np.log(sum_exp)) / beta
    c5 = 1.0 - (2.0 / n) * min_conv
    return f, grad, c5


# --------------------------------------------------------------
# Gradient of the worst shift (used for guided hill‑climb)
def worst_gradient_fft(h):
    n = h.shape[0]
    conv = linear_corr(h, h)
    k = int(np.argmin(conv))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

# Simple hill‑climb that moves mass from a high‑gradient index to a low‑gradient one
def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.7):
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient_fft(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5_fft(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:

    return best_h, best_c
