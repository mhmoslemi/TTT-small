def worst_shift_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g
```

def max_overlap_and_shifts(h):
    n = h.size
    L = 2*n-1
    L_pow = 1 << (L-1).bit_length()
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, L_pow)
    R = np.fft.rfft(rev, L_pow)

    max_val = conv.max()
    idxs = np.where(conv == max_val)[0]
    # shift indices: k = idx - (n-1)

    target = n / 2.0
    max_val, ks = max_overlap_and_shifts(h)
    cur_c = max_val * (2.0 / n)
    step = 0.5
    it = 0
    while time.time() < time_end:
        # compute gradient for worst shifts
        ks = max_overlap_and_shifts(h)[1] # we need ks each iteration
        # get gradient
        g = np.zeros_like(h)

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97
    best_h = None
    best_c = np.inf
    best_n = None

    # initial parent or random base
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        base_n = len(parent)
        h_base = np.array(parent, dtype=np.float64)
        # project to ensure sum
        h_base = project_box_sum(h_base, base_n/2.0)
    else:

def max_overlap_and_shifts(h):
    n = h.size
    L = 2 * n - 1
    L_pow = 1 << (L - 1).bit_length()  # next power of 2
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, L_pow)
    R = np.fft.rfft(rev, L_pow)
    conv = np.fft.irfft(H * R, L_pow)[:L]
    max_val = conv.max()
    idxs = np.where(conv == max_val)[0]
    ks = idxs - (n - 1)
    return max_val, ks

    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ---------- correlation (max overlap) ----------
def max_overlap_and_shifts(h):
    """Return (max_corr, list_of_shifts) where max_corr is the raw
    correlation sum (without dx) and shifts are integer k for which it attains max."""
    n = h.size
    L = 2 * n - 1
    L_pow = 1 << (L - 1).bit_length()  # next power of two
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, L_pow)
    R = np.fft.rfft(rev, L_pow)
    conv = np.fft.irfft(H * R, L_pow)[:L]
    max_val = conv.max()
    idxs = np.where(conv == max_val)[0]
    ks = idxs - (n - 1)
    return max_val, ks


def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    max_val, _ = max_overlap_and_shifts(h)
    return max_val * (2.0 / h.size)


# ---------- subgradient descent on worst shift ----------
def run_descent(h, time_end, rng):
    n = h.size
    target = n / 2.0
    max_val, ks = max_overlap_and_shifts(h)
    cur_c = max_val * (2.0 / n)
    step = 0.5
    it = 0
    while time.time() < time_end:
        # compute subgradient for the active worst shifts
        g = np.zeros_like(h)
        for k in ks:
            if k >= 0:
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
            else:
                kk = -k
                g[kk:] += 1.0 - h[: n - kk]
                g[: n - kk] -= h[kk:]
        g /= max(1, len(ks))
        if np.linalg.norm(g) < 1e-14:
            break

        improved = False
        for _ in range(12):  # backtracking line search
            v = h - step * g
            v = project_box_sum(v, target)
            mval, ks_new = max_overlap_and_shifts(v)
            c = mval * (2.0 / n)
            if c < cur_c - 1e-12:
                h = v
                cur_c = c
                ks = ks_new
                improved = True
                step *= 1.1
                break
            step *= 0.5
        if not improved:
            step *= 0.5

        # occasional random jitter
        if it % 200 == 0:
            v = h + rng.normal(scale=0.02, size=n)
            v = project_box_sum(v, target)
            mval, ks_new = max_overlap_and_shifts(v)
            c = mval * (2.0 / n)
            if c < cur_c:
                h = v
                cur_c = c
                ks = ks_new

        it += 1
        if step < 1e-6:
            break
    return h, cur_c


# ---------- binary hill‑climb (simulated annealing) ----------
def binary_make_feasible(h_bin, rng):
    """Adjust a {0,1} vector to have exactly n/2 ones."""
    n = h_bin.size
    target = n // 2
    cur = int(h_bin.sum())
    while cur < target:
        zeros = np.flatnonzero(h_bin == 0)
        if zeros.size == 0:
            break
        i = rng.choice(zeros)
        h_bin[i] = 1.0
        cur += 1
    while cur > target:
        ones = np.flatnonzero(h_bin == 1)
        if ones.size == 0:
            break
        i = rng.choice(ones)
        h_bin[i] = 0.0
        cur -= 1
    return h_bin


def binary_sa_hill_climb(h_bin, rng, time_end):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    it = 0
    while time.time() < time_end and it < 200000:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        if c < cur_c:
            cur_c = c
            ones[i_idx] = j
            zeros[j_idx] = i
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
    return best_h, best_c


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # ----- initial base vector -----
    parent = globals().get('initial_h_values')
    if parent is not None:
        base_n = len(parent)
        h_base = np.array(parent, dtype=np.float64)
        h_base = project_box_sum(h_base, base_n / 2.0)
    else:
        base_n = 720
        h_base = rng.random(base_n)
        h_base = project_box_sum(h_base, base_n / 2.0)

    # ----- first descent on parent (if any) -----
    if parent is not None:
        now = time.time()
        if now < deadline:
            slice_end = now + min(30.0, (deadline - now) * 0.2)
            h_cur, c_cur = run_descent(h_base.copy(), slice_end, rng)
            if c_cur < best_c:
                best_c, best_h, best_n = c_cur, h_cur.copy(), base_n

    # ----- stochastic restarts -----
    while time.time() < deadline:
        now = time.time()
        remaining = deadline - now
        if remaining < 5.0:
            break
        slice_len = min(25.0, remaining * 0.3)
        slice_end = now + slice_len

        n = base_n
        h0 = rng.random(n)
        h0 = project_box_sum(h0, n / 2.0)

        h_cur, c_cur = run_descent(h0, slice_end, rng)
        if c_cur < best_c:
            best_c, best_h, best_n = c_cur, h_cur.copy(), n

    # ----- up‑sampling attempts -----
    while time.time() < deadline and best_h is not None and best_h.size * 2 <= 1440:
        now = time.time()
        remaining = deadline - now
        if remaining < 8.0:
            break
        slice_len = min(20.0, remaining * 0.5)
        slice_end = now + slice_len

        h_up = np.repeat(best_h, 2)
        h_up = project_box_sum(h_up, h_up.size / 2.0)

        h_cur, c_cur = run_descent(h_up, slice_end, rng)
        if c_cur < best_c:
            best_c, best_h, best_n = c_cur, h_cur.copy(), h_up.size
        else:
            break  # stop if no improvement

    # ----- binary refinement if time permits -----
    if best_h is not None and time.time() < deadline - 5.0:
        sub_end = time.time() + (deadline - time.time()) * 0.5
        h_bin = (best_h >= 0.5).astype(np.float64)
        h_bin = binary_make_feasible(h_bin, rng)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_end)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), best_h.size

    # ----- fallback in the unlikely event of no solution -----
    if best_h is None:
        n = 200
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_box_sum(best_h, n / 2.0)
        best_c = compute_c5(best_h)
        best_n = n

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
