def worst_shift_gradient(h):

    max_val = v.max()
    idxs = np.where(v == max_val)[0]  # potentially multiple

    # For each shift index
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            # slice 0:n-k for i
            g[: n - k] += one_minus_h[k:]
            g[k :] -= h[: n - k]
        else:
            kk = -k
            g[kk :] += one_minus_h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

import time
import numpy as np

# ---------- utilities ----------
def compute_c5_fft(h):
    """Maximum overlap integral using FFT (full linear correlation)."""
    n = h.size
    L = 2 * n - 1
    # real FFTs are faster
    FA = np.fft.rfft(h, n=L)
    FB = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(FA * np.conj(FB), n=L)
    return float(corr.max() * (2.0 / n))


def overlap_vector_fft(h):
    """Full correlation vector (length 2n‑1)."""
    n = h.size
    L = 2 * n - 1
    FA = np.fft.rfft(h, n=L)
    FB = np.fft.rfft(1.0 - h, n=L)
    return np.fft.irfft(FA * np.conj(FB), n=L)


def adjust_sum_exact(h, rng, max_iter=2000):
    """Shift mass among entries so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_shift_gradient(h):
    """Sub‑gradient for the current worst shift(s)."""
    n = h.size
    v = overlap_vector_fft(h)
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += one_minus_h[k:]
            g[k :] -= h[: n - k]
        else:
            kk = -k
            g[kk :] += one_minus_h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g


def projected_gradient_descent(h, base_step, iters, rng, deadline):
    """Projected sub‑gradient descent using the worst shift gradient."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    step = base_step
    for it in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        # scale step by norm to keep updates moderate
        cur_step = step / max(1.0, norm)
        cand_best = h
        cand_c = best_c
        for mult in (1.0, 0.5, 0.25, 2.0):
            step_sz = cur_step * mult
            cand = h - step_sz * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5_fft(cand)
            if c < cand_c:
                cand_c = c
                cand_best = cand
        h = cand_best
        if cand_c < best_c:
            best_c = cand_c
            best_h = h.copy()
        step *= 0.99
    return best_h, best_c


def binary_swap_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill‑climb on a binary vector."""
    n = h_bin.size
    cur_c = compute_c5_fft(h_bin)
    best_c = cur_c
    best_h = h_bin.copy()
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))
    T0 = max(1e-6, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        it += 1
        T = T0 * (0.9995 ** it)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = compute_c5_fft(h_bin)
        accept = False
        if c_new < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c_new - cur_c) / T):
                accept = True
        if accept:
            cur_c = c_new
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
                T0 = max(1e-6, best_c * 0.02)
                it = 0
        else:
            # revert
            h_bin[i] = 1
            h_bin[j] = 0
        if it > 20000:
            break
    return best_h, best_c
