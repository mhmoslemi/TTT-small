def softmax_weights(O, beta):
    bO = beta * O
    m = bO.max()
    expb = np.exp(bO - m)
    w = expb / expb.sum()

def softmax_grad(h, O, beta):
    w = softmax_weights(O, beta)
    # compute shiftplus and shiftminus
    n = h.size
    # compute conv with w and w_rev using FFT
    L = 1 << ((2*n-1).bit_length())

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

# ---------- gradient for worst shift ----------
def worst_gradient(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

# ---------- projected sub‑gradient ----------
def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

# ---------- soft‑max gradient ----------
def softmax_weights(O: np.ndarray, beta: float) -> np.ndarray:
    bO = beta * O
    m = bO.max()
    expb = np.exp(bO - m)
    return expb / expb.sum()

def softmax_grad(h: np.ndarray, O: np.ndarray, beta: float) -> np.ndarray:
    w = softmax_weights(O, beta)
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    h_fft = np.fft.rfft(h, L)
    w_fft = np.fft.rfft(w, L)
    conv_full = np.fft.irfft(h_fft * w_fft, L)
    shiftminus = conv_full[:n]
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, L)
    conv_rev = np.fft.irfft(h_fft * w_rev_fft, L)
    shiftplus = conv_rev[n - 1:n - 1 + n]
    return 1.0 - shiftplus - shiftminus

# ---------- Adam optimiser on soft‑max ----------
def softmax_adam(h0: np.ndarray, beta_schedule, iters_per_beta,
                 target_sum, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() - start_time > time_limit * 0.99:
                return best_h, best_c
            t += 1
            O = np.correlate(h, 1 - h, mode="full")[h.size - 1 :]
            grad = softmax_grad(h, O, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.02 * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_box_sum(h_new, target_sum)
            h = h_new
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c
