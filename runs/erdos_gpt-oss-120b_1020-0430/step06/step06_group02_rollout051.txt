    n = h.size
    weight = n // 2
    # compute correlation using np.dot for each shift (vectorized)
    # We'll use FFT for speed: A = np.fft.ifft(np.fft.fft(h) * np.fft.fft(h)).real.round().astype(int)

def hill_climb(h, C, weight, max_iters, rng):
    n = h.size
    # maintain best C_max = C.max()
    C_max = C.max()
    # while improvement possible
    for it in range(max_iters):
        k_max = int(np.argmax(C))  # shift with max C
        # gather O_set:
        # O_set: indices i where h[i] = 1 and h[(i + k_max) % n] = 0
        # This can be computed vectorized:
        h_i = h
        h_ipk = np.roll(h, -k_max) # h[i + k] using roll negative shift.
        O_mask = (h_i == 1) & (h_ipk == 0)
        # similarly O_comp: indices j where h[j] = 0 and h[j + k_max] = 1
        O_comp_mask = (h_i == 0) & (h_ipk == 1)
        O_idxs = np.flatnonzero(O_mask)
        O_comp_idxs = np.flatnonzero(O_comp_mask)
        if O_idxs.size == 0 or O_comp_idxs.size == 0:
            # cannot improve? break
            break
        # sample several candidate swaps (maybe 50)
        n_trials = min(100, O_idxs.size * O_comp_idxs.size) # but capped
        best_delta_max = C_max
        best_i = None
        best_j = None
        for _ in range(n_trials):
            i = rng.choice(O_idxs)
            j = rng.choice(O_comp_idxs)
            # ensure i != j (they are different by mask)
            # Compute delta array:
            # delta = (h[(j + k) % n] + h[(j - k) % n]) - (h[(i + k) % n] + h[(i - k) % n])
            # We'll precompute arrays for i and j using np.arange.
            # However, we can compute new max quickly using h and k's vect.
            # Build delta via vectorized:
            ks = np.arange(n)
            delta = (h[(j + ks) % n] + h[(j - ks) % n]) - (h[(i + ks) % n] + h[(i - ks) % n])
            # new_C = C + delta
            new_max = np.max(C + delta)
            if new_max < best_delta_max:
                best_delta_max = new_max
                best_i = i
                best_j = j
                best_delta = delta
                # Early exit if new_max = C_max - 1 maybe; but continue to find best
                if best_delta_max <= C_max - 1:
                    # good enough
                    pass
        if best_i is not None and best_delta_max < C_max:
            # Accept swap
            # Update h
            h[best_i] = 0
            h[best_j] = 1
            # Update C: C = C + best_delta
            C = C + best_delta
            C_max = best_delta_max
        else:
            # no improving swap found; perhaps random restart
            break
    return h, C

import time
import math
import numpy as np

def compute_mismatch_array(h: np.ndarray) -> np.ndarray:
    """C[k] = # of ones mismatched with shift k (cyclic)."""
    n = h.size
    weight = n // 2
    fft_h = np.fft.fft(h)
    autoc = np.fft.ifft(fft_h * np.conj(fft_h)).real
    autoc = np.rint(autoc).astype(np.int64)          # A[k] = Σ h[i]h[i+k]
    C = weight - autoc                               # O_k = weight - A[k]
    C[0] = 0                                          # zero shift always 0
    return C


def greedy_hill_climb(
    h: np.ndarray,
    C: np.ndarray,
    ks: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, np.ndarray]:
    """Repeatedly improve the worst shift using swaps from its O_k set."""
    n = h.size
    while time.time() < deadline:
        k_max = int(np.argmax(C))
        h_shift = np.roll(h, -k_max)
        O_mask = (h == 1) & (h_shift == 0)          # positions contributing to O_kmax
        O_comp = (h == 0) & (h_shift == 1)          # complementary positions
        O_idx = np.flatnonzero(O_mask)
        C_idx = np.flatnonzero(O_comp)
        if O_idx.size == 0:
            break
        improved = False
        for _ in range(30):
            i = rng.choice(O_idx)
            j = rng.choice(C_idx)
            delta = (h[(i + ks) % n] + h[(i - ks) % n]) - (h[(j + ks) % n] + h[(j - ks) % n])
            delta[0] = 0                               # keep shift 0 unchanged
            new_max = np.max(C + delta)
            if new_max < C.max():
                h[i] = 0
                h[j] = 1
                C = C + delta
                improved = True
                break
        if not improved:
            break
    return h, C


def annealing_optimize(
    h_start: np.ndarray,
    C_start: np.ndarray,
    ks: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, np.ndarray]:
    """Simulated annealing that focuses on the current worst shift."""
    n = h_start.size
    h = h_start.copy()
    C = C_start.copy()
    cur_max = C.max()
    best_h = h.copy()
    best_max = cur_max
    # temperature schedule
    T = cur_max * 0.5 + 1.0
    decay = 0.9995
    while time.time() < deadline:
        k_max = int(np.argmax(C))
        h_shift = np.roll(h, -k_max)
        O_mask = (h == 1) & (h_shift == 0)
        O_comp = (h == 0) & (h_shift == 1)
        if O_mask.any():
            i = rng.choice(np.flatnonzero(O_mask))
            j = rng.choice(np.flatnonzero(O_comp))
        else:
            # fallback to any random 1/0 pair
            i = rng.integers(n)
            if h[i] == 0:
                continue
            j = rng.integers(n)
            if h[j] == 1 or j == i:
                continue
        delta = (h[(i + ks) % n] + h[(i - ks) % n]) - (h[(j + ks) % n] + h[(j - ks) % n])
        delta[0] = 0
        new_max = np.max(C + delta)
        dE = new_max - cur_max
        if dE < 0 or rng.random() < math.exp(-dE / T):
            h[i] = 0
            h[j] = 1
            C = C + delta
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        T *= decay
        if T < 1e-3:
            T = cur_max * 0.5 + 1.0
    return best_h, compute_mismatch_array(best_h)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose a reasonably large grid (1024) – more freedom than 512
    n = 1024
    target = n // 2

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.uint8)
        if h0.size != n:
            if n % h0.size == 0:
                factor = n // h0.size
                h0 = np.repeat(h0, factor)
            else:
                extra = n - h0.size
                h0 = np.concatenate([h0, rng.choice([0, 1], size=extra, replace=True)])
    else:
        h0 = np.zeros(n, dtype=np.uint8)
        ones = rng.choice(n, target, replace=False)
        h0[ones] = 1

    # enforce exact sum
    cur = int(h0.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.flatnonzero(h0 == 0)
            flip = rng.choice(zeros, diff, replace=False)
            h0[flip] = 1
        else:
            ones = np.flatnonzero(h0 == 1)
            flip = rng.choice(ones, -diff, replace=False)
            h0[flip] = 0

    ks = np.arange(n, dtype=np.int64)

    C0 = compute_mismatch_array(h0)
    best_h = h0.copy()
    best_C = C0.max()
    best_C5 = (2.0 / n) * best_C

    # ----- quick random restarts (≈10% of budget) ------------------------
    restart_deadline = start + 0.10 * budget_s

        h_rand = np.zeros(n, dtype=np.uint8)

        h_rand[ones] = 1
        C_rand = compute_mismatch_array(h_rand)
        per_deadline = time.time() + 0.015 * budget_s

        if C5_rand < best_C5:

    # ----- annealing phase (≈80% of budget) -------------------------------
    anneal_deadline = start + 0.80 * budget_s
    h_opt, C_opt = annealing_optimize(best_h, compute_mismatch_array(best_h), ks, anneal_deadline, rng)
    C5_opt = (2.0 / n) * C_opt.max()
    if C5_opt < best_C5:
        best_h, best_C, best_C5 = h_opt.copy(), C_opt.max(), C5_opt

    # ----- final polishing (remaining time) -------------------------------
    final_deadline = start + 0.98 * budget_s
    best_h, C_final = greedy_hill_climb(best_h, compute_mismatch_array(best_h), ks, final_deadline, rng)
    best_C = C_final.max()
    best_C5 = (2.0 / n) * best_C

    # ----- sanity enforcement ---------------------------------------------
    best_h = np.clip(best_h, 0, 1).astype(np.uint8)
    cur = int(best_h.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.flatnonzero(best_h == 0)

        else:
            ones = np.flatnonzero(best_h == 1)
            flip = rng.choice(ones, -diff, replace=False)
            best_h[flip] = 0

    # recompute final C5 (protect against any rounding issues)
    final_C5 = (2.0 / n) * compute_mismatch_array(best_h).max()
    return best_h.tolist(), float(final_C5), int(n)
