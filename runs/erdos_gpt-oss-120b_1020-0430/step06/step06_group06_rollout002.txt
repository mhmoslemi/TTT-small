def repair_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12: return h
    # if diff >0: need to increase some values
    if diff > 0:
        # indices where h<1
        idx = np.where(h < 1 - 1e-12)[0]
        for i in idx:
            add = min(diff, 1 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12: break
    else: # diff <0
        idx = np.where(h > 1e-12)[0]
        for i in idx:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-12: break
    return h

def compute_c5(h): ... (same)

def enforce_sum(h, rng): ... (same as previous)

def sum_eq_constraint(x): return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # final exact-sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize
    rng = np.random.default_rng(seed)
    start = time.time()
    # init
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n = 512
        h0 = rng.random(n)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # main restart loop
    sigma = 0.05
    beta_seq = (50.0, 500.0, 5000.0, 20000.0, 50000.0, 100000.0)
    max_iters = (300, 200, 150, 100, 80, 60)
    # allocate approx 80% time for restarts
    restart_deadline = start + budget_s * 0.80
    while time.time() < restart_deadline:
        # generate perturbed start
        h_start = best_h + rng.normal(0.0, sigma, size=n)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = enforce_sum(h_start, rng)
        # run optimizations across beta_seq
        for beta, iters in zip(beta_seq, max_iters):
            if time.time() > restart_deadline:
                break
            h_start = slsqp_optimize(h_start, beta, rng, max_iter=iters)
        c_new = compute_c5(h_start)
        if c_new < best_c:
            best_h, best_c = h_start, c_new
        # decay sigma
        sigma = max(0.01, sigma * 0.995)

    # final repair and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for SLSQP constraints
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k corr_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """SLSQP refinement for a given smoothing parameter β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n = 512
        h0 = rng.random(n)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)

    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- initial SLSQP polishing -----
    beta_seq = (50.0, 500.0, 5000.0, 20000.0, 50000.0, 100000.0)
    max_iters = (300, 200, 150, 100, 80, 60)
    for beta, iters in zip(beta_seq, max_iters):
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=iters)
    best_c = compute_c5(best_h)

    # ----- stochastic restarts with SLSQP -----
    sigma = 0.05
    restart_deadline = start + budget_s * 0.80
    restart_cnt = 0
    max_restarts = 200

    while time.time() < restart_deadline and restart_cnt < max_restarts:
        # perturb current best solution
        h_start = best_h + rng.normal(0.0, sigma, size=n)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = enforce_sum(h_start, rng)

        # cascade of SLSQP refinements
        for beta, iters in zip(beta_seq, max_iters):
            if time.time() >= restart_deadline:
                break
            h_start = slsqp_optimize(h_start, beta, rng, max_iter=iters)

        c_new = compute_c5(h_start)
        if c_new < best_c:
            best_h, best_c = h_start, c_new

        sigma = max(0.01, sigma * 0.995)
        restart_cnt += 1

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
