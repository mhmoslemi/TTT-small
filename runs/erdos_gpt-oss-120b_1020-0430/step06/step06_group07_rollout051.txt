def anneal_hill_climb(h, rng, time_budget):
    start = time.time()
    best_h = h.copy()
    best_min = compute_min_cross(best_h)
    current_h = best_h.copy()
    current_min = best_min
    n = h.size
    it = 0
    while time.time() - start < time_budget:
        # temperature schedule
        T = max(0.1, 1.0 - (time.time() - start) / time_budget)  # linear cooling from 1 to 0.1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        # ensure i is a 1 and j a 0; if not, swap
        if current_h[i] == current_h[j]:
            continue
        # swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        min_new = compute_min_cross(current_h)
        delta = min_new - current_min
        if delta >= 0 or rng.random() < np.exp(delta / T):
            current_min = min_new
            if current_min > best_min:
                best_min = current_min
                best_h = current_h.copy()
        else:
            # revert
            current_h[i], current_h[j] = current_h[j], current_h[i]
        it += 1
        # maybe break if best_min reaches 160
        if best_min >= 160:
            break
    return best_h, best_min

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 value for a discretised step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via a simple binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def worst_gradient(h: np.ndarray):
    """Sub‑gradient of the worst (largest) overlap."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))  # worst shift (largest overlap)
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

def random_binary_seq(N: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random binary vector with exactly N/2 ones."""
    h = np.zeros(N, dtype=np.uint8)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1
    return h

def compute_min_cross(h: np.ndarray) -> int:
    """Minimum linear autocorrelation count for binary h."""
    n = h.size
    hf = h.astype(np.float64)
    N2 = 2 * n
    H = np.fft.rfft(hf, N2)
    conv = np.fft.irfft(H * H, N2)
    cross = conv[n - 1:2 * n - 1]          # length n
    cross = np.rint(cross).astype(int)
    return int(cross.min())

def binary_anneal(N: int, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing style optimization of a binary sequence."""
    start = time.time()
    # start from a random feasible binary vector
    h = random_binary_seq(N, rng)
    best_h = h.copy()
    best_min = compute_min_cross(h)
    cur_h = h.copy()
    cur_min = best_min
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(0.1, 0.9 * (1.0 - elapsed / time_budget))
        i = rng.integers(0, N)
        j = rng.integers(0, N)
        if cur_h[i] == cur_h[j]:
            continue
        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_min = compute_min_cross(cur_h)
        delta = new_min - cur_min
        if delta >= 0 or rng.random() < np.exp(delta / T):
            cur_min = new_min
            if cur_min > best_min:
                best_min = cur_min
                best_h = cur_h.copy()
                if best_min >= 160:        # early exit if we hit the next integer level
                    break
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_min

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # Number of discretisation points
    if "initial_h_values" in globals():
        N = len(initial_h_values)
    else:
        N = 512

    # --- Phase 1: binary annealing (≈45% of time) ---
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    binary_time = remaining * 0.45
    if binary_time > 5.0:
        best_bin, _ = binary_anneal(N, rng, binary_time)
    else:
        best_bin = random_binary_seq(N, rng)

    # --- Phase 2: continuous sub‑gradient descent (≈80% of leftover) ---
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    cont_time = remaining * 0.85
    h0 = best_bin.astype(float)
    target = N / 2.0
    if cont_time > 5.0:
        best_h, best_c = subgradient_optimize(h0, target, cont_time, rng)
    else:
        best_h = h0
        best_c = compute_c5(best_h)

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
