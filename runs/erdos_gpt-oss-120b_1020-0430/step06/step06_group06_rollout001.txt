def avg_worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    epsilon = max_c * 1e-12 + 1e-12
    idxs = np.where(corr >= max_c - epsilon)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            kk = abs(k)
            # For both signs, same gradient (since corr symmetric)
            # Use positive kk
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
    if len(idxs) > 0:
        grad /= len(idxs)
    else:
        grad = 1.0 - 2.0 * h  # fallback
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for the SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk repair
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def avg_worst_gradient(h: np.ndarray) -> np.ndarray:
    """Average gradient of all (near) worst‑shift overlaps."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    eps = max_c * 1e-12 + 1e-12
    idxs = np.where(corr >= max_c - eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            kk = abs(k)
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
    if idxs.size:
        grad /= idxs.size
    else:
        grad = 1.0 - 2.0 * h
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_move(h: np.ndarray, rng: np.random.Generator,
                scale: float) -> np.ndarray:
    """Swap two entries or transfer a random amount while preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.2:  # pure swap
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_descent_refine(h: np.ndarray, rng: np.random.Generator,
                           max_iters: int = 1000, step_start: float = 0.02) -> np.ndarray:
    """Line‑search gradient descent using the averaged worst‑shift gradient."""
    n = h.size
    cur_c = compute_c5(h)
    step = step_start
    for _ in range(max_iters):
        grad = avg_worst_gradient(h)
        improved = False
        for _ in range(5):  # back‑track line search
            nh = h - step * grad
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum(nh, rng)
            nc = compute_c5(nh)
            if nc < cur_c:
                h, cur_c = nh, nc
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
        if step < 1e-8:
            break
        step *= 0.99
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        n = 512
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = h0.copy()
        best_c = compute_c5(best_h)

    best_h = enforce_sum(best_h, rng)
    best_c = compute_c5(best_h)

    # ----- simulated annealing -----
    current_h = best_h.copy()
    current_c = best_c
    T = 0.002
    eps_scale = 0.5
    it = 0
    SA_frac = 0.68
    SA_deadline = start + budget_s * SA_frac
    guided_prob = 0.6

    while time.time() < SA_deadline:
        if rng.random() < guided_prob:
            grad = avg_worst_gradient(current_h)
            nh = guided_perturb(current_h, grad, rng, scale=eps_scale)
        else:
            nh = random_move(current_h, rng, scale=eps_scale)
        nc = compute_c5(nh)
        delta = nc - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_h, current_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        it += 1
        if it % 500 == 0:
            T *= 0.995
            eps_scale *= 0.997
            T = max(T, 1e-12)
            eps_scale = max(eps_scale, 2e-6)

    # ----- gradient descent refine -----
    if time.time() < start + budget_s * 0.85:
        refined = gradient_descent_refine(best_h, rng,
                                          max_iters=1200, step_start=0.02)
        rc = compute_c5(refined)
        if rc < best_c:
            best_h, best_c = refined.copy(), rc

    # ----- soft‑max SLSQP passes -----
    beta_seq = (100.0, 1000.0, 5000.0, 20000.0, 80000.0)
    for beta in beta_seq:
        if time.time() > start + budget_s * 0.98:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=500)
        best_c = compute_c5(best_h)

    # ----- possible final perturb + SLSQP -----
    if time.time() < start + budget_s * 0.995:
        pert = best_h.copy()
        pert = np.clip(pert + rng.normal(scale=0.005, size=n), 0.0, 1.0)
        pert = enforce_sum(pert, rng)
        pert = slsqp_optimize(pert, 150000.0, rng, max_iter=400)
        pc = compute_c5(pert)
        if pc < best_c:
            best_h, best_c = pert, pc

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
