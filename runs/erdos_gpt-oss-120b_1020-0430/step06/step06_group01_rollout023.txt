import time
import numpy as np
import scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int,
    step_scale_start: float = 0.9,
    decay_factor: float = 0.95,
) -> tuple[np.ndarray, float]:
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step_scale = step_scale_start
    decay_int = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0.0:
            eps = rng.random() * max_eps * step_scale
            h[i] -= eps
            h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay_int == 0:
            step_scale *= decay_factor
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(
    h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 1500
) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-5
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = np.array(ones, dtype=int)
        zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = ones_arr[rng.integers(len(ones_arr))]
            j = zeros_arr[rng.integers(len(zeros_arr))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                idx_i = np.where(ones_arr == i)[0][0]
                idx_j = np.where(zeros_arr == j)[0][0]
                ones_arr[idx_i] = j
                zeros_arr[idx_j] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s
    TARGET = 0.3808

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n0 = 512
        h0 = rng.random(n0)
        h0 = enforce_sum(h0, rng)
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # binary simulated annealing for global search
    bin_time = total_budget * 0.25
    h_bin, c_bin = binary_sa(n, rng, bin_time)
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin
    if best_c <= TARGET:
        best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
        return best_h.tolist(), float(best_c), int(best_h.size)

    # remaining time for continuous refinement
    while time.time() - start < total_budget * 0.97:
        rem = total_budget - (time.time() - start)
        if rem < 5:
            break

        # ----- guided hill climb (long) -----
        max_iters = int(min(150000, rem * 1000))
        h = best_h.copy()
        # occasional random perturbation before climbing
        if rng.random() < 0.3:
            for _ in range(rng.integers(1, 6)):
                idx = rng.integers(0, n)
                delta = (rng.random() - 0.5) * 0.2
                h[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)
        h_hc, c_hc = guided_hill_climb(
            h, rng, max_iters, step_scale_start=0.9, decay_factor=0.95
        )
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc
            if best_c <= TARGET:
                break

        # ----- SLSQP soft‑max passes -----
        beta_seq = (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0, 50000.0)
        for beta in beta_seq:
            if time.time() - start > total_budget * 0.96:
                break
            h_s = slsqp_optimize(best_h, beta, rng, max_iter=1500)
            c_s = compute_c5(h_s)
            h_r = gradient_refine(h_s, rng, step=0.01, max_iter=4)
            c_r = compute_c5(h_r)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
            elif c_s < best_c:
                best_h, best_c = h_s, c_s
            if best_c <= TARGET:
                break

        # ----- short hill climb (fine) -----
        rem2 = total_budget - (time.time() - start)
        if rem2 < 2:
            break
        short_iters = int(min(10000, rem2 * 800))
        h_sc, c_sc = guided_hill_climb(
            best_h.copy(), rng, short_iters, step_scale_start=0.5, decay_factor=0.9
        )
        if c_sc < best_c:
            best_h, best_c = h_sc, c_sc
            if best_c <= TARGET:
                break

        # ----- tiny stochastic tweaks -----
        if time.time() - start > total_budget * 0.99:
            break
        for _ in range(2000):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc
                if best_c <= TARGET:
                    break

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
