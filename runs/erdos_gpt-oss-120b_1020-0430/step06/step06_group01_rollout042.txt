def max_corr(h):
    # h is array of float 0.0/1.0
    n = h.shape[0]
    # FFT length = next power of two >= 2*n?
    # Use n2 = 2*n
    n2 = 2*n
    H = np.fft.rfft(h[::-1], n=n2)  # rfft for real input lengths
    G = np.fft.rfft(1.0 - h, n=n2)
    corr = np.fft.irfft(H * G, n=n2)
    # take first 2*n-1 entries: index 0..2*n-2 (since length n2)
    max_val = np.max(corr[:2*n-1])
    return max_val

def binary_sa_fast(n, rng, time_budget):
    start = time.time()
    w = n // 2
    # initialize h random binary
    ones_idx = rng.choice(n, w, replace=False)
    mask = np.zeros(n, dtype=bool)
    mask[ones_idx] = True
    zeros_idx = np.setdiff1d(np.arange(n, dtype=int), ones_idx, assume_unique=True)
    idx_pos = np.empty(n, dtype=int)
    idx_pos[ones_idx] = np.arange(w)
    idx_pos[zeros_idx] = np.arange(w)
    h_float = mask.astype(float)
    cur_c = max_corr(h_float) * (2.0 / n)
    best_c = cur_c
    best_mask = mask.copy()
    # for cooling schedule
    T0 = cur_c
    T_end = 1e-5
    # precompute steps per iteration for cooling
    # we need total iterations estimate, we can compute based on elapsed time and remaining.
    # We'll loop while time < budget.
    iter_count = 0
    while time.time() - start < time_budget:
        # temperature update exponential
        frac = (time.time() - start) / time_budget
        T = T0 * ((T_end / T0) ** frac)
        # pick i in ones, j in zeros
        i = ones_idx[rng.integers(w)]
        j = zeros_idx[rng.integers(w)]
        # swap
        mask[i] = False
        mask[j] = True
        h_float[i] = 0.0
        h_float[j] = 1.0
        # evaluate
        new_c = max_corr(h_float) * (2.0 / n)
        delta = new_c - cur_c
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            # update idx arrays
            pos_i = idx_pos[i]
            pos_j = idx_pos[j]
            ones_idx[pos_i] = j
            zeros_idx[pos_j] = i
            idx_pos[i] = pos_j
            idx_pos[j] = pos_i
            if new_c < best_c:
                best_c = new_c
                best_mask = mask.copy()
                # early exit if below target
                if best_c <= 0.3808:
                    break
        else:
            # revert swap
            mask[i] = True
            mask[j] = False
            h_float[i] = 1.0
            h_float[j] = 0.0
        iter_count += 1
    return best_mask.astype(float), best_c

def compute_c5_fast(h):
    n = h.shape[0]
    n2 = 2 * n
    H = np.fft.rfft(h[::-1], n2)
    G = np.fft.rfft(1.0 - h, n2)
    corr = np.fft.irfft(H * G, n2)
    # only first 2*n-1 needed, but max will sit within this anyway
    max_val = np.max(corr[:2 * n - 1])
    return max_val * (2.0 / n)

def binary_sa_multi(n, rng, time_budget):
    start = time.time()
    w = n // 2
    best_mask = None
    best_c = np.inf

    while time.time() - start < time_budget:
        # initialize with best or random
        if best_mask is not None and rng.random() < 0.3:
            mask = best_mask.copy()
        else:
            ones_idx = rng.choice(n, w, replace=False)
            mask = np.zeros(n, dtype=bool)
            mask[ones_idx] = True

        # build arrays as before
        h_float = mask.astype(float)
        cur_c = compute_c5_fast(h_float)

        # parameters: use a schedule for temperature based on loop_time
        loop_time = min(15.0, time_budget - (time.time() - start))
        # run SA for loop_time seconds
        loop_start = time.time()
        # set T0 = cur_c, T_end small
        T0 = cur_c
        T_end = 1e-6
        while time.time() - loop_start < loop_time:
            # fraction for cooling
            frac = (time.time() - loop_start) / loop_time
            T = T0 * ((T_end / T0) ** frac)
            # random swap
            # we need ones and zeros indices; compute each time? Could compute from mask quickly.
            # if we maintain ones_idx and zeros_idx as earlier, we need to create them after each init, which costs O(n).
            # For brevity, we can just randomly select i and j and ensure they are opposite:
            i = rng.integers(n)
            j = rng.integers(n)
            if mask[i] == mask[j]:
                # need opposite; try again few times
                continue
            # swap
            mask[i], mask[j] = not mask[i], not mask[j]  # toggles
            h_float[i] = 1.0 - h_float[i]
            h_float[j] = 1.0 - h_float[j]
            new_c = compute_c5_fast(h_float)
            delta = new_c - cur_c
            if delta <= 0.0 or rng.random() < np.exp(-delta / T):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_mask = mask.copy()
                    if best_c <= 0.3808:
                        return best_mask.astype(float), best_c
            else:
                # revert swap
                mask[i], mask[j] = not mask[i], not mask[j]
                h_float[i] = 1.0 - h_float[i]
                h_float[j] = 1.0 - h_float[j]
        # maybe additional random perturbation if stuck?
        # continue outer while
    return best_mask.astype(float), best_c

import time
import numpy as np

def compute_c5_fast(h: np.ndarray) -> float:
    n = h.shape[0]
    n2 = 2 * n
    H = np.fft.rfft(h[::-1], n2)
    G = np.fft.rfft(1.0 - h, n2)
    corr = np.fft.irfft(H * G, n2)
    max_val = np.max(corr[: 2 * n - 1])
    return max_val * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 20000, step_scale: float = 0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5_fast(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def binary_sa_fast(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    w = n // 2
    ones_idx = rng.choice(n, w, replace=False)
    mask = np.zeros(n, dtype=bool)
    mask[ones_idx] = True
    zeros_idx = np.setdiff1d(np.arange(n, dtype=int), ones_idx, assume_unique=True)
    idx_pos = np.empty(n, dtype=int)
    idx_pos[ones_idx] = np.arange(w)
    idx_pos[zeros_idx] = np.arange(w)
    h = mask.astype(float)
    cur_c = compute_c5_fast(h)
    best_c, best_mask = cur_c, mask.copy()
    T0 = cur_c
    T_end = 1e-5
    while time.time() - start < time_budget:
        frac = (time.time() - start) / time_budget
        T = T0 * ((T_end / T0) ** frac)
        i = ones_idx[rng.integers(w)]
        j = zeros_idx[rng.integers(w)]
        mask[i] = False
        mask[j] = True
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5_fast(h)
        delta = new_c - cur_c
        accept = delta <= 0.0 or rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            pos_i = idx_pos[i]
            pos_j = idx_pos[j]
            ones_idx[pos_i] = j
            zeros_idx[pos_j] = i
            idx_pos[i] = pos_j
            idx_pos[j] = pos_i
            if new_c < best_c:
                best_c, best_mask = new_c, mask.copy()
                if best_c <= 0.3808:
                    break
        else:
            mask[i] = True
            mask[j] = False
            h[i] = 1.0
            h[j] = 0.0
    return best_mask.astype(float), best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # defaults
    best_h = None
    best_c = np.inf
    n = 512

    # initial global solution if provided
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5_fast(h0)
        best_h, best_c, n = h0, c0, h0.size

    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining <= 0:
        return best_h.tolist() if best_h is not None else [], float(best_c), int(n)

    # ---------- binary simulated annealing ----------
    binary_time = remaining * 0.55
    h_bin, c_bin = binary_sa_fast(n, rng, binary_time)
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin

    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining <= 0:
        final_c = compute_c5_fast(best_h) if best_h is not None else np.inf
        return best_h.tolist() if best_h is not None else [], float(final_c), int(best_h.size)

    # ---------- continuous refinement ----------
    max_iters = int(min(200000, remaining * 500))
    h_cont, c_cont = guided_hill_climb(best_h.copy(), rng, max_iters=max_iters, step_scale=0.7)
    if c_cont < best_c:
        best_h, best_c = h_cont, c_cont

    # finer gradient polishing
    h_ref = gradient_refine(best_h.copy(), rng, step=0.01, max_iter=8)
    c_ref = compute_c5_fast(h_ref)
    if c_ref < best_c:
        best_h, best_c = h_ref, c_ref

    # ---------- final stochastic tweaks ----------
    while time.time() - start < total_budget * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5_fast(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # ---------- sanitisation ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fast(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
