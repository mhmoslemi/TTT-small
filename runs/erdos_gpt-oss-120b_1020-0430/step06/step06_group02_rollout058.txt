import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, h, mode='full')
    min_corr = corr.min()
    return 1.0 - (2.0 / n) * min_corr

def enforce_sum_float(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # loop up to n*10 times
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    # after loop, maybe linearly adjust remaining diff
    if abs(diff) > 1e-12:
        # distribute across all entries proportionally within bounds
        # simple: add diff/n to all then clip, iterate
        for _ in range(10):
            add_all = diff / n
            h += np.clip(add_all, -h, 1 - h)  # but need elementwise min
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def enforce_sum_binary(h, rng):
    n = h.size
    target = n // 2
    diff = int(target - h.sum())
    while diff != 0:
        if diff > 0:
            # need to flip a zero to one
            i = rng.integers(0, n)
            if h[i] == 0:
                h[i] = 1
                diff -= 1
        else:
            i = rng.integers(0, n)
            if h[i] == 1:
                h[i] = 0
                diff += 1
    return h

def simulated_annealing_binary(h0, rng, time_limit):
    n = h0.size
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    # temperature schedule
    T = 0.005
    cooling = 0.99995
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # pick random swap
        a = rng.integers(0, n)
        while h[a] != 1:
            a = rng.integers(0, n)
        b = rng.integers(0, n)
        while h[b] != 0:
            b = rng.integers(0, n)
        # swap
        h[a] = 0
        h[b] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[a] = 1
            h[b] = 0
        # update T
        T *= cooling
        it += 1
        if it % 5000 == 0:
            # optional early break if near target
            if best_c5 <= 0.3808:
                break
    return best_h, best_c5

def gradient_descent_refine(h0, rng, time_limit):
    h = h0.astype(np.float64).copy()
    n = h.size
    target = n / 2.0
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    start = time.time()
    step = 0.2
    it = 0
    while time.time() - start < time_limit:
        # compute worst shift gradient
        # compute correlation of h with h
        corr = np.correlate(h, h, mode='full')
        # find shift with min corr (worst product)
        idx = np.argmin(corr) - (n - 1)
        k = abs(idx)
        grad = np.ones_like(h)
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:]    -= h[: n - k]
        # update
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum_float(h_new, rng)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5, best_h = new_c5, h_new.copy()
        # accept if improvement, else maybe revert
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
        # decay step
        step *= 0.9995
        it += 1
        if step < 1e-5:
            break
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
        h_init = np.clip(h_init, 0.0, 1.0)
        n = h_init.size
    else:
        n = 512
        h_init = None
    # If n not power of two? we'll keep it.
    # Prepare initial binary h
    if h_init is not None:
        # round to binary while preserving sum
        h_bin = (h_init > 0.5).astype(np.int16)
        h_bin = enforce_sum_binary(h_bin, rng)
    else:
        # random balanced binary
        h_bin = np.zeros(n, dtype=np.int16)
        h_bin[rng.choice(n, n//2, replace=False)] = 1
    # Compute initial C5
    init_c5 = compute_c5(h_bin.astype(np.float64))
    best_h = h_bin.astype(np.float64)
    best_c = init_c5
    # Time allocations
    total_time = budget_s
    # Phase 1: Simulated annealing binary (≈60% time)
    time_sa = total_time * 0.6
    h_sa, c_sa = simulated_annealing_binary(best_h.astype(np.int16), rng, time_sa)
    if c_sa < best_c:
        best_h = h_sa.astype(np.float64)
        best_c = c_sa
        # early stop if good enough
        if best_c <= 0.3808:
            return best_h.tolist(), float(best_c), int(n)
    # Phase 2: Gradient descent refinement (≈30% time)
    time_gd = total_time * 0.3
    h_gd, c_gd = gradient_descent_refine(best_h, rng, time_gd)
    if c_gd < best_c:
        best_h = h_gd
        best_c = c_gd
    # Phase 3: Random small perturbations (remaining time)
    time_rem = total_time - (time.time() - start)
    if time_rem > 0:
        # small random tweaks with acceptance
        h = best_h.copy()
        cur_c5 = best_c
        T = 0.001
        cooling = 0.9999
        start_rem = time.time()
        while time.time() - start_rem < time_rem:
            # pick swap
            a = rng.integers(0, n)
            if h[a] <= 0.0:
                continue
            b = rng.integers(0, n)
            if h[b] >= 1.0:
                continue
            # perform small epsilon transfer
            eps = min(h[a], 1.0 - h[b]) * rng.random() * 0.05
            if eps <= 0:
                continue
            h[a] -= eps
            h[b] += eps
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                cur_c5 = new_c5
                if new_c5 < best_c:
                    best_c = new_c5
                    best_h = h.copy()
            else:
                h[a] += eps
                h[b] -= eps
            T *= cooling
    # final enforce
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_float(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

   - If iter_since_improve > limit (e.g., 200k), then restart: set h to random balanced binary, recompute cur_c5, reset T to start, iter_since_improve=0.

Better to include the possibility that best found may be from current run; we maintain global best across restarts.

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k))dx for discrete step function."""
    n = h.size
    # use self‑correlation: min P_ℓ gives max overlap with complement
    if np.issubdtype(h.dtype, np.floating):
        corr = np.correlate(h, h, mode="full")
    else:
        corr = np.correlate(h.astype(np.int32), h.astype(np.int32), mode="full")
    min_corr = corr.min()
    return 1.0 - (2.0 / n) * min_corr


def enforce_sum_float(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to have sum exactly n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    # very tiny residual fix
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def enforce_sum_binary(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Flip bits until exactly n/2 ones are present."""
    n = h.size
    target = n // 2
    diff = int(target - h.sum())
    while diff != 0:
        i = rng.integers(0, n)
        if diff > 0 and h[i] == 0:
            h[i] = 1
            diff -= 1
        elif diff < 0 and h[i] == 1:
            h[i] = 0
            diff += 1
    return h


def simulated_annealing_binary(h0: np.ndarray, rng: np.random.Generator,
                               time_limit: float):
    """Binary SA respecting the sum constraint."""
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c

    T = 0.005
    cooling = 0.99995
    iter_since_best = 0
    start = time.time()

    while time.time() - start < time_limit:
        # pick a random 1 and a random 0
        a = rng.integers(0, n)
        while cur_h[a] != 1:
            a = rng.integers(0, n)
        b = rng.integers(0, n)
        while cur_h[b] != 0:
            b = rng.integers(0, n)

        # swap
        cur_h[a] = 0
        cur_h[b] = 1
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
                iter_since_best = 0
        else:
            # revert
            cur_h[a] = 1
            cur_h[b] = 0

        T *= cooling
        iter_since_best += 1

        # restart if we have been stagnant for long
        if iter_since_best > 200_000:
            cur_h = np.zeros(n, dtype=np.int16)
            cur_h[rng.choice(n, n // 2, replace=False)] = 1
            cur_c = compute_c5(cur_h)
            T = 0.005
            iter_since_best = 0

        if best_c <= 0.3808:
            break

    return best_h, best_c


def gradient_descent_refine(h0: np.ndarray, rng: np.random.Generator,
                           time_limit: float):
    """Projected gradient descent on the worst shift."""
    h = h0.astype(np.float64).copy()
    n = h.size
    target = n / 2.0
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    step = 0.2
    start = time.time()

    while time.time() - start < time_limit:
        # correlation of h with itself
        corr = np.correlate(h, h, mode="full")
        idx = np.argmin(corr) - (n - 1)          # shift giving min product
        k = abs(idx)

        grad = np.ones_like(h)
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:]     -= h[: n - k]

        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum_float(h_new, rng)

        new_c = compute_c5(h_new)
        if new_c < best_c:
            best_c, best_h = new_c, h_new.copy()

        if new_c < cur_c:
            h, cur_c = h_new, new_c

        step *= 0.9995
        if step < 1e-5:
            break
        if best_c <= 0.3808:
            break

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_all = time.time()

    # ------------------------------------------------------------------ #
    # Determine size and initialise a binary vector
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
        h_init = np.clip(h_init, 0.0, 1.0)
        n = h_init.size
        h_bin = (h_init > 0.5).astype(np.int16)
        h_bin = enforce_sum_binary(h_bin, rng)
    else:
        n = 512
        h_bin = np.zeros(n, dtype=np.int16)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1

    best_h = h_bin.astype(np.float64)
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------ #
    # Phase 1 : binary simulated annealing (≈60% of time)
    time_sa = budget_s * 0.6
    h_sa, c_sa = simulated_annealing_binary(h_bin, rng, time_sa)
    if c_sa < best_c:
        best_h = h_sa.astype(np.float64)
        best_c = c_sa
        if best_c <= 0.3808:
            return best_h.tolist(), float(best_c), int(n)

    # ------------------------------------------------------------------ #
    # Phase 2 : gradient descent refinement (≈30% of time)
    time_gd = budget_s * 0.3
    h_gd, c_gd = gradient_descent_refine(best_h, rng, time_gd)
    if c_gd < best_c:
        best_h = h_gd
        best_c = c_gd
        if best_c <= 0.3808:
            return best_h.tolist(), float(best_c), int(n)

    # ------------------------------------------------------------------ #
    # Phase 3 : tiny random epsilon transfers until time runs out
    time_rem = budget_s - (time.time() - start_all)
    if time_rem > 0.0:
        cur_h = best_h.copy()
        cur_c = best_c
        T = 0.001
        cooling = 0.9999
        start = time.time()
        while time.time() - start < time_rem:
            # choose a donor (h>0) and a receiver (h<1)
            a = rng.integers(0, n)
            if cur_h[a] <= 0.0:
                continue
            b = rng.integers(0, n)
            if cur_h[b] >= 1.0:
                continue
            eps = min(cur_h[a], 1.0 - cur_h[b]) * rng.random() * 0.05
            if eps <= 0.0:
                continue
            cur_h[a] -= eps
            cur_h[b] += eps
            new_c = compute_c5(cur_h)
            delta = new_c - cur_c
            accept = False
            if delta <= 0.0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur_h.copy()
            else:
                cur_h[a] += eps
                cur_h[b] -= eps
            T *= cooling

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_float(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
