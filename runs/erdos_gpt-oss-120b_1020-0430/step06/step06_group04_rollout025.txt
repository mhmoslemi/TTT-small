def momentum_gradient_descent(h, betas, base_lr, mu, it_per_beta, rng, deadline):
    n = h.size
    m = np.zeros_like(h)
    for beta in betas:
        lr = base_lr
        for it in range(it_per_beta):
            if time.time()>deadline: return h
            g = gradient_smoothmax(h, beta)
            m = mu*m + (1-mu)*g
            # attempt multiple lr multipliers
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = lr*mult
                cand = h - step*m
                np.clip(cand,0,1,out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
            lr *= 0.97  # decaying
    return h

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    max_it = 60000
    while time.time() < deadline and it < max_it:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        # exponential cooling
        T = init_T * (0.99995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0  # reset stagnation counter
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for t in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)  # we need to define worst_shift_gradient
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.98
    return h

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

def momentum_gradient_descent(h, betas, base_lr, mu, it_per_beta, rng, deadline):
    n = h.size
    m = np.zeros_like(h)
    for beta in betas:
        lr = base_lr
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            m = mu * m + (1 - mu) * g
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = h - lr * mult * m
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
            lr *= 0.96
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.98
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    max_iters = 50000
    while time.time() < deadline and it < max_iters:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.99995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None

    best_n = None

            h_bin = np.rint(h_up).astype(float)
            adjust_sum_exact(h_bin, rng)
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)

    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return h

CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                             "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.05)
    it = 0
    max_iters = 100000
    while time.time() < deadline and it < max_iters:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.99997 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None; best_c = np.inf; best_n = None
    base_n = 960
    restarts = 8
    betas = [1, 5, 20, 80, 200, 500, 2000, 5000, 20000, 50000]

    for _ in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if 'initial_h_values' in globals():
            parent = globals()['initial_h_values']
            m = len(parent)
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            ys = np.linspace(0.0, 2.0, m, endpoint=False)
            h = np.interp(xs, ys, parent)
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        h = momentum_gradient_descent(h, betas, 0.30, 0.9, 12, rng, deadline)
        h = worst_shift_refine(h, rng, 0.25, 120, deadline)
        for beta in (2000, 5000, 20000, 50000):
            if time.time() > deadline: break
            h = slsqp_optimize(h, beta, rng, 600)

        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.18)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_bin, n

    if best_h is not None and time.time() < deadline:
        n_up = best_n * 2
        if n_up <= 1920:
            h_up = np.repeat(best_h, 2)
            h_up += rng.normal(scale=0.015, size=h_up.shape)
            np.clip(h_up, 0.0, 1.0, out=h_up)
            adjust_sum_exact(h_up, rng)
            h_up = momentum_gradient_descent(h_up, betas[:6], 0.22, 0.9, 8, rng, deadline)
            h_up = worst_shift_refine(h_up, rng, 0.20, 100, deadline)
            for beta in (2000, 5000, 20000):
                if time.time() > deadline: break
                h_up = slsqp_optimize(h_up, beta, rng, 500)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up
            h_bin = np.rint(h_up).astype(float)
            adjust_sum_exact(h_bin, rng)
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
            h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n_up

    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
