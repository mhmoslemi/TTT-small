def run(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    h = np.zeros(n, dtype=np.uint8)
    h[rng.permutation(n)[:n//2]] = 1
    minC = compute_minC(h)
    best_h = h.copy()
    best_minC = minC
    # compute correlation once maybe for best
    # main loop
    while time.time() - start < budget_s * 0.99:
        # compute correlation for current h, need pos_counts for guided
        corr_full = np.correlate(h, h, mode='full')
        pos_counts = corr_full[n:]  # length n-1
        minC = int(pos_counts.min())
        # if this improved best, update
        if minC > best_minC:
            best_minC = minC
            best_h = h.copy()
        # compute shift(s)
        min_shift = int(np.argmin(pos_counts)) + 1
        # try guided swap
        # compute masks
        max_i = n - min_shift
        mask_i = (h[:max_i] == 1) & (h[min_shift:] == 0)
        mask_j = (h[:max_i] == 0) & (h[min_shift:] == 1)
        if mask_i.any() and mask_j.any():
            # choose random i and j
            idx_i = np.flatnonzero(mask_i)
            idx_j = np.flatnonzero(mask_j)
            i = idx_i[rng.integers(len(idx_i))]
            j = idx_j[rng.integers(len(idx_j))]
            # swap
            h[i], h[j] = h[j], h[i]
            # compute new minC
            new_minC = compute_minC(h)
            # if not improved (or maybe even same), keep? May still later improve.
            # We'll accept if new_minC >= minC (i.e., not decreasing)
            if new_minC >= minC:
                # keep, update best if improved
                if new_minC > best_minC:
                    best_minC = new_minC
                    best_h = h.copy()
                continue
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
        # else or not improvement => random swap with SA
        # random indices
        # pick random index of 1 and 0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # swap
        h[i], h[j] = h[j], h[i]
        new_minC = compute_minC(h)
        # Simulated annealing acceptance
        # Temperature schedule linear: T = T0 * (1 - elapsed/total)
        elapsed = time.time() - start
        T0 = 2.0
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)
        delta = new_minC - minC
        accept = False
        if delta >= 0:
            accept = True
        else:
            if T > 0:
                prob = math.exp(delta / T)  # delta negative
                if rng.random() < prob:
                    accept = True
        if accept:
            minC = new_minC
            if minC > best_minC:
                best_minC = minC
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

    # after loop, compute final C5 bound using best_h
    final_c5 = 1.0 - best_minC * (2.0 / n)
    # return list, float, n
    return best_h.tolist(), float(final_c5), n

def delta_O_for_swap(h, i0, i1):
    n = h.size
    ks = np.arange(n)  # 0 to n-1
    # bool arrays for indices in range
    in_range_i0_plus = i0 + ks < n
    in_range_i1_plus = i1 + ks < n
    i0_plus = np.empty(n, dtype=np.int64)
    i0_plus[in_range_i0_plus] = i0 + ks[in_range_i0_plus]
    # but we need values h[i0 + ks] for those ks; we can just fetch with advanced indexing using np.where.
    # Simpler: compute term1 = np.where(i0+ks < n, -(1 - h[i0+ks]), 0)
    term1 = np.zeros(n, dtype=np.int64)
    idx = i0 + ks[in_range_i0_plus]
    term1[in_range_i0_plus] = -(1 - h[idx])
    # term2 similarly for i1
    term2 = np.zeros(n, dtype=np.int64)
    idx = i1 + ks[in_range_i1_plus]
    term2[in_range_i1_plus] = (1 - h[idx])
    # term3: h[i0 - ks] where i0 - ks >=0
    in_range_i0_minus = i0 - ks >= 0
    term3 = np.zeros(n, dtype=np.int64)
    idx = i0 - ks[in_range_i0_minus]
    term3[in_range_i0_minus] = h[idx]
    # term4: -h[i1 - ks] where i1 - ks >=0
    in_range_i1_minus = i1 - ks >=0
    term4 = np.zeros(n, dtype=np.int64)
    idx = i1 - ks[in_range_i1_minus]
    term4[in_range_i1_minus] = -h[idx]
    delta = term1 + term2 + term3 + term4
    return delta

def compute_delta(h, i0, i1, ks):
    n = h.size
    delta = np.zeros_like(ks, dtype=np.int32)
    # term1: -(1 - h[i0 + k]) for k where i0+k < n
    idx = i0 + ks
    mask = idx < n
    delta[mask] -= (1 - h[idx[mask]])
    # term2: +(1 - h[i1 + k]) for i1 + k < n
    idx = i1 + ks
    mask = idx < n
    delta[mask] += (1 - h[idx[mask]])
    # term3: +h[i0 - k] for i0 - k >= 0
    idx = i0 - ks
    mask = idx >= 0
    delta[mask] += h[idx[mask]]
    # term4: -h[i1 - k] for i1 - k >= 0
    idx = i1 - ks
    mask = idx >= 0
    delta[mask] -= h[idx[mask]]
    return delta

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()

    # initialize balanced binary vector
    h = np.zeros(n, dtype=np.uint8)
    h[rng.permutation(n)[:n//2]] = 1
    # precompute ks
    ks = np.arange(1, n, dtype=np.int32)
    # compute O
    O = np.correlate(h, 1 - h, mode='full')[n:].astype(np.int32)  # length n-1
    Omax = int(O.max())
    best_Omax = Omax
    best_h = h.copy()
    dx = 2.0 / n
    target_c5 = 0.3808
    target_Omax = int(np.floor(target_c5 / dx))
    # parameters
    M = 60  # number of random swaps per iteration
    T0 = 5.0
    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.99:
            break

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        # guided swap based on max O
        k_idx = int(np.argmax(O))
        k = int(ks[k_idx])
        if n - k > 0:
            mask_i = (h[:n - k] == 1) & (h[k:] == 0)
            mask_j = (h[:n - k] == 0) & (h[k:] == 1)
            if mask_i.any() and mask_j.any():
                i = np.flatnonzero(mask_i)[rng.integers(mask_i.sum())]
                j = np.flatnonzero(mask_j)[rng.integers(mask_j.sum())]
                delta = compute_delta(h, i, j, ks)

                new_Omax = int(new_O.max())

                    Omax = new_Omax

        best_new_Omax = Omax
        best_delta = None

import time, math, numpy as np

def compute_delta(h: np.ndarray, i0: int, i1: int, ks: np.ndarray) -> np.ndarray:
    """Delta of O(k)=∑ h[i]*(1-h[i+k]) for k=1..n-1 after swapping
    h[i0]=1↔h[i1]=0.  Returns int32 array of length n-1."""
    n = h.size
    delta = np.zeros_like(ks, dtype=np.int32)

    # -(1 - h[i0 + k])  where i0+k < n
    idx = i0 + ks
    m = idx < n
    delta[m] -= (1 - h[idx[m]])

    # +(1 - h[i1 + k]) where i1+k < n
    idx = i1 + ks
    m = idx < n
    delta[m] += (1 - h[idx[m]])

    # +h[i0 - k] where i0-k >= 0
    idx = i0 - ks
    m = idx >= 0
    delta[m] += h[idx[m]]

    # -h[i1 - k] where i1-k >= 0
    idx = i1 - ks
    m = idx >= 0
    delta[m] -= h[idx[m]]

    return delta


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 512                     # number of samples
    dx = 2.0 / n                # discretisation step
    target_c5 = 0.3808
    target_Omax = int(np.floor(target_c5 / dx))

    # ----- initialise a balanced binary vector --------------------------------
    h = np.zeros(n, dtype=np.uint8)
    h[rng.permutation(n)[: n // 2]] = 1

    ks = np.arange(1, n, dtype=np.int32)          # shifts 1..n-1
    O = np.correlate(h, 1 - h, mode="full")[n:].astype(np.int32)
    Omax = int(O.max())

    best_h = h.copy()
    best_Omax = Omax

    # ---- parameters for the local search ------------------------------------
    M = 60                     # random swaps examined per iteration
    T0 = 5.0                   # initial temperature for SA
    no_improve = 0
    prev_Omax = Omax

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.99:
            break

        # current positions of 1s and 0s
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)

        # ----- guided move using the worst shift --------------------------------
        k_idx = int(np.argmax(O))          # index in ks of the maximal O(k)
        k = int(ks[k_idx])                 # actual shift value (≥1)

        if n - k > 0:
            mask_i = (h[: n - k] == 1) & (h[k:] == 0)
            mask_j = (h[: n - k] == 0) & (h[k:] == 1)
            if mask_i.any() and mask_j.any():
                i = np.flatnonzero(mask_i)[rng.integers(mask_i.sum())]
                j = np.flatnonzero(mask_j)[rng.integers(mask_j.sum())]
                delta = compute_delta(h, i, j, ks)
                new_O = O + delta
                new_Omax = int(new_O.max())
                if new_Omax < Omax:
                    h[i], h[j] = h[j], h[i]
                    O[:] = new_O
                    Omax = new_Omax
                    if Omax < best_Omax:
                        best_Omax = Omax
                        best_h = h.copy()
                    continue

        # ----- random‑swap sampling, keep the best improvement ------------------
        best_new_Omax = Omax
        best_delta = None
        best_pair = None

        for _ in range(M):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            delta = compute_delta(h, i, j, ks)
            new_Omax = int((O + delta).max())
            if new_Omax < best_new_Omax:
                best_new_Omax = new_Omax
                best_delta = delta
                best_pair = (i, j)
                if best_new_Omax <= target_Omax:
                    break

        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            O[:] = O + best_delta
            Omax = best_new_Omax
            if Omax < best_Omax:
                best_Omax = Omax
                best_h = h.copy()
            continue

        # ----- simulated annealing fallback ------------------------------------
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        delta = compute_delta(h, i, j, ks)
        new_Omax = int((O + delta).max())
        T = T0 * max(0.0, 1.0 - elapsed / (budget_s * 0.95))
        if new_Omax < Omax or (T > 0.0 and rng.random() < math.exp((Omax - new_Omax) / max(1e-12, T))):
            h[i], h[j] = h[j], h[i]
            O[:] = O + delta
            Omax = new_Omax
            if Omax < best_Omax:
                best_Omax = Omax
                best_h = h.copy()
            continue

        # ----- stagnation handling ---------------------------------------------
        if Omax == prev_Omax:
            no_improve += 1
        else:
            no_improve = 0
        prev_Omax = Omax

        # restart if stuck for a while (but keep the best found)
        if no_improve > 3000 and elapsed < budget_s * 0.8:
            h = np.zeros(n, dtype=np.uint8)
            h[rng.permutation(n)[: n // 2]] = 1
            O = np.correlate(h, 1 - h, mode="full")[n:].astype(np.int32)
            Omax = int(O.max())
            no_improve = 0
            prev_Omax = Omax
            continue

    final_c5 = best_Omax * dx
    return best_h.astype(float).tolist(), float(final_c5), int(n)
