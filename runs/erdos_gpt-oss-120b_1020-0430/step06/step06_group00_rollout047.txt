def worst_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)   # shift index
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1 - 2*h
    else:
        grad[:n-k] = 1 - h[k:]
        grad[k:] -= h[:n-k]
    return grad, k

def enforce_sum(h, rng):
    n = h.shape[0]
    target = n/2
    diff = target - h.sum()
    attempts = 0
    max_attempts = 5000
    while abs(diff) > 1e-12 and attempts < max_attempts:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    # If still off, small adjustment to random indices proportionally
    if abs(diff) > 1e-12:
        # fallback: scale all values by target / sum
        scale = target / h.sum()
        h *= scale
        # clip again
        np.clip(h, 0, 1, out=h)
    return h

def mutate_h(h, rng, eps_scale=0.05, num_mut=2):
    N = h.shape[0]
    for _ in range(num_mut):
        i = rng.integers(N)
        # random delta within +-eps_scale
        delta = (rng.random()*2 - 1) * eps_scale
        # clip delta to stay within bounds
        if delta > 0:
            delta = min(delta, 1 - h[i])
        else:
            delta = -min(-delta, h[i])
        if delta != 0:
            h[i] += delta
    # after mutations sum may change; enforce sum
    h = enforce_sum(h, rng)
    return h

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using FFT for speed."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    conv = np.fft.irfft(H * G, L)[: 2 * n - 1]
    return float(np.max(conv) * (2.0 / n))


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    max_att = 5000
    while abs(diff) > 1e-12 and attempts < max_att:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    if abs(diff) > 1e-12:
        scale = target / h.sum()
        h *= scale
        np.clip(h, 0.0, 1.0, out=h)
    return h


def random_h(N: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector with the required sum."""
    h = rng.random(N)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray):
    """Gradient of the shift attaining the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def gradient_descent(h: np.ndarray, steps: int, step0: float,
                    rng: np.random.Generator) -> np.ndarray:
    """Iterative refinement using the worst‑shift gradient."""
    n = h.shape[0]
    target = n / 2.0
    step = step0
    for _ in range(steps):
        grad, _ = worst_gradient(h)
        h -= step * grad
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = target - h.sum()
        step *= 0.995
    return h


def mutate_h(h: np.ndarray, rng: np.random.Generator,
             eps_scale: float = 0.05, num_mut: int = 2) -> np.ndarray:
    """Small random perturbations preserving bounds, then enforce sum."""
    n = h.shape[0]
    for _ in range(num_mut):
        i = rng.integers(n)
        delta = (rng.random() * 2.0 - 1.0) * eps_scale
        if delta > 0:
            delta = min(delta, 1.0 - h[i])
        else:
            delta = -min(-delta, h[i])
        if delta != 0.0:
            h[i] += delta
    return enforce_sum(h, rng)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    N = 512                     # primary resolution
    elite_size = 10
    init_pop = 30

    elites = []                 # list of (h, c5)

    # ----- possibly use provided construction -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != N:
            # up‑sample / down‑sample to N
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        elites.append((h0, c0))

    # ----- random initial population -----
    while len(elites) < init_pop:
        h = random_h(N, rng)
        c = compute_c5(h)
        elites.append((h, c))
        if time.time() - start > total_budget * 0.15:
            break

    elites.sort(key=lambda x: x[1])
    elites = elites[: elite_size]
    best_h, best_c = elites[0]

    # ----- evolutionary loop -----
    while True:
        now = time.time()
        if now - start > total_budget * 0.95:
            break

        p1 = elites[rng.integers(elite_size)][0]
        p2 = elites[rng.integers(elite_size)][0]
        alpha = rng.random()
        child = alpha * p1 + (1.0 - alpha) * p2
        np.clip(child, 0.0, 1.0, out=child)
        child = enforce_sum(child, rng)
        child = mutate_h(child, rng,
                         eps_scale=0.05,
                         num_mut=rng.integers(1, 4))
        child = gradient_descent(child, steps=80, step0=0.07, rng=rng)
        c_child = compute_c5(child)

        if c_child < best_c:
            best_h, best_c = child.copy(), c_child

        if c_child < elites[-1][1]:
            elites[-1] = (child.copy(), c_child)
            elites.sort(key=lambda x: x[1])

    # ----- final up‑sample & fine refinement -----
    if time.time() - start < total_budget * 0.99:
        N2 = 1024
        x_old = np.linspace(0.0, 2.0, N, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N2, endpoint=False)
        best_h = np.interp(x_new, x_old, best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_h = gradient_descent(best_h, steps=120, step0=0.04, rng=rng)
        best_c = compute_c5(best_h)
        N = N2

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
