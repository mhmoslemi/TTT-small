import time
import numpy as np
import cvxpy as cp

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that sum equals len(h)/2, staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (largest) overlap integral."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift giving worst overlap
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]              # -h[i+k] term
        grad[k:]    -= h[: n - k]           # -h[i-k] term
    return grad

def milp_optimize(n: int, time_limit: float, rng: np.random.Generator):
    """Solve the integer program that maximizes the minimal overlap count."""
    h = cp.Variable(n, boolean=True)
    C = cp.Variable(integer=True)

    constraints = [cp.sum(h) == n // 2]

    # For each shift create linking variables y and enforce sum(y) >= C
    for k in range(1, n):
        m = n - k
        y = cp.Variable(m, boolean=True)
        constraints += [
            y <= h[:m],
            y <= h[k:],
            y >= h[:m] + h[k:] - 1,
            cp.sum(y) >= C,
        ]

    # Objective: maximize C
    prob = cp.Problem(cp.Maximize(C), constraints)

    # Solve with CBC, respecting the time limit
    prob.solve(solver=cp.CBC, maximumSeconds=time_limit, msg=False)

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(h.value).astype(int)
    # Ensure exact sum constraint (rounding may disturb it)
    diff = (n // 2) - h_val.sum()
    while diff != 0:
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            if zeros.size == 0:
                break
            idx = rng.choice(zeros)
            h_val[idx] = 1
            diff -= 1
        else:
            ones = np.where(h_val == 1)[0]
            if ones.size == 0:
                break
            idx = rng.choice(ones)
            h_val[idx] = 0
            diff += 1

    C_val = int(C.value) if C.value is not None else None
    return h_val, C_val

def downsample_initial(initial, n):
    """Down‑sample a larger initial vector to length n by averaging blocks."""
    init = np.array(initial, dtype=float)
    L = init.size
    factor = L // n
    if factor * n != L:
        # simple interpolation fallback
        x_old = np.linspace(0, 1, L, endpoint=False)
        x_new = np.linspace(0, 1, n, endpoint=False)
        return np.interp(x_new, x_old, init)
    reshaped = init[: factor * n].reshape((n, factor))
    avg = reshaped.mean(axis=1)
    binary = (avg > 0.5).astype(float)
    return binary

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    elapsed = lambda: time.time() - start

    # -----------------------------------------------------------
    # 1. Choose problem size and allocate time for MILP
    n = 64
    milp_time = min(budget_s * 0.80, 800.0)

    # -----------------------------------------------------------
    # 2. Try MILP optimization
    h_bin, C_opt = milp_optimize(n, milp_time, rng)

    # -----------------------------------------------------------
    # 3. If MILP failed, fall back to a down‑sampled initial guess
    if h_bin is None:
        if "initial_h_values" in globals():
            h_bin = downsample_initial(initial_h_values, n)
        else:
            # simple random balanced binary fallback
            h_bin = np.zeros(n)
            h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
        h_bin = enforce_sum(h_bin, rng)

    # -----------------------------------------------------------
    # 4. Compute initial C5 bound
    best_h = h_bin.astype(float)
    best_c = compute_c5(best_h)

    # -----------------------------------------------------------
    # 5. Short gradient‑based refinement (allow fractional values)
    #    Use remaining time
    step0 = 0.2
    it = 0
    while elapsed() < budget_s * 0.98:
        # adapt step size slowly
        step = step0 * (0.985 ** it)
        if step < 1e-12:
            break
        grad = worst_gradient(best_h)
        nh = best_h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, rng)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
            # reset iteration counter to keep exploring with larger steps
            it = 0
            continue
        it += 1

    # -----------------------------------------------------------
    # 6. Final sanity checks and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
