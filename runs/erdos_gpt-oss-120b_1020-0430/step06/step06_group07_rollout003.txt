        T = T0 * (1 - frac)  # linear cooling

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n: try to use 2048 if enough time; else 1024. We'll allocate at least 700 sec for optimization.
    # Rough estimate: if budget > 300, use 2048.
    if budget_s > 400:
        n = 2048
    else:
        n = 1024
    # Initialize h0
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # If size differs from n, we need to upsample or downsample.
        if h0.size != n:
            # interpolate to size n
            x_old = np.linspace(0, 2, h0.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n/2.0)
    else:
        h0 = random_feasible(n, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: subgradient for ~0.4 of budget
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    sub_budget = remaining * 0.4
    if sub_budget > 5:
        h_sub, c_sub = subgradient_opt(best_h, n/2.0, sub_budget, rng)
        if c_sub < best_c:
            best_h, best_c = h_sub, c_sub

        if c_hc < best_c:

    # Sanitize
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n/2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def compute_c5_and_shift(h):
    n = h.size

    min_idx = int(np.argmin(cross))

def worst_gradient(h, shift):
    n = h.size
    grad = np.ones_like(h)
    if shift > 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:] -= h[: n - shift]
    elif shift < 0:
        k = -shift
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    else:  # shift == 0
        grad -= 2 * h
    return grad

def subgradient_opt(h0, target, time_budget, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c, _ = compute_c5_and_shift(h)
    start = time.time()

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

import time
import numpy as np

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def compute_c5_and_shift(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")
    min_idx = int(np.argmin(cross))
    min_cross = cross[min_idx]
    shift = min_idx - (n - 1)
    c5 = (h.sum() - min_cross) * (2.0 / n)
    return c5, shift


def worst_gradient(h: np.ndarray, shift: int):
    n = h.size
    grad = np.ones_like(h)
    if shift > 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:] -= h[: n - shift]
    elif shift < 0:
        k = -shift
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    else:
        grad -= 2.0 * h
    return grad


def subgradient_opt(h0: np.ndarray, target: float, time_budget: float,
                    rng: np.random.Generator):
    h = h0.copy()
    best_h = h.copy()
    best_c, _ = compute_c5_and_shift(h)
    start = time.time()
    t = 0
    while time.time() - start < time_budget:
        t += 1
        c, shift = compute_c5_and_shift(h)
        grad = worst_gradient(h, shift)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c_new, _ = compute_c5_and_shift(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c


def hill_climb(h0: np.ndarray, target: float, time_budget: float,
               rng: np.random.Generator):
    h = h0.copy()
    curr_c, shift = compute_c5_and_shift(h)
    best_h = h.copy()
    best_c = curr_c
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(best_c * 0.1, 1e-12) * (1.0 - frac)  # linear cooling

        # choose move type
        if rng.random() < 0.7:
            # guided move using gradient of worst shift
            grad = worst_gradient(h, shift)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j or h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                # fall back to random pair
                i = rng.integers(0, h.size)
                while h[i] <= 1e-12:
                    i = rng.integers(0, h.size)
                j = rng.integers(0, h.size)
                while h[j] >= 1.0 - 1e-12:
                    j = rng.integers(0, h.size)
        else:
            # random pair
            i = rng.integers(0, h.size)
            while h[i] <= 1e-12:
                i = rng.integers(0, h.size)
            j = rng.integers(0, h.size)
            while h[j] >= 1.0 - 1e-12:
                j = rng.integers(0, h.size)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.5
        if eps <= 0:
            continue

        h[i] -= eps
        h[j] += eps
        new_c, new_shift = compute_c5_and_shift(h)

        if new_c < curr_c or rng.random() < np.exp((curr_c - new_c) / max(T, 1e-12)):
            curr_c = new_c
            shift = new_shift
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
    return best_h, best_c


def upsample_h(h: np.ndarray):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)

    rng = np.random.default_rng(seed)
    start = time.time()

    # Choose resolution based on total budget
    n = 2048 if budget_s > 400 else 1024

    # Initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        h0 = random_feasible(n, rng)

    best_h = h0.copy()
    best_c, _ = compute_c5_and_shift(best_h)

    # Phase 1: sub‑gradient descent (≈40% of time)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    sub_budget = remaining * 0.4
    if sub_budget > 5.0:
        h_sg, c_sg = subgradient_opt(best_h, n / 2.0, sub_budget, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # Phase 2: guided hill‑climb (remaining time)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        h_hc, c_hc = hill_climb(best_h, n / 2.0, remaining, rng)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # Phase 3: optional up‑sample and short polish
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 4.0 and n * 2 <= 4096:
        h_up = upsample_h(best_h)
        sub_budget = remaining * 0.6
        if sub_budget > 2.0:
            h_up_sg, c_up_sg = subgradient_opt(h_up, h_up.size / 2.0,
                                               sub_budget, rng)
            if c_up_sg < best_c:
                best_h, best_c = h_up_sg, c_up_sg
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining > 2.0:

    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c, _ = compute_c5_and_shift(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
