def milp_optimize(N, rng, time_limit):
    import cvxpy as cp
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N/2]
    for k in range(1, N):
        y = cp.Variable(N - k)
        constraints += [
            y <= h[:N-k],
            y <= h[k:],
            y >= h[:N-k] + h[k:] - 1,
            y >= 0,
            cp.sum(y) >= t
        ]
    obj = cp.Maximize(t)
    prob = cp.Problem(obj, constraints)
    try:
        # solver options
        prob.solve(solver=cp.SCIP, timeLimit=time_limit, msg=False)
    except Exception as e:
        try:
            prob.solve(solver=cp.CBC, maxSeconds=time_limit, msg=False)
        except Exception:
            prob.solve(solver=cp.ECOS_BB, max_iters=5000, verbose=False)
    if h.value is None:
        return None, np.inf
    h_val = np.array(h.value).flatten()
    # Round to 0 or 1 respecting sum
    h_val = np.rint(h_val)
    # Ensure exactly N/2 ones
    diff = int(N/2 - h_val.sum())
    if diff > 0:
        # set diff zeros to 1
        zeros = np.where(h_val == 0)[0]
        if len(zeros) >= diff:
            idx = rng.choice(zeros, diff, replace=False)
            h_val[idx] = 1
    elif diff < 0:
        ones = np.where(h_val == 1)[0]
        if len(ones) >= -diff:
            idx = rng.choice(ones, -diff, replace=False)
            h_val[idx] = 0
    return h_val.astype(float), compute_c5(h_val.astype(float))
```

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None
    # initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = enforce_sum(np.clip(h0,0,1), rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size
    # Phase MILP
    remaining = budget_s - (time.time() - start)
    if remaining > 250:
        N = 128
        time_limit = min(remaining*0.9, 300)
        h_bin, c_bin = milp_optimize(N, rng, time_limit)
        if h_bin is not None and c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, N
    # Phase upsample to 512 if best_n < 512 and time left
    remaining = budget_s - (time.time() - start)
    if remaining > 0 and best_h is not None:
        target_n = 512
        if best_n < target_n:
            h_mid = upsample_h(best_h, target_n)
            h_mid = np.clip(h_mid, 0, 1)
            h_mid = enforce_sum(h_mid, rng)
            c_mid = compute_c5(h_mid)
            if c_mid < best_c:
                best_h, best_c, best_n = h_mid, c_mid, target_n
    # Phase local refinement continuous SA for remaining time
    remaining = budget_s - (time.time() - start)
    if remaining > 0 and best_h is not None:
        h_ref, c_ref = continuous_sa(best_h, rng, remaining)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref
            best_n = best_h.size
    # final sanitisation
    best_h = np.clip(best_h, 0, 1)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------- core utilities -----------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        it += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def random_feasible(N: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(N, dtype=float)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1.0
    return h


# ----------------- soft‑max SLSQP -----------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----------------- subgradient descent -----------------
def subgradient_descent(h: np.ndarray, rng: np.random.Generator,
                       time_budget: float, step0: float = 0.05,
                       decay: float = 0.001, max_iter: int = 10000):
    """Subgradient method on max‑overlap objective."""
    start = time.time()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() - start < time_budget and it < max_iter:
        corr = np.correlate(h, 1.0 - h, mode="full")
        maxc = np.max(corr)
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        alpha = step0 / (1.0 + decay * it)
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new <= maxc * (2.0 / n):
            h = h_new
        it += 1
    return best_h, best_c


# ----------------- continuous SA (fallback) -----------------
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c


# ----------------- main driver -----------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf

        best_h = random_feasible(N0, rng)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = N0

    # ----- Phase 1: soft‑max SLSQP on 512 points -----
    remaining = total_budget - (time.time() - start)
    if remaining > 30.0:
        phase_time = min(remaining * 0.35, 300.0)
        t_start = time.time()
        while time.time() - t_start < phase_time:
            h_start = best_h.copy()
            for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
                if time.time() - start > total_budget * 0.96:
                    break
                h_start = slsqp_optimize(h_start, beta, rng, max_iter=600)
            c_new = compute_c5(h_start)
            if c_new < best_c:
                best_h, best_c = h_start.copy(), c_new
                best_n = h_start.size
        # end of Phase 1

    # ----- Phase 2: subgradient descent at 512 -----
    remaining = total_budget - (time.time() - start)
    if remaining > 20.0:
        t_sgd = subgradient_descent(best_h, rng, remaining * 0.4, step0=0.07,

        h_sgd, c_sgd = t_sgd
        if c_sgd < best_c:
            best_h, best_c = h_sgd.copy(), c_sgd
            best_n = h_sgd.size

    # ----- Phase 3: upsample to 1024 -----
    remaining = total_budget - (time.time() - start)
    if remaining > 30.0:
        h_up = upsample_h(best_h, 1024)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up.copy(), c_up
            best_n = h_up.size

    # ----- Phase 4: subgradient descent at 1024 -----
    remaining = total_budget - (time.time() - start)
    if remaining > 20.0:
        h_sgd2, c_sgd2 = subgradient_descent(best_h, rng, remaining * 0.5,
                                             step0=0.05, decay=0.001)
        if c_sgd2 < best_c:
            best_h, best_c = h_sgd2.copy(), c_sgd2
            best_n = h_sgd2.size

    # ----- Phase 5: final continuous SA (if time) -----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        h_sa, c_sa = continuous_sa(best_h, rng, remaining)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa
            best_n = h_sa.size

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
