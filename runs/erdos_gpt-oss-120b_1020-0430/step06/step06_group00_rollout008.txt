def compute_c5(h):
    n = h.size
    m = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1-h, m)
    corr = np.fft.irfft(H*G, m)[:2*n-1]
    return np.max(corr) * (2.0/n)
```

    n = h.shape[0]
    m = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, m)

def genetic_algorithm(initial_h, rng, time_budget, pop_size=30, n_mutations=3):
    start = time.time()
    n = initial_h.size
    target = n/2.0
    # Initialize population
    pop = []
    # include initial_h
    pop.append(initial_h.copy())
    # fill rest
    while len(pop) < pop_size:
        h = rng.random(n)
        # scale to target sum via repair
        h = repair_sum(h, target, rng)
        pop.append(h)
    # precompute scores
    scores = np.array([compute_c5(h) for h in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # GA loop
    # parameters for scheduling step size
    step_initial = 0.25
    step_final = 0.01
    while time.time() - start < time_budget:
        # fraction elapsed
        frac = (time.time() - start) / time_budget
        step = step_initial * (1 - frac) + step_final * frac
        # elitism keep best
        new_pop = [best_h.copy()]
        # generate rest
        while len(new_pop) < pop_size:
            # tournament selection
            i1 = tournament_select(scores, rng)
            i2 = tournament_select(scores, rng)
            p1, p2 = pop[i1], pop[i2]
            # crossover (blend)
            alpha = rng.random()
            child = alpha * p1 + (1 - alpha) * p2
            # mutation
            for _ in range(n_mutations):
                idx = rng.integers(0, n)
                delta = (rng.random()*2 - 1) * step  # uniform in [-step, step]
                child[idx] = np.clip(child[idx] + delta, 0.0, 1.0)
            # repair sum
            child = repair_sum(child, target, rng)
            new_pop.append(child)
        # evaluate
        pop = new_pop
        scores = np.array([compute_c5(h) for h in pop])
        # update best
        cur_best_idx = np.argmin(scores)
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = pop[cur_best_idx].copy()
    return best_h, best_score
```

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    T0, T_end = 0.005, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # direction
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        score_new = compute_c5(h_new)
        delta = score_new - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            if score_new < best_score:
                best_score = score_new
                best_h = h_new.copy()
    return best_h, best_score

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = repair_sum(x, CURRENT_TARGET, rng)
    return x
```

def gradient_refine(h, rng, step=0.015, max_iter=15):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * grad
        np.clip(h, 0.0, 1.0, out=h)
        # repair sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = repair_sum(h, target, rng)
    return h

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * G, m)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)

def repair_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < n * 10:
        i = rng.integers(0, n)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            if inc > 0:
                h[i] += inc
                diff -= inc
        else:
            dec = min(-diff, h[i])
            if dec > 0:
                h[i] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            i = rng.integers(0, n)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def tournament_select(scores: np.ndarray, rng: np.random.Generator, k: int = 3) -> int:
    idx = rng.choice(len(scores), k, replace=False)
    return idx[np.argmin(scores[idx])]

def genetic_algorithm(initial_h: np.ndarray, rng: np.random.Generator,
                     time_budget: float, pop_size: int = 30,
                     n_mutations: int = 3) -> tuple:
    start = time.time()
    n = initial_h.size
    target = n / 2.0
    pop = [initial_h.copy()]
    while len(pop) < pop_size:
        h = rng.random(n)
        pop.append(repair_sum(h, target, rng))
    scores = np.array([compute_c5(h) for h in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    step_initial, step_final = 0.25, 0.01
    while time.time() - start < time_budget:
        frac = (time.time() - start) / time_budget
        step = step_initial * (1 - frac) + step_final * frac
        new_pop = [best_h.copy()]
        while len(new_pop) < pop_size:
            i1 = tournament_select(scores, rng)
            i2 = tournament_select(scores, rng)
            p1, p2 = pop[i1], pop[i2]
            alpha = rng.random()
            child = alpha * p1 + (1 - alpha) * p2
            for _ in range(n_mutations):
                idx = rng.integers(0, n)
                delta = (rng.random() * 2 - 1) * step
                child[idx] = np.clip(child[idx] + delta, 0.0, 1.0)
            child = repair_sum(child, target, rng)
            new_pop.append(child)
        pop = new_pop
        scores = np.array([compute_c5(h) for h in pop])
        cur_best = np.argmin(scores)
        cur_score = scores[cur_best]
        if cur_score < best_score:
            best_score = cur_score
            best_h = pop[cur_best].copy()
    return best_h, best_score

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = repair_sum(x, CURRENT_TARGET, rng)
    return x

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * grad
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = repair_sum(h, target, rng)
    return h

def continuous_sa(h0: np.ndarray, rng: np.random.Generator,
                  time_budget: float) -> tuple:
    start = time.time()
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    T0, T_end = 0.005, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        score_new = compute_c5(h_new)
        delta = score_new - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            if score_new < best_score:
                best_score = score_new
                best_h = h_new.copy()
    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s
    best_h = None
    best_score = np.inf
    if 'initial_h_values' in globals():

        if ga_s < best_score:

            for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
