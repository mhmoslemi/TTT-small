import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    # bisection on lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo+hi)*0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # adjust tiny error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x>1e-12) & (x<1-1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(v - ((lo+hi)*0.5)))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def softmax_grad(h, beta):
    n = h.size
    # correlation O
    O = np.correlate(h, 1.0 - h, mode='full')
    maxO = O.max()
    w = np.exp(beta * (O - maxO))
    w_sum = w.sum()
    w = w / w_sum
    # convs
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    mid = n-1
    g = 1.0 - (conv1[mid:mid+n] + conv2[mid:mid+n])
    return g

import time, math, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(v - lam))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def softmax_grad(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    maxO = O.max()
    w = np.exp(beta * (O - maxO))
    w /= w.sum()
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    mid = n - 1
    return 1.0 - (conv1[mid:mid + n] + conv2[mid:mid + n])

            step = step0 * (1.0 - it / steps) / norm
            h = project_onto_box_sum(h - step * grad, target)
        maybe_update(h)

    # ----- final stochastic refinement --------------------------------------
    stagnation = 0
    while time.time() - start < budget_s * 0.995:
        scale = 0.02 if stagnation % 1000 else 0.2
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=scale)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc
            stagnation = 0
        else:
            stagnation += 1

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
