def compute_c5_fft(h):
    n = h.shape[0]
    f = 1.0 - h
    L = 1 << ((2*n - 1).bit_length())  # next power of 2
    H = np.fft.rfft(h, L)
    F = np.fft.rfft(f, L)
    corr = np.fft.irfft(H * np.conj(F), L)
    # Only first 2*n-1 entries valid (others are zero due to padding)
    max_corr = corr[:2*n - 1].max()
    return max_corr * (2.0 / n)

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time, numpy as np

_fft_len_cache = {}
def _fft_len(n):
    L = _fft_len_cache.get(n)
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
        _fft_len_cache[n] = L
    return L

def project_onto_box_sum(v, target):
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.shape[0]
    L = _fft_len(n)
    H = np.fft.rfft(h, L)
    F = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(F), L)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def compute_corr_fft(h):
    n = h.shape[0]
    L = _fft_len(n)
    H = np.fft.rfft(h, L)
    F = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * np.conj(F), L)[:2*n-1]

def worst_gradient_fft(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        kk = abs(k)
        # The pattern is symmetric for +k and -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine dimension
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 512
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    target = n / 2.0
    h0 = project_onto_box_sum(h0, target)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # ---------- Random sampling phase ----------
    phase_end = start + budget_s * 0.20
    while time.time() < phase_end:
        v = rng.random(n)
        h = project_onto_box_sum(v, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # ---------- Hill‑climbing phase ----------
    h = best_h.copy()
    cur_c = best_c
    scale = 0.5
    last_improve = time.time()
    loop_end = start + budget_s * 0.99
    iter_cnt = 0
    while time.time() < loop_end:
        # gradient of the active shift
        grad = worst_gradient_fft(h)
        # occasional decay of scale
        if iter_cnt and iter_cnt % 2000 == 0:
            scale = max(scale * 0.7, 0.001)
        nh = guided_perturb(h, grad, rng, scale=scale)
        nc = compute_c5(nh)
        # accept if improves current point
        if nc < cur_c:
            h, cur_c = nh, nc
        # record global improvement
        if nc < best_c:
            best_c, best_h = nc, nh.copy()
            last_improve = time.time()
            scale = 0.5
        # if stagnated for a while, jump to a fresh random point
        if time.time() - last_improve > budget_s * 0.10:
            v = rng.random(n)
            h = project_onto_box_sum(v, target)
            cur_c = compute_c5(h)
            last_improve = time.time()
            scale = 0.5
        iter_cnt += 1
    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
