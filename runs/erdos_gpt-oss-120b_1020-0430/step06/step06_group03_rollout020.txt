def continuous_opt(start_h, time_limit, rng):
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    # ensure feasible:
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target)
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    # Subgradient phase (maybe 15% of time)
    sub_time_limit = time_limit * 0.15
    while time.time() - start < sub_time_limit:
        grad = worst_gradient(h)
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = 0.5 / np.sqrt(time.time() - start + 1e-12) / norm
            v = h - step * grad
            h = project_onto_box_sum(v, target)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    # Annealing phase
    anneal_start = time.time()
    while time.time() - start < time_limit:
        # pick random i,j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max < 1e-12:
            continue
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        scale = max(0.01, 0.5 * (1 - t_frac))
        eps = eps_max * scale * rng.random()
        if eps < 1e-12:
            eps = eps_max * 1e-12
        # propose move
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        # temperature schedule
        T0 = best_c * 0.04 + 1e-5
        T_end = 1e-7
        T = T0 * (1 - t_frac) + T_end
        if new_c <= cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
    # Final local refinement (guided hill climb) if time remains
    remaining = time_limit - (time.time() - start)
    if remaining > 1.0:

import time, math, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    decay = max(1, max_iters // 5)
    step = step_scale
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def binary_anneal(start_h, time_limit, rng):
    h = start_h.copy()
    n = h.size
    target = n // 2
    cur = h.sum()
    if cur > target:
        idx = np.where(h == 1)[0]
        to_zero = rng.choice(idx, int(cur - target), replace=False)
        h[to_zero] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        to_one = rng.choice(idx, int(target - cur), replace=False)
        h[to_one] = 1
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    T0, T_end = 0.08, 1e-6
    while time.time() - start < time_limit:
        # pick random 1 and 0
        while True:
            i = rng.integers(n)
            if h[i] == 1:
                break
        while True:
            j = rng.integers(n)
            if h[j] == 0:
                break
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        T = T0 * (1 - t_frac) + T_end
        if new_c <= cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c

def continuous_opt(start_h, time_limit, rng):
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target)
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    # subgradient phase (≈15% of allocated time)
    sub_limit = time_limit * 0.15
    it_sub = 0
    while time.time() - start < sub_limit:
        grad = worst_gradient(h)
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = 0.5 / np.sqrt(it_sub + 1) / norm
            v = h - step * grad
            h = project_onto_box_sum(v, target)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        it_sub += 1
    # annealing phase
    while time.time() - start < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max < 1e-12:
            continue
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        scale = max(0.01, 0.5 * (1 - t_frac))
        eps = eps_max * scale * rng.random()
        eps = max(eps, eps_max * 1e-12)
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        T0, T_end = best_c * 0.04 + 1e-5, 1e-7
        T = T0 * (1 - t_frac) + T_end
        if new_c <= cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i] += eps
            h[j] -= eps
    # final local refinement if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 1.0:
        max_iters = int(remaining * 2000)
        if max_iters > 0:
            hc, cc = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.4)
            if cc < best_c:
                best_h, best_c = hc, cc
    return best_h, best_c
