def project_feasible(h, target):
    # h may be any real numbers
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    N = h.size
    # mask of indices that can be increased if diff>0 or decreased if diff<0
    # We'll loop.
    # Initialize mask = np.ones(N, dtype=bool)
    mask = np.ones(N, dtype=bool)
    max_iter = N
    for _ in range(max_iter):
        free = mask & ((diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0))
        # free indices where we can move
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        # apply delta but clamp to bounds
        h_new = h[idx] + delta
        # find overshoot:
        if diff > 0:
            overshoot = h_new > 1.0
        else:
            overshoot = h_new < 0.0
        if np.any(overshoot):
            # for those overshoot, set to bound and adjust diff accordingly
            # compute actual change for those overshoot
            actual_delta = np.where(overshoot,
                                    (1.0 - h[idx]) if diff > 0 else (-h[idx]),
                                    delta)
            # update h
            h[idx] = np.where(overshoot,
                              1.0 if diff > 0 else 0.0,
                              h_new)
            # recompute diff
            diff = target - h.sum()
            # update mask: set overshoot indices as fixed (mask False)
            mask[idx[overshoot]] = False
        else:
            # no overshoot, just update
            h[idx] = h_new
            diff = target - h.sum()
            break
    # final clip (just in case)
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_and_idx(h):
    N = h.size
    # FFT corr
    L = 1 << ((2*N-1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    corr = np.fft.irfft(H * G, n=L)[:2*N-1]
    idx = int(np.argmax(corr))
    c = corr[idx] * (2.0 / N)
    return idx, c

def project_feasible(v, target):

    # First clip

    cur_sum = v.sum()

    if abs(diff) < 1e-12:
        return v
    N = v.size
    # We'll iteratively adjust free variables.
    # Use a mask for variables that are not at bound in direction of diff
    free = np.ones(N, dtype=bool)
    max_iter = N
    for _ in range(max_iter):
        # Determine which variables can move
        if diff > 0:
            can_move = free & (v < 1.0 - 1e-12)
        else:
            can_move = free & (v > 1e-12)
        idx = np.where(can_move)[0]
        if idx.size == 0:
            # cannot move any further
            break

def gradient_descent(h, target, rng, max_iters, time_budget, start_time):
    step = 0.1
    max_step = 0.5
    min_step = 1e-6
    cur_idx, cur_c = compute_c5_and_idx(h)
    N = h.size
    it = 0
    while it < max_iters and (time.time() - start_time) < time_budget:
        k = abs(cur_idx - (N-1))
        # gradient
        if k == 0:
            grad = 1.0 - 2.0*h
        else:
            grad = np.zeros(N, dtype=h.dtype)
            grad[:N - k] = 1.0 - h[k:]
            grad[k:] -= h[:N - k]
        # propose new h
        h_new = h - step * grad
        # project
        h_new = project_feasible(h_new, target)
        # compute new c and idx
        new_idx, new_c = compute_c5_and_idx(h_new)
        if new_c <= cur_c:
            # accept
            h = h_new
            cur_idx = new_idx
            cur_c = new_c
            step = min(step * 1.05, max_step)
        else:
            step = max(step * 0.5, min_step)
        it += 1
    return h, cur_c

import time
import numpy as np

# ---------- Core correlation utilities ----------
def _conv_fft(h: np.ndarray) -> np.ndarray:
    """Convolution of h with reversed (1‑h) via FFT (full length)."""
    N = h.size
    L = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    return np.fft.irfft(H * G, n=L)[: 2 * N - 1]


def compute_c5_and_idx(h: np.ndarray):
    """Return (index of maximal overlap in correlation, C5 value)."""
    conv = _conv_fft(h)
    corr = conv[::-1]
    idx = int(np.argmax(corr))
    c5 = corr[idx] * (2.0 / h.size)
    return idx, c5


def compute_c5_fast(h: np.ndarray) -> float:
    """C5 using only the maximal convolution value."""
    return np.max(_conv_fft(h)) * (2.0 / h.size)


# ---------- Feasibility helpers ----------
def project_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto [0,1] box and sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    N = v.size
    free = np.ones(N, dtype=bool)
    for _ in range(N):
        if diff > 0:
            can = free & (v < 1.0 - 1e-12)
        else:
            can = free & (v > 1e-12)
        idx = np.where(can)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        if diff > 0:
            overflow = v[idx] + delta > 1.0
        else:
            overflow = v[idx] + delta < 0.0
        if np.any(overflow):
            oidx = idx[overflow]
            if diff > 0:
                actual = 1.0 - v[oidx]
                v[oidx] = 1.0
            else:
                actual = -v[oidx]
                v[oidx] = 0.0
            diff -= actual.sum()
            free[oidx] = False
        else:
            v[idx] += delta
            diff = target - v.sum()
            break
    return np.clip(v, 0.0, 1.0)


def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Exact sum repair by tiny transfers."""
    diff = target - v.sum()
    N = v.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(N)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
            if add <= 0:
                continue
            v[i] += add
            diff -= add
        else:
            sub = min(-diff, v[i])
            if sub <= 0:
                continue
            v[i] -= sub
            diff += sub
    return v


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points over [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def random_feasible_h(N: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^N with sum N/2."""
    v = rng.random(N)
    return project_feasible(v, N / 2.0)


# ---------- Gradient descent on worst shift ----------
def gradient_descent(h: np.ndarray, target: float, rng: np.random.Generator,
                     max_iters: int, deadline: float):
    """Sum‑preserving projected gradient descent on C5."""
    step, max_step, min_step = 0.1, 0.5, 1e-6
    N = h.size
    idx, cur_c = compute_c5_and_idx(h)
    it = 0
    while it < max_iters and time.time() < deadline:
        k = abs(idx - (N - 1))
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.zeros_like(h)
            grad[: N - k] = 1.0 - h[k:]
            grad[k:] -= h[: N - k]
        h_new = project_feasible(h - step * grad, target)
        new_idx, new_c = compute_c5_and_idx(h_new)
        if new_c <= cur_c:
            h, idx, cur_c = h_new, new_idx, new_c
            step = min(step * 1.05, max_step)
        else:
            step = max(step * 0.5, min_step)
        it += 1
    return h, cur_c


# ---------- Main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    N_target = 2048
    target = N_target / 2.0
    best_h = None
    best_c = np.inf

    # ---- start from parent solution if available ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = upsample_h(h0, N_target)
        h0 = project_feasible(h0, target)
        rem = total_budget - (time.time() - start)
        if rem > 5.0:
            deadline = time.time() + rem * 0.30
            h_opt, c_opt = gradient_descent(h0, target, rng,
                                            max_iters=50000,
                                            deadline=deadline)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # ---- random restarts with gradient descent ----
    while True:
        rem = total_budget - (time.time() - start)
        if rem <= 10.0:
            break
        t_restart = min(rem * 0.04, 25.0)
        deadline = time.time() + t_restart
        h0 = random_feasible_h(N_target, rng)
        h_opt, c_opt = gradient_descent(h0, target, rng,
                                        max_iters=50000,
                                        deadline=deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- final fine‑tuning on the best vector ----
    rem = total_budget - (time.time() - start)
    if rem > 5.0 and best_h is not None:

        h_opt, c_opt = gradient_descent(best_h, target, rng,

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
