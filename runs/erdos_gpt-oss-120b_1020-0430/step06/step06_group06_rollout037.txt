def compute_c5_fft(h):
    N = h.size
    dx = 2.0 / N
    comp = 1.0 - h
    L = 2 * N
    H = np.fft.rfft(h, n=L)
    C = np.fft.rfft(comp, n=L)
    conv = np.fft.irfft(H*C, n=L)[:2*N-1] * dx
    return conv.max()

import time

# ---------- helper utilities ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using fast convolution (same as np.correlate, scaled by dx)."""
    n = h.size
    dx = 2.0 / n
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    C = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * C, n=L)[:2 * n - 1] * dx
    return float(conv.max())

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 5) -> np.ndarray:
    """Fine‑tune using gradient of the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # repair sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h
