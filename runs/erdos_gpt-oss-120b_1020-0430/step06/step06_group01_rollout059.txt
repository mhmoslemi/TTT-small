def softmax_obj_and_grad(h, beta, penalty, target):
    n = h.size
    # compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = beta * corr
    max_s = np.max(s)
    # compute stable exp sum
    exp_shifted = np.exp(s - max_s)
    sum_exp = np.sum(exp_shifted)
    f = (max_s + np.log(sum_exp)) / beta
    # weights
    w = exp_shifted / sum_exp  # shape (2n-1)
    # gradient
    grad = np.zeros_like(h)
    # Positive shifts (s >= 0)
    w_pos = w[n-1:]  # length n
    # term1: grad[:i_end] += w_pos[s] * (1 - h[s:]) for each s
    # We can use loop over s:
    for s_shift, w_val in enumerate(w_pos):
        if w_val == 0.0:
            continue
        i_end = n - s_shift
        if i_end > 0:
            grad[:i_end] += w_val * (1 - h[s_shift:i_end + s_shift])
        # second part: grad[s_shift:i_end + s_shift] -= w_val * h[:i_end]
        if i_end > 0:
            grad[s_shift:i_end + s_shift] -= w_val * h[:i_end]
    # Negative shifts
    w_neg = w[:n-1][::-1]  # length n-1, corresponds to shifts -1...-(n-1)
    # w_neg[t-1] corresponds to shift -t
    for t, w_val in enumerate(w_neg, start=1):
        if w_val == 0.0:
            continue
        # contributions to grad[j] for j >= t: + w_val * (1 - h[j - t])
        # for j = t..n-1
        grad[t:] += w_val * (1 - h[:n - t])
        # contributions to grad[j] for j <= n-1 - t: - w_val * h[j + t]
        grad[:n - t] -= w_val * h[t:]
    # penalty gradient
    if penalty != 0.0:
        sum_h = np.sum(h)
        grad += 2.0 * penalty * (sum_h - target)
        f += penalty * (sum_h - target) ** 2
    return f, grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 500:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            cap = np.sum(1.0 - h[mask])
            if cap <= 0:
                break
            inc = diff * (1.0 - h[mask]) / cap
            inc = np.minimum(inc, 1.0 - h[mask])
            h[mask] += inc
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            cap = np.sum(h[mask])
            dec = (-diff) * h[mask] / cap
            dec = np.minimum(dec, h[mask])
            h[mask] -= dec
        diff = target - h.sum()
        it += 1
    # final tiny fix
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- hill climbing ----------
def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.7,
):
    """Gradient‑directed hill climbing."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.015,
    max_iter: int = 10,
) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- SLSQP with soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(
    h0: np.ndarray,
    beta: float,
    rng: np.random.Generator,
    max_iter: int = 800,
) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary simulated annealing ----------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = np.array(ones, dtype=int)
        zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = ones_arr[rng.integers(len(ones_arr))]
            j = zeros_arr[rng.integers(len(zeros_arr))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                idx_i = np.where(ones_arr == i)[0][0]
                idx_j = np.where(zeros_arr == j)[0][0]
                ones_arr[idx_i] = j
                zeros_arr[idx_j] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

        h_cont, c_cont = continuous_optimize(target_n, rng, remaining, init_h=init_h)
        if c_cont < best_c:
            best_h, best_c, best_n = h_cont.copy(), c_cont, target_n

    # ----- final stochastic tweaking -----
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # ----- sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
