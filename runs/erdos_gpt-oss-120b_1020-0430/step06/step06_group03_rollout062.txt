def max_autocorr_fft(h):
    N = len(h)
    # Compute convolution of h and reversed h
    H = np.fft.rfft(h, n=2*N)
    # reversed h:
    rev = h[::-1]
    G = np.fft.rfft(rev, n=2*N)
    c = np.fft.irfft(H * G, n=2*N)
    # c length 2*N
    # indices N (i.e., N-1+1) through 2*N-2 inclusive contain lags 1..N-1
    start = N
    end = 2*N - 1
    # c may have small floating errors, round
    max_val = int(np.rint(np.max(c[start:end])).astype(int))
    return max_val

def max_autocorr_fft(h):
    N = h.shape[0]
    f = np.fft.rfft(h, n=2*N)
    g = np.fft.rfft(h[::-1], n=2*N)
    c = np.fft.irfft(f * g, n=2*N)
    start = N
    end = 2*N - 1
    return int(np.rint(np.max(c[start:end])).astype(int))

def run(seed=42, budget_s=1100, **kwargs):

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def slsqp_optimize(h0, beta, rng, max_iter=800):
    n = h0.size
    target = n / 2.0
    def soft_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def eq_con(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_con}
    res = minimize(soft_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, target)
