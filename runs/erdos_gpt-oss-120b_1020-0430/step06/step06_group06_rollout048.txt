def project_onto_box_sum(v, target):
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = mid
        else:
            high = mid
    y = np.clip(v - high, 0.0, 1.0)
    # adjust rounding errors:
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # Distribute diff to entries not at bounds
        not_full = (y > 1e-12) & (y < 1 - 1e-12)
        if np.any(not_full):
            y[not_full] += diff / np.count_nonzero(not_full)
        else:
            # fallback, adjust any entry
            y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y
```

def random_perturb(h, rng):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:
            return h
        eps = rng.random() * max_transfer * 0.5
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_transfer * 0.5
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
```

import time
import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto [0,1]^n with sum(v)=target (Euclidean)."""
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = mid
        else:
            high = mid
    # final clipping using the tighter bound
    y = np.clip(v - high, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # distribute the tiny residue over non‑boundary entries
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt:
            y[mask] += diff / cnt
        else:
            y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y


def _compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with N points."""
    n = h.size
    n_fft = 2 * n
    conv = np.fft.ifft(np.fft.fft(h, n_fft) * np.fft.fft(1.0 - h, n_fft)).real
    max_val = np.max(conv[: 2 * n - 1])
    return max_val * (2.0 / n)


def _worst_gradient(h: np.ndarray):
    """Gradient of the worst (largest) overlap term."""
    n = h.size
    n_fft = 2 * n
    conv = np.fft.ifft(np.fft.fft(h, n_fft) * np.fft.fft(1.0 - h, n_fft)).real
    i_max = int(np.argmax(conv[: 2 * n - 1]))
    start = max(0, i_max - (n - 1))
    end = min(n, i_max + 1)
    k = np.arange(start, end)
    t = i_max - k
    g = np.zeros(n)
    g[start:end] = 1.0 - 2.0 * h[t]
    return g, i_max, conv


def _random_perturb(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small amount of mass between two coordinates."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0.0:
            return h
        eps = rng.random() * max_transfer * 0.5
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_transfer * 0.5
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- problem size -------------------------------------------------
    N = 2048                      # number of discretisation points
    target_sum = N / 2.0

    # ---- initial function ---------------------------------------------
    if "initial_h_values" in globals():

    else:

    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_box_sum(h, target_sum)

    best_h = h.copy()
    best_c = _compute_c5(best_h)

    # ---- optimisation parameters ---------------------------------------
    step0 = 0.5
    decay = 0.9999
    random_prob = 0.05
    last_improve = 0
    it = 0
    max_iter = 200000

    # ---- main loop ----------------------------------------------------
    while True:
        now = time.time()
        if now - start > budget_s * 0.95:
            break
        if it >= max_iter:
            break

        if rng.random() < random_prob:
            nh = _random_perturb(h, rng)
        else:
            grad, _, _ = _worst_gradient(h)
            alpha = step0 * (decay ** it)
            v = h - alpha * grad
            nh = _project_onto_box_sum(v, target_sum)

        h = nh

        if it % 100 == 0:
            c = _compute_c5(h)
            if c < best_c - 1e-9:
                best_c = c
                best_h = h.copy()
                last_improve = it

        if it - last_improve > 5000:
            # restart from a fresh random point
            h = rng.random(N)
            h = np.clip(h, 0.0, 1.0)
            h = _project_onto_box_sum(h, target_sum)
            last_improve = it

        it += 1

    # ---- binary refinement --------------------------------------------
    # Threshold to obtain a binary candidate with exact sum
    binary_h = np.zeros_like(best_h)
    k = int(target_sum)
    idx = np.argpartition(best_h, -k)[-k:]
    binary_h[idx] = 1.0
    c_binary = _compute_c5(binary_h)
    if c_binary < best_c:

        nh[i], nh[j] = nh[j], nh[i]   # swap one 1 with one 0

            best_h = nh

    final_c = _compute_c5(best_h)
