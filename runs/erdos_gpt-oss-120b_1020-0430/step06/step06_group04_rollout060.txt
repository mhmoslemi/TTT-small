def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * v
    m = np.max(bc)
    exp_term = np.exp(bc - m)
    w = exp_term / np.sum(exp_term)
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)  # shift k
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i:] -= wi * h[: n - i]
        else:
            ii = -i
            g[ii:] += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii:]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

            delta = min(step_frac2, h2[i], 1.0 - h2[j])
            if delta <= 0:
                continue
            cand = h2.copy()
            cand[i] -= delta
            cand[j] += delta
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand.copy()
            if c < compute_c5(h2):
                h2 = cand

        # binary hill‑climb on up‑sampled version
        h2_bin = np.rint(h2).astype(float)
        adjust_sum_exact(h2_bin, rng)

        cur_h2 = h2_bin.copy()
        cur_c2 = compute_c5(cur_h2)
        if cur_c2 < best_c:
            best_c = cur_c2
            best_h = cur_h2.copy()

        for _ in range(20000):
            if time.time() > deadline:
                break
            ones = np.flatnonzero(cur_h2 == 1)
            zeros = np.flatnonzero(cur_h2 == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h2[i] = 0
            cur_h2[j] = 1
            c = compute_c5(cur_h2)
            if c < cur_c2:
                cur_c2 = c
                if c < best_c:
                    best_c = c
                    best_h = cur_h2.copy()
            else:
                cur_h2[i] = 1
                cur_h2[j] = 0

    # ----- finalise ----------------------------------------------------
    final_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(final_h, rng)
    final_c = compute_c5(final_h)
    return final_h.tolist(), float(final_c), int(final_h.shape[0])
