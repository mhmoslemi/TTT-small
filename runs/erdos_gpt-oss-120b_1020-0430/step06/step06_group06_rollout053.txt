def compute_c5(h):
    n = h.size
    # zero pad to length 2n-1
    N = 2*n-1
    # use next power of 2 for FFT efficiency
    P = 1 << (N-1).bit_length()
    # Create arrays
    a = np.zeros(P)
    b = np.zeros(P)
    a[:n] = h
    b[:n] = 1 - h
    # compute convolution via FFT
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)

    max_val = conv.max()
    dx = 2.0 / n
    return max_val * dx

import time
import numpy as np

# ---- basic operations -------------------------------------------------
def _fft_corr(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1−h) using FFT."""
    n = h.size
    N = 2 * n - 1
    P = 1 << (N - 1).bit_length()          # next power of two
    a = np.zeros(P, dtype=float)
    b = np.zeros(P, dtype=float)
    a[:n] = h
    b[:n] = 1.0 - h
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)
    corr = np.fft.irfft(fa * np.conj(fb), n=P)[:N]
    return corr

def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for a step function on [0,2]."""
    n = h.size
    corr = _fft_corr(h)
    return corr.max() * (2.0 / n)

def compute_corr_max_idx(h: np.ndarray):
    """Return (max_correlation, index_shift) where index_shift = k−(n−1)."""
    n = h.size
    corr = _fft_corr(h)
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx] * (2.0 / n)
    shift_idx = max_idx - (n - 1)
    return max_val, shift_idx

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap Oₖ."""
    n = h.size
    _, idx = compute_corr_max_idx(h)
    k = abs(idx)
    g = np.zeros_like(h)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
    return g

# ---- projection onto box [0,1] with a prescribed sum -------------------
def proj_box_sum(h: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 100) -> np.ndarray:
    """Project onto {0≤x_i≤1, sum x_i = target}."""
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            low = mid
        else:
            high = mid
        if high - low < tol:
            break
    lam = (low + high) * 0.5
    return np.clip(h + lam, 0.0, 1.0)

def _sa_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    n = h.size

    if i == j:
        return h.copy()
    if rng.random() < 0.2:          # plain swap
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    # transfer mass from i to j

    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])

            return h.copy()
        eps = rng.random() * max_eps * scale

        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps

    while time.time() < deadline:
        cand = _sa_move(cur, rng, scale)
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur, cur_c = cand, cand_c
            if cand_c < best_c:
                best, best_c = cand.copy(), cand_c
        it += 1
        if it % 1000 == 0:
            T = max(T * 0.995, 1e-12)
            scale = max(scale * 0.997, 1e-6)
    return best, best_c

# ---- sub‑gradient descent ---------------------------------------------
def subgradient_descent(h0: np.ndarray, target: float, deadline: float,
                        rng: np.random.Generator):
    h = h0.copy()
    best = h.copy()
    best_c = compute_c5(h)
    step0 = 0.8
    it = 0
    while time.time() < deadline:
        it += 1
        step = step0 / np.sqrt(it)
        g = worst_gradient(h)
        h_new = proj_box_sum(h - step * g, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best, best_c = h_new.copy(), c_new
        h = h_new
    return best, best_c

# ---- local random search (tiny tweaks) -------------------------------
def local_random_search(h0: np.ndarray, target: float, deadline: float,
                        rng: np.random.Generator):
    best = h0.copy()
    best_c = compute_c5(best)
    n = best.size
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best[i], 1.0 - best[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.1
        nh = best.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        if c_new < best_c:
            best, best_c = nh, c_new
    return best, best_c

# ---- main entry point -------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    n = 1024
    target = n / 2.0

    # ---- initialise ---------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        h0 = np.repeat(base, 2)
        if h0.size > n:
            h0 = h0[:n]
        elif h0.size < n:
            h0 = np.concatenate([h0, np.zeros(n - h0.size)])
        h0 = proj_box_sum(h0, target)
    else:
        h0 = rng.random(n)
        h0 = proj_box_sum(h0, target)

    # ---- time slices --------------------------------------------------
    SA_deadline = start + total * 0.45
    SD_deadline = start + total * 0.90
    RS_deadline = start + total * 0.99

    # ---- simulated annealing -----------------------------------------
    best_h, best_c = simulated_annealing(h0, target, SA_deadline, rng)

    # ---- sub‑gradient descent -----------------------------------------
    best_h, best_c = subgradient_descent(best_h, target, SD_deadline, rng)

    # ---- local random tweaks -----------------------------------------
    if time.time() < RS_deadline:
        best_h, best_c = local_random_search(best_h, target, RS_deadline, rng)

    # ---- final projection & evaluation --------------------------------
    best_h = proj_box_sum(best_h, target)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
