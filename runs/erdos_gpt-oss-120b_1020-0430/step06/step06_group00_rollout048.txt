def ga_optimize(n, rng, time_budget):
    start = time.time()
    popsize = 80
    # target sum = n/2
    target = n/2

    # helper to generate random feasible
    def random_feasible():
        h = np.zeros(n, dtype=float)
        # assign 1 to target positions
        ones = rng.choice(n, int(target), replace=False)
        h[ones] = 1.0
        # optionally perturb a few positions
        # choose m pairs to exchange epsilon
        m_pairs = rng.integers(0, n//10+1)  # up to 10% pairs
        for _ in range(m_pairs):
            i, j = rng.choice(n, 2, replace=False)
            # move epsilon from i to j (if possible)
            eps = rng.random() * 0.3 # up to 0.3
            if h[i] >= eps and h[j] <= 1-eps:
                h[i] -= eps
                h[j] += eps
        return h

    population = [random_feasible() for _ in range(popsize)]
    scores = np.array([compute_c5(ind) for ind in population])

    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # GA loop
    while time.time() - start < time_budget:
        # parent selection via tournament
        parents = []
        for _ in range(popsize):
            i1, i2 = rng.integers(0, popsize, size=2)
            if scores[i1] < scores[i2]:
                parents.append(population[i1])
            else:
                parents.append(population[i2])

        # produce offspring
        offspring = []
        for i in range(0, popsize, 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < popsize else parents[0]
            # crossover (blend)
            alpha = rng.random()
            child1 = (1-alpha) * p1 + alpha * p2
            child2 = alpha * p1 + (1-alpha) * p2
            # mutation
            child1 = mutate(child1, rng)
            child2 = mutate(child2, rng)
            # repair
            child1 = np.clip(child1, 0, 1)
            child1 = enforce_sum(child1, rng)
            child2 = np.clip(child2, 0, 1)
            child2 = enforce_sum(child2, rng)
            offspring.append(child1)
            offspring.append(child2)

        # evaluate offspring
        offspring_scores = np.array([compute_c5(ind) for ind in offspring])

        # select next generation: combine and keep best popsize
        combined = population + offspring
        combined_scores = np.concatenate([scores, offspring_scores])
        # argsort
        idx = np.argsort(combined_scores)
        population = [combined[i] for i in idx[:popsize]]
        scores = combined_scores[idx[:popsize]]

        # update best
        if scores[0] < best_score:
            best_score = scores[0]
            best_h = population[0].copy()

    return best_h, best_score

def mutate(h, rng, pm=0.2):
    # pm fraction of positions to be selected for potential transfer
    n = h.size
    num = int(pm * n)
    if num < 1:
        return h
    # select pairs
    idx = rng.choice(n, size=2*num, replace=False)
    for k in range(num):
        i = idx[2*k]
        j = idx[2*k+1]
        # transfer random epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5  # limited to 0.5
        h[i] -= eps
        h[j] += eps
    return h

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        # select two distinct indices i, j
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        # decide direction
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
        else:
            max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * scale
        if eps <= 0:
            continue
        h_new = h.copy()
        if rng.random() < 0.5:
            # transfer from i to j
            h_new[i] -= eps
            h_new[j] += eps
        else:
            # transfer from j to i
            h_new[j] -= eps
            h_new[i] += eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ------------------------------------------------------------
# Gradient of worst shift and refinement
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = n / 2.0 - h.sum()
    return h

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # choose direction of transfer
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0:
                continue
            h_new = h.copy()
            h_new[j] -= eps
            h_new[i] += eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

# Genetic algorithm (real‑valued)
def ga_optimize(n: int, rng: np.random.Generator, time_budget: float):
    """Simple GA for minimizing C5 at resolution n."""
    start = time.time()
    target = n / 2.0
    popsize = 80

    def random_feasible():
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, int(target), replace=False)
        h[ones] = 1.0
        # small perturbations via pairwise transfers
        pairs = rng.integers(0, n, size=2 * (n // 20))
        for k in range(0, len(pairs), 2):
            i, j = pairs[k], pairs[k + 1]
            if h[i] > 0 and h[j] < 1:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = rng.random() * max_eps * 0.5
                h[i] -= eps
                h[j] += eps
        return h

    population = [random_feasible() for _ in range(popsize)]
    scores = np.array([compute_c5(ind) for ind in population])

    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    def mutate_preserve_sum(h):
        """Pairwise mass transfers preserving the total sum."""
        num = max(1, int(0.3 * n))
        idx = rng.choice(n, size=2 * num, replace=False)
        for k in range(num):
            i, j = idx[2 * k], idx[2 * k + 1]
            if h[i] > 0 and h[j] < 1:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = rng.random() * max_eps * 0.5
                h[i] -= eps
                h[j] += eps
            elif h[j] > 0 and h[i] < 1:
                max_eps = min(h[j], 1.0 - h[i])
                if max_eps <= 0:
                    continue
                eps = rng.random() * max_eps * 0.5
                h[j] -= eps
                h[i] += eps
        return h

    while time.time() - start < time_budget:
        # tournament selection
        parents = []
        for _ in range(popsize):
            i1, i2 = rng.integers(0, popsize, size=2)
            parents.append(population[i1] if scores[i1] < scores[i2] else population[i2])

        offspring = []
        for i in range(0, popsize, 2):
            p1 = parents[i]
            p2 = parents[i + 1] if i + 1 < popsize else parents[0]
            alpha = rng.random()
            child1 = (1 - alpha) * p1 + alpha * p2
            child2 = alpha * p1 + (1 - alpha) * p2
            child1 = mutate_preserve_sum(child1)
            child2 = mutate_preserve_sum(child2)
            child1 = np.clip(child1, 0.0, 1.0)
            child2 = np.clip(child2, 0.0, 1.0)
            child1 = enforce_sum(child1, rng)
            child2 = enforce_sum(child2, rng)
            offspring.append(child1)
            offspring.append(child2)

        offspring_scores = np.array([compute_c5(ind) for ind in offspring])
        combined = population + offspring
        combined_scores = np.concatenate([scores, offspring_scores])
        idx = np.argsort(combined_scores)[:popsize]
        population = [combined[i] for i in idx]
        scores = combined_scores[idx]

        if scores[0] < best_score:
            best_score = scores[0]
            best_h = population[0].copy()

    return best_h, best_score

# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # initial construction, if supplied
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # Phase 1: GA on coarse grid (256)
    remaining = budget_s - (time.time() - start)
    if remaining > 200:
        t_phase = min(remaining * 0.30, 250.0)
        h_ga, c_ga = ga_optimize(256, rng, t_phase)
        if c_ga < best_c:
            best_h, best_c, best_n = h_ga, c_ga, 256

    # Phase 2: Upsample to medium resolution (512)
    remaining = budget_s - (time.time() - start)
    if remaining > 100 and best_h is not None:
        if best_n < 512:
            h_mid = upsample_h(best_h, 512)
        else:
            h_mid = best_h.copy()
        h_mid = np.clip(h_mid, 0.0, 1.0)
        h_mid = enforce_sum(h_mid, rng)
        c_mid = compute_c5(h_mid)
        if c_mid < best_c:
            best_h, best_c, best_n = h_mid, c_mid, 512

    # Phase 3: Gradient refinement on 512
    remaining = budget_s - (time.time() - start)
    if remaining > 50 and best_h is not None and best_n == 512:
        h_ref = gradient_refine(best_h, rng, step=0.015, max_iter=15)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # Phase 4: Upsample to fine resolution (1024)
    remaining = budget_s - (time.time() - start)
    if remaining > 50 and best_h is not None and best_n < 1024:
        h_fine = upsample_h(best_h, 1024)
        h_fine = np.clip(h_fine, 0.0, 1.0)
        h_fine = enforce_sum(h_fine, rng)
        c_fine = compute_c5(h_fine)
        if c_fine < best_c:
            best_h, best_c, best_n = h_fine, c_fine, 1024

    # Phase 5: Continuous SA on the finest grid
    remaining = budget_s - (time.time() - start)
    if remaining > 20 and best_h is not None:
        h_sa, c_sa = continuous_sa(best_h, rng, remaining)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
