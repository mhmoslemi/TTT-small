def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    # k can be negative or positive
    grad = np.zeros_like(h)
    if k >= 0:
        # source terms i from 0 to n - k -1: gradient +1 - h[i+k]
        grad[: n - k] += 1.0 - h[k:]
        # target terms i from k to n-1: gradient -h[i - k]
        grad[k:] -= h[: n - k]
    else:
        # k negative: let p = -k > 0
        p = -k
        # source terms i from p to n-1: i+k >=0, i+k = i - p
        grad[p:] += 1.0 - h[: n - p]
        # target terms i from 0 to n - p -1: i is target of some j i+k = i - p; j = i + p
        grad[: n - p] -= h[p:]

def binary_sa(n, rng, time_budget):
    start = time.time()
    best_h = np.zeros(n, dtype=np.float64)
    best_c = np.inf

    # initial random binary vector
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5_fft(h)
    # track ones and zeros as arrays for O(1) pick
    ones_arr = np.array(ones, dtype=int)
    zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)

    # schedule: geometric temperature decreasing from T0 to T_end
    T0 = 1e-2
    T_end = 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end/T0) ** (elapsed / time_budget)
        # propose swap
        i_idx = rng.integers(len(ones_arr))
        j_idx = rng.integers(len(zeros_arr))
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]
        # swap in h
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            c_curr = c_new
            # update arrays: replace i with j in ones_arr, j with i in zeros_arr
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                # early exit if good enough
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Tolerance threshold
    while abs(diff) > 1e-12:
        # Find indices that can accept positive or negative adjustment
        if diff > 0:
            # need to add mass; choose i where h[i] < 1
            candidates = np.where(h < 1.0)[0]

import time
import numpy as np

# ---------- constraint handling ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- FFT based correlation ----------
def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1‑h) using FFT (linear correlation)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, N)
    B = np.fft.rfft((1.0 - h)[::-1], N)   # reversed second argument
    corr = np.fft.irfft(H * B, N)[: 2 * n - 1]
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for a step function on [0,2]."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


# ---------- worst‑shift gradient ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    k = (n - 1) - idx          # shift giving the maximal correlation
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:]      -= h[: n - k]
    else:  # k < 0
        p = -k
        grad[p:]      += 1.0 - h[: n - p]
        grad[: n - p] -= h[p:]
    return grad


# ---------- binary simulated annealing ----------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    # random binary start with exactly n/2 ones
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)

    # maintain index lists for O(1) random swaps
    ones_arr = np.array(ones, dtype=int)
    zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)

    T0, T_end = 0.08, 1e-5
    best_h, best_c = h.copy(), c_curr

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        i_idx = rng.integers(len(ones_arr))
        j_idx = rng.integers(len(zeros_arr))
        i, j = ones_arr[i_idx], zeros_arr[j_idx]

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / max(T, 1e-12))

        if accept:
            c_curr = c_new
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c


# ---------- projected gradient descent ----------
def projected_gradient_descent(h0: np.ndarray, rng: np.random.Generator,
                              time_budget: float):
    start = time.time()
    cur_h, cur_c = h0.copy(), compute_c5(h0)
    best_h, best_c = cur_h.copy(), cur_c

    while time.time() - start < time_budget:
        grad = worst_gradient(cur_h)
        # linear step‑size decay
        elapsed = time.time() - start
        step = 0.6 * (1.0 - elapsed / time_budget) + 1e-5

        new_h = cur_h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, rng)
        new_c = compute_c5(new_h)

        # greedy accept, occasional random move to escape flat regions
        if new_c < cur_c or rng.random() < 0.001:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target resolution (larger n gives finer discretisation)
    n = 1024

    # ---- binary SA (≈45 % of time) ----
    time_sa = total_budget * 0.45
    best_h, best_c = binary_sa(n, rng, time_sa)

    # ---- projected gradient descent with remaining time ----
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 10.0:
        # use majority of what is left, keep a small reserve for final tweaks
        grad_time = remaining * 0.75
        best_h, best_c = projected_gradient_descent(best_h, rng, grad_time)
        elapsed = time.time() - start
        remaining = total_budget - elapsed

    # ---- final stochastic tweaking (≈5 % of total) ----
    while time.time() - start < total_budget * 0.99:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nh = enforce_sum(nh, rng)
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc
                if best_c <= 0.3808:
                    break

    # ---- sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
