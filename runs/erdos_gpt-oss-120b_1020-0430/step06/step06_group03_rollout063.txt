def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    L = 2*n - 1
    idx = np.arange(n, dtype=np.int32)

    # initialize H
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        H = (h0 > 0.5).astype(np.int8)
        cur = int(H.sum())
        target = n // 2
        if cur > target:
            ones = np.where(H == 1)[0]
            to_flip = rng.choice(ones, cur - target, replace=False)
            H[to_flip] = 0
        elif cur < target:
            zeros = np.where(H == 0)[0]
            to_flip = rng.choice(zeros, target - cur, replace=False)
            H[to_flip] = 1
    else:
        H = np.zeros(n, dtype=np.int8)
        H[rng.choice(n, n//2, replace=False)] = 1

    # precompute structures
    padded = np.concatenate([np.zeros(n-1, dtype=np.int8), H, np.zeros(n-1, dtype=np.int8)])
    C = np.correlate(H, H, mode='full').astype(np.int16)

    current_Omax = compute_Omax(H, C, n, idx)

    best_Omax = current_Omax
    best_H = H.copy()

    # annealing parameters
    max_iter = 2000000  # can be high
    T0 = 5.0
    it = 0
    last_improve = 0
    # local copies to avoid reallocation? We'll allocate tmp each loop
    tmp = np.empty(L, dtype=np.int16)

    while True:
        now = time.time()
        if now - start > budget_s * 0.96:
            break
        if it >= max_iter:
            break

        # choose swap
        i = rng.choice(np.where(H == 1)[0])  # we could compute ones indices each time; not efficient but ok for small n
        j = rng.choice(np.where(H == 0)[0])

        # compute delta C
        tmp[:] = padded[j:j+L] - padded[i:i+L]
        tmp[n-1] = 0  # ensure zero shift unchanged

        # candidate C
        C_cand = C + tmp

        # candidate H copy
        H_cand = H.copy()
        H_cand[i] = 0
        H_cand[j] = 1

        # compute Omax candidate
        Omax_cand = compute_Omax(H_cand, C_cand, n, idx)

        # temperature
        T = T0 * (1 - it / max_iter) + 1e-12

        if Omax_cand <= current_Omax or rng.random() < np.exp((current_Omax - Omax_cand) / T):
            # accept
            H = H_cand
            C = C_cand
            # update padded
            padded[i + n - 1] = 0
            padded[j + n - 1] = 1
            current_Omax = Omax_cand
            if Omax_cand < best_Omax:
                best_Omax = Omax_cand
                best_H = H.copy()
                last_improve = it
        it += 1

        # stagnation restart
        if it - last_improve > max_iter // 10:
            # restart random
            H = np.zeros(n, dtype=np.int8)
            H[rng.choice(n, n//2, replace=False)] = 1

            C = np.correlate(H, H, mode='full').astype(np.int16)
            current_Omax = compute_Omax(H, C, n, idx)
            last_improve = it

    current_Omax = max_overlap(H, C, prefix_buf)

    best_Omax = current_Omax
    best_H = H.copy()

    # lists for fast random swap selection
    ones_list = [i for i, v in enumerate(H) if v]
    zeros_list = [i for i, v in enumerate(H) if not v]

    # SA parameters
    max_iter = 2_000_000
    T0 = 5.0
    it = 0
    last_improve = 0
    tmp = np.empty(L, dtype=np.int16)

    while True:
        now = time.time()
        if now - start > budget_s * 0.96 or it >= max_iter:
            break

        # pick a random 1 and a random 0
        i = rng.choice(ones_list)
        j = rng.choice(zeros_list)

        # delta C for this swap
        tmp[:] = padded[j:j + L] - padded[i:i + L]
        tmp[n - 1] = 0  # keep the zero‑shift term unchanged
        C_cand = C + tmp

        # candidate H
        H_cand = H.copy()
        H_cand[i] = 0
        H_cand[j] = 1

        # compute Omax for candidate
        Omax_cand = max_overlap(H_cand, C_cand, prefix_buf)

        # temperature
        T = T0 * (1.0 - it / max_iter) + 1e-12

        # accept if better or with Metropolis probability
        if Omax_cand <= current_Omax or rng.random() < np.exp((current_Omax - Omax_cand) / T):
            # accept the move
            H = H_cand
            C = C_cand
            padded[i + n - 1] = 0
            padded[j + n - 1] = 1
            current_Omax = Omax_cand

            # update the selection lists
            ones_list.remove(i)
            zeros_list.append(i)
            zeros_list.remove(j)
            ones_list.append(j)

            if Omax_cand < best_Omax:
                best_Omax = Omax_cand
                best_H = H.copy()
                last_improve = it

        it += 1

        # restart if no improvement for a while
        if it - last_improve > max_iter // 10:
            H = np.zeros(n, dtype=np.int8)
            H[rng.choice(n, n // 2, replace=False)] = 1
            padded = np.concatenate([np.zeros(n - 1, dtype=np.int8), H,
                                     np.zeros(n - 1, dtype=np.int8)])
            C = np.correlate(H, H, mode='full').astype(np.int16)
            current_Omax = max_overlap(H, C, prefix_buf)
            ones_list = [i for i, v in enumerate(H) if v]
            zeros_list = [i for i, v in enumerate(H) if not v]
            last_improve = it

    # final evaluation on the best solution found
    C_best = np.correlate(best_H, best_H, mode='full').astype(np.int16)
    final_Omax = max_overlap(best_H, C_best, prefix_buf)
    c5_bound = float(final_Omax * 2.0 / n)

    return best_H.astype(float).tolist(), c5_bound, n
