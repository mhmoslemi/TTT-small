def compute_c5_fft(h):
    n = h.size
    # zero pad to length 2n
    size = 1 << ((2*n - 1).bit_length())
    # compute FFT of h and of (1-h)
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1 - h, n=size)
    corr = np.fft.irfft(H * G, n=size)[:2*n-1]  # linear convolution length 2n-1
    # max correlation
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def worst_shift_gradient(h):
    n = h.shape[0]
    k, _ = worst_shift_k(h)
    grad = np.zeros(n)
    if k >= 0:
        # positive shift
        n_k = n - k
        grad[: n_k] += (1.0 - h[k:])  # derivative w.r.t. h_i (i from 0..n_k-1)
        grad[k:] -= h[: n_k]          # derivative w.r.t. h_{i} for i from k..n-1
    else:
        k = -k
        n_k = n - k
        grad[k:] += (1.0 - h[: n_k])
        grad[: n_k] -= h[k:]
    # scale by 2/n
    return grad * (2.0 / n)

def binary_sa_optimize(h, rng, time_budget):
    start = time.time()
    n = h.size
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    # temperature schedule
    T0 = 0.03  # initial temperature
    T_end = 1e-5
    # total steps maximum guess, not needed
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        # linear schedule: fraction = elapsed / time_budget
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + T_end * frac
        # pick i (1) and j (0)
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                # early exit if below target
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def gradient_transfer_optimize(h, rng, time_budget):
    start = time.time()
    n = h.size
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    # initial step factor
    step_factor = 0.9
    # temperature schedule for accept worsening maybe
    T0 = 0.01
    T_end = 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + T_end * frac
        grad = worst_shift_gradient(h)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if i_max == i_min:
            # no improvement direction, break
            break
        max_transfer = min(h[i_min], 1 - h[i_max])
        if max_transfer <= 1e-12:
            # cannot transfer, maybe random perturb
            # small random perturb
            i = rng.integers(n)
            j = rng.integers(n)
            eps = min(h[i], 1 - h[j]) * rng.random() * 0.01
            if eps > 0:
                h[i] -= eps
                h[j] += eps
                c_curr = compute_c5_fft(h)
            continue
        eps = max_transfer * (rng.random() * 0.5 + 0.5) * step_factor

import time
import numpy as np

# ----------- core utilities -------------------------------------------------
def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT based linear correlation (h * (1-h))."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * G, n=size)[: 2 * n - 1]
    return float(corr.max() * (2.0 / n))


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (scaled by dx = 2/n)."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * G, n=size)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # shift amount (negative allowed)
    grad = np.zeros(n, dtype=float)
    if k >= 0:
        nk = n - k
        grad[:nk] += (1.0 - h[k:])      # ∂/∂h_i where i pairs with i+k
        grad[k:]  -= h[:nk]             # ∂/∂h_{i+k}
    else:
        k = -k
        nk = n - k
        grad[k:]  += (1.0 - h[:nk])
        grad[:nk] -= h[k:]
    return grad * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, target: float = None) -> np.ndarray:
    """Adjust h so that sum(h) = target (default n/2) while staying in [0,1]."""
    if target is None:
        target = h.size / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(h.size)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


# ----------- binary simulated annealing ------------------------------------
def binary_sa_optimize(h: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Binary SA (swap 1↔0) on h for the given time budget."""
    n = h.size
    start = time.time()
    c_curr = compute_c5_fft(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.025, 5e-5
    while time.time() - start < time_budget:
        # temperature linearly decays
        frac = (time.time() - start) / time_budget
        T = T0 * (1 - frac) + T_end * frac

        # pick random 1 and 0 positions
        i = rng.integers(n)
        while h[i] == 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1.0:
            j = rng.integers(n)

        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# ----------- gradient‑based transfer refinement ----------------------------
def gradient_transfer_optimize(h: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Transfer mass from negative‑gradient to positive‑gradient indices."""
    n = h.size
    start = time.time()
    c_curr = compute_c5_fft(h)
    best_h, best_c = h.copy(), c_curr
    step_factor = 0.9
    T0, T_end = 0.012, 2e-6
    while time.time() - start < time_budget:
        frac = (time.time() - start) / time_budget
        T = T0 * (1 - frac) + T_end * frac

        grad = worst_shift_gradient(h)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if i_max == i_min:
            break
        max_transfer = min(h[i_min], 1.0 - h[i_max])
        if max_transfer <= 1e-14:
            # tiny random perturbation to escape flat spots
            i = rng.integers(n)
            j = rng.integers(n)
            eps = min(h[i], 1.0 - h[j]) * rng.random() * 0.005
            if eps > 0:
                h[i] -= eps
                h[j] += eps
                c_curr = compute_c5_fft(h)
            continue

        eps = max_transfer * (rng.random() * 0.5 + 0.5) * step_factor
        h_new = h.copy()
        h_new[i_max] += eps
        h_new[i_min] -= eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    break
        if frac > 0.5:
            step_factor = max(step_factor * 0.999, 0.01)
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024                      # number of points (≤ 1 k as required)

    # ----- initial binary vector -----------------------------------------
    if 'initial_h_values' in globals():
        init_vals = np.array(initial_h_values, dtype=float)
        # interpolate to target size
        x_old = np.linspace(0.0, 2.0, init_vals.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_new, x_old, init_vals)
        idx = np.argsort(interp)[-n // 2:]
        h0 = np.zeros(n, dtype=float)
        h0[idx] = 1.0
    else:
        h0 = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h0[ones] = 1.0

    # ----- binary SA (≈60 % of budget) ------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    binary_budget = remaining * 0.60
    h_best, c_best = binary_sa_optimize(h0.copy(), rng, binary_budget)

    # ----- gradient‑based refinement (rest of budget) --------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 1.0:
        h_best, c_best = gradient_transfer_optimize(h_best.copy(), rng, remaining)

    # ----- final sanitisation ---------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = enforce_sum(h_best, rng)
