import time
import numpy as np
from scipy.optimize import minimize

# ---------- global placeholders for SLSQP ----------
GLOBAL_TARGET = None
GLOBAL_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - GLOBAL_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = GLOBAL_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / GLOBAL_BETA

# ---------- core utilities ----------
def compute_c5(h):
    """max overlap integral for discrete h (full correlation)."""
    n = h.size
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(v, target, eps=1e-12):
    """Project v onto [0,1] with given sum target using bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_smoothmax(h, beta):
    """Gradient of (1/β) log-sum-exp of overlaps."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    exp_shift = np.exp(beta * v - np.max(beta * v))
    w = exp_shift / exp_shift.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k:]       -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:]      += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

def worst_shift_gradient(h):
    """Subgradient based on currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:]       -= h[: n - k]
        else:
            kk = -k
            g[kk:]      += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def smoothmax_descent(h, betas, steps_per_beta, base_step, target, rng, deadline):
    """Projected gradient descent on smooth‑max surrogate."""
    for bi, beta in enumerate(betas):
        steps = steps_per_beta[bi] if isinstance(steps_per_beta, list) else steps_per_beta
        step_scale = base_step / np.sqrt(beta)
        for _ in range(steps):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                break
            best = h
            best_c = compute_c5(h)
            for fac in (1.0, 0.5, 0.25):
                cand = h - step_scale * fac * g
                cand = project_sum(cand, target)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best = cand
            h = best
    return h

def worst_shift_refine(h, target, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best = h
        best_c = compute_c5(h)
        for fac in (1.0, 0.5, 0.25, 0.125):
            cand = h - step * fac * g
            cand = project_sum(cand, target)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best = cand
        h = best
        step *= 0.975
    return h

def slsqp_optimize(h, beta, target, rng, max_iter, deadline):
    global GLOBAL_TARGET, GLOBAL_BETA
    GLOBAL_TARGET = target
    GLOBAL_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # ensure exact sum using projection (fast for final fix)
    x = project_sum(x, target)
    return x

def adjust_binary_sum(vec, target, rng):
    """Flip bits to achieve exact integer sum target."""
    cur = int(round(vec.sum()))
    diff = target - cur
    if diff == 0:
        return vec
    zeros = np.flatnonzero(vec == 0)
    ones = np.flatnonzero(vec == 1)
    if diff > 0:
        choose = rng.choice(zeros, size=diff, replace=False)
        vec[choose] = 1.0
    else:
        choose = rng.choice(ones, size=-diff, replace=False)
        vec[choose] = 0.0
    return vec

def binary_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    T0 = max(1e-5, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0.0
        cur[j] = 1.0
        c = compute_c5(cur)
        T = T0 * (0.9999 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if c < best_c:
                best_c = c
                best = cur.copy()
                it = 0
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        it += 1
        if it > 25000:
            break
    return best, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values", None)
    base_n = len(parent) if parent is not None else 720

    # ensure target is integer (n even)
    target = base_n // 2

    # parameters
    betas = [5, 15, 30, 80, 200, 500, 1500, 4000, 10000, 25000, 50000]
    steps_per_beta = [30] * (len(betas) - 1) + [15]

    restarts = 8
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent is not None:
            h = np.array(parent, copy=True, dtype=np.float64)
            h = project_sum(h, target)
        else:
            h = rng.random(n)
            h = project_sum(h, target)

        # smooth‑max descent
        h = smoothmax_descent(h, betas, steps_per_beta, base_step=0.4,
                              target=target, rng=rng, deadline=deadline)

        # worst‑shift refinement
        h = worst_shift_refine(h, target, rng, base_step=0.15,
                               iters=300, deadline=deadline)

        # final SLSQP polish (high β)
        for beta in (2000, 5000, 10000, 20000, 50000):
            if time.time() > deadline:

            h = slsqp_optimize(h, beta, target, rng,
                               max_iter=1500, deadline=deadline)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # binary rounding + hill‑climb
        h_bin = np.rint(h).astype(np.float64)
        h_bin = adjust_binary_sum(h_bin, target, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_bin, c_bin = binary_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        target_up = n_up // 2
        h_up = np.repeat(best_h, 2)
        h_up = project_sum(h_up, target_up)

        # short smooth‑max descent on upsampled version
        h_up = smoothmax_descent(h_up, betas[:6],
                                 steps_per_beta[:6], base_step=0.25,
                                 target=target_up, rng=rng, deadline=deadline)
        h_up = worst_shift_refine(h_up, target_up, rng,
                                  base_step=0.12, iters=200, deadline=deadline)
        for beta in (2000, 5000, 10000, 20000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, target_up, rng,
                                  max_iter=1200, deadline=deadline)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

        # binary hill‑climb on upsampled solution
        h_bin = np.rint(h_up).astype(np.float64)
        h_bin = adjust_binary_sum(h_bin, target_up, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.15)
        h_bin, c_bin = binary_hill_climb(h_bin, rng, sub_dead)

    # final safety fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n, dtype=np.float64)
        best_h[: n // 2] = 1.0
        best_h = project_sum(best_h, n // 2)
        best_c = compute_c5(best_h)

    # final projection + evaluation
    best_h = project_sum(best_h, best_n // 2)
