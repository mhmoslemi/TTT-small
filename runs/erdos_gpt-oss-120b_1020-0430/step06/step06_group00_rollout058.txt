def worst_shift_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad

def worst_gradient_descent(h, rng, step0=0.02, iters=2000, decay=0.999):
    n = h.size
    target = n / 2.0
    step = step0
    for _ in range(iters):
        # compute gradient for worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for __ in range(5):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = target - h.sum()
        step *= decay
    return h

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr

import time, numpy as np
from scipy.optimize import minimize

# ---------- global helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None
N_GLOBAL = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h

def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

# ---------- binary simulated annealing ----------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

# ---------- worst‑shift gradient descent ----------
def worst_gradient_descent(h: np.ndarray, rng: np.random.Generator,
                          step0: float = 0.02, iters: int = 2000,
                          decay: float = 0.999):
    n = h.size
    target = n / 2.0
    step = step0
    for _ in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for __ in range(5):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = target - h.sum()
        step *= decay
    return h

# ---------- block‑wise SLSQP (periodic) ----------
def block_slsqp_optimize(b0: np.ndarray, L: int, beta: float,
                         rng: np.random.Generator, max_iter: int = 800):
    global CURRENT_TARGET, CURRENT_BETA, N_GLOBAL
    CURRENT_TARGET = L / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * L
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    N = N_GLOBAL
    rep = N // L

    def obj(b):
        h = np.tile(b, rep)
        corr = np.correlate(h, 1.0 - h, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    res = minimize(obj, b0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    b = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - b.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(L)
            add = min(diff, 1.0 - b[i]) if diff > 0 else max(diff, -b[i])
            b[i] += add
            diff = CURRENT_TARGET - b.sum()
    return b

# ---------- continuous simulated annealing ----------
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c

# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    N = 1024
    global N_GLOBAL
    N_GLOBAL = N

    best_h = None
    best_c = np.inf

    # initial guess (if provided)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        if h0.size != N:
            h0 = upsample_h(h0, N)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c = h0, c0

    # Phase 1: binary SA on a coarse grid
    elapsed = time.time() - start
    if budget_s - elapsed > 60:
        t_bin = min((budget_s - elapsed) * 0.2, 250.0)
        h_bin, c_bin = binary_sa(512, rng, t_bin)
        if h_bin.size != N:
            h_bin = upsample_h(h_bin, N)
            h_bin = np.clip(h_bin, 0.0, 1.0)
            h_bin = enforce_sum(h_bin, rng)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin

    # Phase 2: periodic SLSQP on several block sizes
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 30:
        for L in (32, 64, 128):
            if N % L != 0:
                continue
            rep = N // L
            # start block from best_h if available
            if best_h is not None:
                b_start = best_h[:L].copy()
                b_start = enforce_sum(b_start, rng)
            else:
                b_start = None
            restarts = 0
            while True:
                elapsed = time.time() - start
                if budget_s - elapsed < 5:
                    break
                if restarts >= 20:
                    break
                if b_start is not None and restarts == 0:
                    b = b_start.copy()
                else:
                    b = rng.random(L)
                    b = enforce_sum(b, rng)
                for beta in (20., 80., 200., 800., 3000., 8000., 20000., 50000.):
                    if time.time() - start > budget_s * 0.95:
                        break
                    max_it = 800 if beta < 2000 else 1200
                    b = block_slsqp_optimize(b, L, beta, rng, max_it)
                    h_cand = np.tile(b, rep)
                    c_cand = compute_c5(h_cand)
                    if c_cand < best_c:
                        best_h, best_c = h_cand.copy(), c_cand
                # optional local refinement of the current best
                if time.time() - start < budget_s * 0.9 and best_h is not None:
                    h_ref = worst_gradient_descent(best_h, rng,
                                                   step0=0.02, iters=2000, decay=0.999)
                    c_ref = compute_c5(h_ref)
                    if c_ref < best_c:
                        best_h, best_c = h_ref, c_ref
                restarts += 1

    # Phase 3: continuous SA on the finest grid
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 10 and best_h is not None:
        h_fin, c_fin = continuous_sa(best_h, rng, remaining)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    # final sanitisation
    if best_h is None:
        best_h = np.zeros(N, dtype=float)
        best_h[: N // 2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
