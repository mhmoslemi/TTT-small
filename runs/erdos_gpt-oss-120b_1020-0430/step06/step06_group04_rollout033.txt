def project_to_bounds_and_sum(v, target, eps=1e-12):
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    # final projection
    proj = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # Might have slight sum error, adjust via small diff distributing to entries not at bounds.
    diff = target - proj.sum()
    if abs(diff) > eps:
        # distribute diff uniformly among non-bound entries
        free = np.where((proj > 0.0 + eps) & (proj < 1.0 - eps))[0]
        if len(free) > 0:
            proj[free] += diff / len(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_to_bounds_and_sum(v, target, eps=1e-12):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > eps:
        idxs = np.where((proj > eps) & (proj < 1 - eps))[0]
        if idxs.size == 0:
            idxs = np.arange(proj.size)
        i = idxs[0]
        proj[i] = np.clip(proj[i] + diff, 0.0, 1.0)
    return proj

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    offset = n - 1
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    offset = n - 1
    for idx in idxs:
        k = idx - offset
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def worst_shift_refine(h, rng, base_step, iters, deadline):
    n = h.size
    step = base_step
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        cand = h - step * g
        cand = project_to_bounds_and_sum(cand, n/2.0)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        else:
            step *= 0.95
    return best_h

def adam_optimize(h0, betas, steps_per_beta, lr_start, rng, deadline):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    for beta in betas:
        lr = lr_start / (np.sqrt(beta) + 1.0)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_h
            g = gradient_smoothmax(h, beta)
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            cand = h - step
            cand = project_to_bounds_and_sum(cand, target)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand.copy()
            h = cand
    return best_h

def binary_local_search(h_bin, rng, deadline, max_iters=2000):
    cur_h = h_bin.copy()

                           time.time() + (deadline - time.time()) * 0.07)
            h_bin, c_bin = binary_local_search(h_bin, rng,
                                               sub_dead, max_iters=1200)
            if c_bin < best_c:
                best_c, best_h, best_n = c_bin, h_bin.copy(), n_up2

    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_to_bounds_and_sum(best_h, n / 2.0)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = project_to_bounds_and_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
