def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = idx  # signed shift
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1 - 2 * h
    elif k > 0:
        grad[: n - k] = -h[k:]  # derivative wrt h[i] for i< n-k: -h[i+k]
        grad[k:] += -h[: n - k]  # derivative wrt h[i] for i>=k: -h[i-k]
        # also derivative from sum_i h[i]*(1) part: +1 for all i (since sum_i h[i] term)
        grad += 1.0
    else:  # k < 0
        k_abs = -k
        grad[k_abs:] = -h[: n - k_abs]  # similar
        grad[: n - k_abs] += -h[k_abs:]
        grad += 1.0
    return grad
```

        grad[k:] -= h[: n - k]
    elif k < 0:
        k = -k
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]

import time, math, numpy as np
from scipy.optimize import minimize

# -------------------- basic utilities --------------------

def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function discretized on [0,2]."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 1000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0.0:
            if h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
        else:
            if h[i] > 0.0:
                add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of Thue–Morse as floats 0/1."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


# -------------------- gradient for worst shift --------------------

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the max overlap with respect to h (subgradient)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = idx
    grad = np.ones(n)
    if k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    elif k < 0:
        k = -k
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    else:  # k == 0
        grad = 1.0 - 2.0 * h
    return grad


# -------------------- SLSQP with soft‑max --------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1000) -> np.ndarray:
    """Run SLSQP on the smooth max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    return x


# -------------------- main optimisation routine --------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h = thue_morse_bits(n).astype(float)
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- phase 1: simulated annealing (≈60% of budget) -----------------
    phase1_end = start + total_budget * 0.60
    T0 = 0.010  # initial temperature
    c_current = best_c
    h_current = best_h.copy()
    while time.time() < phase1_end:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: move mass from one index to the other
        if rng.random() < 0.5:
            max_delta = min(1.0 - h_current[i], h_current[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h_new = h_current.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            max_delta = min(1.0 - h_current[j], h_current[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h_new = h_current.copy()
            h_new[i] -= delta
            h_new[j] += delta
        c_new = compute_c5(h_new)
        dC = c_new - c_current
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (total_budget * 0.60))
        if dC <= 0.0 or rng.random() < math.exp(-dC / (T + 1e-16)):
            h_current = h_new
            c_current = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                if best_c <= 0.3808:
                    return best_h.tolist(), float(best_c), int(n)

    # ---- phase 2: SLSQP soft‑max refinement (≈30% of budget) -----------
    phase2_end = start + total_budget * 0.90
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
        if time.time() >= phase2_end:
            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=1200)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            if best_c <= 0.3808:
                break

    # ---- phase 3: gradient‑based fine tuning (remaining budget) ---------
    while time.time() < start + total_budget * 0.99:
        grad = worst_gradient(best_h)
        # simple projected step
        step = 0.04
        h_new = best_h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            if best_c <= 0.3808:
                break

    # ---- final sanity check --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
