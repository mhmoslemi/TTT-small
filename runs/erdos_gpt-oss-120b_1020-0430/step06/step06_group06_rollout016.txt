def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

    else:

    # Construct the sine‑based step function: h(x)=½+½·sin(πx) on the grid.
    i = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * i / n)
