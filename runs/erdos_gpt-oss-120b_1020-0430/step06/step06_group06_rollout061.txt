def compute_c5_fft(h):
    n = h.shape[0]
    m = 1 << ((2*n-1).bit_length())
    # compute using real FFT for speed
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * G, m)
    # take first 2*n-1 entries (full linear convolution)
    max_val = np.max(conv[:2*n-1])
    return max_val * (2.0 / n)

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5_np(h: np.ndarray) -> float:
    """Discrete C5 using np.correlate (full linear correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 via FFT based linear correlation."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    H = np.fft.fft(h, m)
    B = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(B)).real
    max_val = np.max(corr[: 2 * n - 1])
    return max_val * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # random walk repair
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max correlation for SLSQP."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def binary_anneal_one_run(n: int, rng: np.random.Generator, deadline: float, inner_iters: int = 20000) -> tuple:
    """Single annealing run on a binary vector of length n."""
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    cur_h = h
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    T0 = 0.5
    while time.time() < deadline and it < inner_iters:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur_h[i] == cur_h[j]:
            it += 1
            continue
        nh = cur_h.copy()
        nh[i], nh[j] = cur_h[j], cur_h[i]
        nc = compute_c5_fft(nh)
        delta = nc - cur_c
        T = max(T0 * (0.999 ** it), 1e-7)
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_c = nc
                best_h = nh.copy()
        it += 1
        if best_c < 0.3808:
            break
    return best_h, best_c


def binary_anneal_multiple(n: int, rng: np.random.Generator, deadline: float) -> tuple:
    """Repeated annealing runs until deadline."""
    best_h = None
    best_c = np.inf
    while time.time() < deadline:
        run_deadline = min(deadline, time.time() + 8.0)  # limit each run

        if c < best_c:

    if best_big is None:
        best_big = np.zeros(n_big, dtype=np.float64)
        best_big[rng.choice(n_big, n_big // 2, replace=False)] = 1.0

    # down‑sample to 512 by averaging consecutive pairs
    n_small = n_big // 2
    h_small = (best_big[0::2] + best_big[1::2]) * 0.5  # sum preserved automatically
    h_small = enforce_sum(h_small, rng)

    # ---- Phase 2: continuous SLSQP refinement ----
    slsqp_deadline = start + budget_s * 0.93
    best_h = h_small.astype(float)
    best_c = compute_c5_np(best_h)
    beta_seq = (50.0, 500.0, 5000.0, 20000.0, 50000.0)
    for beta in beta_seq:
        if time.time() > slsqp_deadline:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
        best_c = compute_c5_np(best_h)

    # ---- Phase 3: tiny random polishing ----
    polish_deadline = start + budget_s * 0.98
    cur_h, cur_c = best_h, best_c
    it = 0
    T0 = 0.004
    while time.time() < polish_deadline:
        i = rng.integers(0, n_small)
        j = rng.integers(0, n_small)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            it += 1
            continue
        eps = max_eps * rng.random() * 0.5
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = np.clip(nh, 0.0, 1.0)
        nc = compute_c5_np(nh)
        delta = nc - cur_c
        T = max(T0 * (0.99 ** it), 1e-9)
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        it += 1

    # ---- Final repair and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_np(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
