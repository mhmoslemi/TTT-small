def proj_bounded_simplex(v, target):
    # projection onto {x: sum x = target, 0 <= x <= 1}
    # Solve for lambda using bisection.
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
```

def worst_shifts_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    inds = np.where(corr == maxc)[0]  # vectorized; may be many but fine
    grads = np.zeros_like(h, dtype=float)
    for idx in inds:
        shift = idx - (n - 1)
        k = abs(shift)
        if k == 0:
            grads += 1.0 - 2.0 * h
        else:
            # grad = ones; then subtract
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
            grads += g
    grads /= len(inds)
    return grads
```

def slsqp_optimize(h0, beta, target, rng, max_iter=2000):
    n = h0.size
    # define softmax objective
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        O = corr * (2.0/n)
        bc = beta * O
        m = np.max(bc)
        # log-sum-exp

    # gradient
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        O = corr * (2.0/n)
        bc = beta * O
        max_bc = np.max(bc)
        exp_shift = np.exp(bc - max_bc)
        w = exp_shift / np.sum(exp_shift)
        # compute gradient = sum_k w_k * grad O_k
        # We compute g = zeros(n)
        # For each shift k (index idx):
        # shift = idx - (n-1)

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


# -------------------------------------------------------------
def proj_bounded_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {x : sum x = target, 0 <= x <= 1}."""
    lo, hi = -1.0, 2.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# -------------------------------------------------------------
def worst_shifts_gradient(h: np.ndarray) -> np.ndarray:
    """Average subgradient of the worst overlap shift(s)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = corr.max()
    # include all shifts within a tiny tolerance
    idxs = np.where(corr >= maxc - 1e-12)[0]
    grad = np.zeros_like(h, dtype=float)
    for idx in idxs:
        shift = idx - (n - 1)
        k = abs(shift)
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
            grad += g
    grad /= len(idxs)
    return grad


# -------------------------------------------------------------
def subgradient_descent(h0: np.ndarray, target: float, deadline: float, rng: np.random.Generator):
    """Projected sub‑gradient method with averaging."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    avg_h = np.zeros_like(h)
    avg_cnt = 0
    t = 0
    a0 = 1.0
    while time.time() < deadline:
        t += 1
        alpha = a0 / np.sqrt(t)
        g = worst_shifts_gradient(h)
        h = proj_bounded_simplex(h - alpha * g, target)
        # occasional random jitter
        if t % 5000 == 0:
            eps = (rng.random(n) - 0.5) * 0.02
            h = proj_bounded_simplex(h + eps, target)
        avg_cnt += 1
        avg_h = (avg_h * (avg_cnt - 1) + h) / avg_cnt
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break
        if avg_cnt % 200 == 0:
            ha = proj_bounded_simplex(avg_h, target)
            ca = compute_c5(ha)
            if ca < best_c:
                best_c, best_h = ca, ha.copy()
    return best_h, best_c


# -------------------------------------------------------------
def slsqp_optimize(h0: np.ndarray, beta: float, target: float, rng: np.random.Generator, max_iter: int = 1500):
    """SLSQP on the smooth (soft‑max) approximation of max overlap."""
    n = h0.size

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        O = corr * (2.0 / n)
        bc = beta * O
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        O = corr * (2.0 / n)
        bc = beta * O
        max_bc = bc.max()
        exp_shift = np.exp(bc - max_bc)
        w = exp_shift / exp_shift.sum()
        g = np.zeros_like(x)
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            shift = idx - (n - 1)
            k = abs(shift)
            if k == 0:
                g += wk * (1.0 - 2.0 * x)
            else:
                g[: n - k] -= wk * x[k:]
                g[k:] -= wk * x[: n - k]
        return (2.0 / n) * g

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="SLSQP", jac=grad, bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return proj_bounded_simplex(x, target)
