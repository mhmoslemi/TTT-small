def compute_corr_fft(h):
    N = h.shape[0]
    L = 2 * N
    fft_h = np.fft.fft(h, n=L)
    fft_g = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(fft_h * np.conj(fft_g)).real
    # Return only first 2*N-1 entries (ignore last zero due to periodic)
    return corr[:2*N-1]
```

def worst_shift_and_gradient(h):
    N = h.shape[0]
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    shift = idx - (N - 1)
    k = abs(shift)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # subtract overlapping
        grad[: N - k] -= h[k:]
        grad[k:] -= h[: N - k]
    return shift, grad, corr[idx] * (2.0 / N)  # also return current C5
```

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    # max epsilon we can move from i to j without breaking bounds.
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
```

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    N = h.shape[0]
    L = 2 * N
    fft_h = np.fft.fft(h, n=L)
    fft_g = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(fft_h * np.conj(fft_g)).real
    return corr[: 2 * N - 1]

def compute_c5_fft(h: np.ndarray) -> float:
    return np.max(compute_corr_fft(h)) * (2.0 / h.shape[0])

def repair_sum_and_clip(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h

def worst_shift_and_gradient(h: np.ndarray):
    N = h.shape[0]
    corr = compute_corr_fft(h)
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)
    k = abs(shift)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] -= h[k:]
        grad[k:] -= h[: N - k]
    return shift, grad, corr[idx] * (2.0 / N)

def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def random_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    N = h.shape[0]
    i, j = rng.integers(0, N, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.2:
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def gradient_descent_refine(h: np.ndarray, rng: np.random.Generator,
                           max_iter: int = 200, init_step: float = 0.05) -> np.ndarray:
    N = h.shape[0]
    target = N / 2.0
    step = init_step
    for it in range(max_iter):
        _, grad, cur_c = worst_shift_and_gradient(h)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = repair_sum_and_clip(h, target, rng)
        new_c = compute_c5_fft(h)
        if new_c > cur_c:
            step *= 0.7
        else:
            step *= 1.02
        step = max(min(step, 0.2), 1e-4)
    return h

def softmax_obj(x: np.ndarray) -> float:
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = repair_sum_and_clip(x, CURRENT_TARGET, rng)
    return x

    best_h = h.copy()

    current_h, current_c = best_h.copy(), best_c

    it = 0

        else:
            nh = random_move(current_h, rng, scale=eps_scale)
        nc = compute_c5_fft(nh)
        delta = nc - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_h, current_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        it += 1
        if it % 500 == 0:

            eps_scale *= 0.997
            T = max(T, 1e-12)
            eps_scale = max(eps_scale, 1e-5)

    # Upscale to fine grid
    h_fine = np.repeat(best_h, 2)
    N = h_fine.size
    target = N / 2.0
    best_c = min(best_c, compute_c5_fft(h_fine))

    remaining = budget_s - (time.time() - start)

    if time.time() < start + budget_s * 0.995:
        h_fine = gradient_descent_refine(h_fine, rng, max_iter=30, init_step=0.01)
        best_c = min(best_c, compute_c5_fft(h_fine))

    # Final repair
    h_fine = np.clip(h_fine, 0.0, 1.0)
    h_fine = repair_sum_and_clip(h_fine, target, rng)
    final_c = compute_c5_fft(h_fine)
