def adam_optimize(h0, target, time_budget, rng):
    n = h0.size; dx = 2.0 / n; m = 1 << (2*n - 1).bit_length()
    ones_fft = np.fft.rfft(np.ones(m), m)  # shape m//2+1
    h = h0.copy()
    m_t = np.zeros_like(h)
    v_t = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    base_lr = 0.1
    b1_pow = beta1
    b2_pow = beta2
    t = 0
    # compute initial c
    H = np.fft.rfft(h, m)
    max_c = np.max(np.fft.irfft(H * np.conj(ones_fft - H), m)) * dx
    best_h = h.copy()
    best_c = max_c

    start = time.time()
    # loop
    while True:
        # time check every 100 iterations
        if t % 100 == 0:
            if time.time() - start >= time_budget:
                break
        # recompute H etc each iteration (need fresh after h changed)
        H = np.fft.rfft(h, m)
        cross_hh_full = np.fft.irfft(H * np.conj(H), m)
        cross_h1mh_full = np.fft.irfft(H * np.conj(ones_fft - H), m)
        max_c = np.max(cross_h1mh_full) * dx
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()
            if best_c <= 0.3808:
                break
        # gradient
        cross_nonneg = cross_hh_full[n-1:]
        k = int(np.argmin(cross_nonneg))
        grad = np.ones_like(h)
        if k > 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        # Adam update
        t += 1
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * (grad ** 2)
        m_hat = m_t / (1 - b1_pow)
        v_hat = v_t / (1 - b2_pow)
        step = base_lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = project_box_sum(h - step, target)
        # update bias powers
        b1_pow *= beta1
        b2_pow *= beta2
    return best_h, best_c

import time
import numpy as np

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    m = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, m)
    ones_fft = np.fft.rfft(np.ones(m), m)
    corr = np.fft.irfft(H * np.conj(ones_fft - H), m)
    return np.max(corr) * dx
