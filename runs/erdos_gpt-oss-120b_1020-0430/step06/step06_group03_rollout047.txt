import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral C5 for discretized h on [0,2]."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    G = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(n)]))
    corr = np.fft.irfft(np.conj(H) * G, n=L)
    return corr.max() * (2.0 / n)


def project_onto_box_sum(v, target, tol=1e-12):
    """Projection of v onto [0,1]^n with exact sum=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > tol:
        mask = (x > tol) & (x < 1.0 - tol)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def worst_gradient(h):
    """Gradient of the worst overlap (the shift attaining max)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h, grad, rng, scale=0.5):
    """Move a unit of mass from a high‑gradient index to a low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h, rng, max_iters=400, step_scale=0.5):
    """Simple hill‑climbing using guided perturbations."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def softmax_gradient(h, beta):
    """Gradient of the smooth max (log‑sum‑exp) of the overlap values."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad
