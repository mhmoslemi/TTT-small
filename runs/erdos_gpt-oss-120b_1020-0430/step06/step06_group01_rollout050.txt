def worst_gradient_from_shift(h, shift):
    n = h.shape[0]
    grad = np.zeros(n)
    s = shift
    if s == 0:
        grad = 1.0 - 2.0 * h
    elif s > 0:
        # i in [0, n-s-1] get +1 - h[i+s]
        grad[:n-s] += 1.0 - h[s:]
        # i in [s, n-1] get -h[i-s]
        grad[s:] -= h[:n-s]
    else: # s < 0
        s_abs = -s
        # i in [s_abs, n-1] get +1 - h[i - s_abs] (since i+s = i - s_abs)
        grad[s_abs:] += 1.0 - h[:n - s_abs]
        # i in [0, n - s_abs - 1] get -h[i + s_abs]
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def enforce_sum_simple(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # if diff >0, need to add diff mass; find index with h <1
    # else need to remove mass from index with h>0.
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = rng.choice(np.where(h < 1.0 - 1e-12)[0])
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.choice(np.where(h > 1e-12)[0])
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h

import time
import numpy as np

TARGET_C5 = 0.3808  # stop early if reached


def enforce_sum_simple(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = rng.choice(np.where(h < 1.0 - 1e-12)[0])
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.choice(np.where(h > 1e-12)[0])
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h


def compute_c5_and_worst(h: np.ndarray):
    """Return (C5 value, shift giving the maximum overlap)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)          # can be negative
    c5 = max_val * (2.0 / n)
    return c5, shift, corr


def worst_gradient_from_shift(h: np.ndarray, shift: int):
    """Gradient of the worst‑shift overlap w.r.t. h."""
    n = h.size
    grad = np.zeros(n, dtype=float)
    s = shift
    if s == 0:
        grad = 1.0 - 2.0 * h
    elif s > 0:
        grad[: n - s] += 1.0 - h[s:]
        grad[s:] -= h[: n - s]
    else:  # s < 0
        s_abs = -s
        grad[s_abs:] += 1.0 - h[: n - s_abs]
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def guided_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.7):
    """Deterministic hill‑climb using the worst‑shift gradient."""
    h = h0.copy()
    c, shift, _ = compute_c5_and_worst(h)
    best_h, best_c = h.copy(), c
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient_from_shift(h, shift)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i, j = rng.choice(n, 2, replace=False)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step_scale
        h[i] -= eps
        h[j] += eps
        c_new, shift_new, _ = compute_c5_and_worst(h)
        if c_new < best_c:
            best_c, best_h = c_new, h.copy()
        if c_new <= c:
            c, shift = c_new, shift_new
        else:
            h[i] += eps
            h[j] -= eps
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
        if best_c <= TARGET_C5:
            break
    return best_h, best_c


def sa_optimize(h0: np.ndarray, rng: np.random.Generator, budget: float):
    """Simulated annealing driven by the worst‑shift gradient."""
    start = time.time()
    h = h0.copy()
    c, shift, _ = compute_c5_and_worst(h)
    best_h, best_c = h.copy(), c
    n = h.size
    T0, T_end = 0.2, 1e-6
    it = 0
    while time.time() - start < budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / budget)
        grad = worst_gradient_from_shift(h, shift)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i, j = rng.choice(n, 2, replace=False)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            pos = np.where(h > 1e-12)[0]
            zero = np.where(h < 1.0 - 1e-12)[0]
            if len(pos) == 0 or len(zero) == 0:
                break
            i = rng.choice(pos)
            j = rng.choice(zero)
            max_eps = min(h[i], 1.0 - h[j])
        eps_factor = 0.5 * (1.0 - elapsed / budget) + 0.01
        eps = max_eps * rng.random() * eps_factor
        if eps <= 0:
            continue
        h[i] -= eps
        h[j] += eps
        c_new, shift_new, _ = compute_c5_and_worst(h)
        delta = c_new - c
        accept = delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12))
        if accept:
            c, shift = c_new, shift_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= TARGET_C5:
                    break
        else:
            h[i] += eps
            h[j] -= eps
        it += 1
        if it % 2000 == 0:
            # small random shake to keep diversity
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * 0.01
                h[i] -= eps
                h[j] += eps
            c, shift, _ = compute_c5_and_worst(h)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target resolution (below 1000 points)
    n_target = 768

    # initialise from parent if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        n0 = h0.size
        if n0 != n_target:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum_simple(h0, rng)
    else:
        h0 = rng.random(n_target)
        h0 = enforce_sum_simple(h0, rng)

    c0, _, _ = compute_c5_and_worst(h0)
    best_h, best_c = h0.copy(), c0

    # allocate time: 80% SA, 20% polishing
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining <= 0:
        final_c = compute_c5_and_worst(best_h)[0]
        return best_h.tolist(), float(final_c), int(best_h.size)

    sa_time = remaining * 0.80
    h_sa, c_sa = sa_optimize(best_h, rng, sa_time)
    if c_sa < best_c:
        best_h, best_c = h_sa, c_sa

    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 0:
        # final deterministic polish
        max_iters = int(min(80000, remaining * 300))
        h_fc, c_fc = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.5)
        if c_fc < best_c:
            best_h, best_c = h_fc, c_fc

    # very short stochastic tweaks until time ends
    while time.time() - start < total_budget * 0.999:
        _, shift, _ = compute_c5_and_worst(best_h)
        grad = worst_gradient_from_shift(best_h, shift)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i, j = rng.choice(best_h.size, 2, replace=False)
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.02
        best_h[i] -= eps
        best_h[j] += eps
        c_new = compute_c5_and_worst(best_h)[0]
        if c_new < best_c:
            best_c = c_new
        else:
            best_h[i] += eps
            best_h[j] -= eps

    # final sanitation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_simple(best_h, rng)
    final_c = compute_c5_and_worst(best_h)[0]

    return best_h.tolist(), float(final_c), int(best_h.size)
