def _make_target_sum(arr, target, rng):
    """Adjust arr in-place so that sum(arr) == target while staying in [0,1]."""
    diff = target - arr.sum()

def _compute_c5(h):

    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1

    rev = (1.0 - h)[::-1]
    G = np.fft.rfft(rev, n=N)
    conv = np.fft.irfft(H * G, n=N)

        # scale delta (larger early, smaller later)
        scale = 0.5 * (T / T0 + 0.01)
        delta = max_tr * rng.random() * scale
        if delta <= 0.0:
            continue

        h_new = cur_h.copy()
        h_new[src] -= delta
        h_new[dst] += delta

        new_c = _compute_c5(h_new)

        # Metropolis acceptance
        if (new_c < cur_c) or (rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-15))):
            cur_h, cur_c = h_new, new_c
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
                if best_c < 0.38079:               # early exit if record beaten
                    time_limit = time.time() + 0.1

        it += 1

    # ---------- gradient polishing ----------
    best_h = _gradient_refine(best_h, target, rng, step=0.02, iters=15)
    final_c = _compute_c5(best_h)

    # final feasibility tidy‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _make_target_sum(best_h, target, rng)

    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
