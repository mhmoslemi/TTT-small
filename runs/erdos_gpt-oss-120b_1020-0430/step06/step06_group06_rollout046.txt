def compute_c5(h):
    # h can be int or float; convert to float
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return np.max(corr) * (2.0 / h.size)

def repair_binary(h):
    # h is int array; ensure sum = target
    current = h.sum()
    diff = target - current
    if diff > 0:
        zeros = np.where(h == 0)[0]
        if zeros.size >= diff:
            rng.shuffle(zeros)
            h[zeros[:diff]] = 1
        else:
            # fill all zeros then flip some ones randomly? but shouldn't happen.
            h[:] = 0
            h[:target] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[: -diff]] = 0
    return h

import time, numpy as np
from scipy.optimize import minimize

# ---------- utility ----------
def compute_c5(h: np.ndarray) -> float:
    """C5 for step function on [0,2] with N points."""
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- gradient refinement ----------
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

# ---------- binary GA ----------
def repair_binary(h: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:diff]] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[: -diff]] = 0
    return h

def ga_optimize(initial_bin: np.ndarray, rng: np.random.Generator,
                time_budget: float, start_time: float) -> np.ndarray:
    n = initial_bin.size
    target = n // 2
    pop_size = 250
    mutation_rate = 0.02
    crossover_prob = 0.7
    tournament_k = 3
    # initialise population
    population = [initial_bin.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        if i == 0:
            continue
        child = population[0].copy()
        mask = rng.random(n) < 0.5
        child[mask] = 1 - child[mask]
        population[i] = repair_binary(child, target, rng)
    best_h = None
    best_c5 = 1e9
    # main loop
    while time.time() - start_time < time_budget:
        c5_vals = np.array([compute_c5(ind) for ind in population])
        idx_min = int(c5_vals.argmin())
        if c5_vals[idx_min] < best_c5:
            best_c5 = float(c5_vals[idx_min])
            best_h = population[idx_min].copy()
        # generate next generation
        new_pop = []
        for _ in range(pop_size):
            t = rng.choice(pop_size, size=tournament_k, replace=False)
            parent1 = population[t[np.argmin(c5_vals[t])]]
            child = parent1.copy()
            if rng.random() < crossover_prob:
                t2 = rng.choice(pop_size, size=tournament_k, replace=False)
                parent2 = population[t2[np.argmax(c5_vals[t2])]]
                mask = rng.random(n) < 0.5
                child[mask] = parent2[mask]
            mutate = rng.random(n) < mutation_rate
            child[mutate] = 1 - child[mutate]
            child = repair_binary(child, target, rng)
            new_pop.append(child)
        population = new_pop
        # quick time check
        if time.time() - start_time > time_budget * 0.99:
            break
    return best_h if best_h is not None else initial_bin

# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initialise base binary vector ---
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # binarise by rounding, then repair exact sum
        h0_bin = np.rint(h0).astype(int)
        n0 = h0_bin.size
        target0 = n0 // 2
        diff0 = target0 - h0_bin.sum()
        if diff0 > 0:
            zeros = np.where(h0_bin == 0)[0]
            rng.shuffle(zeros)
            h0_bin[zeros[:diff0]] = 1
        elif diff0 < 0:
            ones = np.where(h0_bin == 1)[0]
            rng.shuffle(ones)
            h0_bin[ones[: -diff0]] = 0
        # upsample to get finer resolution (factor 2)
        factor = 2
        n = n0 * factor
        h0_bin = np.repeat(h0_bin, factor)
    else:
        n = 1024
        target = n // 2
        h0_bin = np.zeros(n, dtype=int)
        h0_bin[:target] = 1
        rng.shuffle(h0_bin)

    # ------- genetic algorithm phase -------
    ga_time = budget_s * 0.55
    best_bin = ga_optimize(h0_bin, rng, ga_time, start)

    # ------- gradient refinement (binary -> real) -------
    best_h = best_bin.astype(float)
    if time.time() - start < budget_s * 0.7:
        best_h = gradient_refine(best_h, rng, step=0.025, max_iter=12)

    # ------- soft‑max SLSQP polishing -------
    beta_seq = (50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0)
    for beta in beta_seq:
        if time.time() - start > budget_s * 0.95:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=500)
    # final repair
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
