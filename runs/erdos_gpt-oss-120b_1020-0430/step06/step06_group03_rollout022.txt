def compute_c5(h):
    n = h.size
    L = 1 << ((2*n - 1).bit_length())  # next power of 2 >= 2n
    H = np.fft.rfft(h, L)

    max_val = conv.max()

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    conv = np.fft.ifft(H * G, L).real[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator, max_iters: int = 1000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # target resolution
    n = 1024
    target = n / 2.0

    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        if n0 != n:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x, x0, h0)
        else:
            h0 = h0.copy()
    else:
        h0 = np.zeros(n)
        idx = rng.choice(n, int(target), replace=False)
        h0[idx] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start)

    # ---------- softmax‑gradient phases ----------
    beta_list = [20, 80, 300, 1000, 3000, 10000, 30000]
    step_map = {20: 0.6, 80: 0.5, 300: 0.4, 1000: 0.3, 3000: 0.2, 10000: 0.15, 30000: 0.1}

    for beta in beta_list:
        if time_left() < 5.0:
            break
        iters = int(1500 * 256 / n)
        if beta >= 3000:
            iters = int(2500 * 256 / n)
        step0 = step_map.get(beta, 0.1)
        h = best_h.copy()
        for it in range(iters):
            if time_left() < 5.0:
                break
            O = np.correlate(h, 1.0 - h, mode="full")
            Omax = O.max()
            exp_term = np.exp(beta * (O - Omax))
            w = exp_term / exp_term.sum()
            grad = np.zeros_like(h)
            wk = w[n - 1]
            grad += wk * (1.0 - 2.0 * h)
            for k in range(1, n):
                wk = w[n - 1 + k] + w[n - 1 - k]
                if wk == 0.0:
                    continue
                grad += wk
                grad[: n - k] -= wk * h[k:]
                grad[k:] -= wk * h[: n - k]
            norm = np.linalg.norm(grad)
            if norm > 0.0:
                step = step0 * (1.0 - it / iters) / norm
                v = h - step * grad
                h = project_onto_box_sum(v, target)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # local hill climbing after each beta
        if time_left() > 5.0:
            hc, cc = guided_hill_climb(best_h, rng,
                                      max_iters=int(1500 * 256 / n), step_scale=0.5)
            if cc < best_c:
                best_h, best_c = hc, cc

    # ---------- intensive hill climbing ----------
    while time_left() > 3.0:
        # perturb guided by worst gradient
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc
        # occasional random transfer
        if rng.random() < 0.1:
            i, j = rng.choice(n, 2, replace=False)
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 1e-12:
                eps = rng.random() * max_eps * 0.5
                nh2 = best_h.copy()
                nh2[i] -= eps
                nh2[j] += eps
                nh2 = project_onto_box_sum(nh2, target)
                nc2 = compute_c5(nh2)
                if nc2 < best_c:
                    best_h, best_c = nh2, nc2

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
