def softmax_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_beta_corr = np.max(beta * corr)
    # use log-sum-exp stable
    exp_beta_corr = np.exp(beta * corr - max_beta_corr)
    sum_exp = np.sum(exp_beta_corr)
    w = exp_beta_corr / sum_exp

    grad = np.zeros_like(h)
    # For each shift idx (0..2n-2)
    for idx, wk in enumerate(w):
        if wk == 0:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wk * (1 - 2 * h)
        elif k > 0:
            # positive shift
            grad[:n-k] += wk * (1 - h[k:])
            grad[k:] -= wk * h[:n-k]
        else:
            k_abs = -k
            grad[k_abs:] += wk * (1 - h[:n - k_abs])
            grad[:n - k_abs] -= wk * h[k_abs:]
    return corr, grad
```

import time
import numpy as np

# ---------- helper utilities ----------

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution, guaranteed to finish because diff is bounded
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def corr_fft(h: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation of h with (1-h) using FFT.
    Returns array of length 2*len(h)-1 (full mode).
    """
    b = 1.0 - h
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b[::-1], n=L)
    conv = np.fft.irfft(H * B, n=L)
    return conv[: 2 * h.size - 1]


def compute_c5_scaled(h: np.ndarray, L: int) -> float:
    """Maximum overlap multiplied by dx = 2/len(h)."""
    n = h.size
    return corr_fft(h, L).max() * (2.0 / n)

    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent using worst shift, fast via FFT."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5_scaled(h, L)

    decay = max(1, max_iters // 5)
    it = 0
    while it < max_iters and time.time() < time_limit:
        corr = corr_fft(h, L)
        max_corr = corr.max()
        idx = int(corr.argmax())
        k = idx - (n - 1)

        # gradient of the worst shift
        if k == 0:
            grad = 1.0 - 2.0 * h
        elif k > 0:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            ka = -k
            grad = np.ones_like(h)
            grad[ka:] -= h[: n - ka]
            grad[: n - ka] -= h[ka:]

        i = int(grad.argmax())
        j = int(grad.argmin())

        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps

        c = max_corr * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()

        it += 1
        if (it % decay) == 0:
            step_scale = max(step_scale * 0.7, 1e-7)

    return best_h, best_c


# ---------- binary hill climbing (swap) ----------

def binary_hill_climb_fft(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int,
    L: int,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Discrete hill climbing on 0/1 vectors using full swaps."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5_scaled(h, L)

    it = 0
    while it < max_iters and time.time() < time_limit:
        corr = corr_fft(h, L)
        max_corr = corr.max()
        idx = int(corr.argmax())
        k = idx - (n - 1)

        if k == 0:
            grad = 1.0 - 2.0 * h
        elif k > 0:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            ka = -k
            grad = np.ones_like(h)
            grad[ka:] -= h[: n - ka]
            grad[: n - ka] -= h[ka:]

        i = int(grad.argmax())
        j = int(grad.argmin())

        if i != j:
            # swap the binary values
            h[i], h[j] = h[j], h[i]

        c = max_corr * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()

        it += 1

    return best_h, best_c


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512                     # number of discretisation points
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = enforce_sum(h0, rng)
    else:
        h0 = enforce_sum(rng.random(n), rng)

    best_h = h0.copy()
    best_c = compute_c5_scaled(best_h, L)

    # ----- time allocation ------------------------------------------------
    total_budget = budget_s
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    # allocate 95 % to continuous hill climbing, 5 % to binary refinement
    cont_time = remaining * 0.95
    bin_time = remaining - cont_time

    # ----- continuous hill climbing ---------------------------------------
    cont_end = start + cont_time
    # generous iteration bound; actual stop driven by wall‑clock
    max_cont_iters = 10_000_000
    h_cont, c_cont = guided_hill_climb_fft(
        best_h.copy(),
        rng,
        max_iters=max_cont_iters,
        step_scale=1.0,
        L=L,
        time_limit=cont_end,
    )
    if c_cont < best_c:
        best_h, best_c = h_cont, c_cont

    # ----- rounding to binary and binary hill climbing --------------------
    # round the best continuous solution
    idx_desc = np.argsort(-best_h)
    binary_h = np.zeros(n, dtype=float)
    binary_h[idx_desc[: n // 2]] = 1.0
    c_bin = compute_c5_scaled(binary_h, L)
    if c_bin < best_c:
        best_h, best_c = binary_h, c_bin

    # binary refinement within remaining time
    bin_end = start + total_budget
    max_bin_iters = 5_000_000
    h_bin, c_bin = binary_hill_climb_fft(
        best_h.copy(),
        rng,
        max_iters=max_bin_iters,
        L=L,
        time_limit=bin_end,
    )
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin

    # ----- final sanitisation ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_scaled(best_h, L)

    return best_h.tolist(), float(final_c), int(best_h.size)
