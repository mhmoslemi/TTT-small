def project_onto_box_sum(v, target_sum):
    # Return projection of v onto {x: sum x = target_sum, 0<=x<=1}
    # Using bisection on lambda.
    lo = -np.max(v) - 1  # lower bound for lambda maybe -max(v)-1
    hi = 1 - np.min(v) + 1
    # but we need monotonic f(lambda) = sum clip(v+lambda, 0,1)
    # start with lo = -max(v) - 1, hi = 1 - min(v) + 1 ensure bounds
    for _ in range(50):
        mid = (lo + hi) / 2
        clipped = np.clip(v + mid, 0, 1)
        s = clipped.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2, 0, 1)

def binary_sa(n, time_budget, start_time):
    target_sum = n//2
    # initialize random binary
    h = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, target_sum, replace=False)
    h[ones_idx] = 1.0
    ov = np.correlate(h, h, mode='full')
    c5 = 1 - (2.0/n) * np.min(ov)

    best_h = h.copy()
    best_c5 = c5

    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # choose random swap
        i = rng.choice(np.flatnonzero(h == 1))
        j = rng.choice(np.flatnonzero(h == 0))
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        ov = np.correlate(h, h, mode='full')
        c5_new = 1 - (2.0/n) * np.min(ov)
        delta = c5_new - c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # revert
            h[j] = 0.0
            h[i] = 1.0
    return best_h, best_c5

def binary_sa(n, rng, time_budget, start_time):
    target = n // 2
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, target, replace=False)
    h[ones] = 1.0
    # compute C5
    ov = np.correlate(h, h, mode='full')
    c5 = 1 - (2.0 / n) * np.min(ov)
    best_h = h.copy()
    best_c5 = c5

    # schedule
    T0 = 0.05
    T_end = 1e-4

    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        fraction = elapsed / time_budget
        T = T0 * (T_end / T0) ** fraction

        # pick random ones and zeros
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        # swap
        h[i] = 0.0
        h[j] = 1.0

        ov = np.correlate(h, h, mode='full')
        c5_new = 1 - (2.0 / n) * np.min(ov)
        delta = c5_new - c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                if best_c5 <= 0.3808:
                    break
        else:
            # revert swap
            h[j] = 0.0
            h[i] = 1.0

    return best_h, best_c5

def project_onto_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    lo = -np.max(v) - 1.0

            hi = mid
        else:
            lo = mid

        # pick random 1 and 0 positions
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        # swap
        h[i] = 0.0
        h[j] = 1.0
        ov = np.convolve(h, h, mode="full")
        c5_new = 1.0 - (2.0 / n) * np.min(ov)
        delta = c5_new - c5
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c5 = c5_new
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h.copy()
                if best_c5 <= 0.3808:
                    break
        else:
            h[j] = 0.0
            h[i] = 1.0
    return best_h, best_c5


# ---------- subgradient descent ----------
def subgradient_descent(h0: np.ndarray, target_sum: float,
                        time_budget: float, start_time: float):
    n = h0.size
    h = h0.copy()
    ov = np.convolve(h, h, mode="full")
    c5 = 1.0 - (2.0 / n) * np.min(ov)
    best_h, best_c5 = h.copy(), c5
    it = 0
    a = 4.0                         # step‑size coefficient
    while time.time() - start_time < time_budget:
        # worst shift and gradient
        _, grad, _ = worst_shift_and_grad(h)
        it += 1
        step = a / np.sqrt(it)
        v = h + step * grad
        h = project_onto_box_sum(v, target_sum)

        ov = np.convolve(h, h, mode="full")
        c5 = 1.0 - (2.0 / n) * np.min(ov)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
            if best_c5 <= 0.3808:
                break
    return best_h, best_c5


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    n = 512
    target_sum = n / 2.0

    best_h = None
    best_c5 = np.inf

    # initial construction, if supplied
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target_sum)
        c0 = c5_from_h(h0)
        best_h, best_c5 = h0, c0

    # ---------- binary SA (≈30% time) ----------
    binary_time = min(total_budget * 0.30, 300.0)
    binary_deadline = start_time + binary_time
    while time.time() < binary_deadline:
        remaining = binary_deadline - time.time()
        if remaining < 1.0:
            break
        run_len = min(30.0, remaining)
        h_bin, c_bin = binary_sa(n, rng, run_len)
        if c_bin < best_c5:
            best_h, best_c5 = h_bin, c_bin
            if best_c5 <= 0.3808:
                break

    # ---------- subgradient descent (rest of time) ----------
    remaining = total_budget - (time.time() - start_time)
    if remaining > 5.0 and best_h is not None:
        h_sub, c_sub = subgradient_descent(best_h, target_sum,
                                           remaining, time.time())
        if c_sub < best_c5:
            best_h, best_c5 = h_sub, c_sub

        # possible extra restarts with small perturbations
        while time.time() - start_time < total_budget * 0.95:
            left = total_budget - (time.time() - start_time)
            if left < 5.0:
                break
            pert = best_h.copy()
            # small random nudges
            pert += (rng.random(n) - 0.5) * 0.1
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_onto_box_sum(pert, target_sum)
            h_tmp, c_tmp = subgradient_descent(pert, target_sum, left, time.time())
            if c_tmp < best_c5:
                best_h, best_c5 = h_tmp, c_tmp

    # ---------- final tiny tweaks ----------
    while time.time() - start_time < total_budget * 0.999 and best_h is not None:
        _, grad, _ = worst_shift_and_grad(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            h_new = best_h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            c_new = c5_from_h(h_new)
            if c_new < best_c5:
                best_h, best_c5 = h_new, c_new

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target_sum)
    final_c5 = c5_from_h(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
