def compute_c5_fft(h):
    n = h.size
    a = h.astype(np.float64)
    b = 1.0 - a
    N = 2*n - 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    corr = np.fft.irfft(A * np.conj(B), N)
    # corr length N, indices shift = -(n-1) .. (n-1)
    # scaling factor dx = 2 / n
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

enforce_sum_binary(vec, target, rng):
    current = int(vec.sum())
    diff = target - current
    if diff > 0:
        # need more ones: select zeros to turn to ones
        zero_idxs = np.flatnonzero(vec == 0)
        choose = rng.choice(zero_idxs, diff, replace=False)
        vec[choose] = 1
    elif diff < 0:
        one_idxs = np.flatnonzero(vec == 1)
        choose = rng.choice(one_idxs, -diff, replace=False)
        vec[choose] = 0
    return vec

def random_transfer_move(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

import numpy as np
import math

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def random_transfer_move(h: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a small amount of mass between two positions, preserving total sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.5:
        # increase i, decrease j
        max_eps = min(1.0 - h[i], h[j])
        if max_eps <= 0.0:
            return h.copy()
        eps = max_eps * rng.random() * scale
        h2 = h.copy()
        h2[i] += eps
        h2[j] -= eps
    else:
        # decrease i, increase j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            return h.copy()
        eps = max_eps * rng.random() * scale
        h2 = h.copy()
        h2[i] -= eps
        h2[j] += eps
    return h2


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n // 2

    # ----- initialise h ----- #
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target_sum, rng)
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, target_sum, replace=False)] = 1.0

    current_h = h.copy()
    current_c = compute_c5(current_h)
    best_h = current_h.copy()
    best_c = current_c

    # ----- simulated annealing ----- #
    anneal_time = budget_s * 0.70
    init_temp = 0.001
    final_temp = 1e-6

    while time.time() - start < anneal_time:
        elapsed = time.time() - start
        progress = elapsed / anneal_time
        temp = init_temp * (1 - progress) + final_temp * progress
        scale = max(0.5 * (1 - progress), 0.01)

        h_new = random_transfer_move(current_h, rng, scale=scale)
        new_c = compute_c5(h_new)
        delta = new_c - current_c

        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            current_h = h_new
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
        # else reject

    # ----- gradient polishing ----- #
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        refined = gradient_refine(best_h, target_sum, rng, step=0.02, max_iter=30)
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_c = refined_c
            best_h = refined.copy()

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
