def project_onto_box_and_sum(v, target, eps=1e-9, max_iter=100):
    lo = -np.max(v) - 1  # ensure large negative, so v - lo large, clipped to 1 => sum = n
    hi = np.max(v) + 1   # large positive, v - hi negative, clipped to 0 => sum = 0
    # but we can start with lo = -1e5, hi=1e5.
    for _ in range(max_iter):
        mid = (lo + hi)/2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < eps:
            break
    return np.clip(v - (lo+hi)/2, 0, 1)

def project_onto_box_and_sum(v, target, tol=1e-10, max_iter=100):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # try to adjust a free index
        idx = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if idx.size:
            i = idx[0]
            x[i] += diff
            x[i] = np.clip(x[i], 0.0, 1.0)
        else:
            # small fudge
            x += diff / v.size
            x = np.clip(x, 0.0, 1.0)
    return x

    corr = fftconvolve(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

import time
import numpy as np
from scipy.signal import fftconvolve

def _project_box_sum(v: np.ndarray, target: float, tol: float = 1e-10, max_iter: int = 100) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            i = free[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with N points."""
    corr = fftconvolve(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    target = N / 2.0
    total_iters = 5000
    beta_start, beta_end = 5.0, 20000.0
    lr_start, lr_end = 0.5, 0.001
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # ----- initialise h -----
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.size
        if N % n0 == 0:
            factor = N // n0
            h = np.repeat(init, factor)
        else:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, N, endpoint=False)
            h = np.interp(x, x0, init)
        h = np.clip(h, 0.0, 1.0)
    else:
        h = np.zeros(N)
        idx = rng.choice(N, N // 2, replace=False)
        h[idx] = 1.0

    h = _project_box_sum(h, target)

    # ----- Adam state -----
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    best_h = h.copy()
    best_c = _c5(best_h)

    dx = 2.0 / N

    # ----- main optimisation loop -----
    for it in range(1, total_iters + 1):
        if time.time() - start > budget_s * 0.95:
            break

        # schedules
        beta = beta_start * (beta_end / beta_start) ** (it / total_iters)
        lr = lr_start * (1.0 - it / total_iters) + lr_end

        a = 1.0 - h
        corr = fftconvolve(h, a, mode="full")
        max_corr = np.max(corr)

        w = np.exp(beta * (corr - max_corr))
        w /= w.sum()
