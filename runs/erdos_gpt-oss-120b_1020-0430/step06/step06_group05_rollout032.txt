def simulated_annealing(h, target_sum, rng, time_budget, start_time):
    # use SA parameters
    temp = 0.01   # initial temperature
    final_temp = 1e-6
    decay = 0.9995  # per iteration
    best_h = h.copy()
    best_c = compute_c5(best_h)
    c = best_c
    it = 0
    while True:
        if time.time() - start_time > time_budget * 0.97:
            break
        # compute worst shift gradient
        k, grad = worst_shift_and_grad(h)
        # pick alpha randomly (log-uniform)
        alpha = 10 ** rng.uniform(-4, -0.5)  # between 0.0001 and 0.316
        h_new = mass_transfer_move(h, grad, alpha, target_sum, rng)
        c_new = compute_c5(h_new)
        delta = c_new - c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h = h_new
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
        # cooling
        temp = max(final_temp, temp * decay)
        it += 1
        # optionally include random perturbations occasionally
        if it % 500 == 0:
            h_alt = random_perturb(h, rng, scale=0.3)
            c_alt = compute_c5(h_alt)
            if c_alt < c:
                h = h_alt
                c = c_alt
                if c_alt < best_c:
                    best_c = c_alt
                    best_h = h_alt.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----- global parameters for SLSQP soft‑max ----- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp for stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def grad_for_shift_k(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the worst‑shift overlap for a known shift k."""
    n = h.size
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def mass_transfer_move(
    h: np.ndarray,
    grad: np.ndarray,
    alpha: float,
    target: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def gradient_refine(
    h: np.ndarray,
    target: float,
    rng: np.random.Generator,
    step: float = 0.015,
    max_iter: int = 10,
) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_refine(
    h: np.ndarray,
    target: float,
    rng: np.random.Generator,
    beta: float,
    max_iter: int,
) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


def random_h(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible step function."""
    h = rng.beta(0.5, 0.5, size=n)  # U‑shaped distribution encourages extremes
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def simulated_annealing(
    h0: np.ndarray,
    target: float,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Simple SA using mass‑transfer moves directed by the worst shift."""
    end = time.time() + time_limit
    n = h0.size
    h = h0.copy()
    # initial evaluation
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = np.max(corr) * (2.0 / n)
    best_h, best_c = h.copy(), c
    temp = 0.02
    final_temp = 1e-6
    decay = 0.9995

    while time.time() < end:
        # correlation gives both k and c in one pass
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        c = max_val * (2.0 / n)
        k = abs(max_idx - (n - 1))

        grad = grad_for_shift_k(h, k)

        # log‑uniform α ∈ [10⁻⁴, 10⁻⁰·⁵]
        log_alpha = rng.uniform(-4.0, -0.5)
        alpha = 10.0 ** log_alpha

        h_new = mass_transfer_move(h, grad, alpha, target, rng)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        c_new = np.max(corr_new) * (2.0 / n)
