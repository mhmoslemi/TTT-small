import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    """Euclidean projection onto [0,1]^n with sum == target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def slsqp_optimize(h0, beta, rng, max_iter=800):
    n = h0.size
    target = n / 2.0
    def soft_obj(x):
        O = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * O
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def eq_con(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_con}
    res = minimize(soft_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    # initialise
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 512
        target = n / 2.0
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------ #
    # Phase A : continuation on beta with softmax gradient
    beta_schedule = [10, 50, 200, 500, 2000, 8000, 20000, 80000, 200000]
    step_map = {10: 0.7, 50: 0.6, 200: 0.5, 500: 0.45,
                2000: 0.35, 8000: 0.30, 20000: 0.25, 80000: 0.20, 200000: 0.15}
    base_iter = int(1500 * 256 / n)  # ~750 for n=512
    for beta in beta_schedule:
        if time.time() - start > budget_s * 0.78:
            break
        iters = base_iter
        if beta >= 20000:
            iters = int(2000 * 256 / n)
        step0 = step_map.get(beta, 0.15)
        h = best_h.copy()
        for it in range(iters):
            if time.time() - start > budget_s * 0.78:
                break
            grad = softmax_gradient(h, beta)
            norm = np.linalg.norm(grad)
            step = step0 * (1.0 - it / iters) / (norm + 1e-12)
            v = h - step * grad
            h = project_onto_box_sum(v, target)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # a short hill‑climb after each beta
        if time.time() - start < budget_s * 0.85:
            hc, cc = guided_hill_climb(h, rng,
                max_iters=int(800 * 256 / n), step_scale=0.5)
            if cc < best_c:
                best_h, best_c = hc.copy(), cc

    # ------------------------------------------------------------------ #
    # Phase B : aggressive hill climbs
    for scale in (0.6, 0.4, 0.3):
        if time.time() - start > budget_s * 0.90:
            break
        hc, cc = guided_hill_climb(best_h, rng,
                max_iters=int(1500 * 256 / n), step_scale=scale)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    # ------------------------------------------------------------------ #
    # Phase C : SLSQP refinements with very high beta
    for beta in (200000, 400000):
        if time.time() - start > budget_s * 0.95:
            break
        h = slsqp_optimize(best_h, beta, rng, max_iter=1200)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ------------------------------------------------------------------ #
    # Final polishing with tiny subgradient steps
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
