    step = step_init
    for it in range(max_iters):

        i_max = int(np.argmax(grad))

        if max_eps <= 0:
            # pick random pair
            i_max = rng.integers(0, h.size)

    return best_h, best_c

def upsample(h, rng):
    n = h.size
    n2 = n*2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2/2.0)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by binary search on the shift."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def worst_gradient(h: np.ndarray):
    """Gradient of the worst shift overlap O_k."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 500, step_scale: float = 0.7):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

def upsample(h: np.ndarray, rng: np.random.Generator):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def small_step_descent(h: np.ndarray, target: float,
                       max_time: float, step_init: float,
                       rng: np.random.Generator):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_init
    n = h.size
    while time.time() - start < max_time:
        grad, _ = worst_gradient(h)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if i_max == i_min:
            break
        max_eps = min(h[i_max], 1 - h[i_min])
        if max_eps <= 0:
            i_max = rng.integers(0, n)
            i_min = rng.integers(0, n)
            max_eps = min(h[i_max], 1 - h[i_min])
            if max_eps <= 0:
                continue
        eps = step * (0.5 + rng.random())
        if eps > max_eps:
            eps = max_eps
        h[i_max] -= eps
        h[i_min] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        step *= 0.9995
        if step < 1e-6:
            step = step_init
    return best_h, best_c

def multi_start_optimization(n: int, total_time: float,
                             rng: np.random.Generator,
                             best_h0=None, best_c0=np.inf):
    start = time.time()
    best_h = best_h0
    best_c = best_c0
    sub_time = 1.0                     # seconds per sub‑gradient run
    hill_iters = 500                   # iterations for guided hill‑climb
    while time.time() - start < total_time:
        h = random_feasible(n, rng)
        h_sg, c_sg = subgradient_optimize(h, n / 2.0, sub_time, rng)
        h_hc, c_hc = guided_hill_climb(h_sg, rng,
                                        max_iters=hill_iters,
                                        step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    if "initial_h_values" in globals():

    # Use remaining time for fine‑grained descent, possibly after up‑sampling
    remaining = total_budget - (time.time() - start)
    if remaining > 15.0 and n * 2 <= 1024:
        h_up = upsample(best_h, rng)
        # fine descent on finer grid

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
