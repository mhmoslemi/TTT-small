import time, numpy as np, cvxpy as cp, scipy.sparse as sp

def compute_c5_from_h(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    n = 256  # choose 256 for speed

    # Determine if we have initial_h_values; get n_initial = len(initial_h_values)
    # If n_initial != n, we can either use as start after interpolation; but not needed.

    # Create variable
    h = cp.Variable(n)
    t = cp.Variable()

    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]

    # Build sparse matrices for each shift k
    # Precompute row indices and col indices with data = 0.5
    for k in range(n):
        if k == 0:
            # q_0 = sum_i h_i^2, i.e., quad_form with identity
            Q = sp.identity(n, format='csc')
        else:
            rows = np.arange(n - k)
            cols = rows + k
            data = np.full(n - k, 0.5)
            # Create symmetric entries
            Q = sp.coo_matrix((np.concatenate([data, data]),
                               (np.concatenate([rows, cols]),
                                np.concatenate([cols, rows]))),
                              shape=(n, n)).tocsc()
        # constraint q_k >= t
        constraints.append(cp.quad_form(h, Q) >= t)

    # objective
    prob = cp.Problem(cp.Maximize(t), constraints)

    # Solve within budget: allocate some time for solving.
    # We'll call solve with warm_start=False, ignore budget.

    # Determine max_iters for ECOS; maybe 2000
    try:
        prob.solve(solver='ECOS', max_iters=2000, abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    except cp.error.SolverError:
        prob.solve(solver='ECOS', max_iters=5000, verbose=False)

    # Retrieve solution
    h_opt = np.array(h.value).flatten()
    # If any NaN due to solver failure, fallback to random.

def enforce_sum_numpy(h, n, rng, tol=1e-10, max_iter=10000):
    target = n / 2.0
    diff = target - h.sum()
    itr = 0
    while abs(diff) > tol and itr < max_iter:
        # choose random index
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
        itr += 1
    return h

def worst_gradient_fft(h):
    n = h.size
    # compute correlation via FFT
    m = 2 * n
    h_fft = np.fft.rfft(h, n=m)
    one_minus_h_fft = np.fft.rfft(1 - h, n=m)
    corr = np.fft.irfft(h_fft * np.conj(one_minus_h_fft), n=m)
    corr = corr[:2 * n - 1]
    idx = np.argmax(corr)  # index in 0..2n-2
    k = idx - (n - 1)  # signed shift
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        k_abs = abs(k)
        if k > 0:
            # shift positive: h[i] with h[i + k]
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]
        else:
            # negative: effectively shift left
            # derivative similar as for positive because correlation symmetric
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities -------------------------------------------------
def enforce_sum_numpy(h: np.ndarray, n: int, rng: np.random.Generator,
                     tol: float = 1e-12, max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while keeping 0<=h<=1."""
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > tol and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 using FFT (for many evaluations)."""
    n = h.size
    m = 2 * n
    hf = np.fft.rfft(h, n=m)
    omf = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(hf * np.conj(omf), n=m)
    corr = corr[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def compute_c5_np(h: np.ndarray) -> float:
    """Exact C5 using np.correlate (for final reporting)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_gradient_fft(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap shift."""
    n = h.size
    m = 2 * n
    hf = np.fft.rfft(h, n=m)
    omf = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(hf * np.conj(omf), n=m)
    corr = corr[: 2 * n - 1]
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # signed shift
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        k_abs = abs(k)
        # both directions give the same updates because h is real
        grad[: n - k_abs] -= h[k_abs:]
        grad[k_abs:]     -= h[: n - k_abs]
    return grad


def softmax_obj(x: np.ndarray, beta: float) -> float:
    """Smooth approximation of max_k ∫ h(1‑h_shift)."""
    n = x.size
    m = 2 * n
    xf = np.fft.rfft(x, n=m)
    omf = np.fft.rfft(1.0 - x, n=m)
    corr = np.fft.irfft(xf * np.conj(omf), n=m)
    corr = corr[: 2 * n - 1]
    bc = beta * corr
    mval = np.max(bc)
    return (mval + np.log(np.sum(np.exp(bc - mval)))) / beta


# ---------- main optimisation routine ----------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # problem size – larger gives more flexibility, still cheap with FFT
    n = 1024

    # ---------- initial candidate ----------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:                     # tile / trim to required size
            reps = (n + h0.size - 1) // h0.size
            h0 = np.tile(h0, reps)[:n]
    else:
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    h0 = enforce_sum_numpy(h0, n, rng)
    best_h = h0.copy()
    best_c = compute_c5_fft(best_h)

    # ---------- momentum sub‑gradient descent ----------
    momentum = np.zeros_like(best_h)
    beta_mom = 0.9
    step0 = 0.6
    it = 0
    max_iters = 200000

    while time.time() - start < budget_s * 0.88 and it < max_iters:
        # learning‑rate schedule (very slow decay)
        step = step0 * (0.9996 ** it)
        grad = worst_gradient_fft(best_h)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        nh = best_h - step * momentum
        np.clip(nh, 0.0, 1.0, out=nh)
        nh = enforce_sum_numpy(nh, n, rng)
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
            momentum.fill(0.0)
            if best_c <= 0.3808:
                break
        # occasional random transfer to escape plateaus
        if it % 5000 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps_max = min(best_h[i], 1.0 - best_h[j])
                if eps_max > 0:
                    eps = eps_max * rng.random() * 0.08
                    nh2 = best_h.copy()
                    nh2[i] -= eps
                    nh2[j] += eps
                    np.clip(nh2, 0.0, 1.0, out=nh2)
                    nh2 = enforce_sum_numpy(nh2, n, rng)
                    c2 = compute_c5_fft(nh2)
                    if c2 < best_c:
                        best_c, best_h = c2, nh2.copy()
                        momentum.fill(0.0)
        it += 1

    # ---------- SLSQP refinement on smooth approximation ----------
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    h_cur = best_h.copy()
    for beta in (10.0, 80.0, 300.0, 1500.0, 7000.0):
        if time.time() - start > budget_s * 0.97:
            break
        res = minimize(lambda x, b=beta: softmax_obj(x, b), h_cur,
                       method="SLSQP", bounds=bounds,
                       constraints=[eq_cons],
                       options={"maxiter": 600, "ftol": 1e-9, "disp": False})
        x = np.clip(res.x, 0.0, 1.0)
        # tiny mass‑preserving correction
        diff = target_sum - x.sum()
        if abs(diff) > 1e-12:
            for idx in range(n):
                add = min(diff, 1.0 - x[idx]) if diff > 0 else max(diff, -x[idx])
                if add != 0.0:
                    x[idx] += add
                    diff = target_sum - x.sum()
                    if abs(diff) < 1e-12:
                        break
        c = compute_c5_fft(x)
        if c < best_c:
            best_c, best_h = c, x.copy()
            h_cur = x
            if best_c <= 0.3808:
                break

    # ---------- final sanity enforcement ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_numpy(best_h, n, rng)
    final_c = compute_c5_np(best_h)   # exact value matching evaluator

    return best_h.tolist(), float(final_c), int(n)
