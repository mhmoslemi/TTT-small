def project_onto_bounds_and_sum(x, target):
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # compute sum diff
    s = x.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return x
    # Determine indices that are not at bounds
    # We'll use an iterative method: adjust free variables.
    free = np.logical_and(x > 0.0, x < 1.0)
    # If not enough free, we need to adjust some bound variables by moving them within range.
    # For simplicity, we can distribute diff uniformly among all elements then re-clip and repeat a few times.
    # Use for loop up to 10 times.
    for _ in range(10):
        free = np.logical_and(x > 0.0, x < 1.0)
        nfree = free.sum()
        if nfree == 0:
            # nothing free, adjust all uniformly (ignoring bounds)
            x += diff / x.size
        else:
            delta = diff / nfree
            x[free] += delta
        # reclip
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    return x

import time

# ---------- helper functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] using correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_shift_and_gradient(h: np.ndarray):
    """Return the index of the worst shift and the gradient of that overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]          # (1 - h_{j+k}) term
        grad[k:] -= h[: n - k]               # -h_{j-k} term
    return idx, grad


def project_onto_bounds_and_sum(x: np.ndarray, target: float, max_iter: int = 20):
    """Clip to [0,1] and adjust to satisfy sum(x)=target (exact to tol)."""
    x = np.clip(x, 0.0, 1.0)
    tol = 1e-12
    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) < tol:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = free.sum()
        if nfree == 0:
            delta = diff / x.size
            x += delta
        else:
            delta = diff / nfree
            x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # tiny final correction if needed
    s = x.sum()
    diff = target - s
    if abs(diff) >= tol:
        for i in range(x.size):
            if (diff > 0 and x[i] < 1.0) or (diff < 0 and x[i] > 0.0):

                diff -= adj

    return np.clip(x, 0.0, 1.0)
