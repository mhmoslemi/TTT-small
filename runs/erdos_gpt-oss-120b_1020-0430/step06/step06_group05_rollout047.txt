def enforce_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # eligible indices
    if diff > 0:
        eligible = np.where(h < 1)[0]
        caps = 1 - h[eligible]
    else:
        eligible = np.where(h > 0)[0]
        caps = h[eligible]
    # distribute uniformly
    while True:
        if eligible.size == 0:
            break
        total_cap = caps.sum()
        if total_cap <= 1e-12:
            break
        # fraction of diff to allocate
        frac = min(1.0, abs(diff) / total_cap)
        delta = diff * (caps / total_cap) * (1 if diff>0 else 1)
        # but we want delta_i = caps_i * frac * sign(diff)
        delta = caps * frac * np.sign(diff)
        h[eligible] += delta
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # remove saturated indices
        if diff > 0:
            mask = h < 1 - 1e-12
        else:
            mask = h > 1e-12
        eligible = np.where(mask)[0]
        caps = (1 - h[eligible]) if diff > 0 else h[eligible]
    return h

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1 - 2 * h
    else:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    return k, grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global for constraints ----------
CURRENT_TARGET = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def enforce_sum(h, target):
    """Adjust h to have sum = target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h
            mask = cap > 1e-12
        else:
            cap = h
            mask = cap > 1e-12
        if not np.any(mask):
            break
        total_cap = cap[mask].sum()
        if total_cap < 1e-15:
            break
        delta = np.zeros_like(h)
        delta[mask] = cap[mask] * (diff / total_cap)
        h = h + delta
        diff = target - h.sum()
    # final tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            idx = np.where(free)[0][0]
            h[idx] += diff
        else:
            h += diff / h.size
    np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c5(h):
    """Discrete C5 for step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_shift_and_grad(h):
    """Return the worst shift (k) and its sub‑gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h, grad, alpha, target, rng):
    """Move mass from positive‑gradient to negative‑gradient positions."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not np.any(pos) or not np.any(neg):
        return h.copy()
    total_out = h[pos].sum()
    total_in = (1.0 - h[neg]).sum()
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    h_new[pos] = h[pos] * (1.0 - eps / total_out)
    h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    h_new = enforce_sum(h_new, target)
    return h_new


def softmax_obj(x, beta):
    """Smooth max approximation used by SLSQP."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 1024  # number of samples
    target = n / 2.0

    # ---- initialise h ----
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size == n:
            h = init.copy()
        elif n % init.size == 0:
            repeats = n // init.size
            h = np.repeat(init, repeats)
        else:
            x_old = np.linspace(0, 2, init.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, init)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, int(target), replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_c = compute_c5(best_h)

    max_iters = 20000
    alpha0 = 0.5

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.98:
            break

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            if best_c < 0.38078:  # early success
                break

        k, grad = worst_shift_and_grad(h)

        # ----- try aggressive mass‑transfer moves ----- #
        improved = False
        for alpha in (0.5, 0.25, 0.125, 0.0625):
            h_mt = mass_transfer_move(h, grad, alpha, target, rng)
            c_mt = compute_c5(h_mt)
            if c_mt < c:
                h = h_mt
                improved = True
                break
        if improved:
            continue

        # ----- gradient step with projection ----- #
        step = alpha0 / np.sqrt(it + 1)
        h_new = h - step * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = enforce_sum(h_new, target)
        c_new = compute_c5(h_new)

        if c_new < c or rng.random() < 0.001:
            h = h_new
        else:
            # occasional small random swap
            if rng.random() < 0.01:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps > 1e-12:
                        eps = max_eps * rng.random() * 0.2
                        h[i] -= eps
                        h[j] += eps

        # periodic mild re‑exploration
        if (it + 1) % 5000 == 0:
            h = best_h.copy()
            h += rng.normal(scale=0.02, size=n)
            np.clip(h, 0.0, 1.0, out=h)
            h = enforce_sum(h, target)

    # ----- final deterministic refinement ----- #
    h = best_h.copy()
    for it in range(200):
        if time.time() - start > budget_s * 0.96:
            break
        _, grad = worst_shift_and_grad(h)
        step = 0.02 * (0.99 ** it)
        h = h - step * grad
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, target)

    final_c = compute_c5(h)
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()

    # ----- optional soft‑max SLSQP refinement ----- #
    if time.time() - start < budget_s * 0.94:
        global CURRENT_TARGET
        CURRENT_TARGET = target
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        for beta in (2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.95:
                break
            res = minimize(
                softmax_obj,
                best_h,
                args=(beta,),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 800, "ftol": 1e-12, "disp": False},
            )
            x = np.clip(res.x, 0.0, 1.0)
            x = enforce_sum(x, target)
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
