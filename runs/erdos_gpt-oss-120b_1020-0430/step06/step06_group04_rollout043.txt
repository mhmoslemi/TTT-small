def compute_c5_fft(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    L = int(2**np.ceil(np.log2(2*n - 1)))
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa*Fb, n=L)[:2*n-1]
    return float(np.max(conv) * (2.0 / n))

def binary_sa_fast(h, rng, deadline):
    n = h.size
    C_arr, rev_h = compute_initial_arrays(h)
    cur_c = C_arr.max()
    best_c = cur_c
    best_h = h.copy()
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    delta = np.empty(2*n-1, dtype=np.float64)
    # Temperature schedule
    # Set initial T = max(1e-5, cur_c*0.03) maybe.
    init_T = max(1e-5, cur_c * 0.03)
    it = 0
    while time.time() < deadline:
        # Determine temperature
        # Linear decay: T = init_T * max(0, 1 - it / max_it)
        # But we don't know max_it; we can decay exponentially: T *= 0.9999 each iteration
        T = init_T * np.exp(-it * 0.0005)  # adjust
        # pick random i from ones and j from zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Compute delta
        delta.fill(0.0)
        slice_i = slice(n-1-i, n-1-i + n)
        slice_j = slice(n-1-j, n-1-j + n)
        slice_i_rev = slice(i, i + n)
        slice_j_rev = slice(j, j + n)
        delta[slice_i] += h
        delta[slice_j] -= h
        delta[slice_i_rev] += rev_h
        delta[slice_j_rev] -= rev_h
        new_c = (C_arr + delta).max()
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            # accept
            C_arr += delta
            cur_c = new_c
            # update h
            h[i] = 0
            h[j] = 1
            # update rev_h
            rev_h[n-1-i] = 0
            rev_h[n-1-j] = 1
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                it = 0  # reset stagnation?
        it += 1
        # optional break if stagnation after many iters without improvement
        if it > 2000000:  # extremely many iterations may occur; we can break if time near deadline
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95

    # Determine base n
    base_n = 720
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        # If parent length differs from base_n, interpolate
        if len(parent) != base_n:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            parent_interp = np.interp(xt, xs, parent)
        else:
            parent_interp = np.array(parent, copy=False)
    else:
        parent_interp = None

    # initialize best
    best_h = None
    best_c = np.inf
    best_n = base_n

    # number of restarts for SA
    restarts = 3
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        # initial binary vector
        if r == 0 and parent_interp is not None:
            h0 = np.rint(parent_interp).astype(float)
        else:

            h0 = np.zeros(n, dtype=float)

        # Ensure exact sum (though binary with n/2 ones already)
        # but if sum not n/2 due to rounding, adjust:
        adjust_sum_exact(h0, rng)
        # run SA fast
        # time remaining for this restart: allocate proportionally
        # e.g., allocate (deadline - time.time()) / (restarts - r)
        rem_deadline = deadline - (time.time() - start) * 0.0  # will compute later
        # we'll just give each restart until deadline
        h_opt, c_opt = binary_sa_fast(h0, rng, deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = n

    # Possibly upscale to double size if time left
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        # upsample by repeat
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        # Ensure sum exact (should be n_up/2)

        adjust_sum_exact(h_up, rng)

            best_n = n_up

def compute_delta(h, rev_h, i, j, delta):
    n = h.size
    delta.fill(0.0)
    # base terms
    delta[n-1-i : n-1-i + n] += h
    delta[n-1-j : n-1-j + n] -= h
    delta[i : i + n] += rev_h
    delta[j : j + n] -= rev_h
    # const for i+k in range but not j+k
    # i_range: start_i = -i, end_i = n-1-i
    # j_range: start_j = -j, end_j = n-1-j
    start_i = -i
    end_i = n-1-i
    start_j = -j
    end_j = n-1-j
    # Note: these are shift values, need to convert to idx = k + (n-1)
    # intervals where i in range, j not:
    # left part: if start_i < start_j:
    if start_i < start_j:
        idx_start = start_i + n -1
        idx_end = start_j -1 + n -1
        delta[idx_start: idx_end+1] -= 1.0
    # right part: if end_i > end_j:
    if end_i > end_j:
        idx_start = end_j +1 + n -1
        idx_end = end_i + n -1
        delta[idx_start: idx_end+1] -= 1.0
    # intervals where j in range, i not:
    if start_j < start_i:
        idx_start = start_j + n -1
        idx_end = start_i -1 + n -1
        delta[idx_start: idx_end+1] += 1.0

        idx_end = end_j + n -1

    # correction for double-counted pairs
    delta[n-1] -= 2.0
    diff = j - i

    delta[-diff + n -1] += 1.0

def binary_sa_fast(h, rng, deadline):
    n = h.shape[0]
    # precompute correlation using np.correlate
    C_arr = np.correlate(h, 1 - h, mode='full')
    cur_c = C_arr.max()
    best_c = cur_c
    best_h = h.copy()
    # rev_h
    rev_h = h[::-1].copy()
    delta = np.empty(2*n - 1, dtype=np.float64)
    # precompute static values
    n_minus_one = n - 1
    init_T = max(1e-8, float(cur_c) * 0.1)
    alpha = 5e-5
    it = 0
    while time.time() < deadline:
        T = init_T * np.exp(-alpha * it)
        # random i with h[i]==1
        i = rng.integers(0, n)
        while h[i] != 1.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 0.0:
            j = rng.integers(0, n)
        # compute delta
        delta.fill(0.0)
        # base slices
        delta[n_minus_one - i : n_minus_one - i + n] += h
        delta[n_minus_one - j : n_minus_one - j + n] -= h
        delta[i : i + n] += rev_h
        delta[j : j + n] -= rev_h
        # const intervals
        start_i = -i
        end_i = n_minus_one - i
        start_j = -j
        end_j = n_minus_one - j
        # i range not j
        if start_i < start_j:
            idx_start = start_i + n_minus_one
            idx_end = start_j - 1 + n_minus_one
            delta[idx_start: idx_end + 1] -= 1.0
        if end_i > end_j:
            idx_start = end_j + 1 + n_minus_one
            idx_end = end_i + n_minus_one
            delta[idx_start: idx_end + 1] -= 1.0
        # j range not i
        if start_j < start_i:
            idx_start = start_j + n_minus_one
            idx_end = start_i - 1 + n_minus_one
            delta[idx_start: idx_end + 1] += 1.0
        if end_j > end_i:
            idx_start = end_i + 1 + n_minus_one
            idx_end = end_j + n_minus_one
            delta[idx_start: idx_end + 1] += 1.0
        # correction for double counting
        delta[n_minus_one] -= 2.0
        diff = j - i
        delta[diff + n_minus_one] += 1.0
        delta[-diff + n_minus_one] += 1.0

        # compute new max
        new_c = (C_arr + delta).max()
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-12)):
            C_arr += delta
            cur_c = new_c
            # update h and rev_h
            h[i] = 0.0
            h[j] = 1.0
            rev_h[n_minus_one - i] = 0.0
            rev_h[n_minus_one - j] = 1.0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                it = 0  # reset iteration counter for stagnation
        it += 1
        # optional break condition for long loops
        if it > 5000000:
            break
    return best_h, float(best_c)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Modify h in‑place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def binary_sa_fast(h, rng, deadline):
    """Simulated‑annealing on binary vectors using an O(n) delta update."""
    n = h.size
    C_arr = np.correlate(h, 1.0 - h, mode="full")
    cur_c = C_arr.max()
    best_c = cur_c
    best_h = h.copy()
    rev_h = h[::-1].copy()
    delta = np.empty(2 * n - 1, dtype=np.float64)
    n_minus_one = n - 1
    init_T = max(1e-8, float(cur_c) * 0.1)
    alpha = 5e-5
    it = 0
    while time.time() < deadline:
        T = init_T * np.exp(-alpha * it)

        # pick a 1 and a 0
        i = rng.integers(0, n)
        while h[i] != 1.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 0.0:
            j = rng.integers(0, n)

        # ---- delta computation ----
        delta.fill(0.0)
        delta[n_minus_one - i : n_minus_one - i + n] += h
        delta[n_minus_one - j : n_minus_one - j + n] -= h
        delta[i : i + n] += rev_h
        delta[j : j + n] -= rev_h

        # constant offsets where only one side is in range
        start_i = -i
        end_i = n_minus_one - i
        start_j = -j
        end_j = n_minus_one - j

        if start_i < start_j:
            a = start_i + n_minus_one
            b = start_j - 1 + n_minus_one
            delta[a : b + 1] -= 1.0
        if end_i > end_j:
            a = end_j + 1 + n_minus_one
            b = end_i + n_minus_one
            delta[a : b + 1] -= 1.0
        if start_j < start_i:
            a = start_j + n_minus_one
            b = start_i - 1 + n_minus_one
            delta[a : b + 1] += 1.0
        if end_j > end_i:
            a = end_i + 1 + n_minus_one
            b = end_j + n_minus_one
            delta[a : b + 1] += 1.0

        # correct double‑counted pairs (k=0 and k=±(j‑i))
        delta[n_minus_one] -= 2.0
        diff = j - i
        delta[diff + n_minus_one] += 1.0
        delta[-diff + n_minus_one] += 1.0

        new_c = (C_arr + delta).max()

        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (T + 1e-12)):
            C_arr += delta
            cur_c = new_c
            h[i] = 0.0
            h[j] = 1.0
            rev_h[n_minus_one - i] = 0.0
            rev_h[n_minus_one - j] = 1.0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                it = 0  # reset stagnation counter
        it += 1
        if it > 5_000_000:
            break
    return best_h, float(best_c)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96  # safety margin

    base_n = 720
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        if len(parent) != base_n:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            parent_interp = np.interp(xt, xs, parent)
        else:
            parent_interp = np.array(parent, copy=False)
    else:
        parent_interp = None

    # initial binary vector
    if parent_interp is not None:
        h0 = np.rint(parent_interp).astype(float)
    else:
        h0 = np.zeros(base_n, dtype=float)
        h0[: base_n // 2] = 1.0
    adjust_sum_exact(h0, rng)

    # main simulated‑annealing phase
    best_h, best_c = binary_sa_fast(h0.copy(), rng, deadline)

    # short greedy polishing if time permits
    if time.time() < deadline:
        best_h, best_c = greedy_improve(best_h, rng, deadline)

    # optional up‑sampling and second SA phase
    if best_h is not None and time.time() < deadline and base_n * 2 <= 1440:
        up_n = base_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        best_h_up, best_c_up = binary_sa_fast(h_up, rng, deadline)
        if best_c_up < best_c:
            best_h, best_c = best_h_up, best_c_up
            base_n = up_n
        if time.time() < deadline:
            best_h, best_c = greedy_improve(best_h, rng, deadline)

    final_c = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h.tolist(), float(final_c), int(base_n)
