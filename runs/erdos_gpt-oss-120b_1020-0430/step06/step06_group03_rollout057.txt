def compute_c5_fft(h):
    n = h.shape[0]
    N = 2*n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # final refinement
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1-1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - (lo+hi)*0.5))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def differential_evolution(initial_pop, target, runtime, start_time):
    n = initial_pop.shape[1]
    pop = initial_pop.copy()
    pop_f = np.empty(pop.shape[0])
    for i in range(pop.shape[0]):
        pop_f[i] = compute_c5_fft(pop[i])
    best_idx = np.argmin(pop_f)
    best_h = pop[best_idx].copy()
    best_f = pop_f[best_idx]
    F = 0.6
    CR = 0.9
    # generation loop
    gen = 0
    while True:
        # time check
        if time.time() - start_time > runtime:
            break
        gen += 1
        for i in range(pop.shape[0]):
            # random distinct indices
            idxs = rng.choice(pop.shape[0], 3, replace=False)
            while i in idxs:
                idxs = rng.choice(pop.shape[0], 3, replace=False)
            a,b,c = idxs
            mutant = pop[a] + F * (pop[b] - pop[c])
            # clip then project
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, target)
            # crossover
            cross_rand = rng.random(n)
            trial = np.where(cross_rand < CR, mutant, pop[i])
            # ensure feasibility
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, target)
            f_trial = compute_c5_fft(trial)
            if f_trial < pop_f[i]:
                pop[i] = trial
                pop_f[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = trial.copy()
        # optional break if no improvement for many generations
        # but we can rely on runtime limit.
    return best_h, best_f, gen

def hill_climb(h, rng, max_iters, init_step, target):
    best_h = h.copy()
    best_f = compute_c5_fft(best_h)
    step = init_step
    for it in range(max_iters):
        # time check maybe
        # pick i,j random
        i, j = rng.integers(0, h.size, size=2)
        if i == j:
            continue
        # Determine max epsilon to transfer from i to j
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        f_new = compute_c5_fft(new_h)
        if f_new < best_f:
            best_f = f_new
            best_h = new_h
        # schedule step decay
        if (it+1) % (max_iters // 10 + 1) == 0:
            step *= 0.8
    return best_h, best_f

import time
import numpy as np

def project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    N = 2 * n - 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return corr

def compute_c5_fft(h: np.ndarray) -> float:
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    corr = compute_corr_fft(h)
    max_val = corr.max()
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            left = np.zeros_like(h)
            right = np.zeros_like(h)
            if k > 0:
                left[:-k] = h[k:]
                right[k:] = h[:-k]
            else:
                kk = -k
                left[kk:] = h[:-kk]
                right[:-kk] = h[kk:]
            grad += 1.0 - left - right
    grad /= len(idxs)
    return grad

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int, step_init: float, target: float):
    best_h = h.copy()
    best_f = compute_c5_fft(best_h)
    step = step_init
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i = rng.integers(0, h.size)
            j = rng.integers(0, h.size)
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        f_new = compute_c5_fft(new_h)
        if f_new < best_f:
            best_f = f_new
            best_h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_f
