def real_ga(n, popsize, time_budget, rng):
    target = n/2
    # init population
    pop = []
    for _ in range(popsize):
        v = rng.random(n)  # uniform
        v = project_to_feasible(v, target)
        pop.append(v)
    # track best
    best = None
    best_c = np.inf
    # main loop
    start = time.time()
    while time.time() - start < time_budget:
        # tournament selection
        idxs = np.arange(popsize)
        parents = []
        for _ in range(popsize):
            # select 3 random indices
            sel = rng.choice(idxs, size=3, replace=False)
            # pick best
            sel_c = [compute_c5(pop[i]) for i in sel]
            best_sel = sel[np.argmin(sel_c)]
            parents.append(pop[best_sel])
        # crossover and mutation to produce new pop
        new_pop = []
        for i in range(0, popsize, 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < popsize else parents[0]
            # blend crossover
            alpha = rng.random()
            child1 = alpha * p1 + (1-alpha) * p2
            child2 = (1-alpha) * p1 + alpha * p2
            # mutation: add small Gaussian noise to random fraction of elements
            for child in (child1, child2):
                if rng.random() < 0.5:
                    # choose m elements to mutate
                    m = max(1, int(0.05 * n))
                    idxs_mut = rng.choice(n, m, replace=False)
                    child[idxs_mut] += rng.normal(scale=0.1, size=m)
                # project
                child = project_to_feasible(child, target)
                child = np.clip(child, 0.0, 1.0)
                new_pop.append(child)
        # evaluate and keep elite (best from old pop)
        # compute fitness for new pop
        # We'll maybe keep best individual overall
        # replace pop with new_pop (size ~ popsize)
        pop = new_pop[:popsize]
        # evaluate best in pop
        for indiv in pop:
            c = compute_c5(indiv)
            if c < best_c:
                best_c = c
                best = indiv.copy()
                if best_c <= 0.3808:
                    return best, best_c
    if best is None:
        # fallback: return best from initial pop
        best = pop[0]
        best_c = compute_c5(best)
    return best, best_c

    v = np.clip(v, 0.0, 1.0)
    # if sum equals target within tolerance, return
    for _ in range(100):
        s = v.sum()
        diff = target - s

        # compute mask of free variables

            mask = v < 1.0

def binary_ga(n, popsize, time_budget, rng):
    target = n//2 # integer
    # init population
    pop = []
    for _ in range(popsize):
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, target, replace=False)
        h[ones] = 1.0
        pop.append(h)
    best = None; best_c = np.inf
    start = time.time()
    while time.time() - start < time_budget:
        fitness = np.array([compute_c5(ind) for ind in pop])
        idx_best = np.argmin(fitness)
        if fitness[idx_best] < best_c:
            best_c = float(fitness[idx_best]); best = pop[idx_best].copy()
            if best_c <= 0.3808:
                return best, best_c
        # selection
        parents = []
        for _ in range(popsize):
            sel = rng.choice(popsize, size=3, replace=False)
            best_sel = sel[np.argmin(fitness[sel])]
            parents.append(pop[best_sel])
        # crossover and mutation
        new_pop = []
        for i in range(0, popsize, 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < popsize else parents[0]
            # single-point crossover
            cp = rng.integers(1, n)  # crossover point
            child1 = np.concatenate([p1[:cp], p2[cp:]])
            child2 = np.concatenate([p2[:cp], p1[cp:]])
            # mutation: swap bits at probability p_mut_per_individual
            # Choose two indices i,j and swap bits with probability
            # We'll do random swaps
            for child in (child1, child2):
                if rng.random() < 0.5:
                    # perform up to 3 swaps
                    num_swaps = rng.integers(1, 4)
                    for _ in range(num_swaps):
                        i1 = rng.integers(n)
                        i2 = rng.integers(n)
                        child[i1], child[i2] = child[i2], child[i1]
                # repair to have exactly target ones
                cur = int(child.sum())
                diff = cur - target
                if diff > 0:
                    # turn diff ones to zeros
                    ones_idx = np.flatnonzero(child == 1.0)
                    if len(ones_idx) > 0:
                        turn_off = rng.choice(ones_idx, diff, replace=False)
                        child[turn_off] = 0.0
                elif diff < 0:
                    zeros_idx = np.flatnonzero(child == 0.0)
                    if len(zeros_idx) > 0:
                        turn_on = rng.choice(zeros_idx, -diff, replace=False)
                        child[turn_on] = 1.0
                new_pop.append(child)
        pop = new_pop[:popsize]
    if best is None:
        best = pop[0]
        best_c = compute_c5(best)
    return best, best_c

import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    attempts = 0
    n = h.size
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ----------------------------------------------------------------------
# Projection onto bounded simplex (0≤x≤1, sum=target)
def project_to_feasible(v: np.ndarray, target: float, max_iter: int = 200) -> np.ndarray:
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if not np.any(mask):
            break
        delta = diff / np.count_nonzero(mask)
        v[mask] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

# Binary Genetic Algorithm
def binary_ga(n: int, popsize: int, time_budget: float, rng: np.random.Generator):
    target = n // 2
    pop = []
    for _ in range(popsize):
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, target, replace=False)
        h[ones] = 1.0
        pop.append(h)
    best = None
    best_c = np.inf
    start = time.time()
    while time.time() - start < time_budget:
        fitness = np.array([compute_c5(ind) for ind in pop])
        idx_best = np.argmin(fitness)
        if fitness[idx_best] < best_c:
            best_c = float(fitness[idx_best])
            best = pop[idx_best].copy()
            if best_c <= 0.3808:
                return best, best_c
        # tournament selection
        parents = []
        for _ in range(popsize):
            sel = rng.choice(popsize, size=3, replace=False)
            parents.append(pop[sel[np.argmin(fitness[sel])]])
        # crossover & mutation
        new_pop = []
        for i in range(0, popsize, 2):
            p1 = parents[i]
            p2 = parents[i + 1] if i + 1 < popsize else parents[0]
            cp = rng.integers(1, n)  # crossover point
            c1 = np.concatenate([p1[:cp], p2[cp:]])
            c2 = np.concatenate([p2[:cp], p1[cp:]])
            for child in (c1, c2):
                # random swaps
                if rng.random() < 0.5:
                    swaps = rng.integers(1, 4)
                    for _ in range(swaps):
                        i1 = rng.integers(0, n)
                        i2 = rng.integers(0, n)
                        child[i1], child[i2] = child[i2], child[i1]
                # repair to exact number of ones
                cur = int(child.sum())
                diff = cur - target
                if diff > 0:
                    idx = np.flatnonzero(child == 1.0)
                    if len(idx):
                        child[rng.choice(idx, diff, replace=False)] = 0.0
                elif diff < 0:
                    idx = np.flatnonzero(child == 0.0)
                    if len(idx):
                        child[rng.choice(idx, -diff, replace=False)] = 1.0
                new_pop.append(child)
        pop = new_pop[:popsize]
    if best is None:
        best = pop[0]
        best_c = compute_c5(best)
    return best, best_c

# Real-valued Genetic Algorithm
def real_ga(n: int, popsize: int, time_budget: float, rng: np.random.Generator):
    target = n / 2.0
    pop = [project_to_feasible(rng.random(n), target) for _ in range(popsize)]
    best = None
    best_c = np.inf
    start = time.time()
    while time.time() - start < time_budget:
        fitness = np.array([compute_c5(ind) for ind in pop])
        idx_best = np.argmin(fitness)
        if fitness[idx_best] < best_c:
            best_c = float(fitness[idx_best])
            best = pop[idx_best].copy()
            if best_c <= 0.3808:
                return best, best_c
        # tournament selection
        parents = []
        for _ in range(popsize):
            sel = rng.choice(popsize, size=3, replace=False)
            parents.append(pop[sel[np.argmin(fitness[sel])]])
        # crossover & mutation
        new_pop = []
        for i in range(0, popsize, 2):
            p1 = parents[i]
            p2 = parents[i + 1] if i + 1 < popsize else parents[0]
            alpha = rng.random()
            c1 = alpha * p1 + (1 - alpha) * p2
            c2 = (1 - alpha) * p1 + alpha * p2
            for child in (c1, c2):
                # mutate a fraction of entries
                if rng.random() < 0.7:
                    m = max(1, int(0.05 * n))
                    idx = rng.choice(n, m, replace=False)
                    child[idx] += rng.normal(scale=0.08, size=m)
                child = project_to_feasible(child, target)
                child = np.clip(child, 0.0, 1.0)
                new_pop.append(child)
        pop = new_pop[:popsize]
    if best is None:
        best = pop[0]
        best_c = compute_c5(best)
    return best, best_c

# Soft‑max SLSQP optimisation (global helpers)
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# Gradient of the worst shift and fine‑tuning
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
    return g


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

# Continuous simulated annealing
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- Initial solution (if provided) ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---- Phase 0: Binary GA on coarse grid (256) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 20.0:
        t_bin = min(remaining * 0.15, 250.0)
        h_bin, c_bin = binary_ga(256, popsize=30, time_budget=t_bin, rng=rng)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, 256
            if best_c <= 0.3808:
                best_h = np.clip(best_h, 0.0, 1.0)
                best_h = enforce_sum(best_h, rng)
                return best_h.tolist(), float(best_c), int(best_h.size)

    # ---- Phase 1: Upsample to medium resolution (512) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0 and best_h is not None:
        if best_n < 512:
            h_mid = upsample_h(best_h, 512)
            h_mid = np.clip(h_mid, 0.0, 1.0)
            h_mid = enforce_sum(h_mid, rng)
            c_mid = compute_c5(h_mid)
            if c_mid < best_c:
                best_h, best_c, best_n = h_mid, c_mid, 512
        else:
            # already at >=512
            best_h = enforce_sum(best_h, rng)
            best_c = compute_c5(best_h)
            best_n = best_h.size

    # ---- Phase 2: Real‑valued GA on 512 points ----
    remaining = total_budget - (time.time() - start)
    if remaining > 30.0 and best_h is not None and best_n == 512:
        t_real = min(remaining * 0.30, 300.0)
        h_real, c_real = real_ga(512, popsize=30, time_budget=t_real, rng=rng)
        if c_real < best_c:
            best_h, best_c, best_n = h_real, c_real, 512
            if best_c <= 0.3808:
                best_h = np.clip(best_h, 0.0, 1.0)
                best_h = enforce_sum(best_h, rng)
                return best_h.tolist(), float(best_c), int(best_h.size)

    # ---- Phase 3: Soft‑max SLSQP refinement (512) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 20.0 and best_h is not None and best_n == 512:
        h = best_h.copy()
        for beta in (50., 200., 800., 3000., 12000., 40000.):
            if time.time() - start > total_budget * 0.95:
                break
            iters = 800 if beta >= 3000 else 500
            h = slsqp_optimize(h, beta, rng, max_iter=iters)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
                if best_c <= 0.3808:
                    best_h = np.clip(best_h, 0.0, 1.0)
                    best_h = enforce_sum(best_h, rng)
                    return best_h.tolist(), float(best_c), int(best_h.size)

    # ---- Phase 4: Gradient tidy‑up (512) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0 and best_h is not None:
        h = gradient_refine(best_h, rng, step=0.015, max_iter=15)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
            if best_c <= 0.3808:
                best_h = np.clip(best_h, 0.0, 1.0)
                best_h = enforce_sum(best_h, rng)
                return best_h.tolist(), float(best_c), int(best_h.size)

    # ---- Phase 5: Upsample to fine resolution (1024) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 15.0 and best_h is not None:
        if best_n < 1024:
            h_fine = upsample_h(best_h, 1024)
            h_fine = np.clip(h_fine, 0.0, 1.0)
            h_fine = enforce_sum(h_fine, rng)
            c_fine = compute_c5(h_fine)
            if c_fine < best_c:
                best_h, best_c, best_n = h_fine, c_fine, 1024
        else:
            best_n = best_h.size

    # ---- Phase 6: Continuous SA on the finest grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h, c = continuous_sa(best_h, rng, remaining)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- Final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
