def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    # zero shift weight
    wk = w[n-1]
    grad += wk * (1.0 - 2.0 * h)
    # for k = 1 .. n-1
    for k in range(1, n):
        wk = w[n-1 + k] + w[n-1 - k]
        if wk == 0.0:
            continue
        # gradient contribution for shift k:
        # + wk for entries where h appears (since term -h*h_{i+k} etc)?
        # The earlier code added wk for all entries then subtract wk*h shifted.

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 512
        target = n / 2.0
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # optimisation parameters
    total_steps = 300000
    betas = [20, 100, 500, 2000, 8000, 20000, 80000]
    steps_per_beta = max(1, total_steps // len(betas))
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    for beta in betas:
        # reset Adam state for this beta
        m = np.zeros_like(best_h)
        v = np.zeros_like(best_h)
        beta1_pow = 1.0
        beta2_pow = 1.0
        # learning rate scaled with beta
        lr = 0.05 / np.sqrt(beta)
        h_cur = best_h.copy()
        for t in range(1, steps_per_beta + 1):
            if time.time() - start > budget_s * 0.965:
                break
            grad = softmax_gradient(h_cur, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1 - beta1_pow)
            v_hat = v / (1 - beta2_pow)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h_cur - step
            h_new = project_onto_box_sum(h_new, target)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h_cur = h_new
            # occasional random jitter to escape plateaus
            if t % 5000 == 0 and rng.random() < 0.02:

                eps_pair = rng.random() * 0.05

                    h_cur = tmp.copy()

        i = int(np.argmax(grad))

                h_new = best_h.copy()

                    best_h = h_new

    # final random pair perturbations
    while time.time() - start < budget_s * 0.998:
        i, j = rng.integers(n, size=2)
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps_move = max_eps * rng.random()
        h_new = best_h.copy()
        h_new[i] -= eps_move
        h_new[j] += eps_move
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_onto_box_sum(np.clip(best_h, 0.0, 1.0), target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
