Pseudo:

def proj_box_sum(v, target):
    v = np.clip(v, 0, 1)
    residual = target - v.sum()
    while abs(residual) > 1e-12:
        if residual > 0:
            free = np.where(v < 1)[0]
        else:
            free = np.where(v > 0)[0]
        if free.size == 0:
            break
        delta = residual / free.size
        # apply delta but don't exceed bounds
        if residual > 0:
            add = np.minimum(delta, 1 - v[free])
            v[free] += add
            residual -= add.sum()
        else:
            sub = np.maximum(delta, -v[free])  # delta negative
            v[free] += sub
            residual -= sub.sum()
    return v

def proj_box_sum(v, target):
    # Clip to [0,1] first
    v = np.clip(v, 0.0, 1.0)
    residual = target - v.sum()
    # Use iterative water-filling
    max_iter = 1000
    it = 0
    while abs(residual) > 1e-12 and it < max_iter:
        if residual > 0:
            free = np.where(v < 1.0)[0]
        else:
            free = np.where(v > 0.0)[0]
        if free.size == 0:
            break
        delta = residual / free.size
        if residual > 0:
            add = np.minimum(delta, 1.0 - v[free])
            v[free] += add
            residual -= add.sum()
        else:
            sub = np.maximum(delta, -v[free])
            v[free] += sub
            residual -= sub.sum()
        it += 1
    return v

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions (top level)
# ------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with given samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def proj_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while forcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    residual = target - v.sum()
    max_iter = 200
    it = 0
    while abs(residual) > 1e-12 and it < max_iter:
        if residual > 0:
            free = np.where(v < 1.0)[0]
        else:
            free = np.where(v > 0.0)[0]
        if free.size == 0:
            break
        delta = residual / free.size
        if residual > 0:
            add = np.minimum(delta, 1.0 - v[free])
            v[free] += add
            residual -= add.sum()
        else:
            sub = np.maximum(delta, -v[free])
            v[free] += sub
            residual -= sub.sum()
        it += 1
    return v


def upsample(vec: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of vec (defined on [0,2]) to n_target points."""
    n_src = vec.size
    if n_src == n_target:
        return vec.copy()
    x_src = np.linspace(0.0, 2.0, n_src)
    x_tgt = np.linspace(0.0, 2.0, n_target)
    return np.interp(x_tgt, x_src, vec)


# ------------------------------------------------------------
# Core optimisation routine
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------
    # Parameters
    # --------------------------------------------------------
    n = 1024                     # number of samples (even)
    target = n / 2.0
    mu = 0.9                     # momentum
    step0 = 0.30                 # initial step size
    max_iters = 80000

    # --------------------------------------------------------
    # Initialise h
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = upsample(np.array(initial_h_values, dtype=float), n)
    else:
        h0 = np.full(n, 0.5)

    h0 = proj_box_sum(h0, target)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # Main projected gradient descent with occasional jumps
    # --------------------------------------------------------
    h_cur = h0.copy()
    v = np.zeros_like(h_cur)
    last_improve = 0

    it = 0
    while it < max_iters and time.time() - start < budget_s * 0.93:
        # correlation and worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = int(np.argmax(corr) - (n - 1))
        k = abs(idx)

        # gradient of the worst overlap
        grad = np.ones_like(h_cur)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h_cur
        else:
            grad[: n - k] -= h_cur[k:]
            grad[k:] -= h_cur[: n - k]

        # step size (slow decay)
        step = step0 / (1.0 + it * 0.001)

        # momentum update and projection
        v = mu * v - step * grad
        h_new = proj_box_sum(h_cur + v, target)

        # evaluate new candidate
        c_new = compute_c5(h_new)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            last_improve = it
            v[:] = 0.0                     # reset momentum after improvement

        h_cur = h_new                     # continue from new point

        # ----------------------------------------------------
        # Stagnation handling – occasional mass swaps
        # ----------------------------------------------------
        if it - last_improve > 5000:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_transfer = min(h_cur[i], 1.0 - h_cur[j])
                if max_transfer > 0:
                    eps = rng.random() * max_transfer * 0.05
                    h_swap = h_cur.copy()
                    h_swap[i] -= eps
                    h_swap[j] += eps
                    h_swap = proj_box_sum(h_swap, target)
                    c_swap = compute_c5(h_swap)
                    if c_swap < best_c:
                        best_c = c_swap
                        best_h = h_swap.copy()
                        last_improve = it
                        v[:] = 0.0

        # Larger random jumps after longer stagnation
        if it - last_improve > 7000:
            h_cur = best_h + rng.normal(scale=0.02, size=n)
            h_cur = proj_box_sum(np.clip(h_cur, 0.0, 1.0), target)
            v[:] = 0.0
            last_improve = it

        # Full random binary restart after very long stagnation
        if it - last_improve > 15000:
            rand = np.zeros(n)
            rand[rng.choice(n, n // 2, replace=False)] = 1.0
            h_cur = proj_box_sum(rand, target)
            v[:] = 0.0
            last_improve = it

        it += 1

    # --------------------------------------------------------
    # Final fine‑grained random tweaks until time expires
    # --------------------------------------------------------
    while time.time() - start < budget_s * 0.99:
        h_try = best_h + rng.normal(scale=0.01, size=n)
        h_try = proj_box_sum(np.clip(h_try, 0.0, 1.0), target)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # --------------------------------------------------------
    # Final sanitisation and return
    # --------------------------------------------------------
    best_h = proj_box_sum(np.clip(best_h, 0.0, 1.0), target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
