def enforce_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # if diff positive, add mass
    if diff > 0:
        mask = h < 1.0
        capacity = np.sum(1.0 - h[mask])
        if capacity < 1e-12:
            # fallback random
            while diff > 1e-12:
                i = rng.integers(0, h.size)
                cap = 1.0 - h[i]
                if cap <= 0: continue
                delta = min(diff, cap)
                h[i] += delta
                diff -= delta
            return h
        add = diff * (1.0 - h[mask]) / capacity
        h[mask] += add
    else:
        # diff negative, subtract mass
        mask = h > 0.0
        capacity = np.sum(h[mask])
        if capacity < 1e-12:
            while diff < -1e-12:
                i = rng.integers(0, h.size)
                cap = h[i]
                if cap <= 0: continue
                delta = min(-diff, cap)
                h[i] -= delta
                diff += delta
            return h
        sub = -diff * h[mask] / capacity
        h[mask] -= sub
    # small correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # fallback recursive call with limited iterations?
        # just adjust one random index

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)  # index in 0..2n-2
    k = idx - (n - 1)
    if k < 0:
        k = -k
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

import time
import numpy as np

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if diff > 0:
        mask = h < 1.0
        cap = 1.0 - h[mask]
        total = cap.sum()
        if total < 1e-12:
            while diff > 1e-12:
                i = rng.integers(0, n)
                inc = min(diff, 1.0 - h[i])
                if inc <= 0:
                    continue
                h[i] += inc
                diff -= inc
            return h
        h[mask] += diff * cap / total
    else:
        mask = h > 0.0
        cap = h[mask]
        total = cap.sum()
        if total < 1e-12:
            while diff < -1e-12:
                i = rng.integers(0, n)
                dec = min(-diff, h[i])
                if dec <= 0:
                    continue
                h[i] -= dec
                diff += dec
            return h
        h[mask] -= (-diff) * cap / total
    # tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def c5_and_grad(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    c = corr[idx] * (2.0 / n)
    k = idx - (n - 1)
    if k < 0:
        k = -k
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return c, grad


def random_h(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    h = enforce_sum(h, target, rng)
    return h


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024
    target = n / 2.0

    best_h = None
    best_c = np.inf

    # start from upsampled parent if available
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        if n % n0 == 0:
            rep = n // n0
            h0 = np.repeat(base, rep)
        else:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, base)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_c = c0
        if best_c < 0.3808:
            return best_h.tolist(), float(best_c), n

    steps_per_restart = 200
    step0 = 0.1
    restart = 0

    while True:
        if time.time() - start > budget_s * 0.95:
            break
        # generate a start point
        if best_h is not None and rng.random() < 0.1:
            h = best_h.copy()
            for _ in range(3):
                h = random_perturb(h, rng, scale=0.2)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, target, rng)
        else:
            h = random_h(n, target, rng)

        for it in range(steps_per_restart):
            if time.time() - start > budget_s * 0.95:
                break
            c, grad = c5_and_grad(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                if best_c < 0.3808:
                    return best_h.tolist(), float(best_c), n
            step = step0 * (0.95 ** it)
            h -= step * grad
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, target, rng)
            if rng.random() < 0.05:
                h = random_perturb(h, rng, scale=0.2)
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    h = enforce_sum(h, target, rng)
        restart += 1

    # final intensive refinement on the best found
    if best_h is not None:
        h = best_h.copy()
        for it in range(5000):
            if time.time() - start > budget_s * 0.98:
                break
            c, grad = c5_and_grad(h)
            step = 0.02 * (0.99 ** it)
            h -= step * grad
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, target, rng)
        final_c = compute_c5(h)
        if final_c < best_c:
            best_c = final_c
            best_h = h.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
