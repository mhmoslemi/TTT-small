def worst_shift_and_grad(h):
    # compute correlation
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    corr = corr[:2*n-1]
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        k = shift
        if k < n:
            grad[:n - k] = 1.0 - h[k:]
            grad[k:] -= h[:n - k]
    else:
        p = -shift
        if p < n:
            grad[p:] = 1.0 - h[:n - p]
            grad[:n - p] -= h[p:]
    return shift, grad
```

Now mass_transfer_move:

```
def mass_transfer_move(h, grad, alpha, target, rng):
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not np.any(pos) or not np.any(neg):
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out <= 0.0 or total_in <= 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    # clip
    np.clip(h_new, 0.0, 1.0, out=h_new)
    # enforce sum
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new
```

def random_perturb(h, rng, scale=0.2):
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # while loop but limit iterations
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h
```

import time
import numpy as np

# problem size – will be set in run()
n = 0
N = 0


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    size = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, size)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def evaluate_h(h: np.ndarray) -> float:
    """Exact discrete C5 for the current step function."""
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)


def worst_shift_and_grad(h: np.ndarray):
    """Return (shift, grad) for the shift attaining maximal overlap."""
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    corr = corr[: 2 * n - 1]
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # k such that corr[k] is maximal
    grad = np.zeros_like(h)
    if shift >= 0:
        k = shift
        if k < n:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
    else:
        p = -shift
        if p < n:
            grad[p:] = 1.0 - h[: n - p]
            grad[: n - p] -= h[p:]
    return shift, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient to negative‑gradient indices."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not np.any(pos) or not np.any(neg):
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out <= 0.0 or total_in <= 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves the total sum."""
    size = h.size
    i = rng.integers(0, size)
    j = rng.integers(0, size)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def local_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 max_iter: int = 200):
    """Iterative refinement using gradient‑guided mass transfers."""
    best_h = h
    best_c = evaluate_h(best_h)
    alphas = (0.5, 0.25, 0.125, 0.0625, 0.03125,
              0.015625, 0.0078125, 0.00390625, 0.001953125)
    for _ in range(max_iter):
        shift, grad = worst_shift_and_grad(best_h)
        improved = False
        for a in alphas:
            h_new = mass_transfer_move(best_h, grad, a, target, rng)
            c_new = evaluate_h(h_new)
            if c_new < best_c - 1e-12:
                best_h, best_c = h_new, c_new
                improved = True
                break
        if not improved:
            h_new = random_perturb(best_h, rng, scale=0.2)
            c_new = evaluate_h(h_new)
            if c_new < best_c - 1e-12:
                best_h, best_c = h_new, c_new
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    global n, N
    n = 4096
    target = n / 2.0
    N = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1

    # ----- initialise h ----------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.shape[0] == n:
            h = h0.copy()
        elif n % h0.shape[0] == 0:
            reps = n // h0.shape[0]
            h = np.tile(h0, reps)
            h = enforce_sum(h, target, rng)
        else:
            h = np.zeros(n)
            h[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)

    # ----- first refinement -------------------------------------------------
    best_h, best_c = local_refine(h, target, rng, max_iter=300)

    # ----- main search loop ------------------------------------------------
    time_limit = budget_s * 0.97
    while time.time() - start < time_limit:
        # random binary start
        h_start = np.zeros(n)
        h_start[rng.choice(n, n // 2, replace=False)] = 1.0
        h_ref, c_ref = local_refine(h_start, target, rng, max_iter=250)

            best_h, best_c = local_refine(best_h, target, rng, max_iter=800)
            break

    # ----- final sanity check ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = evaluate_h(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
