import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, max_iter: int = 10000) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        it += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 30) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # quick repair
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def continuous_sa_multi_restart(n: int, rng: np.random.Generator,
                               total_time: float, max_restarts: int = 5) -> tuple:
    """Simulated annealing with multiple restarts, preserving the sum."""
    best_h = None
    best_c = np.inf
    start_all = time.time()
    for restart in range(max_restarts):
        elapsed_all = time.time() - start_all
        if elapsed_all >= total_time:
            break
        remaining = total_time - elapsed_all
        # allocate equal remaining budget to remaining restarts
        rest = max_restarts - restart
        slice_budget = remaining / rest
        # initialize feasible vector
        h = rng.random(n)
        h = enforce_sum(h, rng)
        h = np.clip(h, 0.0, 1.0)
        c = compute_c5(h)
        best_h_local = h.copy()
        best_c_local = c
        start_slice = time.time()
        T0, T_end = 0.02, 1e-6
        while True:
            elapsed_slice = time.time() - start_slice
            if elapsed_slice >= slice_budget:
                break
            frac = elapsed_slice / slice_budget
            T = T0 * (T_end / T0) ** frac
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # decide transfer direction
            if rng.random() < 0.5:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = rng.random() * max_eps * (T / T0)
                h_new = h.copy()
                h_new[i] -= eps
                h_new[j] += eps
            else:
                max_eps = min(h[j], 1.0 - h[i])
                if max_eps <= 0:
                    continue
                eps = rng.random() * max_eps * (T / T0)
                h_new = h.copy()
                h_new[i] += eps
                h_new[j] -= eps
            c_new = compute_c5(h_new)
            delta = c_new - c
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                h, c = h_new, c_new
                if c < best_c_local:
                    best_c_local, best_h_local = c, h.copy()
        if best_c_local < best_c:
            best_c, best_h = best_c_local, best_h_local.copy()
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None

    # ------------------------------------------------------------------
