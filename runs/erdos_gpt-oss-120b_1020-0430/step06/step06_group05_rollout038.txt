def compute_c5(h):

def project_sum(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)

            lo = mid

def random_perturb(h, rng, scale=0.1):
    n = h.size
    idx1 = np.where(h > 1e-12)[0]
    idx0 = np.where(h < 1 - 1e-12)[0]

    j = rng.choice(idx0)

        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

import time
import numpy as np

# global variables used by the SLSQP helper
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET (used by SLSQP)."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while keeping Σx = target."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        idx = (y > 1e-12) & (y < 1.0 - 1e-12)
        if idx.any():
            y[idx] += diff / idx.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.05) -> np.ndarray:
    """Transfer a small amount of mass from a 1‑entry to a 0‑entry."""
    idx1 = np.where(h > 1e-12)[0]
    idx0 = np.where(h < 1.0 - 1e-12)[0]
    if idx1.size == 0 or idx0.size == 0:
        return h
    i = rng.choice(idx1)
    j = rng.choice(idx0)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise h ----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        target = n / 2.0
        h = np.clip(h, 0.0, 1.0)
        h = project_sum(h, target)
    else:
        n = 512
        target = n / 2.0
        h = rng.random(n)
        h = project_sum(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # parameters for the soft‑max gradient descent
    beta_schedule = [10, 50, 200, 500, 1000, 2000, 5000, 10000]

    # main optimisation loop
    while time.time() - start < budget_s * 0.94:
        for beta in beta_schedule:
            step = 0.5 / np.sqrt(beta + 1.0)          # step size adapts to β
            it = 0
            while time.time() - start < budget_s * 0.94 and it < 2500:
                # correlation O_k
                O = np.correlate(h, 1.0 - h, mode="full")
                # soft‑max weights
                scaled = beta * O
                m = scaled.max()
                expb = np.exp(scaled - m)
                p = expb / expb.sum()

                # gradient of the soft‑max (via correlations)
                g1 = np.correlate(p, 1.0 - h, mode="full")[n - 1 : n - 1 + n]
                g2 = np.correlate(p[::-1], h, mode="full")[n - 1 : n - 1 + n]
                g = g1 - g2

                # gradient step with projection
                h_new = h - step * g
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_sum(h_new, target)
                h = h_new

                # evaluate C5
                c = O.max() * (2.0 / n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                    if best_c < 0.38079:           # early exit if record is beaten
                        return best_h.tolist(), float(best_c), int(n)

                # occasional random perturbation to escape plateaus
                if it % 250 == 0 and it > 0:
                    h_pert = random_perturb(h, rng, scale=0.05)
                    h_pert = project_sum(h_pert, target)
                    c_pert = compute_c5(h_pert)
                    if c_pert < best_c:
                        best_c = c_pert
                        best_h = h_pert.copy()
                        h = h_pert
                it += 1

    # ----- final polishing with SLSQP (if time permits) ----- #
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        from scipy.optimize import minimize

        n = best_h.size
        target = n / 2.0
        global CURRENT_TARGET, CURRENT_BETA
        CURRENT_TARGET = target
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        # try a few increasing β values
        for beta in (2000.0, 5000.0, 10000.0):
            if time.time() - start > budget_s * 0.99:
                break
            CURRENT_BETA = beta
            res = minimize(
                softmax_obj,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-12, "disp": False},
            )
            x = np.clip(res.x, 0.0, 1.0)
            x = project_sum(x, target)
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
                if best_c < 0.38079:
                    break

    # final sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
