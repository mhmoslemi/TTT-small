def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # maintain lists of ones and zeros indices
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    # set initial temperature based on cur_c
    init_T = max(1e-6, cur_c * 0.05)
    it = 0
    while time.time() < deadline:
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        # temperature schedule
        T = init_T * (0.9999 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            # accept
            cur_c = c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0  # reset stagnation counter
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c

def get_contributors(b, k):
    n = b.size
    if k >= 0:
        # positions where i + k < n
        if k < n:
            i_range = np.arange(0, n - k)
            contrib = i_range[(b[i_range] == 1) & (b[i_range + k] == 0)]
        else:
            contrib = np.array([], dtype=int)
        # out-of-range positions i where i + k >= n
        if k > 0:
            out_range = np.arange(n - k, n)
            contrib = np.concatenate([contrib, out_range[b[out_range] == 1]])
        return contrib
    else:
        kp = -k
        if kp < n:
            i_range = np.arange(kp, n)
            contrib = i_range[(b[i_range] == 1) & (b[i_range - kp] == 0)]

def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    return float(v.max() * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Adjust h in‑place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:  # final tiny fix
        i = rng.integers(0, n)
        h[i] += diff
    np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_smoothmax(h, beta):
    """Gradient of (1/β)·log ∑exp(β·overlap) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)          # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

# ------------------ binary SA ------------------

def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill‑climb on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-6, cur_c * 0.1)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9999 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c

# ------------------ run ------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97   # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate sample sizes (all even)
    n_candidates = [720, 1080, 1440]

    # main loop over resolutions
    for n in n_candidates:
        if time.time() > deadline:
            break
        target_v = int(np.floor(0.3808 * n / 2 + 1e-12))
        restarts = 2
        for r in range(restarts):
            if time.time() > deadline:
                break

            # --- initial fractional vector ---
            parent = globals().get("initial_h_values", None)
            if r == 0 and parent is not None and len(parent) == n:
                h = np.array(parent, dtype=float)
                adjust_sum_exact(h, rng)
            else:
                h = rng.random(n)
                h = h / h.sum() * (n / 2.0)
                np.clip(h, 0.0, 1.0, out=h)
                adjust_sum_exact(h, rng)

            # --- smooth‑max gradient descent ---
            betas = [10, 30, 90, 270, 810, 2430, 7290,
                     21870, 65610, 196830, 590490]
            base_step = 0.2
            it_per_beta = 15
            for beta in betas:
                if time.time() > deadline:
                    break
                for _ in range(it_per_beta):
                    if time.time() > deadline:
                        break
                    g = gradient_smoothmax(h, beta)
                    if np.linalg.norm(g) < 1e-14:
                        continue
                    best_local_h = h
                    best_local_c = compute_c5(h)
                    for mult in (1.0, 0.5, 0.25, 0.75, 1.5):
                        step = base_step * mult
                        cand = h - step * g
                        np.clip(cand, 0.0, 1.0, out=cand)
                        adjust_sum_exact(cand, rng)
                        c = compute_c5(cand)
                        if c < best_local_c:
                            best_local_c = c
                            best_local_h = cand
                    h = best_local_h

            # record best after gradient stage
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

            # --- rounding to binary and SA ---
            k_half = n // 2
            idx = np.argpartition(-h, k_half - 1)[:k_half]
            h_bin = np.zeros(n, dtype=float)
            h_bin[idx] = 1.0
            sub_dead = min(deadline,
                           time.time() + (deadline - time.time()) * 0.10)
            h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
            if c_bin < best_c:
                best_c, best_h, best_n = c_bin, h_bin.copy(), n

            if best_c <= 0.3808:
                # sufficient, exit early
                break

        # --- up‑sample if time permits ---
        if best_h is not None and best_h.size == n and time.time() < deadline and n * 2 <= 2000:
            h_up = np.repeat(best_h, 2)
            adjust_sum_exact(h_up, rng)
            n_up = h_up.size
            betas_up = [10, 30, 90, 270, 810, 2430, 7290]
            base_step_up = 0.15
            it_per_beta_up = 10
            for beta in betas_up:
                if time.time() > deadline:
                    break
                for _ in range(it_per_beta_up):
                    if time.time() > deadline:
                        break
                    g = gradient_smoothmax(h_up, beta)
                    if np.linalg.norm(g) < 1e-14:
                        continue
                    best_local_h = h_up
                    best_local_c = compute_c5(h_up)
                    for mult in (1.0, 0.5, 0.25):
                        step = base_step_up * mult
                        cand = h_up - step * g
                        np.clip(cand, 0.0, 1.0, out=cand)
                        adjust_sum_exact(cand, rng)
                        c = compute_c5(cand)
                        if c < best_local_c:
                            best_local_c = c
                            best_local_h = cand
                    h_up = best_local_h
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c, best_h, best_n = c_up, h_up.copy(), n_up

            # binary rounding + SA on up‑sampled version
            k_half_up = n_up // 2
            idx_up = np.argpartition(-h_up, k_half_up - 1)[:k_half_up]
            h_bin_up = np.zeros(n_up, dtype=float)
            h_bin_up[idx_up] = 1.0
            sub_dead = min(deadline,
                           time.time() + (deadline - time.time()) * 0.08)
            h_bin_up, c_bin_up = binary_sa_hill_climb(h_bin_up, rng, sub_dead)
            if c_bin_up < best_c:
                best_c, best_h, best_n = c_bin_up, h_bin_up.copy(), n_up

            if best_c <= 0.3808:
                break

    # fallback simple construction
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
