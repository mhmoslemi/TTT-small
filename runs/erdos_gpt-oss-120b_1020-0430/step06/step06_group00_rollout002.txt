def binary_sa_fixed_n(n, rng, time_budget, c_threshold=np.inf):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0 = 0.1
    T_end = 1e-5
    while time.time() - start < time_budget:
        # start from random binary vector
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        # maintain arrays of indices
        ones_arr = np.sort(ones)  # maybe needed for operations index
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        # inner loop runs while time left
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            # pick random swap
            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                # swap indices in arrays
                ones_arr[i_idx] = j
                zeros_arr[j_idx] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= c_threshold:
                        return best_h, best_c
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
        # after inner loop, continue with new random init
    if best_h is None:
        # fallback: start with all zeros (invalid sum) but enforce later
        best_h = np.zeros(n, dtype=float)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=40):
    n = h.size
    for it in range(max_iter):
        k = compute_worst_shift(h)
        # compute gradient as earlier
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        # step size decays
        alpha = step * (0.7 ** it)
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)

        target = n / 2.0

        if abs(diff) > 1e-12:

                h[i] += add
                diff = target - h.sum()

    return h

import time
import numpy as np

# ---------------------------------------------------------
# Core evaluation and utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def compute_worst_shift(h: np.ndarray) -> int:
    """Return the shift index (absolute) that attains maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    return abs(idx)

def binary_sa_fixed_n(n: int, rng: np.random.Generator,
                     time_budget: float, c_threshold: float = np.inf):

    start = time.time()

    while time.time() - start < time_budget:
        # fresh random binary vector
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)

        ones_arr = np.sort(ones)
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)

        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)

            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]

            # propose swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr

            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new

                zeros_arr[j_pos] = i

# Gradient refinement on worst shift
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.02, max_iter: int = 40) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        k = compute_worst_shift(h)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        alpha = step * (0.7 ** it)
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ---------------------------------------------------------
# Continuous simulated annealing (sum fixed, bounds [0,1])
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.005, 1e-7

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)

        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1

        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps

        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf

    # ---- initial construction (if provided) ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c = h0, c0

    # ---- Phase 0: binary SA on 512 points ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        t_bin = min(remaining * 0.45, 460.0)
        h_bin, c_bin = binary_sa_fixed_n(512, rng, t_bin, c_threshold=0.3808)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin

    # ---- Phase 1: upsample to 1024 (if needed) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None and best_h.size < 1024:
        h_up = upsample_h(best_h, 1024)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # ---- Phase 2: gradient refinement ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        # allocate a modest portion for refinement
        t_ref = min(remaining * 0.10, 120.0)
        # run refinement for a bounded number of iterations
        h_ref = gradient_refine(best_h, rng, step=0.02, max_iter=40)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref
        # respect time used
        used = time.time() - start
        remaining = total_budget - used

    # ---- Phase 3: continuous SA on finest grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        h_cs, c_cs = continuous_sa(best_h, rng, remaining)
        if c_cs < best_c:
            best_h, best_c = h_cs, c_cs

    # ---- Final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
