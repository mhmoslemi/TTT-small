import time
import numpy as np

def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT for speed."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return corr.max() * (2.0 / n)

compute_c5 = compute_c5_fft  # alias for compatibility


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient to negative‑gradient entries."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 512
    target_sum = n // 2  # integer

    # ----- initialise h (binary) ----- #
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # pick the indices of the largest target_sum entries
        idx = np.argpartition(h0, -target_sum)[-target_sum:]
        h = np.zeros_like(h0)
        h[idx] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, target_sum, replace=False)
        h[idx] = 1.0

    # ----- auxiliary structures for fast swaps ----- #
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    idx_to_onepos = np.full(n, -1, dtype=int)
    idx_to_onepos[ones] = np.arange(ones.size)
    idx_to_zeropos = np.full(n, -1, dtype=int)
    idx_to_zeropos[zeros] = np.arange(zeros.size)

    c_current = compute_c5(h)
    best_h = h.copy()
    best_c = c_current

    # ----- warm‑up to estimate temperature ----- #
    warm_iters = 200
    pos_deltas = []
    for _ in range(warm_iters):
        i = ones[rng.integers(ones.size)]
        j = zeros[rng.integers(zeros.size)]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_current
        if delta > 0:
            pos_deltas.append(delta)
        h[i] = 1.0
        h[j] = 0.0
    if pos_deltas:
        T0 = np.mean(pos_deltas) * 2.0
    else:
        T0 = 0.01
    T_end = 1e-9
    log_T0 = np.log(T0)
    log_T_end = np.log(T_end)

    # ----- Simulated Annealing ----- #
    max_iter = 20_000_000
    it = 0
    while it < max_iter and (time.time() - start) < budget_s * 0.95:
        frac = it / max_iter
        T = np.exp(log_T0 + frac * (log_T_end - log_T0))

        i_idx = rng.integers(ones.size)
        j_idx = rng.integers(zeros.size)
        i = ones[i_idx]
        j = zeros[j_idx]

        # perform the swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_current
        accept = False
        if delta <= 0:
            accept = True
        else:
            # avoid overflow in exp
            arg = -delta / T
            if arg > 700:
                prob = 0.0
            else:
                prob = np.exp(arg)
            accept = rng.random() < prob

        if accept:
            c_current = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            # update index structures
            ones[i_idx] = j
            idx_to_onepos[j] = i_idx
            idx_to_onepos[i] = -1
            zeros[j_idx] = i
            idx_to_zeropos[i] = j_idx
            idx_to_zeropos[j] = -1
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        it += 1

    # ----- optional gradient refinement (fractional) ----- #
    if (time.time() - start) < budget_s * 0.94:
        h_ref = gradient_refine(best_h, float(target_sum), rng,
                                step=0.02, max_iter=30)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    # final enforcement of constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, float(target_sum), rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
