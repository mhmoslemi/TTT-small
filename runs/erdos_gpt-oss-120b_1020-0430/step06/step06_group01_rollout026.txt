def softmax_c5_and_grad(h, beta):
    n = h.size
    # compute corr via FFT
    f = np.fft.rfft(h, n=2*n)
    g = np.fft.rfft(1 - h, n=2*n)
    conv = np.fft.irfft(f * g, n=2*n)
    corr = conv[:2*n-1]  # length 2n-1
    # compute softmax
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    p = exp_bc / sum_exp
    # compute gradient
    p_arr = p  # length 2n-1
    # reverse p
    p_rev = p_arr[::-1]
    # compute conv_rev_oneMinus = convolution(p_rev, 1-h)
    f_pr = np.fft.rfft(p_rev, n=2*n)
    f_one = np.fft.rfft(1 - h, n=2*n)
    conv_rev = np.fft.irfft(f_pr * f_one, n=2*n)
    conv_rev = conv_rev[:2*n-1]  # length 2n-1
    # compute conv_ph = convolution(p, h)
    f_p = np.fft.rfft(p_arr, n=2*n)
    f_h = np.fft.rfft(h, n=2*n)
    conv_ph = np.fft.irfft(f_p * f_h, n=2*n)
    conv_ph = conv_ph[:2*n-1]
    # gradient indices shift: we need at i+(n-1)
    offset = n - 1
    g_vec = conv_rev[offset:offset+n] - conv_ph[offset:offset+n]
    # The objective value: softmax approximation = (max bc + log sum exp) / beta
    soft_c5 = (max_bc + np.log(sum_exp)) / beta
    return soft_c5, g_vec

    diff = target - h.sum()

        if diff > 0:
            # add up to 1-h[i]
            add = min(diff, 1.0 - h[i])
        else:
            # remove up to h[i]

def softmax_c5_and_grad(h, beta):
    n = h.size
    fft_len = 2 * n
    # compute correlation via fft
    fh = np.fft.rfft(h, fft_len)
    f1h = np.fft.rfft(1.0 - h, fft_len)

def lbfgsb_optimize(h0, beta, rng, max_iter=1000, penalty=1e3):
    n = h0.size
    target = n / 2.0
    # closure for objective and gradient using softmax_c5_and_grad
    def obj_grad(x):
        h = x
        soft, g = softmax_c5_and_grad(h, beta)
        # penalty term
        diff = h.sum() - target
        penalty_val = penalty * diff * diff
        soft += penalty_val
        # gradient of penalty: 2*penalty*diff for each component
        g += 2.0 * penalty * diff
        return soft, g
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: obj_grad(x)[0], h0, jac=lambda x: obj_grad(x)[1],
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    enforce_sum(x, rng)
    return x

def lbfgsb_optimize(h0, beta, rng, max_iter=800, penalty=1000.0):
    n = h0.size
    target = n / 2.0
    # define objective and gradient
    def fun(x):
        val, _ = softmax_c5_and_grad(x, beta)
        diff = x.sum() - target
        val += penalty * diff * diff
        return val
    def jac(x):
        _, grad = softmax_c5_and_grad(x, beta)
        diff = x.sum() - target
        grad += 2.0 * penalty * diff
        return grad
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, rng)
    return x

    f1h = np.fft.rfft(1.0 - h, fft_len)
    conv = np.fft.irfft(fh * f1h, fft_len)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        h[0] += diff
    np.clip(h, 0.0, 1.0, out=h)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    fft_len = 2 * n
    corr = np.fft.irfft(
        np.fft.rfft(h, fft_len) * np.fft.rfft(1.0 - h, fft_len), fft_len
    )[: 2 * n - 1]
    idx = np.argmax(corr)
    l = idx - (n - 1)
    k = abs(l)
    grad = np.ones_like(h)
    if k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        grad = 1.0 - 2.0 * h
    return grad


# ------------------ soft‑max objective ------------------

def softmax_c5_and_grad(h: np.ndarray, beta: float):
    """Smooth approximation of max C5 and its gradient."""
    n = h.size
    fft_len = 2 * n

    # correlation
    fh = np.fft.rfft(h, fft_len)
    f1h = np.fft.rfft(1.0 - h, fft_len)
    corr = np.fft.irfft(fh * f1h, fft_len)[: 2 * n - 1]

    # soft‑max value
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    s = np.sum(exp_bc)
    p = exp_bc / s
    soft_val = (m + np.log(s)) / beta

    # gradient via convolutions
    p_rev = p[::-1]

    fpr = np.fft.rfft(p_rev, fft_len)
    f_one = np.fft.rfft(1.0 - h, fft_len)
    conv_rev = np.fft.irfft(fpr * f_one, fft_len)[: 2 * n - 1]

    fp = np.fft.rfft(p, fft_len)
    fh = np.fft.rfft(h, fft_len)
    conv_ph = np.fft.irfft(fp * fh, fft_len)[: 2 * n - 1]

    off = n - 1
    grad = conv_rev[off : off + n] - conv_ph[off : off + n]
    return soft_val, grad


def lbfgsb_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                    max_iter: int = 800, penalty: float = 1000.0) -> np.ndarray:
    """L‑BFGS‑B on soft‑max C5 with a sum‑penalty."""
    n = h0.size
    target = n / 2.0

    def fun(x):
        val, _ = softmax_c5_and_grad(x, beta)
        diff = x.sum() - target
        val += penalty * diff * diff
        return val

    def jac(x):
        _, g = softmax_c5_and_grad(x, beta)
        diff = x.sum() - target
        g += 2.0 * penalty * diff
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, jac=jac, method="L-BFGS-B",
                   bounds=bounds, options={"maxiter": max_iter,
                                           "ftol": 1e-12,
                                           "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    enforce_sum(x, rng)
    return x


# ------------------ binary simulated annealing ------------------

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """0/1 SA for the overlap problem."""
    start = time.time()
    best_h = np.zeros(n, dtype=float)
    best_c = np.inf
    T0, T_end = 0.03, 1e-4

    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, n // 2, replace=False)
        h[ones_idx] = 1.0
        c_curr = compute_c5(h)

        ones = list(ones_idx)
        zeros = [i for i in range(n) if i not in ones_idx]

        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)

            i_pos = rng.integers(len(ones))
            j_pos = rng.integers(len(zeros))
            i, j = ones[i_pos], zeros[j_pos]

            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr

            accept = delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12))
            if accept:
                c_curr = c_new
                ones[i_pos] = j
                zeros[j_pos] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0

    return best_h, best_c


# ------------------ hill‑climb (worst‑gradient) ------------------

def hill_climb_worst(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 20000, step_scale: float = 0.6):
    """Deterministic hill‑climb using the worst‑shift gradient."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size

    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # decay step_scale a little
        if (it + 1) % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-5)

    return best_h, best_c


# ------------------ main entry point ------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # use provided construction if any
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # size we aim for
    target_n = 1024

    # ----- binary simulated annealing (≈30% of time) -----
    binary_time = total_budget * 0.30
    h_bin, c_bin = binary_sa(target_n, rng, binary_time)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin, c_bin, target_n

    # ----- continuous refinement (remaining time) -----
    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        # initialise continuous vector at target_n
        if best_h is not None and best_n == target_n:
            init_h = best_h.copy()
        elif best_h is not None:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            init_h = np.interp(x_new, x_old, best_h)
            init_h = np.clip(init_h, 0.0, 1.0)
            enforce_sum(init_h, rng)
        else:
            init_h = rng.random(target_n)
            enforce_sum(init_h, rng)

        h_cont = init_h
        beta_seq = [200, 800, 3000, 10000, 30000]
        for beta in beta_seq:
            if time.time() - start > total_budget * 0.95:
                break
            h_cont = lbfgsb_optimize(h_cont, beta, rng, max_iter=800, penalty=1000.0)
            c_val = compute_c5(h_cont)
            if c_val < best_c:
                best_h, best_c, best_n = h_cont.copy(), c_val, target_n

        # final hill‑climb
        time_left = total_budget - (time.time() - start)
        if time_left > 5.0:
            max_iters = int(20000 * (time_left / total_budget))
            max_iters = max(max_iters, 2000)
            h_hc, c_hc = hill_climb_worst(best_h.copy(), rng,
                                          max_iters=max_iters, step_scale=0.6)
            if c_hc < best_c:
                best_h, best_c, best_n = h_hc.copy(), c_hc, target_n

    # ----- tiny stochastic tweaks until time runs out -----
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i, j = int(np.argmax(grad)), int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
