import time
import numpy as np

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by binary search on lambda."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (0≤h≤1, sum = n/2)."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst (max) overlap O_k and its shift index."""
    n = h.shape[0]
    cross = np.correlate(h, h, mode="full")
    cross_pos = cross[n - 1 :]                # length n, shifts 0..n-1
    k = int(np.argmin(cross_pos))            # worst shift (minimum product)
    grad = np.ones_like(h)                   # ∂O_k/∂h_i = 1 - (h_{i+k}+h_{i-k})
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def subgradient_optimize_full(
    h0: np.ndarray,
    target: float,
    max_iter: int,
    rng: np.random.Generator,
    a0: float = 1.0,
    mu: float = 0.9,
    time_budget: float = None,
):
    """Projected sub‑gradient ascent on min‑overlap (i.e. descent on C5)."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros_like(h)
    start = time.time()
    for t in range(1, max_iter + 1):
        if time_budget is not None and (time.time() - start) > time_budget:
            break
        grad, _ = worst_gradient(h)
        v = mu * v + grad
        alpha = a0 / np.sqrt(t + 1)
        h = project_box_sum(h - alpha * v, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.6,
    time_budget: float = None,
):
    """Hill‑climb using gradient‑directed mass transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    start = time.time()
    for it in range(max_iters):
        if time_budget is not None and (time.time() - start) > time_budget:
            break
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
                c = compute_c5(nh)
                if c < best_c:
                    best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def upsample(h: np.ndarray) -> np.ndarray:
    """Linear up‑sampling by factor 2 with projection onto the feasible set."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

            best_h, best_c = h_hc, c_hc

    # ---- final sanitisation ----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
