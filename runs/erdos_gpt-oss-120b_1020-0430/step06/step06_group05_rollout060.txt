- random_swap_move(h, rng):
    # swap values between i,j where h[i]!=h[j] ensures sum unchanged.
    n = h.size
    i = rng.integers(0, n)
    # find j where h[j] != h[i]
    mask = (h != h[i])

    if not np.any(mask):
        return h.copy()

    # Use rng.choice on indices:

    j = rng.choice(possible)
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

import time, numpy as np

n_samples = 0

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        caps = 1.0 - h[mask]
        cap_sum = caps.sum()
        if cap_sum < diff:
            # fallback iterative
            while diff > 1e-12:
                i = rng.integers(0, h.size)
                cap = 1.0 - h[i]
                if cap <= 0.0:
                    continue
                delta = min(diff, cap)
                h[i] += delta
                diff = target - h.sum()
            return h
        h[mask] += diff * caps / cap_sum
    else:
        mask = h > 0.0
        caps = h[mask]
        cap_sum = caps.sum()
        if cap_sum < -diff:
            while diff < -1e-12:
                i = rng.integers(0, h.size)
                cap = h[i]
                if cap <= 0.0:
                    continue
                delta = min(-diff, cap)
                h[i] -= delta
                diff = target - h.sum()
            return h
        h[mask] += diff * caps / cap_sum  # diff negative
    np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_softmax(h, beta):
    n = h.size
    c_full = np.correlate(h, 1.0 - h, mode='full')
    # stable softmax
    bc = beta * c_full
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w_sum = np.sum(w)
    w /= w_sum
    # conv1 = conv(w_rev, h)
    conv1 = np.convolve(w[::-1], h, mode='full')[n-1:n-1+n]
    conv2 = np.convolve(w, h, mode='full')[n-1:n-1+n]
    grad = 1.0 - conv1 - conv2
    return grad

def projected_grad_step(h, grad, step, target, rng):
    h_new = h - step * grad
    np.clip(h_new, 0.0, 1.0, out=h_new)
    h_new = enforce_sum(h_new, target, rng)
    return h_new

def random_swap_move(h, rng):
    n = h.size
    i = rng.integers(0, n)
    mask = h != h[i]
    if not np.any(mask):
        return h.copy()
    j = rng.choice(np.flatnonzero(mask))
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def random_perturb_mass(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1

        return h
    eps = max_eps * rng.random() * scale
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

import time
import numpy as np

n_samples = 0  # will be set inside run()


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        caps = 1.0 - h[mask]
        cap_sum = caps.sum()
        if cap_sum < diff:  # fallback
            while diff > 1e-12:
                i = rng.integers(0, h.size)
                cap = 1.0 - h[i]
                if cap <= 0.0:
                    continue
                delta = min(diff, cap)
                h[i] += delta
                diff = target - h.sum()
            return h
        h[mask] += diff * caps / cap_sum
    else:
        mask = h > 0.0
        caps = h[mask]
        cap_sum = caps.sum()
        if cap_sum < -diff:
            while diff < -1e-12:
                i = rng.integers(0, h.size)
                cap = h[i]
                if cap <= 0.0:
                    continue
                delta = min(-diff, cap)
                h[i] -= delta
                diff = target - h.sum()
            return h
        h[mask] += diff * caps / cap_sum  # diff negative
    np.clip(h, 0.0, 1.0, out=h)
    return h


def gradient_softmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max_k O_k."""
    n = h.size
    c_full = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * c_full
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w_sum = np.sum(w)
    w /= w_sum

    term1 = np.zeros(n)
    term2 = np.zeros(n)

    # w index corresponds to shift s = idx - (n-1)
    for idx in range(2 * n - 1):
        ws = w[idx]
        if ws == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            term1[: n - s] += ws * h[s:]
            term2[s:] += ws * h[: n - s]
        else:
            s_abs = -s
            term1[s_abs:] += ws * h[: n - s_abs]
            term2[: n + s] += ws * h[s_abs:]
    grad = 1.0 - term1 - term2
    return grad


def projected_grad_step(h: np.ndarray, grad: np.ndarray, step: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    h_new = h - step * grad
    np.clip(h_new, 0.0, 1.0, out=h_new)
    h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_swap_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    mask = h != h[i]
    if not np.any(mask):
        return h.copy()
    j = rng.choice(np.flatnonzero(mask))
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new


def random_perturb_mass(h: np.ndarray, rng: np.random.Generator,
                        scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new


def random_initial(target: float, rng: np.random.Generator) -> np.ndarray:
    n = int(n_samples)
    h = rng.random(n)
    return enforce_sum(h, target, rng)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    global n_samples
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 1024
    n_samples = n
    target = n / 2.0

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        m = h0.size
        if n % m == 0:
            factor = n // m
            h = np.repeat(h0, factor)
        else:
            xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(xs_new, xs_old, h0)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    else:
        h = random_initial(target, rng)

    c_current = compute_c5(h)
    best_h = h.copy()
    best_c = c_current

    # schedule of (beta, step, max_iter)
    schedule = [
        (500.0, 0.05, 3000),
        (2000.0, 0.02, 3000),
        (5000.0, 0.01, 2000),
        (20000.0, 0.005, 1500),
        (50000.0, 0.0025, 1000),
    ]

    no_improve = 0
    max_no_improve = 5000
    done = False

    for beta, step, max_iter in schedule:
        for it in range(max_iter):
            if time.time() - start > budget_s * 0.99:
                done = True
                break
            grad = gradient_softmax(h, beta)
            h_new = projected_grad_step(h, grad, step, target, rng)
            c_new = compute_c5(h_new)

            if c_new + 1e-12 < c_current:
                h = h_new
                c_current = c_new
                no_improve = 0
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                if best_c < 0.38079:
                    done = True
                    break
            else:
                no_improve += 1
                if rng.random() < 0.01:
                    h_swap = random_swap_move(h, rng)
                    c_swap = compute_c5(h_swap)
                    if c_swap < c_current:
                        h = h_swap
                        c_current = c_swap
                        no_improve = 0
                        if c_swap < best_c:
                            best_c = c_swap
                            best_h = h.copy()
                if it % 100 == 0 and rng.random() < 0.5:
                    h_pert = random_perturb_mass(h, rng, scale=0.1)
                    c_pert = compute_c5(h_pert)
                    if c_pert < c_current:
                        h = h_pert
                        c_current = c_pert
                        no_improve = 0
                        if c_pert < best_c:
                            best_c = c_pert
                            best_h = h.copy()
                if no_improve >= max_no_improve:
                    h = random_initial(target, rng)
                    c_current = compute_c5(h)
                    no_improve = 0
            # early exit if overall done
        if done or time.time() - start > budget_s * 0.99:
            break

    # ----- final high‑beta fine tuning on the best solution -----
    for _ in range(5000):
        if time.time() - start > budget_s * 0.99:
            break
        grad = gradient_softmax(best_h, 100000.0)
        best_h = projected_grad_step(best_h, grad, 0.001, target, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
