def compute_c5_binary(h):
    n = h.size
    c = np.correlate(h, h, mode='full')
    c_nonneg = c[h.size-1:]  # from index n-1 to end
    cmin = c_nonneg.min()
    return 1.0 - (2.0/n) * cmin

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """C5 for a (possibly non‑binary) step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_c5_binary(h: np.ndarray) -> float:
    """C5 for a binary (0/1) step function – uses self‑correlation."""
    n = h.size
    c = np.correlate(h, h, mode="full")
    c_nonneg = c[n - 1 :]
    cmin = c_nonneg.min()
    return 1.0 - (2.0 / n) * cmin


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x


# ---------- main routine ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- obtain initial continuous vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h0 = rng.random(n)
    n = h0.size
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- binary hill climbing (swap 1↔0) ----
    # convert to binary while keeping the correct integral
    h_bin = (h0 > 0.5).astype(float)
    diff = int(n // 2 - h_bin.sum())
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        chosen = rng.choice(zeros, diff, replace=False)
        h_bin[chosen] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        chosen = rng.choice(ones, -diff, replace=False)
        h_bin[chosen] = 0.0

    bin_c = compute_c5_binary(h_bin)
    if bin_c < best_c:
        best_c, best_h = bin_c, h_bin.astype(float).copy()

    # hill‑climbing loop (time‑bounded)
    time_limit = start + budget_s * 0.6
    while time.time() < time_limit:
        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        if best_h[i] != 1.0:
            continue
        j = rng.integers(0, n)
        if best_h[j] != 0.0:

        best_h[j] = 1.0

        else:

            best_h[i] = 1.0
            best_h[j] = 0.0

    target = 0.3808

    # ---- SLSQP refinement with increasing beta ----
    h_float = enforce_sum(best_h.astype(float), rng)

    for beta in betas:

            if best_c <= target:

    step = 0.05

        nh = h_float - step * grad
        nh = np.clip(nh, 0.0, 1.0)

        if c_new < best_c:
            best_c, h_float = c_new, nh.copy()
            if best_c <= target:
                break

    # ---- final sanity check ----
    final_h = enforce_sum(np.clip(h_float, 0.0, 1.0), rng)
    final_c = float(compute_c5(final_h))
    return final_h.tolist(), final_c, int(final_h.size)
