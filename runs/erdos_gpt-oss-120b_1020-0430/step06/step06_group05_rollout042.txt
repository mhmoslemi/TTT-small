def compute_c5(h):
    n = h.size
    q = 1.0 - h
    rev_h = h[::-1]
    m = 2*n - 1
    H = np.fft.rfft(rev_h, m)
    Q = np.fft.rfft(q, m)
    conv = np.fft.irfft(H * Q, m)
    # conv is full convolution rev_h * q; correlation for k >=0 is conv[n-1 + k]
    max_corr = conv[n-1:].max()
    return max_corr * (2.0 / n)
```

import numpy as np

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def compute_c5_float(h):
    n = h.shape[0]
    q = 1.0 - h
    rev_h = h[::-1]
    m = 2*n - 1
    H = np.fft.rfft(rev_h, m)
    Q = np.fft.rfft(q, m)
    conv = np.fft.irfft(H * Q, m)
    max_corr = conv[n-1:].max()
    return max_corr * (2.0 / n)

def worst_shift_and_grad_float(h):
    n = h.shape[0]

import time, math
import numpy as np

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def compute_c5_float(h: np.ndarray) -> float:
    """C5 for a (possibly fractional) step function on [0,2]."""
    n = h.shape[0]
    q = 1.0 - h
    rev_h = h[::-1]
    m = 2 * n - 1
    H = np.fft.rfft(rev_h, m)
    Q = np.fft.rfft(q, m)
    conv = np.fft.irfft(H * Q, m)
    max_corr = conv[n - 1 :].max()
    return max_corr * (2.0 / n)

def worst_shift_and_grad_float(h: np.ndarray):
    """Find shift attaining max overlap and its gradient (continuous version)."""
    n = h.shape[0]
    q = 1.0 - h
    rev_h = h[::-1]
    m = 2 * n - 1
    H = np.fft.rfft(rev_h, m)
    Q = np.fft.rfft(q, m)
    conv = np.fft.irfft(H * Q, m)

    if k == 0:

    else:
        grad[: n - k] -= h[k:]

    step0 = 0.015
    for it in range(12):
        k, grad = worst_shift_and_grad_float(h_float)
        gamma = step0 * (0.5 ** it)
        h_float = h_float - gamma * grad
        np.clip(h_float, 0.0, 1.0, out=h_float)
        diff = target_sum - h_float.sum()
        if abs(diff) > 1e-12:
            h_float = enforce_sum(h_float, target_sum, rng)
        c_new = compute_c5_float(h_float)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_float.copy()

    # ----- final sanity enforcement -----
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = enforce_sum(final_h, target_sum, rng)
    final_c = compute_c5_float(final_h)

    return final_h.tolist(), float(final_c), int(final_h.size)
