def compute_c5_binary(h):
    n = h.size
    # compute circular autocorrelation
    # using FFT of uint8 array for speed
    fft = np.fft.fft(h)
    corr = np.fft.ifft(fft * np.conj(fft)).real
    # round to nearest integer
    overlap = np.rint(corr).astype(np.int64)
    min_overlap = overlap.min()
    c5 = 1.0 - (2.0 / n) * min_overlap
    return float(c5)

    T0 = 0.02
    Tmin = 1e-7

    # estimate 0.0006 per iter
    approx_iter = max(1, int(time_budget / 0.0006))
    for t in range(approx_iter):
        now = time.time()

import time
import numpy as np

def _compute_c5_binary(h: np.ndarray) -> float:
    """Return C5 for a binary (0/1) step function on [0,2]."""
    n = h.size
    # circular autocorrelation via FFT
    fft_h = np.fft.rfft(h)
    corr = np.fft.irfft(fft_h * np.conj(fft_h), n=n)
    # round to nearest integer to avoid floating noise
    corr = np.rint(corr).astype(np.int64)
    min_overlap = int(corr.min())
    return 1.0 - (2.0 / n) * min_overlap

def _binary_from_initial() -> np.ndarray:
    """Create a binary vector (n/2 ones) from the provided initial_h_values."""
    arr = np.array(initial_h_values, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1.0)
    n = arr.size
    # choose the indices of the largest n/2 entries
    k = n // 2
    idx = np.argpartition(arr, -k)[-k:]
    bin_vec = np.zeros(n, dtype=np.int8)
    bin_vec[idx] = 1
    return bin_vec

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random binary vector with exactly n/2 ones."""
    ones = rng.choice(n, size=n // 2, replace=False)
    vec = np.zeros(n, dtype=np.int8)
    vec[ones] = 1
    return vec

def _simulated_annealing_binary(h0: np.ndarray, time_budget: float,
                                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Guided simulated annealing on a binary vector."""
    start = time.time()
    h = h0.copy()
    n = h.size

    # initial state
    fft_h = np.fft.rfft(h)
    corr = np.fft.irfft(fft_h * np.conj(fft_h), n=n)
    corr = np.rint(corr).astype(np.int64)
    current_c = 1.0 - (2.0 / n) * int(corr.min())
    best_h = h.copy()
    best_c = current_c

    # temperature schedule
    T0 = 0.03
    Tmin = 1e-6

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        # linear cooling
        T = max(Tmin, T0 * (1.0 - elapsed / time_budget))

        # worst shift (minimum overlap)
        k_worst = int(np.argmin(corr))

        # try to improve the worst shift
        shifted = np.roll(h, k_worst)
        cand_i = np.where((h == 0) & (shifted == 1))[0]
        cand_j = np.where((h == 1) & (shifted == 0))[0]

        if cand_i.size and cand_j.size:
            i = rng.choice(cand_i)
            j = rng.choice(cand_j)
        else:
            # fallback: random swap of a 0 and a 1
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            i = rng.choice(zeros)
            j = rng.choice(ones)

        # propose swap
        h[i] = 1
        h[j] = 0

        # evaluate new state
        fft_h = np.fft.rfft(h)
        corr_new = np.fft.irfft(fft_h * np.conj(fft_h), n=n)
        corr_new = np.rint(corr_new).astype(np.int64)
        new_c = 1.0 - (2.0 / n) * int(corr_new.min())

        delta = new_c - current_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            # accept
            corr = corr_new
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # reject – revert swap
            h[i] = 0
            h[j] = 1

        if best_c <= 0.3808:
            break

    return best_h, best_c

def _greedy_improve_binary(h: np.ndarray, max_iters: int,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Deterministic greedy improvement: only accept moves that lower C5."""
    n = h.size
    best_h = h.copy()
    best_c = _compute_c5_binary(h)

    for _ in range(max_iters):
        fft_h = np.fft.rfft(h)
        corr = np.fft.irfft(fft_h * np.conj(fft_h), n=n)
        corr = np.rint(corr).astype(np.int64)
        k_worst = int(np.argmin(corr))
        shifted = np.roll(h, k_worst)
        cand_i = np.where((h == 0) & (shifted == 1))[0]
        cand_j = np.where((h == 1) & (shifted == 0))[0]
        if cand_i.size == 0 or cand_j.size == 0:
            break
        i = rng.choice(cand_i)
        j = rng.choice(cand_j)
        h[i] = 1
        h[j] = 0
        new_c = _compute_c5_binary(h)
        if new_c < best_c - 1e-12:
            best_c = new_c
            best_h = h.copy()
        else:
            # revert and stop (no improving move found)
            h[i] = 0
            h[j] = 1
            break

    return best_h, best_c

def _upsample_binary(h: np.ndarray) -> np.ndarray:
    """Double the resolution by repeating each bit."""
    return np.repeat(h, 2)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # Determine base size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        if n % 2 == 1:
            n += 1  # enforce even length
    else:
        n = 512

    # Global best
    best_h = None
    best_c = np.inf

    # Main annealing loop (multiple independent runs)
    while True:
        remaining = total_budget - (time.time() - start)
        if remaining < 8.0:
            break
        run_time = min(30.0, remaining - 5.0)
        # Choose start: occasionally use the binary version of the given initial
        if 'initial_h_values' in globals() and rng.random() < 0.3:
            h0 = _binary_from_initial()
            if h0.size != n:
                h0 = _random_binary(n, rng)
        else:
            h0 = _random_binary(n, rng)

        h_opt, c_opt = _simulated_annealing_binary(h0, run_time, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= 0.3808:
            break

    # Optional up‑sampling to a finer grid
    if best_h is not None and n * 2 <= 1024:
        remaining = total_budget - (time.time() - start)
        if remaining > 12.0:
            h_up = _upsample_binary(best_h)
            n_up = h_up.size
            h_up_opt, c_up_opt = _simulated_annealing_binary(h_up, remaining, rng)
            if c_up_opt < best_c:
                best_c, best_h = c_up_opt, h_up_opt.copy()
                n = n_up

    # Final deterministic greedy polishing
    remaining = total_budget - (time.time() - start)
    if best_h is not None and remaining > 2.0:
        max_greedy = int(remaining * 250)  # roughly 250 iterations per second
        h_polished, c_polished = _greedy_improve_binary(best_h, max_greedy, rng)
        if c_polished < best_c:
            best_c, best_h = c_polished, h_polished.copy()

    # Safety fallback
    if best_h is None:
        best_h = _random_binary(n, rng)
        best_c = _compute_c5_binary(best_h)

    # Ensure proper type for the return
    best_h_float = best_h.astype(np.float64)
    final_c = _compute_c5_binary(best_h.astype(np.int8))

    return best_h_float.tolist(), float(final_c), int(best_h.shape[0])
