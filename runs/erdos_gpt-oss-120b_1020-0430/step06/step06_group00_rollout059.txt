def binary_greedy_search(N, rng, time_budget):
    start = time.time()
    best_h = None
    best_c = np.inf
    iteration = 0
    while time.time() - start < time_budget:
        iteration += 1
        # generate random binary vector
        h = np.zeros(N, dtype=np.int8)
        ones = rng.choice(N, N // 2, replace=False)
        h[ones] = 1
        # local greedy
        while True:
            # compute correlation and worst shift
            corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
            cmax = corr.max()
            if cmax <= best_c:
                # possible improvement; continue
                pass
            # find worst shift
            idx = np.argmax(corr) - (N - 1)
            k = idx
            # compute h_fwd and h_back
            h_fwd = np.zeros(N, dtype=np.int8)
            if k >= 0:
                h_fwd[:N - k] = h[k:]
            else:
                h_fwd[-k:] = h[:N + k]
            h_back = np.zeros(N, dtype=np.int8)
            if k >= 0:
                h_back[k:] = h[:N - k]
            else:
                h_back[:N + k] = h[-k:]
            src = (h == 1) & (h_fwd == 0)
            targ = (h_back == 1)
            s_vals = -src.astype(np.int8) + targ.astype(np.int8)
            plus_k = (h_fwd == 0)
            minus_k = (h_back == 1)
            t_vals = plus_k.astype(np.int8) - minus_k.astype(np.int8)
            # restrict to appropriate indices
            # For ones
            ones_idx = np.where(h == 1)[0]
            s_one = s_vals[ones_idx]
            # For zeros
            zeros_idx = np.where(h == 0)[0]
            t_zero = t_vals[zeros_idx]
            # find best
            i_best = ones_idx[np.argmin(s_one)]
            j_best = zeros_idx[np.argmin(t_zero)]
            delta = s_one.min() + t_zero.min()
            if delta >= 0:
                break
            # swap
            h[i_best] = 0
            h[j_best] = 1
        # after local greedy, compute final c
        final_c = compute_c5(h.astype(float))
        if final_c < best_c:
            best_c = final_c
            best_h = h.copy()
            if best_c <= 0.3808:
                # early exit if target reached
                return best_h, best_c
    return best_h, best_c

        h = h - (step * (0.5 ** it)) * grad

        if abs(diff) > 1e-12:

            for _ in range(500):
                if abs(diff) < 1e-12:
                    break

            best_c = final_c
            best_h = h.copy()
            if best_c <= 0.3808:

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad

def gradient_refine(h, rng, step=0.015, max_iter=15):
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def continuous_sa(h0, rng, time_budget):

    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:

            if max_eps <= 0.0:
                continue

            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:

            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def binary_greedy_search(N: int, rng: np.random.Generator, time_budget: float):
    """Repeated random starts + deterministic greedy descent on the worst shift."""
    start = time.time()
    best_h = None
    best_c = np.inf
    while time.time() - start < time_budget:
        h = np.zeros(N, dtype=np.int8)
        h[rng.choice(N, N // 2, replace=False)] = 1
        while True:
            corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
            idx = np.argmax(corr) - (N - 1)
            k = idx
            # shifted arrays with zero padding (outside interval = 0)
            h_fwd = np.zeros(N, dtype=np.int8)
            if k >= 0:
                h_fwd[: N - k] = h[k:]
            else:
                h_fwd[-k:] = h[: N + k]
            h_back = np.zeros(N, dtype=np.int8)
            if k >= 0:
                h_back[k:] = h[: N - k]
            else:
                h_back[: N + k] = h[-k:]

            src = (h == 1) & (h_fwd == 0)          # 1→0 mismatches for this shift
            targ = (h_back == 1)                  # target side where i−k=1
            s_vals = -src.astype(np.int8) + targ.astype(np.int8)

            plus_k = (h_fwd == 0)                 # (h_{j+k}=0) term for zeros
            minus_k = (h_back == 1)               # (h_{j−k}=1) term for zeros
            t_vals = plus_k.astype(np.int8) - minus_k.astype(np.int8)

            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            s_one = s_vals[ones_idx]
            t_zero = t_vals[zeros_idx]

            i_best = ones_idx[np.argmin(s_one)]
            j_best = zeros_idx[np.argmin(t_zero)]
            delta = s_one.min() + t_zero.min()
            if delta >= 0:
                break
            h[i_best] = 0
            h[j_best] = 1

        final_c = compute_c5(h.astype(float))
        if final_c < best_c:
            best_c = final_c
            best_h = h.copy()
            if best_c <= 0.3808:
                return best_h, best_c
    return best_h, best_c
