```
def enforce_sum_fast(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase sum
        idx = np.where(h < 1)[0]
        # if total capacity insufficient due to rounding errors, just fill to 1
        cap = (1.0 - h[idx]).sum()
        if cap > 0:
            # allocate proportionally
            add = diff * (1.0 - h[idx]) / cap
            h[idx] += add
        # after this diff may be very small
    else:
        # need to decrease sum
        idx = np.where(h > 0)[0]
        cap = h[idx].sum()
        if cap > 0:
            sub = (-diff) * h[idx] / cap
            h[idx] -= sub
    # Clip just in case
    np.clip(h, 0.0, 1.0, out=h)
    # compute residual diff again
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # adjust with small random steps
        for _ in range(10):
            if abs(residual) < 1e-12:
                break
            i = np.random.randint(h.size)
            if residual > 0:
                delta = min(residual, 1.0 - h[i])
            else:
                delta = max(residual, -h[i])
            h[i] += delta
            residual = target - h.sum()
    return h
```

    target_sum = n/2.0

        h = enforce_sum_fast(h, target_sum)
        pop.append(h)

    fitness = np.empty(pop_size)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]
    # DE parameters
    F = 0.8
    CR = 0.9
    # iteration counter
    iter = 0
    # loop until time budget
    while time.time() - start < budget * 0.97:
        for i in range(pop_size):
            # select a,b,c distinct from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # clip and enforce sum
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = enforce_sum_fast(mutant, target_sum)
            # crossover
            trial = np.empty_like(mutant)
            j_rand = rng.integers(0, n)
            for j in range(n):

                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            # enforce sum after crossover
            np.clip(trial, 0.0, 1.0, out=trial)
            trial = enforce_sum_fast(trial, target_sum)
            # evaluate
            f = compute_c5(trial)
            if f < fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_f:

                    best_h = trial.copy()

def enforce_sum_fast(h, target, rng=None):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if diff > 0:
        # increase sum
        idx = np.where(h < 1.0)[0]
        if idx.size == 0:
            # all ones, cannot increase, just return
            return h
        cap = (1.0 - h[idx]).sum()
        if cap > 0:
            add = diff * (1.0 - h[idx]) / cap
            h[idx] += add
    else:
        # decrease sum
        idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            return h
        cap = h[idx].sum()
        if cap > 0:
            sub = (-diff) * h[idx] / cap
            h[idx] -= sub
    # clip
    np.clip(h, 0.0, 1.0, out=h)
    # residual
    residual = target - h.sum()
    if abs(residual) > 1e-12 and rng is not None:
        # fix with small random adjustments up to 10 steps
        for _ in range(10):
            if abs(residual) < 1e-12:
                break
            i = rng.integers(0, n)
            if residual > 0:
                delta = min(residual, 1.0 - h[i])
            else:
                delta = max(residual, -h[i])
            h[i] += delta
            residual = target - h.sum()
    return h
```

def de_optimize(n, rng, budget, init=None):
    start = time.time()
    target = n / 2.0
    pop_size = 30
    pop = np.empty((pop_size, n), dtype=float)
    for i in range(pop_size):
        if i == 0 and init is not None:
            h = init.copy()
        else:
            h = rng.random(n)
        pop[i] = enforce_sum_fast(h, target, rng)
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = float(fitness[best_idx])
    # DE parameters
    F = 0.8
    CR = 0.9
    while time.time() - start < budget * 0.97:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = enforce_sum_fast(mutant, target, rng)
            # crossover
            j_rand = rng.integers(0, n)
            trial = np.empty_like(mutant)
            rand_vals = rng.random(n)
            mask = (rand_vals < CR) | (np.arange(n) == j_rand)
            trial[mask] = mutant[mask]
            trial[~mask] = pop[i][~mask]
            np.clip(trial, 0.0, 1.0, out=trial)
            trial = enforce_sum_fast(trial, target, rng)
            f = compute_c5(trial)
            if f < fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_f:
                    best_f = f
                    best_h = trial.copy()
        # occasional random replace of worst
        if rng.random() < 0.1:
            worst_idxs = np.argsort(fitness)[-3:]
            for wi in worst_idxs:
                new = rng.random(n)
                new = enforce_sum_fast(new, target, rng)
                new_f = compute_c5(new)
                if new_f < fitness[wi]:
                    pop[wi] = new
                    fitness[wi] = new_f
                    if new_f < best_f:
                        best_f = new_f
                        best_h = new.copy()
    # final hill climb on best_h
    elapsed = time.time() - start
    remaining = budget - elapsed
    if remaining > 0.2:
        # simple hill climb using worst_gradient
        h = best_h.copy()
        h = enforce_sum_fast(h, target, rng)
        # number of iterations proportional to remaining
        iters = int(min(200000, remaining * 1000))
        for it in range(iters):
            grad = worst_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = max_eps * 0.5  # half
                    h[i] -= eps
                    h[j] += eps
                    h = enforce_sum_fast(h, target, rng)
        f = compute_c5(h)
        if f < best_f:
            best_f = f
            best_h = h
    return best_h, best_f, n

import time
import numpy as np

# ---------- correlation utilities (FFT) ----------
def _compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) via FFT (full mode)."""
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, N)
    R = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * R, N)
    return conv[: 2 * n - 1]  # real values


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = _compute_corr_fft(h)
    return float(np.max(corr) * (2.0 / h.size))


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap at the worst shift."""
    n = h.size
    corr = _compute_corr_fft(h)
    idx = int(np.argmax(corr))                     # 0 … 2n‑2
    shift = idx - (n - 1)
    k = abs(shift)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


# ---------- feasibility: enforce sum ----------
def enforce_sum_fast(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h (in‑place) so that sum(h) == target while 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            cap = (1.0 - h[idx]).sum()
            if cap > 0:
                add = diff * (1.0 - h[idx]) / cap
                h[idx] += add
    else:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            cap = h[idx].sum()
            if cap > 0:
                sub = (-diff) * h[idx] / cap
                h[idx] -= sub
    np.clip(h, 0.0, 1.0, out=h)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for _ in range(10):
            if abs(residual) < 1e-12:
                break
            i = rng.integers(0, n)
            if residual > 0:
                delta = min(residual, 1.0 - h[i])
            else:
                delta = max(residual, -h[i])
            h[i] += delta
            residual = target - h.sum()
    return h


# ---------- Differential Evolution ----------
def de_optimize(n: int, rng: np.random.Generator, budget: float,
                init: np.ndarray | None = None) -> tuple[np.ndarray, float, int]:
    """Simple DE (rand/1/bin) with occasional random injects."""
    start = time.time()
    target = n / 2.0
    pop_size = 30
    pop = np.empty((pop_size, n), dtype=float)
    for i in range(pop_size):
        if i == 0 and init is not None:
            h = init.copy()
        else:
            h = rng.random(n)
        pop[i] = enforce_sum_fast(h, target, rng)
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = float(fitness[best_idx])

    F = 0.8
    CR = 0.9
    # main DE loop
    while time.time() - start < budget * 0.97:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = enforce_sum_fast(mutant, target, rng)

            j_rand = rng.integers(0, n)
            rand_vals = rng.random(n)
            mask = (rand_vals < CR) | (np.arange(n) == j_rand)
            trial = np.empty_like(mutant)
            trial[mask] = mutant[mask]
            trial[~mask] = pop[i][~mask]
            np.clip(trial, 0.0, 1.0, out=trial)
            trial = enforce_sum_fast(trial, target, rng)

            f = compute_c5(trial)
            if f < fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_f:
                    best_f, best_h = f, trial.copy()
                    if best_f <= 0.3808:
                        return best_h, best_f, n
        # occasional random re‑injection
        if rng.random() < 0.1:
            worst_idxs = np.argsort(fitness)[-3:]
            for wi in worst_idxs:
                new = rng.random(n)
                new = enforce_sum_fast(new, target, rng)
                fnew = compute_c5(new)
                if fnew < fitness[wi]:
                    pop[wi] = new
                    fitness[wi] = fnew
                    if fnew < best_f:
                        best_f, best_h = fnew, new.copy()
                        if best_f <= 0.3808:
                            return best_h, best_f, n

    # post‑DE gradient hill‑climb on the current best
    elapsed = time.time() - start
    remaining = budget - elapsed
    if remaining > 0.2:
        h = best_h.copy()
        h = enforce_sum_fast(h, target, rng)
        iters = int(min(200000, remaining * 1000))
        for _ in range(iters):
            grad = worst_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = max_eps * 0.5
                    h[i] -= eps
                    h[j] += eps
                    h = enforce_sum_fast(h, target, rng)
        f = compute_c5(h)
        if f < best_f:
            best_f, best_h = f, h.copy()
    return best_h, best_f, n


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    total_budget = budget_s

    # main parameters
    n = 800                               # number of samples
    init_up = None
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        init_up = np.interp(x_new, x_old, init)
        init_up = np.clip(init_up, 0.0, 1.0)
        init_up = enforce_sum_fast(init_up, n / 2.0, rng)

    # allocate most of the time to DE
    de_budget = total_budget * 0.90
    best_h, best_c, n_used = de_optimize(n, rng, de_budget, init_up)

    # final tidy‑up (very short)
    elapsed = time.time() - start_total
    if time.time() - start_total < total_budget:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum_fast(best_h, n_used / 2.0, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(n_used)
