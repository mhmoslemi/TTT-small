import time
import numpy as np
from scipy.optimize import minimize

# ---------- global placeholders for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk repair
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max of overlaps, for SLSQP."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = -(1.0 - h[k:])
        grad[k:] -= h[: n - k]
        grad += 1.0
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move mass from high‑gradient to low‑gradient entries."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Swap or transfer while preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.2:  # pure swap
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        n = h0.size
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        n = 512
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = h0.copy()
        best_c = compute_c5(best_h)

    # ----- simulated annealing (global search) -----
    current_h = best_h.copy()
    current_c = best_c
    T = 0.02
    eps_scale = 0.5
    SA_deadline = start + budget_s * 0.45
    it = 0
    while time.time() < SA_deadline:
        if rng.random() < 0.35:
            nh = guided_perturb(current_h,
                                worst_gradient(current_h), rng,
                                scale=eps_scale)
        else:
            nh = random_move(current_h, rng, scale=eps_scale)
        nc = compute_c5(nh)
        delta = nc - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_h, current_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        it += 1
        if it % 800 == 0:
            T *= 0.995
            eps_scale *= 0.996
            T = max(T, 1e-12)
            eps_scale = max(eps_scale, 1e-5)

    # ----- gradient polishing -----
    while time.time() < start + budget_s * 0.55:
        refined = gradient_refine(best_h, rng, step=0.02, max_iter=10)
        rc = compute_c5(refined)
        if rc < best_c:
            best_h, best_c = refined.copy(), rc
        else:
            break

    # ----- soft‑max SLSQP passes -----
    beta_seq = (10.0, 100.0, 1000.0, 5000.0, 20000.0, 80000.0)
    for beta in beta_seq:
        if time.time() > deadline - 30:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
        best_c = compute_c5(best_h)

    # ----- final low‑temperature SA fine‑tune -----
    while time.time() < deadline - 5:
        nh = guided_perturb(best_h,
                            worst_gradient(best_h), rng,
                            scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
