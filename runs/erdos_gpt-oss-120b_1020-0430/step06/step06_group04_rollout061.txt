def gradient_smoothmax_fast(h, beta):
    n = h.size
    v = np.correlate(h, 1-h, mode='full')  # length 2n-1
    # compute softmax weights
    bc = beta * v
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w /= w.sum()
    # Now compute convolution
    one_minus_h = 1.0 - h
    conv1 = np.convolve(w, one_minus_h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    offset = n - 1
    g = conv1[offset: offset + n] - conv2[offset: offset + n]
    return g, v # maybe also return v for computing C5

But w is length 2n-1, and conv with h length n => length 3n-2. offset = n-1 ensures we align to i.

def project_sum_and_bounds(h):
    # Clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # Adjust sum to target n/2 by adding constant diff then reclip iteratively
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    # Simple approach: add diff/n to all entries
    h += diff / n
    # Clip again
    np.clip(h, 0.0, 1.0, out=h)
    # Compute remaining diff, repeat a few times
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # distribute diff proportionally to remaining capacity
        cap_up = 1.0 - h
        cap_down = h
        # If diff positive, we need to increase; distribute to up capacities proportionally
        if diff > 0:
            total_cap = cap_up.sum()
            if total_cap > 0:
                h += diff * (cap_up / total_cap)
        else:
            total_cap = cap_down.sum()
            if total_cap > 0:
                h += diff * (cap_down / total_cap)
        np.clip(h, 0.0, 1.0, out=h)
    # At end ensure sum is within tolerance, else fallback to adjust_sum_exact
    if abs(h.sum() - target) > 1e-8:
        # fallback
        adjust_sum_exact(h, rng)  # need rng as argument; perhaps pass global rng?
    return h

def subgradient_descent(h, iters, base_lr, rng, deadline):
    n = h.size
    best_h = h.copy()
    v = np.correlate(h, 1 - h, mode='full')
    best_c = v.max()
    lr = base_lr
    for i in range(iters):
        if time.time() > deadline:
            break
        v = np.correlate(h, 1 - h, mode='full')
        max_val = v.max()
        # get worst shift indices
        idxs = np.where(v == max_val)[0]
        ks = idxs - (n - 1)
        # compute gradient g
        g = np.zeros_like(h, dtype=float)
        for k in ks:
            if k >= 0:
                g[: n - k] += 1 - h[k:]
                g[k:] -= h[: n - k]
            else:
                kk = -k
                g[kk:] += 1 - h[: n - kk]
                g[: n - kk] -= h[kk:]
        g /= max(1, len(ks))
        # step
        h_new = h - lr * g
        np.clip(h_new, 0, 1, out=h_new)
        adjust_sum_exact(h_new, rng)
        # compute new c
        v_new = np.correlate(h_new, 1 - h_new, mode='full')
        c_new = v_new.max()
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.9995
    return best_h, best_c

import time

# --------------------------------------------------------------
def compute_c5(h):
    """Return C5 for a discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000):
    """Project h onto {0≤h≤1, sum h = n/2} by moving mass between coordinates."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax_fast(h, beta):
    """
    Gradient of (1/β)·log Σ exp(β·overlap) w.r.t. h.
    Returns (gradient, overlaps_vector).
    """
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")       # overlaps for all integer shifts
    bc = beta * v
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= w.sum()
    # convolutions to assemble the gradient
    one_minus_h = 1.0 - h
    conv1 = np.convolve(w, one_minus_h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    off = n - 1
    g = conv1[off:off + n] - conv2[off:off + n]
    return g, v

def subgradient_descent(h0, iters, lr0, rng, deadline):
    """
    Projected sub‑gradient descent on the original C5 (max overlap).
    Returns (best_h, best_raw_overlap).
    """
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    v = np.correlate(h, 1.0 - h, mode="full")
    best_raw = v.max()
    lr = lr0
    for it in range(iters):
        if time.time() > deadline:
            break
        v = np.correlate(h, 1.0 - h, mode="full")
        max_val = v.max()

                h_up = h_new
                lr *= decay

    # ---- final reporting ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
