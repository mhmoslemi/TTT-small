CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    # smooth max of the overlap using log‑sum‑exp
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----- problem specific helpers -------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the shift attaining the maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift giving max overlap
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]    -= h[: n - k]
    return grad


def project_onto_capped_simplex(v: np.ndarray, target: float,
                               max_iter: int = 60,
                               tol: float = 1e-12) -> np.ndarray:
    """
    Euclidean projection of v onto the set {w: 0≤w_i≤1, sum w_i = target}.
    Uses bisection on the threshold t in w_i = clip(v_i - t, 0, 1).
    """
    lo = np.min(v) - 1.0    # guarantees all v_i - lo ≥ 1
    hi = np.max(v)          # guarantees all v_i - hi ≤ 0
    for _ in range(max_iter):
        t = (lo + hi) * 0.5
        w = np.clip(v - t, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = t
        else:
            hi = t
    return np.clip(v - t, 0.0, 1.0)

    a = np.array([0.0])
    while a.size < n:
        a = np.concatenate([a, 1.0 - a])
    return a[:n]

                if best_c <= 0.3808:
                    break

    # ---- Phase 4: final fine‑grained gradient sweeps ------------------------
    phase4_deadline = start + budget_s * 0.99
    while time.time() < phase4_deadline:
        grad = worst_gradient(best_h)
        nh = best_h - 0.02 * grad
        nh = project_onto_capped_simplex(nh, target_sum)
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh.copy(), c
            if best_c <= 0.3808:
                break

    # ---- final sanity enforcement -------------------------------------------
    best_h = project_onto_capped_simplex(best_h, target_sum)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
