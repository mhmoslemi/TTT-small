    N = h.size
    m = 1 << (2*N - 1).bit_length()  # next power of 2 >= 2N-1

def compute_c5_fft(h):
    N = h.shape[0]
    m = 1 << ((2*N - 1).bit_length())
    # compute FFTs
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * B, n=m)
    # take first 2*N-1 entries
    max_val = conv[: 2*N - 1].max()
    return (2.0 / N) * max_val

import time

# ---------------------------------------------------------------
def compute_c5_fft(h: np.ndarray, m: int) -> float:
    """C5 = (2/n) * max_{k} sum_i h_i (1-h_{i+k}) using FFT."""
    n = h.size
    H = np.fft.rfft(h, n=m)
    B = np.empty_like(H)
    B[0] = m - H[0]          # rfft(1) - H
    B[1:] = -H[1:]           # the rest are just -H
    conv = np.fft.irfft(H * B, n=m)
    return (2.0 / n) * conv[: 2 * n - 1].max()


def worst_gradient_binary(arr: np.ndarray) -> np.ndarray:
    """Gradient of the worst overlap (max_k O_k) for a binary array."""
    n = arr.size
    corr = np.correlate(arr, 1.0 - arr, mode="full")

    if k == 0:
        grad = 1.0 - 2.0 * arr
    else:
        grad[: n - k] -= arr[k:]
        grad[k:] -= arr[: n - k]
    return grad
