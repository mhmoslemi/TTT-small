def project_sum_box(h, target, max_iter=10):
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # distribute equally to all entries not at bound
        mask = (h > 0) & (h < 1)
        n_free = mask.sum()
        if n_free == 0:
            # all at bounds, adjust by scaling
            h += diff / len(h)
            h = np.clip(h, 0, 1)
            continue
        delta = diff / n_free
        h[mask] += delta
        # Clip
        np.clip(h, 0, 1, out=h)
    return h

def topk_gradient(h, K=5):

    idxs = np.argpartition(O, -K)[-K:]  # unsorted
    grad = np.zeros_like(h)
    n = h.size
    for idx in idxs:
        shift = idx - (n - 1)  # signed shift
        k = abs(shift)
        # compute grad_O as before
        g = np.ones_like(h)
        if k == 0:
            g = 1 - 2*h
        else:

def adam_optimize(h_initial, max_iter, time_limit, best_c, best_h, rng):
    n = h_initial.size
    target = n/2
    h = h_initial.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02
    beta1_pow = 1.0
    beta2_pow = 1.0
    start = time.time()
    for t in range(1, max_iter+1):
        # time check
        if time.time() - start > time_limit:
            break
        # compute O vector
        corr = np.correlate(h, 1.0 - h, mode='full')
        O = corr * (2.0 / n)  # dx = 2/n
        # top K shifts
        K = 3
        # get indices of top K O values
        idxs = np.argpartition(O, -K)[-K:]
        ks = np.abs(idxs - (n - 1))
        # compute gradient sum
        grad = np.zeros_like(h)
        for k in ks:
            if k == 0:
                grad += 1 - 2 * h
            else:
                # apply gradient
                grad[:n - k] -= h[k:]
                grad[k:] -= h[:n - k]
        grad /= K
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # clip & project sum
        np.clip(h, 0.0, 1.0, out=h)
        h = project_sum_box(h, target)
        # occasional random jiggle
        if rng.random() < 0.001:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(h[i], 1 - h[j]) * rng.random() * 0.02

def project_sum_box(h, target, max_iter=10):
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break

            continue

        np.clip(h, 0.0, 1.0, out=h)

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(h: np.ndarray, target: float, max_iter: int = 10) -> np.ndarray:
    """Project onto the intersection of the box [0,1]^n and the hyperplane sum=h."""
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        n_free = mask.sum()
        if n_free == 0:
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            continue
        delta = diff / n_free
        h[mask] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and enforce the sum constraint."""
    target = h.size / 2.0
    h = np.clip(h, 0.0, 1.0)
    return project_sum_box(h, target)


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence as 0/1 floats."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


# ---------- Adam optimizer with top‑k gradient ----------
def adam_optimize(
    h0: np.ndarray,
    max_iter: int,
    time_limit: float,
    best_c: float,
    best_h: np.ndarray,
    rng: np.random.Generator,
):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.02
    beta1_pow, beta2_pow = 1.0, 1.0
    start = time.time()

    for t in range(1, max_iter + 1):
        if t % 500 == 0 and (time.time() - start > time_limit):
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr * (2.0 / n)

        # top‑k shifts (k = |shift|)
        K = 3
        idxs = np.argpartition(O, -K)[-K:]
        ks = np.abs(idxs - (n - 1))

        grad = np.zeros_like(h)
        for k in ks:
            if k == 0:
                grad += 1.0 - 2.0 * h
            else:
                grad[: n - k] -= h[k:]
                grad[k:] -= h[: n - k]
        grad /= K

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h -= step
        np.clip(h, 0.0, 1.0, out=h)
        h = project_sum_box(h, target)

        # tiny random jiggle to escape plateaus
        if rng.random() < 0.001:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(h[i], 1.0 - h[j]) * rng.random() * 0.02
                h[i] -= delta
                h[j] += delta
                np.clip(h, 0.0, 1.0, out=h)
                h = project_sum_box(h, target)

        if t % 200 == 0:
            c = np.max(corr) * (2.0 / n)
            if c < best_c:
                best_c, best_h = c, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c, time.time() - start

    # final check
    c = np.max(corr) * (2.0 / n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c, time.time() - start


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # determine grid size
    n = len(initial_h_values) if "initial_h_values" in globals() else 512
    target = n / 2.0

    # construct diverse start vectors
    starts = []

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = enforce_sum(h0, rng)
        starts.append(h0)

    if (n & (n - 1)) == 0:          # power of two → Thue‑Morse possible
        starts.append(thue_morse_bits(n))

    rand_bin = np.zeros(n)
    rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(rand_bin)

    for _ in range(3):
        h_rand = rng.random(n)
        h_rand = enforce_sum(h_rand, rng)
        starts.append(h_rand)

    best_h = None
    best_c = np.inf

    def time_left():
        return budget_s - (time.time() - start_time)

    # short Adam runs on every start
    for h_start in starts:
        if time_left() < 5.0:
            break
        max_iter = 20000
        alloc = min(12.0, time_left() * 0.2)
        best_h, best_c, _ = adam_optimize(
            h_start, max_iter, alloc, best_c, best_h if best_h is not None else h_start, rng
        )
        if best_c <= 0.3808:
            break

    # fallback if nothing succeeded
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = enforce_sum(best_h, rng)

    # long refinement on the current best
    remaining = time_left()
    if remaining > 6.0 and best_c > 0.3808:
        max_iter = 250000
        alloc = remaining - 4.0
        best_h, best_c, _ = adam_optimize(best_h, max_iter, alloc, best_c, best_h, rng)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
