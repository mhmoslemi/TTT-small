def compute_c5(h):
    N = h.size
    L = 2 * N
    h_pad = np.zeros(L, dtype=float)
    g_pad = np.zeros(L, dtype=float)
    h_pad[:N] = h
    g_pad[:N] = 1 - h
    # compute linear convolution using FFT
    fft_h = np.fft.rfft(h_pad)
    fft_g = np.fft.rfft(g_pad)
    conv = np.fft.irfft(fft_h * fft_g, n=L)

def binary_sa(n, rng, time_budget):
    start = time.time()
    # initial random binary vector
    ones = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[ones] = 1.0
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr

    # compute zeros array
    zeros = np.setdiff1d(np.arange(n), ones, assume_unique=True)

    # temperature schedule: start T0, end T_end
    T0 = 0.02; T_end = 1e-5

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # propose swap
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            c_curr = c_new
            # update arrays
            ones[i_pos] = j
            zeros[j_pos] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                # early exit if beat target
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # reject: revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def enforce_sum(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    # we will limit number of iterations for safety
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            # add
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        it += 1
    # If not fixed exactly, do deterministic redistribution
    if abs(diff) > 1e-12:
        # compute needed adjustment as small epsilon spread across all
        eps = diff / n
        h += eps
        # clip again and correct sum via recursion (but diff small)
        h = np.clip(h,0,1)
        # final adjust
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            # fallback: scale to meet sum
            scaling = target / h.sum()
            h *= scaling
            h = np.clip(h,0,1)
    return h

import time
import numpy as np

def compute_c5_fft(h: np.ndarray) -> float:
    N = h.size
    L = 2 * N
    h_pad = np.zeros(L, dtype=float)
    g_pad = np.zeros(L, dtype=float)
    h_pad[:N] = h
    g_pad[:N] = 1.0 - h
    conv = np.fft.irfft(np.fft.rfft(h_pad) * np.fft.rfft(g_pad), n=L)
    return conv.max() * (2.0 / N)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        eps = diff / n
        h += eps
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            scaling = target / h.sum()
            h *= scaling
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def adjust_binary_sum(h_bin: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h_bin.size // 2
    cur = int(h_bin.sum())
    while cur > target:
        ones = np.where(h_bin == 1)[0]
        i = rng.choice(ones)
        h_bin[i] = 0
        cur -= 1
    while cur < target:
        zeros = np.where(h_bin == 0)[0]
        i = rng.choice(zeros)
        h_bin[i] = 1
        cur += 1
    return h_bin

def binary_sa(n: int, rng: np.random.Generator, time_budget: float, init_h: np.ndarray = None):
    start = time.time()
    if init_h is None:
        ones = rng.choice(n, n // 2, replace=False)
        h = np.zeros(n, dtype=float)
        h[ones] = 1.0
    else:
        h = init_h.astype(float).copy()
        ones = np.where(h == 1)[0]
    zeros = np.setdiff1d(np.arange(n), ones, assume_unique=True)
    c_curr = compute_c5_fft(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.02, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i, j = ones[i_pos], zeros[j_pos]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            ones[i_pos] = j
            zeros[j_pos] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = h0.copy()
    N = h.size
    c_curr = compute_c5_fft(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.01, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        inc = rng.random() < 0.5
        max_eps = min(h[i], 1.0 - h[j]) if inc else min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * (T / T0)
        if eps <= 0:
            continue
        h_new = h.copy()
        if inc:
            h_new[i] -= eps
            h_new[j] += eps
        else:
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c

def worst_gradient(h: np.ndarray) -> np.ndarray:
    N = h.size
    L = 2 * N
    h_pad = np.zeros(L, dtype=float)
    g_pad = np.zeros(L, dtype=float)
    h_pad[:N] = h
    g_pad[:N] = 1.0 - h
    conv = np.fft.irfft(np.fft.rfft(h_pad) * np.fft.rfft(g_pad), n=L)
    idx = np.argmax(conv) - (N - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:N - k] += 1.0 - h[k:]
        grad[k:] -= h[:N - k]
    return grad

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    N = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        target = N / 2.0
        diff = target - h.sum()
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 500:
            i = rng.integers(0, N)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            attempts += 1
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5_fft(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # Phase: binary SA on 2048 points
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0:
        n_bin = 2048
        t_bin = min(remaining * 0.70, 800.0)
        init_bin = None
        if best_h is not None:
            up = upsample_h(best_h, n_bin)
            bin_vec = (up >= 0.5).astype(float)
            bin_vec = adjust_binary_sum(bin_vec, rng)
            init_bin = bin_vec
        h_bin, c_bin = binary_sa(n_bin, rng, t_bin, init_h=init_bin)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, n_bin
        if best_c <= 0.3808:
            best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
            final_c = compute_c5_fft(best_h)
            return best_h.tolist(), float(final_c), int(best_h.size)

    # Phase: continuous SA refinement
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h_cont, c_cont = continuous_sa(best_h, rng, remaining)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # Phase: gradient tidy‑up if time permits
    remaining = total_budget - (time.time() - start)
    if remaining > 2.0 and best_h is not None:
        h_grad = gradient_refine(best_h, rng, step=0.015, max_iter=15)
        c_grad = compute_c5_fft(h_grad)
        if c_grad < best_c:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
