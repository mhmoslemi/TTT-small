def project_box_sum(x, target):
    # Clip to [0,1]
    # Find lambda via binary search to satisfy sum clip(x - lam) = target.
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo+hi)/2
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return proj
```

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0*h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad
```

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------- helpers ---------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst overlap O_k = Σ h_i (1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]      -= h[: n - k]
    return grad


def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the intersection of the hyperplane sum(x)=target
    and the box 0≤x≤1.
    Solution is clip(x-λ,0,1) with λ chosen by bisection.
    """
    lo, hi = -1.0, 1.0               # guarantees sum(lo)≥target, sum(hi)≤target
    for _ in range(30):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Compatibility wrapper – projects onto the feasible set."""
    return project_box_sum(h, h.size / 2.0)


# ------------------- SLSQP soft‑max refinement ---------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp for numerical stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Run SLSQP on the smooth soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # repair with a quick projection
        x = enforce_sum(x, rng)
    return x


# --------------------------- main -----------------------------------

    start = time.time()

    # ---- initialization -------------------------------------------------
    if "initial_h_values" in globals():

        n = h0.size

    target = n / 2.0
    h = project_box_sum(h0, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- projected subgradient descent ---------------------------------
    a = 1.0                                     # initial step scale
    t = 0
    no_improve = 0
    max_no_improve = 40000

    while True:
        t += 1
        if time.time() - start > budget_s * 0.70:

        grad = worst_gradient(h)
        step = a / (np.sqrt(t))
        h = h - step * grad

            best_c = c
            best_h = h.copy()
            no_improve = 0
            if best_c <= 0.3808:

        else:
            no_improve += 1

        if no_improve >= max_no_improve:

            h = best_h + rng.normal(scale=0.02, size=n)

            no_improve = 0

    # ---- SLSQP refinement (soft‑max) -----------------------------------
    # allocate remaining time proportionally
    remaining = budget_s - (time.time() - start)
