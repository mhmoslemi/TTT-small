def binary_search_opt(h_bin, rng, deadline, target_max):
    # hill climbing: try to reduce max_corr under target_max
    n = h_bin.size

    best_max = int(best_corr.max())

    if best_max <= target_max:

    iter = 0
    no_improve = 0

def weighted_binary_round(h, rng):
    n = h.size
    k = n // 2
    probs = h / h.sum()
    idx = rng.choice(n, size=k, replace=False, p=probs)
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0
    return h_bin
```

def binary_hill_climb_target(h_bin, rng, deadline, target_max):
    n = h_bin.size
    best_h = h_bin.copy()
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    best_max = int(corr.max())
    if best_max <= target_max:
        return best_h, best_max
    no_improve = 0
    it = 0
    while time.time() < deadline and it < 500000:
        if it % 1000 == 0:
            if time.time() > deadline:
                break
        # recompute corr and max
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        max_val = int(corr.max())
        if max_val <= target_max:
            best_h = h_bin.copy()
            best_max = max_val
            break
        worst_idxs = np.where(corr == max_val)[0]
        idx = rng.choice(worst_idxs)
        shift = idx - (n - 1)
        # overlapping ones
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            overlap = (h_bin[i_range] == 1) & (h_bin[i_range + shift] == 0)
            s1 = i_range[overlap]
        else:
            sh = -shift
            i_range = np.arange(sh, n)
            overlap = (h_bin[i_range] == 1) & (h_bin[i_range - sh] == 0)
            s1 = i_range[overlap]
        if s1.size == 0:
            # fallback: any 1
            ones = np.flatnonzero(h_bin == 1)
            i = rng.choice(ones)
        else:
            i = rng.choice(s1)
        zeros = np.flatnonzero(h_bin == 0)
        j = rng.choice(zeros)
        # swap
        h_bin[i] = 0
        h_bin[j] = 1
        # evaluate new max
        new_corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        new_max = int(new_corr.max())
        if new_max <= max_val:
            # accept
            if new_max < best_max:
                best_h = h_bin.copy()
                best_max = new_max
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[i] = 1
            h_bin[j] = 0
            no_improve += 1
        it += 1
        if no_improve >= 10000:
            # randomize using best_h as base
            h_bin = best_h.copy()
            # perform some random swaps
            for _ in range(200):
                ones = np.flatnonzero(h_bin == 1)
                zeros = np.flatnonzero(h_bin == 0)
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_bin[i], h_bin[j] = 0, 1
            no_improve = 0
    return best_h, best_max

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- basic utilities --------------------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# -------------------- smooth‑max gradient --------------------
def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    omh = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * omh[shift:]
            g[shift:]    -= wi * h[: n - shift]
        else:
            sh = -shift
            g[sh:]       += wi * omh[: n - sh]
            g[: n - sh]  -= wi * h[sh:]
    return g

def gradient_descent_smoothmax(h, betas, base_step, steps_per_beta, rng, deadline):
    n = h.size
    for beta in betas:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h

# -------------------- worst‑shift refinement --------------------
def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    n = h.size
    for _ in range(iters):
        if time.time() > deadline:
            break
        v = np.correlate(h, 1.0 - h, mode="full")
        max_val = v.max()
        idxs = np.where(v == max_val)[0]
        g = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                g[: n - k] += 1.0 - h[k:]
                g[k:]    -= h[: n - k]
            else:
                kk = -k
                g[kk:]   += 1.0 - h[: n - kk]
                g[: n - kk] -= h[kk:]
        g /= max(1, len(idxs))
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.95
    return h

# -------------------- SLSQP polish --------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x
