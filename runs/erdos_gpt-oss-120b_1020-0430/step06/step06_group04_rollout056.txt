def worst_shift_gradient_vectorized(h):
    n = h.shape[0]
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idx = np.argmax(v)
    k = idx - (n - 1)
    g = np.zeros_like(h)
    if k >= 0:
        # for j in [0, n-1-k]: derivative includes (1 - h_{j+k})
        g[: n - k] += 1.0 - h[k:]
        # for j in [k, n-1]: derivative includes -h_{j - k}
        g[k:] -= h[: n - k]
    else:
        kk = -k
        # for j in [kk, n-1]: derivative includes (1 - h_{j-kk})
        g[kk:] += 1.0 - h[: n - kk]
        # for j in [0, n-1-kk]: derivative includes -h_{j + kk}
        g[: n - kk] -= h[kk:]
    return g, max_val

    n = h.shape[0]
    best_c = compute_c5(h)
    step = base_step
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g, cur_max = worst_shift_gradient_vectorized(h)
        if np.linalg.norm(g) < 1e-14:
            break
        improved = False
        for mult in (1.0, 0.5, 0.25):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                h = cand
                best_c = c
                improved = True
                break
        if not improved:
            no_improve += 1
            step *= 0.9
            if no_improve >= 200:
                # random perturbation to escape
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                delta = min(0.1, 1.0 - h[i], h[j])
                if delta > 0:
                    h[i] += delta
                    h[j] -= delta
                    adjust_sum_exact(h, rng)
                no_improve = 0
        else:
            no_improve = 0
    return h

            break

        else:

        diff = target - h.sum()
    return h

        g[kk:] += 1.0 - h[: n - kk]

    return np.interp(xt, xs, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97   # safety margin

    n = 1440
    parent = globals().get("initial_h_values", None)

    best_h = None
    best_c = np.inf
    restart = 0

    while time.time() < deadline:
        if restart == 0 and parent is not None:
            h = upsample_h(parent, n)
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # allocate a modest budget of iterations for this restart
        remaining = deadline - time.time()
        iters = max(500, int(remaining / 0.002 * 0.3))
        h = gradient_descent_worst_shift(
            h, iters, base_step=0.15, rng=rng, deadline=deadline
        )
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        restart += 1

    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # intensive final refinement
    if time.time() < deadline:
        best_h = gradient_descent_worst_shift(
            best_h, max_iters=8000, base_step=0.05, rng=rng, deadline=deadline
        )
    if time.time() < deadline:
        best_h = slsqp_optimize(best_h, beta=50000, rng=rng, max_iter=800)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
