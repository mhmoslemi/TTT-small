import time
import numpy as np

# ---------- basic utilities -------------------------------------------------
def compute_O(h: np.ndarray) -> np.ndarray:
    """Return O_k = sum(h) - (h * h) correlation for all shifts."""
    s = h.sum()
    conv = np.correlate(h, h, mode="full")
    return s - conv


def compute_c5(h: np.ndarray) -> float:
    return (2.0 / h.size) * compute_O(h).max()


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            zeros = np.where(h < 1.0)[0]
            if zeros.size == 0:
                break
            i = rng.choice(zeros)
            add = min(diff, 1.0 - h[i])
        else:
            ones = np.where(h > 0.0)[0]
            if ones.size == 0:
                break
            i = rng.choice(ones)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h


# ---------- simulated annealing for binary vectors -------------------------
def simulated_annealing_binary(h0: np.ndarray, rng: np.random.Generator,
                               time_limit: float) -> tuple:
    """Binary SA on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    O = compute_O(h)
    cur_max = O.max()
    best_h = h.copy()
    best_max = cur_max

    offset = n - 1
    pad = np.concatenate([np.zeros(offset), h, np.zeros(offset)])
    shifts = np.arange(-(n - 1), n, dtype=int)          # length 2n‑1
    zero_idx = offset

    start_temp = 1.0
    it = 0
    while time.time() < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # we only move a whole unit (binary swap)
        if hi > 0.5 and hj < 0.5:
            epsilon = 1.0
        elif hj > 0.5 and hi < 0.5:
            i, j = j, i
            epsilon = 1.0
        else:
            continue

        # delta_O for all shifts
        term_i = pad[i + offset + shifts] + pad[i + offset - shifts]
        term_j = pad[j + offset + shifts] + pad[j + offset - shifts]
        delta_O = epsilon * (term_i - term_j)
        delta_O[zero_idx] = 0.0                     # shift 0 stays unchanged

        new_max = (O + delta_O).max()
        temp = start_temp * (0.9995 ** it)
        accept = new_max < cur_max or rng.random() < np.exp(
            -(new_max - cur_max) / (temp + 1e-12)
        )
        if accept:
            O += delta_O
            cur_max = new_max
            # update h and padded version
            h[i] -= epsilon
            h[j] += epsilon
            pad[i + offset] -= epsilon
            pad[j + offset] += epsilon
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        it += 1

        # occasional random reshuffle to inject diversity
        if it % 5000 == 0 and rng.random() < 0.1:
            for _ in range(5):
                a, b = rng.integers(0, n, size=2)
                if h[a] > 0.5 and h[b] < 0.5:
                    h[a] = 0.0
                    h[b] = 1.0
                    pad[a + offset] = 0.0
                    pad[b + offset] = 1.0
            O = compute_O(h)
            cur_max = O.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max

        # delta_O similar to binary case, but correct shift‑0 term
        term_i = pad[i + offset + shifts] + pad[i + offset - shifts]
        term_j = pad[j + offset + shifts] + pad[j + offset - shifts]
        delta_O = epsilon * (term_i - term_j)

        # exact shift‑0 correction for fractional ε
        hi, hj = h[i], h[j]
        delta_O[zero_idx] = 2.0 * epsilon * (hi - hj) - 2.0 * epsilon * epsilon

        new_max = (O + delta_O).max()
        if new_max < cur_max:
            O += delta_O
            cur_max = new_max
            h[i] -= epsilon
            h[j] += epsilon
            pad[i + offset] -= epsilon
            pad[j + offset] += epsilon
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max


# ---------- main entry point ------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = budget_s

    # ---- initial vector (n = 256) ----------------------------------------
    n0 = 256
    if "initial_h_values" in globals():
        full = np.array(initial_h_values, dtype=float)
        # down‑sample by factor two (preserves sum for binary balanced)
        h0 = full[::2]
        if h0.size > n0:
            h0 = h0[:n0]
        elif h0.size < n0:
            h0 = np.concatenate([h0, np.zeros(n0 - h0.size)])
    else:
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)        # ensure exact n0/2 total

    # ---- binary SA on coarse grid ----------------------------------------
    t0_end = start + budget * 0.35
    best_h0, best_max0 = simulated_annealing_binary(h0, rng, t0_end)

    # ---- upscale to n = 512 ------------------------------------------------
    best_h1 = np.repeat(best_h0, 2)
    n1 = best_h1.size                  # 512
    best_h1 = np.clip(best_h1, 0.0, 1.0)  # safety

    # ---- binary SA on fine grid -------------------------------------------
    t1_end = start + budget * 0.70
    best_h1, best_max1 = simulated_annealing_binary(best_h1, rng, t1_end)

    # ---- fractional refinement ---------------------------------------------
    t2_end = start + budget * 0.85
    best_h1, best_max1 = fractional_local_search(best_h1, rng, t2_end)

    # ---- final sanity and bound -------------------------------------------
    best_h1 = enforce_sum(best_h1, rng)
    final_c5 = compute_c5(best_h1)

    return best_h1.tolist(), float(final_c5), int(best_h1.size)
