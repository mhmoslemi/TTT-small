import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto [0,1]^n with sum(v)=target (exact, via bisection)."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # tiny adjustment to hit target exactly
        for i in range(x.size):
            if diff > 0.0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0.0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) <= 1e-12:
                break
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Subgradient of the maximal overlap (worst shift)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)   # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]     -= h[: n - k]
    return grad


# ---------- Thue–Morse ----------
def thue_morse_bits(n: int) -> np.ndarray:
    s = np.array([0.0])
    while s.size < n:
        s = np.concatenate([s, 1.0 - s])
    return s[:n]

    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def random_continuous(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_sum(v, n / 2.0)


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_sum(x, CURRENT_TARGET)
    return x

    # ----- main gradient descent + random swaps -----
    while time.time() - start < budget_s * 0.80:
        step = step0 * (0.998 ** it)
        if step < 1e-7:
            step = 1e-7
        grad = worst_gradient(h)
        new_h = project_sum(h - step * grad, target)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c, best_h, h = c, new_h.copy(), new_h.copy()
            last_improve = it
        else:
            # occasional random mass exchange to escape plateaus
            if rng.random() < 0.25:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    eps = min(h[i], 1.0 - h[j]) * rng.random() * 0.2
                    if eps > 1e-12:
                        h2 = h.copy()
                        h2[i] -= eps
                        h2[j] += eps
                        c2 = compute_c5(h2)
                        if c2 < best_c:
                            best_c, best_h, h = c2, h2.copy(), h2.copy()
                            last_improve = it
        # restart if stuck
        if it - last_improve > 5000:
            h = random_balanced_binary(n, rng)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
                last_improve = it
        it += 1

    # ----- SLSQP refinement with increasing beta -----
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
        if time.time() - start > budget_s * 0.96:
            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=1800)
        c = compute_c5(h_opt)
        if c < best_c:
            best_c, best_h = c, h_opt.copy()
            if best_c <= 0.3808:
                break

    # ----- final fine‑grained local search -----
    for step in (0.08, 0.04, 0.02, 0.01):
        if time.time() - start > budget_s * 0.99:
            break
        for _ in range(500):
            grad = worst_gradient(best_h)
            nh = project_sum(best_h - step * grad, target)
            c = compute_c5(nh)
            if c < best_c:
                best_c, best_h = c, nh.copy()
                if best_c <= 0.3808:
                    break

    # final projection & verification
    best_h = project_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
