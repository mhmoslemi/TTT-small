def compute_c5(h):
    # h is 1d numpy float (length n)
    n = h.shape[0]
    m = 2*n - 1
    # compute correlation via FFT using length L = next power of two >= m
    L = 1 << (m-1).bit_length()
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:m]
    return np.max(corr) * (2.0 / n)

import numpy as np
import time

# ----- helper to enforce sum(h)=n/2 while staying in [0,1] -----
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ----- discrete C5 evaluation (full correlation) -----
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ----- gradient descent on the worst shift -----
def descend(h: np.ndarray, rng: np.random.Generator,
            max_iter: int, init_step: float) -> np.ndarray:
    n = h.size
    step = init_step
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr) - (n - 1))
        k = abs(idx)
        grad = np.ones_like(h)
        if k != 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        step *= 0.995
    return h


# ----- main entry point -----
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024                                 # working resolution

    # ---- initialise best candidate ----
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        if base.size < n:
            rep = n // base.size
            h0 = np.repeat(base, rep)[:n]
        else:
            h0 = base[:n]
        h0 = enforce_sum(h0, rng)
    else:
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- exploratory phase (many short descents) ----
    explore_deadline = start + budget_s * 0.78
    while time.time() < explore_deadline:
        if rng.random() < 0.3:
            # perturb current best a little
            h_start = best_h.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = rng.random() * 0.08
                eps = min(eps, h_start[i], 1.0 - h_start[j])
                h_start[i] -= eps
                h_start[j] += eps
        else:
            # fresh random binary vector with correct sum
            h_start = np.zeros(n)
            h_start[rng.choice(n, n // 2, replace=False)] = 1.0

        h_new = descend(h_start, rng, max_iter=80, init_step=0.02)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # ---- final refinement (longer descent) ----
    final_deadline = start + budget_s * 0.95
    if time.time() < final_deadline:
        h_ref = descend(best_h.copy(), rng, max_iter=500, init_step=0.01)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    # ---- final repair and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    best_c = compute_c5(best_h)
    return best_h.tolist(), float(best_c), int(n)
