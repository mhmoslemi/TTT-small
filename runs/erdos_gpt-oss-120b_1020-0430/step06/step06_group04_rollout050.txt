                    break

def project_box_sum(h, lo, hi, target):
    # Clip to bounds
    np.clip(h, lo, hi, out=h)
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) < 1e-12:
        return
    # Create list of indices where we can increase or decrease
    inc_idxs = np.where(h < hi)[0]
    dec_idxs = np.where(h > lo)[0]
    i = 0
    j = 0
    while abs(diff) > 1e-12:
        if diff > 0:
            if i >= len(inc_idxs):
                break
            idx = inc_idxs[i]
            can = hi - h[idx]
            delta = min(can, diff)
            h[idx] += delta
            diff -= delta
            if h[idx] >= hi - 1e-12:
                i += 1
        else:
            if j >= len(dec_idxs):
                break
            idx = dec_idxs[j]
            can = h[idx] - lo
            delta = min(can, -diff)
            h[idx] -= delta
            diff += delta
            if h[idx] <= lo + 1e-12:
                j += 1
```

def worst_shift_line_refine(h, rng, iters, deadline):
    n = h.size
    target = n/2
    for _ in range(iters):
        if time.time() > deadline:
            break
        # compute overlaps and worst shift
        v = np.correlate(h, 1.0 - h, mode="full")
        max_i = np.argmax(v)
        k = max_i - (n - 1)
        # compute gradient g for this shift
        g = np.zeros_like(h)
        if k >= 0:
            g[: n - k] = 1.0 - h[k:]
            g[k :] -= h[: n - k]
        else:
            kk = -k
            g[kk:] = 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
        # compute max allowable t (preserve bounds)
        pos = g > 0
        neg = g < 0
        t_max = np.inf
        if np.any(pos):
            t_max = min(t_max, np.min(h[pos] / g[pos]))
        if np.any(neg):
            t_max = min(t_max, np.min((1.0 - h[neg]) / (-g[neg])))
        if t_max <= 1e-12:
            continue
        # ternary search
        lo = 0.0
        hi = t_max
        for _ in range(8):
            t1 = lo + (hi - lo)/3
            t2 = hi - (hi - lo)/3
            h1 = h - t1 * g
            h2 = h - t2 * g
            # no need to clip as within bounds
            f1 = np.max(np.correlate(h1, 1.0 - h1, mode="full") * (2.0 / n))
            f2 = np.max(np.correlate(h2, 1.0 - h2, mode="full") * (2.0 / n))
            if f1 < f2:
                hi = t2
            else:
                lo = t1
        t_best = (lo + hi)/2
        h = h - t_best * g
        # Clip for numeric safety
        np.clip(h, 0.0, 1.0, out=h)
        # sum should be preserved; adjust if needed
        project_box_sum(h, 0.0, 1.0, target)
    return h
```

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_box_sum(h, lo, hi, target):
    np.clip(h, lo, hi, out=h)
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) < 1e-12:
        return
    # increase if diff positive
    if diff > 0:
        inc = np.where(h < hi)[0]
        for idx in inc:
            can = hi - h[idx]
            add = min(can, diff)
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        dec = np.where(h > lo)[0]
        for idx in dec:
            can = h[idx] - lo
            sub = min(can, -diff)
            h[idx] -= sub
            diff += sub
            if diff >= -1e-12:
                break

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def pgd_smoothmax(h, betas, base_step, iters_per_beta, rng, deadline):
    n = h.size
    target = n / 2.0
    cur_c = compute_c5(h)
    for beta in betas:
        step = base_step
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return h, cur_c
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            # backtracking
            for _ in range(8):
                cand = h - step * g
                project_box_sum(cand, 0.0, 1.0, target)
                c = compute_c5(cand)
                if c < cur_c - 1e-12:
                    h = cand
                    cur_c = c
                    step = min(step * 1.2, base_step * 3)
                    break
                else:
                    step *= 0.5
            else:
                step = base_step
    return h, cur_c

def worst_shift_line_refine(h, rng, iters, deadline):
    n = h.size
    target = n / 2.0
    cur_c = compute_c5(h)
    for _ in range(iters):
        if time.time() > deadline:
            break
        v = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(v)
        k = max_idx - (n - 1)
        # gradient for worst shift
        g = np.zeros_like(h)
        if k >= 0:
            g[: n - k] = 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk:]   = 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
        # max feasible step
        pos = g > 0
        neg = g < 0
        t_max = np.inf
        if np.any(pos):
            t_max = min(t_max, np.min(h[pos] / g[pos]))
        if np.any(neg):
            t_max = min(t_max, np.min((1.0 - h[neg]) / (-g[neg])))
        if t_max <= 1e-12:
            continue
        lo, hi = 0.0, t_max
        for _ in range(8):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            c1 = compute_c5(h - t1 * g)
            c2 = compute_c5(h - t2 * g)
            if c1 < c2:
                hi = t2
            else:
                lo = t1
        t_best = (lo + hi) / 2.0
        h = h - t_best * g
        np.clip(h, 0.0, 1.0, out=h)
        project_box_sum(h, 0.0, 1.0, target)
        cur_c = compute_c5(h)
    return h, cur_c

CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq","fun":sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                    constraints=cons,
                    options={"maxiter":max_iter,"ftol":1e-12,"disp":False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum precisely
    project_box_sum(x, 0.0, 1.0, n/2.0)
    return x

    rng = np.random.default_rng(seed)
