    n = h0.size
    CURRENT_TARGET = n / 2.0

import time
import numpy as np
import scipy.optimize as opt

# Global variables for SLSQP soft‑max
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 1500):
    """Run SLSQP on the soft‑max for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    if abs(diff) > 1e-12:
        # tiny uniform correction
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            i = rng.integers(0, n)
            h[i] += diff
            h[i] = np.clip(h[i], 0.0, 1.0)
    return h


def compute_O(h: np.ndarray) -> np.ndarray:
    """Exact overlap array O_k = Σ h_i(1‑h_{i+k})."""
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5_from_O(O: np.ndarray, n: int) -> float:
    return np.max(O) * (2.0 / n)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst overlap (max O_k)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def swap_hill_climb(h: np.ndarray, O: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Random swap hill‑climbing that preserves the sum."""
    n = h.size
    pad = n - 1
    H = np.concatenate([np.zeros(pad), h, np.zeros(pad)])
    idx = np.arange(-pad, n)  # length 2n‑1
    max_O = np.max(O)
    start = time.time()
    accepted = 0
    while time.time() - start < time_limit:
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        contrib_i = H[i + pad + idx] + H[i + pad - idx]
        contrib_j = H[j + pad + idx] + H[j + pad - idx]
        delta = contrib_i - contrib_j
        new_max = np.max(O + delta)
        if new_max < max_O - 1e-12:
            # accept swap
            h[i] = 0.0
            h[j] = 1.0
            H[i + pad] = 0.0
            H[j + pad] = 1.0
            O += delta
            max_O = new_max
            accepted += 1
            if accepted % 5000 == 0:
                O = compute_O(h)
                max_O = np.max(O)
                H = np.concatenate([np.zeros(pad), h, np.zeros(pad)])
    return h, O
