    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)/2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))

        else:

import time
import math
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while enforcing sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def softmax_weights(corr: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over correlation values."""
    max_corr = np.max(corr)
    exp_term = np.exp(beta * (corr - max_corr))
    return exp_term / np.sum(exp_term)


def weighted_grad(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max objective."""
    n = h.size
    # correlation of h with the weight kernel w
    corr_hw = np.correlate(h, w, mode="full")
    idx = np.arange(n)
    # f_pos[j] = sum_k w_k * h[j+k]
    f_pos = corr_hw[(2 * n - 2) - idx]
    # f_neg[j] = sum_k w_k * h[j-k]
    f_neg = corr_hw[(2 * n - 2) + idx]
    return 1.0 - (f_pos + f_neg)


def gradient_step(h: np.ndarray, beta: float, step: float):
    """One projected gradient step on the soft‑max surrogate."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(corr, beta)
    grad = weighted_grad(h, w)
    v = h - step * grad
    v = project_sum_box(v, n / 2.0)
    c_new = np.max(corr) * (2.0 / n)  # actual C5 of the *old* point, recompute later
    return v, c_new


def worst_shift_and_grad(h: np.ndarray):
    """Return current C5, the worst shift, and its gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    c = np.max(corr) * (2.0 / n)
    return c, k, grad


def thue_morse_bits(n: int) -> np.ndarray:
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]
