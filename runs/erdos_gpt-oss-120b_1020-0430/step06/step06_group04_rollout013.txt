def targeted_hill_climb_binary(h, rng, deadline):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c

    # Precompute dx as 2.0/n but compute_c5 already multiplies.

    iter = 0
    while time.time() < deadline:
        # compute worst shift(s)
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        maxv = corr.max()
        # Get a shift index random among those attaining max
        # indices where corr == maxv
        idxs = np.where(corr == maxv)[0]
        # pick random idx
        idx = rng.choice(idxs)
        k = idx - (n - 1)
        # compute candidate i and j
        shifted = np.roll(cur_h, -k)
        i_cand = np.where((cur_h == 1) & (shifted == 0))[0]
        j_cand = np.where((cur_h == 0) & (shifted == 1))[0]

        if i_cand.size == 0 or j_cand.size == 0:
            # cannot improve this shift, perhaps try another shift
            # pick random i, j?
            # just continue
            continue

        # pick random i, j
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)

        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0

        new_c = compute_c5(cur_h)

        # Accept if <= cur_c (improvement) else maybe accept with prob
        # temperature schedule: T = init_T * (0.9995 ** iter)
        # init_T based on cur_c
        # but we can also just reject if no improvement
        if new_c <= cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0

        iter += 1

    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for discrete h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Modify h in‑place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- smooth‑max gradient ----------
def gradient_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log∑exp(β·overlap) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)          # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g


def gradient_descent_smoothmax(
    h: np.ndarray,
    betas,
    base_step: float,
    it_per_beta: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """Projected gradient descent on smooth‑max for a schedule of β."""
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h


# ---------- binary targeted hill climbing ----------
def targeted_hill_climb_binary(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """Swap‑based hill climbing that always keeps the max overlap ≤ current best."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c

    while time.time() < deadline:
        # current worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        maxv = corr.max()
        idxs = np.where(corr == maxv)[0]
        idx = rng.choice(idxs)
        k = idx - (n - 1)

        # candidates that would reduce overlap at this shift
        shifted = np.roll(cur_h, -k)
        i_cand = np.where((cur_h == 1) & (shifted == 0))[0]
        j_cand = np.where((cur_h == 0) & (shifted == 1))[0]

        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            new_c = compute_c5(cur_h)
            if new_c <= cur_c:          # accept non‑worsening move
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur_h.copy()
            else:                       # revert
                cur_h[i] = 1.0
                cur_h[j] = 0.0
        else:
            # fallback random swap (still require non‑worsening)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

                    best_c = new_c
                    best_h = cur_h.copy()
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    return best_h, best_c


# ---------- optimisation for a fixed n ----------
def optimise_n(n: int, restarts: int, rng: np.random.Generator, deadline: float):
    """Run several restarts for a given n and return the best binary vector found."""
    best_h_n = None
    best_c_n = np.inf

    parent = globals().get("initial_h_values", None)

    for _ in range(restarts):
        if time.time() > deadline:
            break

        # ----- initialise binary vector -----
        if parent is not None and len(parent) == n:
            h0 = np.rint(parent).astype(float)
            adjust_sum_exact(h0, rng)
        elif parent is not None and len(parent) < n:
            # simple up‑sampling by repetition
            factor = n // len(parent)
            extra = n - factor * len(parent)
            h_up = np.repeat(parent, factor)
            if extra:
                h_up = np.concatenate([h_up, parent[:extra]])
            h0 = np.rint(h_up).astype(float)
            adjust_sum_exact(h0, rng)
        else:
            h0 = np.zeros(n, dtype=float)
            ones = rng.choice(n, size=n // 2, replace=False)
            h0[ones] = 1.0

        # ----- smooth‑max gradient descent (fractional) -----
        h_frac = h0.copy()
        betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 20000]
        it_per_beta = 8 if n <= 720 else 6
        if time.time() < deadline:
            h_frac = gradient_descent_smoothmax(
                h_frac,
                betas,
                base_step=0.25,
                it_per_beta=it_per_beta,
                rng=rng,
                deadline=deadline,
            )

        # ----- round to binary -----
        h_bin = np.rint(h_frac).astype(float)
        adjust_sum_exact(h_bin, rng)

        c_bin = compute_c5(h_bin)
        if c_bin < best_c_n:
            best_c_n, best_h_n = c_bin, h_bin.copy()

        # ----- binary targeted hill climbing -----
        remaining = deadline - time.time()
        if remaining > 5.0:
            sub_dead = time.time() + remaining * 0.6
            h_hc, c_hc = targeted_hill_climb_binary(h_bin, rng, sub_dead)
            if c_hc < best_c_n:
                best_c_n, best_h_n = c_hc, h_hc.copy()

    return best_h_n, best_c_n


# ---------- main entry ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # try n=720 (fast) several times, then n=1440 if time permits
    n_list = [(720, 5), (1440, 2)]

    for n, rest in n_list:
        if time.time() > deadline:
            break
        h_n, c_n = optimise_n(n, rest, rng, deadline)
        if h_n is not None and c_n < best_c:
            best_h, best_c, best_n = h_n, c_n, n

    # fallback simple half‑ones
    if best_h is None:
        n = 180
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
        best_n = n

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
