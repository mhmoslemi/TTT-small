import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function given as a float array."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise ----------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.int8)
        # binarise (values >0.5 become 1)
        h = (h > 0.5).astype(np.int8)
        n = h.size
    else:
        n = 1024
        h = np.zeros(n, dtype=np.int8)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1

    # enforce exact half‑ones
    target_ones = n // 2
    cur = int(h.sum())
    if cur != target_ones:
        if cur < target_ones:
            zeros = np.where(h == 0)[0]
            flip = rng.choice(zeros, target_ones - cur, replace=False)
            h[flip] = 1
        else:
            ones = np.where(h == 1)[0]
            flip = rng.choice(ones, cur - target_ones, replace=False)
            h[flip] = 0

    # initial correlation O_k for k=0..n-1 (linear, O[0]=0)
    O = np.correlate(h, 1 - h, mode="full")[n - 1 :].astype(np.int64)
    O[0] = 0
    cur_max = int(O.max())

    best_h = h.copy()
    best_max = cur_max

    # ----- simulated annealing on binary vectors -----
    T = 3.0                     # initial temperature (integer‑difference scale)
    cooling = 0.9995
    it = 0
    no_improve = 0
    O_new = np.empty_like(O)
    target_max = int(np.floor(0.3808 * n / 2))

    while time.time() - start < budget_s - 5:
        # pick a 1‑position i and a 0‑position j
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)

        # propose new correlation
        O_new[:] = O
        O_new[: n - i] += h[i:]
        O_new[: i + 1] += h[i :: -1]
        O_new[: n - j] -= h[j:]
        O_new[: j + 1] -= h[j :: -1]
        O_new[0] = 0
        new_max = int(O_new.max())
        delta = new_max - cur_max

        if delta <= 0 or (T > 0.0 and rng.random() < np.exp(-delta / T)):
            # accept swap
            h[i] = 0
            h[j] = 1
            O[:] = O_new
            cur_max = new_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                if best_max <= target_max:

        if it % 1000 == 0:
            T *= cooling
            if T < 1e-6:
                T = 0.0

        # occasional restart if stuck
        if no_improve >= 200_000:
            h[:] = 0
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1
            O[:] = np.correlate(h, 1 - h, mode="full")[n - 1 :].astype(np.int64)
            O[0] = 0
            cur_max = int(O.max())
            no_improve = 0
            T = 3.0

        it += 1

    # ----- final evaluation -----
    best_h_float = best_h.astype(float)
    final_c = compute_c5(best_h_float)

    return best_h_float.tolist(), float(final_c), int(n)
