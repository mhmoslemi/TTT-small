def project_onto_box_sum(v, s):
    # v: array
    lo = v.min() - 1  # ensure low enough
    hi = v.max() + 1  # high enough
    for _ in range(50):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0, 1)
        total = h.sum()
        if total > s:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)
```

Implementation details:

```
def project_onto_box_sum(v, s):
    n = v.shape[0]
    # binary search for lambda
    lo = -1.0
    hi = 2.0  # bounds leaning; we want lam such that sum clip(v-lam) = s
    for _ in range(50):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        total = h.sum()
        if total > s:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # due to numerical issues, adjust last bits
    diff = s - h.sum()
    if abs(diff) > 1e-9:
        # adjust by adding/subtracting diff to non-boundary entries
        # Find indices not at 0 or 1
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            # distribute diff equally
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            # adjust at any index with capacity
            for i in range(n):
                cap_up = 1.0 - h[i]
                cap_down = -h[i]
                if diff > 0 and cap_up > 0:
                    add = min(diff, cap_up)
                    h[i] += add
                    diff -= add
                elif diff < 0 and cap_down < 0:
                    sub = min(-diff, -cap_down)
                    h[i] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h
```

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
```

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

        final_refined = gradient_refine(best_h.copy(), rng,
                                        step=0.01, max_iter=5)
        final_c2 = compute_c5(final_refined)
        if final_c2 < best_c:
            best_h, best_c = final_refined, final_c2

    # ---- final repair & return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
