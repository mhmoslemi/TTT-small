```
def compute_c5_fft(h):
    n = h.size
    # compute corr of h and (1 - h) using FFT
    # length for convolution
    N = 2 * n
    hf = np.fft.rfft(h, n=N)
    gf = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(hf * gf, n=N)
    # take the first 2n-1 entries
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

def compute_c5_fft(h):
    n = h.shape[0]
    N = 2 * n
    hf = np.fft.rfft(h, n=N)
    gf = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(hf * np.conj(gf), n=N)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def compute_corr_k_c5(h):

import time
import math
import numpy as np

# ---------- correlation utilities ----------
def _corr_max_k(h: np.ndarray):
    """Return (max_corr, k) for correlation of h with 1-h."""
    n = h.shape[0]
    N = 2 * n
    hf = np.fft.rfft(h, n=N)
    gf = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(hf * np.conj(gf), n=N)[: 2 * n - 1]
    max_val = np.max(corr)
    idx = np.argmax(corr) - (n - 1)
    return max_val, abs(idx)


def compute_c5_fft(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2] using FFT."""
    max_val, _ = _corr_max_k(h)
    return max_val * (2.0 / h.shape[0])


# ---------- feasibility helper ----------
def enforce_sum_feasibility(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1] (tiny corrections)."""
    diff = target - h.sum()
    n = h.shape[0]
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum_feasibility(h, target_sum, rng)
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # ----- simulated annealing -----
    max_moves = 500_000
    T0 = 0.1
    T_end = 1e-6
    log_T_ratio = math.log(T_end / T0)
    moves = 0
    stale = 0  # moves since last improvement

        if time.time() - start > budget_s * 0.96:

        # correlation & worst shift for current h
