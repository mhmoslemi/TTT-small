def proj_bound_sum(v, target, lo=0.0, hi=1.0):
    # v: np.ndarray
    # returns projection w onto [lo,hi] and sum(w)=target
    # algorithm: start with v clipped
    w = np.clip(v, lo, hi)
    # compute sum
    cur = w.sum()
    if abs(cur - target) <= 1e-12:
        return w
    # Determine if we need to increase or decrease
    if cur < target:
        # Need to increase sum
        # free indices are those not at hi
        free = w < hi - 1e-12
        # while need increase
        while True:
            nfree = free.sum()
            if nfree == 0:
                break
            delta = (target - w.sum()) / nfree
            w[free] += delta
            # enforce bounds
            newly_fixed = (w > hi - 1e-12) if (target - w.sum()) > 0 else (w < lo + 1e-12)
            # Actually after increase, we only exceed hi
            excess = w > hi
            if np.any(excess):
                w[excess] = hi
                free = w < hi - 1e-12
            else:
                break
    else:
        # cur > target, need to decrease sum
        free = w > lo + 1e-12
        while True:
            nfree = free.sum()
            if nfree == 0:
                break
            delta = (target - w.sum()) / nfree
            w[free] += delta
            # enforce bounds
            if np.any(w < lo):
                w[w < lo] = lo
                free = w > lo + 1e-12
            else:
                break
    # final clip
    w = np.clip(w, lo, hi)
    return w
```

def proj_sum_bounds(v, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=50):
    # Ensure target feasible: lo*n <= target <= hi*n
    lo_n = lo * v.size
    hi_n = hi * v.size
    if target < lo_n: target = lo_n
    if target > hi_n: target = hi_n
    # bisection on tau
    tau_low = np.min(v) - hi
    tau_high = np.max(v) - lo
    for _ in range(max_iter):
        tau = (tau_low + tau_high) / 2.0
        w = np.clip(v - tau, lo, hi)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            tau_low = tau
        else:
            tau_high = tau
    # after max_iter return w
    w = np.clip(v - tau, lo, hi)
    # final adjustment: adjust slight difference by distributing error among non-bound entries
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute diff proportionally among free variables
        free = (w > lo + 1e-12) & (w < hi - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, lo, hi)
    return w
```

        # enforce exact sum
        x = proj_sum_bounds(x, target)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------
# Core utilities
# -------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def worst_shift_and_gradient(h: np.ndarray):
    """Return (shift_k, gradient_of_worst_shift, full correlation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones(n, dtype=float)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad, corr


def proj_sum_bounds(v: np.ndarray, target: float, lo: float = 0.0, hi: float = 1.0,
                    tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Euclidean projection of v onto {lo<=x<=hi, sum(x)=target}."""
    n = v.size
    lo_n, hi_n = lo * n, hi * n
    if target < lo_n:
        target = lo_n
    if target > hi_n:
        target = hi_n
    tau_low = np.min(v) - hi
    tau_high = np.max(v) - lo
    for _ in range(max_iter):
        tau = (tau_low + tau_high) * 0.5
        w = np.clip(v - tau, lo, hi)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            tau_low = tau
        else:
            tau_high = tau
    # final correction for any residual error
    w = np.clip(v - tau, lo, hi)
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        free = (w > lo + 1e-12) & (w < hi - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, lo, hi)
    return w


# -------------------------------------------------------
# Projected gradient descent on worst shift
# -------------------------------------------------------
def pgd_optimize(h0: np.ndarray, target: float, rng: np.random.Generator,
                 max_steps: int = 150) -> (np.ndarray, float):
    """Gradient descent on the worst‑shift overlap with line search."""
    h = h0.copy()
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_steps):
        _, grad, _ = worst_shift_and_gradient(h)
        d = -grad
        d -= d.mean()                     # enforce zero‑sum direction
        norm = np.linalg.norm(d)
        if norm < 1e-12:
            break
        d /= norm
        improved = False
        for alpha in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            h_new = proj_sum_bounds(h + alpha * d, target)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                h, best_c = h_new, c_new
                improved = True
                break
        if not improved:
            # tiny random transfer as escape move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                eps = rng.random() * max_eps * 0.01
                if eps > 0:
                    h_tmp = h.copy()
                    h_tmp[i] -= eps
                    h_tmp[j] += eps
                    h_tmp = proj_sum_bounds(h_tmp, target)
                    c_tmp = compute_c5(h_tmp)
                    if c_tmp < best_c:
                        h, best_c = h_tmp, c_tmp
    return h, best_c


# -------------------------------------------------------
# Soft‑max SLSQP refinement (optional)
# -------------------------------------------------------
def slsqp_refine(h0: np.ndarray, target: float, rng: np.random.Generator,
                 beta_seq=(20., 80., 200., 800., 3000., 8000., 20000.),
                 max_iter_per_beta: int = 400) -> np.ndarray:
    """Soft‑max SLSQP passes at increasing β."""
    n = h0.size
    x = np.clip(h0, 0.0, 1.0)

    def sum_eq(x):
        return np.sum(x) - target

    cons = {"type": "eq", "fun": sum_eq}
    bounds = [(0.0, 1.0)] * n

    for beta in beta_seq:
        def obj(v):
            corr = np.correlate(v, 1.0 - v, mode="full")
            m = np.max(beta * corr)
            # log‑sum‑exp stable
            return (m + np.log(np.sum(np.exp(beta * corr - m)))) / beta

        res = minimize(obj, x, method="SLSQP", bounds=bounds,
                       constraints=cons,
                       options={"maxiter": max_iter_per_beta,
                                "ftol": 1e-12, "disp": False})
        x = np.clip(res.x, 0.0, 1.0)
        # exact sum repair
        x = proj_sum_bounds(x, target)
    return x

    if remaining > 20.0:
        # allocate about half of the remaining time to SLSQP
        iter_per_beta = max(100, int(remaining * 0.5 / (len((20., 80., 200., 800., 3000., 8000., 20000.)) * 1.0)))
        best_h = slsqp_refine(best_h, target_sum, rng,

    # ------------------------------------------------------------------
    while time.time() - start < total_budget * 0.998:

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))

    best_h = proj_sum_bounds(np.clip(best_h, 0.0, 1.0), target_sum)
