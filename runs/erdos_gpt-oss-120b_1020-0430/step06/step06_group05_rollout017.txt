    b = (1.0 - h).astype(np.float64)[::-1]  # reverse

def repair(child):
    cur_sum = child.sum()
    target = n//2
    if cur_sum > target:
        # need to flip (cur_sum-target) ones to zero
        ones_idx = np.where(child == 1)[0]
        # randomly select indices to flip
        flip = rng.choice(ones_idx, cur_sum - target, replace=False)
        child[flip] = 0
    elif cur_sum < target:
        zeros_idx = np.where(child == 0)[0]
        flip = rng.choice(zeros_idx, target - cur_sum, replace=False)
        child[flip] = 1
    return child

def mutate(child):
    # swap a 1 and a 0
    ones = np.where(child == 1)[0]
    zeros = np.where(child == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return child
    i = rng.choice(ones)
    j = rng.choice(zeros)
    child[i], child[j] = 0, 1
    return child

import time
import numpy as np

# ---------- correlation and objective ----------
def compute_c5_fft(h: np.ndarray) -> float:
    """C5 for a step function given by array h (any dtype, values in [0,1]).
    Uses FFT for speed."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[: 2 * n - 1]
    max_val = conv.max()
    return max_val * (2.0 / n)


# ---------- gradient of the worst shift ----------
def worst_shift_and_grad(h: np.ndarray):
    """Return (signed shift k, gradient) for the shift attaining max overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = idx
    grad = np.ones_like(h)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        l = -k
        grad[l:] -= (1.0 - h[: n - l])
        grad[: n - l] -= h[l:]
    return k, grad


# ---------- helpers for sum constraint ----------
def repair_to_target(arr: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    """Flip bits to obtain exactly `target` ones (array must be binary)."""
    cur = int(arr.sum())
    if cur == target:
        return arr
    if cur > target:
        # need to turn (cur-target) ones into zeros
        ones = np.flatnonzero(arr)
        flip = rng.choice(ones, cur - target, replace=False)
        arr[flip] = 0
    else:
        zeros = np.flatnonzero(arr == 0)
        flip = rng.choice(zeros, target - cur, replace=False)
        arr[flip] = 1
    return arr


def enforce_sum_float(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust a float vector to hit the exact sum `target` while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.shape[0]
    # add mass
    if diff > 0:
        candidates = np.where(h < 1.0)[0]
        while diff > 1e-12 and candidates.size:
            i = rng.choice(candidates)
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if h[i] >= 1.0 - 1e-12:
                candidates = np.where(h < 1.0)[0]
    else:
        diff = -diff
        candidates = np.where(h > 0.0)[0]
        while diff > 1e-12 and candidates.size:
            i = rng.choice(candidates)
            sub = min(h[i], diff)
            h[i] -= sub
            diff -= sub
            if h[i] <= 1e-12:
                candidates = np.where(h > 0.0)[0]
    return h


# ---------- deterministic low‑discrepancy patterns ----------
def sturmian_pattern(alpha: float, n: int, target: int, rng: np.random.Generator) -> np.ndarray:
    """Balanced 0/1 sequence via a Beatty (Sturmian) construction, repaired to target."""
    x = (np.arange(n, dtype=np.float64) * alpha) % 1.0
    arr = (x < 0.5).astype(np.uint8)
    return repair_to_target(arr, target, rng)
