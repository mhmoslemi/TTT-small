def projected_grad_descent(h, beta, step, max_iter, rng, tol=1e-8):
    for it in range(max_iter):

        corr = np.correlate(h, 1 - h, mode="full") * dx   # vector L_k
        maxL = np.max(corr)
        # compute softmax weights
        exp_term = np.exp(beta * (corr - maxL))
        w = exp_term / exp_term.sum()
        # compute gradient
        w_full = w  # already length 2N-1
        conv1 = np.convolve(w_full, 1 - h, mode='full')

import time
import numpy as np

# ---------- basic helpers ----------

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (unnormalized)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


# ---------- smooth‑max gradient descent ----------

def smooth_gradient_descent(h, beta, step0, max_iters, rng, time_limit=None):
    """Projected gradient descent on soft‑max approximation of max_k overlap."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        L = np.correlate(h, 1 - h, mode="full") * dx
        maxL = np.max(L)
        exp_term = np.exp(beta * (L - maxL))
        w = exp_term / exp_term.sum()
        # gradient via convolutions
        conv1 = np.convolve(w, 1 - h, mode="full")
        conv2 = np.convolve(w[::-1], h, mode="full")
        grad = dx * (conv1[n - 1:n - 1 + n] - conv2[n - 1:n - 1 + n])

        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
        step *= 0.995
    return best_h, best_c
