                    if n - k > 0:
                        grad[k:] -= wk * h[: n - k]
                        grad[: n - k] -= wk * h[k:]

            h += -lr * grad
            np.clip(h, 0.0, 1.0, out=h)
            enforce_sum_fast(h, target)

            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
        if time.time() - start > budget_s * 0.95:
            break

    # final fine‑tuning with worst‑gradient hill‑climb
    while time.time() - start < budget_s * 0.998:
        g = worst_gradient(best_h)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if i == j:
            break
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            break
        eps = rng.random() * max_eps * 0.01
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_fast(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
