def compute_c5(h):
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project_box_sum(v, target):
    # binary search for lambda to satisfy sum constraints
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n/2.0)

def worst_gradient(h):
    n = h.size
    cross = np.correlate(h, h, mode='full')[n-1:]
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[:n-k] -= h[k:]
        grad[k:] -= h[:n-k]
    return grad, k
```

def subgradient_descent(h0, target, max_iters, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha0 = 2.0  # initial step scaling
    for t in range(1, max_iters+1):
        grad, _ = worst_gradient(h)
        step = alpha0 / np.sqrt(t + 10)
        h = project_box_sum(h - step * grad, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            # optional early break if extremely good
            if best_c <= 0.3805:
                break
    return best_h, best_c

def guided_hill_climb(h, rng, max_iters=5000, step_scale=0.7):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    decay_interval = max(1, max_iters // 5)
    step_scale_curr = step_scale
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        # find index with max grad (most beneficial to decrease) and min grad (most beneficial to increase)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale_curr
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it+1) % decay_interval == 0:
            step_scale_curr = max(step_scale_curr * 0.7, 1e-4)
    return best_h, best_c
```

def upsample(h, rng):
    n = h.size
    n2 = n*2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2/2.0)
```

import time
import numpy as np

# --------------------------------------------------------------
# Utility functions
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via bisection on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        if proj.sum() > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst shift overlap."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


# --------------------------------------------------------------
# Optimization primitives
def subgradient_descent(h0: np.ndarray, target: float,
                        max_iters: int, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha0 = 2.0
    for t in range(1, max_iters + 1):
        grad, _ = worst_gradient(h)
        step = alpha0 / np.sqrt(t + 10.0)
        h = project_box_sum(h - step * grad, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3805:
                break
    return best_h, best_c


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 5000, step_scale: float = 0.7):
    """Hill‑climb moving mass from high‑gradient to low‑gradient sites."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    decay = max(1, max_iters // 5)
    cur_scale = step_scale
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            eps_max = min(h[i], 1.0 - h[j])
            if eps_max > 0.0:
                eps = rng.random() * eps_max * cur_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            cur_scale = max(cur_scale * 0.7, 1e-4)
    return best_h, best_c


def upsample(h: np.ndarray, rng: np.random.Generator):
    """Linear up‑sampling by factor 2 and re‑projection."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)
