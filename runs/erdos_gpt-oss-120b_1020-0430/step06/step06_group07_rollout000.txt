    best_h = h0.copy()
    best_c = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_c = best_c
    start = time.time()
    # initial temperature as a fraction of best_c, maybe 0.02?
    init_temp = max(0.01, best_c * 0.05)
    # cooling schedule: linear to near zero at budget end.
    while True:
        elapsed = time.time() - start
        if elapsed >= budget:
            break
        # fraction of budget used
        frac = elapsed / budget
        temp = init_temp * (1 - frac)
        # if temp extremely low, just do greedy descent steps
        # pick source and dest
        src = rng.integers(0, n)

import time
import numpy as np

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst‑shift overlap (max_k ∫h(1‑h∘k))."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # autocorrelation for k≥0
    k = int(np.argmin(cross))  # shift giving largest overlap
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def subgradient_optimize(h0: np.ndarray, target: float,
                         budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap with symmetry."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10.0)
        h_new = project_box_sum(h - alpha * grad, target)
        # enforce symmetry (preserves sum and bounds)
        h_new = (h_new + h_new[::-1]) / 2.0
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
    return best_h, best_c


def simulated_annealing(h0: np.ndarray, target: float,
                        budget: float, rng: np.random.Generator):
    """SA with random mass transfers, symmetry, and periodic restarts."""
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_c = best_c
    start = time.time()
    last_improve = start
    init_temp = max(0.01, best_c * 0.15)

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget:
            break

        # restart if stuck too long
        if now - last_improve > 0.2 * budget:
            curr_h = random_feasible(n, rng)
            curr_h = (curr_h + curr_h[::-1]) / 2.0
            curr_h = project_box_sum(curr_h, target)
            curr_c = compute_c5(curr_h)
            last_improve = now

        # temperature (linear cooling)
        frac = elapsed / budget
        temp = init_temp * (1.0 - frac)

        # propose a transfer
        src = rng.integers(0, n)
        if curr_h[src] <= 1e-12:
            continue
        dst = rng.integers(0, n)
        if curr_h[dst] >= 1.0 - 1e-12:
            continue
        max_eps = min(curr_h[src], 1.0 - curr_h[dst])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * 0.5
        new_h = curr_h.copy()
        new_h[src] -= eps
        new_h[dst] += eps

        # enforce symmetry and feasibility
        new_h = (new_h + new_h[::-1]) / 2.0
        new_h = project_box_sum(new_h, target)

        new_c = compute_c5(new_h)

        # Metropolis acceptance
        if new_c < curr_c or rng.random() < np.exp(-(new_c - curr_c) / (temp + 1e-12)):
            curr_h, curr_c = new_h, new_c

                best_h, best_c = new_h.copy(), new_c
                last_improve = now
                if best_c <= 0.3808:
                    break
    return best_h, best_c


def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Linear up‑sampling by factor 2 with sum correction."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

            if c_up_sg < best_c:
                best_h, best_c = h_up_sg, c_up_sg

        # short SA on fine grid with whatever time left
        elapsed = time.time() - start
        remaining2 = total_budget - elapsed
        if remaining2 > 2.0:
            h_up_sa, c_up_sa = simulated_annealing(best_h, best_h.size / 2.0,
                                                   remaining2, rng)
            if c_up_sa < best_c:
                best_h, best_c = h_up_sa, c_up_sa

    # ---- final sanitisation -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
