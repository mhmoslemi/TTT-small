    rng = np.random.default_rng(seed)
    start = time.time()

    if 'initial_h_values' in globals():

        h = np.array(initial_h_values, dtype=float)

        # freeze n

    # ensure sum constraint: we can use repair
    h = enforce_sum(h, rng)

    L = 2 * n

    # compute C
    def compute_c5_fft_local(arr):
        a = np.fft.rfft(arr, L)
        b = np.fft.rfft(np.flip(1 - arr), L)
        corr = np.fft.irfft(a * b, L)[:2*n-1]
        return np.max(corr) * (2.0 / n)

    best_h = h.copy()
    best_C = C

    # parameters
    total_budget = budget_s
    anneal_time = total_budget * 0.75
    gradient_time = total_budget * 0.2
    # For remainder 0.05 seconds for final compute.

    T0 = 0.001
    T_min = 1e-7
    stagnation_counter = 0
    stagnation_limit = 5000
    p_random = 0.1

    # annealing loop
    while time.time() - start < anneal_time:
        elapsed = time.time() - start
        progress = elapsed / anneal_time
        T = max(T0 * (1 - progress), T_min)

        # decide swap type
        if rng.random() < p_random:
            # random swap
            i = rng.integers(0, n)
            while h[i] == 0:

            j = rng.integers(0, n)

                j = rng.integers(0, n)
        else:

            grad = worst_gradient(h)

            j = int(np.argmin(grad))

    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-15:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-15:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

# ------------------------------------------------------------
def compute_c5_full(h: np.ndarray) -> float:
    """Exact C5 using np.correlate (slow but precise)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

# ------------------------------------------------------------
def compute_obj_and_grad(x: np.ndarray, beta: float, lam: float,
                         L: int) -> tuple[float, np.ndarray]:
    """Smooth‐max objective + quadratic penalty, plus its gradient."""
    n = x.size
    sum_h = x.sum()
    # linear auto‑correlation Q_k (size 2n‑1)
    a = np.fft.rfft(x, L)
    Q = np.fft.irfft(a * np.conj(a), L)[: 2 * n - 1]
    O = sum_h - Q                     # O_k for all shifts
    maxO = O.max()
    # stable soft‑max
    expb = np.exp(beta * (O - maxO))
    s = expb.sum()
    smooth = (maxO + np.log(s) / beta)
    # penalty
    diff = sum_h - n / 2.0
    pen = lam * diff * diff
    obj = smooth + pen

    # weights
    w = expb / s                     # sum(w)=1
    # convolution w * x
    conv1 = np.convolve(w, x, mode="full")
    term1 = conv1[(n - 1) : (n - 1) + n]
    w_rev = w[::-1]
    conv2 = np.convolve(w_rev, x, mode="full")
    term2 = conv2[(n - 1) : (n - 1) + n]

    grad = 1.0 - (term1 + term2) + 2.0 * lam * diff
    return float(obj), grad

# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise ------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 512
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    n = h.size
    L = 2 * n                     # FFT length for linear convolutions

    best_h = h.copy()
    best_c = compute_c5_full(best_h)

    # ----- schedule of β (smoothness) --------------------------------
    beta_list = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]
    lam = 100.0                    # penalty weight

    for beta in beta_list:
        if time.time() - start > budget_s * 0.90:
            break
        # L‑BFGS‑B on smooth objective
        res = minimize(
            lambda x: compute_obj_and_grad(x, beta, lam, L),
            best_h,
            method="L-BFGS-B",
            jac=True,
            bounds=[(0.0, 1.0)] * n,
            options={"maxiter": 200, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, rng)
        c = compute_c5_full(x)
        if c < best_c:
            best_c, best_h = c, x.copy()
        # use the refined vector as start for the next β
        best_h = x

    # ----- final random fine‑tuning (cheap) ----------------------------
    while time.time() - start < budget_s * 0.98:
        # move a tiny mass ε from i (higher value) to j (lower)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        diff = best_h[i] - best_h[j]
        if diff <= 1e-6:
            continue
        eps = rng.random() * diff * 0.02
        best_h[i] -= eps
        best_h[j] += eps
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        c = compute_c5_full(best_h)
        if c < best_c:
            best_c = c

    # ----- final sanity ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_full(best_h)

    return best_h.tolist(), float(final_c), int(n)
