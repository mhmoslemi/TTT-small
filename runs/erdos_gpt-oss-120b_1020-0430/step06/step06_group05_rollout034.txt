def compute_corr(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fft_h = np.fft.rfft(h, n=L)
    fft_1h = np.fft.rfft(1.0 - h, n=L)
    prod = fft_h * fft_1h
    corr = np.fft.irfft(prod, n=L)
    return corr[: 2*n-1]

def compute_c5(h):
    corr = compute_corr(h)
    return np.max(corr) * (2.0 / h.size)

import time

# ----- global vars for soft‑max SLSQP ----- #
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple iterative correction
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    hf = np.fft.rfft(h.astype(np.float64), n=L)
    h1f = np.fft.rfft((1.0 - h).astype(np.float64), n=L)
    prod = hf * h1f
    corr = np.fft.irfft(prod, n=L)
    return corr[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = compute_corr_fft(h)
    idx = int(np.argmax(corr)) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h, dtype=np.float64)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def alpha_schedule(it: int, max_it: int):
    """Piece‑wise decreasing list of α values."""
    frac = it / max_it
    if frac < 0.3:
        return (1.0, 0.5, 0.25, 0.125, 0.0625)
    if frac < 0.6:
        return (0.5, 0.25, 0.125, 0.0625, 0.03125)
    if frac < 0.9:
        return (0.25, 0.125, 0.0625, 0.03125, 0.015625)
    return (0.125, 0.0625, 0.03125, 0.015625, 0.0078125)


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize  # local import
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x
