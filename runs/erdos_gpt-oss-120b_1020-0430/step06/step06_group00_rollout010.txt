def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start=time.time()
    total_budget=budget_s

    # compute function
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / h.size)

    def enforce_sum(h):
        # ensure sum == N/2
        target = h.size / 2.0
        # clip first
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        n = h.size
        attempts=0
        while abs(diff) > 1e-12 and attempts < 10000:
            i = rng.integers(n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
            attempts+=1
        return h

    # initial h: if initial_h_values defined use that; else random
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # clip
        np.clip(h0, 0.0, 1.0, out=h0)
        h0 = enforce_sum(h0)
    else:
        N=512
        h0 = rng.random(N)
        h0 = enforce_sum(h0)

    # upsample to 1024 for more freedom
    if h0.size < 1024:
        # linear interpolation
        old_n = h0.size
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, 1024, endpoint=False)
        h = np.interp(x_new, x_old, h0)
    else:
        h = h0.copy()
    # ensure feasible
    h = enforce_sum(h)

    n = h.size
    # initial c5
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr

    # SA parameters
    T0 = 0.08  # initial temperature
    T_end = 1e-7
    total_sa_time = total_budget - (time.time() - start) - 0.5  # keep small margin
    if total_sa_time <= 0:
        total_sa_time = 0
    t0 = time.time()
    while True:
        elapsed = time.time() - t0
        if elapsed > total_sa_time:
            break
        # temperature
        T = T0 * (T_end / T0) ** (elapsed / total_sa_time)
        # step scale
        step_scale = 0.3 * (T / T0)  # up to 30% of donor capacity early
        # choose donor i with h[i] > 1e-12, receiver j with h[j] < 1 - 1e-12

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        np.clip(h0, 0.0, 1.0, out=h0)
        h0 = enforce_sum(h0, rng)
    else:
        n0 = 512
        h0 = rng.random(n0)
        h0 = enforce_sum(h0, rng)

    # upsample to 1024 for more freedom
    if h0.size < 1024:
        h = upsample_h(h0, 1024)
    else:
        h = h0.copy()
    h = enforce_sum(h, rng)

    n = h.size
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr

    # ----- simulated annealing -------------------------------------------
    T0 = 0.08
    T_end = 1e-7
    # reserve a tiny margin for final tidy‑up
    sa_time = max(0.0, total_budget - (time.time() - start) - 0.5)
    t0 = time.time()
    while True:
        elapsed = time.time() - t0
        if elapsed > sa_time:
            break
        # temperature schedule (exponential)
        T = T0 * (T_end / T0) ** (elapsed / sa_time)
        # step scale proportional to temperature
        step_scale = 0.3 * (T / T0)

        # choose donor i (needs positive mass) and receiver j (needs room)
        i = rng.integers(n)
        if h[i] < 1e-12:
            for _ in range(5):

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
