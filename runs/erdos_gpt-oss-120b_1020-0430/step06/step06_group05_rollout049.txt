import time

# global parameters (set inside run)
GLOBAL_N = None
GLOBAL_L = None
TARGET_SUM = None

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # Use simple iterative transfers
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT (full accuracy)."""
    n = GLOBAL_N
    L = GLOBAL_L
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[: 2 * n - 1]
    return float(np.max(corr.real) * (2.0 / n))


def compute_c5_np(h: np.ndarray) -> float:
    """C5 using direct correlation (slower, used for gradient steps)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def make_individual(h: np.ndarray, rng: np.random.Generator):
    """Return dict with h and its C5."""
    c5 = compute_c5_fft(h)
    return {"h": h, "c5": c5}
