def project_onto_box_simplex(x, lower=0, upper=1, target_sum):
    # Clip lower and upper first:
    x = np.clip(x, lower, upper)
    # Compute current sum
    cur_sum = x.sum()
    if np.isclose(cur_sum, target_sum):
        return x
    # Need to adjust sum while staying within bounds
    # Determine set of free indices where 0 < x_i < 1.
    # Use iterative approach: while diff != 0:
    diff = target_sum - cur_sum
    # Keep track of free indices
    free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
    if free.size == 0:
        # all at bounds, cannot adjust -> just return x (maybe diff small)
        return x
    # Distribute diff equally or with clamping
    # We'll compute delta = diff / free.size; add to free; clip again; repeat.
    while abs(diff) > 1e-12 and free.size > 0:
        delta = diff / free.size
        x[free] += delta
        # Clip again and update diff
        x = np.clip(x, lower, upper)
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        # Update free indices (still not at bounds)
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
    return x

def worst_gradient(h, shift):
    n = h.size
    grad = np.ones_like(h)
    k = abs(shift)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        # subtract h[i+k]
        grad[: n - k] -= h[k:]
        # subtract h[i-k]
        grad[k:] -= h[: n - k]
    return grad

def pair_improve(h, shift, rng, max_iter=3):
    n = h.size
    k = abs(shift)
    if k == 0:
        # For shift 0, we can increase variance: push toward bounds
        # Choose i where h[i] not at bound, move towards nearest bound.
        # But we can skip; gradient will deal.
        return h
    # You can perform few attempts
    for _ in range(max_iter):
        # Determine index ranges overlapping region
        # For shift>0: overlapping i in [0, n-k), partner = i+k
        # Choose i_max with largest partner h[i+k] among those where h[i] < 1
        # Choose i_min with smallest partner among those where h[i] > 0
        region_i = np.arange(0, n - k)
        partner_vals = h[region_i + k]
        # Candidates for increase
        inc_mask = h[region_i] < 1.0 - 1e-12
        dec_mask = h[region_i] > 1e-12
        # If no candidates, break
        if not np.any(inc_mask) or not np.any(dec_mask):
            break
        # Among inc candidates, pick i with max partner
        inc_idxs = region_i[inc_mask]
        inc_partners = partner_vals[inc_mask]
        i_inc = inc_idxs[int(np.argmax(inc_partners))]
        # Among dec candidates, pick j with min partner
        dec_idxs = region_i[dec_mask]
        dec_partners = partner_vals[dec_mask]
        i_dec = dec_idxs[int(np.argmin(dec_partners))]
        # If partner at inc <= partner at dec, no improvement
        if h[i_inc + k] <= h[i_dec + k] + 1e-12:
            # Try reverse maybe at other side? maybe pick random
            break
        # Determine delta
        delta = min(1.0 - h[i_inc], h[i_dec])
        # Optionally limit delta to some fraction for stability
        delta = delta * 0.5  # 50% of allowed move
        if delta < 1e-12:
            break
        # Apply
        h[i_inc] += delta
        h[i_dec] -= delta
    return h

import time

from scipy.optimize import minimize

# ---------- core utilities -------------------------------------------------

def compute_c5_fft(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h on [0,2] (FFT version)."""
    n = h.size
    m = 2 * n - 1
    f = np.fft.rfft(h, m)
    g = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(f * np.conj(g), m)
    return np.max(corr) * (2.0 / n)


def worst_shift(h: np.ndarray):
    """Return (shift, value) where the overlap O_k is maximal."""
    n = h.size
    m = 2 * n - 1
    f = np.fft.rfft(h, m)
    g = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(f * np.conj(g), m)
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr[idx]


def worst_gradient(h: np.ndarray, shift: int):
    """Gradient of the worst overlap O_k w.r.t. h."""
    n = h.size
    grad = np.ones_like(h)
    k = abs(shift)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def project_onto_box_sum(x: np.ndarray, target_sum: float):
    """Projection onto 0≤x≤1 and Σx = target_sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-12:
        return x
    free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
    it = 0
    while abs(diff) > 1e-12 and free.size > 0 and it < 10:
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        it += 1
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0][0]
            x[idx] += min(diff, 1.0 - x[idx])
        else:
            idx = np.where(x > 1e-12)[0][0]
            x[idx] -= min(-diff, x[idx])
    return x


def projected_gradient_step(h, grad, step, target_sum):
    """Line‑search on three step sizes, then project."""
    best_h, best_c = None, np.inf
    for factor in (0.5, 1.0, 2.0):
        nh = h - step * factor * grad
        nh = project_onto_box_sum(nh, target_sum)
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h = c, nh
    return best_h, best_c


def pair_improve(h, shift, rng, max_iter=2):
    """Move mass from a low‑partner to a high‑partner for the worst shift."""
    n = h.size
    k = abs(shift)
    if k == 0:
        return h
    for _ in range(max_iter):
        region = np.arange(0, n - k)
        partner = h[region + k]
        inc_mask = h[region] < 1.0 - 1e-12
        dec_mask = h[region] > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            break
        inc_idxs = region[inc_mask]
        dec_idxs = region[dec_mask]
        inc_part = partner[inc_mask]
        dec_part = partner[dec_mask]
        i_inc = inc_idxs[int(np.argmax(inc_part))]
        i_dec = dec_idxs[int(np.argmin(dec_part))]
        if h[i_inc + k] <= h[i_dec + k] + 1e-12:
            break
        delta = min(1.0 - h[i_inc], h[i_dec]) * 0.5
        if delta < 1e-12:
            break
        h[i_inc] += delta
        h[i_dec] -= delta
    return h


def thue_morse_bits(n: int) -> np.ndarray:

    seq = np.zeros(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count("1") % 2
    return seq
