    def fun(x):

        maxc = np.max(corr)
        c = corr - maxc
        expc = np.exp(beta * c)
        sumexp = np.sum(expc)

        L = len(corr)

        conv2 = np.convolve(x, w, mode='full')
        start = L // 2
        grad = conv1[start:start + n] - conv2[start:start + n]
        return f, grad
    # SciPy expects fun to return f only unless jac=True
    # We'll set jac=True and let fun return (f,grad)

    res = minimize(fun, h0, method='SLSQP', jac=True, bounds=bounds,

    diff = target - x.sum()

import time
import numpy as np
from scipy.optimize import minimize

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    O = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(O), n=L)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)

def worst_gradient(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

                         time_budget: float, rng: np.random.Generator):
    start = time.time()
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c

def slsqp_optimize_softmax(h0: np.ndarray, beta: float,
                           rng: np.random.Generator,
                           max_iter: int, time_budget: float):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq",
            "fun": lambda x: np.sum(x) - target,
            "jac": lambda x: np.ones_like(x)}
    def fun(x):
        corr = np.correlate(x, 1 - x, mode="full")
        maxc = np.max(corr)
        c = corr - maxc
        expc = np.exp(beta * c)
        sumexp = np.sum(expc)
        f = (maxc + np.log(sumexp) / beta)
        w = expc / sumexp
        L = corr.size
        w_rev = w[::-1]
        conv1 = np.convolve(1 - x, w_rev, mode="full")
        conv2 = np.convolve(x, w, mode="full")
        start = L // 2
        grad = conv1[start:start + n] - conv2[start:start + n]
        return f, grad
    res = minimize(fun, h0, method="SLSQP", jac=True,
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, n / 2.0)
    else:
        n = 512
        h = random_feasible(n, rng)

    best_h, best_c = h, compute_c5(h)

    # Phase 1: sub‑gradient descent
    sub_budget = time_left() * 0.25
    if sub_budget > 5.0:
        h_sg, c_sg = subgradient_optimize(best_h, n / 2.0, sub_budget, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # Phase 2: guided hill‑climb
    hc_budget = time_left() * 0.15
    if hc_budget > 5.0:
        max_iters = int(min(8000, hc_budget * 300))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                        max_iters=max_iters,
                                        step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # Phase 3: soft‑max SLSQP passes
    betas = (10., 30., 100., 300., 1000., 3000., 10000.)
    while time_left() > 5.0:
        for beta in betas:
            if time_left() < 5.0:
                break
            max_iter = int(min(800, time_left() * 0.2))
            max_iter = max(max_iter, 50)
            h_opt = slsqp_optimize_softmax(best_h, beta, rng,
                                           max_iter=max_iter,
                                           time_budget=time_left())
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # Phase 4: up‑sample and re‑optimise on finer grid
    if n < 1024 and time_left() > 15.0:
        best_h = upsample(best_h, rng)
        n = best_h.size
        best_c = compute_c5(best_h)

        sub_budget = time_left() * 0.20
        if sub_budget > 5.0:
            h_sg, c_sg = subgradient_optimize(best_h, n / 2.0, sub_budget, rng)
            if c_sg < best_c:
                best_h, best_c = h_sg, c_sg

        hc_budget = time_left() * 0.15
        if hc_budget > 5.0:
            max_iters = int(min(8000, hc_budget * 250))
            h_hc, c_hc = guided_hill_climb(best_h, rng,
                                            max_iters=max_iters,
                                            step_scale=0.6)
            if c_hc < best_c:
                best_h, best_c = h_hc, c_hc

        while time_left() > 5.0:
            for beta in betas:
                if time_left() < 5.0:
                    break
                max_iter = int(min(800, time_left() * 0.2))
                max_iter = max(max_iter, 50)
                h_opt = slsqp_optimize_softmax(best_h, beta, rng,
                                               max_iter=max_iter,
                                               time_budget=time_left())
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt

    # Final random restarts (short sub‑gradient)
    while time_left() > 10.0:
        rem = time_left()
        budget = min(5.0, rem - 5.0)
        h_rand = random_feasible(n, rng)
        h_sg, c_sg = subgradient_optimize(h_rand, n / 2.0, budget, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # Sanitize and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
