def compute_c5_fft(h: np.ndarray) -> float:
    """Maximum overlap integral using FFT (full correlation)."""
    n = h.shape[0]
    n2 = 2 * n - 1
    f = np.fft.rfft(h, n=n2)
    g = np.fft.rfft(1.0 - h, n=n2)
    corr = np.fft.irfft(f * g, n=n2)
    return float(corr.max() * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> None:
    """In‑place adjustment so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

def gradient_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log∑exp(β·overlap) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)  # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i:] -= wi * h[: n - i]
        else:
            ii = -i
            g[ii:] += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii:]
    return g

    """Projected gradient descent on the smooth‑max surrogate."""
    for beta in betas:
        # adapt number of inner steps depending on β
        if beta < 5000:
            it_per_beta = 12
        elif beta < 50000:
            it_per_beta = 6
        else:
            it_per_beta = 4
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5_fft(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5_fft(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient using the current worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g


def worst_shift_refine(h: np.ndarray, rng: np.random.Generator,
                       base_step: float, iters: int,
                       deadline: float) -> np.ndarray:
    """Projected sub‑gradient steps focused on the current worst shift."""
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5_fft(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5_fft(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.95
    return h


# ----------------------------------------------------------------------
# Fractional Simulated Annealing
# ----------------------------------------------------------------------
def fractional_sa(h0: np.ndarray, rng: np.random.Generator,
                  deadline: float, T0: float = 1e-3,
                  decay: float = 0.9995, max_iters: int = 500000,
                  delta0: float = 0.2):
    """SA on the fractional space preserving the exact sum."""
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    delta = delta0
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # feasible eps respecting bounds and |eps|≤delta
        low = max(-h[i], h[j] - 1.0, -delta)
        high = min(1.0 - h[i], h[j], delta)
        if low >= high:
            it += 1
            continue
        eps = rng.random() * (high - low) + low
        h[i] += eps
        h[j] -= eps

        c_new = compute_c5_fft(h)
        T = T0 * (decay ** it)
        accept = False
        if c_new < cur_c:
            accept = True
        elif rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-15)):
            accept = True
        if accept:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert
            h[i] -= eps
            h[j] += eps

        it += 1
        # occasional adaptations
        if it % 10000 == 0:
            delta = max(delta * 0.98, 1e-5)
        if it % 50000 == 0:
            adjust_sum_exact(h, rng)
    return best_h, best_c


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # safety margin
    deadline = start + budget_s * 0.95

    best_h = None
    best_c = np.inf
    best_n = None

    base_n = 720
    parent = globals().get("initial_h_values", None)
    if parent is not None and len(parent) == base_n:
        parent_vec = np.array(parent, dtype=float)
    else:
        parent_vec = None

    # ------------------------------------------------------------------
    # Multiple restarts with gradient + SA
    # ------------------------------------------------------------------
    restarts = 3
    for _ in range(restarts):
        if time.time() > deadline:
            break

        n = base_n
        if parent_vec is not None:
            h = parent_vec.copy()
        else:
            h = rng.random(n)
            h /= h.sum()
            h *= n / 2.0
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # smooth‑max gradient descent
        betas = [5, 20, 80, 200, 500, 2000, 5000,
                 20000, 50000, 200000]
        h = gradient_descent_smoothmax(h, betas, base_step=0.25,
                                       rng=rng, deadline=deadline)

        # worst‑shift refinement
        h = worst_shift_refine(h, rng, base_step=0.12,
                               iters=30, deadline=deadline)

        # fractional SA (consume most of remaining time for this restart)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        h, c = fractional_sa(h, rng, sub_dead, T0=1e-3,
                             decay=0.9995, max_iters=300000, delta0=0.2)

        # update global best
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

    # ------------------------------------------------------------------
    # Optional up‑sampling if time permits
    # ------------------------------------------------------------------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)  # tiny correction for fp errors

        # quick SA on the up‑sampled version
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        h_up, c_up = fractional_sa(h_up, rng, sub_dead,
                                   T0=5e-4, decay=0.9997,
                                   max_iters=200000, delta0=0.15)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # ------------------------------------------------------------------
    # Final safety net (fallback)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5_fft(best_h)
        best_n = n

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
