import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # random redistribution until exact
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (subgradient of C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def binary_from_continuous(h: np.ndarray) -> np.ndarray:
    """Create a binary vector with exactly n/2 ones based on largest entries."""
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    b = np.zeros_like(h)
    b[idx] = 1.0
    return b

def project_feasible(y: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of y onto the box [0,1] with a given sum."""
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):

        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

    while time.time() - start < sub_time:
        it += 1
        # worst shift and its gradient
        grad = worst_gradient(h)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # projected step
        y = h - eta * grad
        h = project_feasible(y, target_sum)

        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

        # occasional binary snapshot
        if it % 2000 == 0:
            bh = binary_from_continuous(h)
            cb = compute_c5(bh)
            if cb < best_c:
                best_c = cb
                best_h = bh.copy()

        # small random jitter to escape plateaus
        if it % 5000 == 0:
            jitter = (rng.random(n) - 0.5) * 0.02
            h = np.clip(h + jitter, 0.0, 1.0)
            h = enforce_sum(h, rng)

    # ---------- final SLSQP polish ----------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        beta_seq = (20.0, 80.0, 200.0, 800.0, 3000.0)
        for beta in beta_seq:
            if time.time() - start > budget_s * 0.98:
                break
            h = slsqp_optimize(best_h, beta, rng, max_iter=800)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # ---------- try binary version of the final continuous point ----------
    bh = binary_from_continuous(best_h)
    cb = compute_c5(bh)
    if cb < best_c:
        best_c = cb
        best_h = bh.copy()

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
