import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """C5 for discrete step function using FFT."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.fft(h, N)
    C = np.fft.ifft(H * np.fft.fft(1.0 - h, N)).real
    return C.max() * (2.0 / n)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 512
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.where(x <= 1.0, x, 2.0 - x)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- time allocation -------------------------------------------
    SA_time = budget_s * 0.55
    SLSQP_time = budget_s * 0.35
    final_time = budget_s - SA_time - SLSQP_time
    slsqp_deadline = start + SA_time + SLSQP_time

    # ----- subgradient descent (SA style) ---------------------------
    step = 0.2
    decay = 0.9998
    while time.time() - start < SA_time:
        N = 2 * n
        H = np.fft.fft(h, N)
        C = np.fft.ifft(H * np.fft.fft(1.0 - h, N)).real
        idx = int(C.argmax())
        k = abs(idx - (n - 1))

        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

        cur_c = C.max() * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        step *= decay
        if step < 1e-7:
            step = 1e-7

    # ----- SLSQP refinement -----------------------------------------
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target

    def softmax_obj(x):
        N = 2 * n
        H = np.fft.fft(x, N)
        C = np.fft.ifft(H * np.fft.fft(1.0 - x, N)).real
        bc = CURRENT_BETA * C
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    beta_seq = (20.0, 200.0, 2000.0, 20000.0)
    cur_h = best_h.copy()

    for beta in beta_seq:
        if time.time() > slsqp_deadline:
            break
        CURRENT_BETA = beta
        res = minimize(
            softmax_obj,
            cur_h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 150, "ftol": 1e-12, "disp": False},
        )
        cur_h = np.clip(res.x, 0.0, 1.0)
        cur_h = enforce_sum(cur_h, rng)
        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()

    # ----- final gradient polishing ----------------------------------
    while time.time() - start < start + budget_s - 5:
        N = 2 * n
        H = np.fft.fft(cur_h, N)
        C = np.fft.ifft(H * np.fft.fft(1.0 - cur_h, N)).real
        idx = int(C.argmax())
        k = abs(idx - (n - 1))

        if k == 0:
            grad = 1.0 - 2.0 * cur_h
        else:
            grad = np.ones_like(cur_h)
            grad[: n - k] -= cur_h[k:]
            grad[k:] -= cur_h[: n - k]

        # decaying step size
        elapsed = time.time() - start
        step2 = 0.02 * (0.5 ** (elapsed / budget_s * 5))
        cur_h = cur_h - step2 * grad
        cur_h = np.clip(cur_h, 0.0, 1.0)
        cur_h = enforce_sum(cur_h, rng)

        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        if step2 < 1e-8:
            break

    # ----- final repair & return ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
