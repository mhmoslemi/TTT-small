def subgradient_descent(h, target, max_iters, step_alpha0=0.5, tol=1e-12):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for t in range(1, max_iters+1):
        # compute worst shift list
        corr = np.correlate(h, 1 - h, mode='full')
        # compute scaled version to find max
        O = corr * (2.0 / n)
        max_O = np.max(O)
        # find worst shifts indices
        # tolerance: relative maybe 1e-8
        tol_abs = 1e-9
        worst_idxs = np.where(O >= max_O - tol_abs)[0]
        # compute gradient averaging
        grad = np.zeros_like(h)
        for idx in worst_idxs:
            k = idx - (n - 1)
            # generic gradient
            # we can compute as per earlier
            # For shift k, for i where i+k in range: subtract h[i+k]
            # For i where i-k in range: subtract h[i-k]
            # Add 1 always.
            # Vectorized:
            # create grad_k = np.ones(n)
            # subtract shifted copies
            if k >= 0:
                grad[: n - k] -= h[k:]
                grad[k:] -= h[: n - k]
            else:
                km = -k
                grad[km:] -= h[: n - km]
                grad[: n - km] -= h[km:]
        # average gradient
        grad /= len(worst_idxs)
        # step size schedule
        alpha = step_alpha0 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.7):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    scale = step_scale

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    return proj

import time
import numpy as np
from numpy.random import default_rng

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the box with exact sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst‐shift overlap O_k and the shift value."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = int(np.argmax(corr) - (n - 1))
    grad = np.ones(n)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        km = -k
        grad[km:] -= h[: n - km]
        grad[: n - km] -= h[km:]
    return grad, k


def subgradient_until_time(h: np.ndarray, target: float, max_time: float,
                           step_alpha0: float = 0.5):
    """Projected sub‑gradient descent on the worst shift, time‑limited."""
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    t = 0
    while time.time() - start < max_time:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = step_alpha0 / np.sqrt(t + 10.0)
        h = project_box_sum(h - alpha * grad, target)
        if t % 200 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
                if best_c < 0.3808:
                    break
    return best_h, best_c


def hillclimb_until_time(h: np.ndarray, rng: np.random.Generator,
                         max_time: float, max_iters: int = 20000,
                         step_scale: float = 0.7):
    """Guided hill‑climb using the worst‑gradient direction, time‑limited."""
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    scale = step_scale
    decay = max(1, max_iters // 5)
    it = 0
    while time.time() - start < max_time and it < max_iters:
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * scale
                h[i] -= eps
                h[j] += eps
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        it += 1
        if it % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c
