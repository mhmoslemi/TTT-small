def l_bfgs_optimize(h0, beta, target, pen, rng, max_iter=200):
    n = h0.size
    bounds = [(0.0,1.0)]*n
    def fun(x):
        corr = np.correlate(x, 1-x, mode='full')
        maxbeta = np.max(beta * corr)
        w = np.exp(beta * corr - maxbeta)
        sum_w = np.sum(w)
        obj = (maxbeta + np.log(sum_w))/beta + pen * (np.sum(x) - target)**2
        return obj
    def jac(x):
        corr = np.correlate(x, 1-x, mode='full')
        maxbeta = np.max(beta * corr)
        w = np.exp(beta * corr - maxbeta)
        sum_w = np.sum(w)
        w_norm = w / sum_w
        f = 1 - x
        g1 = np.correlate(w_norm, f, mode='full')[0:n]
        g2 = np.convolve(w_norm, x, mode='full')[0:n]
        grad = g1 - g2 + 2 * pen * (np.sum(x) - target)
        return grad
    res = minimize(fun, h0, method='L-BFGS-B', jac=jac, bounds=bounds, options={'maxfun': max_iter, 'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # final projection to exact sum using project_to_feasible
    x = project_to_feasible(x, target)
    return x

def project_to_feasible(x, target, tol=1e-12, max_iter=100):
    # Clip initial
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) <= tol:
            break
        # free indices where not at bounds
        free = (x > 0.0 + tol) & (x < 1.0 - tol)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            # cannot adjust, break
            break
        delta = diff / cnt
        # add delta to free entries, but ensure not exceed bounds
        x[free] += delta
        # after addition may have overshoot; clip again
        x = np.clip(x, 0.0, 1.0)
    # Final rounding to ensure target via random fix if needed
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # adjust using random indices
        for _ in range(1000):
            if abs(diff) <= 1e-12:
                break
            i = rng.integers(0, x.size)
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """C5 for step function on [0,2] with N points (scaled)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_to_feasible(x: np.ndarray, target: float, rng: np.random.Generator,
                        tol: float = 1e-12, max_iter: int = 100) -> np.ndarray:
    """Clamp to [0,1] and adjust sum to exactly target (water‑filling)."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) <= tol:
            break
        free = (x > tol) & (x < 1.0 - tol)
        cnt = int(free.sum())
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        for _ in range(1000):
            if abs(diff) <= tol:
                break
            i = rng.integers(0, x.size)
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def greedy_improve(h: np.ndarray, deadline: float, target: float,
                   rng: np.random.Generator) -> np.ndarray:
    """Move mass from indices with large A' to those with small A' (subgradient step)."""
    n = h.size
    while time.time() < deadline:
        f = 1.0 - h
        corr = np.correlate(h, f, mode="full")
        idx_max = int(np.argmax(corr))
        k0 = idx_max - (n - 1)

        # A' = f[i+k0] - h[i-k0]
        A = np.zeros(n)
        if k0 >= 0:
            A[: n - k0] += f[k0:]
            A[k0:] -= h[: n - k0]
        else:
            kp = -k0
            A[kp:] += f[: n - kp]
            A[: n - kp] -= h[kp:]

        mask_dec = h > 1e-12
        mask_inc = h < 1.0 - 1e-12
        if not mask_dec.any() or not mask_inc.any():
            break

        dec_idx = np.where(mask_dec)[0]
        inc_idx = np.where(mask_inc)[0]
        L = 20
        if dec_idx.size > L:
            top_dec = dec_idx[np.argpartition(-A[dec_idx], L - 1)[:L]]
        else:
            top_dec = dec_idx
        if inc_idx.size > L:
            bot_inc = inc_idx[np.argpartition(A[inc_idx], L - 1)[:L]]
        else:
            bot_inc = inc_idx

        best_gain = 0.0
        best_pair = None
        for i in top_dec:
            for j in bot_inc:
                gain = A[i] - A[j]
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        if best_pair is None or best_gain <= 1e-12:
            break

        i, j = best_pair
        eps = min(h[i], 1.0 - h[j])
        if eps <= 0.0:
            break
        h[i] -= eps
        h[j] += eps
    return h

def l_bfgs_optimize(h0: np.ndarray, beta: float, target: float,
                    pen: float, rng: np.random.Generator,
                    max_iter: int = 150) -> np.ndarray:
    """L‑BFGS‑B with soft‑max objective and quadratic sum penalty."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(beta * corr)
        w = np.exp(beta * corr - m)
        s = np.sum(w)
        obj = (m + np.log(s)) / beta + pen * (np.sum(x) - target) ** 2
        return obj

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(beta * corr)
        w = np.exp(beta * corr - m)
        s = np.sum(w)
        w_norm = w / s
        f = 1.0 - x
        g1 = np.correlate(w_norm, f, mode="full")[:n]
        g2 = np.convolve(w_norm, x, mode="full")[:n]
        grad = g1 - g2 + 2.0 * pen * (np.sum(x) - target)
        return grad

    res = minimize(fun, h0, method="L-BFGS-B", jac=jac,
                   bounds=bounds,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_to_feasible(x, target, rng)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -----
