def gradient_swap_sa(h, rng, deadline, beta=20000, base_T=None, decay=0.9995, grad_interval=200):
    n = len(h)
    zeros = list(np.flatnonzero(h == 0))
    ones = list(np.flatnonzero(h == 1))
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    if base_T is None:
        base_T = max(1e-6, cur_c * 0.02)  # arbitrary
    T = base_T
    it = 0
    while time.time() < deadline and T > 1e-12:
        if it % grad_interval == 0:
            grad = gradient_smoothmax(cur_h, beta)
            # choose i from zeros with max grad, j from ones with min grad
            # If zeros empty or ones empty, break
            # compute argmax over zeros:
            if len(zeros) == 0 or len(ones) == 0:
                break
            # vectorized selection
            grad_zeros = grad[zeros]
            i_idx = np.argmax(grad_zeros)
            i = zeros[i_idx]

            grad_ones = grad[ones]
            j_idx = np.argmin(grad_ones)
            j = ones[j_idx]
        else:
            # maybe random perturbation: occasionally random swap
            if rng.random() < 0.05:
                i = rng.choice(zeros)
                j = rng.choice(ones)
            else:
                # use previous i,j (kept from last gradient recompute)
                pass

        # perform swap
        cur_h[i] = 1.0
        cur_h[j] = 0.0
        new_c = compute_c5(cur_h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # accept with probability exp(-(new_c - cur_c) / T)
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            # update zero/one lists
            zeros[i_idx] = j  # we replace zero at i with j (j originally one)
            ones[j_idx] = i
            # if new_c < best_c:
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 0.0
            cur_h[j] = 1.0
        it += 1
        T *= decay
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities -------------------
def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """In‑place repair so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ------------------- smooth‑max gradient -------------------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)·log ∑exp(β·overlap) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    """Projected gradient descent on a schedule of β."""
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):
    """Sub‑gradient steps focused on the current worst shift(s)."""
    n = h.size
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        v = np.correlate(h, 1.0 - h, mode="full")
        max_val = v.max()
        idxs = np.where(v == max_val)[0]
        g = np.zeros_like(h, dtype=np.float64)
        for idx in idxs:
            k = idx - (n - 1)
            if k >= 0:
                g[: n - k] += 1.0 - h[k:]
                g[k :]    -= h[: n - k]
            else:
                kk = -k
                g[kk :]   += 1.0 - h[: n - kk]
                g[: n - kk] -= h[kk :]
        g /= max(1, len(idxs))
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.95
    return h

# ------------------- SLSQP polish -------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP on the smooth‑max surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ------------------- binary hill‑climb -------------------
def binary_sa_optimize(h_start, rng, deadline):
    """Simulated‑annealing on binary vectors with sum fixed."""
    n = h_start.size
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    init_T = max(1e-8, cur_c * 0.02)
    T = init_T
    decay = 0.99985
    while time.time() < deadline and T > 1e-12:
        zeros = np.flatnonzero(cur_h == 0)
        ones = np.flatnonzero(cur_h == 1)
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        cur_h[i] = 1.0
        cur_h[j] = 0.0
        new_c = compute_c5(cur_h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 0.0
            cur_h[j] = 1.0
        T *= decay
    return best_h, best_c

    best_h = h.copy()
    best_c = compute_c5(h)
    while time.time() < deadline:
        O = np.correlate(h, 1.0 - h, mode="full")
        O_max = O.max()
        c = O_max * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # absolute worst shifts (ignore sign, drop zero)
        widx = np.where(O == O_max)[0] - (n - 1)
        abs_shifts = {abs(int(k)) for k in widx if k != 0}
        if not abs_shifts:
            break
        # benefit vector: for each position, number of 1's at a worst‑shift distance
        B = np.zeros(n, dtype=int)
        for k in abs_shifts:
            nk = n - k
            B[:nk] += h[k:]
            B[k:] += h[:nk]
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if zeros.size == 0 or ones.size == 0:
            break
        K = min(10, zeros.size, ones.size)
        zero_vals = B[zeros]
        one_vals = B[ones]
        zero_top_idx = np.argpartition(-zero_vals, K - 1)[:K]
        one_bot_idx = np.argpartition(one_vals, K - 1)[:K]
        zeros_top = zeros[zero_top_idx]
        ones_bot = ones[one_bot_idx]
        best_net = -1
        best_pair = None
        for a in zeros_top:
            Ba = B[a]
            for b in ones_bot:
                if a == b:
                    continue
                net = Ba - B[b] - (1 if abs(a - b) in abs_shifts else 0)
                if net > best_net:
                    best_net = net
                    best_pair = (a, b)
        if best_net > 0:
            a, b = best_pair
            h[a] = 1.0
            h[b] = 0.0
            continue
        break
    return best_h, best_c

# ------------------- main entry point -------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        n = len(parent)
        h0 = np.rint(parent).astype(float)
        adjust_sum_exact(h0, rng)
    else:
        n = 720
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    best_h, best_c, best_n = h0.copy(), compute_c5(h0), n

    # --- greedy swap phase (≈15% of time) ---
    greedy_dead = min(deadline, time.time() + (deadline - time.time()) * 0.15)
    h_g, c_g = greedy_swap_optimize(best_h.copy(), rng, greedy_dead)
    if c_g < best_c:
        best_h, best_c = h_g, c_g

    # --- binary simulated annealing (remaining time) ---
    h_sa, c_sa = binary_sa_optimize(best_h.copy(), rng, deadline)
    if c_sa < best_c:
        best_h, best_c = h_sa, c_sa

    # --- optional fractional polishing if time permits ---
    if time.time() < deadline:
        h_frac = best_h.astype(float)
        betas = [5, 20, 80, 200, 500, 1000, 2000, 5000]
        h_frac = gradient_descent_smoothmax(
            h_frac, betas, base_step=0.2, it_per_beta=4, rng=rng, deadline=deadline
        )
        h_frac = worst_shift_refine(h_frac, rng, base_step=0.1, iters=10, deadline=deadline)
        for beta in (2000, 5000, 20000):
            if time.time() > deadline:
                break
            h_frac = slsqp_optimize(h_frac, beta, rng, max_iter=500)
        c_frac = compute_c5(h_frac)
        if c_frac < best_c:
            best_h, best_c = h_frac, c_frac

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
