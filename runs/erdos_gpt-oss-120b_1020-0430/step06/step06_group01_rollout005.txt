def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    # if diff small, return
    if abs(diff) < 1e-12:
        return h
    # adjust iteratively
    # to avoid infinite loop due to floating errors, limit iterations
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        add = max(min(diff, 1.0 - h[i]), -h[i])
        if abs(add) < 1e-14:
            continue
        h[i] += add
        diff = target - h.sum()
    # At end, may correct by scaling slight difference:
    if abs(diff) > 1e-9:
        # distribute evenly small adjustments
        h += diff / n
        h = np.clip(h, 0, 1)
        # final correction
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # fallback: adjust first element
            h[0] = np.clip(h[0] + diff, 0, 1)
    return h
```

import time
import numpy as np

# ---------- helper functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] using np.correlate."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        add = max(min(diff, 1.0 - h[i]), -h[i])
        if abs(add) < 1e-14:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap at the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros_like(h)
    if k > 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kp = -k
        grad[: n - kp] -= h[kp:]
        grad[kp:] += 1.0 - h[: n - kp]
    return grad


def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Simulated annealing on binary vectors (0/1) with sum=n/2."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = np.array(ones, dtype=int)
        zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = ones_arr[rng.integers(len(ones_arr))]
            j = zeros_arr[rng.integers(len(zeros_arr))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[ones_arr == i] = j
                zeros_arr[zeros_arr == j] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


def local_search(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Gradient‑directed hill climbing within a time budget."""
    start = time.time()
    end = start + time_budget
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    step_scale = 0.5
    no_improve = 0
    while time.time() < end:
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
        eps = rng.random() * max_eps * step_scale
        if eps <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            step_scale = min(step_scale * 1.05, 1.0)
        else:
            if rng.random() < 0.001:
                h = h_new
            no_improve += 1
            if no_improve % 500 == 0:
                step_scale = max(step_scale * 0.7, 0.01)
            if no_improve > 2000:
                for _ in range(5):
                    idx = rng.integers(n)
                    delta = (rng.random() - 0.5) * step_scale * max_eps
                    h[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
                h = enforce_sum(h, rng)
                no_improve = 0
    best_h = enforce_sum(best_h, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0, compute_c5(h0), h0.size
    else:
        n0 = 512
        h0 = rng.random(n0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0, compute_c5(h0), n0

    # -------- binary simulated annealing (≈20% budget) --------
    bin_time = total_budget * 0.20
    h_bin, c_bin = binary_sa(1024, rng, bin_time)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin, c_bin, 1024

    # -------- continuous optimisation on several sizes --------
    size_schedule = [(512, 0.12), (1024, 0.28), (2048, 0.48)]
    for size, fraction in size_schedule:
        allocated = total_budget * fraction
        elapsed = time.time() - start
        if elapsed >= total_budget * 0.99:
            break
        remaining = total_budget - elapsed
        allocated = min(allocated, remaining)
        # up‑sample / down‑sample to target size
        if best_n != size:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, size, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = best_h.copy()
        # decide number of restarts
        min_sub = 10.0
        max_restarts = max(1, int(allocated // min_sub))
        sub_budget = allocated / max_restarts
        for _ in range(max_restarts):
            # random perturbation of current h
            h0 = h.copy()
            perturb_cnt = rng.integers(5, 15)
            for __ in range(perturb_cnt):
                idx = rng.integers(size)
                delta = (rng.random() - 0.5) * 0.3
                h0[idx] = np.clip(h0[idx] + delta, 0.0, 1.0)
            h0 = enforce_sum(h0, rng)
            time_left = total_budget - (time.time() - start)
            sub = min(sub_budget, time_left * 0.95)
            if sub <= 0:
                break
            h_opt, c_opt = local_search(h0, rng, sub)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, size
                h = h_opt.copy()
            if time.time() - start >= total_budget * 0.99:
                break

    # -------- final stochastic tweaking --------
    while time.time() - start < total_budget * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            i = rng.integers(best_n)
            j = rng.integers(best_n)
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps <= 0.0:
                break
        eps = rng.random() * max_eps * 0.01
        if eps <= 0.0:
            continue
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < best_c:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
