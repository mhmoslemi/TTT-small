import time
import numpy as np

# ---------------------------------------------------------
# basic utilities
def adjust_sum_exact(h, rng, max_iter=3000):
    """Modify h in-place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------------------------------------------------------
# objective (FFT based for speed)
def compute_c5(h):
    """max overlap integral for discrete h using FFT."""
    n = h.size
    h_pad = np.concatenate([h, np.zeros(n, dtype=np.float64)])
    g_pad = np.concatenate([1.0 - h, np.zeros(n, dtype=np.float64)])
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_pad)
    v = np.fft.irfft(H * np.conj(G), n=2 * n)
    v = v[: 2 * n - 1]
    return float(v.max() * (2.0 / n))

# smooth‑max gradient (loop version – reliable indexing)
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)·log∑exp(β·overlap) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    # shift index 0 corresponds to k = -(n-1)
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = shift_idx - (n - 1)
        if k >= 0:
            # contribution of (1 - h_{i+k})
            g[: n - k] += wi * one_minus_h[k:]
            # contribution of -h_{i}
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

def worst_shift_gradient(h):
    """Sub‑gradient using the currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

    """Projected sub‑gradient steps focused on the current worst shift."""
    step = base_step

    """Simulated‑annealing hill‑climb on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
        if it > 12000:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    # ----- initialise ------------------------------------------------
    parent = globals().get("initial_h_values", None)
    base_n = 720
    if parent is not None:
        if len(parent) == base_n:
            h0 = np.array(parent, dtype=np.float64)
        else:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            h0 = np.interp(xt, xs, parent).astype(np.float64)
    else:
        h0 = rng.random(base_n).astype(np.float64)
        h0 = h0 / h0.sum() * (base_n / 2.0)
        np.clip(h0, 0.0, 1.0, out=h0)
    adjust_sum_exact(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_n = base_n

    # ----- main gradient descent schedule --------------------------------
    # piecewise schedule (beta, iterations)
    schedule = [
        (5, 4000),
        (20, 3000),
        (100, 3000),
        (500, 2500),
        (2000, 2000),
        (5000, 1500),
        (20000, 1000),
        (50000, 500),
    ]
    h = best_h.copy()
    c_h = best_c
    for beta, iters in schedule:
        if time.time() > deadline:
            break
        step_base = 0.25 / np.sqrt(beta)
        for it in range(iters):
            if time.time() > deadline:
                break
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            # simple decaying step multiplier
            step = step_base * (1.0 / (1.0 + it * 0.01))
            cand = h - step * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c_cand = compute_c5(cand)
            if c_cand < c_h:
                h = cand
                c_h = c_cand
                if c_cand < best_c:
                    best_c = c_cand
                    best_h = cand.copy()
            else:
                # occasional uphill move
                if rng.random() < 0.001:
                    h = cand
                    c_h = c_cand

    # ----- refine with worst‑shift sub‑gradient ---------------------------
    h = worst_shift_refine(h, rng, base_step=0.12, iters=60, deadline=deadline)
    c_h = compute_c5(h)
    if c_h < best_c:
        best_c = c_h
        best_h = h.copy()
        best_n = base_n

    # ----- binary rounding + simulated annealing -------------------------
    h_bin = np.rint(h).astype(np.float64)
    adjust_sum_exact(h_bin, rng)
    sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
    h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
        best_n = base_n

    # ----- optional up‑sampling if time permits -------------------------
    if time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h).astype(np.float64)
        adjust_sum_exact(h_up, rng)

        # limited gradient descent on upsampled version
        up_schedule = [
            (5, 2000),
            (20, 1500),
            (100, 1500),
            (500, 1000),
            (2000, 800),
            (5000, 600),
            (20000, 400),
        ]
        h = h_up.copy()
        c_h = compute_c5(h)
        for beta, iters in up_schedule:
            if time.time() > deadline:
                break
            step_base = 0.2 / np.sqrt(beta)
            for it in range(iters):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h, beta)
                if np.linalg.norm(g) < 1e-14:
                    continue
                step = step_base * (1.0 / (1.0 + it * 0.01))
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c_cand = compute_c5(cand)
                if c_cand < c_h:
                    h = cand
                    c_h = c_cand
                    if c_cand < best_c:
                        best_c = c_cand
                        best_h = cand.copy()
                        best_n = n_up

        # final refinement
        h = worst_shift_refine(h, rng, base_step=0.08, iters=40, deadline=deadline)
        c_h = compute_c5(h)
        if c_h < best_c:
            best_c = c_h
            best_h = h.copy()
            best_n = n_up

        # binary polish on up‑sampled solution
        h_bin = np.rint(h).astype(np.float64)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.20)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
            best_n = n_up

    # ----- final safety check -------------------------------------------
    if best_h is None:
        n = 180
        best_h = np.zeros(n, dtype=np.float64)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
