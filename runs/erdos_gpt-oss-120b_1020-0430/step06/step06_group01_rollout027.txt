    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    while time.time() - start < time_budget:
        # compute corr, C5, k*
        corr = np.correlate(h, 1.0 - h, mode='full')
        C5 = np.max(corr) * (2.0 / h.size)
        k_idx = np.argmax(corr) - (h.size - 1)
        k = abs(k_idx)
        # compute gradient for this k
        n = h.size

            grad[k:] -= h[: n - k]

        j = np.argmin(grad)

import time
import numpy as np
from scipy.optimize import minimize

# ----- global variables for softmax SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----- basic utilities -----
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 0:
            h[i] += add
            diff = target - h.sum()
        attempts += 1
    if abs(diff) > 1e-12:
        # final tiny correction on a feasible index
        feasible = np.where(((diff > 0) & (h < 1.0 - 1e-12)) |
                            ((diff < 0) & (h > 1e-12)))[0]
        if feasible.size:
            i = feasible[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


# ----- aggressive hill climb -----
def aggressive_hill_climb(h: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Move mass from high‑gradient to low‑gradient indices, always preserving sum."""
    n = h.size
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    stagnant = 0  # steps without improvement

    while time.time() - start < time_budget:
        # current correlation & worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_c = corr[max_idx] * (2.0 / n)
        k = abs(max_idx - (n - 1))

        # gradient for this shift
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break

        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 1e-12:
            break

        # try full epsilon
        new_h = h.copy()
        new_h[i] -= eps_max
        new_h[j] += eps_max
        new_corr = np.correlate(new_h, 1.0 - new_h, mode="full")
        new_c = np.max(new_corr) * (2.0 / n)

        if new_c < cur_c - 1e-12:
            h = new_h
            cur_c = new_c
            stagnant = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            continue

        # try decreasing factors
        improved = False
        for factor in (0.5, 0.25, 0.125, 0.0625):
            eps = eps_max * factor
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_corr = np.correlate(new_h, 1.0 - new_h, mode="full")
            new_c = np.max(new_corr) * (2.0 / n)
            if new_c < cur_c - 1e-12:
                h = new_h
                cur_c = new_c
                stagnant = 0
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                break

        if not improved:
            stagnant += 1
            if stagnant >= 20:
                # random small perturbation to escape
                i_r = rng.integers(0, n)
                j_r = rng.integers(0, n)
                if i_r != j_r:
                    eps = min(h[i_r], 1.0 - h[j_r]) * rng.random() * 0.2
                    if eps > 1e-12:
                        new_h = h.copy()
                        new_h[i_r] -= eps
                        new_h[j_r] += eps
                        new_corr = np.corrcoef(new_h, 1.0 - new_h)[0, 1]  # dummy, will recompute properly
                        new_c = compute_c5(new_h)
                        if new_c < cur_c - 1e-12:
                            h = new_h
                            cur_c = new_c
                            stagnant = 0
                            if cur_c < best_c:
                                best_c = cur_c
                                best_h = h.copy()
                stagnant = 0

        # occasional renormalisation to fix tiny drift
        if stagnant % 100 == 0:
            diff = n / 2.0 - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, rng)

    return best_h, best_c


# ----- softmax SLSQP refinement -----
def softmax_slsqp_optimize(h0: np.ndarray, betas, rng: np.random.Generator, total_time: float,
                           max_iter_per_beta: int = 800):
    """Run SLSQP on the smooth max with increasing beta."""
    start = time.time()
    n = h0.size
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    for beta in betas:
        if time.time() - start > total_time * 0.99:
            break
        CURRENT_BETA = beta
        max_iter = max_iter_per_beta
        # shorten if time is running low
        if time.time() - start > total_time * 0.6:
            max_iter = max(200, max_iter // 3)

        res = minimize(
            softmax_obj,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        # exact sum correction
        diff = CURRENT_TARGET - x.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += add
                diff = CURRENT_TARGET - x.sum()
                if abs(diff) < 1e-12:
                    break
        c = compute_c5(x)
        if c < best_c:
            best_h, best_c = x, c

    return best_h, best_c


# ----- single trial optimisation (hill climb + SLSQP) -----
def optimize_once(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    hill_time = time_budget * 0.65
    h_hill, c_hill = aggressive_hill_climb(h0, rng, hill_time)
    elapsed = time.time() - start
    remaining = max(0.0, time_budget - elapsed)
    if remaining > 0.0:
        betas = (20., 80., 200., 800., 3000., 8000., 20000.)
        h_slsqp, c_slsqp = softmax_slsqp_optimize(h_hill, betas, rng, remaining, max_iter_per_beta=500)
    else:
        h_slsqp, c_slsqp = h_hill, c_hill

    # tiny gradient‑directed tidy‑up
    h = h_slsqp
    c = c_slsqp
    for _ in range(3):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        eps = min(h[i], 1.0 - h[j]) * 0.01
        if eps <= 0:
            continue
        h_tmp = h.copy()
        h_tmp[i] -= eps
        h_tmp[j] += eps
        c_tmp = compute_c5(h_tmp)
        if c_tmp < c:
            h, c = h_tmp, c_tmp
    return h, c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
