import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discrete h (full linear correlation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Adjust h in‑place so that sum(h) == n/2 exactly (random redistribution)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def binary_hill_climb(h_bin: np.ndarray, rng: np.random.Generator, deadline: float):
    """Simple swap hill‑climb on a binary vector."""
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    while time.time() < deadline:
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    # ----- initialise h -----
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        h = np.array(parent, dtype=np.float64, copy=True)
    else:
        n = 720
        h = rng.random(n)
        h = h / h.sum() * (n / 2.0)
    np.clip(h, 0.0, 1.0, out=h)
    adjust_sum_exact(h, rng)

    n = h.size
    target_sum = n / 2.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # ----- simulated annealing on continuous h -----
    T0, T_min = 0.02, 1e-6
    step0, step_min = 0.5, 0.001
    last_improve = time.time()
    restart_interval = 120.0
    iter_cnt = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * (1.0 - frac) + T_min * frac
        step = step0 * (1.0 - frac) + step_min * frac

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move mass from i to j
        max_allowed = min(h[i], 1.0 - h[j])
        if max_allowed <= 1e-12:
            continue
        eps = step * max_allowed * rng.random()
        if eps <= 0.0:
            continue

        h[i] -= eps
        h[j] += eps

        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                last_improve = time.time()
        else:
            h[i] += eps
            h[j] -= eps

        # occasional sum correction
        if iter_cnt % 2000 == 0:
            s = h.sum()
            if abs(s - target_sum) > 1e-10:
                diff = target_sum - s
                idx = rng.integers(0, n)
                adj = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, -h[idx])
                h[idx] += adj

        # restart if stagnant
        if time.time() - last_improve > restart_interval and time.time() + restart_interval < deadline:
            h = best_h.copy()
            # small random jitter to escape
            jitter = (rng.random(n) - 0.5) * step * 0.1
            h += jitter
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)
            cur_c = compute_c5(h)
            last_improve = time.time()

        iter_cnt += 1

    # ----- binary refinement (if time permits) -----
    if time.time() < deadline:
        h_bin = np.rint(best_h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
        h_bin, c_bin = binary_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # ----- final cleanup -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
