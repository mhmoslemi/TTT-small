def conv_full_fft(a, b):
    n = a.size + b.size - 1
    nfft = 1 << (n-1).bit_length()
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A * B, nfft)[:n]
    return conv

# ---------- basic utilities ----------
def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def conv_full_fft(a, b):
    n = a.size + b.size - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A * B, nfft)[:n]
    return conv

def compute_c5(h):
    n = h.size
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    corr = np.fft.irfft(H * G, nfft)[: 2 * n - 1]
    return float(corr.max() * (2.0 / n))

def gradient_smoothmax_fft(h, beta):
    n = h.size
    # overlap values v
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    v = np.fft.irfft(H * G, nfft)[: 2 * n - 1]
    # soft‑max weights
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    # gradient via convolutions
    w_rev = w[::-1]
    conv1 = conv_full_fft(w_rev, 1.0 - h)
    conv2 = conv_full_fft(w, h)
    off = n - 1
    g = conv1[off : off + n] - conv2[off : off + n]
    return g
