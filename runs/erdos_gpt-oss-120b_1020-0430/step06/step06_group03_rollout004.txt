def softmax_weighted_grad(h, beta):
    n = h.shape[0]
    a = 1 - h
    m = int(2 ** np.ceil(np.log2(2*n - 1)))
    # compute O: correlation h with a
    a_rev = a[::-1]
    fft_h = np.fft.rfft(h, m)
    fft_a_rev = np.fft.rfft(a_rev, m)
    O_full = np.fft.irfft(fft_h * fft_a_rev, m)[:2*n-1]  # corr values
    # O_max for numerical stability
    O_max = O_full.max()
    exp_term = np.exp(beta * (O_full - O_max))
    w = exp_term / exp_term.sum()
    # compute weighted gradient via FFT
    # w_rev
    w_rev = w[::-1]
    # compute conv_rev = a * w_rev (linear conv)
    fft_a = np.fft.rfft(a, m)
    fft_w_rev = np.fft.rfft(w_rev, m)
    conv_rev = np.fft.irfft(fft_a * fft_w_rev, m)[:2*n-1]
    # compute conv2 = h * w
    fft_h = np.fft.rfft(h, m)
    fft_w = np.fft.rfft(w, m)
    conv2 = np.fft.irfft(fft_h * fft_w, m)[:2*n-1]
    grad = conv_rev[n-1:n-1+n] - conv2[n-1:n-1+n]
    return grad, O_full
```

def softmax_weighted_grad(h, beta, m_fft):
    n = h.shape[0]
    a = 1.0 - h
    # compute O = correlate(h, a) via FFT
    # a_rev = a[::-1]
    fft_h = np.fft.rfft(h, m_fft)
    fft_a_rev = np.fft.rfft(a[::-1], m_fft)
    O_full = np.fft.irfft(fft_h * fft_a_rev, m_fft)[:2*n-1]
    O_max = O_full.max()
    # compute weights
    exp_term = np.exp(beta * (O_full - O_max))
    w = exp_term / exp_term.sum()
    # weight arrays
    w_rev = w[::-1]
    # compute conv_rev = a * w_rev
    fft_a = np.fft.rfft(a, m_fft)
    fft_w_rev = np.fft.rfft(w_rev, m_fft)
    conv_rev = np.fft.irfft(fft_a * fft_w_rev, m_fft)[:2*n-1]
    # compute conv2 = h * w
    fft_h = np.fft.rfft(h, m_fft)
    fft_w = np.fft.rfft(w, m_fft)
    conv2 = np.fft.irfft(fft_h * fft_w, m_fft)[:2*n-1]
    grad = conv_rev[n-1:n-1+n] - conv2[n-1:n-1+n]
    return grad, O_full
