def project_box_sum_fast(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # iterate over indices in random order to distribute diff
    indices = rng.permutation(v.size)
    for i in indices:
        if diff == 0:
            break
        if diff > 0:
            max_add = 1.0 - v[i]
            if max_add > 0:
                add = min(diff, max_add)
                v[i] += add
                diff -= add
        else:
            max_sub = v[i]  # amount we can subtract
            if max_sub > 0:
                sub = min(-diff, max_sub)
                v[i] -= sub
                diff += sub
    # after a pass diff should be ~0; if not, loop again (but unlikely)
    if abs(diff) > 1e-12:
        # fallback to binary search
        return project_box_sum(v, target)
    return v

            diff -= add
    else:

            if sub <= 0:
                continue
            v[i] -= sub
            diff += sub
    if abs(diff) > 1e-9:
        # fallback
        v = project_box_sum(v, target)
    return v

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Binary‑search projection onto {0≤x≤1, sum x = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def project_box_sum_fast(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Fast O(n) projection; falls back to binary‑search if needed."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        idxs = np.where(v < 1.0)[0]
        rng.shuffle(idxs)
        for i in idxs:
            if diff <= 0:
                break
            add = min(diff, 1.0 - v[i])
            if add <= 0:
                continue
            v[i] += add
            diff -= add
    else:
        idxs = np.where(v > 0.0)[0]
        rng.shuffle(idxs)
        for i in idxs:
            if diff >= 0:
                break
            sub = min(-diff, v[i])
            if sub <= 0:
                continue
            v[i] -= sub
            diff += sub
    if abs(diff) > 1e-9:
        v = project_box_sum(v, target)
    return v


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample_to_n(h0: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    n0 = h0.size
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x0, h0)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_sum_fast(h_new, n / 2.0, rng)


def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ for a step function using linear correlation (no wrap)."""
    n = h.size
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_rev = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(fft_h * fft_rev, n=L)
    O_full = conv[: 2 * n - 1] * dx
    return float(O_full.max())


def compute_O_full_and_shift(h: np.ndarray, dx: float):
    """Return (max_O, |shift|, full O array, argmax index)."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_rev = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(fft_h * fft_rev, n=L)
    O_full = conv[: 2 * n - 1] * dx
    idx = int(np.argmax(O_full))
    shift = idx - (n - 1)
    k_abs = abs(shift)
    return float(O_full[idx]), k_abs, O_full, idx


def gradient_from_abs_shift(h: np.ndarray, k: int):
    """Gradient ∂O_k/∂h for the linear (zero‑padded) overlap."""
    n = h.shape[0]
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros_like(h)
    grad[: n - k] += 1.0 - h[k:]
    grad[k:] -= h[: n - k]
    return grad


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # grid size (even)
    n = 1024
    target = n / 2.0
    dx = 2.0 / n

    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h = upsample_to_n(h0, n, rng)
    else:
        h = random_feasible(n, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Adam state
    m = np.zeros_like(h)
    v_vec = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5

    no_improve = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main projected Adam optimisation
    while True:
        if time.time() - start >= total_budget - 2.0:
            break

        O_max, k_abs, _, _ = compute_O_full_and_shift(h, dx)

        if O_max < best_c - 1e-12:
            best_c = O_max
            best_h = h.copy()
            no_improve = 0
            if best_c <= 0.3808:
                break
        else:
            no_improve += 1

        grad = gradient_from_abs_shift(h, k_abs)

        m = beta1 * m + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)

        step = -base_lr * m / (np.sqrt(v_vec) + eps)
        h = h + step
        h = project_box_sum_fast(h, target, rng)

        iteration += 1
        if iteration % 5000 == 0:
            base_lr *= 0.9
        if no_improve >= 10000:
            h += rng.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum_fast(h, target, rng)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final lightweight local search (small transfers)
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        attempts = int(min(20000, remaining * 3000))
        for _ in range(attempts):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = (rng.random() - 0.5) * 0.05  # [-0.025,0.025]
            if delta > 0:
                if best_h[i] < delta or best_h[j] > 1.0 - delta:
                    continue
            else:
                if best_h[i] > -delta or best_h[j] < -delta:
                    continue
            nh = best_h.copy()
            nh[i] -= delta
            nh[j] += delta
            c = compute_c5(nh)
            if c < best_c:
                best_c, best_h = c, nh
                if best_c <= 0.3808:
                    break

    # ------------------------------------------------------------------
    # sanitise and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum_fast(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
