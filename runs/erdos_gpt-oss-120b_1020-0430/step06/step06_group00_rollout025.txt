import time

# --------------------------------------------------------------
# Core utilities
def compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] using FFT (correlation)."""
    n = h.size
    # correlation = convolution of reversed h with (1-h)
    conv = np.fft.irfft(
        np.fft.rfft(h[::-1], n=2 * n) * np.fft.rfft(1.0 - h, n=2 * n), n=2 * n
    )
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != 1024:
            h0 = upsample_h(h0, 1024)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5_fft(h0)
        best_h, best_c, best_n = h0, c0, 1024

    # ---- Phase 0: binary SA on fine grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        t_bin = min(remaining * 0.35, 400.0)
        h_bin, c_bin = binary_sa_fft(1024, rng, t_bin, target_c=0.3808)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, 1024
        if best_c <= 0.3808:
            best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
            return best_h.tolist(), float(compute_c5_fft(best_h)), int(best_h.size)

    # ---- Phase 1: continuous SA for fine‑tuning ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        t_cont = min(remaining * 0.45, 400.0)
        h_cont, c_cont = continuous_sa_fft(best_h, rng, t_cont)
        if c_cont < best_c:
            best_h, best_c, best_n = h_cont, c_cont, best_h.size

    # ---- Phase 2: gradient refinement ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        h_ref = gradient_refine(best_h, rng, step=0.015, max_iter=30)
        c_ref = compute_c5_fft(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ---- Final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
