import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """max overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Modify h in‑place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- smooth‑max utilities ----------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)log∑exp(β·overlap) wrt h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:]    -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:]   += wi * one_minus_h[: n - s]
            g[: n - s] -= wi * h[s:]
    return g

def smoothmax_obj(h, beta):
    v = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * v
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

# ---------- projected gradient descent on smooth‑max ----------
def gradient_descent_smoothmax_extended(h, betas, iters_per_beta, rng, deadline):
    """Run back‑tracked projected gradient descent on the smooth‑max."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_h
            g = gradient_smoothmax(h, beta)
            cur_val = smoothmax_obj(h, beta)
            step = 0.5
            # back‑tracking line search
            for _ in range(20):
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                new_val = smoothmax_obj(cand, beta)
                if new_val < cur_val:
                    break
                step *= 0.5
                if step < 1e-8:
                    break
            else:
                continue
            h = cand
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h

def binary_sa_fast(h_bin, rng, deadline, max_iters=1_000_000):
    """SA on binary vectors using O(n) update of autocorrelation."""
    n = h_bin.size
    L = 2 * n - 1
    h_pad = np.concatenate([np.zeros(n - 1, dtype=np.int8),
                            h_bin.astype(np.int8),
                            np.zeros(n - 1, dtype=np.int8)])
    s = np.correlate(h_bin, h_bin, mode="full")
    min_s = s.min()
    c = (n / 2 - min_s) * (2.0 / n)
    best_h = h_bin.copy()
    best_c = c
    T = c * 0.1 + 1e-12
    decay = 0.99995

    while i < max_iters and time.time() < deadline:
        i += 1
        # pick a random 1 and a random 0
        i_idx = rng.integers(0, n)

        while h_bin[j_idx] == 1:
            j_idx = rng.integers(0, n)

        i_off = i_idx + n - 1
        j_off = j_idx + n - 1

        np.subtract(h_pad[j_off:j_off + L], h_pad[i_off:i_off + L], out=delta)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    base_n = 720
    parent = globals().get("initial_h_values", None)

    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        if len(parent) != base_n:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            h = np.interp(xt, xs, parent)
        else:
            h = parent.copy()
    else:
        h = rng.random(base_n)
        h = h / h.sum() * (base_n / 2.0)
    adjust_sum_exact(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---------- phase 1: smooth‑max gradient descent ----------
    betas = [5, 20, 80, 200, 500, 2000, 5000, 20000, 50000, 100000]
    h = gradient_descent_smoothmax_extended(h, betas, iters_per_beta=30,
                                            rng=rng, deadline=deadline)
    c = compute_c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # ---------- phase 2: random perturbation restarts ----------
    for _ in range(3):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(scale=0.02, size=base_n)
        np.clip(h_pert, 0.0, 1.0, out=h_pert)
        adjust_sum_exact(h_pert, rng)
        h_pert = gradient_descent_smoothmax_extended(h_pert, betas,
                                                     iters_per_beta=15,
                                                     rng=rng, deadline=deadline)
        c = compute_c5(h_pert)
        if c < best_c:
            best_c, best_h = c, h_pert.copy()

    # ---------- phase 3: binary simulated annealing ----------
    h_bin = np.rint(best_h).astype(np.int8)
    adjust_sum_exact(h_bin, rng)
    time_left = deadline - time.time()
    if time_left > 5.0:
        # rough estimate of iterations per second (~1800 for n=720)
        max_iters = int(min(2_000_000,
                            max(200_000, time_left * 1800)))
        h_bin_best, c_bin = binary_sa_fast(h_bin, rng, deadline, max_iters)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin_best.astype(float)

    # ---------- phase 4: refine binary solution fractionally ----------
    if time.time() < deadline:
        h_frac = best_h.astype(float)
        adjust_sum_exact(h_frac, rng)
        h_frac = gradient_descent_smoothmax_extended(h_frac, betas,
                                                     iters_per_beta=20,
                                                     rng=rng, deadline=deadline)
        c = compute_c5(h_frac)
        if c < best_c:
            best_c, best_h = c, h_frac.copy()

    # ---------- phase 5: optional up‑sampling ----------
    if time.time() < deadline:
        n_up = best_h.size * 2
        if n_up <= 1440:
            h_up = np.repeat(best_h, 2)
            adjust_sum_exact(h_up, rng)
            h_up = gradient_descent_smoothmax_extended(h_up, betas[:8],
                                                       iters_per_beta=15,
                                                       rng=rng, deadline=deadline)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c, best_h = c_up, h_up.copy()

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
