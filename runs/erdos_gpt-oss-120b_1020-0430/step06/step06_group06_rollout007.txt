def compute_c5(h):
    N = h.size
    # compute correlation of h with 1-h:
    # correlation length 2N-1
    # using FFT: correlate = ifft(fft(h) * conj(fft(1-h))) -> returns circular correlation. We need linear correlation; use zero-padding size 2N.
    size = 2*N
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    # linear correlation at shift k from -(N-1) to N-1 is corr[N-1:2N-1]
    linear = corr[N-1:2*N-1]
    return np.max(linear) * (2.0/N)

CURRENT_BETA = None
CURRENT_TARGET = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    H = np.fft.rfft(x, 2*x.size)
    G = np.fft.rfft(1.0 - x, 2*x.size)
    prod = H*np.conj(G)
    corr = np.fft.irfft(prod, 2*x.size)
    linear = corr[x.size-1:2*x.size-1]
    # compute softmax of beta * corr
    vals = CURRENT_BETA * linear
    m = np.max(vals)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(vals - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    constraints = {"type":"eq","fun":sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=constraints,
                   options={"maxiter":max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x,0.0,1.0)
    # final projection to enforce sum exactly (in case)
    x = enforce_sum(x, n/2.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        if w.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# ---------- C5 ----------
def compute_c5(h):
    n = h.size
    size = 2 * n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    linear = corr[n - 1 : 2 * n - 1]
    return np.max(linear) * (2.0 / n)

def compute_c5_and_grad(h):
    n = h.size
    size = 2 * n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    prod = H * np.conj(G)
    corr = np.fft.irfft(prod, size)
    linear = corr[n - 1 : 2 * n - 1]
    idx = np.argmax(linear)
    max_val = linear[idx]
    k = abs(idx - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return max_val * (2.0 / n), grad

CURRENT_BETA = None
CURRENT_TARGET = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    n = x.size
    size = 2 * n
    H = np.fft.rfft(x, size)
    G = np.fft.rfft(1.0 - x, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    linear = corr[n - 1 : 2 * n - 1] * (2.0 / n)
    vals = CURRENT_BETA * linear
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(vals - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_BETA, CURRENT_TARGET
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_sum(x, n / 2.0)

def gradient_refine(h, rng, step=0.02, max_iter=12):

    for _ in range(max_iter):
        _, grad = compute_c5_and_grad(h)
        h = h - step * grad
        h = project_sum(h, n / 2.0)
        step *= 0.9

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum(h0, h0.size / 2.0)
        n = h0.size
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        n = 512
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = h0.copy()
        best_c = compute_c5(best_h)

    # main restart loop (≈75% of budget)
    main_deadline = start + budget_s * 0.75
    beta_seq = (5.0, 30.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0)

    while time.time() < main_deadline:
        # random start
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        jitter = rng.uniform(-0.03, 0.03, size=n)
        h = np.clip(h + jitter, 0.0, 1.0)
        h = project_sum(h, n / 2.0)

        # annealing (limited by sub‑deadline)
        sa_deadline = min(main_deadline, time.time() + budget_s * 0.15)
        h, c = simulated_annealing(h, rng, sa_deadline)
        if time.time() >= main_deadline:
            break

        # gradient polishing
        h = gradient_refine(h, rng, step=0.02, max_iter=12)
        if time.time() >= main_deadline:
            break

        # sequence of SLSQP passes
        for beta in beta_seq:
            if time.time() >= main_deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=300)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # final refinement (remaining time)
    final_deadline = start + budget_s * 0.95
    best_h = gradient_refine(best_h, rng, step=0.005, max_iter=30)
    while time.time() < final_deadline:
        for beta in (20000.0, 50000.0):
            if time.time() >= final_deadline:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=600)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
