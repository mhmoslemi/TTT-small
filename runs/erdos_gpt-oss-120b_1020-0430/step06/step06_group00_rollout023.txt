def greedy_local_search(h, rng, time_budget):
    start = time.time()
    n = h.size
    dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode='full')
    r = (1.0 - h)[::-1]  # shape (n,)
    cur_c = O.max() * dx
    best_h = h.copy()
    best_c = cur_c
    iter_cnt = 0
    # schedule for delta: start~0.02, end~1e-4
    start_delta = 0.02
    end_delta = 1e-4
    while time.time() - start < time_budget:
        iter_cnt += 1
        # compute delta using exponential decay based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_budget
        delta = start_delta * (end_delta / start_delta) ** frac
        # pick i with h[i] > 0
        i_idx = rng.integers(0, n)
        # ensure proper; if h[i_idx] <= 0, skip maybe
        if h[i_idx] <= 0.0:
            continue
        j_idx = rng.integers(0, n)
        if h[j_idx] >= 1.0:
            continue
        # max allowable delta
        delta = min(delta, h[i_idx], 1.0 - h[j_idx])
        if delta <= 0.0:
            continue
        # compute new O quickly
        O_new = O.copy()
        O_new[i_idx:i_idx+n] -= delta * r
        O_new[j_idx:j_idx+n] += delta * r
        new_c = O_new.max() * dx
        if new_c < cur_c - 1e-12:  # improvement
            # accept
            h[i_idx] -= delta
            h[j_idx] += delta
            O = O_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        # else ignore
    return best_h, best_c
```

def greedy_local_search(h,rng,time_budget):
    start = time.time()
    n = h.size
    dx = 2.0 / n
    O = np.correlate(h,1.0 - h,mode='full')
    cur_c = O.max() * dx
    best_h = h.copy()
    best_c = cur_c
    start_delta = 0.02
    end_delta = 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        delta = start_delta * (end_delta/start_delta) ** frac
        i = rng.integers(0,n)
        if h[i] <= 0.0:
            continue
        j = rng.integers(0,n)
        if h[j] >= 1.0:
            continue
        delta = min(delta, h[i], 1.0 - h[j])
        if delta <= 0.0:
            continue
        r = (1.0 - h)[::-1]
        O_new = O.copy()
        O_new[i:i+n] -= delta * r
        O_new[j:j+n] += delta * r
        new_c = O_new.max() * dx
        if new_c < cur_c - 1e-12:
            h[i] -= delta
            h[j] += delta
            O = np.correlate(h,1.0 - h,mode='full')
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        it += 1
    return np.clip(h, 0.0, 1.0)

    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    global CURRENT_TARGET, CURRENT_BETA

    CURRENT_BETA = beta

    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)

    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
