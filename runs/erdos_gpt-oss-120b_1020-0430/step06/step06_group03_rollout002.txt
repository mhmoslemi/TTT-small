import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def project_onto_box_sum(v, target):
    # Project onto [0,1]^n intersect sum(v)=target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient_and_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    idx = corr.argmax() - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, max_val * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialise h
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        h = project_onto_box_sum(h, target)
    else:
        n = 512
        target = n / 2.0
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    # ensure feasibility
    h = project_onto_box_sum(h, target)
    grad, c_curr = worst_gradient_and_c5(h)
    best_h = h.copy()
    best_c = c_curr

    # ---- pre‑phase: deterministic projected gradient descent ----
    pre_iters = min(2000, int(budget_s * 10))
    pre_step0 = 0.5
    for it in range(pre_iters):
        if time.time() - start > budget_s * 0.55:
            break
        grad, c_curr = worst_gradient_and_c5(h)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        step = pre_step0 * (1.0 - it / pre_iters) / norm
        h_candidate = h - step * grad
        h_candidate = project_onto_box_sum(h_candidate, target)
        _, c_new = worst_gradient_and_c5(h_candidate)
        if c_new < c_curr:
            h = h_candidate
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    # ---- main annealing loop ----
    max_step = 0.5
    min_step = 0.0005
    T0 = 0.002
    while time.time() - start < budget_s * 0.98:
        grad, c_curr = worst_gradient_and_c5(h)
        elapsed = time.time() - start
        rem = max(0.0, 1.0 - elapsed / budget_s)
        step_scale = min_step + (max_step - min_step) * rem
        T = T0 * rem
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            pos_idx = np.where(h > 1e-12)[0]
            neg_idx = np.where(h < 1.0 - 1e-12)[0]
            if pos_idx.size == 0 or neg_idx.size == 0:
                break
            i = rng.choice(pos_idx)
            j = rng.choice(neg_idx)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
        eps = rng.random() * max_eps * step_scale
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        _, c_new = worst_gradient_and_c5(nh)
        delta = c_new - c_curr
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = nh
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    # ---- final fine‑tuning loop ----
    while time.time() - start < budget_s * 0.999:
        grad, c_curr = worst_gradient_and_c5(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            break
        eps = max_eps * 0.005
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        _, c_new = worst_gradient_and_c5(nh)
        if c_new < c_curr:
            h = nh
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
