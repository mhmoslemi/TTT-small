def project_onto_capped_simplex(v, s, max_iter=50):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        cur_sum = x.sum()
        if cur_sum > s:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

import time
import numpy as np

def _project_onto_capped_simplex(v, s, iters=50):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        cur = x.sum()
        if cur > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny fix for numerical drift
    diff = s - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_linear(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def _c5_fft_arr(h):
    N = h.size
    b = 1.0 - h
    H = np.fft.rfft(h, n=2 * N)
    B = np.fft.rfft(b[::-1], n=2 * N)
    corr = np.fft.irfft(H * B, n=2 * N)[: 2 * N - 1]
    return corr * (2.0 / N)

def _grad_shift(h, k):
    N = h.size
    k = abs(k)
    if k == 0:
        return 2.0 * h
    g = np.zeros_like(h)
    g[: N - k] += h[k:]
    g[k:] += h[: N - k]
    return g

def _random_move(h, rng, scale=0.05):
    N = h.size
    i, j = rng.integers(0, N, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    N = 2048
    target_sum = N / 2.0
    best_h = None
    best_c = float('inf')
    # main loop with restarts
    while time.time() - start < budget_s * 0.96:
        # initialise h
        if rng.random() < 0.5:
            # binary random
            h = np.zeros(N)
            ones = rng.choice(N, N // 2, replace=False)
            h[ones] = 1.0
        else:
            # uniform with small noise
            h = np.full(N, 0.5) + rng.uniform(-0.2, 0.2, N)
            h = np.clip(h, 0.0, 1.0)
        h = _project_onto_capped_simplex(h, target_sum)
        # gradient descent on worst shift
        step0 = 0.12
        decay = 0.9995
        max_iters = 80000
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.95:
                break
            O = _c5_fft_arr(h)
            cur_c = O.max()
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            idx = O.argmax()
            k = idx - (N - 1)
            grad = _grad_shift(h, k)
            step = step0 * (decay ** it)
            v = h + step * grad
            h = _project_onto_capped_simplex(v, target_sum)
        # short simulated‑annealing polishing
        T = 0.02
        for _ in range(2000):
            if time.time() - start > budget_s * 0.98:
                break
            nh = _random_move(h, rng, scale=0.05)
            nc = _c5_linear(nh)
            if nc < best_c or rng.random() < np.exp(-(nc - best_c) / T):

                if nc < best_c:
                    best_c = nc
                    best_h = nh.copy()
            T *= 0.999

        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_c = _c5_linear(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    final_c = _c5_linear(best_h)

import time
import numpy as np

def _enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum by moving mass between random coordinates."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def _c5_linear(h: np.ndarray) -> float:
    """Discrete C5 as defined in the problem statement (full linear correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _c5_fft_arr(h: np.ndarray) -> np.ndarray:
    """Array of O_k = ∫ h(x)(1‑h(x+k))dx for all integer shifts (FFT version)."""
    N = h.size
    b = 1.0 - h
    H = np.fft.rfft(h, n=2 * N)
    B = np.fft.rfft(b[::-1], n=2 * N)          # reversed b for linear correlation
    corr = np.fft.irfft(H * B, n=2 * N)[: 2 * N - 1]
    return corr * (2.0 / N)


def _grad_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of O_k w.r.t. h (positive direction reduces O_k)."""
    N = h.size
    k = abs(k)
    if k == 0:
        return 2.0 * h
    g = np.zeros_like(h)
    g[: N - k] += h[k:]
    g[k:] += h[: N - k]
    return g


def _random_move(h: np.ndarray, rng: np.random.Generator, scale: float = 0.05) -> np.ndarray:
    """Transfer a random amount of mass between two coordinates (keeps sum)."""
    N = h.size
    i, j = rng.integers(0, N, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    N = 2048                       # number of discretisation points
    target_sum = N / 2.0

    # ---- initial h -------------------------------------------------
    if rng.random() < 0.5:
        h = np.zeros(N)
        h[rng.choice(N, N // 2, replace=False)] = 1.0
    else:
        h = np.full(N, 0.5) + rng.uniform(-0.2, 0.2, N)
        h = np.clip(h, 0.0, 1.0)
    h = _enforce_sum(h, target_sum, rng)

    best_h = h.copy()
    best_c = _c5_linear(h)

    # ---- subgradient descent on the worst shift --------------------
    step0 = 0.12
    decay = 0.9997
    it = 0
    while True:
        if time.time() - start > budget_s * 0.90:
            break
        O = _c5_fft_arr(h)
        cur_c = O.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        idx = O.argmax()
        k = idx - (N - 1)
        grad = _grad_shift(h, k)
        step = step0 * (decay ** it)
        v = h + step * grad
        v = np.clip(v, 0.0, 1.0)
        h = _enforce_sum(v, target_sum, rng)
        it += 1

    # ---- short simulated‑annealing polishing -----------------------
    T = 0.02
    while True:
        if time.time() - start > budget_s * 0.98:
            break
        nh = _random_move(h, rng, scale=0.05)
        nc = _c5_fft_arr(nh).max()
        if nc < best_c or rng.random() < np.exp(-(nc - best_c) / T):
            h = nh
            if nc < best_c:
                best_c = nc
                best_h = nh.copy()
        T *= 0.999

    # ---- final repair and output -----------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _enforce_sum(best_h, target_sum, rng)
    final_c = _c5_linear(best_h)

    return best_h.tolist(), float(final_c), int(N)
