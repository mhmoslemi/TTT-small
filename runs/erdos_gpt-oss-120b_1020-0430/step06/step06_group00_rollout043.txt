def binary_sa_fast(n, rng, time_budget):
    start = time.time()
    best_h = None
    best_raw = np.inf
    # Pre-calc offset
    offset = n - 1
    total_h = n // 2
    # Keep T0,T_end
    T0 = 0.05
    T_end = 1e-5
    # initialize current h and D
    while time.time() - start < time_budget:
        # start new random binary vector
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, total_h, replace=False)
        h[ones] = 1
        D = np.correlate(h, h, mode='full')
        # compute current raw C
        prefix = np.concatenate(([0], np.cumsum(h, dtype=int)))
        ks = np.arange(n)
        pos_vals = prefix[n - ks] - D[offset + ks]
        neg_vals = total_h - prefix[ks[1:]] - D[offset + ks[1:]]
        curr_raw = max(pos_vals.max(), neg_vals.max())
        # set best
        if curr_raw < best_raw:
            best_raw = curr_raw
            best_h = h.copy()
        # compute elapsed for temperature
        run_start = time.time()
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            frac = (elapsed) / time_budget
            T = T0 * (T_end / T0) ** frac
            # propose swap
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            if h[i] == h[j]:
                continue
            old_h = h.copy()
            # update D with sign=+1
            # Using slice addition
            # D[offset - i : offset - i + n] += old_h
            D[offset - i : offset - i + n] += old_h
            D[offset - j : offset - j + n] -= old_h
            D[offset] += 1  # sign=+1
            # swap in h
            h[i], h[j] = h[j], h[i]
            # compute new raw C
            prefix = np.concatenate(([0], np.cumsum(h, dtype=int)))
            pos_vals = prefix[n - ks] - D[offset + ks]
            neg_vals = total_h - prefix[ks[1:]] - D[offset + ks[1:]]

            else:

                    best_raw = new_raw
                    best_h = h.copy()
                    # early exit if best bound < target
                    if (2.0/n) * best_raw <= 0.3808:

                # revert D using sign=-1 (same old_h)
                D[offset - i : offset - i + n] -= old_h
                D[offset - j : offset - j + n] += old_h
                D[offset] -= 1
                # curr_raw unchanged
        # end inner while loop for this initial vector
        # after this run, maybe start new random init again if time left.
    # after time budget, return best
    return best_h.tolist(), float((2.0/n) * best_raw), n

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ----------------------------------------------------------------------
# Fast binary simulated annealing with O(N) updates
def binary_sa_fast(n: int, rng: np.random.Generator, time_budget: float):
    """
    Binary SA on length n with exactly n/2 ones.
    Returns best h (list), best C5 (float), and n.
    """
    start = time.time()
    offset = n - 1
    total_h = n // 2
    ks = np.arange(n, dtype=int)

    best_h = None
    best_raw = np.iinfo(np.int64).max  # raw Cmax (before scaling)

    # annealing schedule
    T0, T_end = 0.05, 1e-5

    while time.time() - start < time_budget:
        # initialise a random binary vector
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, total_h, replace=False)
        h[ones] = 1
        D = np.correlate(h, h, mode="full")  # length 2n-1, int64
        # current raw C5
        prefix = np.empty(n + 1, dtype=np.int64)
        prefix[0] = 0
        np.cumsum(h, out=prefix[1:])
        pos_vals = prefix[n - ks] - D[offset + ks]
        neg_vals = total_h - prefix[ks[1:]] - D[offset + ks[1:]]
        curr_raw = int(max(pos_vals.max(), neg_vals.max()))
        if curr_raw < best_raw:
            best_raw = curr_raw
            best_h = h.copy()

        # inner SA loop
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            frac = elapsed / time_budget
            T = T0 * (T_end / T0) ** frac

            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if h[i] == h[j]:
                continue

            old_h = h.copy()

            # update D (forward)
            D[offset - i : offset - i + n] += old_h
            D[offset - j : offset - j + n] -= old_h
            D[offset] += 1

            # swap bits
            h[i], h[j] = h[j], h[i]

            # compute new raw C5
            np.cumsum(h, out=prefix[1:])
            pos_vals = prefix[n - ks] - D[offset + ks]
            neg_vals = total_h - prefix[ks[1:]] - D[offset + ks[1:]]
            new_raw = int(max(pos_vals.max(), neg_vals.max()))

            delta = new_raw - curr_raw
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta * (2.0 / n) / max(T, 1e-16)):
                    accept = True

            if accept:
                curr_raw = new_raw
                if new_raw < best_raw:
                    best_raw = new_raw
                    best_h = h.copy()
                    # early stop if target reached
                    if (2.0 / n) * best_raw <= 0.3808:
                        return best_h.tolist(), (2.0 / n) * best_raw, n
            else:
                # revert swap and D
                h[i], h[j] = h[j], h[i]
                D[offset - i : offset - i + n] -= old_h
                D[offset - j : offset - j + n] += old_h
                D[offset] -= 1
                # curr_raw unchanged

    # return best found
    return best_h.tolist(), (2.0 / n) * best_raw, n

# Gradient tidy‑up (same as previous implementation)
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = n / 2.0 - h.sum()
    return h

# Continuous simulated annealing (same as previous)
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        scale = 0.2 * (T / T0)

        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1

        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps

        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- initial solution (if provided) ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---- Phase 0: binary SA on 512 points ----
    remaining = total_budget - (time.time() - start)
    if remaining > 10.0:
        t_bin = min(remaining * 0.7, 800.0)
        h_bin, c_bin, n_bin = binary_sa_fast(512, rng, t_bin)
        if c_bin < best_c:
            best_h = np.array(h_bin, dtype=float)
            best_c = c_bin
            best_n = n_bin

    # ---- Phase 1: upsample to 1024 (if not already) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        if best_n < 1024:
            h_fine = upsample_h(best_h, 1024)
            h_fine = np.clip(h_fine, 0.0, 1.0)
            h_fine = enforce_sum(h_fine, rng)
            c_fine = compute_c5(h_fine)
            if c_fine < best_c:
                best_h, best_c, best_n = h_fine, c_fine, 1024
        else:
            best_n = best_h.size

    # ---- Phase 2: gradient tidy‑up (cheap) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h = gradient_refine(best_h, rng, step=0.015, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c
            best_n = h.size

    # ---- Phase 3: continuous SA on the finest grid ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h, c = continuous_sa(best_h, rng, remaining)
        if c < best_c:
            best_h, best_c = h, c
            best_n = h.size

    # ---- Final sanitisation ----
    if best_h is None:
        # fallback: constant 0.5 function on 512 points
        n = 512
        best_h = np.full(n, 0.5, dtype=float)
        best_n = n
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
