def compute_c5_fast(h, total_sum=None):
    n = h.size
    if total_sum is None:
        total_sum = np.sum(h)
    L = 2*n
    H = np.fft.rfft(h, n=L)  # real FFT for speed
    # autocorrelation via irfft
    conv = np.fft.irfft(H * np.conj(H), n=L)  # length L real values
    inner = conv[:n]  # inner_k for k=0..n-1
    # prefix sum
    prefix = np.cumsum(h)
    # compute overlap for k=1..n-1
    k_arr = np.arange(1, n)
    w_pos = prefix[n-1 - k_arr]   # shape (n-1,)
    w_neg = total_sum - prefix[k_arr - 1]  # suffix sum
    overlap = np.maximum(w_pos - inner[1:], w_neg - inner[1:])
    max_overlap = np.max(overlap) if overlap.size > 0 else 0.0
    return max_overlap * (2.0 / n)
```

def improve_worst_swap(h):
    n = h.size
    total = n // 2
    # Compute inner autocorrelation
    L = 2*n
    H = np.fft.rfft(h, n=L)
    conv = np.fft.irfft(H * np.conj(H), n=L)
    inner = conv[:n]
    prefix = np.cumsum(h)
    k_arr = np.arange(1, n)
    w_pos = prefix[n-1 - k_arr]
    w_neg = total - prefix[k_arr - 1]
    overlap_pos = w_pos - inner[1:]
    overlap_neg = w_neg - inner[1:]
    max_overlap = np.maximum(overlap_pos, overlap_neg)
    idx = np.argmax(max_overlap)
    k = idx + 1
    if overlap_pos[idx] >= overlap_neg[idx]:
        # positive direction
        # find A and B indices
        A_mask = np.zeros(n, dtype=bool)
        A_mask[:n - k] = (h[:n - k] == 1) & (h[k:] == 0)
        A = np.where(A_mask)[0]
        B_mask = np.zeros(n, dtype=bool)
        B_mask[k:] = (h[k:] == 0) & (h[:n - k] == 1)
        B = np.where(B_mask)[0]
    else:
        # negative direction: treat reversed
        # reverse h and compute similarly, then map indices back
        hr = h[::-1]
        # compute A,B for hr positive direction
        # we can reuse same code with hr
        # compute inner for hr not needed; we can directly compute A,B
        A_mask = np.zeros(n, dtype=bool)
        A_mask[:n - k] = (hr[:n - k] == 1) & (hr[k:] == 0)
        A = np.where(A_mask)[0]
        B_mask = np.zeros(n, dtype=bool)
        B_mask[k:] = (hr[k:] == 0) & (hr[:n - k] == 1)
        B = np.where(B_mask)[0]
        # map back to original indices: i_original = n-1 - i_rev
        A = n - 1 - A
        B = n - 1 - B
    # perform swaps as many as possible (limit to length of smaller set)
    num_swaps = min(len(A), len(B))
    if num_swaps == 0:
        return h, False
    # shuffle for randomness
    rng = np.random.default_rng()
    rng.shuffle(A)
    rng.shuffle(B)
    # for first num_swaps indices
    for i_idx, j_idx in zip(A[:num_swaps], B[:num_swaps]):
        # swap values (they are opposite by construction)
        h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
    return h, True

import time
import numpy as np

def compute_c5_fast(h, total=None):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on n points."""
    n = h.size
    if total is None:
        total = np.sum(h)
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    conv = np.fft.irfft(H * np.conj(H), n=L)
    inner = conv[:n]                     # linear autocorrelation
    pref = np.cumsum(h)
    ks = np.arange(1, n)
    w_pos = pref[n - 1 - ks]             # sum_{i<=n-1-k} h_i
    w_neg = total - pref[ks - 1]         # sum_{i>=k} h_i
    overlap = np.maximum(w_pos - inner[1:], w_neg - inner[1:])
    return np.max(overlap) * (2.0 / n)

def swap_reduce_max_overlap(h, rng, deadline):
    """Iteratively swap 1↔0 pairs that strictly lower the current worst overlap."""
    n = h.size
    total = n // 2
    while True:
        if time.time() > deadline:
            break
        L = 2 * n
        H = np.fft.rfft(h, n=L)
        conv = np.fft.irfft(H * np.conj(H), n=L)
        inner = conv[:n]
        pref = np.cumsum(h)
        ks = np.arange(1, n)
        w_pos = pref[n - 1 - ks]
        w_neg = total - pref[ks - 1]
        ov_pos = w_pos - inner[1:]
        ov_neg = w_neg - inner[1:]
        best = np.maximum(ov_pos, ov_neg)
        max_ov = best.max()
        if max_ov == 0:
            break
        idx = best.argmax()
        k = idx + 1
        if ov_pos[idx] >= ov_neg[idx]:
            A = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
            B = np.where((h[k:] == 0) & (h[:n - k] == 1))[0] + k
        else:
            hr = h[::-1]
            A_rev = np.where((hr[:n - k] == 1) & (hr[k:] == 0))[0]
            B_rev = np.where((hr[k:] == 0) & (hr[:n - k] == 1))[0] + k
            A = n - 1 - A_rev
            B = n - 1 - B_rev
        if A.size == 0 or B.size == 0:
            break
        m = min(A.size, B.size, 5)
        i_sel = rng.choice(A, size=m, replace=False)
        j_sel = rng.choice(B, size=m, replace=False)
        h[i_sel], h[j_sel] = h[j_sel], h[i_sel]
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise binary h -----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        h0 = (init >= 0.5).astype(np.int8)
        target = n // 2
        diff = target - h0.sum()
        while diff:
            if diff > 0:
                zero = np.where(h0 == 0)[0]
                h0[rng.choice(zero)] = 1
                diff -= 1
            else:
                one = np.where(h0 == 1)[0]
                h0[rng.choice(one)] = 0
                diff += 1
    else:
        n = 512
        target = n // 2
        h0 = rng.choice([0, 1], size=n).astype(np.int8)
        diff = target - h0.sum()
        while diff:
            if diff > 0:
                zero = np.where(h0 == 0)[0]
                h0[rng.choice(zero)] = 1
                diff -= 1
            else:
                one = np.where(h0 == 1)[0]
                h0[rng.choice(one)] = 0
                diff += 1
    best_h = h0.copy()
    best_c = compute_c5_fast(best_h)
    # ----- main stochastic search -----
    while time.time() - start < budget_s * 0.6:
        # start from best with small random swaps or fresh random
        if rng.random() < 0.3:
            h = best_h.copy()
            for _ in range(8):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = h[j], h[i]
        else:
            h = rng.choice([0, 1], size=n).astype(np.int8)
            diff = target - h.sum()
            while diff:
                if diff > 0:
                    zero = np.where(h == 0)[0]
                    h[rng.choice(zero)] = 1
                    diff -= 1
                else:
                    one = np.where(h == 1)[0]
                    h[rng.choice(one)] = 0
                    diff += 1
        # improve by targeted swaps
        deadline = start + budget_s * 0.6
        h = swap_reduce_max_overlap(h, rng, deadline)
        c = compute_c5_fast(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # ----- final refinement on the best candidate -----
    while time.time() - start < budget_s * 0.95:
        deadline = start + budget_s * 0.95
        best_h = swap_reduce_max_overlap(best_h, rng, deadline)
        best_c = compute_c5_fast(best_h)
    final_c = best_c
    return best_h.tolist(), float(final_c), int(n)
