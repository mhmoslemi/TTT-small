def continuous_sa(h0, rng, budget, size):
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    # Compute initial c
    c_curr, shift_curr, _ = compute_c5_and_shift(h, size)
    best_c = c_curr

    # configure schedule
    T0 = 0.02
    T_end = 1e-5

    it = 0
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget:
            break
        # Temperature
        T = T0 * (T_end / T0) ** (elapsed / budget)
        # compute gradient for current shift
        grad = worst_gradient(h, shift_curr)
        # choose i,j
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # Choose another pair?
            # just skip?
            # fallback: random pair
            i = rng.integers(n)
            j = rng.integers(n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                # continue
                continue
        # step size
        # base eps decays linearly from 0.2 to 0.001
        base_eps = 0.2 * (1 - elapsed / budget) + 0.001
        eps = min(max_eps, base_eps)
        # propose new
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # compute new c and shift
        c_new, shift_new, _ = compute_c5_and_shift(h_new, size)
        # acceptance
        if c_new < c_curr or rng.random() < np.exp(-(c_new - c_curr) / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            shift_curr = shift_new
            # occasionally fix sum due to rounding
            if it % 1000 == 0:
                # enforce exact sum
                diff = n/2.0 - h.sum()
                if abs(diff) > 1e-10:
                    # adjust random index

    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ------------------- continuous simulated annealing -------------------
def continuous_sa(h0: np.ndarray, rng: np.random.Generator,
                 budget: float) -> tuple[np.ndarray, float]:
    start = time.time()
    n = h0.size
    h = h0.copy()
    c_curr, shift_curr, _ = compute_c5_and_shift(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.02, 1e-5
    it = 0
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget:
            break
        T = T0 * (T_end / T0) ** (elapsed / budget)
        grad = worst_gradient(h, shift_curr)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            i = rng.integers(n)
            j = rng.integers(n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
        base_eps = 0.2 * (1.0 - elapsed / budget) + 0.001
        eps = min(max_eps, base_eps)
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new, shift_new, _ = compute_c5_and_shift(h_new)
        if c_new < c_curr or rng.random() < np.exp(-(c_new - c_curr) / max(T, 1e-12)):
            h, c_curr, shift_curr = h_new, c_new, shift_new
            if it % 1000 == 0:
                diff = n / 2.0 - h.sum()
                if abs(diff) > 1e-10:
                    idxs = np.where(h < 1.0)[0] if diff > 0 else np.where(h > 0.0)[0]
                    if idxs.size:
                        k = rng.choice(idxs)
                        adjust = min(diff, 1.0 - h[k]) if diff > 0 else max(diff, -h[k])
                        h[k] += adjust
            if c_curr < best_c:
                best_c, best_h = c_curr, h.copy()
        if it % 5000 == 0 and it > 0:
            # occasional multi‑pair perturbation
            h_tmp = h.copy()
            for _ in range(5):
                a = rng.integers(n)
                b = rng.integers(n)
                if a == b:
                    continue
                eps_max = min(h_tmp[a], 1.0 - h_tmp[b])
                if eps_max <= 0.0:
                    continue
                eps2 = rng.random() * eps_max * 0.5
                h_tmp[a] -= eps2
                h_tmp[b] += eps2
            c_tmp, shift_tmp, _ = compute_c5_and_shift(h_tmp)
            if c_tmp < c_curr or rng.random() < np.exp(-(c_tmp - c_curr) / max(T, 1e-12)):
                h, c_curr, shift_curr = h_tmp, c_tmp, shift_tmp
                if c_curr < best_c:
                    best_c, best_h = c_curr, h.copy()
        it += 1
    return best_h, best_c


# ------------------- main entry point -------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n = 512
        h0 = rng.random(n)
        h0 = enforce_sum(h0, rng)

    n = h0.size
    global FFT_SIZE
    FFT_SIZE = 1 << ((2 * n - 1).bit_length())

    # time splits
    sa_budget = total_budget * 0.70
    slsqp_budget = total_budget * 0.25
    # remaining for final polishing
    remaining = total_budget - (time.time() - start) - sa_budget - slsqp_budget
    final_budget = max(0.0, remaining)

    # ---------- continuous SA ----------
    best_h, best_c = continuous_sa(h0, rng, sa_budget)

    # ---------- SLSQP polishing ----------
    elapsed = time.time() - start
    if elapsed + slsqp_budget < total_budget:
        h = best_h.copy()
        # allocate modest iterations per beta
        beta_seq = (20., 80., 200., 800., 3000., 8000., 20000.)
        per_beta = max(300, int(slsqp_budget * 0.4 / len(beta_seq)))
        for beta in beta_seq:
            if time.time() - start > total_budget * 0.96:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=per_beta)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # ---------- final refinement ----------
    elapsed = time.time() - start
    if elapsed + final_budget < total_budget and best_h is not None:
        # guided hill climb
        max_iters = int(min(50000, final_budget * 500))
        h, c = guided_hill_climb(best_h.copy(), rng, max_iters=max_iters, step_scale=0.7)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # tiny gradient tidy‑up
        h = gradient_refine(best_h.copy(), rng, step=0.01, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # sanitise
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
