def worst_gradient_from_corr(h, corr):
    n = h.size
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    # subtract contributions
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

            lo = lam
        else:
            hi = lam

        mask = (x > 1e-12) & (x < 1 - 1e-12)

import time, math, numpy as np

# ---------- projection onto [0,1] box with exact sum ----------
def project_onto_box_sum(v, target):
    # binary search for lambda in a reasonable interval
    lo, hi = -1.0, 1.0
    for _ in range(32):                     # 2⁻³² ≈ 2·10⁻¹⁰ accuracy
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


# ---------- FFT based correlation ----------
def compute_corr_fft(h, n_fft):
    """linear correlation of h with (1-h) using FFT.
       returns array of length 2*len(h)-1 corresponding to shifts -n+1..n-1"""
    g_rev = (1.0 - h)[::-1]
    fh = np.fft.rfft(h, n=n_fft)
    fg = np.fft.rfft(g_rev, n=n_fft)
    conv = np.fft.irfft(fh * fg, n=n_fft)
    return conv[: 2 * h.size - 1]


def compute_c5_fft(h, n_fft):
    """C5 value for a given h (using FFT correlation)"""
    corr = compute_corr_fft(h, n_fft)
    return corr.max() * (2.0 / h.size)


# ---------- gradients ----------
def worst_gradient_from_corr(h, corr):
    n = h.size
    idx = np.argmax(corr) - (n - 1)          # shift index where overlap is maximal
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:]    -= h[: n - k]
    return grad


def worst_gradient(h):
    """fallback gradient using np.correlate (used only in hill‑climb)"""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:]    -= h[: n - k]
    return grad


# ---------- guided perturbation ----------
def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- hill climbing using the gradient ----------
def guided_hill_climb(h, rng, n_fft, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5_fft(best_h, n_fft)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        # gradient from the current worst shift (fast via FFT)
        corr = compute_corr_fft(h, n_fft)
        grad = worst_gradient_from_corr(h, corr)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5_fft(nh, n_fft)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- problem size (choose a power‑of‑two for cheap FFT) ----
    n = 1024                     #  ≤ 1000 samples as requested, still fast
    target = n / 2.0
    n_fft = 1 << ((2 * n - 1).bit_length())   # next power of two (2048)

    # ---- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n,     endpoint=False)
        h0 = np.interp(x_new, x_old, init)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
    else:
        h0 = rng.random(n)
        h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5_fft(best_h, n_fft)

    # ---- main optimisation loop (multiple random restarts) ----------
    while time.time() - start < budget_s * 0.95:
        # start either from current best (occasionally) or a fresh random point
        if rng.random() < 0.15 and best_c < 0.381:
            h = best_h.copy()
        else:
            h = rng.random(n)
        h = project_onto_box_sum(h, target)

        # how many sub‑gradient steps can we still afford?
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(200000, remaining * 15000))   # optimistic ~15k it/s
        if max_iters < 2000:
            max_iters = 2000

        step0 = 2.0
        for t in range(max_iters):
            if time.time() - start > budget_s * 0.99:
                break
            corr = compute_corr_fft(h, n_fft)
            c5 = corr.max() * (2.0 / n)
            if c5 < best_c:
                best_c, best_h = c5, h.copy()
            grad = worst_gradient_from_corr(h, corr)
            step = step0 / math.sqrt(t + 1)
            v = h - step * grad
            h = project_onto_box_sum(v, target)

        # short hill‑climb on the current best solution
        hc, cc = guided_hill_climb(best_h, rng, n_fft,
                                   max_iters=3000, step_scale=0.4)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5_fft(best_h, n_fft)
    return best_h.tolist(), float(final_c), int(best_h.size)
