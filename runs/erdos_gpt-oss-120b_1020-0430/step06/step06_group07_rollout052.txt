def guided_hill_climb(h0, rng, max_iters=2000, step_scale=0.7):
    h = h0.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # maybe accept if improves else revert? The original code only updates best but keeps h as new even if not improved. We'll keep as is to continue exploring. Good.
        if (it+1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def simulated_anneal(h0, rng, steps, init_temp=0.01, final_temp=1e-6):
    h = h0.copy()
    c_cur = compute_c5(h)
    best_c = c_cur
    best_h = h.copy()
    for step in range(steps):
        # choose two distinct indices with possible moves
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        if i == j:
            continue
        # ensure we can transfer from i to j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        h[i] -= eps
        h[j] += eps
        c_new = compute_c5(h)
        # temperature schedule
        t = init_temp * (final_temp / init_temp) ** (step / steps)
        accept = False
        if c_new < c_cur:
            accept = True
        else:
            # probability of Accept worse
            if np.random.rand() < np.exp((c_cur - c_new) / (t + 1e-12)):
                accept = True
        if accept:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
    return best_h, best_c

def upsample(h, rng):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    # after interpolation, may be out of [0,1] slightly; clip
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

import time
import numpy as np

# --------------------------------------------------------------
# basic utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via binary search on the shift."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in {0≤h≤1, sum h = n/2}."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the shift attaining max O_k = max_h ∫ h(1‑h(·+k))."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))  # shift giving largest O_k
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


# --------------------------------------------------------------
# local optimisation primitives
def subgradient_optimize_iter(h0: np.ndarray, iters: int, target: float):
    """Projected sub‑gradient descent on max‑overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, iters + 1):
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10.0)
        h = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def guided_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed mass transfers."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def simulated_anneal(h0: np.ndarray, rng: np.random.Generator,
                    steps: int, init_temp: float = 0.005,
                    final_temp: float = 1e-7):
    """Random mass‑transfer annealing with decreasing temperature."""
    h = h0.copy()
    c_cur = compute_c5(h)
    best_h = h.copy()
    best_c = c_cur
    for step in range(steps):
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        h[i] -= eps
        h[j] += eps
        c_new = compute_c5(h)
        temp = init_temp * (final_temp / init_temp) ** (step / steps)
        accept = False
        if c_new < c_cur:
            accept = True
        else:
            if rng.random() < np.exp((c_cur - c_new) / (temp + 1e-12)):
                accept = True
        if accept:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] += eps
            h[j] -= eps
    return best_h, best_c


def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Linear up‑sampling by factor 2 with feasibility projection."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    total_budget = float(budget_s)

    best_h = None
    best_c = np.inf

    # include the provided construction if available

        best_h, best_c = init_h, c_init

    # --------------------------------------------------------------

    target_restart = n_restart / 2.0

        # guided hill‑climb
        h_hc, c_hc = guided_hill_climb(
            h_sg, rng, max_iters=1000, step_scale=0.7)

        # keep the best of the two
        if c_sg < best_c:
            best_c, best_h = c_sg, h_sg.copy()
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # optional early stop
        if best_c <= 0.3808:
            break

    # --------------------------------------------------------------
    # Phase B – final polishing on the finer grid
    remaining = total_budget - (time.time() - start)
    if best_h is not None and remaining > 5.0:

        if best_h.size < 512:
            best_h = upsample(best_h, rng)

        # allocate iterations proportional to remaining time
        n_fine = best_h.size
        target_fine = n_fine / 2.0

        # sub‑gradient on fine grid
        iters_sg = int(min(15000, remaining * 5000))
        if iters_sg > 0:
            h_sg2, c_sg2 = subgradient_optimize_iter(
                best_h, iters=iters_sg, target=target_fine)
            if c_sg2 < best_c:
                best_c, best_h = c_sg2, h_sg2.copy()

        # guided hill‑climb on fine grid
        remaining = total_budget - (time.time() - start)
        if remaining > 5.0:
            max_iters_hc = int(min(8000, remaining * 600))
            h_hc2, c_hc2 = guided_hill_climb(
                best_h, rng, max_iters=max_iters_hc, step_scale=0.6)
            if c_hc2 < best_c:
                best_c, best_h = c_hc2, h_hc2.copy()

        remaining = total_budget - (time.time() - start)
        if remaining > 5.0:

            if steps_anneal > 0:

                    best_c, best_h = c_an2, h_an2.copy()

    # final sanitisation
    if best_h is None:

        best_h = np.full(n_fallback, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
