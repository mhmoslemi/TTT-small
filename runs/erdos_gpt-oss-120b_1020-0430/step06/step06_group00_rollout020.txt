import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def gradient_logsumexp(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp approximation of max correlation."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = corr.max()
    exps = np.exp(beta * (corr - maxc))
    w = exps / exps.sum()
    w0 = w[N - 1]               # shift 0
    w_pos = w[N:]               # shifts +1 … +N‑1  (len N‑1)
    w_neg = w[:N - 1]           # shifts -1 … -(N‑1)
    grad = w0 * (1.0 - 2.0 * h)
    h_comp = 1.0 - h
    for k in range(1, N):
        wp = w_pos[k - 1]
        if wp != 0.0:
            grad[k:] += wp * h_comp[: N - k]
        wn = w_neg[k - 1]
        if wn != 0.0:
            grad[: N - k] -= wn * h[k:]
    return grad
