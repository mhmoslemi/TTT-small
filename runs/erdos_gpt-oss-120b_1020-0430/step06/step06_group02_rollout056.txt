def improve_by_pairwise(h, grad, best_c, rng):
    i_max = int(np.argmax(grad))
    i_min = int(np.argmin(grad))
    if i_max == i_min:
        return h, best_c
    delta_max = min(h[i_max], 1.0 - h[i_min])
    if delta_max <= 1e-12:
        return h, best_c
    for f in (0.5, 0.2, 0.1, 0.05, 0.01):
        delta = delta_max * f
        h_new = h.copy()
        h_new[i_max] -= delta
        h_new[i_min] += delta
        # no need to project if we respect bounds and sum unchanged
        c = compute_c5(h_new)
        if c < best_c:
            return h_new, c
    return h, best_c

import time, numpy as np
from scipy.optimize import minimize

def project_sum_bounds(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(x + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return proj

def compute_c5_factory(n):
    dx = 2.0 / n
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    return compute_c5

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

# global for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=1500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

                                               "ftol": 1e-12,
                                               "disp": False})

        improved = False
        for frac in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            delta = delta_max * frac
            nh = best_h.copy()
            nh[i_max] -= delta
            nh[i_min] += delta
            c = compute_c5(nh)
            if c < best_c:
                best_h, best_c = nh, c
                improved = True
                break
        if not improved and rng.random() < 0.3:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = min(best_h[i], 1.0 - best_h[j]) * rng.random() * 0.005
                if eps > 0:
                    nh = best_h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    c = compute_c5(nh)
                    if c < best_c:
                        best_h, best_c = nh, c

    # final projection and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, target_sum)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
