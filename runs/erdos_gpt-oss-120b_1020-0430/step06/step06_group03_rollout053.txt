import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target, max_iter=60):
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def shift_array(h, k):
    n = h.size
    shifted = np.empty_like(h)
    if k >= 0:
        nk = n - k
        shifted[:nk] = h[k:]
        shifted[nk:] = 0.0
    else:
        kk = -k
        shifted[kk:] = h[:n - kk]
        shifted[:kk] = 0.0
    return shifted

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size and starting vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # optional upscale to 1024 if time allows and n divides 1024
        if n < 1024 and 1024 % n == 0:
            factor = 1024 // n
            h0 = np.tile(h0, factor)
            n = h0.size
    else:
        n = 1024
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_sum(h0, target)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)

    step_scale = 0.5
    min_step = 1e-4
    decay = 0.97
    outer_iter = 0

    while time.time() - start < budget_s * 0.98:
        # evaluate worst overlap
        corr = np.correlate(h, 1.0 - h, mode='full')
        Ofull = corr * (2.0 / n)
        cur_c = Ofull.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        w_idx = Ofull.argmax()
        k = w_idx - (n - 1)

        # inner refinement for the current worst shift
        inner_iters = max(2000, n)  # enough work per outer iteration
        for _ in range(inner_iters):
            if time.time() - start >= budget_s * 0.99:
                break

            shifted = shift_array(h, k)

            # source: h>0 and shifted small
            src_mask = (h > 1e-12)
            if not src_mask.any():
                break
            src_shifted = shifted[src_mask]
            src_idx_all = np.nonzero(src_mask)[0]
            i = src_idx_all[np.argmin(src_shifted)]

            # destination: h<1 and shifted large
            dst_mask = (h < 1.0 - 1e-12)
            if not dst_mask.any():
                break
            dst_shifted = shifted[dst_mask]
            dst_idx_all = np.nonzero(dst_mask)[0]
            j = dst_idx_all[np.argmax(dst_shifted)]

            if i == j:
                # fallback random move
                nz = np.where(h > 1e-12)[0]
                nf = np.where(h < 1.0 - 1e-12)[0]
                if nz.size == 0 or nf.size == 0:
                    break
                i = rng.choice(nz)
                j = rng.choice(nf)
                if i == j:
                    continue

            eps = min(h[i], 1.0 - h[j]) * step_scale
            if eps <= 1e-12:
                continue
            h[i] -= eps
            h[j] += eps

        # decay step size
        step_scale = max(step_scale * decay, min_step)
        outer_iter += 1

        # occasional diversification
        if outer_iter % 50 == 0:
            h = best_h.copy()
            # small random swaps (≈1% of points)
            swaps = max(1, n // 100)
            for _ in range(swaps):
                i, j = rng.integers(n), rng.integers(n)
                if i == j:
                    continue
                eps = min(h[i], 1.0 - h[j]) * 0.1 * rng.random()
                if eps > 1e-12:
                    h[i] -= eps
                    h[j] += eps

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
