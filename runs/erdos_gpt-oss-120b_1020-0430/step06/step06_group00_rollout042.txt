def repair_sum(v, target):
    v = np.clip(v, 0, 1)
    s = v.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        # add diff
        while diff > 1e-12:
            # find indices where v < 1
            idx = np.where(v < 1 - 1e-12)[0]
            if len(idx) == 0: break
            # add equally to all
            add_per = diff / len(idx)
            # cap to 1 - v[i]
            cap = 1 - v[idx]
            add = np.minimum(add_per, cap)
            v[idx] += add
            diff = target - v.sum()
    else:
        # diff < 0, subtract
        while diff < -1e-12:
            idx = np.where(v > 1e-12)[0]
            if len(idx) == 0: break
            sub_per = -diff / len(idx)
            cap = v[idx]
            sub = np.minimum(sub_per, cap)
            v[idx] -= sub
            diff = target - v.sum()
    return v

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)  # matching parent
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        # direction random
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps

        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def slsqp_optimize(h0, beta, target_sum, rng, max_iter=800):
    n = h0.size
    # define objective softmax
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # compute log-sum-exp
        lse = np.log(np.sum(np.exp(bc - m))) + m
        return lse / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # precise sum correction using repair_sum
    x = repair_sum(x, target_sum)
    return x

def repair_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return v
    # Adding diff
    if diff > 0:
        # While diff left and capacity exists
        while diff > 1e-12:
            capacity = 1.0 - v
            total_cap = capacity.sum()
            if total_cap < 1e-12:
                break
            add = diff * (capacity / total_cap)  # distribute proportionally
            add = np.minimum(add, capacity)
            v += add
            diff = target - v.sum()
    else:
        # diff < 0
        while diff < -1e-12:
            capacity = v
            total_cap = capacity.sum()
            if total_cap < 1e-12:
                break
            sub = (-diff) * (capacity / total_cap)
            sub = np.minimum(sub, capacity)
            v -= sub
            diff = target - v.sum()
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

This distributes proportionally to capacity, guaranteeing convergence unless capacity zero.

def slsqp_optimize(h0, beta, target_sum, max_iter=200):
    n = h0.size
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        lse = np.log(np.sum(np.exp(bc - m))) + m
        return lse / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = repair_sum(x, target_sum)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def repair_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust linearly to meet the exact sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # distribute the remaining difference proportionally to available capacity
    if diff > 0:
        while diff > 1e-12:
            cap = 1.0 - v
            total = cap.sum()
            if total < 1e-12:
                break
            add = diff * (cap / total)
            add = np.minimum(add, cap)
            v += add
            diff = target - v.sum()
    else:
        while diff < -1e-12:
            cap = v
            total = cap.sum()
            if total < 1e-12:
                break
            sub = -diff * (cap / total)
            sub = np.minimum(sub, cap)
            v -= sub
            diff = target - v.sum()
    return np.clip(v, 0.0, 1.0)


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def slsqp_optimize(h0: np.ndarray, beta: float, target_sum: float, max_iter: int = 200) -> np.ndarray:
    """Soft‑max SLSQP step for a given β."""
    n = h0.size

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        lse = np.log(np.sum(np.exp(bc - m))) + m
        return lse / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return repair_sum(x, target_sum)


def custom_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA that directly works on the minimum autocorrelation (maximises it)."""
    h = h0.copy()
    N = h.size
    full_corr = np.correlate(h, h, mode="full")
    g = full_corr[N - 1 :]  # g[k] = sum_i h_i h_{i+k}
    cur_min = g.min()
    best_h = h.copy()
    best_min = cur_min
    start = time.time()
    T0, T_end = 0.5, 1e-7

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        direction = rng.random() < 0.5  # True: i up, j down
        if direction:
            max_eps = min(1.0 - h[i], h[j])
        else:
            max_eps = min(1.0 - h[j], h[i])
        if max_eps <= 0.0:
            continue
        delta = max_eps * (T / T0) * rng.random()
        if delta <= 0.0:
            continue

        g_new = g.copy()
        if direction:
            # i up
            fl_i = N - i
            if fl_i > 0:
                g_new[:fl_i] += delta * h[i:]
            g_new[: i + 1] += delta * h[i :: -1]
            # j down
            fl_j = N - j
            if fl_j > 0:
                g_new[:fl_j] -= delta * h[j:]
            g_new[: j + 1] -= delta * h[j :: -1]
        else:
            # j up
            fl_j = N - j
            if fl_j > 0:
                g_new[:fl_j] += delta * h[j:]
            g_new[: j + 1] += delta * h[j :: -1]
            # i down
            fl_i = N - i
            if fl_i > 0:
                g_new[:fl_i] -= delta * h[i:]
            g_new[: i + 1] -= delta * h[i :: -1]

        new_min = g_new.min()
        dmin = new_min - cur_min
        if dmin >= 0.0 or rng.random() < np.exp(dmin / max(T, 1e-12)):
            if direction:
                h[i] += delta
                h[j] -= delta
            else:
                h[j] += delta
                h[i] -= delta
            g = g_new
            cur_min = new_min
            if new_min > best_min:
                best_min = new_min
                best_h = h.copy()

    c = 1.0 - (best_min * (2.0 / N))
    return best_h, c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    N = 1024
    target_sum = N / 2.0

    # Initialise
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        if h.size != N:
            h = upsample_h(h, N)
        h = repair_sum(h, target_sum)
    else:
        h = repair_sum(rng.random(N), target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Soft‑max SLSQP refinement (if enough time)
    remaining = budget_s - (time.time() - start)
    if remaining > 150:
        betas = (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0)
        cur = h.copy()
        for beta in betas:
            if time.time() - start > budget_s * 0.9:
                break
            cur = slsqp_optimize(cur, beta, target_sum, max_iter=400)
            c = compute_c5(cur)
            if c < best_c:
                best_c, best_h = c, cur.copy()
        h = cur

    # Custom SA on the minimum autocorrelation
    remaining = budget_s - (time.time() - start)
    if remaining > 10:
        h_sa, c_sa = custom_sa(best_h, rng, remaining)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

    # Final sanitisation
    best_h = repair_sum(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
