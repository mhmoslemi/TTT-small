def compute_c5(h):
    n = h.shape[0]
    L = 1 << (2*n - 1).bit_length()  # next power of 2 >= 2n-1
    # compute FFTs
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    corr = np.fft.irfft(H * G, L)
    # correlation values are from index 0 to 2n-2
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

def simulated_annealing(h, budget, start_time, rng):

    n = len(h)

    best_h = h.copy()
    best_c = compute_c5(h)

    cur_h = h.copy()
    cur_c = best_c

    total_time = budget

    # dynamic parameters
    T0 = max(0.02, (best_c - 0.35) * 0.5)   # guess
    # Ensure T0 > 0
    if T0 <= 0: T0 = 0.01

    iteration = 0

    while time.time() - start_time < total_time:
        elapsed = time.time() - start_time
        # temperature linear cooling
        T = T0 * max(0.0, 1.0 - elapsed / total_time)

        # choose random i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute max_transfer
        hi = cur_h[i]
        hj = cur_h[j]
        max_transfer = min(hi, 1.0 - hj)
        if max_transfer <= 0:
            continue
        # step scale proportional to temperature ratio
        step_scale = T / T0 if T0 > 0 else 0.0
        # ensure positive step scale; set min 0.01
        step_scale = max(step_scale, 0.01)
        eps = max_transfer * rng.random() * step_scale

        # create new h copy
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        # compute new C5
        new_c = compute_c5(new_h)

        # acceptance
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h, cur_c = new_h, new_c

        # update best
        if cur_c < best_c:
            best_c, best_h = cur_c, cur_h.copy()
            # optional early exit
            if best_c <= 0.3808:
                break

        iteration += 1

    return best_h, best_c

import time
import numpy as np

# ------------------- helper functions --------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return max overlap C5 for step function h on [0,2] using FFT."""
    n = h.shape[0]
    # length for linear convolution via FFT
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    corr = np.fft.irfft(H * G, L)
    # correlation values for shifts -(n-1)..(n-1) are first 2*n-1 entries
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)


def project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum x = target} via binary search on λ."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the overlap at the worst (max) shift."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))  # shift giving max overlap
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h0: np.ndarray, budget: float,
                        start_time: float, rng: np.random.Generator):
    """SA on mass‑transfer moves, respecting box and sum constraints."""
    n = h0.shape[0]
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    T0 = 0.03  # initial temperature
    while time.time() - start_time < budget:
        elapsed = time.time() - start_time
        frac = max(0.0, 1.0 - elapsed / budget)
        T = T0 * frac
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        hi = cur_h[i]
        hj = cur_h[j]
        max_transfer = min(hi, 1.0 - hj)
        if max_transfer <= 0.0:
            continue
        # step size proportional to temperature ratio
        step_scale = max(T / T0, 0.01)
        eps = max_transfer * rng.random() * step_scale
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / (T + 1e-12))):
            cur_h, cur_c = new_h, new_c
        if cur_c < best_c:
            best_c, best_h = cur_c, cur_h.copy()
            if best_c <= 0.38078:
                break
    # ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    return best_h, best_c
