def softmax_gradient_fft(h, beta):
    n = h.size
    target = n/2.0
    corr = correlation_fft(h)  # len n
    O = target - corr    # O_k for k>=0
    Omax = O.max()
    exp_vals = np.exp(beta * (O - Omax))
    # compute weight sums: w0 = exp_vals[0], wk = exp_vals[k] * 2 for k>0
    w0 = exp_vals[0]
    wk = np.empty(n)
    wk[0] = w0
    wk[1:] = 2.0 * exp_vals[1:]
    sum_all = wk.sum()
    wk = wk / sum_all  # these wk are now sum of both signs
    # Now compute gradient
    grad = np.zeros_like(h)
    # k=0 contribution:
    grad += wk[0] * (1.0 - 2.0 * h)
    # contributions for k>0
    for k in range(1, n):
        w = wk[k]
        if w == 0.0:
            continue
        grad += w
        grad[: n - k] -= w * h[k:]
        grad[k:] -= w * h[: n - k]
    return grad

import time, numpy as np

def project_onto_box_sum(v, target):
    # binary search for lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # adjust small residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1-1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - (lo+hi)*0.5))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def correlation_fft(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    H_rev = np.fft.rfft(h[::-1], n=L)
    conv = np.fft.irfft(H * H_rev, n=L)
    return conv[n-1:n-1+n]

def compute_c5_fast(h, target):
    corr = correlation_fft(h)
    min_corr = corr.min()
    return (target - min_corr) * (2.0 / h.size)

def worst_gradient(h):
    n = h.size
    corr = correlation_fft(h)
    k = int(corr.argmin())
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

        if i == j:

        max_eps = min(h[i], 1.0 - h[j])

        nh[i] -= eps

    return best_h, best_c

def softmax_gradient_fft(h, beta):
    n = h.size
    target = n / 2.0
    corr = correlation_fft(h)  # len n
    O = target - corr
    Omax = O.max()
    exp_vals = np.exp(beta * (O - Omax))
    wk = np.empty(n, dtype=np.float64)
    wk[0] = exp_vals[0]
    wk[1:] = 2.0 * exp_vals[1:]
    sum_all = wk.sum()
    wk /= sum_all
    grad = np.zeros_like(h)
    grad += wk[0] * (1.0 - 2.0 * h)
    for k in range(1, n):
        w = wk[k]
        if w == 0.0:
            continue
        grad += w
        grad[: n - k] -= w * h[k:]
        grad[k:] -= w * h[: n - k]
    return grad

            if time.time() - start > budget_s * 0.80:
                break
            # iteration counts scaled with n
            iters = int(1200 * 256 / n) if beta >= 2000 else int(800 * 256 / n)
            step0 = step0_map[beta]
            for it in range(iters):
                if time.time() - start > budget_s * 0.80:
                    break
                grad = softmax_gradient_fft(h, beta)
                norm = np.linalg.norm(grad)
                if norm > 0.0:
                    step = step0 * (1.0 - it / iters) / norm
                    v = h - step * grad
                    h = project_onto_box_sum(v, target)
            c = compute_c5_fast(h, target)
            if c < best_c:
                best_h, best_c = h.copy(), c
            # occasional hill‑climb refinement
            if time.time() - start < budget_s * 0.85:
                hc, cc = guided_hill_climb(best_h, rng,
                                           max_iters=int(1000 * 256 / n),
                                           step_scale=0.5)
                if cc < best_c:
                    best_h, best_c = hc.copy(), cc
        # stop early if no time for another full pass
        if time.time() - start > budget_s * 0.80:
            break

    # Final fine‑grained adjustments
    while time.time() - start < budget_s * 0.99:
        grad, _ = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5_fast(nh, target)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5_fast(best_h, target)
    return best_h.tolist(), float(final_c), int(best_h.size)
