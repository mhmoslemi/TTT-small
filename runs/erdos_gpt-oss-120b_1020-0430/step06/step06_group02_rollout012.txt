    # After clipping, values are max(0, min(1, x - lambda)). As lambda increases, x - lambda decreases, so sum decreases.
    # So lambda monotonic decreasing sum, we can binary search.
    # Lower bound for lambda: enough low that x - lambda >= 1 for all -> after clipping sum = n.
    # Lower bound = min(x) - 1? Actually we need lambda <= min(x) - 1 ensures x - lambda >= 1. So lo = min(x) - 1.

    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(max_iter):

        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    return y

        grad[: n - k] += -h[k:]
    return grad


def project_sum_box(x, target, tol=1e-12, max_iter=60):
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > tol:
        y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target = n / 2.0

    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, target)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_c = compute_c5_fft(best_h)

    # Phase 1: aggressive gradient descent
    phase1_end = start + budget_s * 0.70
    step = 0.3
    h_curr = best_h.copy()
    it = 0
    while time.time() < phase1_end:
        grad = worst_gradient_fft(h_curr)
        cand = project_sum_box(h_curr - step * grad, target)
        cand_c = compute_c5_fft(cand)
        if cand_c < best_c:
            best_c = cand_c
            best_h = cand.copy()
            h_curr = cand.copy()
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.5
            if step < 1e-6:
                # random perturbation around the best known point
                h_curr = best_h.copy()
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    eps = rng.random() * min(h_curr[i], 1.0 - h_curr[j]) * 0.1
                    h_curr[i] -= eps
                    h_curr[j] += eps
                h_curr = project_sum_box(h_curr, target)
                step = 0.3
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = rng.random() * min(h_curr[i], 1.0 - h_curr[j]) * 0.2
                h_curr[i] -= eps
                h_curr[j] += eps
                h_curr = project_sum_box(h_curr, target)
        it += 1

    # Phase 2: SLSQP refinement with increasing β
    phase2_end = start + budget_s * 0.90
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
        if time.time() >= phase2_end:
            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=1500)
        c_opt = compute_c5_fft(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            if best_c <= 0.3808:
                break

    # Phase 3: fine‑grained random tweaks
    final_end = start + budget_s * 0.99
    while time.time() < final_end:
        grad = worst_gradient_fft(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.02
                cand = best_h.copy()
                cand[i] -= eps
                cand[j] += eps
                cand = project_sum_box(cand, target)
                c_cand = compute_c5_fft(cand)
                if c_cand < best_c:
                    best_c = c_cand
                    best_h = cand.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, target)
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
