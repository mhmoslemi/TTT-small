def mass_transfer_move_eps(h, pos, neg, total_out, total_in, eps):
    h_new = h.copy()
    if total_out > 0:
        h_new[pos] = h[pos] * (1 - eps/total_out)
    if total_in > 0:
        h_new[neg] = h[neg] + (1 - h[neg]) * (eps/total_in)
    return h_new

import time

from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap == 0.0:
                continue
            delta = min(diff, cap)
            h[i] += delta
        else:
            cap = h[i]
            if cap == 0.0:
                continue
            delta = min(-diff, cap)
            h[i] -= delta
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h, target, rng, beta, max_iter):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

        neg = grad < -1e-12
        improved = False
        if pos.any() and neg.any():
            total_out = np.sum(h[pos])
            total_in = np.sum(1.0 - h[neg])
            max_eps = min(total_out, total_in)
            if max_eps > 0:
                for a in alphas:
                    eps = max_eps * a
                    h_new = mass_transfer_move_eps(h, pos, neg, total_out, total_in, eps)
                    h_new = np.clip(h_new, 0.0, 1.0)
                    diff = target_sum - h_new.sum()
                    if abs(diff) > 1e-12:
                        h_new = enforce_sum(h_new, target_sum, rng)
                    c_new = compute_c5(h_new)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        best_h = h_new.copy()
                    if c_new < cur_c - 1e-13:
                        h = h_new
                        improved = True
                        no_improve = 0
                        break
        if not improved:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = max_eps * rng.random() * 0.1
                    h2 = h.copy()
                    h2[i] -= eps
                    h2[j] += eps
                    c2 = compute_c5(h2)
                    if c2 < best_c - 1e-13:
                        best_c = c2
                        best_h = h2.copy()
                    if c2 < cur_c - 1e-13:
                        h = h2
        it += 1
        if not improved:
            no_improve += 1
            if no_improve > 20000:
                h = np.zeros(n)
                idx0 = rng.choice(n, n // 2, replace=False)
                h[idx0] = 1.0
                h = enforce_sum(h, target_sum, rng)
                no_improve = 0
        if it % 20000 == 0 and time.time() + 5 < time_end:
            h_s = slsqp_refine(h, target_sum, rng, beta=2000.0, max_iter=100)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()
                h = h_s.copy()
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_end = start + budget_s * 0.99

    n = 512
    target = n / 2.0
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
    else:
        h0 = np.zeros(n)
        idx0 = rng.choice(n, n // 2, replace=False)
        h0[idx0] = 1.0

    best_h, best_c = hill_climb(h0, target, rng, time_end)

    # Upsample if time permits
    if time.time() + 200 < time_end:
        h_up = np.repeat(best_h, 2)
        target_up = h_up.size / 2.0
        up_h, up_c = hill_climb(h_up, target_up, rng, time_end)
        if up_c < best_c:
            best_h, best_c = up_h, up_c

    # Final polishing with SLSQP
    if time.time() + 5 < time_end:
        for beta in (2000.0, 5000.0, 20000.0, 50000.0):
            if time.time() > time_end - 2:
                break
            h_pol = slsqp_refine(best_h, best_h.size / 2.0, rng, beta, max_iter=300)
            c_pol = compute_c5(h_pol)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
