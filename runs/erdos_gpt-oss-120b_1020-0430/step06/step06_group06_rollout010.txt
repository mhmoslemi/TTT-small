import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple deterministic correction for tiny drift
    if abs(diff) < 1e-12:
        return h
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k (max over k)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # fallback opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Either swap two entries or transfer a random amount while preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.2:  # pure swap
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- SLSQP with soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- gradient descent on worst shift ----------
def gradient_descent_optimize(h: np.ndarray, rng: np.random.Generator,
                             deadline: float,
                             init_step: float = 0.5,
                             decay: float = 0.9995,
                             min_step: float = 1e-8,
                             max_iter: int = 200000):
    """Projected gradient descent on max‑overlap (using worst‑shift gradient)."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = init_step
    it = 0
    n = h.size
    while time.time() < deadline and step > min_step and it < max_iter:
        g = worst_gradient(h)
        d_raw = -g                      # direction of descent
        d = d_raw - d_raw.mean()        # project to sum‑zero hyperplane
        if np.allclose(d, 0, atol=1e-12):
            break
        pos = d > 0
        neg = d < 0
        # maximal feasible step respecting box constraints
        max_pos = np.min((1.0 - h[pos]) / d[pos]) if np.any(pos) else np.inf
        max_neg = np.min(h[neg] / -d[neg]) if np.any(neg) else np.inf
        max_step = min(max_pos, max_neg)
        actual = step if max_step >= step else max_step
        if actual < min_step:
            break
        h = h + actual * d
        # numeric safety
        h = np.clip(h, 0.0, 1.0)
        if it % 500 == 0:
            h = enforce_sum(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        step *= decay
        it += 1
    return best_h, best_c


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n = 512
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
        h0 = enforce_sum(h0, rng)
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- simulated annealing (≈30% of budget) -----
    SA_deadline = start + budget_s * 0.30
    current_h = best_h.copy()
    current_c = best_c
    T = 0.001
    eps_scale = 0.5
    it = 0
    while time.time() < SA_deadline:
        if rng.random() < 0.4:
            nh = guided_perturb(current_h,
                                 worst_gradient(current_h), rng,
                                 scale=eps_scale)
        else:
            nh = random_move(current_h, rng, scale=eps_scale)
        nc = compute_c5(nh)
        delta = nc - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_h, current_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        it += 1
        if it % 1000 == 0:
            T *= 0.995
            eps_scale *= 0.997
            T = max(T, 1e-12)
            eps_scale = max(eps_scale, 1e-5)

    # ----- SLSQP soft‑max passes (≈30% of budget) -----
    SLSQP_deadline = start + budget_s * 0.60
    beta_seq = (50.0, 500.0, 5000.0, 20000.0)
    for beta in beta_seq:
        if time.time() > SLSQP_deadline:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)
        best_c = compute_c5(best_h)

    # ----- gradient descent on worst shift (remaining budget) -----
    grad_deadline = start + budget_s * 0.98
    best_h, best_c = gradient_descent_optimize(best_h, rng, grad_deadline)

    # final repair
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
