def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # choose n
    n = 1024
    target = n / 2.0
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        # upsample to n if needed
        if init.size != n:
            # repeat or interpolate
            factor = n // init.size
            if n % init.size == 0:
                h0 = np.repeat(init, factor)
            else:
                # linear interpolation
                xs = np.linspace(0, 1, init.size, endpoint=False)
                xs_new = np.linspace(0, 1, n, endpoint=False)
                h0 = np.interp(xs_new, xs, init)
        else:
            h0 = init.copy()
        # ensure constraints
        h0 = np.clip(h0, 0.0, 1.0)
        # scale sum
        h0 = project_onto_box_sum(h0, target)
    else:
        # binary random
        h0 = np.zeros(n)
        h0[rng.choice(n, n//2, replace=False)] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: softmax gradient descent at various beta
    beta_list = [20, 100, 500, 2000, 8000, 20000]
    step0_map = {20:0.6, 100:0.5, 500:0.4, 2000:0.3, 8000:0.25, 20000:0.2}
    for beta in beta_list:
        if time.time() - start > budget_s * 0.70:
            break
        iters = int(1200 * 256 / n)  # bigger than previous
        if beta >= 2000:
            iters = int(1800 * 256 / n)
        step0 = step0_map[beta]
        h = best_h.copy()
        for it in range(iters):
            if time.time() - start > budget_s * 0.70:
                break
            grad = softmax_gradient(h, beta)
            norm = np.linalg.norm(grad)
            if norm == 0:
                continue
            step = step0 * (1.0 - it/iters) / norm
            v = h - step * grad
            h = project_onto_box_sum(v, target)
        c = compute_c5(h)
        if c < best_c - 1e-9:
            best_h, best_c = h.copy(), c
        # after each beta, try guided hill climb
        if time.time() - start < budget_s * 0.75:
            hc, cc = guided_hill_climb(best_h, rng, max_iters=int(800 * 256 / n), step_scale=0.5)
            if cc < best_c:
                best_h, best_c = hc.copy(), cc
    # Phase 2: more hill climbing with random restarts
    while time.time() - start < budget_s * 0.90:
        # random restart: perturb best a bit to escape local minima
        h = best_h.copy()
        # apply a few random swaps
        for _ in range(5):
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * 0.3
            h[i] -= eps
            h[j] += eps
        h = project_onto_box_sum(h, target)
        # hill climb
        hc, cc = guided_hill_climb(h, rng, max_iters=int(1000 * 256 / n), step_scale=0.5)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc
    # Phase 3: SLSQP refinement
    for beta in (2000, 8000, 20000, 50000):
        if time.time() - start > budget_s * 0.95:
            break
        h = slsqp_optimize(best_h, beta, rng, max_iter=800)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    # Phase 4: final fine perturbation
    while time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 for discretized step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target, max_iter=60):
    """Project v onto the box [0,1] while enforcing sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    """Gradient of the worst overlap term O_k."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    """Move a small amount of mass from max‑gradient to min‑gradient index."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    """Simple hill‑climbing using guided perturbations."""
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def softmax_gradient(h, beta):
    """Smooth approximation of the max‑gradient."""
    n = h.shape[0]
    O = np.correlate(h, 1.0 - h, mode="full")
    Omax = O.max()
    w = np.exp(beta * (O - Omax))
    w /= w.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP on the soft‑max objective."""
    n = h0.shape[0]
    target = n / 2.0

    def soft_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def eq_con(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_con}
    res = minimize(
        soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Number of discretisation points – larger gives a finer bound.
    n = 1024
    target = n / 2.0

    # Initialise h (upsample if a parent construction is available).
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            if n % init.size == 0:
                factor = n // init.size
                h0 = np.repeat(init, factor)
            else:
                xs_old = np.linspace(0, 2, init.size, endpoint=False)
                xs_new = np.linspace(0, 2, n, endpoint=False)
                h0 = np.interp(xs_new, xs_old, init)
        else:
            h0 = init.copy()
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
    else:
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- Phase 1: soft‑max gradient descent with decreasing step -----
    t1 = start + budget_s * 0.55
    beta_list = [20, 100, 500, 2000, 8000, 20000]
    step0_map = {20: 0.6, 100: 0.5, 500: 0.4, 2000: 0.3, 8000: 0.25, 20000: 0.2}
    for beta in beta_list:
        if time.time() > t1:
            break
        # iterations scale with n to keep total work roughly constant
        iters = int(2000 * 256 / n)
        if beta >= 2000:
            iters = int(3000 * 256 / n)
        step0 = step0_map[beta]
        h = best_h.copy()
        for it in range(iters):
            if time.time() > t1:
                break
            grad = softmax_gradient(h, beta)
            norm = np.linalg.norm(grad)
            if norm == 0:
                continue
            step = step0 * (1.0 - it / iters) / norm
            v = h - step * grad
            h = project_onto_box_sum(v, target)
        c = compute_c5(h)
        if c < best_c - 1e-9:
            best_h, best_c = h.copy(), c

        # small hill‑climb after each beta
        t_hc = start + budget_s * 0.75
        if time.time() < t_hc:
            hc, cc = guided_hill_climb(
                best_h, rng, max_iters=int(800 * 256 / n), step_scale=0.5
            )
            if cc < best_c:
                best_h, best_c = hc.copy(), cc

    # ----- Phase 2: random restarts + hill‑climb -----
    t2 = start + budget_s * 0.85
    while time.time() < t2:
        h = best_h.copy()
        # a few random mass swaps to leave the basin
        for _ in range(5):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * 0.3
            h[i] -= eps
            h[j] += eps
        h = project_onto_box_sum(h, target)
        # hill‑climb from this perturbed start
        hc, cc = guided_hill_climb(
            h, rng, max_iters=int(1200 * 256 / n), step_scale=0.5
        )
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    # ----- Phase 3: SLSQP refinements -----
    t3 = start + budget_s * 0.95
    for beta in (2000, 8000, 20000, 50000):
        if time.time() > t3:
            break
        h = slsqp_optimize(best_h, beta, rng, max_iter=300)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ----- Phase 4: final fine‑grained perturbations -----
    t_end = start + budget_s * 0.999
    while time.time() < t_end:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
