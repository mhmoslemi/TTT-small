```
def compute_delta_auto(h, i, j):
    n = h.shape[0]
    ks = np.arange(-n+1, n)
    pad = np.pad(h, (n-1, n-1), 'constant')
```

```
def enforce_sum_fast(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # add diff/n uniformly
    h = h + diff / h.size
    # clip
    h = np.clip(h, 0.0, 1.0)
    # recompute diff
    diff = target - h.sum()
    # if still off due to clipping, do random redistribution
    if abs(diff) > 1e-12:
        # use previous random redistribution
        rng = np.random.default_rng()
        # as fallback
        while abs(diff) > 1e-12:
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h
```

    return x

# ----- main entry point -----
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    n = 1024
    target_sum = n / 2.0

    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, h0)
    else:
        h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum_fast(h, target_sum, rng)

    best_h = h.copy()
    best_c = compute_c5(h)

    # time allocation
    slsq_frac = 0.30
    slsq_time = total_budget * slsq_frac
    adam_time = total_budget - slsq_time - 5.0
    adam_end = start + adam_time

    # Adam parameters
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t_iter = 0
    restart_interval = 20000

    while time.time() < adam_end:
        # correlation via FFT
        Nfft = 2 * n
        A = np.fft.rfft(h, n=Nfft)
        B = np.fft.rfft(1.0 - h, n=Nfft)
        corr = np.fft.irfft(A * B.conj(), n=Nfft)[: 2 * n - 1]
        idx = np.argmax(corr)
        k_abs = abs(idx - (n - 1))
        grad = worst_gradient_by_k(h, k_abs)

        t_iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h = h - step
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_fast(h, target_sum, rng)

        if t_iter % 200 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

        if restart_interval and t_iter % restart_interval == 0:
            m.fill(0.0)
            v.fill(0.0)

    # SLSQP refinement (if time permits)
    if time.time() + 5.0 < start + total_budget:
        beta_seq = (20., 80., 200., 800., 3000., 8000., 20000.)
        for beta in beta_seq:
            if time.time() + 5.0 >= start + total_budget:
                break
            h = slsqp_optimize(best_h, beta, rng, max_iter=1000)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_fast(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
