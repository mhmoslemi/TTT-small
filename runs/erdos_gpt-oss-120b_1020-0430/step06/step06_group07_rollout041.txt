def delta_O_move(h, i, j, delta):
    n = h.shape[0]
    s = np.arange(-n+1, n)
    i_plus = i + s
    i_minus = i - s
    j_plus = j + s
    j_minus = j - s

    # valid_i: 0 <= i_plus < n
    valid_i = (i_plus >= 0) & (i_plus < n)
    valid_j = (j_plus >= 0) & (j_plus < n)

    # second_i: 0 <= i_minus < n
    second_i = (i_minus >= 0) & (i_minus < n)
    second_j = (j_minus >= 0) & (j_minus < n)

    # gather h at needed indices
    # Use np.where to set zero where invalid
    h_i_plus = np.where(valid_i, h[i_plus], 0.0)
    h_i_minus = np.where(second_i, h[i_minus], 0.0)
    h_j_plus = np.where(valid_j, h[j_plus], 0.0)
    h_j_minus = np.where(second_j, h[j_minus], 0.0)

               delta * (h_j_plus * valid_j.astype(np.float64) + h_j_minus * second_j.astype(np.float64)))
    return delta_O

def anneal_optimize(h0, time_budget, rng):
    h = h0.copy()
    n = h.size
    O = compute_O(h)
    cur_max = O.max()
    best_h = h.copy()
    best_max = cur_max
    offset = n-1
    L = 2*n-1
    it = 0
    start = time.time()
    # schedule parameters
    max_iters = int(1e7)  # large bound
    initial_temp = cur_max * 2.0 + 1.0  # ensure > cur_max
    final_temp = 0.01
    while time.time() - start < time_budget:
        # compute temp
        # using it / max_iters fraction; if it > max_iters, treat fraction = 1
        frac = it / max_iters
        if frac > 1.0:
            frac = 1.0
        temp = initial_temp * (final_temp / initial_temp) ** frac

        # random move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            it += 1
            continue

        scale = temp / cur_max
        if scale > 1.0:
            scale = 1.0
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            it += 1
            continue

        # compute delta_O via slices
        delta_O = np.zeros(L, dtype=np.float64)
        # h_rev for current h
        h_rev = h[::-1]

        # i valid slice
        start_i_valid = -i + offset
        end_i_valid = (n - 1 - i) + offset
        delta_O[start_i_valid:end_i_valid+1] += delta * (h - 1.0)

        # i second slice
        start_i_second = i - (n - 1) + offset
        end_i_second = i + offset
        delta_O[start_i_second:end_i_second+1] += delta * h_rev

        # j valid slice
        start_j_valid = -j + offset
        end_j_valid = (n - 1 - j) + offset
        delta_O[start_j_valid:end_j_valid+1] += delta * (1.0 - h)

        # j second slice
        start_j_second = j - (n - 1) + offset
        end_j_second = j + offset
        delta_O[start_j_second:end_j_second+1] -= delta * h_rev

        new_O = O + delta_O
        new_max = new_O.max()

        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            # metropolis acceptance
            if rng.random() < np.exp(-(new_max - cur_max) / temp):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            O = new_O
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                # early stop if target reached
                if best_max * (2.0 / n) <= 0.3808:
                    break

        # periodic recompute every 10000 iterations for safety
        if it % 10000 == 0:
            O = compute_O(h)
            cur_max = O.max()
        it += 1

    return best_h, best_max

```python
import time, numpy as np

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo+hi)*0.5
        proj = np.clip(v+lam, 0.0, 1.0)
        if proj.sum() > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)*0.5, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n/2.0)

def compute_O(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    O = compute_O(h)
    return O.max() * (2.0 / h.size)

def anneal_optimize(h0, time_budget, rng):
    h = h0.copy()
    n = h.size
    O = compute_O(h)
    cur_max = O.max()
    best_h = h.copy()
    best_max = cur_max
    offset = n-1
    L = 2*n-1
    start = time.time()
    max_iters = int(5e6)
    init_temp = cur_max*3.0 + 1.0
    final_temp = 0.01
    it = 0
    last_improve = 0
    while time.time() - start < time_budget:
        # temperature
        frac = it / max_iters
        if frac > 1.0:
            frac = 1.0
        temp = init_temp * (final_temp / init_temp) ** frac
        # pick i,j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            it += 1
            continue
        scale = temp / cur_max
        if scale > 1.0:
            scale = 1.0
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            it += 1
            continue
        # compute delta_O via slices
        delta_O = np.zeros(L, dtype=np.float64)
        h_rev = h[::-1]
        start_i_valid = -i + offset
        end_i_valid = (n-1-i) + offset
        delta_O[start_i_valid:end_i_valid+1] += delta * (h - 1.0)
        start_i_second = i - (n-1) + offset
        end_i_second = i + offset
        delta_O[start_i_second:end_i_second+1] += delta * h_rev
        start_j_valid = -j + offset
        end_j_valid = (n-1-j) + offset
        delta_O[start_j_valid:end_j_valid+1] += delta * (1.0 - h)
        start_j_second = j - (n-1) + offset
        end_j_second = j + offset
        delta_O[start_j_second:end_j_second+1] -= delta * h_rev
        cross_idx = (j - i) + offset
        delta_O[cross_idx] += delta * delta
        new_O = O + delta_O
        new_max = new_O.max()
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / temp):
                accept = True
        if accept:
            h[i] -= delta
            h[j] += delta
            O = new_O
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                last_improve = it
                if best_max * (2.0 / n) <= 0.3808:
                    break
        # occasional full recompute
        if it % 20000 == 0:
            O = compute_O(h)
            cur_max = O.max()
        # restart if stalled
        if it - last_improve > 200000:
            # perturb best_h slightly
            h = best_h.copy()
            noise = rng.normal(scale=0.02, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h = project_box_sum(h, n/2.0)
            O = compute_O(h)
            cur_max = O.max()
            last_improve = it
        it += 1
    return best_h, best_max

def upsample(h, rng):
    n = h.size
    n2 = n*2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2/2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n/2.0)
    else:
        n = 512
        h0 = random_feasible(n, rng)
    # phase 1 coarse anneal
    remaining = budget_s - (time.time() - start)
    t1 = remaining * 0.65
    best_h, best_max = anneal_optimize(h0, t1, rng)
    # phase 2 upsample if time permits
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0 and best_h.size * 2 <= 1024:
        h_up = upsample(best_h, rng)
        best_h, best_max = anneal_optimize(h_up, remaining, rng)
    # final sanitise
    n_final = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n_final/2.0)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(n_final)

import time, numpy as np

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        if proj.sum() > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def compute_O(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    O = compute_O(h)
    return O.max() * (2.0 / h.size)

    h = h0.copy()

    best_max = cur_max

    max_iters = int(5e6)

    final_temp = 0.01

    while time.time() - start < time_budget:
        frac = it / max_iters
        if frac > 1.0:
            frac = 1.0
        temp = init_temp * (final_temp / init_temp) ** frac
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            it += 1
            continue
        scale = temp / cur_max
        if scale > 1.0:
            scale = 1.0
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            it += 1

        delta_O = np.zeros(L, dtype=np.float64)

def upsample(h, rng):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)
