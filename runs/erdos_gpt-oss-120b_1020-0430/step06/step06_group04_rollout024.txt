import time
import numpy as np

# ------- basic utilities -------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for discrete h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Project h onto {0≤h_i≤1, sum h_i = n/2} by greedy fixes."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ------- smooth‑max gradient --------------------------------------------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)·log∑exp(β·v) where v are the overlap values."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:] -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:] += wi * one_minus_h[: n - s]
            g[: n - s] -= wi * h[s:]
    return g

# ------- worst‑shift sub‑gradient ----------------------------------------
def worst_shift_gradient(h):
    """Sub‑gradient for the currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

# ------- worst‑shift refinement -----------------------------------------
def worst_shift_refine(h, rng, base_step, iters, deadline):
    """Projected sub‑gradient steps on the true max‑overlap."""
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.95
    return h

# ------- continuous simulated annealing ---------------------------------
def continuous_sa(h, rng, deadline, start_temp_factor=0.02):
    """Metropolis‑type annealing on the true max‑overlap (mass moves)."""
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    start_T = max(cur_c * start_temp_factor, 1e-8)
    end_T = 1e-12
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        T = start_T * ((end_T / start_T) ** (elapsed / total))
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # decide direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 0:
            continue
        eps = max_delta * min(0.5, T * 5.0)
        delta = rng.random() * eps
        cur_h[src] -= delta
        cur_h[dst] += delta
        new_c = compute_c5(cur_h)
        if new_c < cur_c:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12))
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[src] += delta
            cur_h[dst] -= delta
    return best_h, best_c
