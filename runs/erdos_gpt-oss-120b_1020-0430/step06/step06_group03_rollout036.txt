def weighted_gradient(h, beta):
    n = h.size
    # compute O raw via FFT (fast)
    O = corr_full_fft(h)  # length 2n-1, same as np.correlate
    max_O = O.max()
    exp_term = np.exp(beta * (O - max_O))
    w = exp_term / exp_term.sum()   # length 2n-1
    # compute convolution of h with w via FFT (fast)
    # Convert w to time domain? Actually w is already in shift domain, we want convolution of h and w as sequences.
    # Use FFT: conv = np.fft.irfft(np.fft.rfft(h, m) * np.fft.rfft(w, m), n=m) where m = next_power_of_two(2*n-1)
    m = 1 << ((2*n-1)-1).bit_length()
    h_fft = np.fft.rfft(h, n=m)
    w_fft = np.fft.rfft(w, n=m)
    conv = np.fft.irfft(h_fft * w_fft, n=m)[:2*n-1]  # keep length 2n-1
    s = conv[n-1:]   # length n, corresponds to sum_k w_k * h_{j - k}
    # compute g1 via prefix sums
    prefix_w = np.cumsum(w)
    # indices for each j
    idx = 2*n - 1 - np.arange(n) - 1  # 2n-2 - j
    g1 = prefix_w[idx]
    grad = g1 - 2.0 * s
    return grad

import time, numpy as np

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5_fft(h):
    n = h.size
    m = 1 << ((2 * n - 1) - 1).bit_length()
    hf = np.fft.rfft(h, n=m)
    h1f = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(hf * h1f, n=m)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def corr_full_fft(h):
    n = h.size
    m = 1 << ((2 * n - 1) - 1).bit_length()
    hf = np.fft.rfft(h, n=m)
    h1f = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(hf * h1f, n=m)[: 2 * n - 1]
    return corr

def weighted_gradient(h, beta):
    """gradient of the softmax‑smoothed max overlap"""
    n = h.size
    O = corr_full_fft(h)               # raw correlation (no dx)
    M = O.max()
    w = np.exp(beta * (O - M))
    w /= w.sum()                       # length 2n‑1, indices k = -(n-1)..(n-1)

    # convolution h * w (full)
    m_len = 1 << ((2 * n - 1) - 1).bit_length()
    hf = np.fft.rfft(h, n=m_len)
    wf = np.fft.rfft(w, n=m_len)
    conv = np.fft.irfft(hf * wf, n=m_len)[: 2 * n - 1]

    s = conv[n - 1 :]                  # sum_k w_k * h_{j-k}
    prefix_w = np.cumsum(w)
    idx = 2 * n - 2 - np.arange(n)     # 2n-2 - j
    g1 = prefix_w[idx]                 # sum_{k: j+k<N} w_k
    grad = g1 - 2.0 * s
    return grad

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.5):
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    step = step_scale
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
        if time.time() - start > budget_s * 0.999:
            break
    return best_h, best_c

    target = n / 2.0
    h = np.clip(h0, 0.0, 1.0)
    h = project_onto_box_sum(h, target)

    best_h = h.copy()
    best_c = compute_c5_fft(best_h)

    # ---- Adam optimiser on smoothed max -------------------------------
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
