def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # while loop but with cap
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        if diff > 0:
            idx = rng.choice(np.where(h < 1 - 1e-12)[0])
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.choice(np.where(h > 1e-12)[0])
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        it += 1
    return h

def guided_perturb(h, grad, rng, scale):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    # Transfer epsilon from i (positive grad) to j (negative grad)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def gradient_polish(h, rng, step=0.02, iterations=2000):
    n = h.size
    for it in range(iterations):
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1 - 2 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1 - h[k:]
            g[k:] -= h[: n - k]
        # step schedule: step * (0.99**it) maybe
        cur_step = step * (0.995 ** it)
        h = h - cur_step * g
        # project to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            # simple uniform distribution among non-bound entries
            # We'll just use enforce_sum
            h = enforce_sum(h, rng)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP soft‑max optimisation
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Stable soft‑max of the correlation (max‑approximation)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Refine h using SLSQP on the soft‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5(h):
    """Discrete C₅ for a step vector h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, rng):
    """Adjust h so that Σh = N/2 while keeping 0 ≤ h ≤ 1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    # simple random adjustments
    while abs(diff) > 1e-12 and it < 20000:
        if diff > 0:
            idx = rng.choice(np.where(h < 1 - 1e-12)[0])
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.choice(np.where(h > 1e-12)[0])
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        it += 1
    return h


def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros_like(h)
    g[: n - k] += 1.0 - h[k:]
    g[k:] -= h[: n - k]
    return g


def guided_perturb(h, grad, rng, scale=0.5):
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_polish(h, rng, step=0.02, iterations=5000, deadline=None):
    """Many tiny gradient steps on the worst shift."""
    n = h.size
    for it in range(iterations):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        cur_step = step * (0.995 ** it)
        h = h - cur_step * g
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum occasionally
        if (it + 1) % 500 == 0:
            h = enforce_sum(h, rng)
    return h
