import time, numpy as np

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def project_h(x: np.ndarray, s: float) -> np.ndarray:
    low = x.min() - 1.0
    high = x.max()
    for _ in range(30):
        mid = (low + high) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > s:
            low = mid
        else:
            high = mid
    return np.clip(x - (low + high) * 0.5, 0.0, 1.0)

def sym_weights(corr: np.ndarray, beta: float) -> np.ndarray:
    bc = beta * corr
    bc_max = bc.max()
    expv = np.exp(bc - bc_max)
    expv /= expv.sum()
    npt = (len(corr) + 1) // 2
    cen = npt - 1
    w = expv[cen:cen + npt].copy()
    rev = expv[cen - 1 :: -1]
    w[1:] += rev[: npt - 1]
    return w

def smooth_grad(w: np.ndarray) -> np.ndarray:
    cum = np.cumsum(w)
    return 1.0 - cum - cum[::-1]

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

        beta = betas[beta_idx]

        h = h - lr * g
