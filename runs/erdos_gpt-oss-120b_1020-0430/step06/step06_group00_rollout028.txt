import time, numpy as np, math
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h

def upsample_h(h, new_n):
    old_n = h.size
    x_old = np.linspace(0., 2., old_n, endpoint=False)
    x_new = np.linspace(0., 2., new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def _obj_grad(h, beta, lam):
    n = h.size
    target = n / 2.0
    # correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    maxc = np.max(corr)
    w = np.exp(beta * (corr - maxc))
    w_sum = w.sum()
    w_norm = w / w_sum
    # gradients
    # term1: conv(w_norm, (1 - h)[::-1]); need slice N-1 - i
    conv1 = np.convolve(w_norm, (1.0 - h)[::-1])
    term1 = conv1[n-1 - np.arange(n)]
    conv2 = np.convolve(w_norm, h)
    term2 = conv2[np.arange(n) + n - 1]
    grad = term1 - term2
    # penalty gradient
    diff = h.sum() - target
    grad += 2.0 * lam * diff
    # objective
    phi = (maxc + np.log(w_sum)) / beta + lam * diff * diff
    return phi, grad

def lbfgsb_optimize(h0, beta, lam, rng, maxiter=200):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    # wrapper for minimize
    def fun(x):
        phi, grad = _obj_grad(x, beta, lam)
        return phi, grad
    # minimize
    res = minimize(fun, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # final sum repair
    x = enforce_sum(x, rng)
    return x

def final_refine_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    best = h.copy()
    best_c = compute_c5(h)
    N = h.size
    T0, T_end = 0.002, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        # pick another index j distinct
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            # move mass from i to j
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best = h_new.copy()
        # accept with Metropolis
        c_curr = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # multi-resolution schedule: Ns = [256, 512, 1024] (or 128,256,...)
    Ns = [256, 512, 1024]
    beta_seq = [20., 80., 200., 800., 3000., 8000., 20000.]
    lam = 1e4
    for N in Ns:
        if time.time() - start >= total_budget * 0.95:
            break
        # allocate some budget proportionally
        remaining = total_budget - (time.time() - start)
        # allocate half of remaining to this resolution
        alloc = remaining * 0.5
        # upsample best to N if needed
        if best_h is None or best_n != N:
            if best_h is not None:
                h = upsample_h(best_h, N)
            else:
                h = rng.random(N)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = best_h.copy()
        # run L-BFGS-B for each beta
        for beta in beta_seq:
            if time.time() - start >= total_budget * 0.95:
                break
            maxiter = 250
            # we may reduce maxiter for early betas
            if beta < 200:
                maxiter = 150
            h = lbfgsb_optimize(h, beta, lam, rng, maxiter=maxiter)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, N
                if best_c <= 0.3808:
                    # early exit
                    best_h = enforce_sum(best_h, rng)
                    final_c = compute_c5(best_h)
                    return best_h.tolist(), float(final_c), int(best_h.size)
        # after beta loop, maybe run SA for some time
        remaining = total_budget - (time.time() - start)
        if remaining > 0:
            t_sa = min(remaining * 0.2, 30.0)
            h_sa, c_sa = final_refine_sa(h, rng, t_sa)
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, N

    # final sanitization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time, math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global parameters used by the optimizer wrapper (must be top‑level)
CURRENT_BETA = None
CURRENT_LAMBDA = None

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ----------------------------------------------------------------------
def _obj_grad(h: np.ndarray, beta: float, lam: float):
    """Objective (soft‑max + penalty) and its gradient."""
    n = h.size
    target = n / 2.0
    # correlation h with 1‑h
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    # soft‑max weights (stable)
    w = np.exp(beta * (corr - maxc))
    w_sum = w.sum()
    w_norm = w / w_sum
    # weighted gradient of all shifts
    conv1 = np.convolve(w_norm, (1.0 - h)[::-1])
    term1 = conv1[n - 1 - np.arange(n)]
    conv2 = np.convolve(w_norm, h)
    term2 = conv2[np.arange(n) + n - 1]
    grad = term1 - term2
    # penalty gradient (sum constraint)
    diff = h.sum() - target
    grad += 2.0 * lam * diff
    # objective value
    phi = maxc + np.log(w_sum) / beta + lam * diff * diff
    return phi, grad


def _obj_grad_wrapper(x):
    """Wrapper required by scipy (uses globals)."""
    return _obj_grad(x, CURRENT_BETA, CURRENT_LAMBDA)


def lbfgsb_optimize(h0: np.ndarray, beta: float, lam: float,
                    rng: np.random.Generator, maxiter: int = 250) -> np.ndarray:
    """L‑BFGS‑B minimisation of the soft‑max approximation."""
    global CURRENT_BETA, CURRENT_LAMBDA
    CURRENT_BETA = beta
    CURRENT_LAMBDA = lam
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_grad_wrapper, h0, method="L-BFGS-B",
                    jac=True, bounds=bounds,
                    options={"maxiter": maxiter, "ftol": 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    return x


# ----------------------------------------------------------------------
def refine_worst(h0: np.ndarray, rng: np.random.Generator,
                 time_budget: float) -> tuple[np.ndarray, float]:
    """Gradient descent on the worst shift for the remaining time."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    n = h.size
    step0 = 0.02
    while time.time() - start < time_budget:
        # decaying step (5 halvings over whole budget)
        elapsed = time.time() - start
        step = step0 * (0.5 ** (elapsed / time_budget * 5.0))
        # worst shift gradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros(n)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new < c_curr:
            h, c_curr = h_new, c_new
    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Initial solution (parent or random)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        h0 = rng.random(256)
        h0 = enforce_sum(h0, rng)
    best_h, best_c, best_n = h0, compute_c5(h0), h0.size

    # ------------------------------------------------------------------
    # Multi‑resolution optimisation
    Ns = [256, 512, 1024]
    beta_seq = [5., 20., 80., 200., 800., 3000., 8000., 20000., 50000.]
    lam = 5e4

    for N in Ns:
        if time.time() - start > total_budget * 0.95:
            break
        # bring current best to this resolution
        if best_n != N:
            h = upsample_h(best_h, N)
        else:
            h = best_h.copy()
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

        # run L‑BFGS‑B for each β
        for beta in beta_seq:
            if time.time() - start > total_budget * 0.95:
                break
            if beta <= 200:
                iters = 500
            elif beta <= 8000:
                iters = 300
            else:
                iters = 200
            h = lbfgsb_optimize(h, beta, lam, rng, maxiter=iters)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, N
                if best_c <= 0.3808:
                    best_h = enforce_sum(best_h, rng)
                    return best_h.tolist(), float(compute_c5(best_h)), int(best_h.size)

        # after β‑sweep, optional local refinement (worst‑shift)
        remaining = total_budget - (time.time() - start)
        if remaining > 5.0:
            t_ref = min(remaining * 0.10, 30.0)
            h_ref, c_ref = refine_worst(h, rng, t_ref)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref.copy(), c_ref, N

    # ------------------------------------------------------------------
    # Final refinement with all remaining time
    remaining = total_budget - (time.time() - start)
    if remaining > 1.0:
        best_h, best_c = refine_worst(best_h, rng, remaining)

    # ------------------------------------------------------------------
    # Sanitise and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
