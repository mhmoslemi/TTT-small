def guided_hill_climb(h, rng, time_budget, start_time, max_iters=100000, step_scale=0.7):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    decay_interval = max(1, max_iters // 5)
    while it < max_iters and time.time() - start_time < time_budget:
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        it += 1
        if it % decay_interval == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c, h
```

def softmax_obj(x):

    nfft = 1 << ((2*n-1).bit_length())
    fft_x = np.fft.rfft(x, n=nfft)
    fft_rev = np.fft.rfft((1.0 - x)[::-1], n=nfft)

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

import time
import numpy as np
from scipy.optimize import minimize

# ---------- correlation utilities ----------
def _fft_len(n):
    return 1 << ((2 * n - 1).bit_length())

def compute_corr(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    nfft = _fft_len(n)
    fh = np.fft.rfft(h, n=nfft)
    fv = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(fh * np.conj(fv), n=nfft)[: 2 * n - 1]
    return corr

def compute_c5(h: np.ndarray) -> float:
    """Fast C5 using FFT (max correlation scaled to integral)."""
    n = h.shape[0]
    return np.max(compute_corr(h)) * (2.0 / n)

def compute_c5_direct(h: np.ndarray) -> float:
    """Exact C5 using NumPy correlate (used for final reporting)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

# ---------- feasibility ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    max_iter = 10000
    i = 0
    while abs(diff) > 1e-12 and i < max_iter:
        idx = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
                diff -= add
        else:
            sub = min(-diff, h[idx])
            if sub > 0:
                h[idx] -= sub
                diff += sub
        i += 1
    return h

# ---------- gradient of worst shift ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of C5 w.r.t. h for the shift that attains the maximum."""
    n = h.size
    corr = compute_corr(h)
    idx = np.argmax(corr) - (n - 1)      # shift k
    k = idx
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        m = -k
        grad[m:] += 1.0 - h[: n - m]
        grad[: n - m] -= h[m:]
    return grad

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      time_budget: float, start_time: float,
                      max_iters: int = 200000, step_scale: float = 0.6):

    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay_int = max(1, max_iters // 5)

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            eps_max = min(h[i], 1.0 - h[j])

        it += 1

CURRENT_TARGET = None
CURRENT_BETA = None

    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
