def project_onto_capped_simplex(v, target):
    lo = np.min(v) - 1.0  # ensure sum > target
    hi = np.max(v)  # ensures sum < target
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # due to binary search tolerance, sum may be slightly off
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust by adding diff to any non-bound elements
        # find indices where 0<h<1
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size > 0:
            # distribute diff evenly
            add = diff / idx.size
            h[idx] += add
            # clip again
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h0, rng, time_budget):

    N = h0.size
    target = N / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    # step schedule
    step0 = 1.0  # initial step size (maybe need to tune)
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find all indices achieving max within tolerance
        # tolerance relative: say 1e-12 absolute
        tol = 1e-12
        max_idxs = np.where(np.abs(corr - max_corr) <= tol)[0]
        # compute gradient sum
        grad = np.zeros_like(h)
        for idx in max_idxs:
            k = abs(idx - (N - 1))
            if k == 0:
                grad += 1.0 - 2.0 * h
            else:
                grad[: N - k] += 1.0 - h[k:]
                grad[k:] -= h[: N - k]
        # average gradient over number of max shifts
        grad /= max_idxs.size
        # step size schedule: step = step0 / np.sqrt(it)
        step = step0 / np.sqrt(it)
        # update v = h - step * grad
        v = h - step * grad
        # project onto feasible set
        h = project_onto_capped_simplex(v, target)
        # evaluate
        f = compute_c5(h)
        if f < best_f:
            best_f = f
            best_h = h.copy()
            # optional: early break if target reached
            if best_f <= 0.3808:
                break
    return best_h, best_f

def continuous_sa(h0, rng, time_budget):
    start = time.time()
    N = h0.size
    target = N / 2.0
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    # schedule: T0, T_end
    T0, T_end = 0.01, 1e-6  # maybe slower cooling
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)  # scaled magnitude for epsilon
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # random direction
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def binary_sa(N, rng, time_budget):
    start = time.time()
    # random binary vector exact N/2 ones
    h = np.zeros(N, dtype=float)
    ones_idx = rng.choice(N, N // 2, replace=False)
    h[ones_idx] = 1.0
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    # maintain list or array of ones and zeros
    ones_arr = ones_idx.copy()
    zeros_arr = np.setdiff1d(np.arange(N), ones_arr)
    # Temperature schedule
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:

        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # choose random swap
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]

        h[j] = 1.0

def upsample_h(h, new_n):
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection onto {0≤x≤1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(35):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size == 0:
            h += diff / h.size
        else:
            h[idx] += diff / idx.size
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(N: int, rng: np.random.Generator, time_budget: float):
    """Binary simulated annealing (exact N/2 ones) on a coarse grid."""
    start = time.time()
    h = np.zeros(N, dtype=float)
    ones_arr = rng.choice(N, N // 2, replace=False)
    h[ones_arr] = 1.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    zeros_arr = np.setdiff1d(np.arange(N), ones_arr)
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i, j = ones_arr[i_pos], zeros_arr[j_pos]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# ------------------------------------------------------------
def subgradient_descent(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Sub‑gradient descent on max‑overlap with projection onto the feasible set."""
    start = time.time()
    N = h0.size
    target = N / 2.0
    h = h0.copy()
    best_h, best_f = h.copy(), compute_c5(h)
    step0 = 1.0
    it = 0
    while time.time() - start < time_budget:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        # indices attaining the maximum (within tiny tolerance)
        idxs = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            k = abs(idx - (N - 1))
            if k == 0:
                grad += 1.0 - 2.0 * h
            else:
                grad[: N - k] += 1.0 - h[k:]
                grad[k:] -= h[: N - k]
        grad /= idxs.size
        step = step0 / np.sqrt(it)
        v = h - step * grad
        h = project_onto_capped_simplex(v, target)
        f = compute_c5(h)
        if f < best_f:
            best_f, best_h = f, h.copy()
            if best_f <= 0.3808:
                break
    return best_h, best_f

def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Continuous SA on the feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    N = h0.size
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.01, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    return best_h, best_c
