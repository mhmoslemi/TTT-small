import time
import numpy as np

# ----- global variables for SLSQP ----- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max of overlap using CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick for stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def slsqp_refine(h, target, rng, beta, max_iter):
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize  # local import

    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

        delta = compute_delta_O(i, j, h, n, center)
        O_new = O + delta
        cur_max = int(O_new.max())
        if cur_max < best_max:
            h[i] = 0
            h[j] = 1
            O = O_new
            best_h = h.copy()
            best_max = cur_max
            continue
        # occasional uphill move to escape local minima
        if rng.random() < 0.001:
            h[i] = 0
            h[j] = 1
            O = O_new

    # ----- final continuous refinement (SLSQP) if time permits ----- #
    if time.time() - start < budget_s * 0.94:
        h_float = best_h.astype(float)
        for beta in (500.0, 2000.0, 5000.0):
            if time.time() - start > budget_s * 0.96:
                break
            h_float = slsqp_refine(h_float, target_ones, rng, beta=beta, max_iter=200)
            # keep track of the best integer‑rounded version
            cand = np.clip(h_float, 0.0, 1.0)
            cand = enforce_sum(cand, target_ones, rng)
            c_val = compute_c5(cand)
            if c_val < compute_c5(best_h.astype(float)):
                best_h = cand.astype(np.int8)

    # ----- final result ----- #
    best_h = np.clip(best_h, 0, 1)
    final_c = compute_c5(best_h.astype(float))

    return best_h.tolist(), float(final_c), int(best_h.size)
