def compute_c5_fft(h):
    n = h.size
    # Use next power of two for convolution: N = 2**(int(np.ceil(np.log2(2*n-1)))) maybe
    N = 1 << ((2*n-1).bit_length())
    # Compute convolution via FFT: a = h, b = 1 - h reversed
    a = np.fft.rfft(h, N)
    b = np.fft.rfft((1 - h)[::-1], N)
    conv = np.fft.irfft(a*b, N)[:2*n-1]
    # Note: correlation indices: 0 to 2n-2 correspond to shifts -(n-1) ... (n-1). C5 = max(conv) * dx
    return conv.max() * (2.0 / n)

def golden_section_minimize(f, a, b, tol=1e-6, max_iter=50):
    phi = (1+np.sqrt(5))/2
    invphi = 1/phi
    c = b - invphi*(b - a)
    d = a + invphi*(b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - invphi*(b - a)
            fc = f(c)
        else:
            a, fc = c, fd
            c = d
            d = a + invphi*(b - a)
            fd = f(d)
        if abs(b - a) < tol:
            break
    # return point with lower f
    if fc < fd:
        return c, fc
    else:
        return d, fd

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # Use while diff not zero within tolerance:
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            delta = min(diff, cap)
        else:
            cap = h[i]
            if cap <= 0:
                continue
            delta = max(diff, -cap)
        h[i] += delta

    return h

make_h:

def make_h(a, n):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    if a <= 1e-12:
        return np.full(n, 0.5, dtype=float)
    c = (1 - a) / (2 - a)
    h = np.empty(n, dtype=float)
    left = xs < a
    right = xs > 2 - a
    middle = ~left & ~right
    # left linear
    if np.any(left):
        h[left] = 1.0 - (1.0 - c) * (xs[left] / a)
    # right linear
    if np.any(right):
        h[right] = c + (1.0 - c) * ((xs[right] - (2 - a)) / a)
    # middle
    if np.any(middle):
        h[middle] = c
    return h

Note for a close to 0, c ~0.5, left region width zero, but left condition will have none.

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0.0:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            delta = min(diff, cap)
        else:
            cap = h[i]
            if cap <= 0.0:
                continue
            delta = max(diff, -cap)
        h[i] += delta
        diff = target - h.sum()
    return h


def make_h(a: float, n: int) -> np.ndarray:
    """Piecewise‑linear U‑shaped function with area = 1."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    if a <= 1e-12:
        return np.full(n, 0.5, dtype=float)
    c = (1.0 - a) / (2.0 - a)
    h = np.empty(n, dtype=float)
    left = xs < a
    right = xs > 2.0 - a
    middle = ~left & ~right
    if left.any():
        h[left] = 1.0 - (1.0 - c) * (xs[left] / a)
    if right.any():
        h[right] = c + (1.0 - c) * ((xs[right] - (2.0 - a)) / a)
    if middle.any():
        h[middle] = c
    return h


def compute_c5_fft(h: np.ndarray) -> float:
    """Exact discrete C5 using FFT‑based convolution."""
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.fft(h, N)
    b_fft = np.fft.fft((1.0 - h)[::-1], N)
    conv = np.fft.ifft(a_fft * b_fft).real[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def evaluate_a(a: float, n: int, target: float, rng: np.random.Generator) -> float:

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return compute_c5_fft(h)


def worst_shift_and_grad(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(
    h: np.ndarray,
    grad: np.ndarray,
    alpha: float,
    target: float,
    rng: np.random.Generator,
) -> np.ndarray:
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    h_new[pos] = h[pos] * (1.0 - eps / total_out)
    h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new
