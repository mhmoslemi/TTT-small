def continuous_sa_custom(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    target = N/2
    T0 = 0.1  # larger
    T_end = 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # Choose two distinct indices i, j
        i, j = rng.choice(N, size=2, replace=False)
        # Random epsilon bounded by remaining capacity at i to remove and at j to add
        max_remove = min(h[i], 1.0)  # h[i] <=1
        max_add = 1.0 - h[j]
        max_eps = min(max_remove, max_add)
        if max_eps <= 0:
            continue
        # Scale epsilon by temperature and random factor
        eps = rng.random() * max_eps * (T / T0)  # temperature scaling
        if eps <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        # Accept if improvement or with probability
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_h = h_new.copy()
                best_c = c_new
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Basic utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    # Random redistribution; converges quickly for our target.
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# -------------------------------------------------
# Soft‑max SLSQP optimisation (global helpers)
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max approximation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# -------------------------------------------------
# Gradient of the worst shift and refinement
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 30) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# -------------------------------------------------
# Custom continuous simulated annealing (preserves sum exactly)
def continuous_sa_custom(h0: np.ndarray, rng: np.random.Generator,
                        time_budget: float):
    """SA on feasible set preserving sum (move epsilon from i to j)."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.1, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i, j = rng.choice(N, size=2, replace=False)
        max_remove = h[i]
        max_add = 1.0 - h[j]
        max_eps = min(max_remove, max_add)
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * (T / T0)
        if eps <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
    return best_h, best_c


# -------------------------------------------------
# Sub‑gradient descent on the worst shift
def subgradient_descent(h0: np.ndarray, rng: np.random.Generator,
                        time_budget: float):
    """Simple sub‑gradient descent on max‑overlap objective."""
    start = time.time()
    h = h0.copy()
    target = h.size / 2.0
    c = compute_c5(h)
    step = 0.02
    while time.time() - start < time_budget:
        grad = worst_gradient(h)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < c:
            h, c = h_new, c_new
        step *= 0.9995
        if step < 1e-8:
            break
    return h, c


# -------------------------------------------------
# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # ---- Initial solution -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n0 = 512
        h = rng.random(n0)
        h = enforce_sum(h, rng)

    best_h, best_c = h, compute_c5(h)

    # ---- Soft‑max SLSQP refinement (if time) -------------------------------
    if time_left() > 0.15 * budget_s:
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time_left() < 5:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=800)
            best_c = compute_c5(best_h)

    # ---- Gradient tidy‑up --------------------------------------------------
    if time_left() > 2:
        best_h = gradient_refine(best_h, rng, step=0.015, max_iter=30)
        best_c = compute_c5(best_h)

    # ---- Upsample to 1024 points -----------------------------------------
    if best_h.size < 1024 and time_left() > 5:
        h_up = upsample_h(best_h, 1024)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        best_h, best_c = h_up, compute_c5(h_up)

    # ---- Custom continuous SA (remaining time) ----------------------------
    rem = time_left()
    if rem > 5:
        best_h, best_c = continuous_sa_custom(best_h, rng, rem)

    # ---- Sub‑gradient descent for final polish ---------------------------
    rem = time_left()
    if rem > 5:
        best_h, best_c = subgradient_descent(best_h, rng, rem)

    # ---- Final sanitisation ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
