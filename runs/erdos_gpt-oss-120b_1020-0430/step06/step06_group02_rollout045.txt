def project_sum_bounds(v, target):
    v = np.clip(v, 0.0, 1.0)
    # water-filling
    for _ in range(10):
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
        if s < target:
            # distribute deficit to those not at 1
            mask = v < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            inc = (target - s) / cnt
            v[mask] += inc
            v = np.clip(v, 0.0, 1.0)
        else:  # s > target
            mask = v > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            dec = (s - target) / cnt
            v[mask] -= dec
            v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for step function on [0,2] discretized by len(h) points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)


def project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the feasible set {0 <= x_i <= 1, sum x_i = target}.
    Simple water‑filling: repeatedly distribute deficit/excess uniformly.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = x.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:  # excess, shrink
            mask = x > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            dec = diff / cnt
            x[mask] -= dec
        else:  # deficit, grow
            mask = x < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            inc = -diff / cnt
            x[mask] += inc
        x = np.clip(x, 0.0, 1.0)
    return x


def worst_shift_and_grad(h: np.ndarray):
    """
    Return (shift, gradient, max_overlap) for the shift achieving the
    maximal overlap. Gradient is the exact derivative of that overlap
    (scaled by 2/n).
    """
    n = h.size
    factor = 2.0 / n
    # overlaps for all integer shifts (including negative)
    O = factor * np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(O))
    s = idx - (n - 1)  # signed shift
    grad = np.zeros(n)
    if s >= 0:
        i = np.arange(n - s)
        grad[i] += (1.0 - h[i + s])
        grad[i + s] -= h[i]
    else:
        s2 = -s
        i = np.arange(n - s2)
        grad[i + s2] += (1.0 - h[i])
        grad[i] -= h[i + s2]
    grad *= factor
    return s, grad, O.max()


def binaryify(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a balanced binary vector (0/1) derived from h."""
    n = h.size
    target = n // 2
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    cur = ones.size
    if cur > target:
        drop = rng.choice(ones, cur - target, replace=False)
        h_bin = np.zeros(n, dtype=float)
        h_bin[ones] = 1.0
        h_bin[drop] = 0.0
    elif cur < target:
        add = rng.choice(zeros, target - cur, replace=False)
        h_bin = np.zeros(n, dtype=float)
        h_bin[ones] = 1.0
        h_bin[add] = 1.0
    else:
        h_bin = np.zeros(n, dtype=float)
        h_bin[ones] = 1.0
    return h_bin


# ---------- smooth‑max (soft‑max) objective for L‑BFGS ----------

def smooth_max_obj(h, beta, lam, target):
    """Smooth approximation of max_k O_k with quadratic sum penalty."""
    n = h.size
    O = (2.0 / n) * np.correlate(h, 1.0 - h, mode="full")
    betaO = beta * O
    bmax = np.max(betaO)
    exp_term = np.exp(betaO - bmax)
    sm = (bmax + np.log(exp_term.sum())) / beta
    pen = lam * (h.sum() - target) ** 2
    return sm + pen


def smooth_max_grad(h, beta, lam, target):
    """Gradient of smooth_max_obj."""
    n = h.size
    factor = 2.0 / n
    O = (2.0 / n) * np.correlate(h, 1.0 - h, mode="full")
    betaO = beta * O
    bmax = np.max(betaO)
    exp_term = np.exp(betaO - bmax)
    w = exp_term / exp_term.sum()          # softmax weights, sum to 1

    grad = np.zeros(n)
    # weighted sum of shift gradients
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            i = np.arange(n - s)
            grad[i] += wk * (1.0 - h[i + s])
            grad[i + s] -= wk * h[i]
        else:
            s2 = -s
            i = np.arange(n - s2)
            grad[i + s2] += wk * (1.0 - h[i])
            grad[i] -= wk * h[i + s2]
    grad *= factor
    # quadratic penalty gradient
    grad += 2.0 * lam * (h.sum() - target)
    return grad


# ---------- main optimisation routine ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    TIME = lambda: time.time() - start

    # ---- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        target = n / 2.0
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_bounds(h, target)
    else:
        n = 512
        target = n / 2.0
        # random balanced binary start
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- Phase 1: projected gradient descent (PGD) --------------------
    alpha0 = 0.25
    decay = 0.99
    it = 0
    while TIME() < budget_s * 0.45:
        s, grad, cur_c = worst_shift_and_grad(best_h)
        if cur_c < best_c:
            best_c, best_h = cur_c, best_h.copy()
        alpha = max(alpha0 * (decay ** it), 1e-5)
        h_candidate = project_sum_bounds(best_h - alpha * grad, target)
        cand_c = compute_c5(h_candidate)
        if cand_c < best_c:
            best_c, best_h = cand_c, h_candidate.copy()
        else:
            # occasional aggressive perturbation
            if it % 300 == 0 and rng.random() < 0.2:
                eps = rng.normal(scale=0.12, size=n)
                h_tmp = project_sum_bounds(best_h + eps, target)
                tmp_c = compute_c5(h_tmp)
                if tmp_c < best_c:
                    best_c, best_h = tmp_c, h_tmp.copy()
        it += 1
        if TIME() > budget_s * 0.45:
            break

    # ---- Phase 2: L‑BFGS‑B on smooth‑max with increasing β ----------
    beta_list = [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0]
    lam = 1500.0  # penalty strength
    for beta in beta_list:
        if TIME() > budget_s * 0.80:
            break
        def obj(x):
            return smooth_max_obj(x, beta, lam, target)
        def grad(x):
            return smooth_max_grad(x, beta, lam, target)
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            obj,
            best_h,
            method="L-BFGS-B",
            jac=grad,
            bounds=bounds,
            options={"maxiter": 600, "ftol": 1e-13, "disp": False},
        )
        h_opt = project_sum_bounds(np.clip(res.x, 0.0, 1.0), target)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= 0.3808:
            break

    # ---- Phase 3: random restarts + short PGD -----------------------
    restart_budget = budget_s * 0.15
    while TIME() < budget_s * 0.95:
        # generate a new start around the current best
        sigma = 0.15 * (0.9 ** (TIME() / restart_budget))
        h0 = project_sum_bounds(best_h + rng.normal(scale=sigma, size=n), target)
        # short PGD from h0
        alpha = 0.22
        for _ in range(800):
            s, grad, _ = worst_shift_and_grad(h0)
            h0 = project_sum_bounds(h0 - alpha * grad, target)
            alpha *= 0.995
        c0 = compute_c5(h0)
        if c0 < best_c:
            best_c, best_h = c0, h0.copy()
        if best_c <= 0.3808:
            break

    # ---- Phase 4: binary rounding check --------------------------------
    h_bin = binaryify(best_h, rng)
    c_bin = compute_c5(h_bin)
    if c_bin < best_c:
        best_c, best_h = c_bin, h_bin.copy()

    # final sanity
    best_h = project_sum_bounds(np.clip(best_h, 0.0, 1.0), target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
