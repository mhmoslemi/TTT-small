def slsqp_softmax(h0, beta, rng, max_iter=2000):
    n = h0.size
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = {
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target,
        'jac': lambda x: np.ones_like(x)
    }

    # define objective and jac
    def obj_and_grad(x):
        # compute correlation
        corr = np.correlate(x, 1.0 - x, mode='full')
        # softmax
        b_corr = beta * corr
        mx = np.max(b_corr)
        expb = np.exp(b_corr - mx)
        sum_exp = expb.sum()
        f = (mx + np.log(sum_exp)) / beta
        # weights
        w = expb / sum_exp
        # compute symmetric weights
        w_rev = w[::-1]
        s_sym = w + w_rev
        # convolution (same)
        conv = np.convolve(x, s_sym, mode='same')
        grad = 1.0 - conv
        # No scaling factor needed? Already correct.
        # Gradient of constraint already defined.
        return f, grad

    # SciPy's minimize with SLSQP expects separate functions for obj and grad, or combined using 'fun' returning f and 'jac' returning grad.
    # Use 'fun' returning f, and 'jac' returning grad.
    def fun(x):
        f, _ = obj_and_grad(x)
        return f

    def jac(x):
        _, g = obj_and_grad(x)
        return g

    res = minimize(fun, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    # Clip and adjust sum precisely
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # same redistribution as earlier
        for _ in range(2000):
            if abs(diff) < 1e-12: break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def slsqp_softmax(h0, beta, rng, max_iter=2000):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
            'jac': lambda x: np.ones_like(x)}
    # define functions
    def fun(x):
        # correlation
        corr = np.correlate(x, 1.0 - x, mode='full')
        b_corr = beta * corr
        mx = np.max(b_corr)
        expb = np.exp(b_corr - mx)
        sum_exp = expb.sum()
        f = (mx + np.log(sum_exp)) / beta
        return f
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        b_corr = beta * corr
        mx = np.max(b_corr)
        expb = np.exp(b_corr - mx)
        sum_exp = expb.sum()
        w = expb / sum_exp
        w_rev = w[::-1]
        s_sym = w + w_rev
        conv = np.convolve(x, s_sym, mode='same')
        grad = 1.0 - conv
        return grad
    res = minimize(fun, h0, method='SLSQP', jac=jac,

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size

        i = rng.integers(0, n)

            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

def slsqp_softmax(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target,
        'jac': lambda x: np.ones_like(x)
    }
    def fun_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        b_corr = beta * corr
        mx = np.max(b_corr)
        expb = np.exp(b_corr - mx)
        sum_exp = expb.sum()
        f = (mx + np.log(sum_exp)) / beta
        w = expb / sum_exp
        w_rev = w[::-1]
        s_sym = w + w_rev
        conv = np.convolve(x, s_sym, mode='same')
        grad = 1.0 - conv
        return f, grad
    res = minimize(fun_and_grad, h0, method='SLSQP', jac=True,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)

            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

        if best_c <= 0.3808:
