import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    # adjust in-place or return new array
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

def gradient_refine(h, target, rng, step=0.02, max_iter=20):
    # simple gradient descent with decreasing step
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h

def compute_c5(h: np.ndarray) -> float:

                    break

    # ---- optional continuous refinement ----
    if time.time() - start < budget_s * 0.90:
        h_float = best_h.astype(np.float64)
        refined = gradient_refine(h_float, target_sum, rng,
                                 step=0.02, max_iter=30)
        refined_c5 = compute_c5(refined)
        if refined_c5 < compute_c5(best_h):
            best_h = refined

    # final sanity checks
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target_sum) > 1e-12:
        best_h = enforce_sum(best_h, target_sum, rng)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
