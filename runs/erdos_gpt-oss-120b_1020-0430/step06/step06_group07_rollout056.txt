import time
import numpy as np
from scipy.optimize import differential_evolution

# ---------- basic utilities ----------
def sigmoid(x: np.ndarray) -> np.ndarray:
    return 1.0 / (1.0 + np.exp(-x))

def softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum x = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# ---------- C5 evaluation ----------
def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT (same as np.correlate(h,1-h,'full'))."""
    n = h.size
    dx = 2.0 / n
    nfft = 1 << ((2 * n - 1).bit_length())
    h_fft = np.fft.fft(h, nfft)
    rev = (1.0 - h)[::-1]
    rev_fft = np.fft.fft(rev, nfft)
    conv = np.fft.ifft(h_fft * rev_fft).real
    return conv[: 2 * n - 1].max() * dx

# ---------- gradient for worst shift ----------
def worst_gradient(h: np.ndarray):
    """Gradient of the worst (largest) overlap O(k)."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # k >= 0
    k = int(np.argmin(cross))  # shift giving max overlap
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using worst‑gradient directed swaps."""
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def subgradient_optimize(h0: np.ndarray, target: float,
                         timelimit: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap (via worst shift)."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    t = 0
    while time.time() - start < timelimit:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5_fft(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

# ---------- DE objective (piecewise constant) ----------
def de_objective(params: np.ndarray, n: int, m: int) -> float:
    """
    params: [m length logits] + [m-1 height logits]
    returns C5 for the piecewise‑constant function (size n).
    """
    # lengths via softmax -> sum 2
    length_logits = params[:m]
    probs = softmax(length_logits)
    lengths = probs * 2.0  # sum to 2

    # heights for first m-1 intervals
    height_logits = params[m:]
    heights = sigmoid(height_logits)  # in (0,1)

    # compute last height to satisfy integral = 1
    L_last = lengths[-1]
    H_last = (1.0 - np.dot(lengths[:-1], heights)) / L_last
    if H_last < 0.0 or H_last > 1.0:
        return 1e6  # infeasible
    heights_full = np.concatenate([heights, [H_last]])

    # build discrete h on uniform grid
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    cum = np.cumsum(lengths)
    idx = np.searchsorted(cum, xs, side="right")
    h = heights_full[idx]

    # project onto exact constraints (tiny numeric errors)
    h = project_box_sum(h, n / 2.0)
    return compute_c5_fft(h)

# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # grid size (higher gives finer search)
    n = 1024

    # ----- candidate from possible initial_h_values -----
    best_h = None
    best_c = np.inf
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # upsample to n points if needed
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = project_box_sum(h0, n / 2.0)
        c0 = compute_c5_fft(h0)
        best_h, best_c = h0, c0

    # ----- Phase 1: Differential Evolution on piecewise constants -----
    remaining = total_budget - (time.time() - start)
    de_time = min(remaining * 0.30, 30.0)  # cap at 30 s
    if de_time > 5.0:
        de_end = time.time() + de_time
        m = 9  # number of intervals
        dim = 2 * m - 1
        bounds = [(-6.0, 6.0)] * dim

        def callback(xk, convergence):
            return time.time() > de_end

        result = differential_evolution(
            de_objective,
            bounds,
            args=(n, m),
            strategy="best1bin",
            maxiter=200,
            popsize=15,
            tol=1e-6,
            polish=False,
            updating="deferred",
            workers=1,
            seed=rng.integers(1 << 30),
            callback=callback,
            disp=False,
        )
        # recompute h from best parameters
        best_params = result.x
        # reconstruct h (same code as in de_objective)
        length_logits = best_params[:m]
        probs = softmax(length_logits)
        lengths = probs * 2.0
        height_logits = best_params[m:]
        heights = sigmoid(height_logits)
        L_last = lengths[-1]
        H_last = (1.0 - np.dot(lengths[:-1], heights)) / L_last
        heights_full = np.concatenate([heights, [H_last]])
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        cum = np.cumsum(lengths)
        idx = np.searchsorted(cum, xs, side="right")
        h_de = heights_full[idx]
        h_de = project_box_sum(h_de, n / 2.0)
        c_de = compute_c5_fft(h_de)
        if c_de < best_c:
            best_h, best_c = h_de, c_de

    # ----- Phase 2: Sub‑gradient descent (global refinement) -----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        sg_time = remaining * 0.35
        h_sg, c_sg = subgradient_optimize(best_h, n / 2.0, sg_time, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # ----- Phase 3: Guided hill‑climb (local polishing) -----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        # allow roughly 3000 iterations per second
        max_iters = int(min(20000, remaining * 3000))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                        max_iters=max_iters,
                                        step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(n)
