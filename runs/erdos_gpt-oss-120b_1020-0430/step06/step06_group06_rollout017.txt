import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def compute_c5_float(h: np.ndarray) -> float:
    """C5 for (possibly fractional) step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def autocorr_min_lag(h: np.ndarray):
    """Return (minimum autocorrelation value, lag) for binary h."""
    n = h.shape[0]
    H = np.fft.rfft(h, n * 2)
    conv = np.fft.irfft(H * H, n * 2)
    conv = np.rint(conv).astype(int)          # integer counts
    central = n - 1
    left = conv[:central]                     # negative lags
    right = conv[central + 1 : 2 * n - 1]     # positive lags
    min_left = left.min()
    idx_left = left.argmin()
    min_right = right.min()
    idx_right = right.argmin()
    if min_right < min_left:
        return min_right, idx_right + 1                # +1 … +central
    else:
        return min_left, -(idx_left + 1)                # -1 … -central


def compute_c5_binary(h: np.ndarray) -> float:
    """C5 for binary h using the min autocorrelation."""
    m, _ = autocorr_min_lag(h)
    return 1.0 - (2.0 / h.size) * m


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- SLSQP soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

        nc = compute_c5_binary(nh)
        delta = nc - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_h, current_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        T *= 0.9999                 # slow cooling

    # ---- deterministic greedy sweep (no temperature) ----
    sweep_deadline = start + budget_s * 0.85
    while time.time() < sweep_deadline:
        min_val, lag = autocorr_min_lag(best_h)
        if lag == 0:
            break
        if lag > 0:
            shifted = np.concatenate([best_h[lag:], np.zeros(lag, dtype=best_h.dtype)])
        else:
            shifted = np.concatenate([np.zeros(-lag, dtype=best_h.dtype), best_h[: n + lag]])
        zero_to_one = np.where((best_h == 0) & (shifted == 1))[0]
        one_to_zero = np.where((best_h == 1) & (shifted == 0))[0]
        if zero_to_one.size and one_to_zero.size:
            i = rng.choice(zero_to_one)
            j = rng.choice(one_to_zero)
            best_h[i] = 1
            best_h[j] = 0
            best_c = compute_c5_binary(best_h)
        else:
            break

    # ---- continuous refinement with SLSQP ----
    cont_deadline = start + budget_s * 0.97
    best_h_float = best_h.astype(float)
    best_h_float = enforce_sum(best_h_float, rng)
    beta_seq = (50.0, 500.0, 5000.0, 20000.0)
    for beta in beta_seq:
        if time.time() > cont_deadline:
            break
        best_h_float = slsqp_optimize(best_h_float, beta, rng, max_iter=200)
        best_h_float = enforce_sum(best_h_float, rng)

    final_c = compute_c5_float(best_h_float)

    return best_h_float.tolist(), float(final_c), int(n)
