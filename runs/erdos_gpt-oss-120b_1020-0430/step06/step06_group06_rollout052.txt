def enforce_sum_vec(h, target, rng):
    diff = target - h.sum()
    tol = 1e-12
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        if diff > 0:
            free = h < 1 - 1e-14
            cap = 1 - h[free]
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            add = min(diff, total_cap)
            h[free] += add * (cap / total_cap)
        else:
            free = h > 1e-14
            cap = h[free]
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            sub = min(-diff, total_cap)
            h[free] -= sub * (cap / total_cap)
        diff = target - h.sum()
    return h
```

def enforce_sum_vec(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == target while staying in [0,1] (vectorized)."""
    diff = target - h.sum()
    # iterate a few times; should converge quickly because diff is small
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            cap = 1.0 - h[free]
            total = cap.sum()
            if total <= 0:
                break
            add = min(diff, total)
            inc = add * (cap / total)
            h[free] += inc
        else:
            free = h > 1e-14
            cap = h[free]
            total = cap.sum()
            if total <= 0:
                break
            sub = min(-diff, total)
            dec = sub * (cap / total)
            h[free] -= dec
        diff = target - h.sum()
    # fallback random adjustments if still off (very unlikely)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

        _, grad, cur_c = worst_shift_grad(cur_h)

        # schedule step size
        progress = (time.time() - start) / (budget_s * 0.8)
        progress = min(progress, 1.0)
        step = step_initial * (1.0 - progress) + step_final * progress

        # gradient move
        nh = cur_h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum_vec(nh, target_sum, rng)
        nc = max_corr_fft(nh)

        # acceptance
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / max(step, 1e-12)):
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.05, size=n)
            nh = best_h + noise
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum_vec(nh, target_sum, rng)
            cur_h = nh
            cur_c = max_corr_fft(cur_h)

    # ---- SLSQP refinement (remaining time) ----
    beta_seq = (10.0, 100.0, 1000.0, 5000.0, 20000.0)
    for beta in beta_seq:
        if time.time() > start + budget_s * 0.98:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
        best_c = compute_c5_corr(best_h)

    # ---- final repair and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_vec(best_h, target_sum, rng)
    final_c = compute_c5_corr(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
