def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    n = 512
    target_sum = n/2

    # initialize best from initial if exist
    best_h = None
    best_c = np.inf

    # helper: evaluate and store
    def eval_h(h):
        c = compute_c5(h)
        return c

    # Ensure best from initial
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0,0,1)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_c = c0

    # Initialize population
    pop_size = 30
    population = []
    pop_c = []
    # include best_h as first individual if available
    if best_h is not None:
        population.append(best_h.copy())
        pop_c.append(best_c)
    # fill rest
    while len(population) < pop_size:
        h = rng.random(n)
        h = enforce_sum(h, rng)
        c = compute_c5(h)
        population.append(h)
        pop_c.append(c)

    # DE parameters
    F = 0.5
    # maybe CR = 0.9 probability of crossover
    CR = 0.9

    # DE loop
    while time.time() - start < total_budget*0.80:
        for i in range(pop_size):
            # select distinct a,b,c
            idxs = list(range(pop_size))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            # mutation
            mutant = population[a] + F*(population[b] - population[c])
            # crossover
            crossover_mask = rng.random(n) < CR
            # ensure at least one dimension from mutant
            if not np.any(crossover_mask):
                crossover_mask[rng.integers(n)] = True
            trial = np.where(crossover_mask, mutant, population[i])
            # bound and sum repair
            trial = np.clip(trial, 0.0, 1.0)
            trial = enforce_sum(trial, rng)
            # evaluate
            c_trial = compute_c5(trial)
            if c_trial < pop_c[i]:
                population[i] = trial
                pop_c[i] = c_trial
                if c_trial < best_c:
                    best_h = trial.copy()
                    best_c = c_trial
        # maybe occasional random restart for worst individuals
        # Not necessary

    # after DE, apply gradient refinement
    if best_h is not None:
        # gradient_refine
        # use steps maybe 0.02 with 30 iterations (increase)
        refined = gradient_refine(best_h, rng, step=0.02, max_iter=30)
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_h = refined.copy()
            best_c = c_ref

    # final softmax SLSQP on best_h if still time
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None and best_h.size == n:
        # make a copy to avoid modifying original

            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_h = h_s.copy()
                best_c = c_s

    # final check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

    n = h.size
    n2 = 2*n - 1
    H = np.fft.fft(h, n2)

import time
import numpy as np
from scipy.optimize import minimize

TARGET_BOUND = 0.3808

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1.0 - 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            idxs = np.where(h > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            h[0] = np.clip(h[0] + diff2, 0.0, 1.0)
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# ---------- gradient of worst shift ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Differential Evolution ----------
def de_optimize(h0, n, rng, time_budget):
    start = time.time()
    target_sum = n / 2.0
    pop_size = 20
    population = []
    pop_c = []

    if h0 is not None and h0.size == n:
        h = np.clip(h0, 0.0, 1.0)
        h = enforce_sum(h, rng)
        population.append(h)
        pop_c.append(compute_c5(h))

    while len(population) < pop_size:
        h = rng.random(n)
        h = enforce_sum(h, rng)
        population.append(h)
        pop_c.append(compute_c5(h))

    best_idx = int(np.argmin(pop_c))
    best_h = population[best_idx].copy()
    best_c = pop_c[best_idx]

    F = 0.6
    CR = 0.9

    while time.time() - start < time_budget:
        for i in range(pop_size):
            if time.time() - start >= time_budget:
                break
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            crossover = rng.random(n) < CR
            if not np.any(crossover):
                crossover[rng.integers(n)] = True
            trial = np.where(crossover, mutant, population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = enforce_sum(trial, rng)
            c_trial = compute_c5(trial)
            if c_trial < pop_c[i]:
                population[i] = trial
                pop_c[i] = c_trial
                if c_trial < best_c:
                    best_h = trial.copy()
                    best_c = c_trial
                    if best_c <= TARGET_BOUND:
                        return best_h, best_c
    return best_h, best_c
