def binary_sa_hill_climb_long(h_bin, rng, deadline):
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    start_time = time.time()
    # initial temperature as proportion of cur_c
    init_T = max(1e-6, cur_c * 0.03)
    end_T = 1e-8
    while time.time() < deadline:
        # Compute progress
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total if total > 0 else 1.0
        T = init_T * ((end_T / init_T) ** progress)
        # Choose random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Swap
        cur_h[i] = 0.0

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- utilities -------------------
def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000):
    """Force sum(h) == n/2 exactly by tiny adjustments."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# -------------- smooth‑max gradient --------------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β) log‑sum‑exp(β·overlap)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    omh = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = shift_idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * omh[k:]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:] += wi * omh[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g


def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    """Projected gradient descent on smooth‑max for a schedule of β."""
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h


def worst_shift_gradient(h):
    """Sub‑gradient using the worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    maxv = v.max()
    idxs = np.where(v == maxv)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g


def worst_shift_refine(h, rng, base_step, iters, deadline):
    """Projected sub‑gradient on current worst shift."""
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.95
    return h


# -------------- SLSQP on smooth‑max --------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# -------------- binary simulated annealing --------------
def binary_sa_hill_climb_long(h_bin, rng, deadline):
    """Long SA on binary vectors (swap 1↔0)."""
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    start = time.time()
    init_T = max(1e-6, cur_c * 0.03)
    end_T = 1e-8
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        prog = elapsed / total if total > 0 else 1.0
        T = init_T * ((end_T / init_T) ** prog)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / (T + 1e-12)):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c


# ------------------------- main -------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- choose base resolution ----
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        n_parent = len(parent)
        base_n = n_parent * 2 if n_parent * 2 <= 2000 else n_parent
    else:
        base_n = 1440
    if base_n % 2:
        base_n += 1

    # ---- initialise continuous h ----
    if parent is not None:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
        h = np.interp(xt, xs, parent)
    else:
        h = rng.random(base_n)
    h = h / h.sum() * (base_n / 2.0)
    np.clip(h, 0.0, 1.0, out=h)
    adjust_sum_exact(h, rng)

    # ---- smooth‑max gradient descent ----
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000,
             20000, 50000, 100000]
    h = gradient_descent_smoothmax(h, betas, base_step=0.30,
                                   it_per_beta=20, rng=rng,
                                   deadline=deadline)

    # ---- worst‑shift refinement ----
    h = worst_shift_refine(h, rng, base_step=0.15, iters=30,
                           deadline=deadline)

    # ---- SLSQP polish ----
    for beta in (2000, 5000, 20000, 50000, 100000):
        if time.time() > deadline:
            break
        h = slsqp_optimize(h, beta, rng, max_iter=800)

    c = compute_c5(h)
    if c < best_c:
        best_c, best_h, best_n = c, h.copy(), base_n

    # ---- binary SA (long) ----
    h_bin = np.rint(h).astype(float)
    adjust_sum_exact(h_bin, rng)
    h_bin, c_bin = binary_sa_hill_climb_long(h_bin, rng, deadline)
    if c_bin < best_c:
        best_c, best_h, best_n = c_bin, h_bin.copy(), base_n

    # ---- secondary random start if time permits ----
    remaining = deadline - time.time()
    if remaining > 5.0:
        h_rand = rng.random(base_n)
        h_rand = h_rand / h_rand.sum() * (base_n / 2.0)
        np.clip(h_rand, 0.0, 1.0, out=h_rand)
        adjust_sum_exact(h_rand, rng)
        h_rand_bin = np.rint(h_rand).astype(float)
        adjust_sum_exact(h_rand_bin, rng)
        sub_dead = time.time() + remaining * 0.5
        h_rand_bin, c_rand = binary_sa_hill_climb_long(h_rand_bin, rng, sub_dead)
        if c_rand < best_c:
            best_c, best_h, best_n = c_rand, h_rand_bin.copy(), base_n

    # ---- optional up‑sampling and short SA ----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 2000:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        # sum already correct after repeat
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_up, c_up = binary_sa_hill_climb_long(h_up, rng, sub_dead)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # ---- final safety fallback ----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
