import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum x = target} using binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray) -> tuple:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n, k>=0
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap (via worst shift)."""
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c


def upsample(h: np.ndarray, rng: np.random.Generator):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)


# --------------------------------------------------------------
# Linearization + LP refinement (CCP)
def linearize_and_solve(h_old: np.ndarray, target: float):
    """One CCP step: maximize min autocorrelation with linearized constraints."""
    import cvxpy as cp

    n = h_old.shape[0]
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(h) == target,
        h >= 0,
        h <= 1,
    ]

    # pre‑compute const_k = sum_i h_old[i] * h_old[i+k]
    consts = np.empty(n)
    for k in range(n):
        i_end = n - k
        consts[k] = np.dot(h_old[:i_end], h_old[k:])

    for k in range(n):
        i_end = n - k
        hi = h[:i_end]
        hj = h[k:]
        expr = cp.sum(cp.multiply(hi, h_old[k:])) + cp.sum(cp.multiply(h_old[:i_end], hj))
        constraints.append(expr >= t + consts[k])

    prob = cp.Problem(cp.Maximize(t), constraints)
    # ECOS works well for linear programs
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    except Exception:
        return None, None

    if h.value is None:
        return None, None

    h_val = np.clip(h.value, 0.0, 1.0)
    h_val = project_box_sum(h_val, target)
    t_val = t.value if t.value is not None else None
    return h_val, t_val

    best_h = h0

        if c_hc < best_c:

    # --------------------------------------------------------------
    # Phase 3: CCP LP refinement (≈10% of remaining budget)
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        iter_limit = 6
        while remaining > 5.0 and iter_limit > 0:
            h_new, _ = linearize_and_solve(best_h, n0 / 2.0)
            if h_new is None:
                break
            c_new = compute_c5(h_new)
            if c_new < best_c:

            # stop if improvement tiny

            iter_limit -= 1

        if c_up_sg < best_c:

        remaining = total_budget - (time.time() - start)
        if remaining > 5.0:

            h_up_hc, c_up_hc = guided_hill_climb(best_h, rng,
                                                  max_iters=max_iters,
                                                  step_scale=0.6)
            if c_up_hc < best_c:
                best_h, best_c = h_up_hc, c_up_hc
        # final quick CCP on fine grid if time permits
        remaining = total_budget - (time.time() - start)
        if remaining > 5.0 and best_h.size <= 1024:
            h_fine, _ = linearize_and_solve(best_h, best_h.size / 2.0)
            if h_fine is not None:
                c_fine = compute_c5(h_fine)
                if c_fine < best_c:
                    best_h, best_c = h_fine, c_fine

    # --------------------------------------------------------------
    # Phase 5: random restarts of sub‑gradient if any budget left
    remaining = total_budget - (time.time() - start)
    while remaining > 6.0:
        # allocate a modest chunk
        chunk = min(remaining * 0.3, 15.0)
        h_rand = random_feasible(best_h.size, rng)
        h_sg2, c_sg2 = subgradient_optimize(h_rand, best_h.size / 2.0,
                                            chunk, rng)
        if c_sg2 < best_c:
            best_h, best_c = h_sg2, c_sg2
        remaining = total_budget - (time.time() - start)

    # --------------------------------------------------------------
    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
