def worst_gradient(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # For i from 0 to n-1, consider i+k within range
        if k > 0:
            # h_i appears with h_{i+k} and h_{i-k} if i-k>=0
            grad[: n - k] -= h[k:]   # subtract h_{i+k}
            grad[k:] -= h[: n - k]   # subtract h_{i-k}
        else:
            k = -k
            grad[k:] -= h[: n - k]   # i >= k, subtract h_{i-k}
            grad[: n - k] -= h[k:]   # i < n-k, subtract h_{i+k}
    return grad
```

def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    a = np.fft.rfft(h, n=2 * n)
    b = np.fft.rfft(np.flip(1.0 - h), n=2 * n)
    corr = np.fft.irfft(a * b, n=2 * n)[: 2 * n - 1]
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

    # ----- final random fine‑tuning -----
    while time.time() - start < budget_s * 0.999:
        i = rng.integers(0, n2)
        j = rng.integers(0, n2)
        if i == j:
            continue
        max_eps = min(best_h_cont[i], 1.0 - best_h_cont[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.02
        h_tmp = best_h_cont.copy()
        h_tmp[i] -= eps
        h_tmp[j] += eps
        np.clip(h_tmp, 0.0, 1.0, out=h_tmp)
        h_tmp = enforce_sum(h_tmp, rng)
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c_cont:
            best_c_cont = c_tmp
            best_h_cont = h_tmp.copy()

    # ----- final enforce and return -----
    best_h_cont = enforce_sum(np.clip(best_h_cont, 0.0, 1.0), rng)
    final_c = compute_c5(best_h_cont)
    return best_h_cont.tolist(), float(final_c), int(best_h_cont.size)
