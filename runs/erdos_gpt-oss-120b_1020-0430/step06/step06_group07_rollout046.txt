def de_optimize(h0, n, rng, time_budget):
    start = time.time()
    target_sum = n/2.0
    pop_size = 20
    # initialize population
    pop = np.empty((pop_size, n))
    scores = np.empty(pop_size)
    # include h0 as one individual
    pop[0] = project_box_sum(np.clip(h0,0,1), target_sum)
    scores[0] = compute_c5(pop[0])
    for i in range(1, pop_size):
        pop[i] = random_feasible(n, rng)
        scores[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]
    # DE params
    F = 0.7
    CR = 0.9
    # track stagnation
    iter_since_improve = 0
    while time.time() - start < time_budget:
        # iterate each individual
        for i in range(pop_size):
            # time check per inner loop
            if time.time() - start >= time_budget:
                break
            # Choose indices a,b,c distinct; maybe using best
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            # Variant: use best for a (DE/best/1)
            # a = best_idx
            mutant = pop[a] + F * (pop[b] - pop[c])
            # Crossover
            trial = pop[i].copy()
            jrand = rng.integers(0, n)
            cross_mask = rng.random(n) < CR
            cross_mask[jrand] = True
            trial[cross_mask] = mutant[cross_mask]
            # Feasibility
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_box_sum(trial, target_sum)
            ct = compute_c5(trial)
            if ct < scores[i]:
                pop[i] = trial
                scores[i] = ct
                if ct < best_score:
                    best = trial.copy()
                    best_score = ct
                    best_idx = i
                    iter_since_improve = 0
            else:
                iter_since_improve += 1
        # optional diversification: if iter_since_improve > pop_size * 20, reinitialize worst half
        if iter_since_improve > pop_size * 20:
            # reinitialize half of population except best
            # sort indices by scores descending
            order = np.argsort(scores)[::-1]
            for idx in order[:pop_size//2]:
                if idx == best_idx:
                    continue
                pop[idx] = random_feasible(n, rng)
                scores[idx] = compute_c5(pop[idx])
            iter_since_improve = 0
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n/2.0)
    else:
        n = 512
        h0 = random_feasible(n, rng)
    best_h = h0
    best_c = compute_c5(best_h)
    # Allocate time for DE, subgradient, hill, upsample
    time_left = budget_s - (time.time() - start)
    # Phase DE
    if time_left > 5.0:
        de_budget = time_left * 0.5  # 50% for DE
        best_h, best_c = de_optimize(best_h, n, rng, de_budget)
    # Phase local refinement
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        # Use subgradient for some time
        sub_budget = time_left * 0.4
        best_h, best_c = subgradient_optimize(best_h, n/2.0, sub_budget, rng)
    # Phase hill climb
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        max_iters = int(min(4000, time_left * 300))
        best_h, best_c = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.7)
    # Phase upsample & refinement if time left and n*2 <= 1024
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0 and n*2 <= 1024:
        h_up = upsample(best_h, rng)
        n_up = h_up.size
        # Subgradient on upscaled
        sub_budget = time_left * 0.5
        best_h_up, best_c_up = subgradient_optimize(h_up, n_up/2.0, sub_budget, rng)
        # choose better
        if best_c_up < best_c:
            best_h, best_c = best_h_up, best_c_up
        # final hill
        time_left = budget_s - (time.time() - start)
        if time_left > 5.0:
            max_iters = int(min(4000, time_left * 250))
            best_h, best_c = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.6)
    # final sanitization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size/2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via binary search on the Lagrange multiplier."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point in {0≤x≤1, sum x = n/2}."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst‑shift overlap O_k and its shift index."""
    n = h.size
    # auto‑correlation of h (length n, shift 0..n‑1)
    cross = np.correlate(h, h, mode="full")[n - 1 :]
    k = int(np.argmin(cross))  # shift giving maximal O_k
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb moving mass from highest to lowest gradient entries."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h.copy()
        # optional decay of step_scale could be added here
    return best_h, best_c


def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c


def upsample(h: np.ndarray, rng: np.random.Generator):
    """Linear up‑sampling by factor 2 with sum correction."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)


def de_optimize(h0: np.ndarray, n: int, rng: np.random.Generator,
                time_budget: float):
    """Differential Evolution on the C5 objective."""
    start = time.time()
    target = n / 2.0
    pop_size = 20
    # initialise population
    pop = np.empty((pop_size, n))
    scores = np.empty(pop_size)
    pop[0] = project_box_sum(np.clip(h0, 0.0, 1.0), target)
    scores[0] = compute_c5(pop[0])
    for i in range(1, pop_size):
        pop[i] = random_feasible(n, rng)
        scores[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.7
    CR = 0.9
    stagnation = 0
    while time.time() - start < time_budget:
        for i in range(pop_size):
            if time.time() - start >= time_budget:
                break
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            jrand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = mutant[mask]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_box_sum(trial, target)
            ct = compute_c5(trial)
            if ct < scores[i]:
                pop[i] = trial
                scores[i] = ct
                if ct < best_score:
                    best = trial.copy()
                    best_score = ct
                    best_idx = i
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > pop_size * 20:
            order = np.argsort(scores)[::-1]
            for idx in order[:pop_size // 2]:
                if idx == best_idx:
                    continue
                pop[idx] = random_feasible(n, rng)
                scores[idx] = compute_c5(pop[idx])
            stagnation = 0
    return best, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = random_feasible(n, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # Phase 1: Differential Evolution
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        de_budget = remaining * 0.5
        best_h, best_c = de_optimize(best_h, n, rng, de_budget)

    # Phase 2: Sub‑gradient descent
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        sub_budget = remaining * 0.4
        best_h, best_c = subgradient_optimize(best_h, n / 2.0, sub_budget, rng)

    # Phase 3: Guided hill‑climb
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        max_iters = int(min(4000, remaining * 300))
        best_h, best_c = guided_hill_climb(best_h, rng,
                                           max_iters=max_iters, step_scale=0.7)

    # Phase 4: Upsample and final polish (if time permits)
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0 and n * 2 <= 1024:
        h_up = upsample(best_h, rng)
        n_up = h_up.size
        sub_budget = remaining * 0.5
        up_h, up_c = subgradient_optimize(h_up, n_up / 2.0, sub_budget, rng)
        if up_c < best_c:
            best_h, best_c = up_h, up_c
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(min(4000, remaining * 250))
            best_h, best_c = guided_hill_climb(best_h, rng,
                                               max_iters=max_iters,
                                               step_scale=0.6)

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
