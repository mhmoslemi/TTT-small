import time, numpy as np, from scipy.optimize import minimize

global CURRENT_TARGET, CURRENT_BETA

def compute_c5(h):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # Use while loop but limit iterations
    # Use simple algorithm: if diff > 0, fill up random indices
    # else reduce random indices
    # This may be faster than while loop for small diff
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h, target, rng, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0*h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

def mass_transfer_move(h, grad, alpha, target, rng):
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new

def gradient_refine(h, target, rng, step=0.015, max_iter=15):
    n = h.size
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h

def alpha_schedule(it, max_it):
    frac = it / max_it
    if frac < 0.3:
        return (1.0, 0.5, 0.25, 0.125, 0.0625)
    if frac < 0.6:
        return (0.5, 0.25, 0.125, 0.0625, 0.03125)
    if frac < 0.9:
        return (0.25, 0.125, 0.0625, 0.03125, 0.015625)
    return (0.125, 0.0625, 0.03125, 0.015625, 0.0078125)

def random_perturb(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

        factor = n // N0
        remainder = n % N0
        h = np.repeat(h0, factor)
        if remainder > 0:
            h = np.concatenate([h, h0[:remainder]])
        h = np.clip(h, 0.0, 1.0)
        target = n/2.0
        h = enforce_sum(h, target, rng)
        return h
    else:
        # random binary with half ones
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
        return h

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h, target, rng, beta, max_iter=250):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h, grad, alpha, target, rng):
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def gradient_refine(h, target, rng, step=0.015, max_iter=15):
    n = h.size
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def alpha_schedule(it, max_it):
    frac = it / max_it
    if frac < 0.3:
        return (1.0, 0.5, 0.25, 0.125, 0.0625)
    if frac < 0.6:
        return (0.5, 0.25, 0.125, 0.0625, 0.03125)
    if frac < 0.9:
        return (0.25, 0.125, 0.0625, 0.03125, 0.015625)
    return (0.125, 0.0625, 0.03125, 0.015625, 0.0078125)


def random_perturb(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def thue_morse(n):
    """Return binary Thue‑Morse sequence of length n (n power of two)."""
    t = np.empty(n, dtype=float)
    for i in range(n):
        t[i] = (bin(i).count("1") & 1)
    return t


def get_initial_h(n, rng):
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N0 = h0.size
        if n == N0:
            h = np.clip(h0, 0.0, 1.0)
            target = n / 2.0
            return enforce_sum(h, target, rng)
        factor = n // N0
        remainder = n % N0
        h = np.repeat(h0, factor)
        if remainder:
            h = np.concatenate([h, h0[:remainder]])
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        return enforce_sum(h, target, rng)
    # No parent: try Thue‑Morse for powers of two
    if (n & (n - 1)) == 0 and rng.random() < 0.6:
        return thue_morse(n)
    # fallback to random half‑filled binary vector
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h
