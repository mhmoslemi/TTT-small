def stochastic_optimize(h0, rng, budget):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    T0 = 0.02; T_end = 1e-4
    # compute schedule based on total budget
    while time.time() - start < budget:
        # annealing temperature schedule
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / budget)
        # gradient for worst shift
        grad = worst_gradient(h)
        # choose i and j:
        i = np.argmax(grad)
        j = np.argmin(grad)
        # compute max epsilon
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            # fallback: random i,j
            i, j = rng.integers(0, n, size=2)
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
        eps = rng.random() * max_eps
        # propose move in direction of decreasing gradient
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # optional small random perturb
        # no need, we already have randomness.
        # enforce bounds and sum already satisfied
        cc = compute_c5(nh)
        delta = cc - best_c
        if cc < best_c or rng.random() < np.exp(-delta / max(T, 1e-12)):

            if cc < best_c:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.01, n_iter: int = 6) -> np.ndarray:
    for _ in range(n_iter):
        grad = worst_gradient(h)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- stochastic continuous SA ----------
def stochastic_optimize(h_init: np.ndarray, rng: np.random.Generator,
                        budget: float) -> tuple[np.ndarray, float]:
    n = h_init.size
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    T0, T_end = 0.05, 1e-4
    while time.time() - start < budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / budget)
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            i, j = rng.integers(0, n, size=2)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
        eps = rng.random() * max_eps * (T / T0 + 0.05)
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, rng)
        new_c = compute_c5(nh)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, cur_c = nh, new_c
            if new_c < best_c:
                best_c, best_h = new_c, nh.copy()
        if rng.random() < 0.001:  # occasional shake
            h = best_h.copy()
            for _ in range(5):
                a = rng.integers(0, n)
                b = rng.integers(0, n)
                me = min(h[a], 1.0 - h[b])
                if me <= 0:
                    continue
                e = rng.random() * me
                h[a] -= e
                h[b] += e
            h = enforce_sum(h, rng)
            cur_c = compute_c5(h)
    return best_h, best_c


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 512
        h0 = rng.random(n)
    h0 = enforce_sum(h0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # early exit if already good
    if best_c <= 0.3808:
        return best_h.tolist(), float(best_c), int(n)

    # stochastic optimization (≈85% time)
    sa_time = total_budget * 0.85
    best_h, best_c = stochastic_optimize(best_h, rng, sa_time)

    # final refinement with SLSQP and gradient tidy‑up
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        # gradient tidy‑up before SLSQP
        best_h = gradient_refine(best_h, rng, step=0.02, n_iter=4)
        best_c = compute_c5(best_h)
        # soft‑max passes
        beta_seq = (50., 200., 800., 3000., 8000., 20000., 50000.)
        for beta in beta_seq:
            if time.time() - start > total_budget * 0.99:
                break
            max_iter = 2000 if beta >= 800 else 1000
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            best_c = compute_c5(best_h)
        # final gradient polish
        if time.time() - start < total_budget * 0.999:
            best_h = gradient_refine(best_h, rng, step=0.01, n_iter=6)
            best_c = compute_c5(best_h)

    # sanitise and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
