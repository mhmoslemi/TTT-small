def compute_c5_fft(h, L=None):
    n = h.shape[0]
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

    # attempt to adjust iteratively

def slsqp_optimize(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, rng)
    return x

def anneal_binary(h0, rng, time_limit):

    current_h = h0.copy()

    best_h = current_h.copy()

import time, math
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None
L_GLOBAL = None


def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT (same scaling as np.correlate)."""
    n = h.shape[0]
    L = L_GLOBAL
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    corr = corr[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def correlation_fft(x: np.ndarray) -> np.ndarray:
    """Linear correlation of x with 1-x via FFT."""
    n = x.shape[0]
    L = L_GLOBAL
    H = np.fft.rfft(x, n=L)
    G = np.fft.rfft(1.0 - x, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return corr[: 2 * n - 1]


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = correlation_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + math.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to have exact sum n/2 while respecting [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def adjust_sum_exact(h: np.ndarray) -> np.ndarray:
    """Force exact sum n/2 by fixing a single coordinate."""
    n = h.size
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if idx.size == 0:  # should not happen
        return h
    i = idx[0]
    h[i] = min(max(h[i] + diff, 0.0), 1.0)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    """Refine h with soft‑max‑SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    if abs(x.sum() - CURRENT_TARGET) > 1e-12:
        x = enforce_sum(x, rng)
    return x


def init_binary_from_initial(initial, n, rng):
    """Create a binary vector of length n by tiling a processed 512‑sample pattern."""
    h = np.array(initial, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h_bin = (h >= 0.5).astype(float)
    target = h_bin.size // 2
    cur = int(h_bin.sum())
    diff = target - cur
    if diff > 0:
        zero_idx = np.where(h_bin == 0)[0]
        if zero_idx.size >= diff:
            chosen = rng.choice(zero_idx, size=diff, replace=False)
            h_bin[chosen] = 1.0
    elif diff < 0:
        one_idx = np.where(h_bin == 1)[0]
        if one_idx.size >= -diff:
            chosen = rng.choice(one_idx, size=-diff, replace=False)
            h_bin[chosen] = 0.0
    repeats = n // h_bin.size
    big = np.tile(h_bin, repeats)
    if big.size > n:
        big = big[:n]
    return big.astype(float)


def anneal_binary(h0, rng, deadline):
    """Simulated‑annealing on binary vectors (swap moves)."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    it_max = 500_000
    T0 = 0.02
    T_end = 1e-8
    while time.time() < deadline and it < it_max:
        frac = it / it_max
        T = T0 * (1 - frac) + T_end * frac
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # swap
        nc = compute_c5_fft(cur_h)
        delta = nc - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < math.exp(-delta / T)
        if accept:
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # revert
        it += 1
    return best_h, best_c


def compute_c5_np(h_arr: np.ndarray) -> float:
    """Direct correlation using np.correlate (exact as evaluator)."""
    corr = np.correlate(h_arr, 1.0 - h_arr, mode="full")
    return np.max(corr) * (2.0 / h_arr.size)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # resolution – choose a power of two large enough to allow improvement
    n = 4096
    global L_GLOBAL
    L_GLOBAL = 1 << ((2 * n - 1).bit_length())   # 8192

    # initialise
    if "initial_h_values" in globals():
        h0 = init_binary_from_initial(initial_h_values, n, rng)
    else:

        ones = rng.choice(n, n // 2, replace=False)
        h0[ones] = 1.0

    # ----------- binary annealing ---------------
    deadline = start + 0.80 * budget_s
    best_h, best_c = anneal_binary(h0, rng, deadline)

    # ----------- continuous refinement -----------
    now = time.time()
    remaining = start + budget_s - now
    if remaining > 5.0:

        beta_seq = (50.0, 500.0, 5000.0, 20000.0)

            best_c = compute_c5_fft(h_float)

    # ----------- final feasibility fix ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h)

    final_c = compute_c5_np(best_h)
