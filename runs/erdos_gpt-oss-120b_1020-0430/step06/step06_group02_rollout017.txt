# ---------------------- utilities -------------------------

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale/adjust h so that sum(h) == n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = n * 10
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        # tiny residual – distribute equally and clip
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            i = rng.integers(0, n)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def c5_exact(h: np.ndarray) -> float:
    """Exact C5 using np.correlate (the reference implementation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def c5_fast(h: np.ndarray) -> float:
    """Fast C5 via FFT (good for iterative use)."""
    n = h.size
    size = 2 * n
    fa = np.fft.fft(h, size)
    fb = np.fft.fft(1.0 - h, size)
    corr = np.real(np.fft.ifft(fa * np.conj(fb)))[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def worst_shift_index(h: np.ndarray) -> int:
    """Signed shift achieving the maximal overlap (index in -n+1..n-1)."""
    n = h.size
    size = 2 * n
    fa = np.fft.fft(h, size)
    fb = np.fft.fft(1.0 - h, size)
    corr = np.real(np.fft.ifft(fa * np.conj(fb)))[: 2 * n - 1]
    idx = int(np.argmax(corr))
    return idx - (n - 1)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift overlap."""
    n = h.size
    k = worst_shift_index(h)
    k_abs = abs(k)
    grad = np.ones_like(h)
    if k_abs == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k_abs] -= h[k_abs:]
        grad[k_abs:] -= h[: n - k_abs]
    return grad

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k using log‑sum‑exp."""
    n = x.size
    size = 2 * n
    fa = np.fft.fft(x, size)
    fb = np.fft.fft(1.0 - x, size)
    corr = np.real(np.fft.ifft(fa * np.conj(fb)))[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:

    n = h0.size

    res = minimize(

        h0,

        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
