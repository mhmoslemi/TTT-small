def worst_shift(h):
    n = h.shape[0]
    cross = np.correlate(h, h, mode='full')
    cross = cross[n-1:]   # length n
    k = int(np.argmin(cross))  # minimize auto-corr (i.e., max O(k))
    return k
```

def find_swap_indices(h, k):
    n = len(h)
    # For positions where i < n - k
    end = n - k
    if k == 0:
        # O(0) = sum h_i(1-h_i)=0, can't reduce; just return None

def compute_c5_int(h):
    n = h.size
    sum_h = h.sum()
    cross = np.correlate(h, h, mode='full')
    cross = cross[n-1:]  # len n
    max_O = sum_h - cross.min()
    return max_O
```

def max_lfsr_seq(m, taps):
    # taps: list of bit positions (0-indexed) to XOR together for feedback
    state = (1 << m) - 1  # all ones
    seq = np.empty(2**m - 1, dtype=np.uint8)
    for i in range(seq.size):
        out = state & 1
        seq[i] = out
        # compute feedback
        fb = 0
        for t in taps:
            fb ^= (state >> t) & 1
        state = (state >> 1) | (fb << (m - 1))
    return seq

import time
import numpy as np

def compute_max_overlap(h: np.ndarray) -> int:
    """Maximum overlap O = max_k ∫ h(x)(1-h(x+k)) dx scaled by dx=2/n.
    For binary step functions this equals n/2 - min auto‑correlation."""
    n = h.size
    target = n // 2
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    return target - int(cross.min())


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=np.uint8)
    h[perm[: n // 2]] = 1
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024  # number of samples (larger grid gives finer control)

    # initialise with a random binary feasible vector
    cur_h = random_binary(n, rng)
    cur_O = compute_max_overlap(cur_h)
    best_h = cur_h.copy()
    best_O = cur_O
    target_O = 194  # corresponds to C5 <= 0.3808 for n=1024

    # main local‑search loop
    while time.time() - start < budget_s - 0.5:
        # compute worst shift for the current configuration
        cross = np.correlate(cur_h, cur_h, mode="full")[n - 1 :]
        k = int(np.argmin(cross))
        # attempt a directed swap that reduces the overlap for shift k
        i = j = None
        if k != 0:
            end = n - k
            hi = cur_h[:end]
            hi_k = cur_h[k:]
            bad = np.where((hi == 1) & (hi_k == 0))[0]   # (1,0) pairs for shift k
            good = np.where((hi == 0) & (hi_k == 1))[0]  # (0,1) pairs for shift k
            if bad.size and good.size:
                i = int(bad[rng.integers(bad.size)])
                j = int(good[rng.integers(good.size)])
        # if no directed move is possible, perform a random 1↔0 swap
        if i is None:
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            i = int(ones_idx[rng.integers(ones_idx.size)])
            j = int(zeros_idx[rng.integers(zeros_idx.size)])

        # execute the swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # evaluate the new configuration
        new_O = compute_max_overlap(cur_h)

        # accept the swap if it does not increase the worst overlap
        if new_O <= cur_O:
            cur_O = new_O
            if new_O < best_O:
                best_O = new_O
                best_h = cur_h.copy()
                if best_O <= target_O:
                    break  # reached target bound
        else:
            # revert the move
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # final bound in the required format
    best_c = 2.0 * best_O / n
    return best_h.astype(float).tolist(), float(best_c), int(n)
