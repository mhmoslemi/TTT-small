def adjust_sum_binary(h_bin, rng):
    n = h_bin.size
    target = n // 2
    cur = int(h_bin.sum())
    diff = cur - target
    if diff > 0: # too many ones
        # set diff ones to zero
        ones_idx = np.flatnonzero(h_bin == 1)
        # random pick diff
        sel = rng.choice(ones_idx, size=diff, replace=False)
        h_bin[sel] = 0
    elif diff < 0:
        zeros_idx = np.flatnonzero(h_bin == 0)
        sel = rng.choice(zeros_idx, size=-diff, replace=False)
        h_bin[sel] = 1
    return h_bin

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    # shift index: from 0 to 2n-2, shift = idx - (n-1)
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # i from 0 to n-k-1: g[i] += wi * (1 - h[i+k])
            g[: n - k] += wi * one_minus_h[k:]
            # i from k to n-1: g[i] -= wi * h[i-k]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            # i from kk to n-1: g[i] += wi * (1 - h[i-kk])
            g[kk:] += wi * one_minus_h[: n - kk]
            # i from 0 to n-kk-1: g[i] -= wi * h[i+kk]
            g[: n - kk] -= wi * h[kk:]
    return g

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = GLOBAL_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / GLOBAL_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global GLOBAL_TARGET, GLOBAL_BETA
    n = h0.size

def binary_simulated_annealing(h_bin, rng, deadline, init_T_factor=0.02, decay=0.99995):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # Temperature initialization
    T = max(1e-12, cur_c * init_T_factor)
    # maintain ones and zeros lists for O(1) swaps
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    it = 0
    while time.time() < deadline:
        # random indices
        i_idx = rng.integers(0, len(ones))
        j_idx = rng.integers(0, len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        # compute new C
        c_new = compute_c5(cur_h)
        accept = False
        if c_new < cur_c:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-(c_new - cur_c) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_c = c_new
            # update ones and zeros using swap trick
            ones[i_idx] = j
            zeros[j_idx] = i
            if c_new < best_c:
                best_c = c_new
                best_h = cur_h.copy()
                # optionally reset temperature? not needed
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        # decay temperature
        T *= decay
        it += 1
        # Optional early exit if T is extremely low and no improvement for long?
        # Not needed
    return best_h, best_c
