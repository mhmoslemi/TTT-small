def continuous_sa(h0, rng, time_budget, target_c=None):
    start_local = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0 = 0.01
    T_end = 1e-6
    while True:
        now = time.time()
        elapsed = now - start_local
        if elapsed >= time_budget:
            break
        # linear cooling
        T = max(T_end, T0 * (1 - elapsed / time_budget))
        # scale for epsilon
        # we choose random i,j
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # choose direction randomly
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = max_eps * (T / T0) * rng.random()
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = max_eps * (T / T0) * rng.random()
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        # No need to enforce sum because mass moved exactly.
        # Determine c_new
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        # early exit
        if target_c is not None and best_c <= target_c:
            break
    return best_h, best_c

def binary_sa(n, rng, time_budget, target_c=None):
    start_local = time.time()
    best_h = None
    best_c = np.inf
    # current best we might start from random? We'll generate random each iteration.
    # For time efficiency, we can generate a fresh random binary vector each iteration, then SA loop.
    # We'll attempt many SA runs within time_budget.
    # For each run, we start with a random binary vector and perform local swaps.
    # We could limit inner loops to avoid too many operations.
    # Simpler: single SA run that iterates swaps until time budget expire.
    # Starting vector:
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    # create arrays of indices for ones and zeros
    ones_arr = ones.copy()
    zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
    T0 = 0.03
    T_end = 1e-5
    while True:
        now = time.time()
        elapsed = now - start_local
        if elapsed >= time_budget:
            break
        T = max(T_end, T0 * (1 - elapsed / time_budget))
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            # update indices
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if target_c is not None and best_c <= target_c:
                    return best_h, best_c
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def enforce_sum(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff = target - h.sum()
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff = target - h.sum()
        attempts += 1
    # final clip
    np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_refine(h, rng, step=0.015, max_iter=15):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        # apply step with decreasing magnitude
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add:
                h[i] += add
                diff = target - h.sum()
        else:
            sub = min(-diff, h[i])
            if sub:
                h[i] -= sub
                diff = target - h.sum()
        attempts += 1
    np.clip(h, 0.0, 1.0, out=h)
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def binary_sa(n: int, rng: np.random.Generator, time_budget: float, target_c: float = None):
    """Binary simulated annealing on a vector with exactly n/2 ones."""
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    ones_arr = ones.copy()
    zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
    T0, T_end = 0.03, 1e-5
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        T = max(T_end, T0 * (1 - (now - start) / time_budget))
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i, j = ones_arr[i_pos], zeros_arr[j_pos]
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            ones_arr[i_pos], zeros_arr[j_pos] = j, i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if target_c is not None and best_c <= target_c:
                    return best_h, best_c
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c


def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float, target_c: float = None):
    """Simulated annealing on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    N = h.size
    T0, T_end = 0.01, 1e-6
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        T = max(T_end, T0 * (1 - (now - start) / time_budget))
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = max_eps * (T / T0) * rng.random()
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = max_eps * (T / T0) * rng.random()
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_curr = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                if target_c is not None and best_c <= target_c:
                    break
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.015, max_iter: int = 15):
    """Iterative refinement using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h
