def simulated_annealing(n, target_sum, rng, time_limit):
    # initialize
    h = np.zeros(n, dtype=float)
    # randomly assign half ones
    h[: n//2] = 1.0
    rng.shuffle(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    current_c = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_c = current_c

    # parameters
    T0 = 0.001
    Tmin = 1e-6
    max_iter = 10_000_000  # high but break early by time

    start = time.time()
    it = 0
    while True:
        if time.time() - start > time_limit:
            break
        # pick i (1) and j (0)
        # loop to find i:
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        # compute new_corr
        # compute h_rev = h[::-1] (before swap)
        h_rev = h[::-1]
        # apply swap delta
        # start_i = n-1 - i
        start_i = n - 1 - i
        start_j = n - 1 - j
        # create new_corr copy? we can compute delta and add to current corr vector to get new_corr
        new_corr = corr.copy()
        # term1
        new_corr[start_i : start_i + n] += -1 * q
        new_corr[start_j : start_j + n] += 1 * q
        # term2
        new_corr[i : i + n] += 1 * h_rev
        new_corr[j : j + n] += -1 * h_rev
        # term3
        center = n - 1
        new_corr[center] -= 2
        offset = j - i
        new_corr[center + offset] += 1
        new_corr[center - offset] += 1
        # compute new C
        new_c = np.max(new_corr) * (2.0 / n)
        # temperature decays
        # compute fraction of SA time used
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + Tmin * frac
        # accept?
        if new_c < current_c or rng.random() < np.exp(-(new_c - current_c) / T):
            # accept swap
            # update h, q, corr, current_c
            h[i] = 0.0
            h[j] = 1.0
            q[i] = 1.0
            q[j] = 0.0
            corr = new_corr
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        # optionally update best from current if best_c not improved
        it += 1
        if it % 10000 == 0:
            # check time again
            if time.time() - start > time_limit:
                break
    return best_h, best_c

import time
import numpy as np

# ----- globals for SLSQP ----- #
CURRENT_TARGET = None
CURRENT_BETA = None

# ----- basic helpers ----- #
def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple iterative redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] using correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


# ----- fast swap update ----- #
def apply_swap(h: np.ndarray, q: np.ndarray, i: int, j: int) -> np.ndarray:
    """Return new correlation after swapping h[i]=1 with h[j]=0 (binary case)."""
    n = h.size
    # slices
    start_i = n - 1 - i
    start_j = n - 1 - j
    h_rev = h[::-1]        # length n
    new_corr = q_corr.copy()  # q_corr will be provided as global for speed
    # term1: delta_h * q
    new_corr[start_i : start_i + n] += -1.0 * q
    new_corr[start_j : start_j + n] += 1.0 * q
    # term2: h * delta_q
    new_corr[i : i + n] += 1.0 * h_rev
    new_corr[j : j + n] += -1.0 * h_rev
    # term3: delta_h * delta_q
    center = n - 1
    new_corr[center] -= 2.0
    offset = j - i
    new_corr[center + offset] += 1.0
    new_corr[center - offset] += 1.0
    return new_corr


# ----- simulated annealing ----- #
def simulated_annealing(n: int, target_sum: float, rng: np.random.Generator,
                       time_limit: float, h_start: np.ndarray = None):
    """Binary SA using fast O(n) correlation updates."""
    if h_start is None:
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)
    else:
        h = (h_start >= 0.5).astype(float)
        h = enforce_sum(h, target_sum, rng)

    q = 1.0 - h
    # initial full correlation (global variable for speed)
    global q_corr
    q_corr = np.correlate(h, q, mode="full")
    corr = q_corr.copy()
    current_c = np.max(corr) * (2.0 / n)

    best_h = h.copy()
    best_c = current_c
    best_iter = 0

    start = time.time()
    it = 0
    # temperature schedule parameters
    T0 = 0.001
    Tmin = 1e-6
    restart_limit = 200_000  # iterations without improvement before restart

    while True:
        if time.time() - start > time_limit:
            break
        # pick a 1 and a 0
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        # compute new correlation
        new_corr = apply_swap(h, q, i, j)
        new_c = np.max(new_corr) * (2.0 / n)

        # temperature
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + Tmin * frac

        # acceptance
        if new_c < current_c or rng.random() < np.exp(-(new_c - current_c) / T):
            # accept swap
            h[i] = 0.0
            h[j] = 1.0
            q[i] = 1.0
            q[j] = 0.0
            corr = new_corr
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                best_iter = it
                if best_c < 0.3808:
                    # excellent, stop early
                    break

        # restart if stale
        if it - best_iter > restart_limit and (time.time() - start) < time_limit * 0.9:
            h = np.zeros(n, dtype=float)
            h[: n // 2] = 1.0
            rng.shuffle(h)
            q = 1.0 - h
            q_corr = np.correlate(h, q, mode="full")
            corr = q_corr.copy()
            current_c = np.max(corr) * (2.0 / n)
            best_iter = it

        it += 1
        if it % 10_000 == 0:
            # periodic time check
            if time.time() - start > time_limit:
                break

    return best_h, best_c


# ----- main entry point ----- #
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # choose a reasonably large n
    n = 2048
    target_sum = n / 2.0

    # initialise from parent if available
    h_start = None
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

    time_for_sa = budget_s * 0.60

        n, target_sum, rng, time_for_sa, h_start=h_start
    )

    # ----- gradient refinement phase ----- #
    time_elapsed = time.time() - start
    time_left = budget_s - time_elapsed
    if time_left > budget_s * 0.15:
        # a few gradient rounds
        best_h = gradient_refine(best_h, target_sum, rng, step=0.02, max_iter=30)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c
            best_h = best_h.copy()
        time_elapsed = time.time() - start
        time_left = budget_s - time_elapsed

    # ----- final SLSQP refinement (if time permits) ----- #
    if time_left > budget_s * 0.05:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            h_s = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=200)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
