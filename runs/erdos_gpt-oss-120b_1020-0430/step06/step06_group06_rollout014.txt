def repair_sum(h, target_sum, rng):
    current = int(h.sum())
    diff = target_sum - current
    if diff > 0:
        zeros = np.where(h == 0)[0]
        chosen = rng.choice(zeros, size=diff, replace=False)
        h[chosen] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        chosen = rng.choice(ones, size=-diff, replace=False)
        h[chosen] = 0
    return h

    # init population

    # replace one individual with best of parent if available: If initial_h_values present, we can upsample:

    best_h = population[best_idx].copy()
    best_c5 = c5s[best_idx]

    while time.time() - start < budget_s - 2:
        # evaluate fitness if not done
        # compute sorted indices
        sorted_idx = np.argsort(c5s)  # ascending
        # elites
        new_population = [population[i].copy() for i in sorted_idx[:elite]]
        # generate rest
        while len(new_population) < pop_size:
            # select parents via tournament selection
            # For each parent, do tournament of size 3
            def select():
                participants = rng.integers(0, pop_size, size=3)
                best = participants[np.argmin(c5s[participants])]
                return population[best]
            parent1 = select()
            parent2 = select()
            # crossover
            mask = rng.random(n) < 0.5
            child = np.where(mask, parent1, parent2).astype(np.float64)
            # mutation
            mut_mask = rng.random(n) < p_mut
            child[mut_mask] = 1.0 - child[mut_mask]
            # repair sum
            child = repair_sum(child, target_sum, rng)
            new_population.append(child)
        population = new_population
        # evaluate new population
        c5s = np.empty(pop_size)
        for i, h in enumerate(population):
            c5s[i] = compute_c5(h)
        # update best
        idx = np.argmin(c5s)
        if c5s[idx] < best_c5:
            best_c5 = c5s[idx]
            best_h = population[idx].copy()
    # After loop, local improvement
    # hill climbing on best_h
    # Keep swapping bits until time runs out
    # Use rng for picking random 1 and 0 indices
    while time.time() - start < budget_s - 0.5:
        # choose random 1-index and 0-index
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        best_h[i], best_h[j] = 0.0, 1.0
        new_c5 = compute_c5(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # revert
            best_h[i], best_h[j] = 1.0, 0.0
    # Ensure final repair: sum

def repair_sum(h, target, rng):
    current = int(h.sum())
    diff = target - current
    if diff > 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) < diff:
            # unlikely
            zeros = np.arange(len(h))[h == 0]
        chosen = rng.choice(zeros, size=diff, replace=False)
        h[chosen] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        chosen = rng.choice(ones, size=-diff, replace=False)
        h[chosen] = 0
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with N points."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return float(np.max(corr) * (2.0 / n))

def repair_sum(h: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    """Adjust binary array h so that sum(h) == target."""
    current = int(round(h.sum()))
    diff = target - current
    if diff > 0:
        zeros = np.where(h == 0)[0]
        chosen = rng.choice(zeros, size=diff, replace=False)
        h[chosen] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        chosen = rng.choice(ones, size=-diff, replace=False)
        h[chosen] = 0.0
    return h

def random_solution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly n//2 ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def tournament_select(pop, c5s, rng, size=3):
    """Select one individual via tournament."""
    participants = rng.integers(0, len(pop), size=size)
    best = participants[np.argmin(c5s[participants])]
    return pop[best]
