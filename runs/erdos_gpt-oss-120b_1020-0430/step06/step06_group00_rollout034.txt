def enforce_sum(h, rng):

    # Simple redistribution: find indices where we can add/subtract.

    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        # choose random index with capacity
        if diff > 0:
            # need to add
            idx = rng.choice(np.where(h < 1.0)[0])
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = rng.choice(np.where(h > 0.0)[0])
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        attempts += 1
    # final clip just in case

    return h

def local_optimize(h0, rng, max_iters=200, step0=0.1, step_decay=0.95):
    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        # gradient
        k, grad, _ = worst_shift_and_gradient(h)
        d = -grad
        # Project onto sum zero
        d -= d.mean()
        # Compute max feasible step
        pos = d > 1e-12
        neg = d < -1e-12
        max_step = np.inf
        if np.any(pos):
            max_step = min(max_step, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            max_step = min(max_step, np.min((-h[neg]) / d[neg]))
        if max_step == np.inf:
            # all zero direction?
            break
        step = step0 * (step_decay ** it)
        step = min(step, max_step)
        if step <= 1e-14:
            continue
        h_new = h + step * d
        # Clip and enforce sum

            best_h = h_new.copy()

        if c_new < c_curr:

    return best_h

def worst_shift_and_gradient(h):
    n = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    max_c = corr.max() * (2.0 / n)
    return k, grad, max_c

Now compute_corr_fft:

def compute_corr_fft(h):
    n = h.size
    rev = (1.0 - h)[::-1]
    # zero pad to length 2*n
    fft_h = np.fft.rfft(h, n=2*n)
    fft_rev = np.fft.rfft(rev, n=2*n)
    conv = np.fft.irfft(fft_h * fft_rev, n=2*n)
    return conv

def upsample_h(h, new_n):
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

Now slsqp_optimize and global variables:

CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # return log-sum-exp / beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Fast correlation via FFT
def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    n = h.size
    rev = (1.0 - h)[::-1]
    fft_h = np.fft.rfft(h, n=2 * n)
    fft_rev = np.fft.rfft(rev, n=2 * n)
    return np.fft.irfft(fft_h * fft_rev, n=2 * n)


def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx  (discrete version)."""
    return compute_corr_fft(h).max() * (2.0 / h.size)


# --------------------------------------------------------------
# Sum‑preserving projection into [0,1]^n
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:
        if diff > 0.0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        attempts += 1
    return np.clip(h, 0.0, 1.0)


# --------------------------------------------------------------
# Gradient of the worst (largest) overlap shift
def worst_shift_and_gradient(h: np.ndarray):
    n = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    max_c = corr.max() * (2.0 / n)
    return k, grad, max_c


# --------------------------------------------------------------
# Projected gradient descent on the worst shift
def local_optimize(h0: np.ndarray, rng: np.random.Generator,
                   max_iters: int = 200, step0: float = 0.1,
                   step_decay: float = 0.95) -> np.ndarray:
    h = h0.copy()
    n = h.size
    target = n / 2.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    for it in range(max_iters):
        _, grad, _ = worst_shift_and_gradient(h)
        d = -grad
        d -= d.mean()                     # keep sum unchanged
        pos = d > 1e-12
        neg = d < -1e-12
        max_step = np.inf
        if np.any(pos):
            max_step = min(max_step, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            max_step = min(max_step, np.min((-h[neg]) / d[neg]))
        if max_step == np.inf:
            break
        step = min(step0 * (step_decay ** it), max_step)
        if step <= 1e-14:
            continue
        h_new = np.clip(h + step * d, 0.0, 1.0)
        diff = h_new.sum() - target
        if abs(diff) > 1e-12:
            h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new < c_curr:
            h, c_curr = h_new, c_new
    return best_h


# --------------------------------------------------------------
# Random‑restart local search on a fixed grid size
def random_restart_local_search(n: int, end_time: float, rng: np.random.Generator,
                               step0: float = 0.2, decay: float = 0.96,
                               max_iters: int = 80):
    best_h, best_c = None, np.inf
    while time.time() < end_time:
        h0 = enforce_sum(rng.random(n), rng)
        h_opt = local_optimize(h0, rng, max_iters=max_iters,
                               step0=step0, step_decay=decay)
        c = compute_c5(h_opt)
        if c < best_c:
            best_c, best_h = c, h_opt.copy()
    return best_h, best_c


# --------------------------------------------------------------
# Linear up‑sampling
def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# --------------------------------------------------------------
# Soft‑max SLSQP (global refinement)
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    end_total = start + budget_s

    best_h = None
    best_c = np.inf

    def maybe_update(h):
        nonlocal best_h, best_c
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- initial solution (if provided) ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        maybe_update(h0)

    # fall back to a random start if nothing else
    if best_h is None:
        h0 = enforce_sum(rng.random(256), rng)
        maybe_update(h0)

    # ---- Phase A: low‑resolution random local search (n=128) ----
    if time.time() < end_total:
        phase_end = min(end_total, time.time() + min(0.10 * budget_s, 120.0))
        h_a, c_a = random_restart_local_search(128, phase_end, rng)
        if h_a is not None:
            maybe_update(h_a)

    # ---- Phase B: upsample to 256 and local refinement ----
    if best_h is not None and time.time() < end_total:
        h_b = upsample_h(best_h, 256)
        h_b = enforce_sum(np.clip(h_b, 0.0, 1.0), rng)
        phase_end = min(end_total, time.time() + min(0.12 * budget_s, 180.0))
        # limit iterations so we stay within the allocated window
        max_iters = 400
        h_opt = local_optimize(h_b, rng, max_iters=max_iters,
                               step0=0.2, step_decay=0.97)
        maybe_update(h_opt)

    # ---- Phase C: upsample to 512, SLSQP + local refinement ----
    if best_h is not None and time.time() < end_total:
        h_c = upsample_h(best_h, 512)
        h_c = enforce_sum(np.clip(h_c, 0.0, 1.0), rng)
        # SLSQP with increasing β
        remaining = end_total - time.time()
        slsqp_end = time.time() + remaining * 0.20
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() > slsqp_end:
                break
            iters = 600 if beta < 2000 else 300
            h_c = slsqp_optimize(h_c, beta, rng, max_iter=iters)
        maybe_update(h_c)
        # local gradient descent
        if time.time() < end_total:
            remaining = end_total - time.time()
            max_iters = min(800, int(remaining / 0.001))
            h_opt = local_optimize(h_c, rng, max_iters=max_iters,
                                   step0=0.15, step_decay=0.96)
            maybe_update(h_opt)

    # ---- Phase D: upsample to 1024, extensive local search ----
    if best_h is not None and time.time() < end_total:
        h_d = upsample_h(best_h, 1024)
        h_d = enforce_sum(np.clip(h_d, 0.0, 1.0), rng)
        remaining = end_total - time.time()
        if remaining > 5.0:
            est_iter = int(remaining / 0.001)
            max_iters = min(est_iter, 8000)
            h_opt = local_optimize(h_d, rng, max_iters=max_iters,
                                   step0=0.12, step_decay=0.965)
            maybe_update(h_opt)

    # ---- Phase E: final SLSQP fine‑tuning (if time permits) ----
    if best_h is not None and time.time() < end_total:
        remaining = end_total - time.time()
        if remaining > 5.0:
            for beta in (20000., 40000., 80000.):
                iters = 200
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=iters)
                best_h = enforce_sum(best_h, rng)
                best_c = compute_c5(best_h)

    # final sanitisation
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
