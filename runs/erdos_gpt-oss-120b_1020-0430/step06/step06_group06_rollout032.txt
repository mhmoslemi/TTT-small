def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n
    # Convert to float
    hf = np.fft.rfft(h.astype(np.float64), n=L)
    # compute reversed complement
    rev_comp = np.flip(1.0 - h.astype(np.float64))
    gf = np.fft.rfft(rev_comp, n=L)
    conv = np.fft.irfft(hf * gf, n=L)
    # The linear correlation length is 2*n-1
    max_val = np.max(conv[:2*n-1])
    return max_val * (2.0 / n)

import time, math, numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise binary step function -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        target = n // 2
        top = np.argpartition(h0, -target)[-target:]
        h = np.zeros(n, dtype=np.uint8)
        h[top] = 1
    else:
        n = 512
        target = n // 2
        h = np.zeros(n, dtype=np.uint8)
        h[rng.choice(n, target, replace=False)] = 1

    factor = 2.0 / n
    L = 2 * n

    # ----- FFT helpers -----
    def correlation_fft(arr: np.ndarray) -> np.ndarray:
        """Linear correlation of arr with (1-arr) using FFT."""
        hf = np.fft.rfft(arr.astype(np.float64), n=L)
        rev = np.flip(1.0 - arr.astype(np.float64))
        gf = np.fft.rfft(rev, n=L)
        conv = np.fft.irfft(hf * gf, n=L)
        return conv[:2 * n - 1]

    def compute_c5(arr: np.ndarray) -> float:
        return np.max(correlation_fft(arr)) * factor

    # ----- greedy monotone improvement -----
    def greedy_improve(arr: np.ndarray) -> np.ndarray:
        cur = compute_c5(arr)
        while True:
            corr = correlation_fft(arr)
            idx = int(np.argmax(corr))
            k = idx - (n - 1)
            if k == 0 or corr[idx] <= 0:
                # no positive overlapping shift left
                break

            kk = k if k > 0 else -k
            if k > 0:
                i_range = slice(0, n - kk)
                a = arr[i_range]
                b = arr[kk:]
            else:  # k < 0
                i_range = slice(-k, n)
                a = arr[i_range]
                b = arr[: n + k]   # n+k = n-kk
            # candidates that improve this shift
            i_cand = np.where((a == 1) & (b == 0))[0]

            if i_cand.size and j_cand.size:
                i_loc = rng.choice(i_cand)
                j_loc = rng.choice(j_cand)
                i = i_range.start + i_loc
                j = i_range.start + j_loc

                arr[j] = 1
                new = compute_c5(arr)
                if new <= cur:

                arr[i] = 1
                arr[j] = 0
            # no improving swap for this k; discard it
            corr[idx] = -np.inf
            if np.max(corr) <= 0:
                break
