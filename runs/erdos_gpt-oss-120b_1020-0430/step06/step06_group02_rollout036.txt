def enforce_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # increase
        # find indices where h < 1
        idx = np.where(h < 1.0)[0]
        for i in idx:
            room = 1.0 - h[i]
            add = min(diff, room)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # decrease
        idx = np.where(h > 0.0)[0]
        diff = -diff
        for i in idx:
            room = h[i]
            sub = min(diff, room)
            h[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    # if still diff, use random adjustments (unlikely)
    return h

def softmax_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    exps = np.exp(beta * (corr - max_corr))
    weight = exps / exps.sum()
    grad = np.zeros_like(h)
    # loop over lags
    for idx, w in enumerate(weight):
        s = idx - (n - 1)
        if s >= 0:
            i_start = 0
            i_end = n - s
        else:
            i_start = -s
            i_end = n
        i = np.arange(i_start, i_end)
        i_s = i + s
        grad[i] += w * (1.0 - h[i_s])
        grad[i_s] -= w * h[i]
    return grad

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h to have sum == target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        for i in idx:
            room = 1.0 - h[i]
            add = min(diff, room)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.where(h > 0.0)[0]
        for i in idx:
            room = h[i]
            sub = min(diff, room)
            h[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

    rng = np.random.default_rng(seed)
    start = time.time()
    budget = float(budget_s)

    # ----- initialization -------------------------------------------------

    else:
        n = 512
        h = rng.random(n)
    target = n / 2.0
    h = enforce_sum(h, target)

    best_h = h.copy()
    best_c = compute_c5(h)

    def maybe_update(candidate_h, candidate_c):
        nonlocal best_h, best_c
        if candidate_c < best_c:
            best_c = candidate_c
            best_h = candidate_h.copy()

    # ----- coarse subgradient descent (30 % of budget) --------------------
    step0 = 0.2
    it = 0
    max_iters = 20000
    while it < max_iters and time.time() - start < budget * 0.3:
        step = step0 * (0.99 ** it)
        grad = worst_gradient(h)
        nh = h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, target)
        c = compute_c5(nh)
        maybe_update(nh, c)
        h = nh
        it += 1
        if best_c <= 0.3808:

    # ----- Adam on smooth max (up to 90 % of budget) --------------------
    betas = [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0, 200000.0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for beta in betas:
        if time.time() - start >= budget * 0.9:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        # allocate modest iterations per beta

            t += 1

            max_corr = corr.max()

            w = exps / exps.sum()

            grad = np.zeros_like(h)
            for idx, w_l in enumerate(w):
