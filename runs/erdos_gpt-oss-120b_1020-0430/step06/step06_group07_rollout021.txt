def solve_one(n, start_h, rng, time_budget):
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    # aggressive swaps
    while time.time() - start < time_budget * 0.7:  # 70% of allotted for this run
        grad, k = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] <= 0 or grad[j] >= 0:
            break
        eps = min(h[i], 1 - h[j])
        if eps <= 0:
            break
        h[i] -= eps
        h[j] += eps
        # sum unchanged, box constraints maintained.
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            if best_c <= target:
                return best_h, best_c
    # subgradient polishing with decreasing step size
    t=0
    while time.time() - start < time_budget:
        t+=1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        # step = project_box_sum(h - alpha*grad, n/2)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= target: break
    return best_h, best_c

import time
import numpy as np

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] using autocorrelation."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    o_max = n / 2.0 - np.min(cross)
    return o_max * (2.0 / n)


def worst_gradient(h: np.ndarray):
    """Gradient of the worst‑shift overlap O_k and its shift index."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via bisection on the Lagrange multiplier."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample_to_n(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear up‑sample `h` to length `n_target` and re‑project to the feasible set."""
    x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_target / 2.0)


def solve_one_restart(h0: np.ndarray, n: int, rng: np.random.Generator,
                      time_budget: float, target_c: float):
    """Aggressive swap descent followed by projected sub‑gradient polishing."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    # ---- aggressive swap phase (≈70% of allotted time) ----
    aggressive_limit = time_budget * 0.7
    while time.time() - start < aggressive_limit:
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] <= 0 or grad[j] >= 0:
            break
        eps = min(h[i], 1.0 - h[j])
        if eps <= 0:
            break
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= target_c:
                return best_h, best_c

    # ---- projected sub‑gradient polishing (remaining time) ----
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h = project_box_sum(h - alpha * grad, n / 2.0)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= target_c:
                break
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    target_c = 0.3808

    # ---- choose grid size -------------------------------------------------

        best_c = compute_c5(best_h)
