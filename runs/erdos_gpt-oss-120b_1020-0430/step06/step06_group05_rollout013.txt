def project_simplex_box(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
```

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_sum_and_bounds(v: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x : 0 ≤ x ≤ 1, sum(x) = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    h = project_to_sum_and_bounds(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- optimisation parameters -----------------------------------------
    beta_list = [5, 15, 40, 120, 400, 1500, 6000, 25000]
    max_iter_per_beta = 2000
    step0 = 0.5
    no_improve = 0

    total_iters = 0

    while time.time() - start < time_limit:
        for beta in beta_list:

            while it < max_iter_per_beta and time.time() - start < time_limit:
