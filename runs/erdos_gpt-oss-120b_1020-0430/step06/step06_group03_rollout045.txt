def compute_c5_fft(h):
    n = h.size
    N = 1 << (2*n - 1).bit_length()  # next power of two >=2n
    # compute FFT of h and (1-h)
    fh = np.fft.rfft(h, n=N)
    f1mh = np.fft.rfft(1 - h, n=N)
    conv = np.fft.irfft(fh * f1mh, n=N)
    # conv length N, but we need indices 0..2n-2 for full correlation (zero padded)
    corr = conv[: 2*n - 1]
    return np.max(corr) * (2.0 / n)

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def local_descent(h, steps, step_scale, target):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for s in range(steps):
        grad = worst_gradient(cur_h)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        step = step_scale * (1.0 - s / steps) / norm
        v = cur_h - step * grad
        cur_h = project_onto_box_sum(v, target)
        cur_c = compute_c5(cur_h)
    return cur_h, cur_c

def sa_optimize(h0, target, time_budget, start_time, rng):
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    T0 = best_c * 0.2 + 1e-4
    it = 0
    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        T = T0 * (1.0 - frac)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_down = cur_h[i]
        max_up = 1.0 - cur_h[j]
        if max_down + max_up == 0.0:
            continue
        eps = (rng.random() * (max_down + max_up)) - max_down
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        delta = nc - cur_c
        accept = False
        if delta <= 0.0:
            accept = True
        elif T > 0.0 and rng.random() < math.exp(-delta / T):
            accept = True
        if accept:
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_c, best_h = nc, nh.copy()
        it += 1
        if it % 20000 == 0:
            cur_h, cur_c = local_descent(cur_h, steps=200, step_scale=0.5, target=target)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter):
    n = h0.size
    target = n / 2.0
    def soft_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def eq_con(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_con}
    res = minimize(soft_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                              "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, target)

def resample_h(old_h, n_new):
    n_old = len(old_h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    init_h = globals().get('initial_h_values', None)

    best_overall_h = None
    best_overall_c = float('inf')

    # List of (n, fraction of remaining time) pairs
    phases = [(256, 0.35), (200, 0.30), (300, 0.25)]

    for n, frac in phases:
        remaining = budget_s - (time.time() - start)
        if remaining <= 0.0:
            break
        t_alloc = remaining * frac
        target = n / 2.0

        if init_h is not None:
            h0 = resample_h(np.asarray(init_h, dtype=float), n)
        else:
            h0 = np.zeros(n)
            ones = rng.choice(n, n // 2, replace=False)
            h0[ones] = 1.0

        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)

        h_opt, c_opt = sa_optimize(h0, target, t_alloc, time.time(), rng)

        if c_opt < best_overall_c:
            best_overall_c, best_overall_h = c_opt, h_opt.copy()

    # Final refinement with SLSQP and a bit of local descent
    remaining = budget_s - (time.time() - start)
    if best_overall_h is not None and remaining > 5.0:
        n = best_overall_h.size
        target = n / 2.0
        # a short SLSQP cascade
        for beta, iters in ((2000, int(remaining * 0.3)), (8000, int(remaining * 0.3)), (20000, int(remaining * 0.3))):
            if iters < 10:
                continue
            best_overall_h = slsqp_optimize(best_overall_h, beta, max_iter=iters)
        # a final local descent
        best_overall_h, best_overall_c = local_descent(best_overall_h, steps=2000, step_scale=0.4, target=target)

    # Ensure feasibility
    best_overall_h = np.clip(best_overall_h, 0.0, 1.0)
    best_overall_h = project_onto_box_sum(best_overall_h, best_overall_h.size / 2.0)
    final_c = compute_c5(best_overall_h)

    return best_overall_h.tolist(), float(final_c), int(best_overall_h.size)
