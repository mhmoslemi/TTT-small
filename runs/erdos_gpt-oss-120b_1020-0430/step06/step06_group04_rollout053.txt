def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    # Determine initial N (choose 720 as base)
    base_n = 720
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        # parent may be same length or smaller; we can use as is if same length else interpolate.
        n_parent = len(parent)
        if n_parent == base_n:
            h = np.array(parent, dtype=np.float64)
        else:
            xs = np.linspace(0.0, 2.0, n_parent, endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            h = np.interp(xt, xs, parent)
    else:
        # random fractional vector with sum=base_n/2
        h = rng.random(base_n)
        h = h / h.sum() * (base_n/2)
        np.clip(h, 0.0, 1.0, out=h)
    adjust_sum_exact(h, rng)

    # obtain initial best
    best_h = h.copy()
    best_c = compute_c5(best_h)
    best_n = base_n

    # define beta schedule
    beta_schedule = [10, 50, 200, 1000, 5000, 20000, 50000, 100000]

    # main loop
    for beta in beta_schedule:
        if time.time()>deadline:
            break
        # number of iterations per beta depends on remaining time
        # allocate more to early betas? Actually later betas important. we can compute remaining budget and assign.
        # simple: set iters = 200. but ensure time.
        iters = 200
        step_size = 0.2
        for it in range(iters):
            if time.time()>deadline:
                break
            g = gradient_smoothmax(best_h, beta)
            # Project gradient onto sum constraint hyperplane:
            g = g - np.mean(g)  # because sum(g)=0 ensures moving along hyperplane
            # Step
            cand = best_h - step_size * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
                # maybe increase step size a bit
                step_size = min(step_size*1.1, 0.5)
            else:
                step_size = max(step_size*0.8, 0.01)
        # optional after each beta: small refinement on worst shift
        if time.time()>deadline:
            break
        # worst shift refinement
        for _ in range(10):
            if time.time()>deadline:
                break
            g_ws = worst_shift_gradient(best_h)
            g_ws = g_ws - np.mean(g_ws)
            cand = best_h - step_size * g_ws
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        # optional SLSQP polish at this beta:
        if time.time()>deadline:
            break
        # We need to define SLSQP objective for given beta
        # Let's call slsqp_optimize
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()
    # final polish with highest beta
    if time.time()<deadline:
        best_h = slsqp_optimize(best_h, 200000, rng, max_iter=500)
        best_c = compute_c5(best_h)

    # possibly try binary rounding and hill climbing if time remains
    if time.time()<deadline:
        # binary rounding
        h_bin = np.rint(best_h).astype(np.float64)
        adjust_sum_exact(h_bin, rng)
        # hill climb
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, deadline)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin.copy()
    # optional upsample if time left
    if time.time()<deadline and best_n*2<=1440:
        n_up = best_n*2
        # upsample by repeating each element twice
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        # quick refinement similar to above but fewer iterations
        for beta in [20000,50000,100000]:
            if time.time()>deadline: break
            for _ in range(50):
                if time.time()>deadline: break
                g = gradient_smoothmax(h_up,beta)
                g = g - np.mean(g)
                cand = h_up - 0.2*g
                np.clip(cand,0.0,1.0,out=cand)
                adjust_sum_exact(cand,rng)
                if compute_c5(cand) < compute_c5(h_up):
                    h_up = cand
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()
            best_n = n_up

    final_c = compute_c5(best_h)
    best_h = np.clip(best_h,0.0,1.0)
    adjust_sum_exact(best_h, rng)
    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np
from scipy.optimize import minimize

# ---------------------------------------------------------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

# ---------- SLSQP smooth‑max helper ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = (c < cur_c) or (rng.random() < np.exp(-(c - cur_c) / T))
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    base_n = 720
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        n_parent = len(parent)
        if n_parent == base_n:
            h = np.array(parent, dtype=np.float64)
        else:
            xs = np.linspace(0.0, 2.0, n_parent, endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            h = np.interp(xt, xs, parent)
    else:
        h = rng.random(base_n)
        h = h / h.sum() * (base_n / 2.0)
        np.clip(h, 0.0, 1.0, out=h)
    adjust_sum_exact(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)
    best_n = base_n

    beta_schedule = [10, 50, 200, 1000, 5000, 20000, 50000, 100000]
    step_size = 0.2

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        iters = 200
        for _ in range(iters):
            if time.time() > deadline:
                break
            g = gradient_smoothmax(best_h, beta)
            g -= np.mean(g)                     # keep sum constraint
            cand = best_h - step_size * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
                step_size = min(step_size * 1.1, 0.5)
            else:
                step_size = max(step_size * 0.8, 0.01)

        # small worst‑shift refinement
        for _ in range(10):
            if time.time() > deadline:
                break
            g_ws = worst_shift_gradient(best_h)
            g_ws -= np.mean(g_ws)
            cand = best_h - step_size * g_ws
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand

        # SLSQP polish
        if time.time() > deadline:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()
            best_n = base_n

    # final high‑beta polish
    if time.time() < deadline:
        best_h = slsqp_optimize(best_h, 200000, rng, max_iter=600)
        best_c = compute_c5(best_h)

    # binary rounding & hill climb if time permits
    if time.time() < deadline:
        h_bin = np.rint(best_h).astype(np.float64)
        adjust_sum_exact(h_bin, rng)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, deadline)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin.copy()
            best_n = base_n

    # optional up‑sample for extra refinement
    if (time.time() < deadline) and (best_n * 2 <= 1440):
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        for beta in (20000, 50000, 100000):
            if time.time() > deadline:
                break
            for _ in range(50):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h_up, beta)
                g -= np.mean(g)
                cand = h_up - 0.2 * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                if compute_c5(cand) < compute_c5(h_up):
                    h_up = cand
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up
        # binary hill climb on up‑sampled version
        if time.time() < deadline:
            h_bin = np.rint(h_up).astype(np.float64)
            adjust_sum_exact(h_bin, rng)
            h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, deadline)
            if c_bin < best_c:
                best_c, best_h, best_n = c_bin, h_bin.copy(), n_up

    final_c = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    return best_h.tolist(), float(final_c), int(best_n)
