def binary_sa(h0, deadline):
    n = h0.size
    L_full = 2*n - 1
    s_idx = np.arange(L_full, dtype=np.int32)
    # big arrays length
    L_big = 4*n + 5
    offset = 2*n + 2  # plenty
    ar_big = np.zeros(L_big, dtype=np.int8)
    br_big = np.zeros(L_big, dtype=np.int8)
    ar = h0.astype(np.int8)
    br = 1 - ar
    ar_big[offset:offset+n] = ar
    br_big[offset:offset+n] = br
    v = np.correlate(ar, br, mode='full')
    v = v.astype(np.int32)
    max_v = v.max()
    # target max to achieve
    target_max = 0.3808 * n / 2.0  # float
    # initial temperature
    T0 = max(1.0, (max_v - target_max) * 0.5)
    # best tracking
    best_h = ar.copy()
    best_max = max_v
    best_v = v.copy()
    rng = np.random.default_rng()
    start = time.time()
    last_improve = start

    while True:
        now = time.time()
        if now >= deadline:
            break
        # temperature schedule
        progress = (now - start) / (deadline - start)
        T = T0 * (1 - progress)
        if T < 1e-9:
            T = 1e-9
        # pick i,j opposite
        i = rng.integers(n)
        if ar[i] == 1:
            # need zero j
            while True:
                j = rng.integers(n)
                if ar[j] == 0:
                    break
        else:  # ar[i]==0
            while True:
                j = rng.integers(n)
                if ar[j] == 1:
                    break
        # compute delta
        # pre compute constant offsets for i,j
        idx_i_minus = i + 2*(n-1) - s_idx + offset
        idx_i_plus = i + s_idx + offset
        idx_j_minus = j + 2*(n-1) - s_idx + offset
        idx_j_plus = j + s_idx + offset
        # fetch arrays:
        # note we can use br_big[idx_i_minus] etc.
        delta = (- br_big[idx_i_minus] + ar_big[idx_i_plus] + br_big[idx_j_minus] - ar_big[idx_j_plus]).astype(np.int32)
        new_v = v + delta
        new_max = new_v.max()
        # acceptance
        accept = False
        if new_max <= max_v:
            accept = True
        else:
            # probability
            if rng.random() < np.exp(-(new_max - max_v) / T):
                accept = True
        if accept:
            # update structures
            v = new_v
            max_v = new_max
            # swap values in ar, br
            ar_i = ar[i]
            ar_j = ar[j]
            # they're opposite
            ar[i] = 0
            ar[j] = 1
            br[i] = 1
            br[j] = 0
            # update big arrays
            ar_big[offset + i] = 0
            br_big[offset + i] = 1
            ar_big[offset + j] = 1
            br_big[offset + j] = 0
            # update best
            if max_v < best_max:
                best_max = max_v
                best_h = ar.copy()
                best_v = v.copy()
                last_improve = now
        # optional break if stuck for long without improvement
        if now - last_improve > 30.0:
            # break to restart
            break

    # after loop, return best_h and best_max
    return best_h.astype(np.int8), best_max

def adjust_binary_sum(h, target, rng):
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        choose = rng.choice(zeros, size=diff, replace=False)
        h[choose] = 1
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        choose = rng.choice(ones, size=-diff, replace=False)
        h[choose] = 0
    return h

def adjust_binary_sum(h, target, rng):

    cur = int(h.sum())

    diff = target - cur

    if diff > 0:

        zeros = np.flatnonzero(h == 0)

        chosen = rng.choice(zeros, size=diff, replace=False)

        h[chosen] = 1

    elif diff < 0:

        ones = np.flatnonzero(h == 1)

        chosen = rng.choice(ones, size=-diff, replace=False)

        h[chosen] = 0

    return h

def run(seed=42, budget_s=1100, **kwargs):

    import time, numpy as np

    rng = np.random.default_rng(seed)

    start = time.time()

    deadline = start + budget_s * 0.97

    parent = globals().get('initial_h_values', None)

    if parent is not None:

        n = len(parent)

    else:

        n = 720

    target_C5 = 0.3808

    best_h = None

    best_max = np.iinfo(np.int32).max

    # number of restarts wise on time

    restarts = 3

    for r in range(restarts):

        if time.time() >= deadline:

            break

        # initialize h0

        if r == 0 and parent is not None:

            h0 = np.rint(parent).astype(np.int8)

            h0 = adjust_binary_sum(h0, n // 2, rng)

        else:

            h0 = np.zeros(n, dtype=np.int8)

            ones_idx = rng.choice(n, size=n // 2, replace=False)

            h0[ones_idx] = 1

        # run SA

        h_candidate, max_candidate = binary_sa(h0, deadline, rng, target_C5)

        if max_candidate < best_max:

            best_max = max_candidate

            best_h = h_candidate.copy()

        # early exit if good enough

        if best_max <= target_C5 * n / 2.0:

            break

    if best_h is None:

        # fallback simple half ones

        best_h = np.zeros(n, dtype=np.int8)

        best_h[: n // 2] = 1

        best_max = np.correlate(best_h, 1 - best_h, mode='full').max()

    # compute final C5

    final_c = float(best_max) * (2.0 / n)

    # ensure values are floats 0.0/1.0

    best_h = best_h.astype(np.float64)

    best_h = np.clip(best_h, 0.0, 1.0)

    # Ensure sum correct

    cur_sum = best_h.sum()

    target_sum = n / 2.0

    if abs(cur_sum - target_sum) > 1e-12:

        # adjust if needed

        diff = int(round(target_sum - cur_sum))

        if diff > 0:

            zeros = np.flatnonzero(best_h == 0)

            chosen = rng.choice(zeros, size=diff, replace=False)

            best_h[chosen] = 1.0

        elif diff < 0:

            ones = np.flatnonzero(best_h == 1)

            chosen = rng.choice(ones, size=-diff, replace=False)

            best_h[chosen] = 0.0

    return best_h.tolist(), final_c, int(n)

import time
import numpy as np

def adjust_binary_sum(h, target, rng):
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        if zeros.size:
            chosen = rng.choice(zeros, size=diff, replace=False)
            h[chosen] = 1
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        if ones.size:
            chosen = rng.choice(ones, size=-diff, replace=False)
            h[chosen] = 0
    return h

def binary_sa(h0, deadline, rng, target_c5=0.3808):
    n = h0.size
    ar = h0.astype(np.int32)
    br = 1 - ar
    offset = n - 1
    L_full = 2 * n - 1
    s_idx = np.arange(L_full, dtype=np.int32)
    L_big = 3 * n - 2
    ar_big = np.zeros(L_big, dtype=np.int32)
    br_big = np.zeros(L_big, dtype=np.int32)
    ar_big[offset:offset + n] = ar
    br_big[offset:offset + n] = br
    v = np.correlate(ar, br, mode="full").astype(np.int32)
    max_v = v.max()
    target_max = target_c5 * n / 2.0
    T0 = max(1.0, (max_v - target_max) * 0.5)
    best_h = ar.copy()
    best_max = max_v
    last_improve = time.time()
    center_idx = n - 1
    two_n_minus_2 = 2 * (n - 1)
    start = time.time()

    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start) / (deadline - start)
        T = max(T0 * (1 - progress), 1e-12)

        # pick a 1 and a 0
        while True:
            p = rng.integers(n)
            if ar[p] == 1:
                break
        while True:
            q = rng.integers(n)
            if ar[q] == 0:
                break

        const_p_minus = p + two_n_minus_2
        const_q_minus = q + two_n_minus_2

        idx_p_minus = const_p_minus - s_idx
        idx_q_minus = const_q_minus - s_idx
        idx_p_plus = p + s_idx
        idx_q_plus = q + s_idx

        delta = (
            ar_big[idx_p_minus]
            - ar_big[idx_q_minus]
            - br_big[idx_p_plus]
            + br_big[idx_q_plus]
        )
        # apply second order term
        delta[center_idx] -= 2
        d1 = (q - p) + (n - 1)
        d2 = (p - q) + (n - 1)
        delta[d1] += 1
        delta[d2] += 1

        new_v = v + delta
        new_max = new_v.max()

        accept = False
        if new_max <= max_v:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - max_v) / T):
                accept = True

        if accept:
            v = new_v
            max_v = new_max
            # update vectors
            ar[p] = 0
            ar[q] = 1
            br[p] = 1
            br[q] = 0
            ar_big[offset + p] = 0
            br_big[offset + p] = 1
            ar_big[offset + q] = 1
            br_big[offset + q] = 0
            if max_v < best_max:
                best_max = max_v
                best_h = ar.copy()
                last_improve = now

        if now - last_improve > 30.0:
            break

    return best_h.astype(np.int8), best_max

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
