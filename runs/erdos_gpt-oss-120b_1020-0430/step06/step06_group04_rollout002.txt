def compute_c5_fft(h):

    # pad to length 2*n-1
    size = 2*n-1

    m = 1 << (size-1).bit_length()
    # compute FFT of h and (1-h)
    fh = np.fft.rfft(h, n=m)

    conv = np.fft.irfft(fh * fbh, n=m)
    # conv contains circular convolution of length m; linear with padding
    # extract first size elements
    linear = conv[:size]
    return linear.max() * (2.0 / n)

def simulated_annealing(h0, rng, deadline):
    h = h0.copy()
    n = h.shape[0]
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    # temperature schedule
    T0 = cur_val * 0.2 + 1e-4
    it = 0
    last_improve_it = 0
    while time.time() < deadline:
        it += 1
        # choose move type
        if rng.random() < 0.1:
            # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # swap values
            hi, hj = h[i], h[j]
            if hi == hj:
                continue
            h[i], h[j] = hj, hi
            new_val = compute_c5(h)
            dE = new_val - cur_val
            T = T0 * (0.9995 ** it)
            if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                    last_improve_it = it
            else:
                h[i], h[j] = hi, hj
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi = h[i]; hj = h[j]
            lo = max(-hi, hj - 1.0)
            hiDelta = min(1.0 - hi, hj)
            if lo >= hiDelta:
                continue
            delta = rng.random() * (hiDelta - lo) + lo
            h[i] = hi + delta
            h[j] = hj - delta
            new_val = compute_c5(h)
            dE = new_val - cur_val
            T = T0 * (0.9995 ** it)
            if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                    last_improve_it = it
            else:
                # revert
                h[i] = hi
                h[j] = hj

        # restart if stuck
        if it - last_improve_it > 200000:
            # random perturb on best_h
            h = best_h.copy()
            # add small noise to few indices while preserving sum
            k = max(1, n // 200)  # ~0.5% of points

            # apply noise and then project to [0,1] and adjust sum
            h[idxs] += noise
            np.clip(h, 0.0, 1.0, out=h)
            # adjust sum exact
            adjust_sum_exact(h, rng)
            cur_val = compute_c5(h)
            last_improve_it = it
    return best_h, best_val

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g
```

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

import time
import numpy as np

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """In‑place modification of h so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- worst‑shift gradient ----------
def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient coming from the current worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g


def gradient_refine(h: np.ndarray, rng: np.random.Generator, steps: int, deadline: float) -> np.ndarray:
    """Few steps of worst‑shift descent with line search."""
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        cur_c = compute_c5(h)
        best_h, best_c = h, cur_c
        for alpha in (0.2, 0.1, 0.05, 0.02):
            cand = h - alpha * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
    return h


# ---------- simulated annealing on continuous h ----------
def simulated_annealing(h0: np.ndarray, rng: np.random.Generator, deadline: float) -> tuple[np.ndarray, float]:
    """SA that preserves the integral exactly by pairwise adjustments."""
    h = h0.copy()
    n = h.size
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val

    T = cur_val * 0.5 + 0.001  # initial temperature
    decay = 0.99995
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1
        # choose move type
        if rng.random() < 0.1:  # swap move (preserves sum automatically)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            if hi == hj:
                continue
            h[i], h[j] = hj, hi
            new_val = compute_c5(h)
            dE = new_val - cur_val
            if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_h = new_val, h.copy()
                    last_improve = it
            else:
                h[i], h[j] = hi, hj
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hiDelta = min(1.0 - hi, hj)
            if lo >= hiDelta:
                continue
            delta = rng.random() * (hiDelta - lo) + lo
            h[i] = hi + delta
            h[j] = hj - delta
            new_val = compute_c5(h)
            dE = new_val - cur_val
            if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_h = new_val, h.copy()
                    last_improve = it
            else:
                h[i], h[j] = hi, hj

        # temperature decay
        T = max(T * decay, 1e-12)

        # restart if stuck for a long while
        if it - last_improve > 300_000:
            h = best_h.copy()
            k = max(1, n // 200)
            idxs = rng.choice(n, size=k, replace=False)
            h[idxs] += rng.normal(scale=0.02, size=k)
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)
            cur_val = compute_c5(h)
            last_improve = it

    return best_h, best_val


# ---------- main entry ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96  # safety margin

    # ----- base grid -----
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        n0 = len(parent)
        h0 = np.array(parent, dtype=np.float64)
    else:
        n0 = 720
        h0 = rng.random(n0)
        h0 = h0 / h0.sum() * (n0 / 2.0)
        np.clip(h0, 0.0, 1.0, out=h0)
        adjust_sum_exact(h0, rng)

    # ----- phase 1: SA on base grid -----
    phase1_dead = start + (budget_s * 0.55)
    if phase1_dead > deadline:
        phase1_dead = deadline
    h_best, c_best = simulated_annealing(h0, rng, phase1_dead)

    # ----- short gradient polish -----
    h_best = gradient_refine(h_best, rng, steps=30, deadline=deadline)
    c_best = compute_c5(h_best)

    # ----- optional up‑sampling -----
    now = time.time()
    if now + 30 < deadline:
        n_up = n0 * 2
        h_up = np.repeat(h_best, 2).astype(np.float64)
        adjust_sum_exact(h_up, rng)

        # SA on upsampled version with remaining time
        h_up_best, c_up_best = simulated_annealing(h_up, rng, deadline)

        # gradient polish upsampled
        h_up_best = gradient_refine(h_up_best, rng, steps=30, deadline=deadline)
        c_up_best = compute_c5(h_up_best)

        if c_up_best < c_best:
            h_best, c_best, n0 = h_up_best, c_up_best, n_up

    # ----- final safety clip & exact sum -----
    np.clip(h_best, 0.0, 1.0, out=h_best)
    adjust_sum_exact(h_best, rng)
    final_c = compute_c5(h_best)

    return h_best.tolist(), float(final_c), int(n0)
