def conv_fft(a, b):
    n = a.size + b.size - 1
    N = 1 << (n-1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    c = np.fft.irfft(C, n=N)[:n]
    return c

    # Project v onto [0,1]^n with sum s.
    n = v.size

    v_clipped = np.clip(v, 0.0, 1.0)
    # If sum already equals s (within tolerance), return.
    sum_v = v_clipped.sum()

        return v_clipped
    # If sum_v > s need to decrease some values; if less, increase.
    # Use water-filling: treat initial values v_clipped, find λ such that sum(min(max(v_i - λ, 0), 1)) = s.
    # Equivalent to projection onto capped simplex.
    # We can use binary search for λ between -1 and 1.
    lo = -1.0
    hi = 1.0
    for _ in range(50):

        proj = np.clip(v_clipped - mid, 0.0, 1.0)
        total = proj.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v_clipped - (lo+hi)/2.0, 0.0, 1.0)
    return proj

def conv_fft(a, b):
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    c = np.fft.irfft(C, n=N)[:n]
    return c

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, target: float) -> np.ndarray:
    """Adjust h so that sum(h)=target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 20000, step_scale: float = 0.5):
    """Gradient‑directed hill climbing on the worst shift."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    target = n / 2.0
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- FFT convolution ----------
def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    c = np.fft.irfft(C, n=N)[:n]
    return c


# ---------- smooth‑min gradient ascent ----------
def gradient_ascent_step(h: np.ndarray, beta: float, eta: float,
                        target: float, rng: np.random.Generator):
    """One ascent step on the smooth min of self‑correlations."""
    n = h.size
    corr = np.correlate(h, h, mode="full")
    Q = corr[n - 1 :]                     # Q_k for k=0..n-1
    Q_min = Q.min()
    # soft‑min weights (stable)
    w = np.exp(-beta * (Q - Q_min))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # backward contribution
    back = conv_fft(w, h)[:n]
    # forward contribution (use reversed w)
    fwd = conv_fft(w[::-1], h)[n - 1 : 2 * n - 1]
    grad = back + fwd                     # ∇S = back + fwd
    h_new = h + eta * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, rng, target)
    return h_new, Q_min


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----- initialise ----- 
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 512
        h = rng.random(n)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- smooth‑min ascent ----- 
    beta_seq = [5.0, 20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0, 50000.0]
    eta = 0.2
    for beta in beta_seq:
        if time.time() - start > total_budget * 0.90:
            break
        # allocate iterations based on remaining time (roughly 300 steps per second)
        remaining = total_budget - (time.time() - start)
        max_iters = int(min(30000, remaining * 300))
        for it in range(max_iters):
            if time.time() - start > total_budget * 0.95:
                break
            h, Q_min = gradient_ascent_step(h, beta, eta, target, rng)
            c = 1.0 - (2.0 * Q_min) / n       # exact C5 from Q_min
            if c < best_c:
                best_c, best_h = c, h.copy()
                if best_c < 0.3808:          # early stop if already good enough
                    break
        eta *= 0.7
        # occasional small random tweak to escape plateaus
        if rng.random() < 0.2 and time.time() - start < total_budget * 0.95:
            h += (rng.random(n) - 0.5) * 0.05
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng, target)

    # ----- final local refinement ----- 
    if time.time() - start < total_budget * 0.97:
        best_h, best_c = guided_hill_climb(best_h.copy(), rng,
                                           max_iters=20000, step_scale=0.5)

    # ----- sanitise and return ----- 
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
