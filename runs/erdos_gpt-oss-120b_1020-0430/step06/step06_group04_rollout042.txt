def compute_delta(b, i, j, offset):
    n = b.shape[0]
    L = 2*n - 1
    delta = np.zeros(L, dtype=np.int32)
    # make padded view
    b_padded = np.concatenate([np.zeros(offset, dtype=np.int8), b, np.zeros(offset, dtype=np.int8)])
    # i contributions
    slice_i = b_padded[i : i + L]
    delta_i = -1
    delta += delta_i * (slice_i + slice_i[::-1])
    # j contributions
    slice_j = b_padded[j : j + L]
    delta_j = 1
    delta += delta_j * (slice_j + slice_j[::-1])
    # cross term
    idx1 = (j - i) + offset
    idx2 = (i - j) + offset
    delta[idx1] -= 1
    delta[idx2] -= 1
    # correct for double-count at zero shift
    delta[offset] -= delta_i * b[i] + delta_j * b[j]
    return delta

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = shift_idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    n = h.size
    while time.time() < deadline:
        for beta in betas:
            for _ in range(it_per_beta):
                if time.time() > deadline:
                    return h
                g = gradient_smoothmax(h, beta)
                if np.linalg.norm(g) < 1e-14:
                    continue
                best_local = h
                best_c = compute_c5(h)
                for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                    step = base_step * mult
                    cand = h - step * g
                    np.clip(cand, 0.0, 1.0, out=cand)
                    adjust_sum_exact(cand, rng)
                    c = compute_c5(cand)
                    if c < best_c:
                        best_c = c
                        best_local = cand
                h = best_local
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        h0 = np.array(parent, dtype=np.float64)
        n = h0.size
    else:
        n = 720
        h0 = rng.random(n)
        h0 = h0 / h0.sum() * (n / 2.0)
        np.clip(h0, 0.0, 1.0, out=h0)

    # binary rounding of initial vector
    b = np.rint(h0).astype(np.int8)
    target_ones = n // 2
    cur = b.sum()
    diff = target_ones - cur
    if diff > 0:
        zeros = np.flatnonzero(b == 0)
        idx = rng.choice(zeros, size=diff, replace=False)
        b[idx] = 1
    elif diff < 0:
        ones = np.flatnonzero(b == 1)
        idx = rng.choice(ones, size=-diff, replace=False)
        b[idx] = 0

    # autocorrelation and auxiliary data
    a = np.correlate(b, b, mode="full").astype(np.int32)
    offset = n - 1
    L = 2 * n - 1
    min_a = a.min()
    best_c = 1.0 - 2.0 * min_a / n
    best_h = b.astype(np.float64).copy()
    target_min_a = int(np.ceil((1.0 - 0.3808) * n / 2.0))

    # simulated‑annealing on binary vector
    T = 0.5
    decay = 0.9995
    b_padded = np.concatenate([np.zeros(offset, dtype=np.int8), b, np.zeros(offset, dtype=np.int8)])

    while time.time() < deadline:
        min_a = a.min()
        if min_a >= target_min_a:
            best_c = 1.0 - 2.0 * min_a / n
            best_h = b.astype(np.float64).copy()
            break

        worst_idxs = np.where(a == min_a)[0]
        idx_shift = worst_idxs[rng.integers(0, worst_idxs.size)]
        k = idx_shift - offset

        # candidate positions that could increase a_k
        if k >= 0:
            max_i = n - k
            i_range = np.arange(max_i)
            mask_i = (b[i_range] == 0) & (b[i_range + k] == 1)
            i_cand = i_range[mask_i]
            mask_j = (b[i_range] == 1) & (b[i_range + k] == 0)
            j_cand = i_range[mask_j]
        else:
            d = -k
            i_range = np.arange(d, n)
            mask_i = (b[i_range] == 0) & (b[i_range - d] == 1)
            i_cand = i_range[mask_i]
            mask_j = (b[i_range] == 1) & (b[i_range - d] == 0)
            j_cand = i_range[mask_j]

        if i_cand.size and j_cand.size:
            i = i_cand[rng.integers(0, i_cand.size)]
            j = j_cand[rng.integers(0, j_cand.size)]
        else:
            ones = np.flatnonzero(b)
            zeros = np.flatnonzero(~b)
            i = ones[rng.integers(0, ones.size)]
            j = zeros[rng.integers(0, zeros.size)]

        # compute delta of autocorrelation caused by swapping i (1) and j (0)
        slice_i = b_padded[i : i + L].astype(np.int32)
        slice_j = b_padded[j : j + L].astype(np.int32)
        delta_i = -1
        delta_j = 1
        delta = delta_i * (slice_i + slice_i[::-1]) + delta_j * (slice_j + slice_j[::-1])
        idx1 = (j - i) + offset
        idx2 = (i - j) + offset
        delta[idx1] -= 1
        delta[idx2] -= 1
        delta[offset] -= delta_i * b[i] + delta_j * b[j]

        new_a = a + delta
        new_min_a = new_a.min()
        accept = False
        if new_min_a > min_a:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(min_a - new_min_a) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            b[i] = 1
            b[j] = 0
            a = new_a
            b_padded = np.concatenate([np.zeros(offset, dtype=np.int8), b, np.zeros(offset, dtype=np.int8)])
            cur_c = 1.0 - 2.0 * new_min_a / n
            if cur_c < best_c:
                best_c = cur_c
                best_h = b.astype(np.float64).copy()
        T *= decay

    # optional fractional polish if time remains
    if time.time() < deadline:
        h_frac = best_h.copy()
        betas = [20, 80, 200, 800, 2000]
        step = 0.2
        h_frac = adjust_sum_exact(h_frac, rng)
        for beta in betas:
            for _ in range(6):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h_frac, beta)
                cand = h_frac - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand.copy()
                h_frac = cand

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)

import time
import numpy as np

# ------------------- basic utilities -------------------
def compute_c5(h: np.ndarray) -> float:
    """C₅ for a (floating) step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Project h onto {0≤h≤1, sum(h)=n/2} by local moves."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log∑exp(β·overlap) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k:] -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:] += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g


def gradient_descent_smoothmax(
    h: np.ndarray,
    betas,
    base_step: float,
    it_per_beta: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """Simple projected GD on the smooth‑max surrogate."""
    n = h.size
    while time.time() < deadline:
        for beta in betas:
            for _ in range(it_per_beta):
                if time.time() > deadline:
                    return h
                g = gradient_smoothmax(h, beta)
                if np.linalg.norm(g) < 1e-14:
                    continue
                best_local = h
                best_val = compute_c5(h)
                for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                    step = base_step * mult
                    cand = h - step * g
                    np.clip(cand, 0.0, 1.0, out=cand)
                    adjust_sum_exact(cand, rng)
                    val = compute_c5(cand)
                    if val < best_val:
                        best_val = val
                        best_local = cand
                h = best_local
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        h0 = np.array(parent, dtype=np.float64)
        n = h0.size
    else:
        n = 720
        h0 = rng.random(n)
        h0 = h0 / h0.sum() * (n / 2.0)
        np.clip(h0, 0.0, 1.0, out=h0)

    # ----- binary rounding and exact sum -----
    b = np.rint(h0).astype(np.int8)
    target_ones = n // 2
    cur = b.sum()
    diff = target_ones - cur
    if diff > 0:
        zeros = np.flatnonzero(b == 0)
        b[rng.choice(zeros, size=diff, replace=False)] = 1
    elif diff < 0:
        ones = np.flatnonzero(b == 1)
        b[rng.choice(ones, size=-diff, replace=False)] = 0

    # ----- autocorrelation and initial bound -----
    a = np.correlate(b, b, mode="full").astype(np.int32)
    offset = n - 1
    L = 2 * n - 1
    min_a = a.min()
    best_c = 1.0 - 2.0 * min_a / n
    best_h = b.astype(np.float64).copy()
    target_min_a = int(np.ceil((1.0 - 0.3808) * n / 2.0))

    # ----- simulated annealing on the binary vector -----
    T = 0.5
    decay = 0.9995
    b_padded = np.concatenate(
        [np.zeros(offset, dtype=np.int8), b, np.zeros(offset, dtype=np.int8)]
    )

    while time.time() < deadline:
        min_a = a.min()
        if min_a >= target_min_a:
            best_c = 1.0 - 2.0 * min_a / n
            best_h = b.astype(np.float64).copy()
            break

        # worst shift(s)
        worst_idxs = np.where(a == min_a)[0]
        idx_shift = worst_idxs[rng.integers(0, worst_idxs.size)]
        k = idx_shift - offset

        # candidate positions that could raise a_k
        if k >= 0:
            max_i = n - k
            i_range = np.arange(max_i)
            i_mask = (b[i_range] == 0) & (b[i_range + k] == 1)
            i_cand = i_range[i_mask]
            j_mask = (b[i_range] == 1) & (b[i_range + k] == 0)
            j_cand = i_range[j_mask]
        else:
            d = -k
            i_range = np.arange(d, n)
            i_mask = (b[i_range] == 0) & (b[i_range - d] == 1)
            i_cand = i_range[i_mask]
            j_mask = (b[i_range] == 1) & (b[i_range - d] == 0)
            j_cand = i_range[j_mask]

        if i_cand.size and j_cand.size:
            i = i_cand[rng.integers(0, i_cand.size)]
            j = j_cand[rng.integers(0, j_cand.size)]
        else:
            zeros = np.flatnonzero(b == 0)
            ones = np.flatnonzero(b == 1)
            i = zeros[rng.integers(0, zeros.size)]
            j = ones[rng.integers(0, ones.size)]

        # ----- delta of autocorrelation caused by setting i←1, j←0 -----
        delta_i = 1 - b[i]          # will be +1 (i currently 0)
        delta_j = -b[j]             # will be -1 (j currently 1)

        slice_i = b_padded[i : i + L].astype(np.int32)
        slice_j = b_padded[j : j + L].astype(np.int32)

        delta = delta_i * (slice_i + slice_i[::-1]) + delta_j * (slice_j + slice_j[::-1])

        # diagonal (s=0) contribution Δ_i²+Δ_j²
        delta[offset] += delta_i * delta_i + delta_j * delta_j

        # cross contributions for shifts ±(j‑i)
        cross = delta_i * delta_j
        if cross != 0:
            idx1 = (j - i) + offset
            idx2 = (i - j) + offset
            delta[idx1] += cross
            delta[idx2] += cross

        new_a = a + delta
        new_min_a = new_a.min()

        accept = False
        if new_min_a > min_a:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(min_a - new_min_a) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            b[i] = 1
            b[j] = 0
            a = new_a
            b_padded = np.concatenate(
                [np.zeros(offset, dtype=np.int8), b, np.zeros(offset, dtype=np.int8)]
            )
            cur_c = 1.0 - 2.0 * new_min_a / n
            if cur_c < best_c:
                best_c = cur_c
                best_h = b.astype(np.float64).copy()
        T *= decay

    # ----- optional fractional polishing if time permits -----
    if time.time() < deadline:
        h_frac = best_h.copy()
        h_frac = adjust_sum_exact(h_frac, rng)
        betas = [20, 80, 200, 800, 2000]
        step = 0.2
        for beta in betas:
            for _ in range(6):
                if time.time() > deadline:
                    break
                g = gradient_smoothmax(h_frac, beta)
                cand = h_frac - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand.copy()
                h_frac = cand

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
