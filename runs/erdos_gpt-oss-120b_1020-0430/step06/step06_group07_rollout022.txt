    pos[zeros] = np.arange(w)
    # compute A
    A_full = np.correlate(h, h, mode='full')
    A = A_full[n - 1:].astype(np.int32)
    return h, ones, zeros, pos, A

def simulated_annealing(h, ones, zeros, pos, A, w, n, target_minA, rng, time_budget):
    start = time.time()
    cur_minA = A[1:].min()
    best_h = h.copy()
    best_minA = cur_minA
    best_A = A.copy()
    while time.time() - start < time_budget:
        # temperature linear
        elapsed = time.time() - start
        T = max(1e-4, 1.0 - elapsed / time_budget) # from 1 to 0
        # propose swap
        i = ones[rng.integers(w)]
        j = zeros[rng.integers(w)]
        delta = compute_delta_swap(i, j, h)
        new_A = A + delta
        new_minA = new_A[1:].min()
        delta_cur = new_minA - cur_minA
        if delta_cur > 0 or rng.random() < np.exp(delta_cur / T):
            # accept
            # update structures
            p = pos[i]
            q = pos[j]
            ones[p] = j
            zeros[q] = i
            pos[i] = q
            pos[j] = p
            h[i] = 0
            h[j] = 1
            A[:] = new_A
            cur_minA = new_minA
            if cur_minA > best_minA:
                best_minA = cur_minA
                best_h = h.copy()
                best_A = A.copy()
                if best_minA >= target_minA:
                    break
    return best_h, best_minA, best_A, ones, zeros, pos

def greedy_improve(h, ones, zeros, pos, A, w, n, target_minA, rng):
    cur_minA = A[1:].min()
    while True:
        worst_k = A[1:].argmin() + 1
        k = worst_k
        # shift_plus: h shifted left by k (i.e., h[i+k]), assign zeros for overflow
        shift_plus = np.empty(n, dtype=np.uint8)
        shift_plus[:n - k] = h[k:]
        shift_plus[n - k:] = 0
        # shift_minus: h shifted right by k (i.e., h[i-k]), zeros for underflow
        shift_minus = np.empty(n, dtype=np.uint8)
        shift_minus[k:] = h[:n - k]
        shift_minus[:k] = 0
        loss = shift_plus[ones] + shift_minus[ones]
        gain = shift_plus[zeros] + shift_minus[zeros]
        min_loss = loss.min()
        max_gain = gain.max()
        if max_gain <= min_loss:
            break
        # find indices with these values (choose random among ties)
        i_candidates = ones[loss == min_loss]
        j_candidates = zeros[gain == max_gain]
        i_idx = i_candidates[rng.integers(len(i_candidates))]
        j_idx = j_candidates[rng.integers(len(j_candidates))]
        # compute delta before swapping
        delta = compute_delta_swap(i_idx, j_idx, h)
        # apply swap
        p = pos[i_idx]
        q = pos[j_idx]
        ones[p] = j_idx
        zeros[q] = i_idx
        pos[i_idx] = q
        pos[j_idx] = p
        h[i_idx] = 0
        h[j_idx] = 1
        A += delta
        cur_minA = A[1:].min()
        if cur_minA > target_minA:
            break
    return h, A, cur_minA

def adam_optimize(h0, target, beta, rng, time_budget):
    # h0: initial h satisfying sum target
    h = h0.copy()
    n = h.size
    # Adam parameters
    lr = 0.1  # initial learning rate
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    while time.time() - start < time_budget:
        t += 1
        # compute O_k (max objective) and gradient of softmax
        # correlation of h and (1 - h)
        corr_full = np.correlate(h, 1 - h, mode='full')
        O = (2.0 / n) * corr_full[n - 1 :]  # length n
        max_O = O.max()
        # compute softmax probabilities
        exp_term = np.exp(beta * (O - max_O))
        p = exp_term / exp_term.sum()
        # compute sum_forward via convolution
        # conv_f = np.convolve(h, p[::-1], mode='full')
        # sum_forward = conv_f[n - 1 : n - 1 + n]
        conv_f = np.convolve(h, p[::-1], mode='full')
        sum_forward = conv_f[n - 1 : n - 1 + n]
        # sum_backward via convolution p * h
        conv_b = np.convolve(p, h, mode='full')
        sum_backward = conv_b[:n]
        grad = 1.0 - sum_forward - sum_backward
        # adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h + step
        # project
        h_new = project_box_sum(h_new, target)
        # evaluate
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

import time
import numpy as np

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


# ----------------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ----------------------------------------------------------------------
def adam_optimize(h0: np.ndarray, target_sum: float, beta: float,
                  rng: np.random.Generator, time_budget: float):
    """Adam on the soft‑max approximation of max O_k."""
    h = h0.copy()
    n = h.size
    lr = 0.12
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    t = 0
    while time.time() - start < time_budget:
        t += 1
        # O_k for all k ≥ 0
        corr_full = np.correlate(h, 1 - h, mode="full")
        O = (2.0 / n) * corr_full[n - 1 :]          # length n
        max_O = O.max()
        # soft‑max probabilities
        exp_term = np.exp(beta * (O - max_O))
        p = exp_term / exp_term.sum()
        # convolution for forward and backward sums
        conv_f = np.convolve(h, p[::-1], mode="full")
        sum_forward = conv_f[n - 1 : n - 1 + n]      # length n
        conv_b = np.convolve(p, h, mode="full")
        sum_backward = conv_b[:n]                    # length n
        # gradient of soft‑max objective
        grad = (2.0 / n) * (1.0 - sum_forward - sum_backward)
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h + step
        h_new = project_box_sum(h_new, target_sum)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # Choose resolution (larger n if we have enough time)
    n = 1024 if total_budget > 650 else 512
    target_sum = n / 2.0

    # Initial feasible point
    h0 = random_feasible(n, rng)
    best_h, best_c = h0, compute_c5(h0)

    # Reserve a small fraction for final polishing
    reserve = total_budget * 0.06
    time_available = total_budget - (time.time() - start) - reserve
    if time_available <= 0:
        time_available = 0.0

    # ----- Phase 1 : Adam with increasing β (soft‑max) -----
    betas = (1.0, 5.0, 20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0)
    per_beta = max(0.5, time_available / len(betas))
    h_cur = best_h
    for beta in betas:
        elapsed = time.time() - start
        if elapsed + per_beta > total_budget * 0.93:
            break
        h_opt, c_opt = adam_optimize(
            h_cur, target_sum, beta, rng, per_beta
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        h_cur = h_opt

    # ----- Phase 2 : Guided hill‑climb on the current best -----
    remaining = total_budget - (time.time() - start) - reserve * 0.5
    if remaining > 2.0:
        max_iters = int(min(6000, remaining * 250))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                        max_iters=max_iters,
                                        step_scale=0.7)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # ----- Final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
