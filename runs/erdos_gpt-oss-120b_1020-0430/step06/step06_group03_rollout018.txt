import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def local_hill_climb(h, c, rng, start, limit):
    step = 0.2
    while time.time() - start < limit:
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        nc = compute_c5(nh)
        if nc < c:
            h, c = nh, nc
        step *= 0.9995
    return h, c

def random_anneal(h, c, rng, start, limit):
    n = h.size
    target = n / 2.0
    temp = 0.01
    while time.time() - start < limit:
        i, j = rng.choice(n, 2, replace=False)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.5
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        delta = nc - c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h, c = nh, nc
        temp *= 0.9995
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial vector -----
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 256
        target = n / 2.0
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
        h0 = h0 + 0.1 * rng.random(n)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- evolutionary search -----
    pop = 30
    elite = 5
    sigma0 = 0.3
    # create initial population
    population = np.empty((pop, n))
    for i in range(pop):
        if i == 0:
            population[i] = best_h.copy()
        else:
            child = best_h + sigma0 * rng.normal(size=n)
            child = np.clip(child, 0.0, 1.0)
            child = project_onto_box_sum(child, target)
            population[i] = child

    while time.time() - start < budget_s * 0.85:
        # evaluate
        c_vals = np.empty(pop)
        for i in range(pop):
            c_vals[i] = compute_c5(population[i])
        elite_idx = np.argsort(c_vals)[:elite]
        # update global best
        if c_vals[elite_idx[0]] < best_c:
            best_c = float(c_vals[elite_idx[0]])
            best_h = population[elite_idx[0]].copy()
        # generate next generation
        next_pop = np.empty_like(population)
        next_pop[:elite] = population[elite_idx]
        j = elite
        while j < pop:
            parent = population[rng.choice(elite_idx)]
            progress = (time.time() - start) / (budget_s * 0.85)
            sigma = sigma0 * (1 - progress) ** 2 + 0.02
            child = parent + sigma * rng.normal(size=n)
            child = np.clip(child, 0.0, 1.0)
            child = project_onto_box_sum(child, target)
            next_pop[j] = child
            j += 1
        population = next_pop

    # ----- local refinements -----
    # hill climbing on worst-gradient direction
    best_h, best_c = local_hill_climb(
        best_h, best_c, rng, start, start + budget_s * 0.95
    )
    # random annealing moves
    best_h, best_c = random_anneal(
        best_h, best_c, rng, start, start + budget_s * 0.99
    )
    # final tiny guided perturbations
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # ensure feasibility before reporting
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
