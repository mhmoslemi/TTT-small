def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

def gradient_refine(h, target, rng, step=0.015, max_iter=10):
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h

import time
import numpy as np

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Fast C5 using FFT (full convolution)."""
    n = h.shape[0]
    N = 2 * n - 1
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(H * G, n=N)
    return conv.max() * (2.0 / n)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            delta = min(cap, diff)
            h[i] += delta
        else:
            cap = -h[i]
            if cap >= 0.0:
                continue
            delta = max(cap, diff)  # both negative
            h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def gradient_refine(h, target, rng, step=0.015, max_iter=10):
    """Local gradient descent using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_refine(h, target, rng, beta, max_iter):
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize

    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- problem size ---- #
    n = 1024                     # more points → finer resolution
    target_sum = n / 2.0
    pop_size = 80

    # ---- population init ---- #
    pop = []
    pop_c = []

    if "initial_h_values" in globals():
        hi = np.array(initial_h_values, dtype=float)
        if hi.size * 2 == n:
            hi_up = np.repeat(hi, 2)
        elif n % hi.size == 0:
            hi_up = np.repeat(hi, n // hi.size)
        else:
            hi_up = np.interp(
                np.linspace(0, 2, n, endpoint=False),
                np.linspace(0, 2, hi.size, endpoint=False),
                hi,
            )
        hi_up = np.clip(hi_up, 0.0, 1.0)
        hi_up = enforce_sum(hi_up, target_sum, rng)
        pop.append(hi_up)
        pop_c.append(compute_c5(hi_up))

    while len(pop) < pop_size:
        idx = rng.choice(n, n // 2, replace=False)
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        pop.append(h)
        pop_c.append(compute_c5(h))

    best_idx = int(np.argmin(pop_c))
    best_h = pop[best_idx].copy()
    best_c = pop_c[best_idx]

    # ---- main evolutionary loop ---- #
    while time.time() - start < budget_s * 0.85:
        # tournament selection (size 2)
        i1, j1 = rng.integers(pop_size, size=2)
        parent1 = pop[i1] if pop_c[i1] <= pop_c[j1] else pop[j1]

        i2, j2 = rng.integers(pop_size, size=2)
        parent2 = pop[i2] if pop_c[i2] <= pop_c[j2] else pop[j2]

        # crossover
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent1, parent2).astype(np.float64)
        child = enforce_sum(child, target_sum, rng)

        # mutation: swap a 1 and a 0
        if rng.random() < 0.3:
            ones = np.where(child == 1)[0]
            zeros = np.where(child == 0)[0]
            if ones.size and zeros.size:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                child[i] = 0
                child[j] = 1

        # small epsilon transfer (preserves sum)
        if rng.random() < 0.2:
            pos = np.where(child > 0)[0]
            neg = np.where(child < 1)[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                eps = min(0.1, child[i]) * rng.random()
                child[i] -= eps
                child[j] += eps

        # occasional gradient polishing
        if rng.random() < 0.05:
            child = gradient_refine(child, target_sum, rng, step=0.02, max_iter=10)

        c_child = compute_c5(child)

        # replace worst if better
        worst_idx = int(np.argmax(pop_c))
        if c_child < pop_c[worst_idx]:
            pop[worst_idx] = child
            pop_c[worst_idx] = c_child

        # update global best
        if c_child < best_c:
            best_c = c_child
            best_h = child.copy()
            if best_c < 0.38078:
                break

    # ---- final polishing on the best solution ---- #
    best_h = gradient_refine(best_h, target_sum, rng, step=0.015, max_iter=30)
    final_c = compute_c5(best_h)

    # optional SLSQP refinement if time remains
    if time.time() - start < budget_s * 0.98:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            best_h = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=200)
            final_c = compute_c5(best_h)
            if final_c < best_c:
                best_c = final_c

    # ---- enforce constraints finally ---- #
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
