def anneal_binary(h, rng, max_iter, T0):
    n = h.shape[0]
    half = n // 2
    # Setup ones, zeros, pos arrays
    ones = np.flatnonzero(h).astype(np.int64)  # shape (half,)
    zeros = np.flatnonzero(1 - h).astype(np.int64)  # shape (half,)
    pos = np.empty(n, dtype=np.int64)
    pos[ones] = np.arange(half)
    pos[zeros] = np.arange(half)
    # current C5
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    # temperature schedule
    T_final = 1e-8
    decay = math.exp(math.log(T_final / T0) / max_iter)
    T = T0
    for it in range(max_iter):
        # pick random i from ones, j from zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        # update h array
        h[i] = 0.0
        h[j] = 1.0
        # update pos, ones, zeros
        idx_i = pos[i]
        idx_j = pos[j]
        ones[idx_i] = j
        zeros[idx_j] = i
        pos[i] = idx_j
        pos[j] = idx_i
        # compute new C5
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            # revert ones, zeros, pos
            ones[idx_i] = i
            zeros[idx_j] = j
            pos[i] = idx_i
            pos[j] = idx_j
        # temperature decay
        T *= decay
    return best_h, best_c

def compute_c5(h):
    n = h.shape[0]
    conv = np.convolve(h, h, mode='full')
    min_overlap = conv.min()
    # conv includes self-overlap for shift 0: sum_i h_i * h_i = sum_i h_i = n/2. That's okay.
    # Note conv includes negative shifts too (same values).
    # So C5 = 1 - (2.0 / n) * min_overlap

        total = space.sum()

import time, math, numpy as np, scipy.optimize as opt

# ---------- core evaluation ----------
def compute_c5(h: np.ndarray) -> float:
    """C5 = 1 - (2/n) * min_i conv_i, where conv = h*h (full)."""
    n = h.shape[0]
    min_overlap = np.min(np.convolve(h, h, mode="full"))
    return 1.0 - (2.0 / n) * min_overlap


# ---------- projection onto simplex with box ----------
def enforce_sum_continuous(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h so that sum(h)=target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        space = 1.0 - h
        total = space.sum()
        if total > 0:
            h = h + diff * space / total
    else:
        space = h
        total = space.sum()
        if total > 0:
            h = h + diff * space / total
    # tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        i = np.argmin(np.abs(h - 0.5))
        h[i] += diff
    np.clip(h, 0.0, 1.0, out=h)
    return h


# ---------- subgradient of worst shift ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    conv = np.convolve(h, h, mode="full")
    idx = np.argmin(conv) - (n - 1)          # signed shift giving min overlap
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]    -= h[: n - k]
    return grad


# ---------- soft‑max (for SLSQP) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    n = x.shape[0]
    target = n / 2.0
    conv = np.convolve(x, x, mode="full")
    O = target - conv                     # O(k) = sum h - overlap
    bc = CURRENT_BETA * O
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 1200) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final projection if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum_continuous(x, CURRENT_TARGET)
    return x


# ---------- continuous simulated annealing ----------
def anneal_continuous(h: np.ndarray, rng: np.random.Generator,
                     time_limit: float, T0: float = 0.04) -> tuple:
    n = h.size
    target = n / 2.0
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    T_final = 1e-12
    total = time_limit - time.time()
    check_every = 5000
    next_check = check_every
    it = 0
    while True:
        now = time.time()
        if now >= time_limit:
            break
        # linear cooling
        frac = (now - (time_limit - total)) / total
        T = T_final + (T0 - T_final) * (1.0 - frac)

        # choose i with h[i]>0 and j with h[j]<1
        i = rng.integers(n)
        while h[i] <= 1e-12:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] >= 1.0 - 1e-12:
            j = rng.integers(n)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.07  # up to 7% of possible transfer
        h[i] -= eps
        h[j] += eps

        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps

        it += 1
        if it == next_check:
            next_check += check_every
            # cheap time check
            if time.time() >= time_limit:
                break
    return best_h, best_c


# ---------- main routine ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initial candidate (if provided) ---
    best_h = None
    best_c = float("inf")
    best_n = None
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        # enforce exact sum
        h0 = enforce_sum_continuous(h0, n0 / 2.0)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, n0

    # --- continuous annealing phase ---
    # allocate ~60% of budget to annealing
    anneal_limit = start + budget_s * 0.60
    n = 512  # primary resolution
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    h = enforce_sum_continuous(h, n / 2.0)

    h, c = anneal_continuous(h, rng, anneal_limit)
    if c < best_c:
        best_h, best_c, best_n = h.copy(), c, n
    if best_c <= 0.3808:
        final_c = compute_c5(best_h)
        return best_h.tolist(), float(final_c), int(best_n)

    # --- subgradient refinement (up to ~80% of budget) ---
    subgrad_limit = start + budget_s * 0.80
    steps = [0.25, 0.12, 0.06, 0.03, 0.015]
    target = best_h.size / 2.0
    h = best_h.copy()
    cur_c = best_c
    for step in steps:
        while time.time() < subgrad_limit:
            grad = worst_gradient(h)
            nh = h - step * grad
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum_continuous(nh, target)
            new_c = compute_c5(nh)
            if new_c < cur_c:
                h, cur_c = nh, new_c
                if new_c < best_c:
                    best_h, best_c = nh.copy(), new_c
                if best_c <= 0.3808:
                    final_c = compute_c5(best_h)
                    return best_h.tolist(), float(final_c), int(best_n)

    # --- SLSQP soft‑max polishing (remaining time) ---
    remain = start + budget_s - time.time()
    if remain > budget_s * 0.05:
        betas = [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0]
        for beta in betas:
            if time.time() >= start + budget_s * 0.99:
                break
            opt_h = slsqp_optimize(h, beta, max_iter=1000)
            c = compute_c5(opt_h)
            if c < best_c:
                best_h, best_c = opt_h.copy(), c
                h = opt_h
                if best_c <= 0.3808:
                    break

    # final safety projection & return
    best_h = enforce_sum_continuous(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
