import time
import math
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# basic utilities
# -------------------------------------------------------------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 20000:
        idx = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
        else:
            add = max(diff, -h[idx])
        if add != 0.0:
            h[idx] += add
            diff = target - h.sum()
        it += 1
    return h


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 using the min autocorrelation formulation."""
    n = h.size
    corr = np.correlate(h, h, mode="full")
    conv = corr[n - 1 :]                # shifts 0 .. n-1
    min_conv = np.min(conv[1:])         # ignore shift 0
    return 1.0 - min_conv / (n / 2.0)   # C5 = 1 - min_conv/(n/2)


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def thue_morse_binary(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.zeros(n, dtype=np.float64)
    for i in range(n):
        seq[i] = bin(i).count("1") & 1
    return seq


# -------------------------------------------------------------
# Simulated annealing on the binary space
# -------------------------------------------------------------
def simulated_annealing(h_start: np.ndarray,
                        time_limit: float,
                        rng: np.random.Generator) -> (np.ndarray, float):
    """Return a binary vector with large min autocorrelation."""
    n = h_start.size
    h = np.rint(h_start).astype(np.float64)       # force binary
    # initial min autocorrelation
    corr = np.correlate(h, h, mode="full")
    conv = corr[n - 1 :]
    min_conv = np.min(conv[1:])

    best_h = h.copy()
    best_min = min_conv

    # temperature based on possible improvement
    potential = (n / 2.0) - min_conv
    T = max(0.1, potential * 0.5)
    decay = 0.9999

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # pick a random 1 and a random 0 to swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform the swap
        h[i] = 0.0
        h[j] = 1.0

        # evaluate new min autocorrelation
        corr = np.correlate(h, h, mode="full")
        conv = corr[n - 1 :]
        new_min = np.min(conv[1:])

        delta = new_min - min_conv
        if delta > 0 or rng.random() < math.exp(delta / T):
            # accept
            min_conv = new_min
            if min_conv > best_min:
                best_min = min_conv
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

        it += 1
        if it % 1000 == 0:
            T = max(1e-12, T * decay)

    return best_h, best_min


# -------------------------------------------------------------
# Continuous refinement with soft‑max SLSQP
# -------------------------------------------------------------
def slsqp_refine(h0: np.ndarray,
                 rng: np.random.Generator,
                 max_iter: int = 800) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def eq_sum(x):
        return np.sum(x) - target

    # start with a moderate beta, then increase in a loop
    x = np.clip(h0, 0.0, 1.0)

    for beta in (5.0, 20.0, 80.0, 300.0, 1200.0, 5000.0):
        def softmax_obj(v):
            corr = np.correlate(v, 1.0 - v, mode="full")
            bc = beta * corr
            m = np.max(bc)
            # log‑sum‑exp (stable)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        cons = {"type": "eq", "fun": eq_sum}
        res = minimize(
            softmax_obj,
            x,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        # tiny repair if needed
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            x = enforce_sum(x, rng)

    return x
