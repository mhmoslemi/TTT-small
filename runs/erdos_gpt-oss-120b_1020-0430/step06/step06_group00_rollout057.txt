def enforce_sum(h, rng, tol=1e-12, max_iter=1000):
    target = h.size/2.0
    diff = target - h.sum()
    iter_count = 0
    while abs(diff) > tol and iter_count < max_iter:
        if diff > 0:
            free = h < 1.0
            nfree = free.sum()
            if nfree == 0:
                break
            inc = diff / nfree
            # Ensure we don't exceed 1
            max_inc = (1.0 - h[free]).min()
            inc = min(inc, max_inc)
            h[free] += inc
        else:
            free = h > 0.0
            nfree = free.sum()
            if nfree == 0:
                break
            dec = -diff / nfree
            max_dec = h[free].min()
            dec = min(dec, max_dec)
            h[free] -= dec
        diff = target - h.sum()
        iter_count += 1
    # if still off, use random adjustments
    while abs(diff) > tol:
        i = rng.integers(h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h
```

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad

def binary_sa(n, rng, time_budget, target_c=None):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    half = n // 2
    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, half, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        # maintain arrays
        ones_arr = ones
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr, assume_unique=True)
        # inner SA
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            # pick swap indices
            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]
            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                # swap indices in arrays
                ones_arr[i_idx] = j
                zeros_arr[j_idx] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if target_c is not None and best_c <= target_c:
                        return best_h, best_c
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator, tol=1e-12, max_iter=100) -> np.ndarray:
    target = h.size / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > tol and it < max_iter:
        if diff > 0:
            free = h < 1.0
            nfree = free.sum()
            if nfree == 0:
                break
            inc = diff / nfree
            max_inc = (1.0 - h[free]).min()
            inc = min(inc, max_inc)
            h[free] += inc
        else:
            free = h > 0.0
            nfree = free.sum()
            if nfree == 0:
                break
            dec = -diff / nfree
            max_dec = h[free].min()
            dec = min(dec, max_dec)
            h[free] -= dec
        diff = target - h.sum()
        it += 1
    while abs(diff) > tol:
        i = rng.integers(h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
    return g

def gradient_descent(h0: np.ndarray, rng: np.random.Generator,
                     time_budget: float, step0=0.2, decay=0.999) -> np.ndarray:
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() - start < time_budget:

        g = worst_gradient(h)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)

        h = h_new

        it += 1

def binary_sa(n: int, rng: np.random.Generator, time_budget: float,
              target_c=None) -> tuple[np.ndarray, float]:
    start = time.time()
    best_h = None
    best_c = np.inf
    half = n // 2
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, half, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr, assume_unique=True)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

                ones_arr[i_idx] = j

                if c_new < best_c:

                h[j] = 0.0
    if best_h is None:

    return best_h, best_c
