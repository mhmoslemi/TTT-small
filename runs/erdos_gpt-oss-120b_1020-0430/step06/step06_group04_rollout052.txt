def gradient_smoothmax_fast(h, beta):
    n = h.size
    v = compute_v_fft(h)  # returns v of length 2n-1
    # compute softmax weights
    mv = np.max(beta * v)
    exp_term = np.exp(beta * v - mv)
    w = exp_term / np.sum(exp_term)
    # compute convolutions
    L = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fft_h = np.fft.rfft(h, L)
    fft_w = np.fft.rfft(w, L)
    fft_w_rev = np.fft.rfft(w[::-1], L)
    conv1 = np.fft.irfft(fft_w_rev * fft_h, L)[:2 * n - 1]  # w_rev * h
    conv2 = np.fft.irfft(fft_w * fft_h, L)[:2 * n - 1]      # w * h
    g = 1.0 - conv1[n - 1 : n - 1 + n] - conv2[n - 1 : n - 1 + n]
    return g
```

def adjust_sum_exact(h, rng, max_iter=10000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final zip if any small diff
    return h

def gradient_smoothmax_fast(h, beta):
    n = h.size
    L = int(2 ** np.ceil(np.log2(2*n-1)))
    # compute v
    fft_h = np.fft.rfft(h, L)
    fft_one_minus_h = np.fft.rfft(1.0 - h, L)
    v = np.fft.irfft(fft_h * fft_one_minus_h, L)[:2*n-1]
    # softmax
    mv = np.max(beta * v)
    exp_term = np.exp(beta * v - mv)
    w = exp_term / np.sum(exp_term)
    # convolutions
    fft_w = np.fft.rfft(w, L)
    fft_w_rev = np.fft.rfft(w[::-1], L)
    conv1 = np.fft.irfft(fft_w_rev * fft_h, L)[:2*n-1]  # w_rev * h
    conv2 = np.fft.irfft(fft_w * fft_h, L)[:2*n-1]      # w * h
    # gradient
    g = 1.0 - conv1[n-1:n-1+n] - conv2[n-1:n-1+n]
    return g
```

def adam_optimize(h0, betas, steps_per_beta, alpha, rng, deadline):
    h = h0.copy()
    n = h.size
    for beta in betas:
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return h
            t += 1
            g = gradient_smoothmax_fast(h, beta)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
            h -= step
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng, max_iter=2000)
    return h

def binary_sa_optimize(h_bin, rng, deadline, max_iters=1_000_000):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5_fft(cur)
    best = cur.copy()
    best_c = cur_c
    # indices
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    pos_one[ones] = np.arange(len(ones))
    pos_zero[zeros] = np.arange(len(zeros))
    it = 0
    init_T = max(1e-8, cur_c * 0.02)
    while time.time() < deadline and it < max_iters:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        c = compute_c5_fft(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            # update index structures
            i_pos = pos_one[i]
            j_pos = pos_zero[j]
            ones[i_pos] = j
            zeros[j_pos] = i
            pos_one[i] = -1
            pos_one[j] = i_pos
            pos_zero[j] = -1
            pos_zero[i] = j_pos
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            # revert
            cur[i] = 1
            cur[j] = 0
        it += 1
    return best, best_c

import time
import numpy as np

# ---------- utilities ----------
def adjust_sum_exact(h, rng, max_iter=2000):
    """Adjust h in‑place so that sum(h) == len(h)/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5_fft(h):
    """Maximum overlap integral using FFT (discrete version)."""
    n = h.size
    L = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fft_h = np.fft.rfft(h, L)
    fft_one_minus_h = np.fft.rfft(1.0 - h, L)
    conv = np.fft.irfft(fft_h * fft_one_minus_h, L)[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))

def gradient_smoothmax_fast(h, beta):
    """Gradient of smooth‑max surrogate (1/β)·log Σ exp(β·v)."""
    n = h.size
    L = int(2 ** np.ceil(np.log2(2 * n - 1)))
    # v = convolution h * (1‑h)
    fft_h = np.fft.rfft(h, L)
    fft_one_minus_h = np.fft.rfft(1.0 - h, L)
    v = np.fft.irfft(fft_h * fft_one_minus_h, L)[: 2 * n - 1]
    # soft‑max weights
    mv = np.max(beta * v)
    exp_term = np.exp(beta * v - mv)
    w = exp_term / np.sum(exp_term)
    # convolutions w_rev * h and w * h
    fft_w = np.fft.rfft(w, L)
    fft_w_rev = np.fft.rfft(w[::-1], L)
    conv1 = np.fft.irfft(fft_w_rev * fft_h, L)[: 2 * n - 1]  # w_rev * h
    conv2 = np.fft.irfft(fft_w * fft_h, L)[: 2 * n - 1]      # w * h
    # gradient: 1 – Σ w·h_{j+k} – Σ w·h_{j‑k}
    g = 1.0 - conv1[n - 1 : n - 1 + n] - conv2[n - 1 : n - 1 + n]
    return g

def adam_optimize(h0, betas, steps_per_beta, alpha, rng, deadline):
    """Projected Adam on smooth‑max surrogate for a schedule of β."""
    h = h0.copy()
    for beta in betas:
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return h
            t += 1
            g = gradient_smoothmax_fast(h, beta)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
            h -= step
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng, max_iter=2000)
    return h

    """Simulated‑annealing on binary vectors (swap moves, sum fixed)."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5_fft(cur)
    best = cur.copy()
    best_c = cur_c

    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    pos_one = np.full(n, -1, dtype=int)

        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 2160
    ):
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)

        # brief smooth‑max refinement on the up‑sampled vector
        h_up = adam_optimize(
            h_up,
            betas=[5, 20, 80],
            steps_per_beta=100,
            alpha=0.15,
            rng=rng,
            deadline=deadline,
        )

        # binary SA (30 % of remaining time)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
        h_up, c_up = binary_sa_optimize(h_up, rng, sub_dead, max_iters=2_000_000)

        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()
            best_n = n_up

    # safety fallback
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5_fft(best_h)

    # final sanity checks
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng, max_iter=2000)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
