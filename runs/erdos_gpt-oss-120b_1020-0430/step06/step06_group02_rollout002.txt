def project_box_sum(x, target):
    low = np.min(x) - 1.0
    high = np.max(x)
    for _ in range(50):
        mid = (low+high)/2
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            high = mid
        else:
            low = mid
    return np.clip(x - (low+high)/2, 0.0, 1.0)

def gradient_fbeta(h, beta):
    n = h.size
    cross = np.convolve(h, h, mode='full')
    O = (n/2.0) - cross
    # O is length 2n-1
    # softmax
    betaO = beta * O
    max_betaO = np.max(betaO)
    exp_betaO = np.exp(betaO - max_betaO)
    w = exp_betaO / exp_betaO.sum()
    # w_rev
    w_rev = w[::-1]
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w_rev, h, mode='full')
    # indices offset n-1
    offset = n - 1
    grad = - (conv1[offset:offset+n] + conv2[offset:offset+n])
    return grad

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of x onto the intersection of the box [0,1]^n
    and the hyperplane sum = target.
    Uses binary search on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0  # guarantees sum(lo) > target > sum(hi)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def gradient_fbeta(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_beta(h) = (1/beta) * log sum_k exp(beta * O_k)
    where O_k = sum_i h[i] (1 - h[i+k]).
    """
    n = h.size
    cross = np.convolve(h, h, mode="full")          # length 2n-1
    O = (n * 0.5) - cross                           # O_k for k = -(n-1)..(n-1)
    betaO = beta * O
    max_betaO = np.max(betaO)
    w = np.exp(betaO - max_betaO)
    w /= w.sum()                                    # softmax weights
    w_rev = w[::-1]

    conv1 = np.convolve(w, h, mode="full")          # length 3n-2
    conv2 = np.convolve(w_rev, h, mode="full")
    off = n - 1
    grad = - (conv1[off:off + n] + conv2[off:off + n])
    return grad


def worst_gradient_vec(h: np.ndarray) -> np.ndarray:
    """
    Gradient of the overlap for the worst shift (max_k O_k).
    Used for a final fine‑grained local search.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)      # signed shift giving worst overlap
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target_sum = None

    # ----- Determine dimension and initial candidate --------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        base = np.array(initial_h_values, dtype=float)
        base = np.clip(base, 0.0, 1.0)
        target_sum = n * 0.5
        base = project_box_sum(base, target_sum)
    else:
        n = 512
        target_sum = n * 0.5
        base = None

    # ----- Generate a few diverse starting points ----------------------
    starts = []
    if base is not None:
        starts.append(base)
    for _ in range(4):
        rnd = rng.random(n)
        starts.append(project_box_sum(rnd, target_sum))

    best_h = None
    best_c = np.inf

    def time_left(fraction: float = 1.0) -> bool:
        return (time.time() - start) < budget_s * fraction

    # ----- Phase 1: low‑beta gradient descent on all starts ------------
    low_beta = 10.0

    iters_low = 5000
    for h0 in starts:

        c = compute_c5(h)
        if c < best_c:

            if best_c <= 0.3808:
                break
        if not time_left(0.70):
            break
    if best_h is None:
        # fallback to a simple balanced binary vector
        best_h = np.zeros(n)
        best_h[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = project_box_sum(best_h, target_sum)
        best_c = compute_c5(best_h)

    # ----- Phase 2: increasing beta refinement ------------------------
    beta_schedule = [50, 200, 1000, 5000, 20000, 80000]

            if best_c <= 0.3808:

    while time_left(0.999):

        if c_prop < best_c:
            best_c, best_h = c_prop, h_prop.copy()
            if best_c <= 0.3808:
                break

    # ----- Final sanity enforcement ------------------------------------
    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
