# -------------------- helper functions --------------------

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_gradient(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the worst‑shift overlap for shift k."""
    n = h.size

        g[start:] = 1.0 - h[: n - kp]
        # subtract h[i+kp] for i <= n-1-kp
        g[: n - kp] -= h[kp:]
    return g
