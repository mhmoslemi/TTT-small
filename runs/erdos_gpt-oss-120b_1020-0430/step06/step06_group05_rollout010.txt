def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # For shift k, correlation involves pairs (i,i+k) for i=0..n-k-1
        # Contribution = h[i]*(1-h[i+k]) + h[i+k]*(1-h[i])? Wait correlation using full mode 
        # produce sum over i h[i]*(1-h[i+k]) for all i such that i+k < n (others i+k out of range considered zero).

        # For i in [0,n-k-1]: derivative of term h[i]*(1-h[i+k]) w.r.t h[i] = (1 - h[i+k])
        # For i in [k,n-1]: derivative of term h[i-k]*(1 - h[i]) w.r.t h[i] = -h[i-k]

        # For i < k: only the first part? Actually i < k => i+k < n, but i-k is negative so no term. So grad[i] = (1 - h[i+k]).
        # For i >= n-k: i+k >= n, so term does not exist, but i-k >=0, so grad[i] = -h[i-k].

        # That matches.
        grad[: n - k] -= h[k:]

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    # We'll adjust by random indices.
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            delta = min(diff, cap)
        else:
            cap = -h[i]
            if cap >= 0:
                continue
            delta = max(diff, cap)  # diff negative, cap negative
        h[i] += delta
        diff = target - h.sum()
    return h
```

import time, numpy as np

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0: continue
            delta = min(diff, cap)
        else:
            cap = -h[i]
            if cap >= 0: continue
            delta = max(diff, cap)  # both negative
        h[i] += delta
        diff = target - h.sum()
    return h

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1 - 2 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad

def gradient_refine(h, target, rng, step=0.015, max_iter=12):
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h

CURRENT_BETA = None

def softmax_obj(x):

    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    CURRENT_TARGET = target
    CURRENT_BETA = beta

                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

    """Discrete C5 for a step function on [0,2]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
