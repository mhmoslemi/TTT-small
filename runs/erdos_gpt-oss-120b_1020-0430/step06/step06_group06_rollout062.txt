def softmax_gradient(h, beta):
    n = h.size
    b = 1 - h
    N = 2*n
    H = np.fft.fft(h, N)
    B = np.fft.fft(b, N)
    y = np.fft.ifft(H * np.conj(B)).real[:2*n-1]   # correlation

    # compute softmax weights
    y_max = np.max(y)
    exp_y = np.exp(beta * (y - y_max))
    w = exp_y / exp_y.sum()   # shape (2n-1,)

    # compute conv2 = w * h (as linear conv)
    L = 2*n-1 + n - 1  # = 3n-2
    N2 = 1 << ((L - 1).bit_length())
    W = np.fft.fft(w, N2)
    H2 = np.fft.fft(h, N2)
    conv2_full = np.fft.ifft(W * H2).real
    conv2 = conv2_full[:n]  # because for j from 0..n-1, conv2_full[j] gives sum_k w_k * h_{j - k}
    # compute conv1: w_rev * h
    w_rev = w[::-1]
    W_rev = np.fft.fft(w_rev, N2)
    conv1_full = np.fft.ifft(W_rev * H2).real
    # need appropriate offset: conv1_full offset = (len(w_rev)-1) = 2n-2. So conv1[j] = conv1_full[j + (2n-2)]
    offset = 2*n - 2
    conv1 = conv1_full[offset:offset+n]   # slice length n

    grad = -(conv1 + conv2)
    # For k=0 double counting? Let's see: In expression grad_j = -(sum_k w_k*(h_{j+k} + h_{j-k})). For k=0, term appears twice (h_{j}+h_{j}) = 2h_j. That's correct. So our conv1+conv2 includes contributions from k=0 twice? Let's verify:

    # conv1 includes w_0 * h_{j+0}= w_0 * h_j.
    # So sum = 2 w_0 h_j, matches gradient for k=0: -2 w_0 h_j. Good.
    return grad

    n = h.size

    y_max = np.max(y)
    exp_y = np.exp(beta * (y - y_max))
    w = exp_y / exp_y.sum()

import time
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.size + b.size - 1
    N = next_pow2(n)
    A = np.fft.fft(a, N)
    B = np.fft.fft(b, N)
    return np.fft.ifft(A * B).real[:n]

def correlation_hb(h: np.ndarray) -> np.ndarray:
    n = h.size
    b = 1.0 - h
    N = next_pow2(2 * n - 1)
    H = np.fft.fft(h, N)
    B = np.fft.fft(b[::-1], N)
    y = np.fft.ifft(H * B).real[: 2 * n - 1]
    return y

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    y = correlation_hb(h)
    return 1.0 - (2.0 / n) * y.max()

def softmax_gradient(h: np.ndarray, beta: float):
    n = h.size
    y = correlation_hb(h)
    y_max = y.max()
    exp_y = np.exp(beta * (y - y_max))
    w = exp_y / exp_y.sum()                     # length 2n-1
    # conv_minus = (w * h)[n-1:n-1+n]
    conv_minus_full = fft_convolve(w, h)
    conv_minus = conv_minus_full[n - 1 : n - 1 + n]
    # conv_plus = ((w reversed) * h)[n-1:n-1+n]
    w_rev = w[::-1]
    conv_plus_full = fft_convolve(w_rev, h)
    conv_plus = conv_plus_full[n - 1 : n - 1 + n]
    grad = 1.0 - (conv_plus + conv_minus)
    return grad, y_max
