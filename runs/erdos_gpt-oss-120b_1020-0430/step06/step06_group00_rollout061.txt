def compute_c5_fft(h):
    n = h.shape[0]
    n2 = 2 * n
    # Use rfft for real input
    f1 = np.fft.rfft(h, n=n2)
    f2 = np.fft.rfft((1 - h)[::-1], n=n2)
    c = np.fft.irfft(f1 * f2, n=n2)
    # c length n2, disregard last point at n2-1? Actually c has length n2.
    # The linear convolution of length n2-1, the last element is zero when one of the zero-padded inputs is at full length? Might be zero but keep.
    max_val = c[:n2-1].max()

def continuous_sa_fast(h, rng, time_budget):
   start = time.time()
   n = h.size
   target_sum = n/2.0
   c_cur = compute_c5_fft(h)
   best_h = h.copy()
   best_c = c_cur
   T0 = 0.02
   T_end = 1e-5
   while time.time() - start < time_budget:
        elapsed = time.time() - start
        fraction = elapsed / time_budget
        T = T0 * (T_end / T0) ** fraction
        # select i, j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * (T/T0)  # scaling
        if eps <= 0:
            continue
        # propose new vector
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_cur
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                # Early exit if very low
                if best_c <= 0.3808:
                    break

import time
import numpy as np

# --------------------------------------------------------------
# Core utilities
def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 via FFT (full correlation of h with 1‑h)."""
    n = h.shape[0]
    n2 = 2 * n
    f1 = np.fft.rfft(h, n=n2)
    f2 = np.fft.rfft((1 - h)[::-1], n=n2)
    c = np.fft.irfft(f1 * f2, n=n2)
    return c[: n2 - 1].max() * (2.0 / n)


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto [0,1]^n ∩ {sum = target} via binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        clipped = np.clip(v + mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def get_worst_shift(h: np.ndarray):
    """Return worst shift k (relative to centre) and its correlation value."""
    n = h.shape[0]
    n2 = 2 * n
    f1 = np.fft.rfft(h, n=n2)
    f2 = np.fft.rfft((1 - h)[::-1], n=n2)
    c = np.fft.irfft(f1 * f2, n=n2)
    idx = int(c[: n2 - 1].argmax())
    k = idx - (n - 1)
    return k, c[idx]


def worst_shift_gradient(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the overlap for the shift attaining the maximal value."""
    n = h.shape[0]
    k = abs(k)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros_like(h)
    g[: n - k] = 1.0 - h[k:]
    g[k:] -= h[: n - k]
    return g


# --------------------------------------------------------------
# Continuous simulated annealing (fast, sum‑preserving moves)
def continuous_sa_fast(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    c_cur = compute_c5_fft(h)
    best_h, best_c = h.copy(), c_cur
    T0, T_end = 0.02, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * (T / T0)
        if eps <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - c_cur
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, c_cur = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
                if best_c <= 0.3808:
                    break
    return best_h, best_c


# --------------------------------------------------------------
# Projected gradient refinement on the worst shift
def projected_gradient_refine(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    c_cur = compute_c5_fft(h)
    step0 = 0.02
    step = step0
    while time.time() - start < time_budget:
        k, _ = get_worst_shift(h)
        g = worst_shift_gradient(h, k)
        # simple backtracking line search
        improved = False
        for _ in range(5):
            h_new = project_sum_box(h - step * g, target)
            c_new = compute_c5_fft(h_new)
            if c_new < c_cur - 1e-12:
                h, c_cur = h_new, c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            step = step0
        if step < 1e-8:
            break
    return h, c_cur


# --------------------------------------------------------------
# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    N = 1024  # final resolution (still fast)

    # ---- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if h.size != N:
            h = upsample_h(h, N)
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, N / 2.0)
    else:
        h = rng.random(N)
        h = project_sum_box(h, N / 2.0)

    best_h, best_c = h.copy(), compute_c5_fft(h)

    # ---- Phase 1: fast continuous SA --------------------------------
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 4.0:
        t_sa = remaining * 0.65
        h_sa, c_sa = continuous_sa_fast(best_h, rng, t_sa)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # ---- Phase 2: projected gradient refinement ----------------------
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 0.5:
        h_pg, c_pg = projected_gradient_refine(best_h, rng, remaining)
        if c_pg < best_c:
            best_h, best_c = h_pg, c_pg

    # ---- Final sanitisation -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, best_h.size / 2.0)
    final_c = compute_c5_fft(best_h)
    best_c = float(final_c)

    return best_h.tolist(), best_c, int(best_h.size)
