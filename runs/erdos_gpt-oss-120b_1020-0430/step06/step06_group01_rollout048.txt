        if remaining <= 0:
            break
        # allocate some iterations, maybe max_iter = int(min(2000, remaining*200))
        max_iter = int(min(2000, remaining * 200))
        step0 = 0.5 / np.sqrt(beta)  # smaller steps for larger beta
        for it in range(max_iter):
            # compute gradient of obj
            corr_full = np.correlate(h, h, mode='full')
            c = corr_full[h.size-1:]  # length n
            w = np.exp(-beta * c)

import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_sum_box(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros_like(h)
    grad[: n - k] -= h[k:]   # -h_{i+k}
    grad[k:] -= h[: n - k]   # -h_{i-k}
    return grad

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c

def pgd_optimize(h0, rng, beta_seq, time_budget):
    start = time.time()
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    for beta in beta_seq:
        # allocate some iterations based on remaining time
        remaining = time_budget - (time.time() - start)
        if remaining <= 0:
            break
        max_iter = int(min(2000, max(10, remaining * 200)))  # at least few
        step0 = 0.5 / np.sqrt(beta)
        # compute initial objective
        corr_full = np.correlate(h, h, mode='full')
        c = corr_full[n-1:]
        w = np.exp(-beta * c)
        obj = np.log(w.sum()) / beta
        for it in range(max_iter):
            # gradient
            corr_full = np.correlate(h, h, mode='full')
            c = corr_full[n-1:]
            w = np.exp(-beta * c)
            W = w.sum()
            left = np.convolve(w[::-1], h, mode='full')[n-1: n-1+n]
            right = np.convolve(w, h, mode='full')[:n]
            grad = - (left + right) / W
            # line search
            step = step0
            improved = False
            for _ in range(12):
                h_new = project_sum_box(h - step * grad, target)
                corr_full_new = np.correlate(h_new, h_new, mode='full')
                c_new = corr_full_new[n-1:]
                w_new = np.exp(-beta * c_new)
                obj_new = np.log(w_new.sum()) / beta
                if obj_new < obj - 1e-14:
                    h = h_new
                    obj = obj_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                # no further improvement for this beta
                break
            # optional stop if gradient small
            if np.linalg.norm(grad) * step < 1e-7:
                break
        # after beta, may recompute c5 for logging but not needed
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto [0,1] with sum == target using bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = Σ h_i(1‑h_{i+k})."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros_like(h)
    grad[: n - k] -= h[k:]      # -h_{i+k}
    grad[k:] -= h[: n - k]      # -h_{i‑k}
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using the worst‑shift gradient."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c


def pgd_optimize(h0: np.ndarray, rng: np.random.Generator,
                 beta_seq, time_budget: float) -> np.ndarray:
    """Projected gradient descent on smooth surrogate of min autocorrelation."""
    start = time.time()
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    for beta in beta_seq:
        # allocate iterations based on remaining time
        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        max_iter = int(min(2000, max(10, remaining * 200)))
        step0 = 0.5 / np.sqrt(beta)
        # initial objective value
        c_full = np.correlate(h, h, mode="full")
        c = c_full[n - 1:]
        w = np.exp(-beta * c)
        obj = np.log(w.sum()) / beta
        for _ in range(max_iter):
            # gradient of the surrogate
            c_full = np.correlate(h, h, mode="full")
            c = c_full[n - 1:]
            w = np.exp(-beta * c)
            W = w.sum()
            left = np.convolve(w[::-1], h, mode="full")[n - 1:n - 1 + n]
            right = np.convolve(w, h, mode="full")[:n]
            grad = - (left + right) / W
            # backtracking line search
            step = step0
            improved = False
            for _ in range(12):
                h_new = project_sum_box(h - step * grad, target)
                c_full_new = np.correlate(h_new, h_new, mode="full")
                c_new = c_full_new[n - 1:]
                w_new = np.exp(-beta * c_new)
                obj_new = np.log(w_new.sum()) / beta
                if obj_new < obj - 1e-14:
                    h, obj = h_new, obj_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                break
            if np.linalg.norm(grad) * step < 1e-7:
                break
    return h
