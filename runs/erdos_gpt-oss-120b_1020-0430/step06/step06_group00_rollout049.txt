def compute_c5(h):
    n = h.shape[0]
    if n <= 128:
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n)
    else:
        nfft = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(h, n=nfft)
        fb = np.fft.rfft(1.0 - h, n=nfft)
        conv = np.fft.irfft(fa * fb, n=nfft)[:2 * n - 1]
        return np.max(conv) * (2.0 / n)

def enforce_sum(h, rng):

    diff = target - h.sum()
    n = h.size
    # handle diff >0: need to add tot diff, randomly fill positions not at 1.
    # diff <0: need to subtract
    # We'll perform iteration similar to previous, but clamp.
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:

        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:

import time
import numpy as np

# ------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N) using FFT for speed."""
    n = h.size
    if n <= 128:
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * (2.0 / n))
    # FFT convolution
    nfft = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, n=nfft)
    fb = np.fft.rfft(1.0 - h, n=nfft)
    conv = np.fft.irfft(fa * fb, n=nfft)[: 2 * n - 1]
    return float(np.max(conv) * (2.0 / n))


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto the affine set sum(h)=n/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff = target - h.sum()
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff = target - h.sum()
        attempts += 1
    if abs(diff) > 1e-10:
        # deterministic fill if still off
        if diff > 0:
            for i in range(n):
                add = min(diff, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    diff -= add
                if diff <= 1e-12:
                    break
        else:
            for i in range(n):
                sub = min(-diff, h[i])
                if sub > 0:
                    h[i] -= sub
                    diff += sub
                if diff >= -1e-12:
                    break
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

# Binary simulated annealing (0/1 vectors)
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Binary SA with exactly n/2 ones, using FFT for fast C5 evaluation."""
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    zeros_idx = np.setdiff1d(np.arange(n), ones_idx, assume_unique=True)
    ones = list(ones_idx)
    zeros = list(zeros_idx)

    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    T0, T_end = 0.1, 1e-6
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            cur_c = new_c
            ones[i_pos] = j
            zeros[j_pos] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

# Continuous simulated annealing
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    N = h.size
    T0, T_end = 0.05, 1e-7
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.5 * (T / T0)

        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1

        direction = rng.random() < 0.5
        if direction:
            max_eps = min(h[i], 1.0 - h[j])
        else:
            max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        if eps <= 0.0:
            continue

        h_new = h.copy()
        if direction:
            h_new[i] -= eps
            h_new[j] += eps
        else:
            h_new[i] += eps
            h_new[j] -= eps

        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c


# ------------------------------------------------------------
# Gradient based refinement (sub‑gradient of worst shift)
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 80) -> np.ndarray:
    """Iterative sub‑gradient descent with diminishing step size."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h -= step * (0.5 ** it) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

        c_ref = compute_c5(h_ref)
        if c_ref < best_c:

    # --------------------------------------------------------

    # --------------------------------------------------------
