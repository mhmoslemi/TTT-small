def compute_c5_fft(h):
    n = h.shape[0]
    L = 2*n
    H = np.fft.rfft(h, L)
    B = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(H * B, L)
    return np.max(c) * (2.0 / n)
```

def simulated_annealing_bin(h, start_temp, end_temp, rng, time_limit):
    n = len(h)
    cur_h = h.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        # linear cooling
        frac = (time.time() - start) / (time_limit - start)
        temp = start_temp * (1 - frac) + end_temp * frac
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5_fft(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c

import time, math, numpy as np

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5_brute(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, L)
    B = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(H * B, L)
    return np.max(c) * (2.0 / n)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def binarize_and_adjust(v, target):
    h = (v > 0.5).astype(float)
    cur = int(h.sum())
    diff = int(target - cur)
    if diff > 0:
        zeros = np.where(h == 0)[0]
        order = np.argsort(v[zeros])
        sel = zeros[order[-diff:]]
        h[sel] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        order = np.argsort(v[ones])
        sel = ones[order[: -diff]]
        h[sel] = 0.0
    return h

def simulated_annealing_bin(h, start_temp, end_temp, rng, time_limit):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    while time.time() < time_limit:
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)
        temp = start_temp * (1 - frac) + end_temp * frac
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5_fft(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            cur_c = new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c
