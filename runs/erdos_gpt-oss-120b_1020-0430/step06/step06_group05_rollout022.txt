def gradient_descent(h, target, rng, time_limit, start):
    n = h.size
    step0 = 0.2
    it = 0
    best = compute_c5_fft(h)
    while time.time() - start < time_limit:
        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        shift = idx - (n-1)
        k = abs(shift)
        # compute grad
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.zeros_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        # step size decay
        step = step0 * (0.99 ** it)
        it += 1
        h_new = h - step * grad
        # clip
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # enforce sum
        h_new = enforce_sum(h_new, target, rng)
        c_new = compute_c5_fft(h_new)
        if c_new < best - 1e-12:
            best = c_new
            h = h_new
        # else maybe accept with some prob? Not needed.
        # Break if step becomes very small
        if step < 1e-6:
            break
    return h, best
```

def coordinate_search(h, target, rng, time_limit, start):
    n = h.size
    best = compute_c5_fft(h)
    eps_scale = 0.2
    it = 0
    while time.time() - start < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # compute feasible epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * eps_scale

            # maybe increase eps_scale slowly?
            pass
        it += 1

def compute_corr_fft(h):
    n = h.size
    # pad length = 2*n - 1
    sz = 2 * n - 1
    # next power of two for efficient FFT?
    m = 1 << (sz - 1).bit_length()
    # compute FFT of h and reversed (1-h)
    H = np.fft.rfft(h, n=m)
    rev = (1 - h)[::-1]
    R = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(H * R, n=m)[:sz]
    return conv
```

Using rfft/irfft for real arrays gives speed gains. m is power of 2 ≥ sz.

def enforce_sum(h, target, rng):
    # Simple iterative approach: distribute diff to random indices
    diff = target - h.sum()
    n = h.size
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        # pick random index
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
        else:
            cap = -h[i]
        if cap == 0.0:
            it += 1
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
        it += 1
    return h

def slsqp_refine(h, target, rng, beta, max_iter):
    n = h.size
    # Global variables for objective
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        lambda x: softmax_obj(x),
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x
```

Define:

```
def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
```

import time
import numpy as np

# ---------- Helper functions ----------

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            it += 1
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
        it += 1
    return h

def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1-h) using FFT (real)."""
    n = h.size
    sz = 2 * n - 1
    # pad to next power of two
    m = 1 << (sz - 1).bit_length()
    H = np.fft.rfft(h, n=m)
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(H * R, n=m)[:sz]
    return conv

def compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 for step function h on [0,2] using FFT."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for shift achieving max overlap."""
    n = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    k = abs(shift)

    else:

    return k, grad

def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Transfer a fraction alpha of out‑mass to in‑mass according to sign of grad."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)

    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new

CURRENT_TARGET = None
CURRENT_BETA = None

def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h, target, rng, beta, max_iter):
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:

    return x
