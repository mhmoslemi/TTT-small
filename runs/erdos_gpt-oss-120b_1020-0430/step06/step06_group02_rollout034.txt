def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # while loop adjusting random index:
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    # If still not perfect, adjust first element with residual
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def thue_morse_bits(n):
    seq = np.zeros(1, dtype=float)
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]

# ------------------- core utilities ------------------------------------ #
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    max_iter = 10000
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue‑Morse sequence (0/1)."""
    seq = np.zeros(1, dtype=float)
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]

# ------------------- soft‑max SLSQP ----------------------------------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP on the soft‑max surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        if add != 0.0:
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
        it += 1
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial candidates -----------------------------------------
    candidates = []

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        candidates.append((h0, compute_c5(h0)))

    tm = thue_morse_bits(512)
    candidates.append((tm, compute_c5(tm)))

    n0 = 512
    rand_bin = np.zeros(n0, dtype=float)
    rand_bin[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    candidates.append((rand_bin, compute_c5(rand_bin)))

    best_h, best_c = min(candidates, key=lambda t: t[1])
    n = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)

    # ---- greedy hill climbing ----------------------------------------
    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.98:
            break

        grad = worst_gradient(best_h)
        idx_pos = int(np.argmax(grad))
        idx_neg = int(np.argmin(grad))

        remaining = budget_s - elapsed
        max_eps_allowed = max(1e-4, 0.1 * (remaining / budget_s))

        max_eps = min(best_h[idx_pos], 1.0 - best_h[idx_neg], max_eps_allowed)
        moved = False
        if max_eps > 0 and grad[idx_pos] > 0 and grad[idx_neg] < 0:
            eps = max_eps * rng.random()
            cand = best_h.copy()
            cand[idx_pos] -= eps
            cand[idx_neg] += eps
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
                moved = True

        # occasional random tweaks
        if (iteration % 500 == 0) or not moved:
            for _ in range(5):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                max_eps2 = min(best_h[i], 1.0 - best_h[j], max_eps_allowed)
                if max_eps2 <= 0:
                    continue
                eps2 = max_eps2 * rng.random() * 0.05
                cand2 = best_h.copy()
                cand2[i] -= eps2
                cand2[j] += eps2
                c2 = compute_c5(cand2)
                if c2 < best_c:
                    best_c, best_h = c2, cand2
                    break

        iteration += 1
        if best_c <= 0.3808:
            break

    # ---- optional SLSQP refinement ------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 8.0 and best_c > 0.3808:
        for beta in (50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0):
            if time.time() - start >= budget_s * 0.99:
                break
            opt_h = slsqp_optimize(best_h, beta, rng, max_iter=800)
            c = compute_c5(opt_h)
            if c < best_c:
                best_c, best_h = c, opt_h
                if best_c <= 0.3808:
                    break

    # ---- final sanity enforcement ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
