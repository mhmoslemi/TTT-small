import time
import numpy as np

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple water-filling style
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def gradient_step_line_search(h, grad, target, rng, fractions=(0.2, 0.5, 0.8, 1.0)):
    """Take a step along -grad with a line search respecting box constraints."""
    pos = grad > 0
    neg = grad < 0
    # max α keeping h - α*grad in [0,1]
    max_alpha_pos = np.where(pos, h / grad, np.inf)
    max_alpha_neg = np.where(neg, (1.0 - h) / (-grad), np.inf)
    alpha_max = np.min(np.where(pos | neg, np.minimum(max_alpha_pos, max_alpha_neg), np.inf))
    if not np.isfinite(alpha_max) or alpha_max <= 0:
        return h
    best_h = h
    best_c = compute_c5(h)
    for f in fractions:
        alpha = alpha_max * f
        h_new = np.clip(h - alpha * grad, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    return best_h


def mass_transfer_move(h, grad, alpha, target, rng):
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h, rng, scale=0.2):
    """Small random transfer that preserves total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def slsqp_refine(h, target, rng, beta, max_iter=600):
    """SLSQP on the smooth max (softmax) objective."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


def crossover(h1, h2, rng):
    """One‑point crossover preserving the sum constraint."""
    n = h1.size
    cut = rng.integers(1, n)
    child = np.concatenate([h1[:cut], h2[cut:]])
    child = np.clip(child, 0.0, 1.0)
    child = enforce_sum(child, n / 2.0, rng)
    return child


def local_optimize(h, target, rng, max_iters=30):
    """Iterative improvement using gradient steps and fallback moves."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    for _ in range(max_iters):
        _, grad = worst_shift_and_grad(cur_h)
        new_h = gradient_step_line_search(cur_h, grad, target, rng)
        new_c = compute_c5(new_h)
        if new_c < cur_c - 1e-12:
            cur_h, cur_c = new_h, new_c
            continue
        improved = False
        for alpha in (0.5, 0.25, 0.125, 0.0625, 0.03125):
            cand = mass_transfer_move(cur_h, grad, alpha, target, rng)
            cand_c = compute_c5(cand)
            if cand_c < cur_c - 1e-12:
                cur_h, cur_c = cand, cand_c
                improved = True
                break
        if not improved:
            cand = random_perturb(cur_h, rng, scale=0.2)
            cand_c = compute_c5(cand)
            if cand_c < cur_c - 1e-12:
                cur_h, cur_c = cand, cand_c
        # stop early if no progress
        if not improved and new_c >= cur_c - 1e-12:
            break
    return cur_h, cur_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
    else:
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    top_solutions = [(best_h, best_c)]

    # ---- main evolutionary loop ----
    K = 10
    main_budget = budget_s * 0.85
    while time.time() - start < main_budget:
        # candidate generation
        if rng.random() < 0.35 or len(top_solutions) < 2:
            cand = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            cand[idx] = 1.0
        else:
            i, j = rng.integers(0, len(top_solutions), size=2)
            p1, _ = top_solutions[i]
            p2, _ = top_solutions[j]
            cand = crossover(p1, p2, rng)
        cand = enforce_sum(cand, target_sum, rng)

        # local refinement
        cand_h, cand_c = local_optimize(cand, target_sum, rng, max_iters=30)

        # update best and archive
        if cand_c < best_c - 1e-12:
            best_h, best_c = cand_h.copy(), cand_c
        top_solutions.append((cand_h, cand_c))
        top_solutions.sort(key=lambda x: x[1])
        if len(top_solutions) > K:
            top_solutions = top_solutions[:K]

    # ---- final SLSQP polishing ----
    remaining = budget_s - (time.time() - start)
    if remaining > budget_s * 0.05:
        for beta in (5000.0, 10000.0, 20000.0, 50000.0, 100000.0):
            if time.time() - start > budget_s * 0.99:
                break
            refined = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=500)
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_h, best_c = refined, c_ref

    # ---- final enforcement ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
