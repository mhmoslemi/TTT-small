def compute_c5_fft(h):
    N = h.size
    L = 1 << (2*N-1).bit_length()
    H = np.fft.rfft(h, n=L)
    corr = np.fft.irfft(H * np.conj(H), n=L)[:2*N-1]  # linear autocorrelation
    min_corr = corr.min()
    c5 = 1.0 - (2.0/N) * min_corr
    return c5, min_corr

def binary_sa_fft(N, rng, time_budget):
    start = time.time()
    # initial random binary vector
    h = np.zeros(N, dtype=float)
    ones = rng.choice(N, N//2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    c5, min_corr = compute_c5_fft(h)
    best_c = c5
    best_min_corr = min_corr
    # precompute indices for fast swap? maybe just use arrays each iteration.
    T0, T_end = 0.02, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end/T0) ** (elapsed / time_budget)
        # pick random zero and one
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        if zeros_idx.size == 0 or ones_idx.size == 0:
            break
        i = rng.choice(zeros_idx)
        j = rng.choice(ones_idx)
        # propose swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_c5, new_min_corr = compute_c5_fft(h_new)
        delta = new_c5 - c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            h = h_new
            c5 = new_c5
            min_corr = new_min_corr
            if new_c5 < best_c:
                best_c = new_c5
                best_h = h_new.copy()
                best_min_corr = new_min_corr
                # early exit if below target
                if best_c <= 0.3808:
                    break
    return best_h, best_c

import time
import numpy as np

# -------------------------------------------------------------
def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 evaluation using linear autocorrelation via FFT."""
    N = h.size
    L = 1 << (2 * N - 1).bit_length()          # next power of two >= 2N-1
    H = np.fft.rfft(h, n=L)
    corr = np.fft.irfft(H * np.conj(H), n=L)[: 2 * N - 1]
    corr = np.maximum(corr, 0.0)               # guard against tiny negatives
    min_corr = corr.min()
    return 1.0 - (2.0 / N) * min_corr


# -------------------------------------------------------------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 20000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


# -------------------------------------------------------------
def binary_sa_fft(N: int, rng: np.random.Generator, time_budget: float):
    """Binary simulated annealing with exact N/2 ones, using FFT for evaluation."""
    start = time.time()
    h = np.zeros(N, dtype=float)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1.0

    best_h = h.copy()
    best_c = compute_c5_fft(h)

    T0, T_end = 0.02, 1e-5

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        # pick a random 0 and a random 1 and propose a swap
        i = rng.integers(N)
        while h[i] != 0.0:
            i = rng.integers(N)
        j = rng.integers(N)
        while h[j] != 1.0:
            j = rng.integers(N)

        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0

        c_new = compute_c5_fft(h_new)
        delta = c_new - best_c if c_new < best_c else c_new - best_c  # dummy for acceptance

        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                if best_c <= 0.3808:
                    break
    return best_h, best_c

def continuous_sa_fft(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated annealing on the continuous feasible set (sum fixed, bounds [0,1])."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.002, 1e-7

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.2 * (T / T0)

        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1

        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps

        c_new = compute_c5_fft(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    N = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (N - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: N - k] = 1.0 - h[k:]
            g[k:] -= h[: N - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum
        target = N / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(N)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h to a new number of sample points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
