def anneal_hill_climb(h0, rng, max_iters=5000, init_temp=0.01, final_temp=1e-5, step_scale=0.5):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    for t in range(max_iters):
        T = init_temp + (final_temp - init_temp) * (t / max_iters)
        # choose i and j
        i = rng.integers(n)
        attempts = 0
        while h[i] <= 1e-12 and attempts < 10:
            i = rng.integers(n)
            attempts += 1
        j = rng.integers(n)
        attempts = 0
        while h[j] >= 1 - 1e-12 and attempts < 10:
            j = rng.integers(n)
            attempts += 1
        if h[i] <= 1e-12 or h[j] >= 1 - 1e-12:
            continue
        max_eps = min(h[i], 1 - h[j])
        eps = max_eps * rng.random() * step_scale
        if eps <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / (T+1e-12)):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
    return best_h, best_c

def upsample(h, new_n):
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # ensure within [0,1]
    h_new = np.clip(h_new, 0.0, 1.0)

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Projection onto the box [0,1] and a given sum
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# ----------------------------------------------------------------------
# Random feasible vector (0≤h≤1, sum = n/2)
def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

# ----------------------------------------------------------------------
# Compute max overlap O_k and the corresponding C5 bound (using FFT for speed)
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    size = 2 * n
    # linear correlation via FFT
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(n)]))
    corr = np.fft.irfft(H * G, n=size)
    return np.max(corr) * (2.0 / n)

# ----------------------------------------------------------------------
# Worst shift gradient (used by sub‑gradient descent)
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the max‑overlap objective
def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / math.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

# Simulated‑annealing hill climb (fast, works on any n)
def anneal_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000,
                      init_temp: float = 0.01,
                      final_temp: float = 1e-5,
                      step_scale: float = 0.6):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    for t in range(max_iters):
        T = init_temp + (final_temp - init_temp) * (t / max_iters)
        # pick i (decrease) and j (increase)
        i = rng.integers(n)
        attempt = 0
        while h[i] <= 1e-12 and attempt < 10:
            i = rng.integers(n)
            attempt += 1
        j = rng.integers(n)
        attempt = 0
        while h[j] >= 1.0 - 1e-12 and attempt < 10:
            j = rng.integers(n)
            attempt += 1
        if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        eps = max_eps * rng.random() * step_scale
        if eps <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        if new_c < cur_c or rng.random() < math.exp((cur_c - new_c) / (T + 1e-12)):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
    return best_h, best_c

# ----------------------------------------------------------------------
# Upsample by linear interpolation
def upsample_interp(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_sum(h_new, new_n / 2.0)

# ----------------------------------------------------------------------
# Upsample by repeating each entry (works when new_n is a multiple of old_n)
def upsample_repeat(h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = h.size
    factor = new_n // old_n
    h_up = np.repeat(h, factor)
    # handle possible off‑by‑one
    if h_up.size > new_n:
        h_up = h_up[:new_n]
    elif h_up.size < new_n:
        extra = new_n - h_up.size
        h_up = np.concatenate([h_up, h_up[:extra]])
    return project_box_sum(h_up, new_n / 2.0)

# Soft‑max SLSQP optimisation (global variables required)
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
