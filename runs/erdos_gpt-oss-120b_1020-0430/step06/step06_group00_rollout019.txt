    iter_count = 0

        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - (len(h)-1)
        k = abs(k)
        n = len(h)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1 - 2*h
        else:
            grad[:n-k] = 1 - h[k:]
            grad[k:] -= h[:n-k]
        # compute max feasible step size without bounds

def enforce_sum(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    attempts = 0
    n = h.size
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff = target - h.sum()
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff = target - h.sum()
        attempts += 1
    # If still diff > tolerance, adjust first index
    if abs(diff) > 1e-12:
        # just distribute diff to first index
        i = 0
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

import time
import numpy as np

# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function on [0,2] with equidistant samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff = target - h.sum()
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff = target - h.sum()
        attempts += 1
    # fallback (should rarely trigger)
    if abs(diff) > 1e-12:
        i = 0
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# -------------------------------------------------------------
def gradient_descent(h_start: np.ndarray, rng: np.random.Generator,
                     time_limit: float):
    """Projected gradient descent on the worst shift."""
    start = time.time()
    h = h_start.copy()
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    while time.time() - start < time_limit:
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = abs(idx - (N - 1))

        # gradient of that shift (derivative of overlap)
        grad = np.zeros_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            n_k = N - k
            grad[:n_k] = 1.0 - h[k:]
            grad[k:] -= h[:n_k]

        # feasible step size so that 0<=h-step*grad<=1
        pos = grad > 0
        neg = grad < 0
        max_step = np.inf
        if np.any(pos):
            max_step = min(max_step, np.min(h[pos] / grad[pos]))
        if np.any(neg):
            max_step = min(max_step, np.min((1.0 - h[neg]) / (-grad[neg])))
        if max_step <= 0.0 or not np.isfinite(max_step):
            # do a tiny random transfer and continue
            idx_pos = np.where(h > 0.0)[0]
            idx_neg = np.where(h < 1.0)[0]
            if idx_pos.size == 0 or idx_neg.size == 0:
                break
            i = rng.choice(idx_pos)
            j = rng.choice(idx_neg)
            eps = min(h[i], 1.0 - h[j], 0.01)
            if eps <= 0.0:
                break
            h[i] -= eps
            h[j] += eps
            continue

        step = max_step * 0.9  # stay away from the boundary
        h_new = h - step * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = enforce_sum(h_new, rng)

        cur_c = compute_c5(h_new)
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h_new.copy()
            if best_c <= 0.3808:
                return best_h, best_c
        h = h_new

    return best_h, best_c

def hill_climb(h_start: np.ndarray, rng: np.random.Generator, time_limit: float):
    """Simple random transfer hill‑climbing (sum preserved)."""
    start = time.time()
    h = h_start.copy()
    N = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    while time.time() - start < time_limit:
        i = rng.integers(0, N)
        if h[i] <= 0.0:
            continue
        j = rng.integers(0, N)
        if h[j] >= 1.0:
            continue
        eps = min(h[i], 1.0 - h[j], 0.02)
        if eps <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        cur_c = compute_c5(h_new)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_new.copy()
            h = h_new
        # else keep old h
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ---------------------------------------------------------
    # Initial construction (1024 samples)
    N0 = 1024
    if 'initial_h_values' in globals():
