def compute_c5(h):
    n = h.size
    # pad to length 2*n
    N = 2*n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def enforce_sum(v, target, rng):
    # v is float numpy array, we need sum(v)=target, with bounds [0,1]
    # Clip first
    np.clip(v, 0.0, 1.0, out=v)
    diff = target - v.sum()
    n = v.size
    # With random adjustments until close
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
            v[i] += add
        else:
            sub = min(-diff, v[i])
            v[i] -= sub
        diff = target - v.sum()
        attempts += 1
    return v

def worst_shift_gradient(h):
    n = h.size
    # compute correlation
    # maybe use same fast correlation
    N = 2*n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1-h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[:2*n-1]
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

import time
import numpy as np

# ---------- core utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using FFT for speed."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Project v onto {0≤v≤1, sum(v)=target} via random redistribution."""
    np.clip(v, 0.0, 1.0, out=v)
    diff = target - v.sum()
    n = v.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
            v[i] += add
        else:
            sub = min(-diff, v[i])
            v[i] -= sub
        diff = target - v.sum()
        attempts += 1
    return v


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(h) to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def combined_gradient(h: np.ndarray, M: int = 3):
    """Gradient obtained by averaging the top M shift gradients."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[: 2 * n - 1]
    max_corr = np.max(corr)
    m = min(M, corr.size)
    if m == corr.size:
        idxs = np.arange(corr.size)
    else:
        idxs = np.argpartition(corr, -m)[-m:]
    grad = np.zeros_like(h)
    for j in idxs:
        k = abs(j - (n - 1))
        if k == 0:
            grad[:] += 1.0 - 2.0 * h
        else:
            grad[: n - k] += 1.0 - h[k:]
            grad[k:] -= h[: n - k]
    grad /= m
    return grad, max_corr


def gradient_descent_run(h0: np.ndarray, target: float,
                         rng: np.random.Generator, time_budget: float):
    """Projected gradient descent with softmax‑style gradient."""
    start = time.time()
    n = h0.size
    h = h0.copy()
    grad, max_corr = combined_gradient(h)
    c_curr = max_corr * (2.0 / n)
    best_h = h.copy()
    best_c = c_curr
    it = 0
    while time.time() - start < time_budget:
        step = 0.05 * (0.995 ** it)  # start larger than previous version
        h_new = h - step * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)
        enforce_sum(h_new, target, rng)
        grad_new, max_corr_new = combined_gradient(h_new)
        c_new = max_corr_new * (2.0 / n)
        # SA‑style acceptance
        if c_new <= c_curr or rng.random() < np.exp(-(c_new - c_curr) / max(step, 1e-12)):
            h, grad, c_curr = h_new, grad_new, c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        it += 1
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target resolution
    n_target = 1024
    target_sum = n_target / 2.0

    # ---- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        enforce_sum(h0, h0.size / 2.0, rng)
        if h0.size != n_target:
            h = upsample_h(h0, n_target)
            enforce_sum(h, target_sum, rng)
        else:
            h = h0.copy()
    else:
        h = rng.random(n_target)
        enforce_sum(h, target_sum, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- iterative gradient descent cycles -------------------------------
    while True:
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        if remaining < 12.0:  # leave a few seconds for final polishing
            break
        # allocate a chunk (max 30 s, at most 40 % of what is left)
        chunk = min(30.0, remaining * 0.4)
        # perturb current best
        h_start = best_h.copy()
        eps = 0.05
        h_start += rng.uniform(-eps, eps, size=n_target)
        np.clip(h_start, 0.0, 1.0, out=h_start)
        enforce_sum(h_start, target_sum, rng)
        local_h, local_c = gradient_descent_run(h_start, target_sum, rng, chunk)
        if local_c < best_c:
            best_c, best_h = local_c, local_h.copy()

    # ---- final short simulated‑annealing refinement ----------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 4.0:
        h_cur = best_h.copy()
        c_cur = best_c
        n = n_target
        T0, T_end = 0.001, 1e-7
        while time.time() - start < total_budget - 2.0:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / total_budget)
            scale = 0.2 * (T / T0)
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if rng.random() < 0.5:
                max_eps = min(h_cur[i], 1.0 - h_cur[j])
                if max_eps <= 0.0:
                    continue
                eps = rng.random() * max_eps * scale
                if eps <= 0.0:
                    continue
                h_new = h_cur.copy()
                h_new[i] -= eps
                h_new[j] += eps
            else:
                max_eps = min(h_cur[j], 1.0 - h_cur[i])
                if max_eps <= 0.0:
                    continue
                eps = rng.random() * max_eps * scale
                if eps <= 0.0:
                    continue
                h_new = h_cur.copy()
                h_new[i] += eps
                h_new[j] -= eps
            c_new = compute_c5(h_new)
            delta = c_new - c_cur
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                h_cur, c_cur = h_new, c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h_new.copy()

    # ---- final sanitisation -----------------------------------------------
    enforce_sum(best_h, target_sum, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_target)
