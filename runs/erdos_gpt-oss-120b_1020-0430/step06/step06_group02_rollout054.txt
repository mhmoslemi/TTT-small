def gradient_descent(h, rng, max_iter, time_limit, start_time):
    best_h, best_c = h, compute_c5(h)
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        grad = worst_gradient(best_h)
        # direction -grad
        new_h, new_c = line_search(best_h, grad, rng)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
            # optionally reset iteration counter or continue.
        else:
            # If no improvement, maybe random perturbation.
            # do small random transfer
            i, j = rng.integers(0, best_h.size, size=2)
            if i != j:
                max_eps = min(best_h[i], 1.0 - best_h[j])
                if max_eps > 0:
                    eps = rng.random() * max_eps * 0.02
                    nh = best_h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    nh = np.clip(nh,0,1)
                    nh = enforce_sum(nh, rng)
                    c = compute_c5(nh)
                    if c < best_c:
                        best_h, best_c = nh, c
        # Optionally early break if best_c < target.
        if best_c <= 0.3808:
            break
    return best_h, best_c

def line_search(h, grad, rng):
    # compute d = -grad
    d = -grad
    # compute max step
    # avoid division by zero
    pos = d > 1e-12
    neg = d < -1e-12
    steps = []
    if np.any(pos):
        steps.append(((1.0 - h[pos]) / d[pos]).min())
    if np.any(neg):
        steps.append((h[neg] / -d[neg]).min())
    if not steps:
        return h, compute_c5(h)
    step_max = min(steps)
    # start with step_max
    step = step_max
    cur_c = compute_c5(h)
    min_step = step_max * 1e-6
    while step >= min_step:
        h_new = h + step * d
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new)
        c_new = compute_c5(h_new)
        if c_new < cur_c - 1e-14:
            return h_new, c_new
        step *= 0.5
    # if no improvement, return original
    return h, cur_c

def gradient_descent(h_initial, rng, max_iter, time_limit, start_time):

    best_c = compute_c5(h)
    best_h = h.copy()
    it = 0

        grad = worst_gradient(h)
        h_new, c_new = line_search(h, grad, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # random small transfer
            i, j = rng.integers(0, h.size, size=2)

                        h = h2

def slsqp_optimize(h0, beta, rng, max_iter):
    # as described above with gradient
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        max_bc = np.max(bc)
        w = np.exp(bc - max_bc)
        w = w / w.sum()
        # compute convs
        conv_w = np.convolve(w, x, mode='full')
        conv_rev_w = np.convolve(w[::-1], x, mode='full')
        offset_w = n - 1
        offset_rev = 2 * n - 2
        grad_vec = 1.0 - conv_rev_w[offset_rev:offset_rev + n] - conv_w[offset_w:offset_w + n]
        return grad_vec

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = opt.minimize(obj, h0, method='SLSQP', jac=grad, bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    sum_diff = target - x.sum()
    if abs(sum_diff) > 1e-12:
        # simple repair
        x = enforce_sum(x)
    return x

    for i in range(n):

import time
import numpy as np
import scipy.optimize as opt

# ------------------ basic utilities -----------------------------------


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """
    Gradient of the worst overlap O_k w.r.t. h.
    O_k = sum_i h[i] * (1 - h[i+k]).
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def enforce_sum(h: np.ndarray) -> np.ndarray:
    """Scale / shift h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = np.sum(1.0 - h)
        if cap > 0:
            h = h + diff * (1.0 - h) / cap
    else:
        cap = np.sum(h)
        if cap > 0:
            h = h + diff * h / cap
    # final tiny correction
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        i = 0 if diff2 > 0 else np.argmax(h)
        h[i] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


# ------------------ line search ---------------------------------------


def line_search(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator) -> tuple:
    """Backtracking line search along -grad, returns (new_h, new_c5)."""
    d = -grad
    pos = d > 1e-12
    neg = d < -1e-12
    steps = []
    if np.any(pos):
        steps.append(((1.0 - h[pos]) / d[pos]).min())
    if np.any(neg):
        steps.append((h[neg] / -d[neg]).min())
    if not steps:
        return h, compute_c5(h)
    step_max = min(steps)
    step = step_max
    cur_c = compute_c5(h)
    min_step = step_max * 1e-6
    attempts = 0
    while step >= min_step and attempts < 15:
        h_new = h + step * d
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new)
        c_new = compute_c5(h_new)
        if c_new < cur_c - 1e-14:
            return h_new, c_new
        step *= 0.5
        attempts += 1
    return h, cur_c


# ------------------ gradient descent ----------------------------------


def gradient_descent(h0: np.ndarray, rng: np.random.Generator,
                    max_iter: int, time_limit: float, start_time: float) -> tuple:
    """Projected gradient descent with line search."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    it = 0
    while it < max_iter and time.time() - start_time < time_limit:
        h = best_h
        grad = worst_gradient(h)
        h_new, c_new = line_search(h, grad, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        else:
            # small random transfer as fallback
            i, j = rng.integers(0, h.size, size=2)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = rng.random() * max_eps * 0.02
                    h2 = h.copy()
                    h2[i] -= eps
                    h2[j] += eps
                    h2 = enforce_sum(h2)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()
        it += 1
        if best_c <= 0.3808:
            break
    return best_h, best_c


# ------------------ soft‑max SLSQP ------------------------------------


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int) -> np.ndarray:
    """SLSQP on the smooth soft‑max approximation (gradient supplied)."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        max_bc = np.max(bc)
        w = np.exp(bc - max_bc)
        w = w / w.sum()
        # S_minus = sum_k w_k * x[i - k]
        conv_w = np.convolve(w, x, mode="full")
        s_minus = conv_w[n - 1 : n - 1 + n]
        # S_plus = sum_k w_k * x[i + k] using convolution with reversed x
        conv_rev = np.convolve(w, x[::-1], mode="full")
        s_plus = conv_rev[n - 1 :: -1][:n]
        return 1.0 - s_plus - s_minus

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = opt.minimize(obj, h0, method="SLSQP", jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x)
    return x


# ------------------ helper constructors -------------------------------


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue‑Morse sequence as 0/1 floats."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count("1") % 2
    return seq


def random_balanced(n: int, rng: np.random.Generator) -> np.ndarray:
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


# ------------------ main run ------------------------------------------
