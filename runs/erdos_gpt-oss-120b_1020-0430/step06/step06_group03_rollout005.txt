def softmax_gradient_fft(h, beta):
    n = h.shape[0]
    # correlation O
    m = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    O = np.fft.irfft(H * G, m)[:2*n-1]
    Omax = O.max()
    # compute softmax weights w
    # Use exp(beta*(O - Omax)) for stability
    exp_term = np.exp(beta * (O - Omax))

    conv = np.fft.irfft(W * H, m)[:2*n-1]
    grad = wk0 * (1.0 - h) - conv[n-1 : n-1 + n]
    return grad

def softmax_gradient_fft(h, beta):
    n = h.shape[0]
    # compute next power of two m >= 2*n-1
    m = 1 << ((2*n-1).bit_length())
    # FFT of h
    H = np.fft.rfft(h, m)
    # FFT of 1-h
    G = np.fft.rfft(1.0 - h, m)
    # correlation O
    O = np.fft.irfft(H * G, m)[:2*n-1]
    Omax = O.max()
    # softmax weights
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    wk0 = w[n-1]
    # convolution of w with h
    W = np.fft.rfft(w, m)
    conv = np.fft.irfft(W * H, m)[:2*n-1]
    # gradient
    grad = wk0 * (1.0 - h) - conv[n-1:n-1+n]
    return grad

import time, numpy as np

# ---------- utilities ----------
def project_onto_box_sum(v, target):
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def correlation_full_fft(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * G, m)[: 2 * n - 1]
    return conv

def compute_c5_fft(h):
    n = h.shape[0]
    corr = correlation_full_fft(h)
    return corr.max() * (2.0 / n)

def compute_c5_corr(h):
    n = h.shape[0]
    return np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n)

def worst_gradient(h):
    n = h.shape[0]
    corr = correlation_full_fft(h)
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5_corr(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5_corr(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    best_c = compute_c5_corr(best_h)
    return best_h, best_c

def softmax_gradient_fft(h, beta):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    O = np.fft.irfft(H * G, m)[: 2 * n - 1]
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    wk0 = w[n - 1]
    W = np.fft.rfft(w, m)
    conv = np.fft.irfft(W * H, m)[: 2 * n - 1]
    grad = wk0 * (1.0 - h) - conv[n - 1 : n - 1 + n]
    return grad

    if beta <= 200:
        return 2.0
    if beta <= 3000:
        return 1.0
