def worst_gradient_for_shift(h, k):
    n = h.size
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # k may be negative; use absolute.
        k_abs = abs(k)
        grad[: n - k_abs] -= h[k_abs:]
        grad[k_abs:] -= h[: n - k_abs]
    return grad

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    target_n = 1024
    # Build initial h
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        # upsample to target_n by repeating proportionally
        # If len(init) == target_n, just copy.
        if init.size == target_n:
            h0 = init.copy()
        else:
            # simple repeat factor = target_n // init.size
            factor = target_n // init.size
            rem = target_n % init.size
            h0 = np.repeat(init, factor)
            if rem > 0:
                h0 = np.concatenate([h0, init[:rem]])
        # Ensure sum target_n/2
        target = target_n / 2.0
        # Project onto box sum
        h0 = project_onto_box_sum(h0, target)
    else:
        target_n = 1024
        target = target_n / 2.0

    best_k = best_k

import time, math, numpy as np

def _project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _worst_gradient(h, k):
    n = h.size
    g = np.ones_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        kk = abs(k)
        g[: n - kk] -= h[kk:]
        g[kk:] -= h[: n - kk]
    return g

def _c5_fft_with_idx(h):
    n = h.size
    N = 2 * n
    f = np.fft.rfft(h, n=N)
    g = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(f * np.conj(g), n=N)
    idx = int(np.argmax(corr[:N - 1]))
    maxc = corr[idx]
    k = idx - (n - 1)
    return maxc * (2.0 / n), k

def _refine_grad(h, iters):
    best_h, best_c = h.copy(), _c5_fft_with_idx(h)[0]
    cur_h = h.copy()
    for it in range(iters):
        cur_c, cur_k = _c5_fft_with_idx(cur_h)
        grad = _worst_gradient(cur_h, cur_k)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            break
        eps = max_eps * (1.0 - it / iters) * 0.3
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        cnew = _c5_fft_with_idx(nh)[0]
        if cnew < best_c:

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_target = 1024
    target_sum = n_target / 2.0

    # ---- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n_target:
            if n_target % h0.size == 0:
                factor = n_target // h0.size
                h0 = np.repeat(h0, factor)
            else:
                # linear interpolation to required size
                x_old = np.linspace(0, 2, h0.size, endpoint=False)
                x_new = np.linspace(0, 2, n_target, endpoint=False)
                h0 = np.interp(x_new, x_old, h0)
        h0 = _project_onto_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c, best_k = _c5_fft_with_idx(best_h)

    temp = 0.001
    it = 0

    # ---- simulated annealing -----------------------------------------
    while time.time() - start < budget_s * 0.93:
        if it % 2 == 0:
            grad = _worst_gradient(cur_h, cur_k)

            i = rng.choice(ones)

        temp *= 0.99997

    remaining = budget_s - (time.time() - start)
