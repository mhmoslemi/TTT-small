import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Maximum overlap integral (discrete version)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Project h onto {0≤h_i≤1, sum(h)=n/2} by greedy swaps."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- smooth‑max and its gradient ----------
def smoothmax_obj(h, beta):
    v = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * v
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def gradient_smoothmax(h, beta):
    """∇ of smooth‑max (log‑sum‑exp) w.r.t. h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * v
    m = np.max(bc)
    exp_term = np.exp(bc - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    omh = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * omh[shift:]
            g[shift:]      -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:]          += wi * omh[: n - s]
            g[: n - s]     -= wi * h[s:]
    return g

# ---------- worst‑shift subgradient ----------
def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:]      -= h[: n - k]
        else:
            kk = -k
            g[kk:]      += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

# ---------- refinement helpers ----------
def gradient_descent_smoothmax(h, betas, step, it_per_beta, rng, deadline):
    """Projected GD on smooth‑max."""
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0):
                cand = h - step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h

    """Sub‑gradient descent on the true max."""
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.95
    return h

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth‑max with analytic gradient."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        return smoothmax_obj(x, beta)

    def grad(x):
        return gradient_smoothmax(x, beta)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        jac=grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum repair
    adjust_sum_exact(x, rng)
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    """Swap‑based simulated‑annealing on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    T0 = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = T0 * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < math.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    best_h = None
    best_c = float("inf")
    best_n = None

    # ----- base size and parent handling -----
    base_n = 720
    parent = globals().get("initial_h_values")
    if parent is not None:
        # ensure even length
        if len(parent) % 2:
            parent = parent[:-1]
        # interpolate parent to base_n if needed
        if len(parent) != base_n:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            parent_interp = np.interp(xt, xs, parent)
        else:
            parent_interp = np.array(parent, dtype=float)
    else:
        parent_interp = None

    betas = [5, 20, 80, 200, 500, 2000, 5000, 20000, 50000]

    # ----- multiple restarts -----
    max_restarts = 12
    for r in range(max_restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent_interp is not None:
            h = parent_interp.copy()
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # smooth‑max GD (more iterations)
        h = gradient_descent_smoothmax(
            h,
            betas,
            step=0.26,
            it_per_beta=20,
            rng=rng,
            deadline=deadline,
        )

        # worst‑shift refinement
        h = worst_shift_refine(h, rng, step=0.13, iters=60, deadline=deadline)

        # SLSQP polishing (high β)
        for beta in (2000, 5000, 20000, 50000):
            if time.time() > deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=500)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n
            if best_c < 0.38078:  # early kill switch
                break

        # binary SA on rounded version
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n
            if best_c < 0.38078:
                break

    # ----- up‑sampling if time permits -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)

        # quick GD
        h_up = gradient_descent_smoothmax(
            h_up, betas[:6], step=0.22, it_per_beta=12, rng=rng, deadline=deadline
        )
        h_up = worst_shift_refine(h_up, rng, step=0.11, iters=30, deadline=deadline)
        for beta in (2000, 5000, 20000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=400)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

        # binary SA on up‑sampled solution
        h_bin = np.rint(h_up).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n_up

    # ----- fallback -----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
