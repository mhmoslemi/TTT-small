def hill_climb(h0, rng, target, max_iters, eps_start, eps_end, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        # compute gradient
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # choose random pair
            idx = rng.choice(len(h), size=2, replace=False)
            i, j = idx[0], idx[1]
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
        # schedule epsilon
        eps_frac = (max_iters - it) / max_iters  # decreasing
        eps = max_eps * eps_frac * eps_start  # maybe too large
        # We want eps in [eps_end, eps_start]
        # Option: eps = eps_end + (eps_start - eps_end) * ((max_iters - it) / max_iters)
        eps = eps_end + (eps_start - eps_end) * ((max_iters - it) / max_iters)
        eps = rng.random() * max_eps * eps
        # update
        h[i] -= eps
        h[j] += eps
        # compute C5
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def compute_c5(h):

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)

            break
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            break
        eps = max_eps * 1e-4
        best_h[i] -= eps
        best_h[j] += eps
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c
        if rng.random() < 0.01:
            best_h = project_onto_box_sum(best_h, target)

    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = project_onto_box_sum(final_h, target)
    final_c = compute_c5(final_h)
    return final_h.tolist(), float(final_c), int(final_h.size)
