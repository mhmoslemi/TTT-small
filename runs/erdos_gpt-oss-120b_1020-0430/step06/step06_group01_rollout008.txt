def project_onto_bounds_sum(v, target):
    # project v onto [0,1] and sum target.
    # Inverse: solution of min ||x - v||^2 s.t. sum x = target, 0 <= x <= 1.
    # Use algorithm: clamp v to [0,1] then adjust via water-filling.
    # Actually known algorithm: sort v, then find lambda such that sum(min(max(v_i - lambda, 0), 1)) = target.
    # We can implement via binary search on lambda within [min(v)-1, max(v)].
    # Since n relatively small, binary search fine.

    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def binary_sa_n(n, rng, time_budget):
    start = time.time()
    # initial binary vector with n/2 ones
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    ones_set = set(ones.tolist())
    zeros_set = set(range(n)) - ones_set
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    T0 = 0.02
    T_end = 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick random one and zero to swap
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            # accept
            c_curr = c_new
            # update sets
            ones_set.remove(i)
            zeros_set.remove(j)
            ones_set.add(j)
            zeros_set.add(i)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # reject, revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

def project_onto_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def binary_sa_n(n, rng, time_budget, init_bin=None):
    start = time.time()
    if init_bin is not None:
        h = init_bin.astype(float).copy()
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    # enforce exact half‑ones
    diff = int(h.sum()) - n // 2
    if diff != 0:
        if diff > 0:
            idx = np.where(h == 1.0)[0]
            turn = rng.choice(idx, diff, replace=False)
            h[turn] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            turn = rng.choice(idx, -diff, replace=False)
            h[turn] = 1.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    T0, T_end = 0.02, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(n)
        val_i = h[i]
        j = rng.integers(n)
