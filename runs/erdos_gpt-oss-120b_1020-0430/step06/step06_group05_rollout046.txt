def run_ga(rng, time_limit, start_time, n, target_sum):

    pop_size = 150

    # init pop

    pop = [] # list of np.ndarray uint8

    pop_c5 = [] # list of float

    # candidate include binary from initial_h_values if available

    # fill rest random

    while len(pop) < pop_size:

        h = np.zeros(n, dtype=np.uint8)

        idx = rng.choice(n, target_sum, replace=False)

        h[idx] = 1

        pop.append(h)

        pop_c5.append(compute_c5(h.astype(float)))  # compute with float

    best_idx = np.argmin(pop_c5)

    best_h = pop[best_idx].copy()

    best_c = pop_c5[best_idx]

    # GA loop

    while time.time() - start_time < time_limit:

        # select parents via tournament

        def tournament_select(t=5):

            candidates = rng.integers(0, pop_size, size=t)

            best = candidates[np.argmin([pop_c5[i] for i in candidates])]

            return pop[best]

        p1 = tournament_select()

        p2 = tournament_select()

        # crossover uniform

        mask = rng.random(n) < 0.5

        child = np.where(mask, p1, p2).astype(np.uint8)

        # adjust sum

        s = child.sum()

        if s > target_sum:

            # need to turn off (s-target_sum) ones

            ones_idx = np.where(child == 1)[0]

            turn_off = rng.choice(ones_idx, s - target_sum, replace=False)

            child[turn_off] = 0

        elif s < target_sum:

            zeros_idx = np.where(child == 0)[0]

            turn_on = rng.choice(zeros_idx, target_sum - s, replace=False)

            child[turn_on] = 1

        # mutation: swap random pairs

        if rng.random() < 0.3:  # mutation prob

            num_swaps = rng.integers(1, 4)  # 1-3 swaps

            for _ in range(num_swaps):

                ones_idx = np.where(child == 1)[0]

                zeros_idx = np.where(child == 0)[0]

                if len(ones_idx)==0 or len(zeros_idx)==0:

                    break

                i = rng.choice(ones_idx)

                j = rng.choice(zeros_idx)

                child[i], child[j] = child[j], child[i]  # swap

        child_c = compute_c5(child.astype(float))

        # replacement: if child better than worst

        worst_idx = np.argmax(pop_c5)

        if child_c < pop_c5[worst_idx]:

            pop[worst_idx] = child

            pop_c5[worst_idx] = child_c

            if child_c < best_c:

                best_c = child_c

                best_h = child.copy()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n/2.0

    # initial continuous candidate from provided values
    best_cont = None
    best_cont_c = np.inf
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = enforce_sum(init_h, target_sum, rng)
        best_cont = init_h
        best_cont_c = compute_c5(best_cont)

    # GA time allocation
    remaining = budget_s - (time.time() - start)
    ga_time = max(remaining * 0.7, 0.1)  # allocate 70% if enough

    # Run GA
    best_bin, best_bin_c = run_ga(rng, ga_time, start, n, target_sum)

    # Update best continuous if binary better
    if best_bin_c < best_cont_c:
        best_cont = best_bin.astype(float)
        best_cont_c = best_bin_c

    # Local continuous refinement
    # Use gradient_refine with many iterations
    if time.time() - start < budget_s * 0.95:
        best_cont = gradient_refine(best_cont, target_sum, rng, step=0.02, max_iter=30)
        best_cont_c = compute_c5(best_cont)

    # SLSQP refinement if time
    if time.time() - start < budget_s * 0.95:
        # try several betas
        for beta in (2000.0, 5000.0, 10000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            tmp = slsqp_refine(best_cont, target_sum, rng, beta=beta, max_iter=600)
            tmp_c = compute_c5(tmp)
            if tmp_c < best_cont_c:
                best_cont = tmp
                best_cont_c = tmp_c

    # final enforce sum
    best_cont = np.clip(best_cont, 0.0, 1.0)
    best_cont = enforce_sum(best_cont, target_sum, rng)
    final_c = compute_c5(best_cont)

    return best_cont.tolist(), float(final_c), int(best_cont.size)

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    h_new[pos] = h[pos] * (1.0 - eps / total_out)
    h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def alpha_schedule(it: int, max_it: int):
    """Piecewise decreasing alphas."""
    frac = it / max_it
    if frac < 0.3:
        return (1.0, 0.5, 0.25, 0.125, 0.0625)
    if frac < 0.6:
        return (0.5, 0.25, 0.125, 0.0625, 0.03125)
    if frac < 0.9:
        return (0.25, 0.125, 0.0625, 0.03125, 0.015625)
    return (0.125, 0.0625, 0.03125, 0.015625, 0.0078125)


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(softmax_obj, h, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


def run_ga(rng, end_time, start_time, n, target_sum):
    """Simple genetic algorithm for binary candidates."""
    pop_size = 150
    pop = []
    pop_c5 = []
    # initialise population
    while len(pop) < pop_size:
        h = np.zeros(n, dtype=np.uint8)
        idx = rng.choice(n, int(target_sum), replace=False)
        h[idx] = 1
        val = compute_c5(h.astype(float))
        pop.append(h)
        pop_c5.append(val)
    best_idx = int(np.argmin(pop_c5))
    best_h = pop[best_idx].copy()
    best_c = pop_c5[best_idx]

    def tournament():
        idxs = rng.integers(0, pop_size, size=5)
        best_local = idxs[0]
        best_val = pop_c5[best_local]
        for ix in idxs[1:]:
            if pop_c5[ix] < best_val:
                best_val = pop_c5[ix]
                best_local = ix
        return pop[best_local]

    while time.time() < end_time:
        p1 = tournament()
        p2 = tournament()
        mask = rng.random(n) < 0.5
        child = np.where(mask, p1, p2).astype(np.uint8)

        s = child.sum()
        if s > target_sum:
            ones = np.where(child == 1)[0]
            child[rng.choice(ones, int(s - target_sum), replace=False)] = 0
        elif s < target_sum:
            zeros = np.where(child == 0)[0]
            child[rng.choice(zeros, int(target_sum - s), replace=False)] = 1

        if rng.random() < 0.3:
            for _ in range(rng.integers(1, 4)):
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                child[i], child[j] = child[j], child[i]

        child_c = compute_c5(child.astype(float))
        worst_idx = int(np.argmax(pop_c5))
        if child_c < pop_c5[worst_idx]:
            pop[worst_idx] = child
            pop_c5[worst_idx] = child_c
            if child_c < best_c:
                best_c = child_c
                best_h = child.copy()
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0

    # initial continuous candidate if supplied
    best_cont = None
    best_cont_c = np.inf
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = enforce_sum(init_h, target_sum, rng)
        best_cont = init_h
        best_cont_c = compute_c5(best_cont)

    # allocate time for GA (≈70%)
    time_left = budget_s - (time.time() - start)
    ga_time = max(time_left * 0.7, 0.1)
    ga_end = start + ga_time

    # run GA
    best_bin, best_bin_c = run_ga(rng, ga_end, start, n, target_sum)

    if best_bin_c < best_cont_c:
        best_cont = best_bin.astype(float)
        best_cont_c = best_bin_c

    # gradient refinement (if time permits)
    if time.time() - start < budget_s * 0.95:
        best_cont = gradient_refine(best_cont, target_sum, rng,
                                   step=0.02, max_iter=30)
        best_cont_c = compute_c5(best_cont)

    # SLSQP refinement with several betas
    if time.time() - start < budget_s * 0.95:
        for beta in (2000.0, 5000.0, 10000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            tmp = slsqp_refine(best_cont, target_sum, rng,
                               beta=beta, max_iter=600)
            tmp_c = compute_c5(tmp)
            if tmp_c < best_cont_c:
                best_cont = tmp
                best_cont_c = tmp_c

    # final enforcement and output
    best_cont = np.clip(best_cont, 0.0, 1.0)
    best_cont = enforce_sum(best_cont, target_sum, rng)
    final_c = compute_c5(best_cont)

    return best_cont.tolist(), float(final_c), int(best_cont.size)
