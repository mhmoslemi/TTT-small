import time
import numpy as np

# -----------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for vector h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """In‑place modify h so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# -----------------------------------------------------------

    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    omh = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * omh[shift:]
            g[shift:]    -= wi * h[: n - shift]
        else:
            kk = -shift
            g[kk:]      += wi * omh[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    """Projected gradient descent on the smooth‑max surrogate."""
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-15:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 0.125):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h

# -----------------------------------------------------------
def worst_shift_gradient(h):
    """Sub‑gradient using all currently worst shifts."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:]      -= h[: n - k]
        else:
            kk = -k
            g[kk:]    += 1.0 - h[: n - kk]
            g[: n-kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def worst_shift_refine(h, rng, base_step, iters, deadline):
    """Projected sub‑gradient descent focused on worst shift."""
    best_c = compute_c5(h)
    best_h = h
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-15:
            break
        for mult in (1.0, 0.5, 0.25, 0.125):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.96
    return h

# -----------------------------------------------------------
def local_move_refine(h, rng, max_iters, deadline):
    """Move mass from high to low contribution positions on the worst shift."""
    n = h.size
    best_c = compute_c5(h)
    best_h = h
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        if shift >= 0:
            contrib = h[: n - shift] * (1.0 - h[shift:])
            i_hi = np.argmax(contrib)
            i_lo = np.argmin(contrib)
        else:
            kk = -shift
            contrib = h[kk:] * (1.0 - h[: n - kk])
            i_hi = np.argmax(contrib) + kk
            i_lo = np.argmin(contrib) + kk
        if i_hi == i_lo:
            continue
        eps = min(0.02, h[i_hi], 1.0 - h[i_lo])
        if eps < 1e-12:
            continue
        cand = h.copy()
        cand[i_hi] -= eps
        cand[i_lo] += eps
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand
            h = cand
            continue
        # try half epsilon
        eps2 = eps * 0.5
        cand[i_hi] += eps - eps2
        cand[i_lo] -= eps - eps2
        c2 = compute_c5(cand)
        if c2 < best_c:
            best_c = c2
            best_h = cand
            h = cand
    return best_h

# -----------------------------------------------------------
def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill‑climb on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    if not ones or not zeros:
        return best_h, best_c
    init_T = max(1e-6, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 12000:
            break
    return best_h, best_c

# -----------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    base_n = 720
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        if parent.size != base_n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
            parent = np.interp(xt, xs, parent)
    else:
        parent = None

    restarts = 5
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent is not None:
            h = parent.copy()
        else:
            h = rng.random(n)
            h *= (n / 2.0) / h.sum()
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # smooth‑max gradient descent
        betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]
        h = gradient_descent_smoothmax(h, betas, base_step=0.3,
                                       it_per_beta=20, rng=rng,
                                       deadline=deadline)

        # worst‑shift refinement
        h = worst_shift_refine(h, rng, base_step=0.15, iters=200,
                               deadline=deadline)

        # local move refinement
        h = local_move_refine(h, rng, max_iters=3000, deadline=deadline)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # binary rounding + SA
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # optional up‑sampling
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        np.clip(h_up, 0.0, 1.0, out=h_up)
        # up‑sampled refinement (lighter)
        betas_up = [30, 100, 300, 1000, 3000, 10000]
        h_up = gradient_descent_smoothmax(h_up, betas_up, base_step=0.2,
                                          it_per_beta=12, rng=rng,
                                          deadline=deadline)
        h_up = worst_shift_refine(h_up, rng, base_step=0.12, iters=120,
                                  deadline=deadline)
        h_up = local_move_refine(h_up, rng, max_iters=1500, deadline=deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up
        # binary SA on up‑sampled
        h_bin = np.rint(h_up).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n_up

    # final safety fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
