    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # List of indices for ones and zeros
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    no_improve = 0
    swaps = 0
    while time.time() < deadline and swaps < max_swaps and no_improve < no_improve_limit:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        if c < cur_c:
            cur_c = c
            # update sets
            ones = ones[ones != i]  # remove i
            zeros = zeros[zeros != j]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            # record best
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
            no_improve = 0
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
            no_improve += 1
        swaps += 1
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:] -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:] += wi * one_minus_h[: n - s]
            g[: n - s] -= wi * h[s:]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_c = compute_c5(h)
            best_h = h
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h

def worst_shift_refine(h, rng, base_step, max_iters, deadline):
    step = base_step
    for i in range(max_iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_c = compute_c5(h)
        best_h = h
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.98
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    v = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * v
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum exactly
    adjust_sum_exact(x, rng)
    return x

def binary_hill_climb(h_bin, rng, deadline, max_swaps=150000, no_improve_limit=15000):
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    no_improve = 0
    swaps = 0
    n = cur_h.size
    while time.time() < deadline and swaps < max_swaps and no_improve < no_improve_limit:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        if c < cur_c:
            cur_c = c
            # update index arrays
            # remove i from ones, add j
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
            no_improve = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
            no_improve += 1
        swaps += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96  # safety margin

    parent = globals().get("initial_h_values", None)
    base_n = 720
    best_h = None
    best_c = np.inf
    best_n = None

    # Prepare parent if exists
    if parent is not None and len(parent) == base_n:
        parent_h = np.array(parent, dtype=np.float64)
    else:
        parent_h = None

    restarts = 4
    betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    it_per_beta = 12

    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent_h is not None:
            h = parent_h.copy()
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # smoothmax gradient descent
        h = gradient_descent_smoothmax(h, betas, base_step=0.35,
                                      it_per_beta=it_per_beta,
                                      rng=rng, deadline=deadline)

        # worst shift refinement
        h = worst_shift_refine(h, rng, base_step=0.15, max_iters=200,
                               deadline=deadline)

        # final SLSQP polish with a few high betas
        for beta in (30000, 100000):
            if time.time() > deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # binary hill climb on rounded version
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_bin, c_bin = binary_hill_climb(h_bin, rng, sub_dead,
                                         max_swaps=200000,
                                         no_improve_limit=20000)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # final safety fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities ------------------- #
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000):
    """Repair h in‑place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ------------------- smooth‑max helpers ------------------- #
def gradient_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log Σ exp(β·overlap)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:] -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:] += wi * one_minus_h[: n - s]
            g[: n - s] -= wi * h[s:]
    return g


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient using the currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g


# ------------------- descent procedures ------------------- #
def gradient_descent_smoothmax(
    h: np.ndarray,
    betas: list,
    base_step: float,
    it_per_beta: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """Projected gradient descent on the smooth‑max surrogate."""
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_c = compute_c5(h)
            best_h = h
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h


def worst_shift_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    base_step: float,
    max_iters: int,
    deadline: float,
) -> np.ndarray:
    """Iterative refinement using the worst‑shift sub‑gradient."""
    step = base_step
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_c = compute_c5(h)
        best_h = h
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.98
    return h


# ------------------- SLSQP polish ------------------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    v = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * v
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(
    h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 800
) -> np.ndarray:
    """SLSQP on the smooth‑max surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


# ------------------- binary hill‑climb ------------------- #
def binary_hill_climb(
    h_bin: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_swaps: int = 150000,
    no_improve_limit: int = 15000,
) -> tuple:
    """Deterministic hill‑climb on a binary vector (swap 1↔0)."""
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    no_improve = 0
    swaps = 0
    while time.time() < deadline and swaps < max_swaps and no_improve < no_improve_limit:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        if c < cur_c:
            cur_c = c
            # update index collections
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
            no_improve = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
            no_improve += 1
        swaps += 1
    return best_h, best_c
