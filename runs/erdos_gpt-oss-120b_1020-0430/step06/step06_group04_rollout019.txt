import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = shift_idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k :]    -= wi * h[: n - k]
        else:
            kk = -k
            g[kk :]   += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk :]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    n = h.size
    for beta in betas:
        for it in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):
    n = h.size
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.95
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=1200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                             'ftol': 1e-12,
                                             'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c

def continuous_sa_hill_climb(h_cont, rng, deadline):
    n = h_cont.size
    cur_h = h_cont.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    total = max(deadline - start, 1e-12)
    step_counter = 0
    while time.time() < deadline:
        # choose distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # randomly decide direction
        if rng.random() < 0.5:
            # increase i, decrease j
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                continue
            # temperature factor
            remaining = deadline - time.time()
            T = max(1e-6, remaining / total)
            delta = max_delta * T
            if delta < 1e-12:
                continue
            cur_h[i] += delta
            cur_h[j] -= delta
        else:
            # increase j, decrease i
            max_delta = min(1.0 - cur_h[j], cur_h[i])
            if max_delta <= 0:
                continue
            remaining = deadline - time.time()
            T = max(1e-6, remaining / total)
            delta = max_delta * T
            if delta < 1e-12:
                continue
            cur_h[j] += delta
            cur_h[i] -= delta
        c = compute_c5(cur_h)
        # acceptance
        remaining = deadline - time.time()
        T = max(1e-6, remaining / total)
        if c < cur_c:
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            # probability to accept worse
            # scale temperature to objective magnitude
            accept_prob = np.exp(-(c - cur_c) / (T * 0.01))
            if rng.random() < accept_prob:
                cur_c = c
            else:
                # revert move
                if rng.random() < 0.5:
                    # revert increase i, decrease j
                    cur_h[i] -= delta
                    cur_h[j] += delta
                else:
                    cur_h[j] -= delta
                    cur_h[i] += delta
        step_counter += 1
        if step_counter > 200000:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += wi * one_minus_h[k:]
            g[k:]      -= wi * h[: n - k]
        else:
            kk = -k
            g[kk:]     += wi * one_minus_h[: n - kk]
            g[: n - kk] -= wi * h[kk:]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:]      -= h[: n - k]
        else:
            kk = -k
            g[kk:]     += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.95
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons, options={"maxiter": max_iter,
                                  "ftol": 1e-12,
                                  "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c

def continuous_sa_hill_climb(h_cont, rng, deadline):
    n = h_cont.size
    cur_h = h_cont.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    total = max(deadline - start, 1e-12)
    steps = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                continue
            remaining = deadline - time.time()
            T = max(1e-6, remaining / total)
            delta = max_delta * T
            if delta < 1e-12:
                continue
            cur_h[i] += delta
            cur_h[j] -= delta
            applied = (i, j, delta)
        else:
            max_delta = min(1.0 - cur_h[j], cur_h[i])
            if max_delta <= 0:
                continue
            remaining = deadline - time.time()
            T = max(1e-6, remaining / total)
            delta = max_delta * T
            if delta < 1e-12:
                continue
            cur_h[j] += delta
            cur_h[i] -= delta
            applied = (j, i, delta)
        c = compute_c5(cur_h)
        remaining = deadline - time.time()
        T = max(1e-6, remaining / total)
        if c < cur_c:
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            accept = np.exp(-(c - cur_c) / (T * 0.01))
            if rng.random() >= accept:
                a, b, d = applied
                cur_h[a] -= d
                cur_h[b] += d
        steps += 1
        if steps > 200000:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    base_n = 720
    parent = globals().get('initial_h_values')
    if parent is not None:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
        parent_interp = np.interp(xt, xs, parent)
    else:
        parent_interp = None

    betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000]
    restarts = 3
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent_interp is not None:
            h = parent_interp.copy()
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        h = gradient_descent_smoothmax(h, betas, base_step=0.2,
                                       it_per_beta=15, rng=rng,
                                       deadline=deadline)
        h = worst_shift_refine(h, rng, base_step=0.12, iters=70,
                               deadline=deadline)
        for beta in (2000, 5000, 20000, 50000):
            if time.time() > deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

    # finer local search on the best candidate
    if best_h is not None and time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.4
        h_cont, c_cont = continuous_sa_hill_climb(best_h, rng, sub_dead)
        if c_cont < best_c:
            best_c, best_h, best_n = c_cont, h_cont.copy(), best_n
        if time.time() < deadline:
            h_bin = np.rint(best_h).astype(float)
            adjust_sum_exact(h_bin, rng)
            sub_dead = time.time() + (deadline - time.time()) * 0.2
            h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
            if c_bin < best_c:
                best_c, best_h, best_n = c_bin, h_bin.copy(), best_n

    # up‑sample and repeat refinement if time permits
    if (best_h is not None and time.time() < deadline
            and best_n * 2 <= 1440):
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_descent_smoothmax(h_up, betas[:8],
                                          base_step=0.15,
                                          it_per_beta=10,
                                          rng=rng,
                                          deadline=deadline)
        h_up = worst_shift_refine(h_up, rng, base_step=0.10,
                                 iters=40, deadline=deadline)
        for beta in (2000, 5000, 20000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

        if time.time() < deadline:
            sub_dead = time.time() + (deadline - time.time()) * 0.25
            h_cont_up, c_cont_up = continuous_sa_hill_climb(h_up, rng, sub_dead)
            if c_cont_up < best_c:
                best_c, best_h, best_n = c_cont_up, h_cont_up.copy(), n_up

        if time.time() < deadline:
            h_bin_up = np.rint(h_up).astype(float)
            adjust_sum_exact(h_bin_up, rng)
            sub_dead = time.time() + (deadline - time.time()) * 0.15
            h_bin_up, c_bin_up = binary_sa_hill_climb(h_bin_up, rng, sub_dead)
            if c_bin_up < best_c:
                best_c, best_h, best_n = c_bin_up, h_bin_up.copy(), n_up

    # safety fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
