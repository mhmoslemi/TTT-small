def anneal_optimize(h, current_c, rng, max_seconds):
    start = time.time()
    T = current_c * 0.1 if current_c > 0 else 0.01
    final_T = 1e-9
    decay = 0.99995
    n = h.size
    while time.time() - start < max_seconds - 0.01:  # ensure a small safety margin
        # choose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # small vs large eps
        if rng.random() < 0.2:  # large move
            eps = max_eps
        else:
            eps = rng.random() * max_eps
        # apply move
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5_fast(h)
        delta = new_c - current_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_c = new_c
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        T = max(T * decay, final_T)
    return h, current_c

import time
import numpy as np

def compute_c5_fast(h: np.ndarray) -> float:
    """C5 using FFT (linear correlation)."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, n=m)
    rev_g = (1.0 - h)[::-1]                       # reversed (1‑h)
    GR = np.fft.rfft(rev_g, n=m)
    corr = np.fft.irfft(H * GR, n=m)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via bisection on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the box with sum=n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation to twice the resolution, then re‑project."""
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)


def anneal_optimize(h: np.ndarray, cur_c: float,
                   rng: np.random.Generator, max_seconds: float) -> tuple:
    """Simulated annealing on the feasible polytope."""
    start = time.time()
    T = max(cur_c * 0.1, 0.01)
    final_T = 1e-9
    decay = 0.99995
    n = h.size
    while time.time() - start < max_seconds - 0.01:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps if rng.random() < 0.2 else rng.random() * max_eps
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5_fast(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
        else:
            h[i] += eps
            h[j] -= eps
        T = max(T * decay, final_T)
    return h, cur_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    elapsed = lambda: time.time() - start

    # ---- initialise best solution -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = random_feasible(n, rng)

    best_h = h0.copy()
    best_c = compute_c5_fast(best_h)

    cur_h = best_h.copy()
    cur_c = best_c
    no_improve = 0

    # ---- main optimisation loop ---------------------------------------------------
    while elapsed() < budget_s - 2.0:
        rem = budget_s - elapsed()

        # optional up‑sampling (once)
        if cur_h.size < 1024 and rem > 40.0:
            up_h = upsample(cur_h, rng)
            up_c = compute_c5_fast(up_h)
            cur_h, cur_c = up_h, up_c
            if up_c < best_c:
                best_h, best_c = up_h, up_c
                no_improve = 0

        # annealing chunk
        chunk = min(rem - 2.0, 15.0)
        if chunk <= 0.0:
            break
        cur_h, cur_c = anneal_optimize(cur_h, cur_c, rng, chunk)

        if cur_c < best_c:
            best_h, best_c = cur_h.copy(), cur_c
            no_improve = 0
        else:
            no_improve += 1

        # restart if stuck
        if no_improve >= 5 and rem > 20.0:
            n = cur_h.size
            cur_h = random_feasible(n, rng)
            cur_c = compute_c5_fast(cur_h)
            no_improve = 0

        # early exit if target achieved
        if best_c <= 0.3808:
            break

    # ---- final sanitisation -------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
