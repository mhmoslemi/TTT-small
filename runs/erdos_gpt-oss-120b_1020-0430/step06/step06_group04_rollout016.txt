def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # full length 2n-1, center at n-1 corresponds k=0.
    # C5 = max corr * dx
    return float(np.max(corr) * (2.0 / n))

def slsqp_optimize(h0, beta, max_iter, rng):

   n = h0.size
   target = n/2.0
   bounds = [(0.0,1.0)] * n
   cons = [{'type':'eq','fun': lambda x: np.sum(x)-target,
            'jac': lambda x: np.ones_like(x)}]
   # We'll use jac for objective: gradient_smoothmax
   res = minimize(lambda x: softmax_obj(x,beta), h0, method='SLSQP', jac=lambda x: gradient_smoothmax(x,beta),
                  bounds=bounds, constraints=cons,
                  options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
   x = np.clip(res.x,0,1)
   # adjust sum exactly
   adjust_sum_exact(x, rng)
   return x

    corr = np.correlate(x,1-x,mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
