def worst_shift_and_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)  # shift (positive or negative)
    val = corr[idx] * (2.0 / n)
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        if k > 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:  # k < 0
            kk = -k
            grad[kk:] -= h[:n - kk]
            grad[:n - kk] -= h[kk:]
    return k, val, grad

        j = order[0]

    h = np.zeros(n, dtype=float)

                    break

            h[i] = 1.0
            h[j] = 0.0

    if abs(diff) < 1e-12:
        return h
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            if inc <= 0:
                continue
            h[i] += inc
            diff -= inc
        else:
            dec = min(-diff, h[i])
            if dec <= 0:
                continue
            h[i] -= dec
            diff += dec
    return h


def worst_shift_and_gradient(h: np.ndarray):
    """
    Return (k, C5 value, gradient) for the shift that attains the maximum overlap.
    Gradient is the derivative of the overlap with respect to h for that shift.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)                # shift (may be negative)
    val = corr[idx] * (2.0 / n)      # C5 for this shift

    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return k, val, grad


# -------------------------------------------------
# binary simulated annealing (n = 512)
# -------------------------------------------------
def binary_sa_n(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    # random binary start with exactly n/2 ones
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    cur_c = compute_c5(h)
    best_c, best_h = cur_c, h.copy()
    T0, T1 = 0.02, 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T1 / T0) ** (elapsed / time_budget)

        # pick a random 1 and a random 0 to swap
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# -------------------------------------------------
# deterministic greedy descent (continuous)
# -------------------------------------------------
def greedy_descent(h0: np.ndarray, rng: np.random.Generator, time_limit: float):
    """
    Repeatedly move mass from the index with the largest gradient
    to the index with the smallest gradient (full possible epsilon)
    while it reduces the C5 value.
    """
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    n = h.size
    no_improve = 0

    while time.time() - start < time_limit:
        k, val, grad = worst_shift_and_gradient(h)

        # candidate pair (i = max grad, j = min grad)
        order = np.argsort(grad)          # ascending
        i = order[-1]
        j = order[0]
        eps = min(h[i], 1.0 - h[j])

        if eps <= 0:
            # try a few alternative pairs
            found = False
            for ii in order[::-1][:5]:
                for jj in order[:5]:
                    eps2 = min(h[ii], 1.0 - h[jj])
                    if eps2 > 1e-12 and grad[ii] - grad[jj] > 1e-12:
                        i, j, eps = ii, jj, eps2
                        found = True
                        break
                if found:
                    break
            if not found:
                break

        diff = grad[i] - grad[j]
        if diff <= 1e-12:
            break

        # apply full epsilon transfer
        h[i] -= eps
        h[j] += eps
        new_val = compute_c5(h)

        if new_val + 1e-12 < cur_val:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_h = new_val, h.copy()
        else:
            # revert and try a tiny random tweak
            h[i] += eps
            h[j] -= eps
            no_improve += 1
            if no_improve > 200:
                # random small move to escape plateau
                ii = rng.integers(0, n)
                jj = rng.integers(0, n)
                if ii != jj:
                    eps2 = min(h[ii], 1.0 - h[jj])
                    if eps2 > 1e-12:
                        eps2 *= rng.random() * 0.05
                        h[ii] -= eps2
                        h[jj] += eps2
                        cur_val = compute_c5(h)
                        if cur_val < best_val:
                            best_val, best_h = cur_val, h.copy()
                no_improve = 0
    return best_h, best_val


# -------------------------------------------------
# main entry point
# -------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = float(budget_s)

    # start from parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        best_h = None
        best_c = np.inf

    # 1) binary simulated annealing (≈30% time)
    binary_time = total_budget * 0.30
    h_bin, c_bin = binary_sa_n(512, rng, binary_time)
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin

    # 2) greedy continuous descent (majority of remaining time)
    remaining = total_budget - (time.time() - start)
    if remaining > 0.0 and best_h is not None:
        descent_time = remaining * 0.80
        h_des, c_des = greedy_descent(best_h, rng, descent_time)
        if c_des < best_c:
            best_h, best_c = h_des, c_des

    # 3) short random‑restart greedy passes until time runs out
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        now = time.time()
        left = total_budget - (now - start)
        if left < 1.0:
            break
        sub_time = min(4.0, left * 0.5)
        h_tmp = best_h.copy()
        n = h_tmp.size
        # slight random perturbation
        for _ in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            eps = min(h_tmp[i], 1.0 - h_tmp[j])
            if eps > 0:
                eps *= rng.random() * 0.05
                h_tmp[i] -= eps
                h_tmp[j] += eps
        h_tmp, c_tmp = greedy_descent(h_tmp, rng, sub_time)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
