def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    obj = (m + np.log(sum_exp)) / beta
    return obj

def softmax_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    O = corr * dx
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp  # length 2n-1, sum to 1
    # compute cross-correlation w with x
    cross = np.correlate(w, x, mode='full')
    # compute gradient: -dx * (cross[k+(n-1)] + cross[-k+(n-1)])
    n = x.size
    idx = np.arange(- (n-1), n)  # shift index
    grad = -dx * (cross[idx + (n-1)] + cross[-idx + (n-1)])
    return grad

def slsqp_optimize(h0, beta, rng, max_iter):
   global CURRENT_TARGET, CURRENT_BETA
   n = h0.size
   CURRENT_TARGET = n/2
   CURRENT_BETA = beta

   bounds = [(0.0, 1.0)]*n
   constraints = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}

   def obj(x):
       corr = np.correlate(x, 1.0 - x, mode='full')
       bc = CURRENT_BETA * corr
       m = np.max(bc)
       exp_bc = np.exp(bc - m)
       return (m + np.log(np.sum(exp_bc))) / CURRENT_BETA

   def grad(x):
       corr = np.correlate(x, 1.0 - x, mode='full')
       # Compute O's = corr*dx -> but gradient uses w from exp_bc
       bc = CURRENT_BETA * corr
       m = np.max(bc)
       exp_bc = np.exp(bc - m)
       sum_exp = np.exp(m) * np.sum(exp_bc)  # not needed
       w = exp_bc / np.sum(exp_bc)  # length 2n-1
       dx = 2.0 / n
       # cross correlation w with x
       cross = np.correlate(w, x, mode='full')
       nmid = n-1
       # vectorized grad
       idx = np.arange(n)
       grad = -dx * (cross[idx + nmid] + cross[-idx + nmid])
       return grad

   res = minimize(obj, h0, method='SLSQP', jac=grad, bounds=bounds,
                  constraints=constraints,
                  options={'maxiter': max_iter, 'ftol':1e-9, 'disp':False})
   x = np.clip(res.x, 0.0, 1.0)
   # enforce sum again
   diff = CURRENT_TARGET - x.sum()
   if abs(diff) > 1e-12:
       # simple fix
       x = enforce_sum(x, rng)

   return x

def enforce_sum(h, rng):
    n = h.size
    target = n/2
    diff = target - h.sum()
    # tolerance
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase sum; add to indices with h<1
        capacity = 1.0 - h
        total = capacity.sum()
        if total > 0:
            # Proportional addition
            h = h + diff * capacity / total
            # clip in case of rounding
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback random
            for _ in range(1000):
                i = rng.integers(0, n)
                if h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if diff <= 0:
                        break
    else:
        # need to decrease sum
        capacity = h
        total = capacity.sum()

import time

# Global placeholders for the current optimization parameters

CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Return C5 for the discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 0:
            h = h + diff * cap / tot
            h = np.clip(h, 0.0, 1.0)
        else:
            for _ in range(1000):
                i = rng.integers(0, n)
                if h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if diff <= 0:
                        break
    else:
        cap = h
        tot = cap.sum()
        if tot > 0:
            h = h + diff * cap / tot
            h = np.clip(h, 0.0, 1.0)
        else:
            for _ in range(1000):
                i = rng.integers(0, n)
                if h[i] > 0.0:
                    add = max(diff, -h[i])
                    h[i] += add
                    diff -= add
                    if diff >= 0:
                        break
    # final tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(1000):
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
            elif diff < 0 and h[i] > 0.0:
                add = max(diff, -h[i])
            else:
                continue
            h[i] += add
            diff = target - h.sum()
            if abs(diff) <= 1e-12:
                break
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Subgradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence as 0/1 floats."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max corr."""
    global CURRENT_BETA
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def softmax_grad(x: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max objective."""
    global CURRENT_BETA
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / np.sum(exp_bc)          # weights, length 2n-1
    cross = np.correlate(w, x, mode="full")
    mid = n - 1
    idx = np.arange(n)
    grad = -(cross[idx + mid] + cross[-idx + mid])
    return grad


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int) -> np.ndarray:
    """Run SLSQP on the soft‑max objective with analytic gradient."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(softmax_obj, h0, method="SLSQP", jac=softmax_grad,
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    return x


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # initial solution
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        best_h = h.copy()
        best_c = compute_c5(best_h)
    else:
        n = 512
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = enforce_sum(h, rng)
        best_c = compute_c5(best_h)

    n = best_h.size

    # ------------------------------------------------------------------
    # quick subgradient phase (≈20% of budget)
    sub_budget = budget_s * 0.2
    step0 = 0.2
    it = 0
    while time.time() - start < sub_budget:
        step = step0 * (0.99 ** it)
        if step < 1e-13:
            break
        grad = worst_gradient(best_h)
        nh = best_h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, rng)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
        if it % 500 == 0 and rng.random() < 0.4:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps = min(best_h[i], 1.0 - best_h[j]) * 0.05
                if eps > 0:
                    nh2 = best_h.copy()
                    nh2[i] -= eps
                    nh2[j] += eps
                    nh2 = enforce_sum(np.clip(nh2, 0.0, 1.0), rng)
                    c2 = compute_c5(nh2)
                    if c2 < best_c:
                        best_c, best_h = c2, nh2.copy()
        it += 1

    # ------------------------------------------------------------------
    # SLSQP with analytic gradient (≈60% of budget)
    betas = [10, 100, 500, 2000, 8000, 32000, 128000]
    for beta in betas:
        if time.time() - start > budget_s * 0.85:
            break
        remaining = budget_s - (time.time() - start)
        max_iter = min(1500, max(200, int(remaining * 2)))
        # try a few perturbed starts for robustness
        for _ in range(2):
            start_h = best_h.copy()
            # tiny random jitter respecting bounds and sum
            eps = rng.random(n) * 0.02
            start_h += eps - eps.mean() * (2.0 / n)  # keep total roughly constant
            start_h = np.clip(start_h, 0.0, 1.0)
            start_h = enforce_sum(start_h, rng)
            h_opt = slsqp_optimize(start_h, beta, rng, max_iter)
            c = compute_c5(h_opt)
            if c < best_c:
                best_c, best_h = c, h_opt.copy()
                if best_c <= 0.3808:
                    break
        if best_c <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Final annealing using projected gradient on soft‑max (≈20% budget)
    beta = 128000
    steps = 0
    max_steps = 4000
    while steps < max_steps and time.time() - start < budget_s * 0.99:
        # compute gradient of soft‑max at current beta
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        bc = beta * corr
        m = np.max(bc)
        w = np.exp(bc - m)
        w /= w.sum()
        cross = np.correlate(w, best_h, mode="full")
        mid = n - 1
        idx = np.arange(n)
        grad = -(cross[idx + mid] + cross[-idx + mid])
        step = 0.05 * (0.995 ** steps)
        nh = best_h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, rng)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
        steps += 1

    # ------------------------------------------------------------------
    # final sanity check
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
