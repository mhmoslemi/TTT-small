def correct_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # try up to n attempts
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    # if still not exact, fallback to project_box_sum small binary search
    if abs(diff) > 1e-9:
        h = project_box_sum(h, target)
    return h

def grad_descent(h0, betas, iters_per_beta, rng, target_sum, time_budget_start, total_budget):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    global_iter = 0
    start_time = time.time()
    for beta in betas:
        for _ in range(iters_per_beta):
            # check time
            if time.time() - time_budget_start > total_budget * 0.99:
                return best_h, best_c
            # compute cross
            cross = np.correlate(h, h, mode='full')[n-1:n-1+n]
            O = 1.0 - (2.0 / n) * cross
            # softmax weights
            beta_O = beta * O
            m = np.max(beta_O)
            exp_beta_O = np.exp(beta_O - m)
            w = exp_beta_O / exp_beta_O.sum()
            # gradient
            a = np.correlate(w, h, mode='full')[n-1:n-1+n]
            b = np.convolve(h, w, mode='full')[0:n]
            grad = -(2.0 / n) * (a + b)
            # step size
            step = 0.5 / np.sqrt(global_iter + 1)
            # update

            # clip and correct sum
            np.clip(h_new, 0.0, 1.0, out=h_new)
            h_new = correct_sum(h_new, target_sum, rng)

            c = compute_c5(h_new)
            if c < best_c:
                best_c, best_h = c, h_new.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
            h = h_new
            global_iter += 1
    return best_h, best_c

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:

        return h.copy()

    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def upsample_to_n(h, target_n, rng):
    old_n = h.size
    if old_n == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)

import time
import numpy as np

# --------------------------------------------------------------
# basic utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum x = target} using a binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def correct_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Exact sum correction using the binary‑search projector if needed."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    if abs(diff) > 1e-9:
        h = project_box_sum(h, target)
    return h


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    v = np.clip(v, 0.0, 1.0)
    return correct_sum(v, n / 2.0, rng)


def upsample_to_n(h: np.ndarray, target_n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear up‑sampling to a new resolution."""
    old_n = h.size
    if old_n == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return correct_sum(h_new, target_n / 2.0, rng)


# --------------------------------------------------------------
# gradient for the worst shift (used by hill‑climb)
def worst_gradient(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 : n - 1 + n]
    # shift with minimal cross (excluding 0)
    k = int(np.argmin(cross[1:]) + 1)
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# --------------------------------------------------------------
# soft‑max based projected gradient descent
def grad_descent(h0: np.ndarray, betas, iters_per_beta, rng,
                 target_sum, start_time, total_budget):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    global_iter = 0
    for beta in betas:
        for _ in range(iters_per_beta):
            # time guard
            if time.time() - start_time > total_budget * 0.99:
                return best_h, best_c
            # autocorrelation for all shifts
            cross = np.correlate(h, h, mode="full")[n - 1 : n - 1 + n]
            O = 1.0 - (2.0 / n) * cross
            # soft‑max weights
            betaO = beta * O
            m = np.max(betaO)
            exp_term = np.exp(betaO - m)
            w = exp_term / exp_term.sum()
            # gradient of the soft‑max objective
            a = np.correlate(w, h, mode="full")[n - 1 : n - 1 + n]
            b = np.convolve(h, w, mode="full")[0:n]
            grad = -(2.0 / n) * (a + b)
            # step size
            step = 0.5 / np.sqrt(global_iter + 1)
            # projected update
            h_new = h - step * grad
            np.clip(h_new, 0.0, 1.0, out=h_new)
            h_new = project_box_sum(h_new, target_sum)
            # evaluate
            c = compute_c5(h_new)
            if c < best_c:
                best_c, best_h = c, h_new.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
            h = h_new
            global_iter += 1
    return best_h, best_c

    start = time.time()

    # --------------------------------------------------------------
    # Phase 1: soft‑max projected gradient descent
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        # allocate ~70 % of the remaining time to gradient descent
