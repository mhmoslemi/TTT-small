def proj_simplex(v):
    # Project onto {x >=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np
from scipy.optimize import minimize

# -------------------- basic utilities --------------------

def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx, discretized."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    """Project onto [0,1]^n with sum equal to target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    """Subgradient of the worst (max) overlap term."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    """Move mass from highest to lowest gradient coordinate."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# -------------------- smooth (softmax) descent --------------------

def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def softmax_descent(h, beta, n_iters, step0, rng):
    n = h.size
    target = n / 2.0
    for it in range(n_iters):
        grad = softmax_gradient(h, beta)
        norm = np.linalg.norm(grad)
        step = step0 * (1.0 - it / n_iters) / norm if norm > 0 else 0.0
        v = h - step * grad
        h = project_onto_box_sum(v, target)
    return h

# -------------------- aggressive hill climbing --------------------

def aggressive_hill_climb(h0, rng, max_iters=2000, step_scale=1.0):
    """Projected subgradient hill climbing using worst gradient."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.9, 1e-6)
    return best_h, best_c

# -------------------- main entry point --------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512               # number of sample points (must be even)
    target = n // 2       # ∫h = 1  ⇒ sum(h) = n/2

    # ---- initial feasible vector ----
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target)
    else:
        h = np.zeros(n)
        h[rng.choice(n, target, replace=False)] = 1.0

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- smooth (softmax) descent with increasing beta ----
    beta_list = [100, 500, 2000, 8000, 20000]
    for beta in beta_list:
        if time.time() - start > budget_s * 0.45:
            break
        # number of iterations tuned for the budget
        n_iters = int(2500 * 256 / n)  # ≈1250 for n=512
        h = softmax_descent(h, beta, n_iters, step0=0.5, rng=rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- aggressive hill climbing with restarts ----
    chunk_iters = int(2500 * 256 / n)   # ≈1250
    while time.time() - start < budget_s * 0.95:
        # 70% start from current best, 30% from fresh binary
        if rng.random() < 0.7:
            h_start = best_h.copy()
        else:
            h_start = np.zeros(n)
            h_start[rng.choice(n, target, replace=False)] = 1.0
        h_local, c_local = aggressive_hill_climb(h_start, rng,
                                                 max_iters=chunk_iters,
                                                 step_scale=1.0)
        if c_local < best_c:
            best_c, best_h = c_local, h_local.copy()

    # ---- final fine‑tuning with tiny perturbations ----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.01)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # ---- enforce exact feasibility and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
