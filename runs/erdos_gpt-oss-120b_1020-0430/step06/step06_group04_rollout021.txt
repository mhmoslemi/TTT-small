def continuous_sa(h, rng, deadline):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    init_T = max(1e-6, cur_c * 0.015)
    T = init_T
    decay = 0.9999  # per iteration
    max_step = 0.2  # max delta per move

    # Determine approximate time per iteration initially
    # We'll just loop while time left > small threshold
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose delta
        delta = rng.uniform(-max_step, max_step)
        # clamp for bounds
        hi = cur_h[i] + delta
        hj = cur_h[j] - delta
        if hi > 1:
            delta = 1 - cur_h[i]
        elif hi < 0:
            delta = -cur_h[i]
        # after adjusting delta for i, ensure j bound
        if cur_h[j] - delta > 1:
            delta = cur_h[j] - 1
        elif cur_h[j] - delta < 0:
            delta = cur_h[j]
        if abs(delta) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c = compute_c5(new_h)

        # acceptance
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        T *= decay
        # if temperature too low, reset maybe
        if T < 1e-8:
            T = init_T
    return best_h, best_c
```

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h, best_c = h, compute_c5(h)
            for mul in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = h - base_step * mul * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            h = best_h
    return h

def worst_shift_gradient(h):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h, best_c = h, compute_c5(h)
        for mul in (1.0, 0.5, 0.25, 2.0):
            cand = h - step * mul * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand
        h = best_h
        step *= 0.95
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < math.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c

def continuous_sa(h, rng, deadline):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    init_T = max(1e-6, cur_c * 0.015)
    T = init_T
    decay = 0.9999
    max_step = 0.2
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if rng.random() < 0.01:
            if rng.random() < 0.5:
                delta = min(1.0 - cur_h[i], cur_h[j])
            else:
                delta = -min(cur_h[i], 1.0 - cur_h[j])
        else:
            delta = rng.uniform(-max_step, max_step)
            hi = cur_h[i] + delta
            hj = cur_h[j] - delta
            if hi > 1.0:
                delta = 1.0 - cur_h[i]
            elif hi < 0.0:
                delta = -cur_h[i]
            if cur_h[j] - delta > 1.0:
                delta = cur_h[j] - 1.0
            elif cur_h[j] - delta < 0.0:
                delta = cur_h[j]
        if abs(delta) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        T *= decay
        if T < 1e-8:
            T = init_T
    return best_h, best_c
