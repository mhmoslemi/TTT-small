import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via a binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def worst_gradient(h: np.ndarray):
    """Gradient of the worst‑shift overlap."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- Initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # binary approximation: keep top n/2 entries as 1
        idx = np.argsort(-h0)
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx[: n // 2]] = 1.0
    else:
        n = 512
        h_bin = np.zeros(n, dtype=float)
        h_bin[rng.choice(n, size=n // 2, replace=False)] = 1.0

    cur_c = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c = cur_c

    # ----- Phase 1: deterministic hill‑climb -----------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15.0:
        hc_iters = int(min(5000, remaining * 250))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                        max_iters=hc_iters, step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc
        remaining = budget_s - (time.time() - start_time)

    # ----- Phase 2: Simulated Annealing -----------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30.0:
        sa_time = min(300.0, remaining * 0.45)
        # estimate sigma from a few random swaps
        sig_vals = []
        temp_h = best_h.copy()
        temp_c = best_c
        for _ in range(200):
            i = rng.integers(0, n)
            while temp_h[i] == 0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while temp_h[j] == 1:
                j = rng.integers(0, n)
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            new_c = compute_c5(temp_h)
            sig_vals.append(new_c - temp_c)
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
        sigma = np.std(sig_vals) if sig_vals else 0.0
        T = max(sigma, 1e-4)
        alpha = 0.9996

        cur_h = best_h.copy()
        cur_c = best_c
        end_sa = time.time() + sa_time - 0.5
        while time.time() < end_sa:
            i = rng.integers(0, n)
            while cur_h[i] == 0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while cur_h[j] == 1:
                j = rng.integers(0, n)
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_c = compute_c5(cur_h)
            delta = new_c - cur_c
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / (T + 1e-12)):
                    accept = True
            if accept:
                cur_c = new_c
                if cur_c < best_c:
                    best_c, best_h = cur_c, cur_h.copy()
                    if best_c <= 0.3808:
                        break
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            T *= alpha
        remaining = budget_s - (time.time() - start_time)

    # ----- Phase 3: projected sub‑gradient descent ------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        h_sg, c_sg = subgradient_optimize(best_h, n / 2.0,
                                          remaining * 0.70, rng)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # ----- Phase 4: final guided hill‑climb ------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        max_iters = int(min(4000, remaining * 250))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                        max_iters=max_iters, step_scale=0.6)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ----- Sanitise and return --------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
