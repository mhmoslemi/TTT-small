def worst_gradient(h):
    n = h.size
    cross = np.correlate(h, h, mode='full')[n-1:]  # length n, cross[k] = Σ_i h_i h_{i+k} for k >=0
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[:n-k] -= h[k:]
        grad[k:] -= h[:n-k]
    return grad, k

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()  # nothing
    # compute max epsilon that maintains bounds
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):

    return best_h, best_c5

def project_box_sum(v, target):
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    return proj

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function on [0,2] (discrete)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector projected onto the feasible set."""
    return project_box_sum(rng.random(n), n / 2.0)


def upsample(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Linear up‑sampling by factor 2 with re‑projection."""
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n * 2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, (n * 2) / 2.0)


def upsample_to_n(h: np.ndarray, n_target: int, rng: np.random.Generator) -> np.ndarray:
    """Repeated doubling until length n_target is reached."""
    while h.size < n_target:
        h = upsample(h, rng)
    return h


# ------------------------------------------------------------
def optimize_from_h(h0: np.ndarray, rng: np.random.Generator,
                    time_budget: float, max_steps: int):
    """RMSprop on the smooth‑max approximation of the max‑overlap."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros_like(h)
    start = time.time()
    beta_start, beta_end = 50.0, 5000.0
    for t in range(1, max_steps + 1):
        if time.time() - start > time_budget:
            break
        beta = beta_start + (beta_end - beta_start) * (t - 1) / max(max_steps - 1, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = corr.max()
        # soft‑max weights
        exps = np.exp(beta * (corr - max_corr))
        w = exps / exps.sum()
        # gradient = Σ_k w_k * (1-h) shifted by k
        A = 1 - h
        conv = np.convolve(A, w[::-1], mode="full")
        grad = conv[n - 1:n - 1 + n]
        # RMSprop update
        v = 0.9 * v + 0.1 * (grad * grad)
        step = 0.2 * grad / (np.sqrt(v) + 1e-8)
        h = project_box_sum(h - step, target)
        # current C5 (no extra corr needed)
        c5 = max_corr * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            if best_c5 <= 0.3808:
                break
    return best_h, best_c5


# ------------------------------------------------------------
def worst_gradient(h: np.ndarray):
    """Gradient of the worst‑shift overlap (max O_k)."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1:]  # Σ h_i h_{i+k}
    k = int(np.argmin(cross))                     # shift giving max O_k
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5):
    """Transfer ε from a high‑grad index to a low‑grad index."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Simple hill‑climb using the worst‑shift gradient."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c5 = compute_c5(nh)
        if c5 < best_c5:
            best_c5, best_h, h = c5, nh.copy(), nh
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c5
