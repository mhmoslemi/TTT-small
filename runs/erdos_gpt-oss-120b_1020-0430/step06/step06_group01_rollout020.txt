def project_to_sum_bounds(h, target):
   diff = target - h.sum()
   if abs(diff) < 1e-12:
       return h
   # Copy to avoid modify original? Already have h_new.
   # Use water-filling
   free = np.where(((diff > 0) & (h < 1)) | ((diff < 0) & (h > 0)))[0]
   while abs(diff) > 1e-12 and free.size > 0:
       inc = diff / free.size
       # compute possible adjustments limited by bounds
       if inc > 0:
           # max increase limited by 1-h
           cap = 1 - h[free]
           actual = np.minimum(inc, cap)
       else:
           cap = -h[free]
           actual = np.maximum(inc, cap)  # inc negative
       h[free] += actual
       diff = target - h.sum()
       # update free
       free = np.where(((diff > 0) & (h < 1 - 1e-12)) | ((diff < 0) & (h > 1e-12)))[0]
   # At the end, if diff remains due to floating error, adjust first free index

# ------------------- helpers ------------------- #

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function on [0,2] (full linear correlation)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def project_to_sum_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and modify to attain exact sum = target."""
    # Small tolerance for floating errors
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # At most 2 passes per element (each pass saturates at least one index)
    for _ in range(2 * n):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            delta = np.minimum(inc, 1.0 - h[free])
            h[free] += delta
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            delta = np.maximum(inc, -h[free])
            h[free] += delta
        diff = target - h.sum()
    # Final tiny correction (should be negligible)
    if abs(diff) > 1e-12:
        i = 0
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def worst_shift_and_grad(h: np.ndarray, total_sum: float):
    """
    Return (k, max_overlap, gradient) where k is the shift (can be negative)
    attaining the maximal overlap O_k = Σ h[i]·(1‑h[i+k]) (linear,
    zero‑padded outside), max_overlap is that value, and gradient is
    dO_k/dh (the subgradient for the worst shift).
    """
    n = h.size
    L = 2 * n
    # linear correlation via FFT (zero‑padded)
    fft_h = np.fft.rfft(h, n=L)
    fft_rev = np.fft.rfft(h[::-1], n=L)
    corr = np.fft.irfft(fft_h * fft_rev, n=L)[: 2 * n - 1]  # length 2n‑1
    # overlap values O_k = total_sum - Σ h[i]·h[i+k]
    O = total_sum - corr
    idx = int(np.argmax(O))
    max_val = O[idx]
    k = idx - (n - 1)  # shift can be negative
    # gradient for this shift
    grad = np.ones_like(h)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        s = -k
        grad[: n - s] -= h[s:]
        grad[s:] -= h[: n - s]
    return k, max_val, grad


# ------------------- main optimisation ------------------- #

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # final buffer for sanitisation
    buffer = 4.0
    # target size – try a finer discretisation if time permits
    n_target = 1024
    target_sum = n_target / 2.0

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # interpolate to the target size
        x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(x_new, x_old, h0)
    else:
        h = rng.random(n_target)

    np.clip(h, 0.0, 1.0, out=h)
    h = project_to_sum_bounds(h, target_sum)

    # best solution tracker
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- core descent loop -------------------------------------------
    iteration = 0
    while time.time() - start < budget_s - buffer:
        # compute worst shift, its value and gradient
        k, max_val, grad = worst_shift_and_grad(h, target_sum)

        # update best if improved
        if max_val < best_c:
            best_c = max_val
            best_h = h.copy()

        # step size schedule – linear decay to a small floor
        elapsed = time.time() - start
        frac = elapsed / (budget_s - buffer)
        step0 = 0.5
        min_step = 1e-4
        step = max(step0 * (1.0 - frac), min_step)

        # gradient descent step (move opposite to gradient)
        h_new = h - step * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = project_to_sum_bounds(h_new, target_sum)
        h = h_new

        iteration += 1

        # occasional random jitter to escape flat regions
        if iteration % 200000 == 0:
            jitter = (rng.random(n_target) - 0.5) * 0.02
            h = np.clip(h + jitter, 0.0, 1.0)
            h = project_to_sum_bounds(h, target_sum)

    # ----- final sanitisation -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_bounds(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
