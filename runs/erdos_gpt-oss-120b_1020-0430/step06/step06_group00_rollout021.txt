def continuous_sa(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    # parameters:
    T0 = 0.2  # maybe
    T_end = 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)  # exponential cooling
        # choose two distinct indices i, j
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        # propose move: shift epsilon from i to j or reverse with equal prob
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
        else:
            max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * (T / T0)  # scale with temperature
        if eps <= 0:
            continue
        h_new = h.copy()
        if rng.random() < 0.5:
            # move from i to j
            h_new[i] -= eps
            h_new[j] += eps
        else:
            # move from j to i
            h_new[i] += eps
            h_new[j] -= eps
        # ensure within bounds due to eps limited
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad
```

def binary_hill_climb(N, rng, time_budget):
    start = time.time()
    best_h = np.zeros(N, dtype=float)
    best_c = np.inf
    # Pre-allocate arrays for ones/zeros updates
    while time.time() - start < time_budget:
        # random initialization
        h = np.zeros(N, dtype=float)
        ones = rng.choice(N, N//2, replace=False)
        h[ones] = 1.0
        zeros = np.setdiff1d(np.arange(N), ones, assume_unique=True)
        # convert to list for fast element assignment
        ones = list(ones)
        zeros = list(zeros)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # hill climbing loop
        # set a limit for attempts without improvement to avoid infinite loops
        no_improve = 0
        max_no_improve = 2000
        while time.time() - start < time_budget and no_improve < max_no_improve:
            i_pos = rng.integers(len(ones))
            j_pos = rng.integers(len(zeros))
            i = ones[i_pos]
            j = zeros[j_pos]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h)
            if new_c < c - 1e-12:
                c = new_c
                # update lists
                ones[i_pos] = j
                zeros[j_pos] = i
                no_improve = 0
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        # go to next restart
    return best_h, best_c
```

def run(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # Phase 0: binary hill climb at N=256
    t_remaining = total_budget - (time.time() - start)
    if t_remaining <= 0:
        return [], 0.0, 0

    t_binary = min(t_remaining * 0.55, 700)  # leaving buffer
    best_h_bin, best_c_bin = binary_hill_climb(256, rng, t_binary)

    # Phase 1: upsample to 512 and enforce sum
    t_remaining = total_budget - (time.time() - start)
    if t_remaining <= 0:
        best_h = best_h_bin
        best_c = best_c_bin
    else:
        h_up = upsample_h(best_h_bin, 512)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        best_h = h_up
        best_c = compute_c5(best_h)

        # Phase 2: continuous SA
        t_remaining = total_budget - (time.time() - start)
        if t_remaining > 0:
            t_cont = t_remaining * 0.85
            h_cont, c_cont = continuous_sa(best_h, rng, t_cont)
            if c_cont < best_c:
                best_h, best_c = h_cont, c_cont

        # Phase 3: gradient refinement
        t_remaining = total_budget - (time.time() - start)
        if t_remaining > 0:
            h_ref = gradient_refine(best_h, rng, step=0.015, max_iter=20)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:

    # Final sanitization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

def continuous_sa(h0, rng, time_budget):
    start = time.time()

    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0 = 0.05
    T_end = 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # direction: 0 means i->j, 1 means j->i
        direction = rng.integers(2)
        if direction == 0:
            max_eps = min(h[i], 1.0 - h[j])
        else:

        if max_eps <= 0:

            h_new[i] -= eps

            c_curr = c_new

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > 1e-12 and it < 200000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        it += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    idx = np.argmax(np.correlate(h, 1.0 - h, mode="full")) - (n - 1)
    k = abs(idx)
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
    return g


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        g = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                else:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                diff = n / 2.0 - h.sum()
    return h


def binary_hill_climb(N: int, rng: np.random.Generator, time_budget: float):
    """Greedy hill climbing on binary vectors with exactly N/2 ones."""
    start = time.time()
    best_h = np.zeros(N, dtype=float)
    best_c = np.inf
    max_no_improve = 2000
    while time.time() - start < time_budget:
        h = np.zeros(N, dtype=float)
        ones = rng.choice(N, N // 2, replace=False)
        h[ones] = 1.0
        zeros = np.setdiff1d(np.arange(N), ones)
        ones = list(ones)
        zeros = list(zeros)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        no_imp = 0
        while time.time() - start < time_budget and no_imp < max_no_improve:
            i_pos = rng.integers(len(ones))
            j_pos = rng.integers(len(zeros))
            i = ones[i_pos]
            j = zeros[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h)
            if new_c < c - 1e-12:
                c = new_c
                ones[i_pos] = j
                zeros[j_pos] = i
                no_imp = 0
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_imp += 1
    return best_h, best_c


def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated annealing on the continuous feasible set."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.05, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        direction = rng.integers(2)  # 0: i->j, 1: j->i
        if direction == 0:
            max_eps = min(h[i], 1.0 - h[j])
        else:
            max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * (T / T0)
        if eps <= 0:
            continue
        h_new = h.copy()
        if direction == 0:
            h_new[i] -= eps
            h_new[j] += eps
        else:
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

        h_bin, c_bin = binary_hill_climb(256, rng, t_bin)

        # Upsample to 512 and enforce sum
        h_up = upsample_h(h_bin, 512)
        h_up = np.clip(h_up, 0.0, 1.0)

        c_up = compute_c5(h_up)
        if c_up < best_c:

        # Phase 1: continuous simulated annealing
