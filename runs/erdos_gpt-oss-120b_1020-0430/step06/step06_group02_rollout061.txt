def project_onto_feasible(x, target, max_iter=50, tol=1e-9):
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0

    i = rng.integers(0, n)

    h_new[i] -= eps

def project_onto_feasible(x, target, tol=1e-9):
    lo, hi = -1.0, 1.0 # ensures solution in range
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # correction due to numeric errors:
    diff = target - y.sum()
    if abs(diff) > 1e-8:
        # distribute diff over non-bound entries
        mask = (y > 0) & (y < 1)
        cnt = mask.sum()
        if cnt > 0:
            y[mask] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y
```

import time
import numpy as np
from scipy.optimize import minimize

# --------------------- core utilities ---------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_feasible(x: np.ndarray, target: float, tol: float = 1e-9) -> np.ndarray:
    """Euclidean projection of x onto {0≤y≤1, sum(y)=target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-8:
        mask = (y > 0.0) & (y < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            y[mask] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y


def subgrad_and_corr(h: np.ndarray):
    """Return subgradient of the worst overlap and its raw (unscaled) value."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    o_raw = corr[idx]
    shift = idx - (n - 1)
    k = abs(shift)
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, o_raw


def random_transfer(h: np.ndarray, rng: np.random.Generator, max_frac: float = 0.05) -> np.ndarray:
    """Move a random amount of mass from one index to another (keeps sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    eps = min(h[i], 1.0 - h[j]) * rng.uniform(0.0, max_frac)
    if eps <= 0.0:
        return h
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 1000) -> np.ndarray:
    """Refine a candidate using SLSQP on a smoothed max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # repair exact sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            for _ in range(2000):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
                if add != 0.0:
                    x[i] += add
                    diff = CURRENT_TARGET - x.sum()
                    if abs(diff) <= 1e-12:
                        break
    return x
