```python
def enforce_sum_simple(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Identify free positions
    if diff > 0:
        free = h < 1 - 1e-12
        if not free.any():
            return h
        inc = np.minimum(1 - h[free], diff * (1 - h[free]) / np.sum(1 - h[free]))
        h[free] += inc
        # If still diff not zero due to rounding, adjust with small loop
    else: # diff < 0
        free = h > 1e-12
        if not free.any():
            return h
        dec = np.minimum(h[free], -diff * h[free] / np.sum(h[free]))
        h[free] -= dec
    # Recurse if not exact
    return enforce_sum_simple(h, target) if abs(target - h.sum()) > 1e-12 else h
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple random adjustments; converges fast when |diff| is modest
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            cap = 1.0 - h[i]
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            cap = -h[i]
        delta = np.sign(diff) * min(abs(diff), abs(cap))
        h[i] += delta
        diff = target - h.sum()
    return h

def random_perturb_small(h: np.ndarray, rng: np.random.Generator, scale: float = 0.05) -> np.ndarray:
    """Small mass transfer preserving the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

def local_optimize(h_init: np.ndarray, target: float, rng: np.random.Generator,
                   time_end: float, best_global_c: float) -> tuple:
    """Projected gradient descent on the soft‑max of the overlap."""
    n = h_init.size
    h = h_init.copy()
    c_cur = compute_c5(h)

    beta_start = 5.0
    beta_max = 20000.0
    max_iters = 800
    # geometric increase of beta so that beta_max is reached at the last iteration
    beta_factor = np.exp(np.log(beta_max / beta_start) / max_iters)
    beta = beta_start
    base_step = 0.02

    best_h_local = h.copy()
    best_c_local = c_cur

    for it in range(max_iters):
        if time.time() >= time_end:
            break

        # soft‑max probabilities of the overlaps
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        exp_shifted = np.exp(beta * (corr - max_corr))
        p = exp_shifted / exp_shifted.sum()   # Σ p = 1

        # forward and backward weighted sums
        forward = np.convolve(p[::-1], h, mode="full")[n - 1:n - 1 + n]
        backward = np.convolve(p, h, mode="full")[n - 1:n - 1 + n]

        grad = 1.0 - forward - backward          # ∇ of soft‑max objective
        step = base_step / np.sqrt(beta)

        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)

        c_new = compute_c5(h_new)

        # accept if improvement (or tiny random acceptance to escape plateaus)
        if c_new <= c_cur or rng.random() < 0.001:
            h = h_new
            c_cur = c_new
            if c_new < best_c_local:
                best_c_local = c_new
                best_h_local = h_new.copy()
            if c_new < best_global_c:
                best_global_c = c_new

        # early stop if we already beat the target bound
        if best_global_c < 0.38079:
            break

        beta = min(beta_max, beta * beta_factor)

    return best_h_local, best_c_local, best_global_c

        if best_c < 0.38079:
            break

        # generate next starting point: small perturbation of the best found
        h_current = random_perturb_small(best_h, rng, scale=0.07)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
