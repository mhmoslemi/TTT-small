def simple_sa(h, rng, time_budget):
    best = h.copy()
    best_c = compute_c5(best)
    curr = h.copy()
    curr_c = best_c
    N = h.size
    start = time.time()
    while time.time() - start < time_budget:
        # choose two indices i,j
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i: j += 1
        # compute max epsilon that maintains bounds
        # choose direction random
        if rng.random() < 0.5:
            max_eps = min(curr[i], 1.0 - curr[j])
            if max_eps <= 0: continue
            eps = max_eps * rng.random() * 0.2  # scaled
            new = curr.copy()
            new[i] -= eps
            new[j] += eps
        else:
            max_eps = min(curr[j], 1.0 - curr[i])
            if max_eps <= 0: continue
            eps = max_eps * rng.random() * 0.2
            new = curr.copy()
            new[i] += eps
            new[j] -= eps
        new_c = compute_c5(new)
        # temperature schedule
        elapsed = time.time() - start
        T = 0.05 * (1e-5 / 0.05) ** (elapsed / time_budget)
        if new_c < curr_c or rng.random() < np.exp(-(new_c - curr_c) / max(T, 1e-12)):
            curr, curr_c = new, new_c
            if new_c < best_c:
                best, best_c = new.copy(), new_c
    return best, best_c

def enforce_sum_target(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # attempt proportional adjustment
    if diff > 0:
        capacity = 1.0 - h
        total = capacity.sum()
        if total > 0:
            add = diff * capacity / total
            h = h + add
    else:
        deficit = h
        total = deficit.sum()
        if total > 0:
            sub = -diff * deficit / total
            h = h - sub
    # clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # fallback simple loop
        for _ in range(10000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(h.size)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
    return h
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def enforce_sum_target(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        total = capacity.sum()
        if total > 0:
            h = h + diff * capacity / total
    else:
        deficit = h
        total = deficit.sum()
        if total > 0:
            h = h + diff * deficit / total   # diff is negative
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # fallback simple redistribution
        for _ in range(10000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(h.size)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Iterative refinement using the gradient of the worst‑shift overlap."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_target(h, target, rng)
    return h


def simple_sa(h: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simple simulated annealing on the continuous feasible set."""
    best = h.copy()
    best_c = compute_c5(best)
    curr = h.copy()
    curr_c = best_c
    N = h.size
    start = time.time()
    while time.time() - start < time_budget:
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        # try moving mass from i to j
        max_eps = min(curr[i], 1.0 - curr[j])
        if max_eps > 0:
            eps = max_eps * rng.random() * 0.2
            new = curr.copy()
            new[i] -= eps
            new[j] += eps
        else:
            # opposite direction
            max_eps = min(curr[j], 1.0 - curr[i])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.2
            new = curr.copy()
            new[i] += eps
            new[j] -= eps
        new = np.clip(new, 0.0, 1.0)
        c_new = compute_c5(new)
        elapsed = time.time() - start
        T0, T_end = 0.05, 1e-5
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        if c_new < curr_c or rng.random() < np.exp(-(c_new - curr_c) / max(T, 1e-12)):
            curr, curr_c = new, c_new
            if c_new < best_c:
                best, best_c = new.copy(), c_new
    return best, best_c

    start = time.time()
    total_budget = budget_s

    # ---- parameters ----
    N = 1024                     # grid size
    target = N / 2.0
    P = 40                       # population size
    pc = 0.9                     # crossover probability
    pm = 0.1                     # mutation probability
    max_step = 0.05              # max epsilon in mutation

    # ---- initial population ----
    pop_h = []
    pop_c = np.empty(P)
    have_initial = 'initial_h_values' in globals()

    if have_initial:

        if init_h.size != N:

        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = enforce_sum_target(init_h, target, rng)

        pop_c[0] = compute_c5(init_h)
        start_idx = 1

        ones = rng.choice(N, N // 2, replace=False)
        h[ones] = 1.0
        h = enforce_sum_target(h, target, rng)
        pop_h.append(h)

    idx = np.argsort(pop_c)

    best_c = pop_c[0]

    gen = 0

    while True:

        for _ in range(P):
            # tournament selection for parent 1

            p1_idx = a if pop_c[a] <= pop_c[b] else b
            # tournament selection for parent 2

            p2_idx = a if pop_c[a] <= pop_c[b] else b
            p1 = pop_h[p1_idx]
            p2 = pop_h[p2_idx]

            # crossover
            if rng.random() < pc:
                cut = rng.integers(1, N - 1)
                child = np.concatenate((p1[:cut], p2[cut:]))

                child = p1.copy()

            if rng.random() < pm:

                j_idx = rng.integers(N - 1)

                    child[i_idx] -= eps
                    child[j_idx] += eps
                else:

                    if max_eps > 0:
                        eps = max_eps * rng.random() * max_step
                        child[i_idx] += eps
                        child[j_idx] -= eps

            new_c.append(compute_c5(child))
