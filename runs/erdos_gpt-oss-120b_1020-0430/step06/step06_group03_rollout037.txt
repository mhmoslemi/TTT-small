import time
import numpy as np

def compute_c5(h):
    """Return C₅ for a step function h on [0,2] (discretized)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def project_onto_box_sum(v, target):
    """
    Project vector v onto the box [0,1]^n while forcing the sum to be `target`.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny fix for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def grad_for_shift(h, k):
    """
    Exact gradient of the overlap term for shift k:
        O(k) = sum_{i} h[i] * (1 - h[i+k])   (indices out of range are ignored)
    """
    n = h.size
    if k > 0:
        grad = np.zeros_like(h)
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    elif k < 0:
        p = -k
        grad = np.zeros_like(h)
        grad[p:] = 1.0 - h[: n - p]
        grad[: n - p] -= h[p:]
    else:
        grad = 1.0 - 2.0 * h
    return grad


def softmax_gradient(h, beta):
    """Gradient of the smooth approximation of max_k O(k)."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    # central term (k=0)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    # Initialise h
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
    else:
        n = 512
        h = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    target = n / 2.0
    h = project_onto_box_sum(h, target)

    dx = 2.0 / n

    # best solution seen so far
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # current solution (used in the greedy phase)
    cur_h = h.copy()
    corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_c = corr.max() * dx

    no_improve = 0
    # ------------------------------------------------------------------ #
    # Phase 1: greedy transport / hill climbing (≈78% of time)
    while time.time() - start < budget_s * 0.78:
        # locate worst shift
        s = int(np.argmax(corr))
        k = s - (n - 1)

        # gradient for that shift
        grad = grad_for_shift(cur_h, k)

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        diff = grad[i_max] - grad[i_min]
        if diff < 1e-12:
            no_improve += 1
            # tiny random nudge to escape flat region
            if no_improve % 250 == 0:
                i, j = rng.integers(n, size=2)
                eps = min(cur_h[i], 1.0 - cur_h[j]) * 0.01
                if eps > 0:
                    cur_h[i] -= eps
                    cur_h[j] += eps
                    corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                    cur_c = corr.max() * dx
            continue

        eps_max = min(cur_h[i_max], 1.0 - cur_h[i_min])
        if eps_max < 1e-12:
            no_improve += 1
            continue

        # step size that shrinks as we consume the time budget
        elapsed = time.time() - start
        fraction = max(0.0, (budget_s - elapsed) / budget_s)
        step_scale = max(1e-6, fraction * 0.5)
        eps = eps_max * step_scale

        # propose move
        new_h = cur_h.copy()
        new_h[i_max] -= eps
        new_h[i_min] += eps

        new_corr = np.correlate(new_h, 1.0 - new_h, mode="full")
        new_c = new_corr.max() * dx

        # update global best
        if new_c < best_c - 1e-12:
            best_c = new_c
            best_h = new_h.copy()
            no_improve = 0

        # accept improvement locally
        if new_c < cur_c - 1e-12:
            cur_h = new_h
            corr = new_corr
            cur_c = new_c
            no_improve = 0
        else:
            no_improve += 1

        # occasional restart when stuck
        if no_improve > 5000:
            cur_h = best_h.copy()
            i, j = rng.integers(n, size=2)
            eps = min(cur_h[i], 1.0 - cur_h[j]) * 0.01
            if eps > 0:
                cur_h[i] -= eps
                cur_h[j] += eps
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            cur_c = corr.max() * dx
            no_improve = 0

    # ------------------------------------------------------------------ #
    # Phase 2: soft‑max (smooth max) gradient descent to polish (≈95% time)
    while time.time() - start < budget_s * 0.95:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        # increase beta as we get closer to the deadline
        if remaining > budget_s * 0.25:
            beta = 2000
        elif remaining > budget_s * 0.05:
            beta = 8000
        else:
            beta = 20000

        grad = softmax_gradient(best_h, beta)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        # step size decays with remaining time
        step = max(1e-8, (remaining / budget_s) * 0.3) / norm
        v = best_h - step * grad
        best_h = project_onto_box_sum(v, target)
        cand_c = compute_c5(best_h)
        if cand_c < best_c:
            best_c = cand_c

    # ------------------------------------------------------------------ #
    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
