def subgradient_descent(h_init, rng, time_budget):
    start = time.time()
    best_h = h_init.copy()
    best_c = compute_c5(best_h)
    h = h_init.copy()
    n = h.size
    it = 0
    # parameters
    step0 = 0.05 # may be tuned
    while time.time() - start < time_budget:
        it += 1
        g = worst_gradient(h)
        # step schedule: alpha = step0 / sqrt(it)
        alpha = step0 / np.sqrt(it)
        h_new = h - alpha * g
        # clip
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # sum correction
        diff = n/2 - h_new.sum()
        # distribute diff across random indices, up to 10 steps to correct
        if abs(diff) > 1e-12:
            # we will adjust gradually
            for _ in range(20):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[i])
                else:
                    add = max(diff, -h_new[i])
                h_new[i] += add
                diff = n/2 - h_new.sum()
        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            # if reached target, break early
            if best_c <= 0.3808:
                break
    return best_h, best_c

import time
import numpy as np

# --------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # vectorized proportional adjustment
        if diff > 0:
            cap = 1.0 - h
        else:
            cap = -h
        cap_sum = cap.sum()
        if cap_sum > 0:
            factor = min(1.0, diff / cap_sum) if diff > 0 else max(-1.0, diff / cap_sum)
            h += cap * factor
    # residual random fix
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst‑shift overlap (max_k)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad

# Local optimisation
def subgradient_descent(h0: np.ndarray, rng: np.random.Generator,
                        time_budget: float):
    """Projected sub‑gradient descent on f(h)=max_k L_k(h)."""
    start = time.time()
    h = h0.copy()
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    step = 0.1
    while time.time() - start < time_budget:
        g = worst_gradient(h)
        h_new = h - step * g
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < c:
            h, c = h_new, c_new
            step = min(step * 1.1, 0.2)
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
                if best_c <= 0.3808:
                    break
        else:
            step *= 0.5
            if step < 1e-5:
                step = 0.1
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 20) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, rng)
    return h


def binary_round_best(h: np.ndarray, rng: np.random.Generator,
                      trials: int = 10):
    """Round to binary vectors (exactly n/2 ones) and keep the best C5."""
    n = h.size
    best_hb, best_cb = None, np.inf
    for _ in range(trials):
        noise = rng.random(n) * 1e-12
        idx = np.argsort(-(h + noise))
        hb = np.zeros(n, dtype=float)
        hb[idx[: n // 2]] = 1.0
        cb = compute_c5(hb)
        if cb < best_cb:
            best_cb, best_hb = cb, hb.copy()
    return best_hb, best_cb
