def compute_c5(h):
    n = h.size
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1 - h, L)
    conv = np.fft.ifft(H * np.conj(G)).real
    max_val = conv.max()
    return max_val * (2.0 / n)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Fast C5 using FFT‑based cross‑correlation."""
    n = h.size
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real
    return corr.max() * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto sum(h)=n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        i = 0
        h[i] = np.clip(h[i] + residual, 0.0, 1.0)
    return h


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the current worst overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- SLSQP helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    H = np.fft.fft(x, 2 * x.size)
    G = np.fft.fft(1.0 - x, 2 * x.size)
    corr = np.fft.ifft(H * np.conj(G)).real
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if add != 0.0:
                x[i] += add
                diff = CURRENT_TARGET - x.sum()
                if abs(diff) <= 1e-12:
                    break
    return x


# ---------- main entry ----------

    target_c = 0.3808

            else:

    else:

    best_h = h_bin.copy()

    # ----- binary local search (targeted swaps) -----
    time_binary = budget_s * 0.65
    L = 2 * n
    while time.time() - start < time_binary:
        # correlation & worst shift
        H = np.fft.fft(best_h, L)
        G = np.fft.fft(1.0 - best_h, L)
        corr = np.fft.ifft(H * np.conj(G)).real
        max_idx = corr.argmax()
        k = max_idx if max_idx < n else max_idx - L

        # find i (1->0) and j (0->1) that directly affect this shift
        if k >= 0:
            rng_i = np.arange(n - k)
            mask_i = (best_h[rng_i] == 1) & (best_h[rng_i + k] == 0)
            mask_j = (best_h[rng_i] == 0) & (best_h[rng_i + k] == 1)
        else:
            ka = -k
            rng_i = np.arange(ka, n)
            mask_i = (best_h[rng_i] == 1) & (best_h[rng_i - ka] == 0)
            mask_j = (best_h[rng_i] == 0) & (best_h[rng_i - ka] == 1)

        i_cand = rng_i[mask_i]
        j_cand = rng_i[mask_j]
        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            # random balanced swap
            i = rng.choice(np.flatnonzero(best_h))
            j = rng.choice(np.flatnonzero(1 - best_h))
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_h, best_c = h_new, c_new
            if best_c <= target_c:
                break
        # occasional random perturbation
        if rng.random() < 0.01:
            flip = max(1, n // 100)
            ones_idx = np.flatnonzero(best_h)
            zeros_idx = np.flatnonzero(1 - best_h)
            if ones_idx.size >= flip and zeros_idx.size >= flip:
                i_sel = rng.choice(ones_idx, size=flip, replace=False)
                j_sel = rng.choice(zeros_idx, size=flip, replace=False)
                h_tmp = best_h.copy()
                h_tmp[i_sel] = 0.0
                h_tmp[j_sel] = 1.0
                c_tmp = compute_c5(h_tmp)
                if c_tmp < best_c:
                    best_h, best_c = h_tmp, c_tmp
                    if best_c <= target_c:
                        break

    # ----- continuous refinement with SLSQP -----
    time_slsqp = budget_s * 0.30
    h_cont = best_h.astype(float)
    beta_vals = (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0)
    for beta in beta_vals:
        if time.time() - start > time_binary + time_slsqp:
            break
        h_opt = slsqp_optimize(h_cont, beta, rng, max_iter=1500)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            if best_c <= target_c:
                break
        h_cont = h_opt  # warm‑start next beta

    # ----- fine‑grained subgradient sweeps -----
    step_vals = (0.05, 0.02, 0.01, 0.005, 0.002)
    for step in step_vals:
        if time.time() - start > budget_s * 0.98:
            break
        for _ in range(300):
            grad = worst_gradient(best_h)
            h_new = best_h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
                if best_c <= target_c:

        if best_c <= target_c:

    # ----- final sanity check -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
