def project_bounds_sum(x, target):
    y = np.clip(x, 0.0, 1.0)
    diff = target - y.sum()
    # water-filling
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = y < 1.0 - 1e-12
        else:
            mask = y > 1e-12
        if not np.any(mask):
            break
        # average needed per eligible
        cnt = mask.sum()
        avg = diff / cnt
        if diff > 0:
            # max we can add to each is (1 - y)
            max_add = (1.0 - y[mask]).min()
            add = min(avg, max_add)
        else:
            max_sub = y[mask].min()
            add = max(avg, -max_sub)
        y[mask] += add
        diff = target - y.sum()
    # final clipping due to numeric errors
    y = np.clip(y, 0.0, 1.0)
    # adjust any residual diff by adding to first element not at bound
    if abs(diff) > 1e-10:
        # find an index where we can adjust
        if diff > 0:
            idx = np.where(y < 1.0 - 1e-10)[0]
        else:
            idx = np.where(y > 1e-10)[0]
        if idx.size > 0:
            i = idx[0]
            y[i] = min(max(y[i] + diff, 0.0), 1.0)
    return y

    else:

    h = project_bounds_sum(h0, target)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- global variables for SLSQP -------------------
CURRENT_TARGET = None
CURRENT_BETA = None

# ------------------- utility functions -------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = target while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def project_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the intersection of [0,1]^n and the hyperplane sum = target."""
    y = np.clip(x, 0.0, 1.0)
    diff = target - y.sum()
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = y < 1.0 - 1e-12
        else:
            mask = y > 1e-12
        if not np.any(mask):
            break
        cnt = mask.sum()
        avg = diff / cnt
        if diff > 0:
            max_add = (1.0 - y[mask]).min()
            add = min(avg, max_add)
        else:
            max_sub = y[mask].min()
            add = max(avg, -max_sub)
        y[mask] += add
        diff = target - y.sum()
    # tiny residual fix
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(y < 1.0 - 1e-10)[0]
        else:
            idx = np.where(y > 1e-10)[0]
        if idx.size:
            i = idx[0]
            y[i] = min(max(y[i] + diff, 0.0), 1.0)
    return y


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_descent(h0: np.ndarray, target: float, time_limit: float,
                     rng: np.random.Generator) -> tuple:
    """Run projected gradient descent until time_limit, return best h and its C5."""
    n = h0.size
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c = corr.max() * (2.0 / n)
    best_h = h.copy()
    step0 = 0.5
    step_decay = 0.9998
    it = 0
    jitter_every = 3000
    while time.time() < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        c = corr.max() * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # gradient of the worst shift
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        step = step0 * (step_decay ** it)
        h_new = h - step * g
        h_new = project_bounds_sum(h_new, target)
        if it % jitter_every == 0 and it > 0:
            noise = rng.normal(scale=0.02, size=n)
            h_new = h_new + noise
            h_new = project_bounds_sum(h_new, target)
        h = h_new
        it += 1
    return best_h, best_c


# ------------------- main entry point -------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- problem size -----
    n = 1024
    target = n // 2

    # ----- initial construction -----
    if "initial_h_values" in globals():
        h0 = np.repeat(np.array(initial_h_values, dtype=float), 2)
    else:
        h0 = np.zeros(n)
        h0[rng.choice(n, target, replace=False)] = 1.0
    # make it slightly continuous
    h0 = h0 + rng.uniform(-0.05, 0.05, size=n)
    h0 = project_bounds_sum(h0, target)

    # ----- two gradient phases -----
    phase1_end = start + budget_s * 0.35
    best_h1, best_c1 = gradient_descent(h0, target, phase1_end, rng)

    phase2_end = start + budget_s * 0.70
    # fresh random start for phase 2
    h_rand = rng.uniform(0.0, 1.0, size=n)
    h_rand = project_bounds_sum(h_rand, target)
    best_h2, best_c2 = gradient_descent(h_rand, target, phase2_end, rng)

    if best_c2 < best_c1:
        best_h, best_c = best_h2, best_c2
    else:
        best_h, best_c = best_h1, best_c1

    # ----- short local polishing on the best candidate -----
    for _ in range(12):
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * best_h
        else:
            g = np.ones_like(best_h)
            g[: n - k] -= best_h[k:]
            g[k:] -= best_h[: n - k]
        best_h = project_bounds_sum(best_h - 0.05 * g, target)

    # ----- SLSQP refinement (if time permits) -----
    slsqp_deadline = start + budget_s * 0.95
    if time.time() < slsqp_deadline:
        global CURRENT_TARGET, CURRENT_BETA
        CURRENT_TARGET = target
        beta_seq = (20.0, 200.0, 2000.0, 10000.0)
        for beta in beta_seq:

            CURRENT_BETA = beta
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)

    # ----- final repair and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
