def binary_sa_anneal(h, rng, deadline, max_k=5):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)  # using FFT
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = cur_c * 0.02
    T_min = 1e-9
    it = 0
    while time.time() < deadline:
        # temperature decay
        T = max(T0 * (0.9995 ** it), T_min)
        # decide k
        if rng.random() < 0.2:
            k = rng.integers(2, max_k + 1)
        else:
            k = 1
        # indices
        ones_idx = np.flatnonzero(cur_h == 1.0)
        zeros_idx = np.flatnonzero(cur_h == 0.0)
        if len(ones_idx) < k or len(zeros_idx) < k:
            continue
        i_swap = rng.choice(ones_idx, k, replace=False)
        j_swap = rng.choice(zeros_idx, k, replace=False)
        # swap
        cur_h[i_swap] = 0.0
        cur_h[j_swap] = 1.0
        c_new = compute_c5(cur_h)
        delta = c_new - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i_swap] = 1.0
            cur_h[j_swap] = 0.0
        it += 1
    return best_h, best_c

    """Maximum overlap integral for a step function h (FFT version)."""
    n = h.size

    # ----- worst‑shift refinement -----
    if time.time() < deadline:
        h_ws = worst_shift_refine(best_h, rng, base_step=0.12,
                                 iters=30, deadline=deadline)
        c_ws = compute_c5(h_ws)
        if c_ws < best_c:
            best_c, best_h = c_ws, h_ws.copy()

    # ----- final SLSQP polishing -----
    for beta in (2000, 5000, 20000, 50000):
        if time.time() > deadline:
            break
        h_sl = slsqp_optimize(best_h, beta, rng, max_iter=600)
        c_sl = compute_c5(h_sl)
        if c_sl < best_c:
            best_c, best_h = c_sl, h_sl.copy()

    # ----- final safety adjustment -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(base_n)
