def binary_from_initial(arr, target, rng):
    h = (np.array(arr) >= 0.5).astype(np.int8)
    cur = h.sum()
    diff = target - cur
    N = h.size
    if diff > 0:
        # need to add diff ones: flip zeros to ones
        zero_idx = np.flatnonzero(h == 0)
        to_flip = rng.choice(zero_idx, diff, replace=False)
        h[to_flip] = 1
    elif diff < 0:
        # need to remove -diff ones: flip ones to zeros
        one_idx = np.flatnonzero(h == 1)
        to_flip = rng.choice(one_idx, -diff, replace=False)
        h[to_flip] = 0
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def binary_from_initial(arr, target, rng):
    """Convert initial_h_values (float) to binary with exact sum."""
    h = (np.array(arr) >= 0.5).astype(np.int8)
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        pick = rng.choice(zeros, diff, replace=False)
        h[pick] = 1
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        pick = rng.choice(ones, -diff, replace=False)
        h[pick] = 0
    return h


def greedy_optimize(h: np.ndarray, target: int, rng, time_limit: float):
    """Greedy improvement: swap a bad and a good index for the worst shift."""
    N = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)

    while time.time() < time_limit:
        corr = np.correlate(h, 1 - h, mode="full")
        Omax = corr.max()
        idx = corr.argmax()
        shift = idx - (N - 1)

        # find bad (1→0) and good (0→1) positions for this shift
        if shift >= 0:
            seg_len = N - shift
            h_seg = h[:seg_len]
            h_shift = h[shift:]
            bad_mask = (h_seg == 1) & (h_shift == 0)
            good_mask = (h_seg == 0) & (h_shift == 1)
            bad_idx = np.flatnonzero(bad_mask)
            good_idx = np.flatnonzero(good_mask)
        else:
            s = -shift
            seg_len = N - s
            h_seg = h[s:]
            h_shift = h[:seg_len]
            bad_mask = (h_seg == 1) & (h_shift == 0)
            good_mask = (h_seg == 0) & (h_shift == 1)
            bad_idx = np.flatnonzero(bad_mask) + s
            good_idx = np.flatnonzero(good_mask) + s

        if bad_idx.size == 0 or good_idx.size == 0:
            break  # no improving swap for this worst shift

        i = rng.choice(bad_idx)
        j = rng.choice(good_idx)
        h[i], h[j] = h[j], h[i]  # swap

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
