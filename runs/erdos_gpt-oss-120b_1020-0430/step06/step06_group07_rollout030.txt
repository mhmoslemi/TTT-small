def compute_min_cross_corr(h):
    n = h.shape[0]
    H = np.fft.rfft(h, n*2)
    conv = np.fft.irfft(H * np.conj(H), n*2)
    cross = conv[n-1:n-1+n]  # length n
    # cross[0] = sum(h^2) = sum(h) = n//2
    # ignore k=0
    min_cross = int(np.min(cross[1:]))
    return min_cross
```

def binary_hill_climb(h, rng, max_iters, temp_schedule=None):
    n = h.size
    best_h = h.copy()
    best_m = compute_min_cross_corr(best_h)
    cur_h = h.copy()
    cur_m = best_m
    for it in range(max_iters):
        # select random 1 index and 0 index
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_m = compute_min_cross_corr(cur_h)
        # accept criteria
        accept = False
        if new_m > cur_m:
            accept = True
        else:
            if temp_schedule is not None:
                T = temp_schedule(it)
                # Acceptance probability: exp((new_m - cur_m)/T)
                # Since change is integer negative, T must be >0
                if T > 0 and rng.random() < np.exp((new_m - cur_m) / T):
                    accept = True
        if accept:
            cur_m = new_m
            if new_m > best_m:
                best_m = new_m
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # possible restart after no improvement for long?
        # optional
    return best_h, best_m
```

    h[:n//2] = 1
    rng.shuffle(h)
    # compute cross via FFT
    cross = np.fft.irfft(np.fft.rfft(h, n*2) * np.fft.rfft(h, n*2).conj(), n*2)
    cross = np.rint(cross[n-1:n-1+n]).astype(np.int32)
    best_h = h.copy()
    best_m = cross[1:].min()
    cur_h = h
    cur_cross = cross
    cur_m = best_m
    for it in range(max_iters):
        # choose i (1) and j (0)
        i = rng.integers(0, n)

            i = rng.integers(0, n)

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        # worst gradient
        grad, k = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step_scale
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        # optionally decay step_scale
        if (it+1) % (max_iters//5) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

import time
import numpy as np

# ---------- utilities ----------
def project_box_sum(v, target):
    """Project onto {0≤x≤1, sum x = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h):
    """C5 for a step function on [0,2] – fast FFT version."""
    n = h.shape[0]
    H = np.fft.rfft(h, n * 2)
    G = np.fft.rfft(1.0 - h, n * 2)
    conv = np.fft.irfft(H * G, n * 2)
    return float(conv[n - 1:n - 1 + n].max() * (2.0 / n))


def worst_gradient(h):
    """Sub‑gradient of the worst‑shift overlap for continuous h."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1:]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


# ---------- binary‑search specific ----------
def apply_swap(cross, h, i, j):
    """Update aperiodic auto‑correlation after swapping a 1 at i with a 0 at j."""
    n = h.shape[0]  # h[i]==1, h[j]==0
    if i < n - 1:
        limit = n - 1 - i
        cross[1 : limit + 1] -= h[i + 1 : i + limit + 1]
    if i > 0:
        limit = i
        cross[1 : limit + 1] -= h[i - 1 : i - limit - 1 : -1]
    if j < n - 1:
        limit = n - 1 - j
        cross[1 : limit + 1] += h[j + 1 : j + limit + 1]
    if j > 0:
        limit = j
        cross[1 : limit + 1] += h[j - 1 : j - limit - 1 : -1]


def initial_legendre_h(n, rng):
    """Balanced binary start based on a Legendre‑type sequence (p=509)."""
    p = 509
    seq = np.zeros(p, dtype=np.int8)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            seq[i] = 1
    h = np.zeros(n, dtype=np.int8)
    h[:p] = seq
    needed = n // 2 - int(h.sum())
    zero_idx = np.flatnonzero(h == 0)
    rng.shuffle(zero_idx)
    h[zero_idx[:needed]] = 1
    rng.shuffle(h)
    return h


def binary_local_search_full(n, rng, time_budget, init_h=None):
    """Binary hill‑climb + simulated annealing to maximise min aperiodic autocorrelation."""
    start = time.time()
    if init_h is not None:
        cur_h = init_h.copy()
    else:
        cur_h = np.zeros(n, dtype=np.int8)
        cur_h[: n // 2] = 1
        rng.shuffle(cur_h)

    H = np.fft.rfft(cur_h, n * 2)
    conv = np.fft.irfft(H * np.conj(H), n * 2)
    cur_cross = np.rint(conv[n - 1 : n - 1 + n]).astype(np.int32)
    cur_m = cur_cross[1:].min()
    best_h = cur_h.copy()
    best_m = cur_m
    last_improve = 0
    it = 0
    T0, T_end = 5.0, 0.001

    while time.time() - start < time_budget:
        it += 1
        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        while cur_h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while cur_h[j] == 1:
            j = rng.integers(0, n)

        # perform swap
        apply_swap(cur_cross, cur_h, i, j)
        cur_h[i], cur_h[j] = 0, 1
        new_m = cur_cross[1:].min()

        # acceptance test
        accept = False
        if new_m > cur_m:
            accept = True
        else:
            frac = (time.time() - start) / time_budget
            T = T0 * (1.0 - frac) + T_end * frac
            if T > 0.0 and rng.random() < np.exp((new_m - cur_m) / T):
                accept = True

        if accept:
            cur_m = new_m
            if new_m > best_m:
                best_m = new_m
                best_h = cur_h.copy()
                last_improve = it
                if 1.0 - best_m / (n / 2) <= 0.3808:
                    break
        else:
            # revert the move
            cur_h[i], cur_h[j] = 1, 0
            apply_swap(cur_cross, cur_h, i, j)

        # stagnation restart
        if it - last_improve > 50000:
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[: n // 2] = 1
            rng.shuffle(cur_h)
            H = np.fft.rfft(cur_h, n * 2)
            conv = np.fft.irfft(H * np.conj(H), n * 2)
            cur_cross = np.rint(conv[n - 1 : n - 1 + n]).astype(np.int32)
            cur_m = cur_cross[1:].min()
            last_improve = it

    return best_h, best_m


# ---------- continuous refinement ----------
def subgradient_optimize(h0, target, time_budget, rng):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
        h = h_new
    return best_h, best_c


def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % (max_iters // 5) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def continuous_improve(h, rng, time_budget):
    start = time.time()
    remaining = time_budget
    if remaining > 4.0:
        sub_budget = remaining * 0.6
        h, _ = subgradient_optimize(h, h.size / 2.0, sub_budget, rng)
        remaining = time_budget - (time.time() - start)
    if remaining > 2.0:
        max_iters = int(min(5000, remaining * 300))
        h, _ = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
    return h
